- Perform some number of operations (possibly zero) so that your left hand (if H_i is L) or your right hand (if H_i is R) is holding part T_i.
  Here, you may move the other hand not specified by H_i.

Under the settings and constraints of this problem, it can be proved that any instructions are achievable.
Find the minimum total number of operations required to follow all the instructions.

import sys
def solve():
    input=sys.stdin.readline
    N,Q=map(int,input().split())
    INF=10**18
    dp=[INF]*N
    dp[1]=0
    prev_type='L'
    prev_pos=0
    for _ in range(Q):
        H,T=input().split()
        t=int(T)-1
        # build pref/suff from dp
        pref=[INF]*(N+1)
        best=INF
        for d in range(1,N):
            v=dp[d]-d
            if v<best: best=v
            pref[d]=best
        pref[N]=best
        suff=[INF]*(N+1)
        best=INF
        for d in range(N-1,0,-1):
            v=dp[d]+d
            if v<best: best=v
            suff[d]=best
        # suff[N]=INF, suff[0]? not needed
        def dt(c):
            if c <= 0:
                return suff[1] - c
            elif c >= N:
                return c + pref[N]
            else:
                left = c + pref[c]
                right = suff[c] - c
                return left if left < right else right
        ndp=[INF]*N
        if H == 'L':
            if prev_type == 'L':
                A=(t - prev_pos) % N
                # L->L
                for e in range(1,N):
                    c1 = A + e
                    v1 = A + dt(c1)
                    c2 = e - (N - A)
                    v2 = (N - A) + dt(c2)
                    ndp[e] = v1 if v1 < v2 else v2
            else: # prev_type R -> L
                A=(t - prev_pos) % N
                B = (-A) % N
                c0=B-N; c1=B; c2=B+N
                b0=dt(c0); b1=dt(c1); b2=dt(c2)
                for e in range(1,N):
                    v = b0 + abs(e-c0)
                    v2 = b1 + abs(e-c1)
                    if v2 < v: v = v2
                    v3 = b2 + abs(e-c2)
                    if v3 < v: v = v3
                    ndp[e]=v
        else: # H == 'R'
            if prev_type == 'L': # L->R
                A=(t - prev_pos) % N
                c0=A-N; c1=A; c2=A+N
                b0=dt(c0); b1=dt(c1); b2=dt(c2)
                for e in range(1,N):
                    v = b0 + abs(e-c0)
                    v2 = b1 + abs(e-c1)
                    if v2 < v: v = v2
                    v3 = b2 + abs(e-c2)
                    if v3 < v: v = v3
                    ndp[e]=v
            else: # prev_type R -> R
                A=(t - prev_pos) % N
                for e in range(1,N):
                    c1 = e - A
                    v1 = A + dt(c1)
                    c2 = e + N - A
                    v2 = (N - A) + dt(c2)
                    ndp[e] = v1 if v1 < v2 else v2

        prev_pos=t
        prev_type=H

import sys

def solve():
    input = sys.stdin.buffer.readline
    N, Q = map(int, input().split())
    INF = 10**18

    prev_pos = 0

        H, T = input().split()
        t = int(T) - 1

        best = INF

            v = dp[d] - d
            if v < best:
                best = v
            pref[d] = best
        prefN = best

        suff = [INF] * (N + 1)
        best = INF
        for d in range(N - 1, 0, -1):
            v = dp[d] + d
            if v < best:
                best = v
            suff[d] = best
        suff1 = suff[1]

        ndp = [INF] * N

        if H == b'L':
            if prev_type == b'L':
                A = (t - prev_pos) % N
                NmA = N - A

                for e in range(1, N):
                    c = A + e
                    if c >= N:
                        v1 = c + prefN
                    else:
                        left = c + pref[c]
                        right = suff[c] - c
                        v1 = left if left < right else right
                    v1 += A

                    c = e - NmA
                    if c <= 0:
                        v2 = suff1 - c
                    else:
                        left = c + pref[c]
                        right = suff[c] - c
                        v2 = left if left < right else right
                    v2 += NmA

                    ndp[e] = v1 if v1 < v2 else v2

            else:
                A = (t - prev_pos) % N
                B = (-A) % N

                b0 = suff1 - (B - N)
                if B == 0:

                b2 = (B + N) + prefN

                    v2 = b1 + abs(e - B)
                    if v2 < v:

                    v3 = b2 + ((B + N) - e)
                    if v3 < v:
                        v = v3
                    ndp[e] = v

        else:
            if prev_type == b'L':
                A = (t - prev_pos) % N

                b0 = suff1 - (A - N)
                if A == 0:
                    b1 = suff1
                else:
                    left = A + pref[A]
                    right = suff[A] - A
                    b1 = left if left < right else right
                b2 = (A + N) + prefN
