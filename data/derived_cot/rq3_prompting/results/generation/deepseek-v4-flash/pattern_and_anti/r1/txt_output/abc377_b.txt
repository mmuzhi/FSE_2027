```python
import sys

def main():
    S = [sys.stdin.readline().strip() for _ in range(8)]

    row_has_piece = [False] * 8
    col_has_piece = [False] * 8

    for i in range(8):
        for j in range(8):
            if S[i][j] == '#':
                row_has_piece[i] = True
                col_has_piece[j] = True

    ans = 0
    for i in range(8):
        if row_has_piece[i]:
            continue
        for j in range(8):
            if not col_has_piece[j]:
                ans += 1

    print(ans)

if __name__ == "__main__":
    main()
```
