import sys

def solve():
    data = list(map(int, sys.stdin.read().split()))
    if not data:
        return

    N = data[0]
    A = data[1:1 + N]

    M = 1 << 100
    limit = M - A[-1] - 1

    pow2 = [1 << i for i in range(100)]
    masks = [(1 << L) - 1 for L in range(101)]

    ch0 = [-1]
    ch1 = [-1]
    wt = [0]

    def new_node():
        ch0.append(-1)
        ch1.append(-1)
        wt.append(0)
        return len(ch0) - 1

    def add_prefix(r, L):
        node = 0
        for _ in range(L):
            if r & 1:
                nxt = ch1[node]
                if nxt == -1:
                    nxt = new_node()
                    ch1[node] = nxt
                node = nxt
            else:
                nxt = ch0[node]
                if nxt == -1:
                    nxt = new_node()
                    ch0[node] = nxt
                node = nxt
            r >>= 1
        wt[node] += 1

    # For each offset A, B = -A mod 2^100.
    # P(i + A) = 1 iff i == B + 2^v (mod 2^(v+2)) for v = v2(i + A).
    for a in A:
        B = (-a) % M
        for v in range(100):
            L = v + 2 if v < 99 else 100
            r = (B + pow2[v]) & masks[L]
            add_prefix(r, L)

    def has_valid(d, r):
        # Does the residue class r mod 2^d contain a valid i?
        if r == 0:
            return (1 << d) <= limit
        return r <= limit

    sys.setrecursionlimit(1000000)
    ans = 0

    def dfs(node, depth, r, acc):
        nonlocal ans

        if not has_valid(depth, r):
            return

        acc += wt[node]

        if depth == 100:
            if acc > ans:
                ans = acc
            return

        # Child bit = 0
        if ch0[node] == -1:
            if has_valid(depth + 1, r):
                if acc > ans:
                    ans = acc
        else:
            dfs(ch0[node], depth + 1, r, acc)

        # Child bit = 1
        if ch1[node] == -1:
            r1 = r | pow2[depth]
            if has_valid(depth + 1, r1):
                if acc > ans:
                    ans = acc
        else:
            dfs(ch1[node], depth + 1, r | pow2[depth], acc)

    dfs(0, 0, 0, 0)
    print(ans)

if __name__ == "__main__":
    solve()
