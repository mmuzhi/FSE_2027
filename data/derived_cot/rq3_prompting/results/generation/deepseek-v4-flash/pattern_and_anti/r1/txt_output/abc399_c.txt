import sys

class DSU:
    def __init__(self, n):
        self.parent = list(range(n))
        self.size = [1] * n

    def find(self, x):
        while self.parent[x] != x:
            self.parent[x] = self.parent[self.parent[x]]
            x = self.parent[x]
        return x

    def union(self, a, b):
        ra = self.find(a)
        rb = self.find(b)
        if ra == rb:
            return False
        if self.size[ra] < self.size[rb]:
            ra, rb = rb, ra
        self.parent[rb] = ra
        self.size[ra] += self.size[rb]
        return True


def main():
    input = sys.stdin.buffer.readline

    N, M = map(int, input().split())
    dsu = DSU(N)

    deleted = 0
    for _ in range(M):
        u, v = map(int, input().split())
        u -= 1
        v -= 1

        if not dsu.union(u, v):
            deleted += 1

    print(deleted)


if __name__ == "__main__":
    main()
