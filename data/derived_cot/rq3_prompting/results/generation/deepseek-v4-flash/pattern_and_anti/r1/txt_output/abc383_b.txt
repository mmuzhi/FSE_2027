The AtCoder company office can be represented as a grid of H rows and W columns. Let (i, j) denote the cell at the i-th row from the top and j-th column from the left.

```python
import sys

def main():
    data = sys.stdin.read().split()
    if not data:
        return

    H = int(data[0])
    W = int(data[1])
    D = int(data[2])
    grid = data[3:3 + H]

    floors = []
    for i in range(H):
        for j, ch in enumerate(grid[i]):
            if ch == '.':
                floors.append((i, j))

    F = len(floors)

    masks = []
    for i, (r1, c1) in enumerate(floors):
        mask = 0
        for j, (r2, c2) in enumerate(floors):
            if abs(r1 - r2) + abs(c1 - c2) <= D:
                mask |= 1 << j
        masks.append(mask)

    ans = 0
    for i in range(F):
        for j in range(i + 1, F):
            ans = max(ans, (masks[i] | masks[j]).bit_count())

    print(ans)

if __name__ == "__main__":
    main()
```
