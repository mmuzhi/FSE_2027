```python
import sys

S = sys.stdin.readline().strip()
pos = {ch: i for i, ch in enumerate(S, 1)}

ans = 0
for a, b in zip("ABCDEFGHIJKLMNOPQRSTUVWXYZ", "BCDEFGHIJKLMNOPQRSTUVWXYZ"):
    ans += abs(pos[a] - pos[b])

print(ans)
```
