import sys

def main():
    S = sys.stdin.readline().strip()
    n = len(S)
    ans = n

    i = 0
    while i < n:
        if S[i] == '0':
            j = i
            while j < n and S[j] == '0':
                j += 1
            ans -= (j - i) // 2
            i = j
        else:
            i += 1

    print(ans)

if __name__ == "__main__":
    main()
