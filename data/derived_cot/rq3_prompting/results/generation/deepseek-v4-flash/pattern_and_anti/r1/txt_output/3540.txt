```python
class Solution:
    def stringHash(self, s: str, k: int) -> str:
        result = []
        
        for i in range(0, len(s), k):
            substring = s[i:i + k]
            total = sum(ord(c) - ord('a') for c in substring)
            hashed_char = total % 26
            result.append(chr(ord('a') + hashed_char))
        
        return ''.join(result)
```
