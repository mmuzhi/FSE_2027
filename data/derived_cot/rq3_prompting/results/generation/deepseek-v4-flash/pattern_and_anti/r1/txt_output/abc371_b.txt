import sys

def main():
    data = sys.stdin.read().split()
    if not data:
        return

    N = int(data[0])
    M = int(data[1])

    has_eldest_son = [False] * (N + 1)
    ans = []

    idx = 2
    for _ in range(M):
        a = int(data[idx])
        b = data[idx + 1]
        idx += 2

        if b == 'M' and not has_eldest_son[a]:
            has_eldest_son[a] = True
            ans.append("Yes")
        else:
            ans.append("No")

    print("\n".join(ans))

if __name__ == "__main__":
    main()
