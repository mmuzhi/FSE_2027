```python
from typing import List

class Solution:
    def minBitwiseArray(self, nums: List[int]) -> List[int]:
        ans = []
        for num in nums:
            # Count consecutive 1 bits from the least significant side.
            trailing_ones = 0
            temp = num
            while temp & 1:
                trailing_ones += 1
                temp >>= 1

            if trailing_ones == 0:
                ans.append(-1)
            else:
                ans.append(num - (1 << (trailing_ones - 1)))
        return ans
```

```python
from typing import List

class Solution:
    def minBitwiseArray(self, nums: List[int]) -> List[int]:
        ans = []
        
        for num in nums:
            trailing_ones = 0
            temp = num
            
            while temp & 1:
                trailing_ones += 1
                temp >>= 1
            
            if trailing_ones == 0:
                ans.append(-1)
            else:
                ans.append(num - (1 << (trailing_ones - 1)))
        
        return ans
```
