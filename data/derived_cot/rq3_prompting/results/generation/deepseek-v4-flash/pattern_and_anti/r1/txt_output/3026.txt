class Solution:
    def minimumPossibleSum(self, n: int, target: int) -> int:
        MOD = 10**9 + 7
        k = min(n, target // 2)
        ans = k * (k + 1) // 2
        rem = n - k
        if rem:
            ans += rem * (2 * target + rem - 1) // 2
        return ans % MOD

class Solution:
    def minimumPossibleSum(self, n: int, target: int) -> int:
        MOD = 10**9 + 7
        k = min(n, target // 2)
        ans = k * (k + 1) // 2
        rem = n - k
        if rem:
            ans += rem * (2 * target + rem - 1) // 2
        return ans % MOD
