You are given an integer array nums. This array contains n elements, where exactly n - 2 elements are special numbers. One of the remaining two elements is the sum of these special numbers, and the other is an outlier.

class Solution:
    def getLargestOutlier(self, nums: List[int]) -> int:
        total = sum(nums)
        cnt = Counter(nums)
        ans = -10**18

from typing import List
from collections import Counter

class Solution:
    def getLargestOutlier(self, nums: List[int]) -> int:
        total = sum(nums)
        freq = Counter(nums)
        ans = -10**18
        for x in nums:
            diff = total - x
            if diff & 1:
                continue
            s = diff // 2
            if freq[s] > (1 if s == x else 0):
                ans = max(ans, x)
        return ans

from typing import List
from collections import Counter

class Solution:
    def getLargestOutlier(self, nums: List[int]) -> int:
        total = sum(nums)
        freq = Counter(nums)
        ans = -10**18

```python
        for s, count in freq.items():
            outlier = total - 2 * s
            if outlier in freq:
                if outlier != s or count >= 2:
                    ans = max(ans, outlier)

        return ans
```
