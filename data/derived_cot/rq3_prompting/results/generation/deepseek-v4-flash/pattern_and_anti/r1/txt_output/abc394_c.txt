import sys, heapq
def main():
    S = sys.stdin.readline().strip()
    n = len(S)
    chars = list(S)
    heap = []
    for i in range(n-1):
        if chars[i] == 'W' and chars[i+1] == 'A':
            heapq.heappush(heap, i)
    while heap:
        i = heapq.heappop(heap)
        if i < 0 or i+1 >= n or chars[i] != 'W' or chars[i+1] != 'A':
            continue
        chars[i] = 'A'
        chars[i+1] = 'C'
        j = i - 1
        if j >= 0 and chars[j] == 'W' and chars[j+1] == 'A':
            heapq.heappush(heap, j)
    print(''.join(chars))
if __name__ == "__main__": main()

import sys
import heapq

def main():

    n = len(S)
    s = list(S)

    heap = [i for i in range(n - 1) if s[i] == 'W' and s[i + 1] == 'A']

        i = heapq.heappop(heap)
        if i + 1 >= n or s[i] != 'W' or s[i + 1] != 'A':
            continue

        s[i + 1] = 'C'
        if i > 0 and s[i - 1] == 'W':
            heapq.heappush(heap, i - 1)

    sys.stdout.write(''.join(s) + '\n')

if __name__ == "__main__":
