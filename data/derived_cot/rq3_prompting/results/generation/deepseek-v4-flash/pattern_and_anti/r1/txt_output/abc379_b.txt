import sys

def main():
    data = sys.stdin.read().split()
    if not data:
        return
    N = int(data[0])
    K = int(data[1])
    S = data[2]
    ans = 0
    current = 0
    for ch in S:
        if ch == 'O':
            current += 1
        else:
            ans += current // K
            current = 0
    ans += current // K
    print(ans)

if __name__ == "__main__":
    main()
