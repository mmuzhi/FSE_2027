```python
class Solution:
    def findValidPair(self, s: str) -> str:
        cnt = [0] * 10
        for ch in s:
            cnt[int(ch)] += 1

        for i in range(len(s) - 1):
            a = s[i]
            b = s[i + 1]
            if a != b and cnt[int(a)] == int(a) and cnt[int(b)] == int(b):
                return s[i:i + 2]

        return ""
```
