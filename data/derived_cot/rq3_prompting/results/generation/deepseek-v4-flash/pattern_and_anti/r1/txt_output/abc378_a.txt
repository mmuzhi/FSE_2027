=== ANSWER ===
```python
import sys
from collections import Counter

def main():
    A = list(map(int, sys.stdin.read().split()))
    cnt = Counter(A)
    ans = sum(v // 2 for v in cnt.values())
    print(ans)

if __name__ == "__main__":
    main()
```
