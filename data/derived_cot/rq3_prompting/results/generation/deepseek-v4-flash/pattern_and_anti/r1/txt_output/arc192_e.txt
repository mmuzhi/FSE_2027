import sys
from array import array

MOD = 998244353

def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    W,H,L,R,D,U = data
    max_n = W + H + 4

    fact = array('I', [1]) * (max_n + 1)
    for i in range(1, max_n + 1):
        fact[i] = (fact[i-1] * i) % MOD

    invfact = array('I', [1]) * (max_n + 1)
    invfact[max_n] = pow(fact[max_n], MOD - 2, MOD)
    for i in range(max_n, 0, -1):
        invfact[i-1] = (invfact[i] * i) % MOD

    inv = array('I', [0]) * (max_n + 1)
    for i in range(1, max_n + 1):
        inv[i] = (fact[i-1] * invfact[i]) % MOD

    def comb(n, k):
        if k < 0 or k > n:
            return 0
        return fact[n] * invfact[k] % MOD * invfact[n-k] % MOD

    s_cache = {}
    def S(I, J):
        if I < 0 or J < 0:
            return 0
        key = (I, J)
        if key in s_cache:
            return s_cache[key]
        val = (comb(I + J + 4, J + 2) - (I + J + 4)) % MOD
        s_cache[key] = val
        return val

    def pref(a, b):
        if a < 0 or b < 0:
            return 0
        q = (S(W, H) - S(W - a - 1, H) - S(W, H - b - 1) + S(W - a - 1, H - b - 1)) % MOD
        area = ((a + 1) % MOD) * ((b + 1) % MOD) % MOD
        return (q - area) % MOD

    total_full = (S(W, H) - ((W + 1) % MOD) * ((H + 1) % MOD)) % MOD

    sum_M = (pref(R, U) - pref(L - 1, U) - pref(R, D - 1) + pref(L - 1, D - 1)) % MOD

    B0 = None
    if L > 0 or D > 0:
        B0 = comb(W + H - L - D + 2, W - L + 1)

    vertical = 0
    if L > 0:
        A = comb(L + D + 1, L)
        B = B0
        for y in range(D, U + 1):
            term = ((A - 1) % MOD) * ((B - 1) % MOD) % MOD
            vertical += term
            if vertical >= MOD:
                vertical -= MOD
            if y < U:
                A = A * (L + y + 2) % MOD * inv[y + 2] % MOD
                nB = W + H - L - y + 2
                B = B * (H - y + 1) % MOD * inv[nB] % MOD

    horizontal = 0
    if D > 0:

    ans = (total_full - sum_M - vertical - horizontal) % MOD

import sys
from array import array

MOD = 998244353


def solve():
    W, H, L, R, D, U = map(int, sys.stdin.buffer.read().split())

    max_n = W + H + 4

    fact = array('I', [1]) * (max_n + 1)
    for i in range(1, max_n + 1):
        fact[i] = fact[i - 1] * i % MOD

    invfact = array('I', [1]) * (max_n + 1)
    invfact[max_n] = pow(fact[max_n], MOD - 2, MOD)
    for i in range(max_n, 0, -1):
        invfact[i - 1] = invfact[i] * i % MOD

    inv = array('I', [0]) * (max_n + 1)
    for i in range(1, max_n + 1):
        inv[i] = fact[i - 1] * invfact[i] % MOD

    def comb(n, k):
        if k < 0 or k > n:
            return 0
        return fact[n] * invfact[k] % MOD * invfact[n - k] % MOD

    s_cache = {}

    def S(I, J):
        if I < 0 or J < 0:
            return 0
        key = (I, J)
        if key in s_cache:
            return s_cache[key]
        val = (comb(I + J + 4, J + 2) - (I + J + 4)) % MOD
        s_cache[key] = val
        return val

    def pref(a, b):
        if a < 0 or b < 0:
            return 0
        q = (
            S(W, H)
            - S(W - a - 1, H)
            - S(W, H - b - 1)
            + S(W - a - 1, H - b - 1)
        ) % MOD
        area = (a + 1) % MOD * ((b + 1) % MOD) % MOD
        return (q - area) % MOD

    total_full = (S(W, H) - (W + 1) % MOD * ((H + 1) % MOD)) % MOD

    # Paths whose starting point is inside the missing rectangle.
    sum_M = (
        pref(R, U)
        - pref(L - 1, U)
        - pref(R, D - 1)
        + pref(L - 1, D - 1)
    ) % MOD

    # Common value: C(W+H-L-D+2, W-L+1)
    B0 = None
    if L > 0 or D > 0:
        B0 = comb(W + H - L - D + 2, W - L + 1)

    # Paths entering the missing rectangle from the left.
    vertical = 0
    if L > 0:
        A = comb(L + D + 1, L)          # C(L+y+1, L) at y=D
        B = B0                          # C(W+H-L-y+2, W-L+1) at y=D

        for y in range(D, U + 1):
            term = ((A - 1) % MOD) * ((B - 1) % MOD) % MOD
            vertical += term
            if vertical >= MOD:
                vertical -= MOD

            if y < U:
                A = A * (L + y + 2) % MOD * inv[y + 2] % MOD
                nB = W + H - L - y + 2
                B = B * (H - y + 1) % MOD * inv[nB] % MOD

    # Paths entering the missing rectangle from below.
    horizontal = 0
    if D > 0:
        Cval = comb(L + D + 1, L + 1)   # C(x+D+1, x+1) at x=L
        E = B0                          # C(W+H-x-D+2, W-x+1) at x=L

        for x in range(L, R + 1):
            term = ((Cval - 1) % MOD) * ((E - 1) % MOD) % MOD
            horizontal += term
            if horizontal >= MOD:
                horizontal -= MOD

            if x < R:
                Cval = Cval * (x + D + 2) % MOD * inv[x + 2] % MOD
                nE = W + H - x - D + 2
                E = E * (W - x + 1) % MOD * inv[nE] % MOD

    ans = (total_full - sum_M - vertical - horizontal) % MOD
