```python
from typing import List
from math import gcd

class Solution:
    def maxLength(self, nums: List[int]) -> int:
        n = len(nums)
        ans = 0

        for i in range(n):
            prod = 1
            cur_lcm = 1
            cur_gcd = 0

            for j in range(i, n):
                x = nums[j]
                prod *= x
                cur_gcd = gcd(cur_gcd, x)
                cur_lcm = cur_lcm * x // gcd(cur_lcm, x)

                if prod == cur_lcm * cur_gcd:
                    ans = max(ans, j - i + 1)

        return ans
```
