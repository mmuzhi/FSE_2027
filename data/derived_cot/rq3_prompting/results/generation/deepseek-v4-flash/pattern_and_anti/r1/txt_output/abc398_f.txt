import sys

def prefix_function(s):
    n = len(s)
    pi = [0] * n
    for i in range(1, n):
        j = pi[i - 1]
        while j > 0 and s[i] != s[j]:
            j = pi[j - 1]
        if s[i] == s[j]:
            j += 1
        pi[i] = j
    return pi

def main():

    k = n - q
    ans = S + S[:k][::-1]

def prefix_function(s):
    n = len(s)

```python
import sys

def prefix_function(s):
    n = len(s)
    pi = [0] * n
    for i in range(1, n):
        j = pi[i - 1]
        c = s[i]
        while j > 0 and c != s[j]:
            j = pi[j - 1]
        if c == s[j]:
            j += 1
        pi[i] = j
    return pi

def main():
    S = sys.stdin.readline().strip()
    n = len(S)

    rev = S[::-1]
    pi = prefix_function(rev)

    q = 0
    for ch in S:
        while q > 0 and ch != rev[q]:
            q = pi[q - 1]
        if ch == rev[q]:
            q += 1

    k = n - q
    ans = S + S[:k][::-1]
    print(ans)

if __name__ == "__main__":
    main()
