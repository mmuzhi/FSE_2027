from typing import List
import heapq

class Solution:
    def resultsArray(self, queries: List[List[int]], k: int) -> List[int]:
        heap = []
        ans = []
        for x, y in queries:
            d = abs(x) + abs(y)
            if len(heap) < k:
                heapq.heappush(heap, -d)
            else:
                heapq.heappushpop(heap, -d)
            if len(heap) == k:
                ans.append(-heap[0])
            else:
                ans.append(-1)
        return ans

=== ANSWER ===
```python
from typing import List
import heapq

class Solution:
    def resultsArray(self, queries: List[List[int]], k: int) -> List[int]:
        heap = []
        ans = []

        for x, y in queries:
            dist = abs(x) + abs(y)

            if len(heap) < k:
                heapq.heappush(heap, -dist)
            else:
                heapq.heappushpop(heap, -dist)

            if len(heap) == k:
                ans.append(-heap[0])
            else:
                ans.append(-1)

        return ans
```
