```
import sys
import heapq

MOD = 998244353

class Entry:
    __slots__ = ('w','s','rep','ver')
    def __init__(self, w, s, rep, ver):
        self.w = w
        self.s = s
        self.rep = rep
        self.ver = ver
    def __lt__(self, other):
        return self.w * other.s > other.w * self.s

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    it = iter(data)
    T = next(it)
    out = []
    for _ in range(T):
        N = next(it)
        p = [0]*(N+1)
        for i in range(1, N+1):
            p[i] = next(it)
        p[0] = -1
        a = [0]*(N+1)
        total_a = 0
        for i in range(1, N+1):
            val = next(it)
            a[i] = val
            total_a += val

        dsu = list(range(N+1))
        W = [0]*(N+1)
        S = [0]*(N+1)
        F = [0]*(N+1)
        ver = [0]*(N+1)
        for i in range(1, N+1):
            W[i] = a[i]
            S[i] = 1
            F[i] = a[i] % MOD

        heap = []
        for i in range(1, N+1):
            heap.append(Entry(W[i], S[i], i, ver[i]))
        heapq.heapify(heap)

        def find(x):
            while dsu[x] != x:
                dsu[x] = dsu[dsu[x]]
                x = dsu[x]
            return x

        while heap:
            e = heapq.heappop(heap)
            rep = e.rep
            if rep == 0:
                continue
            if dsu[rep] != rep:
                continue
            if W[rep] != e.w or S[rep] != e.s or ver[rep] != e.ver:
                continue
            parent = find(p[rep])
            # parent should not be rep
            if parent == rep:
                continue
            inc = S[parent] * W[rep]
            F[parent] = (F[parent] + F[rep] + inc) % MOD
            W[parent] += W[rep]
            S[parent] += S[rep]
            dsu[rep] = parent
            if parent != 0:
                ver[parent] += 1
                heapq.heappush(heap, Entry(W[parent], S[parent], parent, ver[parent]))

        ans = F[0] * pow(total_a % MOD, MOD-2, MOD) % MOD
        out.append(str(ans))
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    main()
```

```
import heapq

class Entry:
    __slots__ = ('w','s','rep','ver')
    def __init__(self, w, s, rep, ver):

        self.s = s
```
