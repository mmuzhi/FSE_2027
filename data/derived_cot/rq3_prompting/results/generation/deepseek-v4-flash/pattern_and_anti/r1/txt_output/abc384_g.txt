class BIT:
  def __init__(n):
    self.n=n
    self.c=[0]*(n+1)
    self.s=[0]*(n+1)
  def add(idx, dc, ds):
    n=self.n; c=self.c; s=self.s
    while idx<=n:
      c[idx]+=dc; s[idx]+=ds; idx += idx & -idx
  def query(idx):
    c=self.c; s=self.s; cnt=0; sm=0
    while idx>0:
      cnt += c[idx]; sm += s[idx]; idx -= idx & -idx
    return cnt, sm

def hilbert_order(x, y, pow=21, rot=0):
    if pow == 0:
        return 0
    hpow = 1 << (pow - 1)
    seg = (0 if x < hpow else 1) * 2 + (0 if y < hpow else 1)
    seg = (seg + rot) & 3
    rotate_delta = [3, 0, 0, 1]
    nx = x & (hpow - 1)
    ny = y & (hpow - 1)
    nrot = (rot + rotate_delta[seg]) & 3
    sub_square_size = 1 << (2 * pow - 2)
    ans = seg * sub_square_size
    add = hilbert_order(nx, ny, pow - 1, nrot)
    if seg == 1 or seg == 2:
        ans += add
    else:

def hilbert_order(x, y, pow=17, rot=0):
    if pow == 0:
        return 0
    hpow = 1 << (pow - 1)
    seg = (0 if x < hpow else 1) * 2 + (0 if y < hpow else 1)
    seg = (seg + rot) & 3
    rotate_delta = [3, 0, 0, 1]
    nx = x & (hpow - 1)
    ny = y & (hpow - 1)
    nrot = (rot + rotate_delta[seg]) & 3
    sub_square_size = 1 << (2 * pow - 2)
    ord = seg * sub_square_size
    add = hilbert_order(nx, ny, pow - 1, nrot)

```python
import sys
from array import array


def hilbert_order(x, y, pow, rot=0):
    if pow == 0:
        return 0
    hpow = 1 << (pow - 1)

    if x < hpow:
        seg = 0 if y < hpow else 3
    else:
        seg = 1 if y < hpow else 2

    seg = (seg + rot) & 3
    rotate_delta = (3, 0, 0, 1)

    nx = x & (hpow - 1)
    ny = y & (hpow - 1)
    nrot = (rot + rotate_delta[seg]) & 3

    sub = 1 << (2 * pow - 2)
    ans = seg * sub
    add = hilbert_order(nx, ny, pow - 1, nrot)

    if seg == 1 or seg == 2:
        ans += add
    else:
        ans += sub - add - 1
    return ans


class WaveletMatrix:
    def __init__(self, ranks, values):
        n = len(ranks)
        self.max_rank = max(ranks) if n else 0
        self.max_log = max(1, self.max_rank.bit_length())

        self.pref0 = []
        self.pref_sum0 = []
        self.mid = []

        cur_rank = ranks[:]
        cur_val = values[:]

        for bit in range(self.max_log - 1, -1, -1):
            pref0 = array('i', [0]) * (n + 1)
            pref_sum0 = array('q', [0]) * (n + 1)

            zero_rank = []
            zero_val = []
            one_rank = []
            one_val = []

            zcnt = 0
            ssum = 0

            for i in range(n):
                if (cur_rank[i] >> bit) & 1:
                    one_rank.append(cur_rank[i])
                    one_val.append(cur_val[i])
                else:
                    zcnt += 1
                    ssum += cur_val[i]
                    zero_rank.append(cur_rank[i])
                    zero_val.append(cur_val[i])

                pref0[i + 1] = zcnt
                pref_sum0[i + 1] = ssum

            self.mid.append(zcnt)
            self.pref0.append(pref0)
            self.pref_sum0.append(pref_sum0)

            cur_rank = zero_rank + one_rank
            cur_val = zero_val + one_val


def main():
    input = sys.stdin.buffer.readline

    N = int(input())
    A = list(map(int, input().split()))
    B = list(map(int, input().split()))

    K = int(input())
    queries = []
    for idx in range(K):
        x, y = map(int, input().split())
        queries.append((x, y, idx))

    all_vals = sorted(set(A + B))
    max_rank = len(all_vals) - 1

    val_to_rank = {v: i for i, v in enumerate(all_vals)}
    rankA = [val_to_rank[a] for a in A]
    rankB = [val_to_rank[b] for b in B]

    prefA = [0] * (N + 1)
    prefB = [0] * (N + 1)
    for i in range(N):
        prefA[i + 1] = prefA[i] + A[i]
        prefB[i + 1] = prefB[i] + B[i]

    wmA = WaveletMatrix(rankA, A)
    wmB = WaveletMatrix(rankB, B)

    pref0A = wmA.pref0
    pref_sum0A = wmA.pref_sum0
    midA = wmA.mid
    logA = wmA.max_log

    pref0B = wmB.pref0
    pref_sum0B = wmB.pref_sum0
    midB = wmB.mid
    logB = wmB.max_log

    def query_A_abs(value, rank, x,
                    pref0=pref0A, pref_sum0=pref_sum0A, mid=midA,
                    log=logA, max_rank=max_rank, pref=prefA):
        if x == 0:
            return 0
        if rank >= max_rank:
            cnt = x
            sm = pref[x]
        else:
            upper = rank + 1
            l = 0
            r = x
            cnt = 0
            sm = 0
            shift = log - 1

            for level in range(log):
                p0 = pref0[level]
                zl = p0[l]
                zr = p0[r]

                if (upper >> shift) & 1:
                    cnt += zr - zl
                    ps = pref_sum0[level]
                    sm += ps[r] - ps[l]
                    m = mid[level]
                    l = m + (l - zl)
                    r = m + (r - zr)
                else:
                    l = zl
                    r = zr

                shift -= 1

        return value * (2 * cnt - x) + pref[x] - 2 * sm

    def query_B_abs(value, rank, y,
                    pref0=pref0B, pref_sum0=pref_sum0B, mid=midB,
                    log=logB, max_rank=max_rank, pref=prefB):
        if y == 0:
            return 0
        if rank >= max_rank:
            cnt = y
            sm = pref[y]
        else:
            upper = rank + 1
            l = 0
            r = y
            cnt = 0
            sm = 0
            shift = log - 1

            for level in range(log):
                p0 = pref0[level]
                zl = p0[l]
                zr = p0[r]

                if (upper >> shift) & 1:
                    cnt += zr - zl
                    ps = pref_sum0[level]
                    sm += ps[r] - ps[l]
                    m = mid[level]
                    l = m + (l - zl)
                    r = m + (r - zr)
                else:
                    l = zl
                    r = zr

                shift -= 1

        return value * (2 * cnt - y) + pref[y] - 2 * sm

    pow_bits = max(1, N.bit_length())
    queries = [(x, y, idx, hilbert_order(x, y, pow_bits)) for x, y, idx in queries]
    queries.sort(key=lambda q: q[3])

    ans = [0] * K

    cur_x = 0
    cur_y = 0
    cur_ans = 0

    for x, y, idx, _ in queries:
        while cur_x < x:
            a = A[cur_x]
            r = rankA[cur_x]
            cur_ans += query_B_abs(a, r, cur_y)
            cur_x += 1

        while cur_x > x:
            cur_x -= 1
            a = A[cur_x]
            r = rankA[cur_x]
            cur_ans -= query_B_abs(a, r, cur_y)

        while cur_y < y:
            b = B[cur_y]
            r = rankB[cur_y]
            cur_ans += query_A_abs(b, r, cur_x)
            cur_y += 1

        while cur_y > y:
            cur_y -= 1
            b = B[cur_y]
            r = rankB[cur_y]
            cur_ans -= query_A_abs(b, r, cur_x)

        ans[idx] = cur_ans

    print("\n".join(map(str, ans)))


if __name__ == "__main__":
    main()
```
