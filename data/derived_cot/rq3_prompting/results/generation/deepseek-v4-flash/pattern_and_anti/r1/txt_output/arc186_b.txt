def solve():
    import sys
    MOD=998244353
    data=list(map(int, sys.stdin.buffer.read().split()))
    if not data: return
    N=data[0]
    A=[0]+data[1:1+N]  # 1-indexed
    left=[0]*(N+1)
    right=[0]*(N+1)
    st=[]
    root=0
    for i in range(1,N+1):
        a=A[i]
        last=0
        if a==0:
            while st:
                last=st.pop()
            root=i
            if last:
                left[i]=last
        else:
            while st and st[-1]!=a:
                last=st.pop()
            # st[-1] == a
            if last:
                left[i]=last
            right[a]=i
        st.append(i)
    # compute sizes
    size=[0]*(N+1)
    order=[]
    stack=[root]
    while stack:
        v=stack.pop()
        order.append(v)
        l=left[v]
        if l:
            stack.append(l)
        r=right[v]
        if r:
            stack.append(r)
    for v in reversed(order):
        s=1
        l=left[v]
        if l:
            s += size[l]
        r=right[v]
        if r:
            s += size[r]
        size[v]=s
    fact=1
    for i in range(2,N+1):
        fact = fact * i % MOD
    denom=1
    for v in range(1,N+1):
        denom = denom * size[v] % MOD
    ans = fact * pow(denom, MOD-2, MOD) % MOD
    print(ans)

import sys

MOD = 998244353

def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    N = data[0]
    left = [0]*(N+1)
    right = [0]*(N+1)
    st = []
    root = 0
    for i in range(1, N+1):
        a = data[i]
        last = 0
        if a == 0:
            while st:
                last = st.pop()
            root = i
            if last:
                left[i] = last
        else:
            while st and st[-1] != a:
                last = st.pop()
            if last:
                left[i] = last
            right[a] = i
        st.append(i)
    del data
    del st

    size = [0]*(N+1)
    order = []
    stack = [root]
    while stack:
        v = stack.pop()
        order.append(v)
        l = left[v]
        if l:
            stack.append(l)
        r = right[v]
        if r:
            stack.append(r)

    fact = 1
    for x in range(2, N+1):
        fact = fact * x % MOD

    denom = 1
    for v in reversed(order):
        s = 1
        l = left[v]
        if l:
            s += size[l]
        r = right[v]
        if r:
            s += size[r]
        size[v] = s
        denom = denom * s % MOD

    ans = fact * pow(denom, MOD-2, MOD) % MOD
    print(ans)

if __name__ == "__main__":
    solve()
