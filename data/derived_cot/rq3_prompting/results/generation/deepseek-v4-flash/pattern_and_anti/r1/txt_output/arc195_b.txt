def main():

    N = data[0]

    cntA = {}

        if x == -1:

        else:
            cntA[x] = cntA.get(x, 0) + 1

    q = 0
    for x in B:
        if x == -1:
            q += 1
        else:
            cntB[x] = cntB.get(x, 0) + 1

    P = p + q
    if P >= N:
        print("Yes")
        return

    T = N - P
    if T <= 1:
        print("Yes")
        return

    if len(cntA) > len(cntB):

def main():

    cntA = {}

        else:
            cntA[x] = cntA.get(x, 0) + 1

    cntB = {}
    q = 0
    for x in B:
        if x == -1:
            q += 1
        else:
            cntB[x] = cntB.get(x, 0) + 1

    P = p + q
    T = N - P
    if T <= 1:
        print("Yes")
        return

    L = max(max(cntA), max(cntB))

    if len(cntA) > len(cntB):
        cntA, cntB = cntB, cntA

    itemsA = sorted(cntA.items(), key=lambda x: x[1], reverse=True)

            b = keysB[j]

import sys
from bisect import bisect_left

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    N = data[0]
    A = data[1:1+N]
    B = data[1+N:1+2*N]

    cntA = {}
    p = 0
    for x in A:
        if x == -1:
            p += 1
        else:
            cntA[x] = cntA.get(x, 0) + 1

    cntB = {}
    q = 0
    for x in B:
        if x == -1:
            q += 1
        else:
            cntB[x] = cntB.get(x, 0) + 1

    P = p + q
    T = N - P
    if T <= 1:
        print("Yes")
        return

    L = max(max(cntA), max(cntB))

    if len(cntA) > len(cntB):
        cntA, cntB = cntB, cntA

    itemsA = sorted(cntA.items(), key=lambda x: x[1], reverse=True)
    keysB = sorted(cntB.keys())
    countsB = [cntB[k] for k in keysB]
    lenB = len(keysB)

    SHIFT = 12
    MASK = (1 << SHIFT) - 1
    arr = []
    append = arr.append

    for a, ca in itemsA:
        idx = bisect_left(keysB, L - a)
        for j in range(idx, lenB):
            b = keysB[j]
            cb = countsB[j]
            val = ca if ca < cb else cb
            append(((a + b) << SHIFT) | val)

    arr.sort()

    best = 0
    i = 0
    n = len(arr)
    while i < n:
        s = arr[i] >> SHIFT
        total = 0
        while i < n and (arr[i] >> SHIFT) == s:
            total += arr[i] & MASK
            i += 1
        if total > best:
            best = total
            if best >= T:
                print("Yes")
                return

    print("No")

if __name__ == "__main__":
    main()
