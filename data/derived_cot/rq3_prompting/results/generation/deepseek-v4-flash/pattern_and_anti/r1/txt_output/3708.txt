from typing import List

class Solution:
    def zigzagTraversal(self, grid: List[List[int]]) -> List[int]:
        rows = len(grid)
        cols = len(grid[0])
        res = []
        take = True
        for r in range(rows):
            if r % 2 == 0:
                c_iter = range(cols)
            else:
                c_iter = range(cols - 1, -1, -1)
            for c in c_iter:
                if take:
                    res.append(grid[r][c])
                take = not take
        return res

from typing import List

class Solution:
    def zigzagTraversal(self, grid: List[List[int]]) -> List[int]:
        rows = len(grid)
        cols = len(grid[0])
        ans = []
        take = True

        for r in range(rows):
            if r % 2 == 0:
                indices = range(cols)
            else:
                indices = range(cols - 1, -1, -1)

            for c in indices:
                if take:
                    ans.append(grid[r][c])
                take = not take

        return ans
