=== ANSWER ===
```python
import sys
import math

def main():
    input = sys.stdin.buffer.readline
    N = int(input())

    px, py = 0, 0
    ans = 0.0

    for _ in range(N):
        x, y = map(int, input().split())
        ans += math.hypot(x - px, y - py)
        px, py = x, y

    ans += math.hypot(px, py)

    print(f"{ans:.15f}")

if __name__ == "__main__":
    main()
```
