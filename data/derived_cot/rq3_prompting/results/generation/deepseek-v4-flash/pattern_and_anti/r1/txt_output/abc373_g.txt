- 1 \leq N \leq 300
- 0 \leq A_i, B_i, C_i, D_i \leq 5000 (1 \leq i \leq N)

- (A_i, B_i) \neq (C_j, D_j) (1 \leq i, j \leq N)

def hungarian(a): # a is n x n matrix
    n = len(a)
    u = [0.0]*(n+1)
    v = [0.0]*(n+1)
    p = [0]*(n+1) # p[j] = row assigned to column j
    way = [0]*(n+1)
    for i in range(1,n+1):
        p[0] = i
        j0 = 0
        minv = [INF]*(n+1)
        used = [False]*(n+1)
        while True:
            used[j0] = True
            i0 = p[j0]
            delta = INF
            j1 = 0
            for j in range(1,n+1):
                if not used[j]:
                    cur = a[i0-1][j-1] - u[i0] - v[j]
                    if cur < minv[j]:
                        minv[j] = cur
                        way[j] = j0
                    if minv[j] < delta:
                        delta = minv[j]
                        j1 = j
            for j in range(0,n+1):
                if used[j]:
                    u[p[j]] += delta
                    v[j] -= delta
                else:
                    minv[j] -= delta
            j0 = j1
            if p[j0] == 0:
                break
        while True:
            j1 = way[j0]
            p[j0] = p[j1]
            j0 = j1
            if j0 == 0:
                break
    assignment = [0]*(n+1)
    for j in range(1,n+1):
        assignment[p[j]] = j
    return assignment[1:] # row -> col

```python
import sys
import math

def hungarian(a):
    n = len(a)
    u = [0.0] * (n + 1)
    v = [0.0] * (n + 1)
    p = [0] * (n + 1)
    way = [0] * (n + 1)
    INF = 1e100
    for i in range(1, n + 1):
        p[0] = i
        j0 = 0
        minv = [INF] * (n + 1)
        used = [False] * (n + 1)
        while True:
            used[j0] = True
            i0 = p[j0]
            row = a[i0 - 1]
            ui0 = u[i0]
            delta = INF
            j1 = 0
            for j in range(1, n + 1):
                if not used[j]:
                    cur = row[j - 1] - ui0 - v[j]
                    if cur < minv[j]:
                        minv[j] = cur
                        way[j] = j0
                    if minv[j] < delta:
                        delta = minv[j]
                        j1 = j
            for j in range(1, n + 1):
                if used[j]:
                    u[p[j]] += delta
                    v[j] -= delta
                else:
                    minv[j] -= delta
            u[p[0]] += delta
            v[0] -= delta
            j0 = j1
            if p[j0] == 0:
                break
        while True:
            j1 = way[j0]
            p[j0] = p[j1]
            j0 = j1
            if j0 == 0:
                break
    ans = [0] * n
    for j in range(1, n + 1):
        ans[p[j] - 1] = j
    return ans

def main():
    input = sys.stdin.readline
    N = int(input())
    P = [tuple(map(int, input().split())) for _ in range(N)]
    Q = [tuple(map(int, input().split())) for _ in range(N)]
    cost = [[math.hypot(px - qx, py - qy) for qx, qy in Q] for px, py in P]
    ans = hungarian(cost)
    print(*ans)

if __name__ == "__main__":
    main()

    v = [0.0]*(n+1)

        p[0] = i
        j0 = 0
        minv = [INF]*(n+1)
        used = [False]*(n+1)
        while True:
            used[j0] = True
            i0 = p[j0]
            row = a[i0-1]

                    if cur < minv[j]:

                        way[j] = j0
                    if minv[j] < delta:

import sys
import math


def orient(a, b, c):
    return (b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0])


def hungarian(a):
    n = len(a)
    n1 = n + 1
    u = [0.0] * n1
    v = [0.0] * n1
    p = [0] * n1
    way = [0] * n1
    INF = 1e100
    rng = range(1, n1)

    for i in rng:
        p[0] = i
        j0 = 0
        minv = [INF] * n1
        used = [False] * n1

        while True:
            used[j0] = True
            i0 = p[j0]
            row = a[i0 - 1]
            ui0 = u[i0]
            delta = INF

            for j in rng:
                if not used[j]:
                    cur = row[j - 1] - ui0 - v[j]
                    if cur < minv[j]:
                        minv[j] = cur
                        way[j] = j0
                    if minv[j] < delta:
                        delta = minv[j]
                        j1 = j

            for j in rng:
                if used[j]:
                    u[p[j]] += delta
                    v[j] -= delta
                else:
                    minv[j] -= delta

            u[i] += delta
            j0 = j1
            if p[j0] == 0:
                break

        while True:
            j1 = way[j0]
            p[j0] = p[j1]
            j0 = j1
            if j0 == 0:
                break

    ans = [0] * n
    for j in rng:
        ans[p[j] - 1] = j
    return ans


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    idx = 0
    N = data[idx]
    idx += 1

    P = []
    for _ in range(N):
        P.append((data[idx], data[idx + 1]))
        idx += 2

    Q = []
    for _ in range(N):
        Q.append((data[idx], data[idx + 1]))
        idx += 2

    sqrt = math.sqrt
    cost = []
    for px, py in P:
        row = []
        for qx, qy in Q:
            dx = px - qx
            dy = py - qy
            row.append(sqrt(dx * dx + dy * dy))
        cost.append(row)

    ans = hungarian(cost)

    # Safety uncrossing. Normally the minimum Euclidean matching is already non-crossing.
    while True:
        swapped = False
        for i in range(N):
            for j in range(i + 1, N):
                a = P[i]
                b = Q[ans[i] - 1]
                c = P[j]
                d = Q[ans[j] - 1]

                if (orient(a, b, c) * orient(a, b, d) < 0 and
                        orient(c, d, a) * orient(c, d, b) < 0):
                    ans[i], ans[j] = ans[j], ans[i]
                    swapped = True
                    break
            if swapped:
                break
        if not swapped:
            break

    print(*ans)


if __name__ == "__main__":
    main()
