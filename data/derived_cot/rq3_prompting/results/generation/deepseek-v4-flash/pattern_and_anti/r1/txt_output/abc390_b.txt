```python
import sys

def main():
    data = list(map(int, sys.stdin.read().split()))
    N = data[0]
    A = data[1:]

    for i in range(1, N - 1):
        if A[i - 1] * A[i + 1] != A[i] * A[i]:
            print("No")
            return

    print("Yes")

if __name__ == "__main__":
    main()
```
