```python
import sys

def main():
    S = sys.stdin.readline().strip()
    N = len(S)
    ans = 0

    for j in range(N):
        if S[j] != 'B':
            continue

        max_d = min(j, N - 1 - j)
        for d in range(1, max_d + 1):
            if S[j - d] == 'A' and S[j + d] == 'C':
                ans += 1

    print(ans)

if __name__ == "__main__":
    main()
```
