=== ANSWER ===
```python
import sys
from collections import Counter

def main():
    S = sys.stdin.readline().strip()

    if len(S) % 2 != 0:
        print("No")
        return

    for i in range(0, len(S), 2):
        if S[i] != S[i + 1]:
            print("No")
            return

    cnt = Counter(S)
    if all(v == 2 for v in cnt.values()):
        print("Yes")
    else:
        print("No")

if __name__ == "__main__":
    main()
```
