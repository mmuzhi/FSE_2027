from typing import List
from bisect import bisect_left

class Solution:
    def gcdValues(self, nums: List[int], queries: List[int]) -> List[int]:
        M = max(nums)
        freq = [0] * (M + 1)
        for x in nums:
            freq[x] += 1

        pair_count = [0] * (M + 1)
        for d in range(1, M + 1):
            c = 0
            for m in range(d, M + 1, d):
                c += freq[m]
            pair_count[d] = c * (c - 1) // 2

        exact = [0] * (M + 1)
        for d in range(M, 0, -1):
            total = pair_count[d]
            for m in range(d + d, M + 1, d):
                total -= exact[m]
            exact[d] = total

        pref = [0] * (M + 1)
        running = 0
        for g in range(1, M + 1):
            running += exact[g]
            pref[g] = running

        bl = bisect_left
        return [bl(pref, q + 1, 1, M + 1) for q in queries]

from typing import List
from bisect import bisect_left

class Solution:
    def gcdValues(self, nums: List[int], queries: List[int]) -> List[int]:
        M = max(nums)

        freq = [0] * (M + 1)
        for x in nums:
            freq[x] += 1

        # pair_count[d] = number of pairs whose gcd is divisible by d
        pair_count = [0] * (M + 1)
        for d in range(1, M + 1):
            cnt = 0
            for m in range(d, M + 1, d):
                cnt += freq[m]
            pair_count[d] = cnt * (cnt - 1) // 2

        # exact[d] = number of pairs whose gcd is exactly d
        exact = [0] * (M + 1)
        for d in range(M, 0, -1):
            total = pair_count[d]
            for m in range(d + d, M + 1, d):
                total -= exact[m]
            exact[d] = total

        # prefix counts over gcd values
        pref = [0] * (M + 1)
        running = 0
        for g in range(1, M + 1):
            running += exact[g]
            pref[g] = running

        bl = bisect_left
        return [bl(pref, q + 1, 1, M + 1) for q in queries]
