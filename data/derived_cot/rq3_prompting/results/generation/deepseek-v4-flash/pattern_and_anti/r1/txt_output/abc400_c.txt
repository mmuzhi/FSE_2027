import sys
from math import isqrt

def main():
    N = int(sys.stdin.readline())

    ans = 0
    p = 2
    while p <= N:
        limit = isqrt(N // p)
        ans += (limit + 1) // 2
        p <<= 1

    print(ans)

    main()
