=== ANSWER ===
```python
import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    N = data[0]
    P = data[1:1 + N]
    Q = data[1 + N:1 + 2 * N]

    person_with_bib = [0] * N
    for person, bib in enumerate(Q):
        person_with_bib[bib - 1] = person

    ans = [0] * N
    for bib in range(1, N + 1):
        person = person_with_bib[bib - 1]
        target_person = P[person] - 1
        ans[bib - 1] = Q[target_person]

    print(*ans)

if __name__ == "__main__":
    main()
```
