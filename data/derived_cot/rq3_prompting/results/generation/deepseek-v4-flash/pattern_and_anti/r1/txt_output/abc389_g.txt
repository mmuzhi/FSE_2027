def solve():

    k = N//2
    MAX_EDGES = N*(N-1)//2

    SHIFT = 90
    MASK = (1 << SHIFT) - 1

    # comb up to MAX_DEG

    for i in range(MAX_DEG+1):

        comb[i][i] = 1
        for j in range(1,i):

    for i in range(N,0,-1):
        invfact[i-1] = invfact[i]*i % P

    def pack(coeffs):
        res=0
        for i,val in enumerate(coeffs):
            if val:
                res |= val << (SHIFT*i)
        return res

    def reduce_packed(packed):
        res=0
        for i in range(MAX_DEG+1):
            val = (packed >> (SHIFT*i)) & MASK
            if val:
                val %= P
                if val:
                    res |= val << (SHIFT*i)
        return res

    trans = [[0]*(N+1) for _ in range(N+1)]
    for s in range(1, N+1):
        for n in range(1, N - s + 1):
            Cint = n*(n-1)//2
            coeffs = [0]*(MAX_DEG+1)
            invf = invfact[n]

```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return

    N = int(data[0])
    P = int(data[1])

    k = N // 2
    MAX_EDGES = N * (N - 1) // 2
    MAX_DEG = MAX_EDGES

    # Pack polynomial coefficients in base 2^SHIFT.
    # 2^90 is safely larger than any accumulated coefficient before reduction.
    SHIFT = 90
    MASK = (1 << SHIFT) - 1

    # Binomial coefficients modulo P up to MAX_DEG
    comb = [[0] * (MAX_DEG + 1) for _ in range(MAX_DEG + 1)]
    for i in range(MAX_DEG + 1):
        comb[i][0] = 1
        comb[i][i] = 1
        for j in range(1, i):
            comb[i][j] = (comb[i - 1][j - 1] + comb[i - 1][j]) % P

    # Factorials and inverse factorials
    fact = [1] * (N + 1)
    for i in range(1, N + 1):
        fact[i] = fact[i - 1] * i % P

    invfact = [1] * (N + 1)
    invfact[N] = pow(fact[N], P - 2, P)
    for i in range(N, 0, -1):
        invfact[i - 1] = invfact[i] * i % P

    def pack(coeffs):
        res = 0
        for i, v in enumerate(coeffs):
            if v:
                res += v << (SHIFT * i)
        return res

    def reduce_packed(packed):
        res = 0
        i = 0
        while packed and i <= MAX_DEG:
            v = packed & MASK
            if v:
                v %= P
                if v:
                    res += v << (SHIFT * i)
            packed >>= SHIFT
            i += 1
        return res

    # trans[s][n] = ((y^s - 1)^n) * y^{C(n,2)} / n! in the basis y = 1+x
    trans = [[0] * (N + 1) for _ in range(N + 1)]
    for s in range(1, N + 1):
        for n in range(1, N - s + 1):
            internal = n * (n - 1) // 2
            coeffs = [0] * (MAX_DEG + 1)
            invf = invfact[n]

            for j in range(n + 1):
                q = s * j + internal
                if q > MAX_DEG:
                    continue
                v = comb[n][j] * invf % P
                if (n - j) & 1:
                    v = (-v) % P
                coeffs[q] = (coeffs[q] + v) % P

            trans[s][n] = pack(coeffs)

    # dp_by_t[t] maps (parity of current level, last level size, even_count)
    # to a packed polynomial in y-basis.
    dp_by_t = [dict() for _ in range(N + 1)]

    # Level 0: vertex 1 only, even distance, size 1, even_count = 1.
    dp_by_t[1][(0, 1, 1)] = 1

    for t in range(1, N):
        states = dp_by_t[t]
        if not states:
            continue

        max_n = N - t

        for (p, s, e), packed in states.items():
            if e > k or (t - e) > k:
                continue

            packed = reduce_packed(packed)
            if packed == 0:
                continue

            for n in range(1, max_n + 1):
                np = 1 - p

                if np == 0:  # new level has even distance
                    ne = e + n
                    if ne > k:
                        continue
                else:        # new level has odd distance
                    ne = e
                    if t - e + n > k:
                        continue

                tr = trans[s][n]
                if tr == 0:
                    continue

                contrib = packed * tr
                key = (np, n, ne)
                d = dp_by_t[t + n]
                d[key] = d.get(key, 0) + contrib

        dp_by_t[t] = None

    # Sum all final states with even_count = k.
    final_packed = 0
    final_states = dp_by_t[N]
    if final_states:
        for (p, s, e), packed in final_states.items():
            if e == k:
                final_packed += reduce_packed(packed)

    # Convert from y-basis to ordinary x-basis.
    final_coeffs = [0] * (MAX_DEG + 1)
    i = 0
    while final_packed and i <= MAX_DEG:
        v = final_packed & MASK
        if v:
            final_coeffs[i] = v % P
        final_packed >>= SHIFT
        i += 1

    fact_mod = fact[N - 1]
    ans = []

    for m in range(N - 1, MAX_DEG + 1):
        total = 0
        for r in range(m, MAX_DEG + 1):
            cr = final_coeffs[r]
            if cr:
                total = (total + cr * comb[r][m]) % P
        ans.append(str(total * fact_mod % P))

    print(" ".join(ans))


if __name__ == "__main__":
    solve()
```
