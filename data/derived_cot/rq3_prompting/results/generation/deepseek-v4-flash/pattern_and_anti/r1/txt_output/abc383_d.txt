def sieve(n):
    if n < 2:
        return []
    is_prime = bytearray(b'\x01') * (n + 1)
    is_prime[0] = 0
    is_prime[1] = 0
    for i in range(2, math.isqrt(n) + 1):
        if is_prime[i]:
            start = i * i
            is_prime[start:n + 1:i] = b'\x00' * (((n - start) // i) + 1)
    return [i for i in range(n + 1) if is_prime[i]]

```python
import sys
import math
from bisect import bisect_right


def sieve(n):
    if n < 2:
        return []
    is_prime = bytearray(b'\x01') * (n + 1)
    is_prime[0] = 0
    is_prime[1] = 0

    for i in range(2, math.isqrt(n) + 1):
        if is_prime[i]:
            start = i * i
            is_prime[start:n + 1:i] = b'\x00' * (((n - start) // i) + 1)

    return [i for i in range(n + 1) if is_prime[i]]


def main():
    N = int(sys.stdin.readline())
    M = math.isqrt(N)
    primes = sieve(M)

    ans = 0

    # n = p^2 * q^2, where p and q are distinct primes
    for i, p in enumerate(primes):
        if p * p > M:
            break
        limit = M // p
        ans += bisect_right(primes, limit) - (i + 1)

    # n = p^8
    for p in primes:
        if p ** 8 > N:
            break
        ans += 1

    print(ans)


if __name__ == "__main__":
    main()
```
