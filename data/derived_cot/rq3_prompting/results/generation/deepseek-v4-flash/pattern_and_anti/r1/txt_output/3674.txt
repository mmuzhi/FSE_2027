```
from typing import List

class Solution:
    def countNonDecreasingSubarrays(self, nums: List[int], k: int) -> int:
        n = len(nums)
        nge = [n] * (n + 1)
        stack = []
        for i in range(n - 1, -1, -1):
            while stack and nums[stack[-1]] <= nums[i]:
                stack.pop()
            if stack:
                nge[i] = stack[-1]
            stack.append(i)
        
        pref = [0] * (n + 1)
        for i in range(n - 1, -1, -1):
            pref[i] = nums[i] * (nge[i] - i) + pref[nge[i]]
        
        LOG = (n + 1).bit_length()
        up = [nge]
        for _ in range(1, LOG):
            prev = up[-1]
            curr = [prev[prev[i]] for i in range(n + 1)]
            up.append(curr)
        
        pref_sum = [0] * (n + 1)
        for i, x in enumerate(nums):
            pref_sum[i + 1] = pref_sum[i] + x
        
        def get_cost(l: int, r: int) -> int:
            cur = l
            for j in range(LOG - 1, -1, -1):
                nxt = up[j][cur]
                if nxt <= r:
                    cur = nxt
            m = cur
            path_sum = pref[l] - pref[nge[m]]
            max_sum = path_sum - nums[m] * (nge[m] - r - 1)
            return max_sum - (pref_sum[r + 1] - pref_sum[l])
        
        ans = 0
        r = -1
        for l in range(n):
            if r < l - 1:
                r = l - 1
            while r + 1 < n and get_cost(l, r + 1) <= k:
                r += 1
            ans += r - l + 1
        return ans
```

```python
from typing import List

class Solution:
    def countNonDecreasingSubarrays(self, nums: List[int], k: int) -> int:
        n = len(nums)

        # nge[i] = first index to the right with nums[index] > nums[i]
        nge = [n] * (n + 1)
        stack = []
        for i in range(n - 1, -1, -1):
            while stack and nums[stack[-1]] <= nums[i]:
                stack.pop()
            if stack:
                nge[i] = stack[-1]
            stack.append(i)

        # pref[i] = sum of block contributions from i along its NGE chain
        pref = [0] * (n + 1)
        for i in range(n - 1, -1, -1):
            pref[i] = nums[i] * (nge[i] - i) + pref[nge[i]]

        # Binary lifting over NGE parent pointers
        LOG = (n + 1).bit_length()
        up = [nge]
        for _ in range(1, LOG):
            prev = up[-1]
            up.append([prev[prev[i]] for i in range(n + 1)])

        pref_sum = [0] * (n + 1)
        for i, x in enumerate(nums):
            pref_sum[i + 1] = pref_sum[i] + x

        levels = range(LOG - 1, -1, -1)

        def cost(l: int, r: int) -> int:
            # Highest ancestor of l whose index is <= r
            cur = l
            for j in levels:
                nxt = up[j][cur]
                if nxt <= r:
                    cur = nxt

            m = cur
            path_sum = pref[l] - pref[nge[m]]
            max_prefix_sum = path_sum - nums[m] * (nge[m] - r - 1)
            return max_prefix_sum - (pref_sum[r + 1] - pref_sum[l])

        ans = 0
        r = -1

        for l in range(n):
            if r < l - 1:
                r = l - 1

            while r + 1 < n and cost(l, r + 1) <= k:
                r += 1

            ans += r - l + 1

        return ans
```
