class Solution:
    def numberOfSubstrings(self, s: str, k: int) -> int:
        n = len(s)
        arr = [ord(c) - 97 for c in s]
        ans = 0

        for left in range(n):
            cnt = [0] * 26
            for right in range(left, n):
                idx = arr[right]
                cnt[idx] += 1
                if cnt[idx] >= k:
                    ans += n - right
                    break

        return ans
