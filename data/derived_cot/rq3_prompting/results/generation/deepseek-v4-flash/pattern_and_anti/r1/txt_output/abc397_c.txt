import sys

def main():
    input = sys.stdin.buffer.readline
    N = int(input())
    A = list(map(int, input().split()))

    pref = [0] * (N + 1)
    seen = [False] * (N + 1)
    cnt = 0
    for i, x in enumerate(A, 1):
        if not seen[x]:
            seen[x] = True
            cnt += 1
        pref[i] = cnt

    seen = [False] * (N + 1)
    cnt = 0
    ans = 0
    for j in range(N, 1, -1):
        x = A[j - 1]
        if not seen[x]:
            seen[x] = True
            cnt += 1
        val = pref[j - 1] + cnt
        if val > ans:
            ans = val

    print(ans)

if __name__ == "__main__":
    main()
