import sys
from bisect import bisect_left, bisect_right

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    it = iter(data)

    N = next(it)
    X = [next(it) for _ in range(N)]
    P = [next(it) for _ in range(N)]

    pref = [0] * (N + 1)
    for i in range(N):
        pref[i + 1] = pref[i] + P[i]

    Q = next(it)
    ans = []

    for _ in range(Q):
        L = next(it)
        R = next(it)

        left = bisect_left(X, L)
        right = bisect_right(X, R)

        ans.append(str(pref[right] - pref[left]))

    sys.stdout.write("\n".join(ans))

if __name__ == "__main__":
    main()
