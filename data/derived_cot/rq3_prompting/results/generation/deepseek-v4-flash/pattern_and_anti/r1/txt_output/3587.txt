```python
    def maxScore(self, n: int, k: int, stayScore: List[List[int]], travelScore: List[List[int]]) -> int:
```

```python
from typing import List

class Solution:
    def maxScore(self, n: int, k: int, stayScore: List[List[int]], travelScore: List[List[int]]) -> int:
        dp = [0] * n
        NEG = -10**18

        for day in range(k):
            ndp = [NEG] * n

            for curr in range(n):
                base = dp[curr]

                # Stay in the current city
                ndp[curr] = max(ndp[curr], base + stayScore[day][curr])

                # Move to another city
                for dest in range(n):
                    if dest == curr:
                        continue
                    ndp[dest] = max(ndp[dest], base + travelScore[curr][dest])

            dp = ndp

        return max(dp)
```
