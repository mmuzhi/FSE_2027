import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    N = data[0]
    A = data[1:]

    last = [0] * (N + 1)
    ans = 0
    cur = 0

    for i, x in enumerate(A, 1):
        cur += i - last[x]
        ans += cur
        last[x] = i

    print(ans)

if __name__ == "__main__":
    main()
