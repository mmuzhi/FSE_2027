import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    N, M = data[0], data[1]
    A = data[2:2 + N]
    B = data[2 + N:2 + N + M]

    max_v = max(max(A), max(B))
    INF = N + 1

    first = [INF] * (max_v + 2)

    for i, a in enumerate(A, 1):
        if first[a] == INF:
            first[a] = i

    for v in range(1, max_v + 1):
        if first[v - 1] < first[v]:
            first[v] = first[v - 1]

    ans = []
    for b in B:
        if first[b] == INF:
            ans.append("-1")
        else:
            ans.append(str(first[b]))

    sys.stdout.write("\n".join(ans))

if __name__ == "__main__":
    main()
