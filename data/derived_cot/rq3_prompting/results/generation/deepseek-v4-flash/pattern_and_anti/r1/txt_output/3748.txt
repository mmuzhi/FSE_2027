```python
from typing import List

class Solution:
    def sortMatrix(self, grid: List[List[int]]) -> List[List[int]]:
        n = len(grid)
        for d in range(-(n - 1), n):
            if d >= 0:
                i, j = d, 0
            else:
                i, j = 0, -d

            vals = []
            r, c = i, j
            while r < n and c < n:
                vals.append(grid[r][c])
                r += 1
                c += 1

            if d >= 0:
                vals.sort(reverse=True)
            else:
                vals.sort()

            r, c = i, j
            idx = 0
            while r < n and c < n:
                grid[r][c] = vals[idx]
                r += 1
                c += 1
                idx += 1

        return grid
```
