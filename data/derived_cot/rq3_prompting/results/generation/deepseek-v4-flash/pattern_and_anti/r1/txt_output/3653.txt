from typing import List

class Solution:
    def maxSubarraySum(self, nums: List[int], k: int) -> int:
        INF = 10**30
        min_pref = [INF] * k
        min_pref[0] = 0
        ans = -INF
        pref = 0

        for i, x in enumerate(nums, 1):
            pref += x
            r = i % k

            if min_pref[r] != INF:
                ans = max(ans, pref - min_pref[r])

            if pref < min_pref[r]:
                min_pref[r] = pref

        return ans
