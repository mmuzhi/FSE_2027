```python
import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    N = data[0]
    A = data[1:1 + N]

    count = {}
    for x in A:
        count[x] = count.get(x, 0) + 1

    best_value = -1
    best_label = -1

    for i, x in enumerate(A, 1):
        if count[x] == 1 and x > best_value:
            best_value = x
            best_label = i

    print(best_label if best_label != -1 else -1)

if __name__ == "__main__":
    main()
```
