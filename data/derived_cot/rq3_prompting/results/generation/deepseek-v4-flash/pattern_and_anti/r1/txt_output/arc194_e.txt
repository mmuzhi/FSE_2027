import sys

def get_residues(s, X, Y):
    zeros = []
    ones = []
    for idx, ch in enumerate(s, start=1):
        if ch == '0':
            zeros.append(idx % Y)
        else:
            ones.append(idx % X)
    return zeros, ones

def main():
    input = sys.stdin.readline
    N,X,Y = map(int,input().split())
    S = input().strip()
    T = input().strip()
    zs, os = get_residues(S,X,Y)
    zt, ot = get_residues(T,X,Y)
    if zs == zt and os == ot:
        print("Yes")
    else:
        print("No")
