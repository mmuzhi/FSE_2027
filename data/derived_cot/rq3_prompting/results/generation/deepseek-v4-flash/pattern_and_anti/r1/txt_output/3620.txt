from typing import List

class Solution:
    def maxDistinctElements(self, nums: List[int], k: int) -> int:
        nums.sort()
        ans = 0
        last = -10**30
        for x in nums:
            lo = x - k
            hi = x + k
            if lo > last:
                cur = lo
            else:
                cur = last + 1
            if cur <= hi:
                ans += 1
                last = cur
        return ans

from typing import List

class Solution:
    def maxDistinctElements(self, nums: List[int], k: int) -> int:
        nums.sort()
        ans = 0
        last = -10**30
        for x in nums:
            val = max(x - k, last + 1)
            if val <= x + k:
                ans += 1
                last = val
        return ans
