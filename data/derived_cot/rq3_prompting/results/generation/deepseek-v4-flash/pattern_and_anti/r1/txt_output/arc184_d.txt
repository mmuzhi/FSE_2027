def main():

import sys

MOD = 998244353

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    N = data[0]
    perm = [0] * (N + 1)

    idx = 1
    for _ in range(N):
        x = data[idx]
        y = data[idx + 1]
        idx += 2
        perm[x] = y

    # 2D prefix sum of real points
    M = N + 2
    grid = [[0] * M for _ in range(M)]
    for x in range(1, N + 1):
        grid[x][perm[x]] = 1

    pref = [[0] * M for _ in range(M)]
    for i in range(M):
        row_sum = 0
        for j in range(M):
            row_sum += grid[i][j]
            pref[i][j] = row_sum + (pref[i - 1][j] if i > 0 else 0)

    def good(ax, ay, bx, by):
        # Rectangle: ax < x < bx, by < y < ay
        lx = ax + 1
        rx = bx - 1
        ly = by + 1
        ry = ay - 1

        if lx > rx or ly > ry:
            return True

        cnt = pref[rx][ry] - pref[lx - 1][ry] - pref[rx][ly - 1] + pref[lx - 1][ly - 1]
        if cnt == 0:
            return True
        if cnt == 1:
            return False

        for x in range(lx, rx + 1):
            y = perm[x]
            if ly <= y <= ry:
                # Points SW of (x, y) inside the rectangle
                slx, srx = lx, x - 1
                sly, sry = ly, y - 1
                if slx <= srx and sly <= sry:
                    sw = pref[srx][sry] - pref[slx - 1][sry] - pref[srx][sly - 1] + pref[slx - 1][sly - 1]
                else:
                    sw = 0

                if sw == 0:
                    # Points NE of (x, y) inside the rectangle
                    nlx, nrx = x + 1, rx
                    nly, nry = y + 1, ry
                    if nlx <= nrx and nly <= nry:
                        ne = pref[nrx][nry] - pref[nlx - 1][nry] - pref[nrx][nly - 1] + pref[nlx - 1][nly - 1]
                    else:
                        ne = 0

                    if ne == 0:
                        return False

        return True

    # virtual start: (0, N+1), virtual end: (N+1, 0)
    verts = [(0, N + 1)]
    for x in range(1, N + 1):
        verts.append((x, perm[x]))
    verts.append((N + 1, 0))

    dp = [0] * (N + 2)
    dp[0] = 1

    for i in range(N + 2):
        if dp[i] == 0:
            continue
        ax, ay = verts[i]
        for j in range(i + 1, N + 2):
            bx, by = verts[j]
            if ay > by:
                if good(ax, ay, bx, by):
                    dp[j] = (dp[j] + dp[i]) % MOD

    print(dp[N + 1] % MOD)


if __name__ == "__main__":
    main()
