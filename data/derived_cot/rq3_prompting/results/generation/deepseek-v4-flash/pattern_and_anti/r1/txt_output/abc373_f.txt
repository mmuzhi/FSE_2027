You will be given a competitive programming problem.
There are N types of items. The i-th type of item has a weight of w_i and a value of v_i. Each type has 10^{10} items available.
Takahashi is going to choose some items and put them into a bag with capacity W. He wants to maximize the value of the selected items while avoiding choosing too many items of the same type. Hence, he defines the happiness of choosing k_i items of type i as k_i v_i - k_i^2. He wants to choose items to maximize the total happiness over all types while keeping the total weight at most W. Calculate the maximum total happiness he can achieve.

Input

The input is given from Standard Input in the following format:
N W
w_1 v_1
w_2 v_2
\vdots
w_N v_N

Output

Print the answer.

Constraints


- 1 \leq N \leq 3000
- 1 \leq W \leq 3000
- 1 \leq w_i \leq W
- 1 \leq v_i \leq 10^9
- All input values are integers.

Sample Input 1

2 10
3 4
3 2

Sample Output 1

5

By choosing 2 items of type 1 and 1 item of type 2, the total happiness can be 5, which is optimal.
Here, the happiness for type 1 is 2 \times 4 - 2^2 = 4, and the happiness for type 2 is 1 \times 2 - 1^2 = 1.
The total weight is 9, which is within the capacity 10.

```python
import sys

def solve():
    input = sys.stdin.buffer.readline
    N, W = map(int, input().split())
    NEG = -10**30
    NEG_HALF = NEG // 2
    dp = [NEG] * (W + 1)
    dp[0] = 0

    ha = []
    hb = []

    for _ in range(N):
        w, v = map(int, input().split())
        ndp = [NEG] * (W + 1)

        limit = w
        tmp = W - w + 1
        if tmp < limit:
            limit = tmp

        dp_local = dp
        ndp_local = ndp
        W_local = W
        w_local = w
        v_local = v
        NEG_HALF_LOCAL = NEG_HALF

        for r in range(limit):
            ha.clear()
            hb.clear()
            ptr = 0
            m = 0
            c = r
            base = 0
            vplus = 0
            step = 1

            while c <= W_local:
                old = dp_local[c]
                if old > NEG_HALF_LOCAL:
                    a = m << 1
                    b = old - vplus

                    while len(ha) >= 2:
                        a1 = ha[-2]
                        b1 = hb[-2]
                        a2 = ha[-1]
                        b2 = hb[-1]
                        if (b1 - b2) * (a - a2) >= (b2 - b) * (a2 - a1):
                            ha.pop()
                            hb.pop()
                        else:
                            break

                    ha.append(a)
                    hb.append(b)

                    if ptr >= len(ha):
                        ptr = len(ha) - 1

                if ha:
                    while ptr + 1 < len(ha) and ha[ptr] * m + hb[ptr] <= ha[ptr + 1] * m + hb[ptr + 1]:
                        ptr += 1
                    best = ha[ptr] * m + hb[ptr]
                    ndp_local[c] = base + best

                base += v_local - step
                vplus += v_local + step
                step += 2
                m += 1
                c += w_local

        if limit < w:
            ndp_local[limit:w] = dp_local[limit:w]

        dp = ndp

    print(max(dp))

if __name__ == "__main__":
    solve()
```

```python
import sys

def solve():
    input = sys.stdin.buffer.readline
    N, W = map(int, input().split())

    NEG = -10**30
    NEG_HALF = NEG // 2

    dp = [NEG] * (W + 1)
    dp[0] = 0

    ha = []
    hb = []

    for _ in range(N):
        w, v = map(int, input().split())
        ndp = [NEG] * (W + 1)

        # residues r with r + w <= W need CHT;
        # the rest can only keep the old value.
        limit = min(w, W - w + 1)

        dp_local = dp
        ndp_local = ndp
        W_local = W
        w_local = w
        v_local = v
        NEG_HALF_LOCAL = NEG_HALF

        for r in range(limit):
            ha.clear()
            hb.clear()

            ptr = 0
            m = 0
            c = r

            base = 0       # v*m - m*m
            vplus = 0      # v*m + m*m
            step = 1       # 2*m + 1

            while c <= W_local:
                old = dp_local[c]

                if old > NEG_HALF_LOCAL:
                    a = m << 1
                    b = old - vplus

                    while len(ha) >= 2:
                        a1 = ha[-2]
                        b1 = hb[-2]
                        a2 = ha[-1]
                        b2 = hb[-1]

                        if (b1 - b2) * (a - a2) >= (b2 - b) * (a2 - a1):
                            ha.pop()
                            hb.pop()
                        else:
                            break

                    ha.append(a)
                    hb.append(b)

                    if ptr >= len(ha):
                        ptr = len(ha) - 1

                if ha:
                    while (
                        ptr + 1 < len(ha)
                        and ha[ptr] * m + hb[ptr] <= ha[ptr + 1] * m + hb[ptr + 1]
                    ):
                        ptr += 1

                    best = ha[ptr] * m + hb[ptr]
                    ndp_local[c] = base + best

                base += v_local - step
                vplus += v_local + step
                step += 2
                m += 1
                c += w_local

        if limit < w:
            ndp_local[limit:w] = dp_local[limit:w]

        dp = ndp

    print(max(dp))


if __name__ == "__main__":
    solve()
```
