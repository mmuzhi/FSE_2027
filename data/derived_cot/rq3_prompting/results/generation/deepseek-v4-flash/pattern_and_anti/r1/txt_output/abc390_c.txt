```python
import sys

def main():
    input = sys.stdin.readline
    H, W = map(int, input().split())
    S = [input().strip() for _ in range(H)]

    min_r, max_r = H, -1
    min_c, max_c = W, -1

    for i in range(H):
        for j, ch in enumerate(S[i]):
            if ch == '#':

                if i > max_r:
                    max_r = i
                if j < min_c:
                    min_c = j
                if j > max_c:
                    max_c = j

    if max_r == -1:

        return

    for i in range(min_r, max_r + 1):
        if S[i].find('.', min_c, max_c + 1) != -1:
            print("No")
            return

    print("Yes")

if __name__ == "__main__":
```

```python
import sys

def main():
    input = sys.stdin.readline
    H, W = map(int, input().split())
    S = [input().strip() for _ in range(H)]

    min_r, max_r = H, -1
    min_c, max_c = W, -1

    for i in range(H):
        for j, ch in enumerate(S[i]):
            if ch == '#':
                min_r = min(min_r, i)
                max_r = max(max_r, i)
                min_c = min(min_c, j)
                max_c = max(max_c, j)

    if max_r == -1:
        print("No")
        return

    for i in range(min_r, max_r + 1):
        if S[i].find('.', min_c, max_c + 1) != -1:
            print("No")
            return

    print("Yes")

if __name__ == "__main__":
    main()
```
