from typing import List

class Solution:
    def minValidStrings(self, words: List[str], target: str) -> int:
        trie = [{}]
        for w in words:
            node = 0
            for ch in w:
                c = ord(ch) - 97
                nxt = trie[node].get(c)
                if nxt is None:
                    nxt = len(trie)
                    trie[node][c] = nxt
                    trie.append({})
                node = nxt

        t = [ord(ch) - 97 for ch in target]
        n = len(target)

        jumps = 0
        cur_end = 0
        farthest = 0

        for i in range(n):
            if i > cur_end:
                return -1

            node = 0
            j = i
            while j < n:
                nxt = trie[node].get(t[j])
                if nxt is None:
                    break
                node = nxt
                j += 1

            if j > farthest:
                farthest = j

            if farthest >= n:
                return jumps + 1

            if i == cur_end:
                jumps += 1
                cur_end = farthest

        return -1

from typing import List

class Solution:
    def minValidStrings(self, words: List[str], target: str) -> int:
        max_nodes = sum(len(w) for w in words) + 1
        trie = [-1] * (max_nodes * 26)
        node_count = 1

        for w in words:
            node = 0
            for ch in w:
                c = ord(ch) - 97
                idx = node * 26 + c
                nxt = trie[idx]
                if nxt == -1:
                    nxt = node_count
                    node_count += 1
                    trie[idx] = nxt
                node = nxt

        t = [ord(ch) - 97 for ch in target]
        n = len(target)

        jumps = 0
        cur_end = 0
        farthest = 0

        for i in range(n):
            if i > cur_end:
                return -1

            node = 0
            j = i
            while j < n:
                nxt = trie[node * 26 + t[j]]
                if nxt == -1:
                    break
                node = nxt
                j += 1

            if j > farthest:
                farthest = j

            if farthest >= n:
                return jumps + 1

            if i == cur_end:
                jumps += 1
                cur_end = farthest

        return -1
```

```python
from typing import List

class Solution:
    def minValidStrings(self, words: List[str], target: str) -> int:
        max_nodes = sum(len(w) for w in words) + 1
        trie = [-1] * (max_nodes * 26)
        node_count = 1

        for w in words:
            node = 0
            for ch in w:
                c = ord(ch) - 97
                idx = node * 26 + c
                nxt = trie[idx]
                if nxt == -1:
                    nxt = node_count
                    node_count += 1
                    trie[idx] = nxt
                node = nxt

        t = [ord(ch) - 97 for ch in target]
        n = len(target)

        jumps = 0
        cur_end = 0
        farthest = 0

        for i in range(n):
            if i > cur_end:
                return -1

            node = 0
            j = i
            while j < n:
                nxt = trie[node * 26 + t[j]]
                if nxt == -1:
                    break
                node = nxt
                j += 1

            if j > farthest:
                farthest = j

            if farthest >= n:
                return jumps + 1

            if i == cur_end:
                jumps += 1
                cur_end = farthest

        return -1
