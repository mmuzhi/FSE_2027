import sys

def main():
    data = sys.stdin.read().split()
    if not data:
        return

    N = int(data[0])
    M = int(data[1])

    LIMIT = 10 ** 9
    total = 0
    term = 1

    for _ in range(M + 1):
        total += term
        if total > LIMIT:
            print("inf")
            return
        term *= N

    print(total)

if __name__ == "__main__":
    main()
