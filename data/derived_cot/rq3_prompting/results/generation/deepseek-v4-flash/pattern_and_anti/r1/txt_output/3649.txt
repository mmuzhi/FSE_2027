class Solution:
    def findMinimumTime(self, strength: List[int], K: int) -> int:
        n = len(strength)

class Solution:
    def findMinimumTime(self, strength: List[int], K: int) -> int:
        n = len(strength)
        total_masks = 1 << n
        INF = 10**18
        dp = [INF] * total_masks
        dp[0] = 0

        pop = [0] * total_masks
        for mask in range(1, total_masks):

from typing import List

class Solution:
    def findMinimumTime(self, strength: List[int], K: int) -> int:
        n = len(strength)
        total = 1 << n
        INF = 10**18
        dp = [INF] * total
        dp[0] = 0

        pop = [0] * total
        for mask in range(1, total):
            pop[mask] = pop[mask >> 1] + (mask & 1)

        for mask in range(total):
            if dp[mask] == INF:
                continue
            x = 1 + K * pop[mask]
            for i in range(n):
                if mask & (1 << i):
                    continue
                wait = (strength[i] + x - 1) // x
                nxt = mask | (1 << i)
                if dp[mask] + wait < dp[nxt]:
                    dp[nxt] = dp[mask] + wait

        return dp[total - 1]

from typing import List

class Solution:
    def findMinimumTime(self, strength: List[int], K: int) -> int:
        n = len(strength)
        total = 1 << n
        INF = 10**18

        dp = [INF] * total
        dp[0] = 0

        pop = [0] * total
        for mask in range(1, total):
            pop[mask] = pop[mask >> 1] + (mask & 1)

        for mask in range(total):
            if dp[mask] == INF:
                continue

            x = 1 + K * pop[mask]

            for i in range(n):
                if mask & (1 << i):
                    continue

                wait = (strength[i] + x - 1) // x
                nxt = mask | (1 << i)
                dp[nxt] = min(dp[nxt], dp[mask] + wait)

        return dp[total - 1]
