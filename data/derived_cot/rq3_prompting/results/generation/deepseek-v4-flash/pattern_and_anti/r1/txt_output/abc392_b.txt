=== ANSWER ===
```python
import sys

def main():
    data = list(map(int, sys.stdin.read().split()))
    N, M = data[0], data[1]
    A = data[2:2 + M]

    present = [False] * (N + 1)
    for x in A:
        present[x] = True

    missing = [i for i in range(1, N + 1) if not present[i]]

    print(len(missing))
    if missing:
        print(*missing)
    else:
        print()

if __name__ == "__main__":
    main()
```
