```python
from typing import List
from itertools import permutations

class Solution:
    def totalNumbers(self, digits: List[int]) -> int:
        seen = set()
        
        for a, b, c in permutations(digits, 3):
            if a != 0 and c % 2 == 0:
                seen.add(a * 100 + b * 10 + c)
        
        return len(seen)
```
