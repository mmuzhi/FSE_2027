=== ANSWER ===
```python
from typing import List

class Solution:
    def minElement(self, nums: List[int]) -> int:
        def digit_sum(x: int) -> int:
            total = 0
            while x > 0:
                total += x % 10
                x //= 10
            return total

        return min(digit_sum(num) for num in nums)
```
