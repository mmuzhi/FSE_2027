```python
from typing import List

class Solution:
    def assignElements(self, groups: List[int], elements: List[int]) -> List[int]:
        limit = max(groups)
        ans = [-1] * (limit + 1)
        seen = [False] * (limit + 1)

        for idx, e in enumerate(elements):
            if e > limit or seen[e]:
                continue
            seen[e] = True

            for multiple in range(e, limit + 1, e):
                if ans[multiple] == -1:
                    ans[multiple] = idx

        return [ans[g] for g in groups]
```
