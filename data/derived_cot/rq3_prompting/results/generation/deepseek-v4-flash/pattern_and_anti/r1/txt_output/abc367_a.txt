=== ANSWER ===
```python
import sys

def main():
    A, B, C = map(int, sys.stdin.readline().split())

    if B < C:
        sleeping = B <= A < C
    else:
        sleeping = A >= B or A < C

    print("No" if sleeping else "Yes")

if __name__ == "__main__":
    main()
```
