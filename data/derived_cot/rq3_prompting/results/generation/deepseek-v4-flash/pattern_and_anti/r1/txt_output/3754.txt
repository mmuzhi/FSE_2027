```python
class Solution:
    def maxDistance(self, s: str, k: int) -> int:
        ans = 0

        for sx in (1, -1):
            for sy in (1, -1):
                cur = 0
                bad = 0

                for ch in s:
                    if ch == 'N':
                        v = sy
                    elif ch == 'S':
                        v = -sy
                    elif ch == 'E':
                        v = sx
                    else:  # 'W'
                        v = -sx

                    cur += v
                    if v == -1:
                        bad += 1

                    ans = max(ans, cur + 2 * min(k, bad))

        return ans
```
