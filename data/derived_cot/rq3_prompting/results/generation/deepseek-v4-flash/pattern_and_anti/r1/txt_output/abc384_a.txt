import sys

def main():
    data = sys.stdin.read().split()
    N = int(data[0])
    c1 = data[1]
    c2 = data[2]
    S = data[3]

    ans = ''.join(c2 if ch != c1 else ch for ch in S)
    print(ans)

if __name__ == "__main__":
    main()
