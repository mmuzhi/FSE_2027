```
from typing import List

class Solution:
    def validSequence(self, word1: str, word2: str) -> List[int]:
        n, m = len(word1), len(word2)
        p0 = [0]*(n+1)
        p1 = [0]*(n+1)
        p0[n] = m - 1
        p1[n] = m - 1
        for i in range(n-1, -1, -1):
            c = word1[i]
            p0_next = p0[i+1]
            p1_next = p1[i+1]
            if p0_next >= 0 and c == word2[p0_next]:
                p0[i] = p0_next - 1
            else:
                p0[i] = p0_next
            p1[i] = p1_next
            if p1_next >= 0 and (c == word2[p1_next] or p1_next == p0_next):
                p1[i] = p1_next - 1
        if p1[0] >= 0:
            return []
        ans = []
        prev = -1
        used = False
        for j in range(m):
            i = prev + 1
            while i < n:
                if used:
                    ok = (word1[i] == word2[j] and p0[i+1] < j + 1)
                else:
                    if word1[i] == word2[j]:
                        ok = p1[i+1] < j + 1
                    else:
                        ok = p0[i+1] < j + 1
                if ok:
                    break
                i += 1
            if i == n:
                return []
            ans.append(i)
            if not used and word1[i] != word2[j]:
                used = True
            prev = i
        return ans
```

```python
from typing import List

class Solution:
    def validSequence(self, word1: str, word2: str) -> List[int]:
        n, m = len(word1), len(word2)

        # p0[i]: first unmatched index in word2 after exactly matching
        #        word2 from the right inside word1[i:].
        # p1[i]: same, but at most one mismatch is allowed.
        p0 = [0] * (n + 1)
        p1 = [0] * (n + 1)
        p0[n] = m - 1
        p1[n] = m - 1

        for i in range(n - 1, -1, -1):
            c = word1[i]
            p0_next = p0[i + 1]
            p1_next = p1[i + 1]

            if p0_next >= 0 and c == word2[p0_next]:
                p0[i] = p0_next - 1
            else:
                p0[i] = p0_next

            p1[i] = p1_next
            if p1_next >= 0 and (c == word2[p1_next] or p1_next == p0_next):
                p1[i] = p1_next - 1

        if p1[0] >= 0:
            return []

        ans = []
        prev = -1
        used = False

        for j in range(m):
            i = prev + 1
            while i < n:
                if used:
                    ok = word1[i] == word2[j] and p0[i + 1] < j + 1
                else:
                    ok = (
                        (word1[i] == word2[j] and p1[i + 1] < j + 1)
                        or p0[i + 1] < j + 1
                    )

                if ok:
                    break
                i += 1

            if i == n:
                return []

            ans.append(i)
            if not used and word1[i] != word2[j]:
                used = True
            prev = i

        return ans
```
