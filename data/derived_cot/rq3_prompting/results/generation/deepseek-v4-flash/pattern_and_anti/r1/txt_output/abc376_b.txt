It is guaranteed that only achievable instructions are given.

Details

import sys
def main():
    input=sys.stdin.readline
    N,Q=map(int,input().split())
    l=1; r=2
    ans=0
    for _ in range(Q):
        H,T = input().split()
        T=int(T)
        if H=='L':
            a,b,o = l,T,r
        else:
            a,b,o = r,T,l
        cw=(b-a)%N
        if cw==0:
            dist=0
        else:
            if ((o-a)%N) < cw:
                dist=N-cw
            else:
                dist=cw
        ans += dist
        if H=='L':
            l=T
        else:
            r=T
    print(ans)
if __name__=="__main__": main()

def main():
    data = sys.stdin.read().split()
    if not data:
        return
    N = int(data[0])
    Q = int(data[1])
    left = 1
    right = 2
    ans = 0
    idx = 2
    for _ in range(Q):
        h = data[idx]
        t = int(data[idx + 1])
        idx += 2

        if h == 'L':
            a, b, other = left, t, right
        else:
            a, b, other = right, t, left

        cw = (b - a) % N
        if cw == 0:
            dist = 0
        elif ((other - a) % N) < cw:
            dist = N - cw
        else:
            dist = cw

        ans += dist

        if h == 'L':
            left = t
        else:
            right = t

    print(ans)

if __name__ == "__main__":

import sys

def main():
    data = sys.stdin.read().split()
    if not data:
        return

    N = int(data[0])
    Q = int(data[1])

    left = 1
    right = 2
    ans = 0
    idx = 2

    for _ in range(Q):
        h = data[idx]
        t = int(data[idx + 1])
        idx += 2

        if h == 'L':
            a, b, other = left, t, right
        else:
            a, b, other = right, t, left

        cw = (b - a) % N

        if cw == 0:
            dist = 0
        elif ((other - a) % N) < cw:
            dist = N - cw
        else:
            dist = cw

        ans += dist

        if h == 'L':
            left = t
        else:
            right = t

    print(ans)

if __name__ == "__main__":
    main()
