```python
import sys
from collections import deque

def main():
    input = sys.stdin.buffer.readline

    H, W = map(int, input().split())
    S = [input().strip() for _ in range(H)]

    A, B, C, D = map(int, input().split())
    s = (A - 1) * W + (B - 1)
    t = (C - 1) * W + (D - 1)

    N = H * W
    dist = [-1] * N
    done = bytearray(N)

    dq = deque([s])
    dist[s] = 0

    DOT = ord('.')

    while dq:
        v = dq.popleft()
        if done[v]:
            continue
        done[v] = 1

        d = dist[v]
        if v == t:
            print(d)
            return

        r = v // W
        c = v - r * W

        # Move / kick to adjacent cell
        if r > 0:
            ni = v - W
            if S[r - 1][c] == DOT:
                if dist[ni] < 0 or d < dist[ni]:
                    dist[ni] = d
                    dq.appendleft(ni)
            else:
                nd = d + 1
                if dist[ni] < 0 or nd < dist[ni]:
                    dist[ni] = nd
                    dq.append(ni)

        if r + 1 < H:
            ni = v + W
            if S[r + 1][c] == DOT:
                if dist[ni] < 0 or d < dist[ni]:
                    dist[ni] = d
                    dq.appendleft(ni)
            else:
                nd = d + 1
                if dist[ni] < 0 or nd < dist[ni]:
                    dist[ni] = nd
                    dq.append(ni)

        if c > 0:
            ni = v - 1
            if S[r][c - 1] == DOT:
                if dist[ni] < 0 or d < dist[ni]:
                    dist[ni] = d
                    dq.appendleft(ni)
            else:
                nd = d + 1
                if dist[ni] < 0 or nd < dist[ni]:
                    dist[ni] = nd
                    dq.append(ni)

        if c + 1 < W:
            ni = v + 1
            if S[r][c + 1] == DOT:
                if dist[ni] < 0 or d < dist[ni]:
                    dist[ni] = d
                    dq.appendleft(ni)
            else:
                nd = d + 1
                if dist[ni] < 0 or nd < dist[ni]:
                    dist[ni] = nd
                    dq.append(ni)

        # Front kick to the cell 2 steps away
        if r >= 2:
            ni = v - 2 * W
            nd = d + 1
            if dist[ni] < 0 or nd < dist[ni]:
                dist[ni] = nd
                dq.append(ni)

        if r + 2 < H:
            ni = v + 2 * W
            nd = d + 1
            if dist[ni] < 0 or nd < dist[ni]:
                dist[ni] = nd
                dq.append(ni)

        if c >= 2:
            ni = v - 2
            nd = d + 1
            if dist[ni] < 0 or nd < dist[ni]:
                dist[ni] = nd
                dq.append(ni)

        if c + 2 < W:
            ni = v + 2
            nd = d + 1
            if dist[ni] < 0 or nd < dist[ni]:
                dist[ni] = nd
                dq.append(ni)

if __name__ == "__main__":
    main()
```
