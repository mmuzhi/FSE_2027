import sys
from bisect import bisect_right

def make_sums(arr):
    sums = [0]
    for x in arr:
        sums += [s + x for s in sums]
    sums.sort()
    return sums

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    n = data[0]
    K = data[1:1+n]
    total = sum(K)
    mid = n // 2

    left = make_sums(K[:mid])
    right = make_sums(K[mid:])

    target = total // 2
    best = 0

    for a in left:
        if a > target:
            break
        idx = bisect_right(right, target - a) - 1
        if idx >= 0:
            s = a + right[idx]
            if s > best:
                best = s

    print(total - best)

if __name__ == "__main__":

import sys
from bisect import bisect_right

def make_sums(arr):
    sums = [0]
    for x in arr:
        sums += [s + x for s in sums]
    sums.sort()
    return sums

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n = data[0]
    K = data[1:1 + n]
    total = sum(K)

    mid = n // 2
    left = make_sums(K[:mid])
    right = make_sums(K[mid:])

    target = total // 2
    best = 0

    for a in left:
        if a > target:
            break
        idx = bisect_right(right, target - a) - 1
        if idx >= 0:
            best = max(best, a + right[idx])

    print(total - best)
