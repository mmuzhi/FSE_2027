=== ANSWER ===
```python
import sys

def main():
    input = sys.stdin.readline
    N, D = map(int, input().split())
    S = input().strip()

    total_cookies = S.count('@')
    keep_cookies = total_cookies - D

    ans = []
    seen = 0

    for ch in S:
        if ch == '@':
            if seen < keep_cookies:
                ans.append('@')
            else:
                ans.append('.')
            seen += 1
        else:
            ans.append('.')

    print(''.join(ans))

if __name__ == "__main__":
    main()
```
