import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data: return
    N = data[0]
    P = data[1:]
    bit = [0]*(N+2)
    def add(idx, val):
        while idx <= N:
            bit[idx] += val
            idx += idx & -idx
    def sum_prefix(idx):
        s=0
        while idx>0:
            s += bit[idx]
            idx -= idx & -idx
        return s
    ans=0
    for i, v in enumerate(P, start=1):
        le = sum_prefix(v)  # seen values <= v
        b = (i-1) - le      # seen values > v
        if b:
            ans += b * i - b*(b+1)//2
        add(v,1)
    print(ans)

if __name__ == "__main__":
    main()

import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    N = data[0]
    P = data[1:1 + N]

        s = 0
        x = v

            s += bit[x]
            x -= x & -x

        # Number of larger values before v
        c = (i - 1) - s

        # Insertion sort contribution for this value
        ans += c * i - c * (c + 1) // 2

        # Add v to Fenwick tree
        x = v
        while x <= N:
            bit[x] += 1
            x += x & -x

    print(ans)
