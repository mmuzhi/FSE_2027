def solve():
    import sys
    data = sys.stdin.buffer.read().split()
    K = int(data[0]); S=data[1]; T=data[2]
    if len(S)>len(T): S,T=T,S
    n=len(S); m=len(T)
    if m-n>K: print("No"); return
    L=2*K+1
    INF=10**9
    prev=[INF]*L
    # base i=0
    # d from -K..K, j=d
    for d in range(0, min(K,m)+1): # d>=0
        prev[d+K]=d
    # d negative remain INF
    # If m < K, d>m remain INF.
    for i in range(1, n+1):
        curr=[INF]*L
        lo = -K
        if -i > lo: lo = -i
        hi = K
        if m - i < hi: hi = m - i
        si = S[i-1]
        start = lo + K
        end = hi + K
        # local variables
        prev_l=prev; curr_l=curr; T_l=T; INF_l=INF; L_l=L; K_l=K; i_l=i
        for idx in range(start, end+1):
            j = i_l + idx - K_l
            # delete
            if idx + 1 < L_l:
                best = prev_l[idx + 1] + 1
            else:
                best = INF_l
            if j:
                # insert
                if idx > 0:
                    v = curr_l[idx - 1] + 1
                    if v < best: best = v
                # substitute
                v = prev_l[idx] + (si != T_l[j - 1])
                if v < best: best = v
            curr_l[idx] = best
        prev = curr
    ans = prev[(m-n)+K]
    print("Yes" if ans <= K else "No")

import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    K = int(data[0])
    S = data[1]
    T = data[2]

    if len(S) > len(T):
        S, T = T, S
    n = len(S)
    m = len(T)

    if m - n > K:
        print("No")
        return
    if S == T:
        print("Yes")
        return

    L = 2 * K + 1
    INF = 10**9
    prev = [INF] * (L + 2)

    # i = 0
    # d = j - i = j, valid d in [0, min(K, m)]
    for d in range(min(K, m) + 1):
        prev[d + K + 1] = d

    for i in range(1, n + 1):
        curr = [INF] * (L + 2)
        start = K + 1 - i
        if start < 1:
            start = 1
        end = K + 1 + m - i
        if end > L:
            end = L
        si = S[i - 1]
        offset = i - K - 1
        # local variables for speed
        prev_l = prev
        curr_l = curr
        T_l = T
        INF_l = INF
        for p in range(start, end + 1):
            j = p + offset
            best = prev_l[p + 1] + 1
            if j:
                v = curr_l[p - 1] + 1
                if v < best:
                    best = v
                v = prev_l[p] + (si != T_l[j - 1])
                if v < best:
                    best = v
            curr_l[p] = best
        prev = curr

    ans = prev[(m - n) + K + 1]
    print("Yes" if ans <= K else "No")

if __name__ == "__main__":
    solve()
