```python
import sys

def main():
    data = sys.stdin.read().split()
    if not data:
        return

    Q = int(data[0])
    stack = [0] * 100
    idx = 1
    ans = []

    for _ in range(Q):
        c = int(data[idx])
        idx += 1

        if c == 1:
            x = int(data[idx])
            idx += 1
            stack.append(x)
        else:
            ans.append(str(stack.pop()))

    print("\n".join(ans))

if __name__ == "__main__":
    main()
```
