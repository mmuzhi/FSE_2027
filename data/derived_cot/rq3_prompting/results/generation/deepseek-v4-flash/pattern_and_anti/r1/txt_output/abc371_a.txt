```python
import sys
from itertools import permutations

def main():
    S_AB, S_AC, S_BC = sys.stdin.readline().split()

    for p in permutations("ABC"):
        rank = {name: i for i, name in enumerate(p)}

        if ((S_AB == '<') == (rank['A'] < rank['B']) and
            (S_AC == '<') == (rank['A'] < rank['C']) and
            (S_BC == '<') == (rank['B'] < rank['C'])):
            print(p[1])
            return

if __name__ == "__main__":
    main()
