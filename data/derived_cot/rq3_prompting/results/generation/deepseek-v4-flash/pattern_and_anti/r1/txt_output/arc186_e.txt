```
import sys

MOD = 998244353

def main():
    data = list(map(int, sys.stdin.read().split()))
    if not data:
        return
    N, M, K = data[0], data[1], data[2]
    X = data[3:3+M]

    fact = [1]*(K+1)
    for i in range(1,K+1):
        fact[i] = fact[i-1]*i % MOD
    invfact = [1]*(K+1)
    invfact[K] = pow(fact[K], MOD-2, MOD)
    for i in range(K,0,-1):
        invfact[i-1] = invfact[i]*i % MOD

    def C(n,r):
        if r<0 or r>n:
            return 0
        return fact[n]*invfact[r]%MOD*invfact[n-r]%MOD

    pows = [[1]*(N+1) for _ in range(K)]
    for b in range(K):
        row = pows[b]
        for t in range(1,N+1):
            row[t] = row[t-1]*b % MOD

    size = K-1
    T = [0]*(N+1)
    for t in range(N+1):
        res = 0
        for j in range(size+1):
            term = C(size, j) * pows[size-j][t] % MOD
            if j & 1:
                res -= term
            else:
                res += term
        T[t] = res % MOD

    H_same = [0]*(N+1)
    for s in range(1,N+1):
        H_same[s] = T[s-1]

    g = [0]*(N+1)
    g[0] = 1
    mult = (K-2) % MOD
    for u in range(1,N+1):
        g[u] = g[u-1] * mult % MOD

    conv = [0]*(N+1)
    for u in range(N+1):
        gu = g[u]
        if gu == 0:
            continue
        limit = N - u
        for t in range(limit+1):
            if T[t]:
                conv[u+t] = (conv[u+t] + gu * T[t]) % MOD

    H_diff = [0]*(N+1)
    for s in range(2,N+1):
        H_diff[s] = conv[s-2]

    nz_same = [(s, h) for s, h in enumerate(H_same) if h]
    nz_diff = [(s, h) for s, h in enumerate(H_diff) if h]

    dp = T[:]  # i=1

    for i in range(2, M+1):
        if X[i-1] == X[i-2]:
            nz = nz_same
        else:
            nz = nz_diff
        new = [0]*(N+1)
        for q, val in enumerate(dp):
            if val == 0:
                continue
            max_s = N - q
            for s, h in nz:
                if s > max_s:
                    break
                new[q+s] += val * h
        for idx in range(N+1):
            new[idx] %= MOD
        dp = new

    print(dp[N] % MOD)

if __name__ == "__main__":
    main()
```

import sys

MOD = 998244353

def main():
    data = list(map(int, sys.stdin.read().split()))
    if not data:
        return

    N, M, K = data[0], data[1], data[2]
    X = data[3:3 + M]

    # factorials for combinations
    fact = [1] * (K + 1)
    for i in range(1, K + 1):
        fact[i] = fact[i - 1] * i % MOD

    invfact = [1] * (K + 1)
    invfact[K] = pow(fact[K], MOD - 2, MOD)
    for i in range(K, 0, -1):
        invfact[i - 1] = invfact[i] * i % MOD

    def C(n, r):
        if r < 0 or r > n:
            return 0
        return fact[n] * invfact[r] % MOD * invfact[n - r] % MOD

    # pows[b][t] = b^t mod MOD
    pows = [[1] * (N + 1) for _ in range(K)]
    for b in range(K):
        row = pows[b]
        for t in range(1, N + 1):
            row[t] = row[t - 1] * b % MOD

    # T[t]: words of length t over a fixed (K-1)-letter alphabet
    # that contain every letter at least once.
    size = K - 1
    T = [0] * (N + 1)
    for t in range(N + 1):
        res = 0
        for j in range(size + 1):
            term = C(size, j) * pows[size - j][t] % MOD
            if j & 1:
                res -= term
            else:
                res += term
        T[t] = res % MOD

    # If X_i == X_{i-1}, the appended symbol itself completes the arch.
    H_same = [0] * (N + 1)
    for s in range(1, N + 1):
        H_same[s] = T[s - 1]

    # If X_i != X_{i-1}, we need R = (some symbols) + X_{i-1}.
    # The number of such R of length r is (K-2)^(r-1).
    g = [0] * (N + 1)
    g[0] = 1
    mult = (K - 2) % MOD
    for i in range(1, N + 1):
        g[i] = g[i - 1] * mult % MOD

    conv = [0] * (N + 1)
    for u in range(N + 1):
        if g[u] == 0:
            continue
        for t in range(N - u + 1):
            if T[t]:
                conv[u + t] = (conv[u + t] + g[u] * T[t]) % MOD

    H_diff = [0] * (N + 1)
    for s in range(2, N + 1):
        H_diff[s] = conv[s - 2]

    nz_same = [(s, h) for s, h in enumerate(H_same) if h]
    nz_diff = [(s, h) for s, h in enumerate(H_diff) if h]

    # dp for missing prefix of length 1
    dp = T[:]

    for i in range(2, M + 1):
        if X[i - 1] == X[i - 2]:
            nz = nz_same
        else:
            nz = nz_diff

        ndp = [0] * (N + 1)

        for q, val in enumerate(dp):
            if val == 0:
                continue
            max_s = N - q
            for s, h in nz:
                if s > max_s:
                    break
                ndp[q + s] += val * h

        for i in range(N + 1):
            ndp[i] %= MOD

        dp = ndp

    print(dp[N] % MOD)


if __name__ == "__main__":
    main()
