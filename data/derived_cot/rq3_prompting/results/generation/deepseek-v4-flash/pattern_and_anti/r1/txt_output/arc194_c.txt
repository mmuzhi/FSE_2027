import sys
from bisect import bisect_left

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    N = data[0]
    A = data[1:1+N]
    B = data[1+N:1+2*N]
    C = data[1+2*N:1+3*N]

    D=[]; U=[]; K=[]
    for a,b,c in zip(A,B,C):
        if a == b:
            if a == 1:
                K.append(c)
        else:
            if a == 1:
                D.append(c)
            else:
                U.append(c)

    D.sort()
    U.sort()
    K.sort()

    def pair_min(arr):
        L = len(arr)
        res = 0
        for i, v in enumerate(arr):
            res += v * (L - 1 - i)
        return res

    prefD = [0]
    for v in D:
        prefD.append(prefD[-1] + v)
    prefU = [0]
    for v in U:
        prefU.append(prefU[-1] + v)

    const = pair_min(D) + prefU[-1] + pair_min(U)

    Ld = len(D)
    Lu = len(U)

    def F(x, arr, pref, L):
        p = bisect_left(arr, x)
        return pref[p] + x * (L - p)

    m = len(K)
    sufK = [0]*(m+1)
    sufPairMin = [0]*(m+1)
    sufCrossD = [0]*(m+1)
    sufCrossU = [0]*(m+1)

    for i in range(m-1, -1, -1):
        x = K[i]
        sufK[i] = sufK[i+1] + x
        sufPairMin[i] = sufPairMin[i+1] + x * (m - 1 - i)
        sufCrossD[i] = sufCrossD[i+1] + F(x, D, prefD, Ld)
        sufCrossU[i] = sufCrossU[i+1] + F(x, U, prefU, Lu)

    Ksum = sufK[0]
    n = Ld + Lu
    ans = 10**30

    for t in range(m+1):
        s = m - t
        Ssum = sufK[t]
        Vsum = Ksum - Ssum
        M = n + 2*s
        total = const + Vsum * M + 2 * sufPairMin[t] + sufCrossD[t] + sufCrossU[t] + Ssum
        if total < ans:
            ans = total

    print(ans)

if __name__ == "__main__":
    main()

import sys
from bisect import bisect_left


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    N = data[0]
    A = data[1:1 + N]
    B = data[1 + N:1 + 2 * N]
    C = data[1 + 2 * N:1 + 3 * N]
    del data

    D = []  # A=1, B=0
    U = []  # A=0, B=1
    K = []  # A=1, B=1

    for a, b, c in zip(A, B, C):
        if a == b:
            if a == 1:
                K.append(c)
        else:
            if a == 1:
                D.append(c)
            else:
                U.append(c)

    del A, B, C

    D.sort()
    U.sort()
    K.sort()

    def pair_min(arr):
        L = len(arr)
        res = 0
        for i, v in enumerate(arr):
            res += v * (L - 1 - i)
        return res

    prefD = [0]
    for v in D:
        prefD.append(prefD[-1] + v)

    prefU = [0]
    for v in U:
        prefU.append(prefU[-1] + v)

    const = pair_min(D) + prefU[-1] + pair_min(U)

    Ld = len(D)
    Lu = len(U)

    def F(x, arr, pref, L):
        p = bisect_left(arr, x)
        return pref[p] + x * (L - p)

    m = len(K)

    sufK = [0] * (m + 1)
    sufPairMin = [0] * (m + 1)
    sufCrossD = [0] * (m + 1)
    sufCrossU = [0] * (m + 1)

    for i in range(m - 1, -1, -1):
        x = K[i]
        sufK[i] = sufK[i + 1] + x
        sufPairMin[i] = sufPairMin[i + 1] + x * (m - 1 - i)
        sufCrossD[i] = sufCrossD[i + 1] + F(x, D, prefD, Ld)
        sufCrossU[i] = sufCrossU[i + 1] + F(x, U, prefU, Lu)

    Ksum = sufK[0]
    n = Ld + Lu

    ans = 10**30

    for t in range(m + 1):
        s = m - t
        Ssum = sufK[t]
        Vsum = Ksum - Ssum
        M = n + 2 * s

        total = (
            const
            + Vsum * M
            + 2 * sufPairMin[t]
            + sufCrossD[t]
            + sufCrossU[t]
            + Ssum
        )

        if total < ans:
            ans = total

    print(ans)


if __name__ == "__main__":
    main()
