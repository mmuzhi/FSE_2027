        lim_s = min(K - 1, n)
        for j in range(lim_s + 1):
            rhs += s[j] * p[n - j]

        lim_q = min(K, n)
        for j in range(1, lim_q + 1):
            rhs -= q[j] * (n - j + 1) * p[n - j + 1]

        p[n + 1] = (rhs % mod) * inv[n + 1] % mod

    fact = 1
    for i in range(2, N + 1):
        fact = fact * i % mod

    integral = [0] * 7

    for d in active:
        ad = a[d]
        r = p[0]
        val = r * inv[1] % mod

        for m in range(1, N):
            r = (p[m] - ad * r) % mod
            val = (val + r * inv[m + 1]) % mod

        integral[d] = val

    ans = 0
    for d in active:
        ans = (ans + sum_len[d] * integral[d]) % mod

    ans = ans * fact % mod
    print(ans)


if __name__ == "__main__":
    main()
