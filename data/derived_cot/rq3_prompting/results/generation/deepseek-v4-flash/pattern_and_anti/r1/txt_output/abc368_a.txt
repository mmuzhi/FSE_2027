```python
import sys

def main():
    data = list(map(int, sys.stdin.read().split()))
    N, K = data[0], data[1]
    A = data[2:2 + N]

    ans = A[N - K:] + A[:N - K]
    print(*ans)

if __name__ == "__main__":
    main()
```
