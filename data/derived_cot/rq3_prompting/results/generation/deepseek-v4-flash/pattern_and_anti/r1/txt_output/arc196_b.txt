        ok = True
        for row in S:
            if row.count(b'A') & 1:
                ok = False
                break
        if not ok:
            out.append("0")
            continue
        N = H + W
        parent = list(range(N))
        weight = [0]*N
        size = [1]*N
        comp = N
        col_parity = [0]*W
        py_col = [0]*W

        def find(x):
            root = x
            parity = 0
            while parent[root] != root:
                parity ^= weight[root]
                root = parent[root]
            # path compression
            cur = x
            cur_parity = parity
            while parent[cur] != cur:
                nxt = parent[cur]
                w = weight[cur]
                parent[cur] = root
                weight[cur] = cur_parity
                cur_parity ^= w
                cur = nxt
            return root, parity

        def unite(u, v, w):
            ru, pu = find(u)
            rv, pv = find(v)
            if ru == rv:
                return 0 if (pu ^ pv) == w else -1
            if size[ru] < size[rv]:
                ru, rv = rv, ru
            parent[rv] = ru
            weight[rv] = w ^ pu ^ pv
            size[ru] += size[rv]
            return 1

        ok = True
        for i, row in enumerate(S):
            px = 0
            # row parity already ok
            for j, ch in enumerate(row):

                col_parity[j] ^= a

                if i == 0:
                    py = 0

```python
            if row.count(b'A') & 1:
                ok = False
                break
        if not ok:
            ans.append("0")
            continue

        N = H + W
        parent = [-1] * N
        weight = [0] * N
        comp = N

        def find(x):
            root = x
            parity = 0
            while parent[root] >= 0:
                parity ^= weight[root]
                root = parent[root]
            cur = x
            cur_parity = parity
            while parent[cur] >= 0:
                nxt = parent[cur]
                w = weight[cur]
                parent[cur] = root
                weight[cur] = cur_parity
                cur_parity ^= w
                cur = nxt
            return root, parity

        def unite(u, v, w):
            ru, pu = find(u)
            rv, pv = find(v)
            if ru == rv:
                return 0 if (pu ^ pv) == w else -1
            if parent[ru] > parent[rv]:
                ru, rv = rv, ru
            d = w ^ pu ^ pv
            parent[ru] += parent[rv]
            parent[rv] = ru
            weight[rv] = d
            return 1

        col_parity = [0] * W
        py_col = [0] * W
        ok = True

        for i, row in enumerate(S):
            px = 0
            for j, ch in enumerate(row):
                a = 1 if ch == A else 0
                col_parity[j] ^= a
                if j > 0:
                    px ^= a
                if i == 0:
                    py = 0
                else:
                    py = py_col[j] ^ a

                if a == 0:
                    b = 1 ^ px ^ py
                    res = unite(i, H + j, b)
                    if res == -1:
                        ok = False
                        break
                    elif res == 1:
                        comp -= 1

                if i > 0:
                    py_col[j] ^= a
            if not ok:
                break

        if not ok:
            ans.append("0")
            continue
        if any(col_parity):
            ans.append("0")
            continue
        ans.append(str(pow(2, comp, MOD)))

    sys.stdout.write("\n".join(ans))
```

```python
import sys

MOD = 998244353
A = ord('A')


def solve():
    input = sys.stdin.buffer.readline
    T = int(input())
    ans = []

    for _ in range(T):
        H, W = map(int, input().split())
        S = [input().strip() for _ in range(H)]

        # Row parity condition for x variables
        ok = True
        for row in S:
            if row.count(b'A') & 1:
                ok = False
                break

        if not ok:
            ans.append("0")
            continue

        N = H + W
        parent = [-1] * N
        weight = [0] * N
        comp = N

        def find(x):
            root = x
            parity = 0
            while parent[root] >= 0:
                parity ^= weight[root]
                root = parent[root]

            cur = x
            cur_parity = parity
            while parent[cur] >= 0:
                nxt = parent[cur]
                w = weight[cur]
                parent[cur] = root
                weight[cur] = cur_parity
                cur_parity ^= w
                cur = nxt

            return root, parity

        def unite(u, v, w):
            # enforce val[u] xor val[v] = w
            ru, pu = find(u)
            rv, pv = find(v)

            if ru == rv:
                return 0 if (pu ^ pv) == w else -1

            if parent[ru] > parent[rv]:
                ru, rv = rv, ru

            d = w ^ pu ^ pv
            parent[ru] += parent[rv]
            parent[rv] = ru
            weight[rv] = d
            return 1

        col_parity = [0] * W
        py_col = [0] * W  # XOR of rows 1..i-1 for each column

        ok = True

        for i, row in enumerate(S):
            px = 0  # XOR of a[i][1..j]

            for j, ch in enumerate(row):
                a = 1 if ch == A else 0

                col_parity[j] ^= a

                if j > 0:
                    px ^= a

                if i == 0:
                    py = 0
                else:
                    py = py_col[j] ^ a

                if a == 0:  # Type B
                    b = 1 ^ px ^ py
                    res = unite(i, H + j, b)

                    if res == -1:
                        ok = False
                        break
                    elif res == 1:
                        comp -= 1

                if i > 0:
                    py_col[j] ^= a

            if not ok:
                break

        if not ok:
            ans.append("0")
            continue

        if any(col_parity):
            ans.append("0")
            continue

        ans.append(str(pow(2, comp, MOD)))

    sys.stdout.write("\n".join(ans))


if __name__ == "__main__":
    solve()
```
