from typing import List

class Solution:
    def maxScore(self, grid: List[List[int]]) -> int:
        m = len(grid)
        rows_with_value = [0] * 101
        for r in range(m):
            for v in grid[r]:
                rows_with_value[v] |= 1 << r
        
        dp = [-1] * (1 << m)
        dp[0] = 0
        
        for v in range(1, 101):
            rows = rows_with_value[v]
            if rows == 0:
                continue
            ndp = dp[:]
            for mask in range(1 << m):
                cur = dp[mask]
                if cur == -1:

                    nmask = mask | bit
                    if cur + v > ndp[nmask]:

            dp = ndp
        
        return max(dp)

```python
from typing import List

class Solution:
    def maxScore(self, grid: List[List[int]]) -> int:
        m = len(grid)
        rows_with_value = [0] * 101

        for r in range(m):
            for v in grid[r]:
                rows_with_value[v] |= 1 << r

        full_mask = (1 << m) - 1
        dp = [-1] * (1 << m)
        dp[0] = 0

        for v in range(1, 101):
            rows = rows_with_value[v]
            if rows == 0:
                continue

            ndp = dp[:]

            for mask in range(1 << m):
                cur = dp[mask]
                if cur == -1:
                    continue

                available = rows & (full_mask ^ mask)
                while available:
                    bit = available & -available
                    nmask = mask | bit
                    score = cur + v
                    if score > ndp[nmask]:
                        ndp[nmask] = score
                    available -= bit

            dp = ndp

        return max(dp)
```
