```python
from typing import List
from math import gcd

class Solution:
    def maxScore(self, nums: List[int]) -> int:
        def score(arr: List[int]) -> int:
            if not arr:
                return 0

            g = arr[0]
            l = arr[0]

            for x in arr[1:]:
                g = gcd(g, x)
                l = l // gcd(l, x) * x

            return g * l

        ans = score(nums)

        for i in range(len(nums)):
            removed = nums[:i] + nums[i + 1:]
            if removed:
                ans = max(ans, score(removed))

        return ans
```
