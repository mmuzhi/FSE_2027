```python
from typing import List

class Solution:
    def countPathsWithXorValue(self, grid: List[List[int]], k: int) -> int:
        MOD = 10**9 + 7
        m, n = len(grid), len(grid[0])

        # dp[j][x] = number of ways to reach current row, column j with XOR x
        dp = [[0] * 16 for _ in range(n)]
        dp[0][grid[0][0]] = 1

        # First row: only left moves
        for j in range(1, n):
            val = grid[0][j]
            ndp = [0] * 16
            for x in range(16):
                ndp[x] = dp[j - 1][x ^ val]
            dp[j] = ndp

        # Remaining rows
        for i in range(1, m):
            # First column: only top moves
            val = grid[i][0]
            ndp = [0] * 16
            for x in range(16):
                ndp[x] = dp[0][x ^ val]
            dp[0] = ndp

            for j in range(1, n):
                val = grid[i][j]
                ndp = [0] * 16
                for x in range(16):
                    s = dp[j][x ^ val] + dp[j - 1][x ^ val]
                    if s >= MOD:
                        s -= MOD
                    ndp[x] = s
                dp[j] = ndp

        return dp[n - 1][k]
```
