```python
from typing import List

class Solution:
    def maxWeight(self, pizzas: List[int]) -> int:
        pizzas.sort(reverse=True)
        n = len(pizzas)
        days = n // 4
        
        odd_days = (days + 1) // 2
        even_days = days // 2
        
        ans = sum(pizzas[:odd_days])
        
        idx = odd_days + 1
        for _ in range(even_days):
            ans += pizzas[idx]
            idx += 2
            
        return ans
```
