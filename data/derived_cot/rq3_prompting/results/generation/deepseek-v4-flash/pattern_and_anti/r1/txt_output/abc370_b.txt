```python
import sys

def main():
    data = list(map(int, sys.stdin.read().split()))
    N = data[0]
    idx = 1

    A = [None] * (N + 1)
    for i in range(1, N + 1):
        A[i] = [0] + data[idx:idx + i]
        idx += i

    x = 1
    for k in range(1, N + 1):
        if x >= k:
            x = A[x][k]
        else:
            x = A[k][x]

    print(x)

if __name__ == "__main__":
    main()
```
