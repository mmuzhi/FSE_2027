```python
from typing import List

class Solution:
    def solveQueries(self, nums: List[int], queries: List[int]) -> List[int]:
        n = len(nums)
        pos = {}
        for i, x in enumerate(nums):
            pos.setdefault(x, []).append(i)
        
        ans = [-1] * n
        for p_list in pos.values():
            m = len(p_list)
            if m == 1:
                continue
            for idx, p in enumerate(p_list):
                prev = p_list[idx - 1]
                nxt = p_list[(idx + 1) % m]
                d1 = abs(p - prev)
                d1 = min(d1, n - d1)
                d2 = abs(p - nxt)
                d2 = min(d2, n - d2)
                ans[p] = min(d1, d2)
        
        return [ans[q] for q in queries]
```

    def solveQueries(self, nums: List[int], queries: List[int]) -> List[int]:
        n = len(nums)
        positions = {}
