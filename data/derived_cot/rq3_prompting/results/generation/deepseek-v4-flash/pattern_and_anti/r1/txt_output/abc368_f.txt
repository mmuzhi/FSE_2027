import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    N = data[0]
    A = data[1:1 + N]
    M = max(A)

    spf = [0] * (M + 1)
    for i in range(2, M + 1):
        if spf[i] == 0:
            spf[i] = i
            if i * i <= M:
                for j in range(i * i, M + 1, i):
                    if spf[j] == 0:
                        spf[j] = i

    omega = [0] * (M + 1)
    for i in range(2, M + 1):
        omega[i] = omega[i // spf[i]] + 1

    xor_sum = 0
    for a in A:
        xor_sum ^= omega[a]

    print("Anna" if xor_sum else "Bruno")

if __name__ == "__main__":
