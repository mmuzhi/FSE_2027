```
def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:

    N = data[0]
    del data[0]

    # For small N, direct O(N^2) is faster than the big-int convolution.
    if N <= 3000:
        ss = set(S)
        S.sort()
        ans = 0
        for i in range(1, N):
            b = S[i]
            two_b = b + b
            for j in range(i):
                if two_b - S[j] in ss:
                    ans += 1
        print(ans)
        return

    M = max(S)
    # Pack S as bits in base 2^20. Since N <= 10^6 < 2^20,
    # convolution coefficients fit in one 20-bit digit without carries.
    buf = bytearray((20 * M) // 8 + 1)
    for x in S:
        # bit position = 20*x = 8*(2*x + x//2) + 4*(x&1)
        buf[(x << 1) + (x >> 1)] = 1 << ((x & 1) << 2)

        coef = unpack4(prod_bytes, 5 * b)[0] & mask
        ans += (coef - 1) // 2

    print(ans)

if __name__ == "__main__":
    solve()
```

```python
import sys
from struct import Struct

try:
    import numpy as np
except ImportError:
    np = None


def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    N = data[0]
    del data[0]
    S = data

    # Small N: direct O(N^2) counting is faster.
    if N <= 3000:
        ss = set(S)
        S.sort()
        ans = 0
        for i in range(1, N):
            b = S[i]
            tb = b << 1
            for j in range(i):
                if tb - S[j] in ss:
                    ans += 1
        print(ans)
        return

    M = max(S)

    # Fast path using numpy FFT if available.
    if np is not None:
        f = np.zeros(M + 1, dtype=np.float64)
        f[S] = 1.0

        n = 1 << (2 * M).bit_length()
        F = np.fft.rfft(f, n)
        conv = np.fft.irfft(F * F, n)

        vals = np.array(S, dtype=np.int64)
        coef = np.rint(conv[2 * vals]).astype(np.int64)
        ans = int(np.sum((coef - 1) // 2))
        print(ans)
        return

    # Exact fallback using Python big integers.
    # Pack S as bits in base 2^20.
    # Since N <= 10^6 < 2^20, convolution coefficients fit without carries.
    buf = bytearray((20 * M) // 8 + 1)
    for x in S:
        # bit position = 20*x = 8*(2*x + x//2) + 4*(x&1)
        buf[(x << 1) + (x >> 1)] |= 1 << ((x & 1) << 2)

    P = int.from_bytes(buf, "little")
    prod = P * P

    # For middle b, coefficient of x^(2b) starts at bit 40*b = byte 5*b.
    prod_bytes = prod.to_bytes(5 * M + 4, "little")

    unpack4 = Struct("<I").unpack_from
    mask = (1 << 20) - 1

    ans = 0
    for b in S:
        coef = unpack4(prod_bytes, 5 * b)[0] & mask
        ans += (coef - 1) // 2

    print(ans)


if __name__ == "__main__":
    solve()
```
