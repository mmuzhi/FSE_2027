=== ANSWER ===
```python
import sys

def main():
    A, B, C = map(int, sys.stdin.read().split())
    S = A + B + C

    ok = False

    if S % 2 == 0 and max(A, B, C) == S // 2:
        ok = True

    if S % 3 == 0 and A == B == C:
        ok = True

    print("Yes" if ok else "No")

if __name__ == "__main__":
    main()
```
