```python
import sys
import heapq


def main():
    input = sys.stdin.buffer.readline

    N = int(input())
    adj = [[] for _ in range(N + 1)]
    mate = [0] * (N + 1)

    # First N/2 edges are the given perfect matching.
    for _ in range(N // 2):
        a, b = map(int, input().split())
        mate[a] = b
        mate[b] = a
        adj[a].append(b)
        adj[b].append(a)

    # Remaining edges.
    for _ in range(N // 2 - 1):
        a, b = map(int, input().split())
        adj[a].append(b)
        adj[b].append(a)

    # Find a centroid.
    parent = [0] * (N + 1)
    parent[1] = -1
    order = [1]
    idx = 0
    while idx < len(order):
        v = order[idx]
        idx += 1
        for u in adj[v]:
            if u == parent[v]:
                continue
            parent[u] = v
            order.append(u)

    sub = [1] * (N + 1)
    for v in reversed(order):
        s = 1
        for u in adj[v]:
            if parent[u] == v:
                s += sub[u]
        sub[v] = s

    c = 1
    best = N + 1
    for v in range(1, N + 1):
        mx = N - sub[v]
        for u in adj[v]:
            if parent[u] == v:
                if sub[u] > mx:
                    mx = sub[u]
        if mx < best:
            best = mx
            c = v

    # Root the tree at the centroid c.
    branches = adj[c]
    B = len(branches)

    branch_id = [-1] * (N + 1)
    for i, r in enumerate(branches):
        branch_id[r] = i

    parent_c = [0] * (N + 1)
    parent_c[c] = -1
    children = [[] for _ in range(N + 1)]

    stack = [c]
    while stack:
        v = stack.pop()
        for u in adj[v]:
            if u == parent_c[v]:
                continue
            parent_c[u] = v
            children[v].append(u)
            if v != c:
                branch_id[u] = branch_id[v]
            stack.append(u)

    # Reorder children:
    # If v is not matched to its parent, its matched child must be visited first.
    for v in range(1, N + 1):
        if v == c:
            continue
        if mate[v] != parent_c[v]:
            m = mate[v]
            ch = children[v]
            children[v] = [m] + [u for u in ch if u != m]

    # Postorder of each branch gives the order in which its vertices become removable leaves.
    queues = [[] for _ in range(B)]
    stack = [(c, 0)]
    while stack:
        v, idx = stack[-1]
        if idx < len(children[v]):
            u = children[v][idx]
            stack[-1] = (v, idx + 1)
            stack.append((u, 0))
        else:
            stack.pop()
            if v != c:
                queues[branch_id[v]].append(v)

    size = [len(q) for q in queues]
    ptr = [0] * B

    # D is the branch containing the current matching partner of c.
    d = branch_id[mate[c]]

    heap = [(-size[i], i) for i in range(B) if size[i] > 0]
    heapq.heapify(heap)

    def clean():
        while heap:
            neg, i = heap[0]
            if size[i] == 0 or size[i] != -neg:
                heapq.heappop(heap)
            else:
                break

    def get_other(d):
        clean()
        if not heap:
            return -1

        top1 = heapq.heappop(heap)
        if top1[1] != d:
            heapq.heappush(heap, top1)
            return top1[1]

        # top1 is d; need another non-empty branch.
        clean()
        if not heap:
            heapq.heappush(heap, top1)
            return -1

        top2 = heapq.heappop(heap)
        heapq.heappush(heap, top1)
        return top2[1]

    ans = []

    while True:
        b = get_other(d)
        if b == -1:
            break

        x = queues[d][ptr[d]]
        ptr[d] += 1
        size[d] -= 1

        y = queues[b][ptr[b]]
        ptr[b] += 1
        size[b] -= 1

        ans.append(f"{x} {y}")

        heapq.heappush(heap, (-size[d], d))
        heapq.heappush(heap, (-size[b], b))

        d = b

    # Final pair: centroid and the last remaining vertex.
    for i in range(B):
        if size[i] > 0:
            v = queues[i][ptr[i]]
            ans.append(f"{c} {v}")
            break

    sys.stdout.write("\n".join(ans))


if __name__ == "__main__":
    main()
```
