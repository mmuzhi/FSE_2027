A string T is called an 11/22 string when it satisfies all of the following conditions:

import sys

def main():
    data = sys.stdin.read().split()
    N = int(data[0])
    S = data[1]

    if N % 2 == 0:
        print("No")
        return

    mid = N // 2

    if S[mid] != '/':
        print("No")
        return

    for ch in S[:mid]:
        if ch != '1':
            print("No")
            return

    for ch in S[mid + 1:]:
        if ch != '2':
            print("No")
            return

    print("Yes")

if __name__ == "__main__":
    main()
