```python
import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    N, D = data[0], data[1]
    snakes = []
    idx = 2

    for _ in range(N):
        t = data[idx]
        l = data[idx + 1]
        idx += 2
        snakes.append((t, l))

    ans = []

    for k in range(1, D + 1):
        max_weight = 0
        for t, l in snakes:
            weight = t * (l + k)
            if weight > max_weight:
                max_weight = weight
        ans.append(str(max_weight))

    print("\n".join(ans))

if __name__ == "__main__":
    main()
```
