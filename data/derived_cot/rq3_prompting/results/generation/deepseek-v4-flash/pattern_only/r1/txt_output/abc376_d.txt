import sys
from collections import deque

def main():
    input=sys.stdin.readline
    N,M=map(int,input().split())
    rev=[[] for _ in range(N)]
    out1=[]
    for _ in range(M):
        a,b=map(int,input().split())
        a-=1; b-=1
        rev[b].append(a)
        if a==0:
            out1.append(b)
    INF=10**18
    dist=[INF]*N
    dist[0]=0
    q=deque([0])
    while q:
        v=q.popleft()
        for to in rev[v]:
            if dist[to]==INF:
                dist[to]=dist[v]+1
                q.append(to)
    ans=INF
    for v in out1:
        if dist[v] < INF:
            ans=min(ans, dist[v]+1)
    print(ans if ans<INF else -1)

if __name__=="__main__": main()

import sys
from collections import deque

def main():
    input = sys.stdin.buffer.readline
    N, M = map(int, input().split())
    rev = [[] for _ in range(N)]
    from_one = []

    for _ in range(M):
        a, b = map(int, input().split())
        a -= 1
        b -= 1
        rev[b].append(a)
        if a == 0:
            from_one.append(b)

    INF = 10**18
    dist = [INF] * N
    dist[0] = 0
    q = deque([0])

    while q:
        v = q.popleft()
        nd = dist[v] + 1
        for u in rev[v]:
            if dist[u] == INF:
                dist[u] = nd
                q.append(u)

    ans = INF
    for v in from_one:
        if dist[v] != INF:
            ans = min(ans, dist[v] + 1)

    print(-1 if ans == INF else ans)

if __name__ == "__main__":
