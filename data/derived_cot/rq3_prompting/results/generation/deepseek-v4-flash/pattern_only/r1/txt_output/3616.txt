from typing import List

class Solution:
    def countValidSelections(self, nums: List[int]) -> int:

        ans = 0
        pref = 0

            if x == 0:

                if left == right:
                    ans += 2
                elif abs(left - right) == 1:
                    ans += 1

            pref += x

        return ans
