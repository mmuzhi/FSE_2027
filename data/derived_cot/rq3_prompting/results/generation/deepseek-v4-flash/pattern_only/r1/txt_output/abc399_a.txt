=== ANSWER ===
```python
import sys

def main():
    data = sys.stdin.read().split()
    N = int(data[0])
    S = data[1]
    T = data[2]

    ans = sum(1 for a, b in zip(S, T) if a != b)
    print(ans)

if __name__ == "__main__":
    main()
```
