```python
from typing import List

class Solution:
    def largestInteger(self, nums: List[int], k: int) -> int:
        n = len(nums)
        window_count = {}

        for start in range(n - k + 1):
            seen = set()
            for i in range(start, start + k):
                seen.add(nums[i])

            for x in seen:
                window_count[x] = window_count.get(x, 0) + 1

        ans = -1
        for x, cnt in window_count.items():
            if cnt == 1 and x > ans:
                ans = x

        return ans
```
