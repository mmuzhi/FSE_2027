import sys

def main():
    input = sys.stdin.buffer.readline
    N, K = map(int, input().split())
    f = [int(x) - 1 for x in input().split()]
    A = list(map(int, input().split()))

    if K == 0:
        print(' '.join(map(str, A)))
        return

    indeg = [0] * N
    for p in f:
        indeg[p] += 1

    q = [i for i in range(N) if indeg[i] == 0]
    in_cycle = [True] * N
    qi = 0
    while qi < len(q):
        v = q[qi]
        qi += 1
        in_cycle[v] = False
        p = f[v]
        indeg[p] -= 1
        if indeg[p] == 0:
            q.append(p)

    indeg = None
    q = None

    cycle_id = [-1] * N
    cycle_pos = [-1] * N
    cycles = []
    ans = [0] * N

    for i in range(N):
        if in_cycle[i] and cycle_id[i] == -1:
            cid = len(cycles)
            cyc = []
            v = i
            while cycle_id[v] == -1:
                cycle_id[v] = cid
                cycle_pos[v] = len(cyc)
                cyc.append(v)
                v = f[v]
            L = len(cyc)
            cycles.append(cyc)
            kmod = K % L
            for idx, node in enumerate(cyc):
                ans[node] = cyc[(idx + kmod) % L]

    rev_head = [-1] * N
    rev_next = [-1] * N
    for v in range(N):
        if in_cycle[v]:
            continue
        p = f[v]
        rev_next[v] = rev_head[p]
        rev_head[p] = v

    f = None

    for root in range(N):
        if not in_cycle[root]:
            continue
        path = [root]
        stack_v = [root]
        stack_e = [rev_head[root]]
        cid_root = cycle_id[root]
        root_pos = cycle_pos[root]
        L = len(cycles[cid_root])
        while stack_v:
            v = stack_v[-1]
            e = stack_e[-1]
            if e == -1:
                stack_v.pop()
                stack_e.pop()
                path.pop()
                continue
            child = e
            stack_e[-1] = rev_next[e]
            path.append(child)
            d = len(path) - 1
            if K <= d:
                ans[child] = path[d - K]
            else:
                rem = (K - d) % L
                ans[child] = cycles[cid_root][(root_pos + rem) % L]
            stack_v.append(child)
            stack_e.append(rev_head[child])

    sys.stdout.write(' '.join(str(A[ans[i]]) for i in range(N)) + '\n')

        p = f[v]
        indeg[p] -= 1
        if indeg[p] == 0:
            q.append(p)

    indeg = None
    q = None

    cycle_id = [-1] * N
    cycle_pos = [-1] * N
    cycles = []
    ans = [0] * N

    for i in range(N):
        if in_cycle[i] and cycle_id[i] == -1:
            cid = len(cycles)
            cyc = []
            v = i
            while cycle_id[v] == -1:
                cycle_id[v] = cid
                cycle_pos[v] = len(cyc)
                cyc.append(v)
                v = f[v]

            L = len(cyc)
            cycles.append(cyc)

            kmod = K % L
            for idx, node in enumerate(cyc):
                ans[node] = cyc[(idx + kmod) % L]

    rev_head = [-1] * N
    rev_next = [-1] * N

    for v in range(N):
        if in_cycle[v]:
            continue
        p = f[v]
        rev_next[v] = rev_head[p]
        rev_head[p] = v

    f = None

    for root in range(N):
        if not in_cycle[root]:
            continue

        path = [root]
        stack_v = [root]
        stack_e = [rev_head[root]]

        cid_root = cycle_id[root]
        root_pos = cycle_pos[root]
        L = len(cycles[cid_root])

        while stack_v:
            v = stack_v[-1]
            e = stack_e[-1]

            if e == -1:
                stack_v.pop()
                stack_e.pop()
                path.pop()
                continue

            child = e
            stack_e[-1] = rev_next[e]

            path.append(child)
            d = len(path) - 1

            if K <= d:
                ans[child] = path[d - K]
            else:
                rem = (K - d) % L
                ans[child] = cycles[cid_root][(root_pos + rem) % L]

            stack_v.append(child)
            stack_e.append(rev_head[child])

    sys.stdout.write(" ".join(str(A[ans[i]]) for i in range(N)) + "\n")


if __name__ == "__main__":
    main()
