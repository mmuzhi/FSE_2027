import sys

MOD = 998244353


def is_good(vals):
    n = len(vals)
    if n == 0:
        return True
    if n == 1:
        return False

    suf = [0] * (n + 1)
    suf[n] = -1
    for i in range(n - 1, -1, -1):
        suf[i] = max(suf[i + 1], vals[i])

    pref_min = 10**9
    for i, v in enumerate(vals):
        if v < pref_min and v > suf[i + 1]:
            return False
        pref_min = min(pref_min, v)

    return True


def main():
    input = sys.stdin.readline
    N = int(input())

    p = [0] * N
    for _ in range(N):
        x, y = map(int, input().split())
        p[x - 1] = y

    start_good = [False] * N
    for j in range(N):
        vals = [p[k] for k in range(j) if p[k] > p[j]]
        start_good[j] = is_good(vals)

    end_good = [False] * N
    for i in range(N):
        vals = [p[k] for k in range(i + 1, N) if p[k] < p[i]]
        end_good[i] = is_good(vals)

    all_good = is_good(p[:])

    rect_good = [[False] * N for _ in range(N)]
    for i in range(N):
        pi = p[i]
        for j in range(i + 1, N):
            pj = p[j]
            if pi > pj:
                vals = []
                for k in range(i + 1, j):
                    v = p[k]
                    if pj < v < pi:
                        vals.append(v)
                rect_good[i][j] = is_good(vals)

    dp = [0] * N

    for j in range(N):
        if start_good[j]:
            dp[j] = 1

        pj = p[j]
        for i in range(j):
            if p[i] > pj and rect_good[i][j]:
                dp[j] += dp[i]
                if dp[j] >= MOD:
                    dp[j] -= MOD

    ans = 1 if all_good else 0

    for j in range(N):
        if end_good[j]:
            ans += dp[j]
            if ans >= MOD:
                ans -= MOD

    print(ans)


if __name__ == "__main__":
    main()
