```python
from typing import List

class Solution:
    def beautifulSplits(self, nums: List[int]) -> int:
        n = len(nums)

        z = [0] * n
        left = right = 0
        for i in range(1, n):
            if i <= right:
                z[i] = min(right - i + 1, z[i - left])
            while i + z[i] < n and nums[z[i]] == nums[i + z[i]]:
                z[i] += 1
            if i + z[i] - 1 > right:
                left, right = i, i + z[i] - 1
        z[0] = n

        ans = 0

        # Condition 1: nums1 is a prefix of nums2.
        for i in range(1, (n - 1) // 2 + 1):
            if z[i] >= i:
                ans += n - 2 * i

        # Condition 2: nums2 is a prefix of nums3.
        # val[i] = LCP(nums[i:], nums[i+L:]) for current L.
        val = [0] * (n + 1)
        max_l = (n - 1) // 2

        for L in range(1, max_l + 1):
            m = n - L
            val[m] = 0
            limit = n - 2 * L

            for i in range(m - 1, 0, -1):
                if nums[i] == nums[i + L]:
                    val[i] = val[i + 1] + 1
                else:
                    val[i] = 0

                if i <= limit and val[i] >= L:
                    ans += 1  # this split satisfies condition 2
                    if i <= L and z[i] >= i:
                        ans -= 1  # it was already counted by condition 1

        return ans
```

```python
from typing import List

class Solution:
    def beautifulSplits(self, nums: List[int]) -> int:
        n = len(nums)
        if n < 3:
            return 0

        # Z-function: z[i] = LCP(nums[0:], nums[i:])
        z = [0] * n
        l = r = 0
        for i in range(1, n):
            if i <= r:
                z[i] = min(r - i + 1, z[i - l])
            while i + z[i] < n and nums[z[i]] == nums[i + z[i]]:
                z[i] += 1
            if i + z[i] - 1 > r:
                l, r = i, i + z[i] - 1
        z[0] = n

        ans = 0

        # Condition 1: nums1 is a prefix of nums2.
        # nums1 = nums[:i], nums2 starts at i.
        # Need z[i] >= i and len(nums2) >= i, i.e. j >= 2*i.
        for i in range(1, (n - 1) // 2 + 1):
            if z[i] >= i:
                ans += n - 2 * i

        # Condition 2: nums2 is a prefix of nums3.
        # For fixed L = len(nums2), count i such that
        # LCP(nums[i:], nums[i+L:]) >= L and 2*L <= n-i.
        val = [0] * (n + 1)
        max_l = (n - 1) // 2

        for L in range(1, max_l + 1):
            m = n - L
            val[m] = 0
            limit = n - 2 * L

            for i in range(m - 1, 0, -1):
                if nums[i] == nums[i + L]:
                    val[i] = val[i + 1] + 1
                else:
                    val[i] = 0

                if i <= limit and val[i] >= L:
                    ans += 1  # condition 2 holds

                    # Subtract if this split was already counted by condition 1.
                    # Condition 1 also needs len(nums2) >= len(nums1), i.e. L >= i.
                    if i <= L and z[i] >= i:
                        ans -= 1

        return ans
```
