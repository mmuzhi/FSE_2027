```python
from typing import List

class Solution:
    def resultsArray(self, nums: List[int], k: int) -> List[int]:
        n = len(nums)
        good = 0

        for i in range(k - 1):
            if nums[i + 1] == nums[i] + 1:
                good += 1

        ans = []

        for i in range(n - k + 1):
            if i > 0 and k > 1:
                if nums[i] == nums[i - 1] + 1:
                    good -= 1
                if nums[i + k - 1] == nums[i + k - 2] + 1:
                    good += 1

            ans.append(nums[i + k - 1] if good == k - 1 else -1)

        return ans
