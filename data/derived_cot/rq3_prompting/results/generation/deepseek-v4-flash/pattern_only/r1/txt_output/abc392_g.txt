    be = bytearray(len_even)
    bo = bytearray(len_odd)

        if v & 1:

            bo[p >> 3] |= 1 << (p & 7)
        else:
            p = K * (v // 2)
            be[p >> 3] |= 1 << (p & 7)

import sys


def solve_bigint(N, vals):
    M = max(vals)
    K = max(1, N.bit_length())
    mask = (1 << K) - 1

    len_even = (K * (M // 2)) // 8 + 1
    len_odd = (K * ((M - 1) // 2)) // 8 + 1

    be = bytearray(len_even)
    bo = bytearray(len_odd)

    for v in vals:
        p = K * (v >> 1)
        if v & 1:
            bo[p >> 3] |= 1 << (p & 7)
        else:
            be[p >> 3] |= 1 << (p & 7)

    PE = int.from_bytes(be, "little")
    PO = int.from_bytes(bo, "little")

    PE_sq = PE * PE
    PO_sq = PO * PO

    # R[d] = number of ordered pairs (x, y) in S with x + y = 2d
    R = PE_sq + (PO_sq << K)

    byte_len = max((K * M) // 8 + 3, (K * (M + 1) + 7) // 8)
    rb = R.to_bytes(byte_len, "little")

    ans = 0
    for v in vals:
        p = K * v
        off = p >> 3
        rem = p & 7
        val = (rb[off] | (rb[off + 1] << 8) | (rb[off + 2] << 16)) >> rem
        c = val & mask
        ans += (c - 1) // 2

    return ans


def main():
    try:
        import numpy as np
    except ImportError:
        data = list(map(int, sys.stdin.buffer.read().split()))
        if not data:
            return
        N = data[0]
        vals = data[1:1 + N]
        print(solve_bigint(N, vals))
        return

    data = np.array(sys.stdin.buffer.read().split(), dtype=np.int64)
    if data.size == 0:
        return

    N = int(data[0])
    vals = data[1:1 + N]
    M = int(vals.max())

    n = 1 << (2 * M).bit_length()
    f = np.zeros(n, dtype=np.float64)
    f[vals] = 1.0

    F = np.fft.rfft(f)
    F *= F
    conv = np.rint(np.fft.irfft(F, n)).astype(np.int64)

    ans = int(np.sum((conv[2 * vals] - 1) // 2))
    print(ans)


if __name__ == "__main__":
    main()
