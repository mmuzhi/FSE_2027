class Solution:
    def maxPossibleScore(self, start: List[int], d: int) -> int:
        start.sort()
        n = len(start)

        def ok(score: int) -> bool:
            prev = start[0]
            for i in range(1, n):
                cur = max(start[i], prev + score)
                if cur > start[i] + d:
                    return False
                prev = cur
            return True

        lo, hi = 0, start[-1] + d - start[0]
        while lo < hi:
            mid = (lo + hi + 1) // 2
            if ok(mid):
                lo = mid
            else:
                hi = mid - 1
        return lo
