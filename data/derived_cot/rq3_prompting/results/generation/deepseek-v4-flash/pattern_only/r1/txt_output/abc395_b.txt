```python
import sys

def main():
    N = int(sys.stdin.read().strip())
    ans = []

    for i in range(1, N + 1):
        row = []
        for j in range(1, N + 1):
            layer = min(i, N + 1 - i, j, N + 1 - j)
            row.append('#' if layer % 2 == 1 else '.')
        ans.append(''.join(row))

    print('\n'.join(ans))

if __name__ == "__main__":
    main()
```
