```python
import sys

def main():
    mod = 998244353
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    N, K = data[0], data[1]
    A = data[2:2 + N]

    # coeff[t] = C(K, t) * (-1)^t mod mod
    coeff = []
    c = 1
    for t in range(K + 1):
        if t & 1:
            coeff.append((mod - c) % mod)
        else:
            coeff.append(c % mod)
        c = c * (K - t) // (t + 1)

    pref = [0] * (K + 1)
    pref[0] = 1  # P_0^0

    pows = [1] * (K + 1)
    ans = 0
    prefix = 0

    for a in A:
        prefix = (prefix + a) % mod

        pows[0] = 1
        for e in range(1, K + 1):
            pows[e] = pows[e - 1] * prefix % mod

        term = 0
        for t in range(K + 1):
            term += (coeff[t] * pows[K - t] % mod) * pref[t]

        ans = (ans + term) % mod

        pref[0] += 1
        if pref[0] >= mod:
            pref[0] -= mod

        for e in range(1, K + 1):
            pref[e] += pows[e]
            if pref[e] >= mod:
                pref[e] -= mod

    print(ans)

if __name__ == "__main__":
    main()
```
