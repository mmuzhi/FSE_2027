You will be given a competitive programming problem.
There are N buildings, building 1, building 2, \ldots, building N, arranged in this order in a straight line from west to east. Building 1 is the westernmost, and building N is the easternmost. The height of building i\ (1\leq i\leq N) is H_i.
For a pair of integers (i,j)\ (1\leq i\lt j\leq N), building j can be seen from building i if the following condition is satisfied.

- There is no building taller than building j between buildings i and j. In other words, there is no integer k\ (i\lt k\lt j) such that H_k > H_j.

You are given Q queries. In the i-th query, given a pair of integers (l_i,r_i)\ (l_i\lt r_i), find the number of buildings to the east of building r_i (that is, buildings r_i + 1, r_i + 2, \ldots, N) that can be seen from both buildings l_i and r_i.

Input

The input is given from Standard Input in the following format:
N Q
H_1 H_2 \ldots H_N
l_1 r_1
l_2 r_2
\vdots
l_Q r_Q

Output

Print Q lines. The i-th line (1 \leq i \leq Q) should contain the answer to the i-th query.

Constraints


- 2 \leq N \leq 2 \times 10^5
- 1 \leq Q \leq 2 \times 10^5
- 1 \leq H_i \leq N
- H_i\neq H_j\ (i\neq j)
- 1 \leq l_i < r_i \leq N
- All input values are integers.

import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    it = iter(data)
    N = next(it); Q = next(it)
    H = [0]*(N+1)
    for i in range(1,N+1):
        H[i] = next(it)
    L = [0]*(N+1)
    stack = []
    for i in range(1,N+1):
        hi = H[i]
        while stack and H[stack[-1]] < hi:
            stack.pop()
        if stack:
            L[i] = stack[-1]
        else:
            L[i] = 0
        stack.append(i)
    buckets = [[] for _ in range(N+1)]  # L values 0..N
    for i in range(1,N+1):
        buckets[L[i]].append(i)
    queries = []
    for idx in range(Q):
        l = next(it); r = next(it)
        queries.append((l, r, idx))
    queries.sort(key=lambda x: x[0])
    bit = [0]*(N+1)
    def add(i):
        while i <= N:
            bit[i] += 1
            i += i & -i
    def sum_(i):
        s=0
        while i>0:
            s += bit[i]
            i -= i & -i
        return s
    ans = [0]*Q

            for j in buckets[cur]:
                add(j)
                added += 1
            cur += 1
        ans[idx] = added - sum_(r)
    sys.stdout.write("\n".join(map(str, ans)))

import sys

def main():
    input = sys.stdin.buffer.readline
    N, Q = map(int, input().split())
    H = [0] + list(map(int, input().split()))

    L = [0] * (N + 1)
    st = []
    for i in range(1, N + 1):
        hi = H[i]
        while st and H[st[-1]] <= hi:
            st.pop()
        if st:
            L[i] = st[-1]
        st.append(i)

    buckets = [[] for _ in range(N + 1)]
    for i in range(1, N + 1):
        buckets[L[i]].append(i)

    queries = []
    for idx in range(Q):
        l, r = map(int, input().split())
        queries.append((l, r, idx))

    queries.sort(key=lambda x: x[0])

    bit = [0] * (N + 1)

    def add(i):
        while i <= N:
            bit[i] += 1
            i += i & -i

    def sum_prefix(i):
        s = 0
        while i > 0:
            s += bit[i]
            i -= i & -i
        return s

    ans = [0] * Q
    cur = 0
    added = 0
    for l, r, idx in queries:
        while cur <= l:
            for j in buckets[cur]:
                add(j)
                added += 1
            cur += 1
        ans[idx] = added - sum_prefix(r)

    sys.stdout.write("\n".join(map(str, ans)))

if __name__ == "__main__":
    main()

import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    p = 0
    N = data[p]; Q = data[p+1]; p += 2

    H = [0] * (N + 1)
    for i in range(1, N + 1):
        H[i] = data[p]
        p += 1

    queries = []
    for idx in range(Q):
        l = data[p]
        r = data[p + 1]
        p += 2
        queries.append((l, r, idx))

    del data

    L = [0] * (N + 1)
    st = []
    for i in range(1, N + 1):
        hi = H[i]
        while st and H[st[-1]] <= hi:
            st.pop()
        if st:
            L[i] = st[-1]
        st.append(i)

    buckets = [[] for _ in range(N + 1)]
    for i in range(1, N + 1):
        buckets[L[i]].append(i)

    queries.sort(key=lambda x: x[0])

    bit = [0] * (N + 1)

    def add(i):
        while i <= N:
            bit[i] += 1
            i += i & -i

    def pref(i):
        s = 0
        while i > 0:
            s += bit[i]
            i -= i & -i
        return s

    ans = [0] * Q
    cur = 0
    added = 0
    for l, r, idx in queries:
        while cur <= l:
            for j in buckets[cur]:
                add(j)
                added += 1
            cur += 1
        ans[idx] = added - pref(r)

    sys.stdout.write("\n".join(map(str, ans)))

if __name__ == "__main__":
    main()

import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    p = 0

    N = data[p]
    Q = data[p + 1]
    p += 2

    H = [0] * (N + 1)
    for i in range(1, N + 1):
        H[i] = data[p]
        p += 1

    queries = []
    for idx in range(Q):
        l = data[p]
        r = data[p + 1]
        p += 2
        queries.append((l, r, idx))

    del data

    # L[i] = nearest index to the left of i with height > H[i], or 0
    L = [0] * (N + 1)
    st = []
    for i in range(1, N + 1):
        hi = H[i]
        while st and H[st[-1]] <= hi:
            st.pop()
        if st:
            L[i] = st[-1]
        st.append(i)

    buckets = [[] for _ in range(N + 1)]
    for i in range(1, N + 1):
        buckets[L[i]].append(i)

    queries.sort(key=lambda x: x[0])

    bit = [0] * (N + 1)

    def add(i):
        while i <= N:
            bit[i] += 1
            i += i & -i

    def pref(i):
        s = 0
        while i > 0:
            s += bit[i]
            i -= i & -i
        return s

    ans = [0] * Q
    cur = 0
    added = 0

    for l, r, idx in queries:
        while cur <= l:
            for j in buckets[cur]:
                add(j)
                added += 1
            cur += 1

        ans[idx] = added - pref(r)

    sys.stdout.write("\n".join(map(str, ans)))


if __name__ == "__main__":
    main()
