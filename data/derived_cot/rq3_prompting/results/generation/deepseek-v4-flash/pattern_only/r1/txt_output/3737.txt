```python
from typing import List

class Solution:
    def minCost(self, n: int, cost: List[List[int]]) -> int:
        m = n // 2
        states = [(0,1),(0,2),(1,0),(1,2),(2,0),(2,1)]
        INF = 10**30

        dp = []
        for x, y in states:
            dp.append(cost[0][x] + cost[n-1][y])

        for k in range(1, m):
            ndp = [INF] * 6
            for s, (x, y) in enumerate(states):
                base = cost[k][x] + cost[n-1-k][y]
                best = INF
                for p, (a, b) in enumerate(states):
                    if a != x and b != y:
                        if dp[p] < best:
                            best = dp[p]
                ndp[s] = best + base
            dp = ndp

        return min(dp)
```

```python
from typing import List

class Solution:
    def minCost(self, n: int, cost: List[List[int]]) -> int:
        m = n // 2
        states = [(0, 1), (0, 2), (1, 0), (1, 2), (2, 0), (2, 1)]
        INF = 10**30

        dp = [cost[0][x] + cost[n - 1][y] for x, y in states]

        for k in range(1, m):
            ndp = [INF] * 6
            left_cost = cost[k]
            right_cost = cost[n - 1 - k]

            for i, (x, y) in enumerate(states):
                base = left_cost[x] + right_cost[y]
                best_prev = INF

                for j, (a, b) in enumerate(states):
                    if a != x and b != y:
                        best_prev = min(best_prev, dp[j])

                ndp[i] = best_prev + base

            dp = ndp

        return min(dp)
```
