import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    N = data[0]
    A = data[1:1 + N]
    M = max(A)

    # omega[n] = number of prime factors of n counted with multiplicity
    omega = [0] * (M + 1)

    for i in range(2, M + 1):
        if omega[i] == 0:  # i is prime
            p = i
            pk = p
            while pk <= M:
                for j in range(pk, M + 1, pk):
                    omega[j] += 1
                pk *= p

    xor_sum = 0
    for a in A:
        xor_sum ^= omega[a]

    print("Anna" if xor_sum else "Bruno")

if __name__ == "__main__":
