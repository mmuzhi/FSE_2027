```python
from typing import List

class Solution:
    def sumOfGoodNumbers(self, nums: List[int], k: int) -> int:
        n = len(nums)
        total = 0

        for i, x in enumerate(nums):
            good = True

            if i - k >= 0 and x <= nums[i - k]:
                good = False
            if i + k < n and x <= nums[i + k]:
                good = False

            if good:
                total += x

        return total
```
