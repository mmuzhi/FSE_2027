from typing import List

class Solution:
    def numOfUnplacedFruits(self, fruits: List[int], baskets: List[int]) -> int:
        n = len(baskets)

        size = 1
        while size < n:
            size <<= 1

        seg = [0] * (2 * size)

        for i, cap in enumerate(baskets):
            seg[size + i] = cap

        for i in range(size - 1, 0, -1):
            seg[i] = max(seg[i << 1], seg[i << 1 | 1])

        def first_available(need: int) -> int:
            if seg[1] < need:
                return -1

            idx = 1
            while idx < size:
                left = idx << 1
                if seg[left] >= need:
                    idx = left
                else:
                    idx = left | 1

            pos = idx - size
            return pos if pos < n else -1

        def remove_basket(pos: int) -> None:
            idx = size + pos
            seg[idx] = -1
            idx >>= 1

            while idx:
                seg[idx] = max(seg[idx << 1], seg[idx << 1 | 1])
                idx >>= 1

        unplaced = 0

        for fruit in fruits:
            pos = first_available(fruit)

            if pos == -1:
                unplaced += 1
            else:
                remove_basket(pos)

        return unplaced
