```python
import sys

def main():
    data = list(map(int, sys.stdin.read().split()))
    N, D = data[0], data[1]

    snakes = []
    idx = 2
    for _ in range(N):
        T = data[idx]
        L = data[idx + 1]
        idx += 2
        snakes.append((T, L))

    ans = []
    for k in range(1, D + 1):
        best = 0
        for T, L in snakes:
            weight = T * (L + k)
            if weight > best:
                best = weight
        ans.append(str(best))

    print("\n".join(ans))

if __name__ == "__main__":
    main()
```
