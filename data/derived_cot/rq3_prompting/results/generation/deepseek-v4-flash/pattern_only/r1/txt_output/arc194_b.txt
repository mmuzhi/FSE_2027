import sys
def main():
    data=list(map(int, sys.stdin.buffer.read().split()))
    if not data: return
    N=data[0]; P=data[1:]
    bit=[0]*(N+2)
    def add(i,v):
        while i<=N:
            bit[i]+=v; i+=i&-i
    def sum_(i):
        s=0
        while i>0:
            s+=bit[i]; i-=i&-i
        return s
    ans=0
    for idx,x in enumerate(P, start=1):
        less = sum_(x-1)
        c = (idx-1) - less
        ans += c * idx - c*(c+1)//2
        add(x,1)
    print(ans)
if __name__=="__main__": main()

import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n = data[0]
    bit = [0] * (n + 2)
    ans = 0

    for j in range(1, n + 1):
        x = data[j]

        # count previous elements smaller than x
        s = 0
        i = x - 1
        while i > 0:
            s += bit[i]
            i -= i & -i

        c = (j - 1) - s  # previous elements greater than x
        ans += c * j - c * (c + 1) // 2

        # add x
        i = x
        while i <= n:
            bit[i] += 1
            i += i & -i

    print(ans)

if __name__ == "__main__":
    main()
