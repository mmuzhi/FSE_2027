```python
import sys

def main():
    S = sys.stdin.readline().strip()
    ans = 0
    zero_run = 0

    for ch in S:
        if ch == '0':
            zero_run += 1
        else:
            ans += (zero_run + 1) // 2 + 1
            zero_run = 0

    ans += (zero_run + 1) // 2
    print(ans)

if __name__ == "__main__":
    main()
