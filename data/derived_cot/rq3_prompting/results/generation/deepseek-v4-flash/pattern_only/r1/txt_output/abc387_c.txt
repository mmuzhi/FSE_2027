```python
import sys

MAX_DIGITS = 20
POW = [[1] * MAX_DIGITS for _ in range(10)]
for d in range(1, 10):
    for e in range(1, MAX_DIGITS):
        POW[d][e] = POW[d][e - 1] * d


def count_snake_upto(n):
    if n < 10:
        return 0

    s = str(n)
    length = len(s)
    ans = 0

    # All Snake numbers with fewer digits
    for m in range(2, length):
        for d in range(1, 10):
            ans += POW[d][m - 1]

    # Same length, smaller leading digit
    first = int(s[0])
    for d in range(1, first):
        ans += POW[d][length - 1]

    # Same length, same leading digit
    d = first
    ok = True
    for i in range(1, length):
        digit = int(s[i])
        rem = length - 1 - i

        # Choose a smaller digit at this position
        ans += min(d, digit) * POW[d][rem]

        # If N's digit is not allowed, N itself is not Snake
        if digit >= d:
            ok = False
            break

    if ok:
        ans += 1

    return ans


def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return

    L = int(data[0])
    R = int(data[1])

    print(count_snake_upto(R) - count_snake_upto(L - 1))


if __name__ == "__main__":
    main()
```
