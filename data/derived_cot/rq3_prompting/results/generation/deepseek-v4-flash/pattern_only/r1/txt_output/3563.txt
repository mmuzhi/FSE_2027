```
class Solution:
    def maxScore(self, grid: List[List[int]]) -> int:
        R = len(grid)
        C = len(grid[0])
```

```python
from typing import List

class Solution:
    def maxScore(self, grid: List[List[int]]) -> int:
        R = len(grid)
        C = len(grid[0])

        rows_with_value = [0] * 101
        for r in range(R):
            for c in range(C):
                v = grid[r][c]
                rows_with_value[v] |= 1 << r

        full_mask = (1 << R) - 1
        dp = [-1] * (1 << R)
        dp[0] = 0

        for v in range(1, 101):
            row_mask = rows_with_value[v]
            if row_mask == 0:
                continue

            old = dp[:]

            for mask in range(1 << R):
                if old[mask] < 0:
                    continue

                available = row_mask & (full_mask ^ mask)
                while available:
                    bit = available & -available
                    nmask = mask | bit
                    dp[nmask] = max(dp[nmask], old[mask] + v)
                    available -= bit

        return max(dp)
```
