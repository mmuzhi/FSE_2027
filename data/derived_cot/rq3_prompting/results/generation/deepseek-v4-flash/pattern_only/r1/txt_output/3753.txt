```python
class Solution:
    def maxDifference(self, s: str) -> int:
        freq = {}
        for ch in s:
            freq[ch] = freq.get(ch, 0) + 1

        max_odd = -1
        min_even = float('inf')

        for cnt in freq.values():
            if cnt % 2 == 1:
                max_odd = max(max_odd, cnt)
            else:
                min_even = min(min_even, cnt)

        return max_odd - min_even
```
