```python
from typing import List
import heapq

class Solution:
    def resultsArray(self, queries: List[List[int]], k: int) -> List[int]:
        heap = []
        ans = []

        for x, y in queries:
            d = abs(x) + abs(y)

            if len(heap) < k:
                heapq.heappush(heap, -d)
            elif d < -heap[0]:
                heapq.heapreplace(heap, -d)

            ans.append(-heap[0] if len(heap) == k else -1)

        return ans
```
