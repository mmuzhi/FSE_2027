```python
import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    q = data[0]

    counts = [0] * 1000001
    distinct = 0
    ans = []

    idx = 1
    for _ in range(q):
        t = data[idx]
        idx += 1

        if t == 1:
            x = data[idx]
            idx += 1
            if counts[x] == 0:
                distinct += 1
            counts[x] += 1

        elif t == 2:
            x = data[idx]
            idx += 1
            counts[x] -= 1
            if counts[x] == 0:
                distinct -= 1

        else:
            ans.append(str(distinct))

    sys.stdout.write("\n".join(ans))

if __name__ == "__main__":
    main()
```
