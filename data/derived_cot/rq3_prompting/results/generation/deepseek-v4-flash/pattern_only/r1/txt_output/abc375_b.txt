```python
import sys
import math

def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return

    n = int(data[0])
    px = py = 0
    ans = 0.0
    idx = 1

    for _ in range(n):
        x = int(data[idx])
        y = int(data[idx + 1])
        idx += 2

        ans += math.hypot(x - px, y - py)
        px, py = x, y

    ans += math.hypot(px, py)
    print(ans)

if __name__ == "__main__":
    main()
```
