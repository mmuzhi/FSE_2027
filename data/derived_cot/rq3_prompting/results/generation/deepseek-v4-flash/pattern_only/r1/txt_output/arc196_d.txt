You will be given a competitive programming problem.
There are N towns, numbered 1,2,\ldots,N, arranged in a line in this order.
There are N-1 roads connecting adjacent towns: road j\,(1 \leq j \leq N-1) connects towns j and j+1. For each road j, you can set a strength w_j (an integer that may be negative).

Process Q queries. For the k-th query (1 \le k \le Q), if it is possible to set the strengths of the roads so that the requirements of all people L_k, L_k + 1, \ldots, R_k are satisfied, print Yes; otherwise, print No.

```
INF = 10**9
NEG = -1

class SegMin:
    __slots__ = ('n','size','data')
    def __init__(self, n):
        self.n = n
        size = 1
        while size < n: size <<= 1
        self.size = size
        self.data = [INF] * (2*size)
    def update(self, pos, val):
        i = pos + self.size
        self.data[i] = val
        i >>= 1
        while i:
            left = self.data[i<<1]; right = self.data[i<<1|1]
            self.data[i] = left if left < right else right
            i >>= 1
    def query(self, l, r):
        if l > r: return INF
        l += self.size; r += self.size
        res = INF
        data = self.data
        while l <= r:
            if l & 1:
                if data[l] < res: res = data[l]
                l += 1
            if not (r & 1):
                if data[r] < res: res = data[r]
                r -= 1
            l >>= 1; r >>= 1
        return res
```

```
import sys

INF = 10**9
NEG = -1

class SegMin:
    __slots__ = ('size','data')
    def __init__(self, n):
        size=1
        while size<n: size<<=1
        self.size=size
        self.data=[INF]*(2*size)
    def update(self, pos, val):
        i=pos+self.size
        data=self.data
        data[i]=val
        i>>=1
        while i:
            left=data[i<<1]; right=data[i<<1|1]
            nv = left if left < right else right
            if data[i] == nv:
                break
            data[i]=nv
            i>>=1
    def query(self,l,r):
        if l>r: return INF
        l+=self.size; r+=self.size
        data=self.data
        res=INF
        while l<=r:
            if l&1:
                if data[l] < res: res=data[l]
                l+=1
            if not (r&1):
                if data[r] < res: res=data[r]
                r-=1
            l>>=1; r>>=1
        return res
```

```python
import sys

INF = 10**9
NEG = -1


class SegMin:
    __slots__ = ("size", "data")

    def __init__(self, n):
        size = 1
        while size < n:
            size <<= 1
        self.size = size
        self.data = [INF] * (2 * size)

    def update(self, pos, val):
        i = pos + self.size
        data = self.data
        data[i] = val
        i >>= 1
        while i:
            left = data[i << 1]
            right = data[i << 1 | 1]
            nv = left if left < right else right
            if data[i] == nv:
                break
            data[i] = nv
            i >>= 1

    def query(self, l, r):
        if l > r:
            return INF
        l += self.size
        r += self.size
        data = self.data
        res = INF
        while l <= r:
            if l & 1:
                if data[l] < res:
                    res = data[l]
                l += 1
            if not (r & 1):
                if data[r] < res:
                    res = data[r]
                r -= 1
            l >>= 1
            r >>= 1
        return res


class SegMax:
    __slots__ = ("size", "data")

    def __init__(self, n):
        size = 1
        while size < n:
            size <<= 1
        self.size = size
        self.data = [NEG] * (2 * size)

    def update(self, pos, val):
        i = pos + self.size
        data = self.data
        data[i] = val
        i >>= 1
        while i:
            left = data[i << 1]
            right = data[i << 1 | 1]
            nv = left if left > right else right
            if data[i] == nv:
                break
            data[i] = nv
            i >>= 1

    def query(self, l, r):
        if l > r:
            return NEG
        l += self.size
        r += self.size
        data = self.data
        res = NEG
        while l <= r:
            if l & 1:
                if data[l] > res:
                    res = data[l]
                l += 1
            if not (r & 1):
                if data[r] > res:
                    res = data[r]
                r -= 1
            l >>= 1
            r >>= 1
        return res


def main():

    N, M, Q = data[0], data[1], data[2]
    idx = 3

    L = [0] * (M + 1)
    R = [0] * (M + 1)

    for i in range(1, M + 1):

        T = data[idx + 1]

        if S < T:

            L[i] = S - 1
            R[i] = T - 1
        else:

            L[i] = T - 1
            R[i] = S - 1

    qR = [0] * Q
    for i in range(Q):
        qL[i] = data[idx]
        qR[i] = data[idx + 1]
        idx += 2

    min0_q = min0.query

    max0_q = max0.query
    max1_q = max1.query

    min1_u = min1.update
    max0_u = max0.update
```
