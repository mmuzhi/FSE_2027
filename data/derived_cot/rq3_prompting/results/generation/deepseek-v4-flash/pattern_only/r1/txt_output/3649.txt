```python
from typing import List

class Solution:
    def findMinimumTime(self, strength: List[int], K: int) -> int:
        n = len(strength)
        dp = [10**18] * (1 << n)
        dp[0] = 0

        for mask in range(1 << n):
            j = mask.bit_count()
            if j == n:
                continue

            x = 1 + j * K

            for i in range(n):
                if mask & (1 << i):
                    continue

                wait = (strength[i] + x - 1) // x
                nmask = mask | (1 << i)
                dp[nmask] = min(dp[nmask], dp[mask] + wait)

        return dp[(1 << n) - 1]
```
