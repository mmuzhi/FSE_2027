def can_delete_one(longer, shorter):
    i = j = 0
    skipped = False
    ln = len(longer)

```python
import sys

def can_delete_one(longer, shorter):
    i = j = 0
    skipped = False
    ln = len(longer)
    ls = len(shorter)

    while i < ls and j < ln:
        if shorter[i] == longer[j]:
            i += 1
            j += 1
        else:
            if skipped:
                return False
            skipped = True
            j += 1

    return i == ls and (j == ln or (not skipped and j == ln - 1))


def main():
    data = sys.stdin.read().split()
    if not data:
        return

    K = int(data[0])
    S = data[1]
    T = data[2]

    if K == 0:
        print("Yes" if S == T else "No")
        return

    n, m = len(S), len(T)

    if S == T:
        print("Yes")
        return

    if abs(n - m) > 1:
        print("No")
        return

    if n == m:
        mismatches = 0
        for a, b in zip(S, T):
            if a != b:
                mismatches += 1
                if mismatches > 1:
                    print("No")
                    return
        print("Yes")
    elif n < m:
        print("Yes" if can_delete_one(T, S) else "No")
    else:
        print("Yes" if can_delete_one(S, T) else "No")
```
