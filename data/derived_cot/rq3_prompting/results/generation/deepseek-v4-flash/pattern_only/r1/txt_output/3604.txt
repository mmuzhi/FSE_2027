    def numberOfWays(self, n: int, x: int, y: int) -> int:
        MOD = 10**9 + 7
        base = (1 - y) % MOD
        # factorials
        fact = [1]*(x+1)
        for i in range(1,x+1):
            fact[i] = fact[i-1]*i % MOD
        invfact = [1]*(x+1)
        invfact[x] = pow(fact[x], MOD-2, MOD)
        for i in range(x,0,-1):
            invfact[i-1] = invfact[i]*i % MOD

            if r<0 or r>x: return 0
            return fact[x]*invfact[r]%MOD*invfact[x-r]%MOD
        # precompute powers
        pow_y = [1]*(x+1)
        pow_base = [1]*(x+1)

        return ans

class Solution:
    def numberOfWays(self, n: int, x: int, y: int) -> int:
        MOD = 10**9 + 7
        base = (1 - y) % MOD

        fact = [1] * (x + 1)
        for i in range(1, x + 1):
            fact[i] = fact[i - 1] * i % MOD

        invfact = [1] * (x + 1)
        invfact[x] = pow(fact[x], MOD - 2, MOD)
        for i in range(x, 0, -1):
            invfact[i - 1] = invfact[i] * i % MOD

        pow_y = [1] * (x + 1)
        pow_base = [1] * (x + 1)

```python
class Solution:
    def numberOfWays(self, n: int, x: int, y: int) -> int:
        MOD = 10**9 + 7
        base = (1 - y) % MOD

        fact = [1] * (x + 1)
        for i in range(1, x + 1):
            fact[i] = fact[i - 1] * i % MOD

        invfact = [1] * (x + 1)
        invfact[x] = pow(fact[x], MOD - 2, MOD)
        for i in range(x, 0, -1):
            invfact[i - 1] = invfact[i] * i % MOD

        pow_y = [1] * (x + 1)
        pow_base = [1] * (x + 1)
        for i in range(1, x + 1):
            pow_y[i] = pow_y[i - 1] * y % MOD
            pow_base[i] = pow_base[i - 1] * base % MOD

        ans = 0
        fact_x = fact[x]

        for i in range(1, x + 1):
            comb = fact_x * invfact[i] % MOD * invfact[x - i] % MOD
            term = comb * pow(i, n, MOD) % MOD
            term = term * pow_y[i] % MOD
            term = term * pow_base[x - i] % MOD
            ans = (ans + term) % MOD

        return ans
```
