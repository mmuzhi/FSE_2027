```python
from typing import List

class Solution:
    def stringSequence(self, target: str) -> List[str]:
        res = []
        cur = []

        for ch in target:
            cur.append('a')
            res.append(''.join(cur))

            while cur[-1] != ch:
                cur[-1] = chr(ord(cur[-1]) + 1)
                res.append(''.join(cur))

        return res
