=== ANSWER ===
```python
import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    N = data[0]
    A = data[1:]

    freq = {}
    for a in A:
        freq[a] = freq.get(a, 0) + 1

    best_value = -1
    answer = -1

    for i, a in enumerate(A, start=1):
        if freq[a] == 1 and a > best_value:
            best_value = a
            answer = i

    print(answer)

if __name__ == "__main__":
    main()
```
