=== ANSWER ===
```python
import sys

def main():
    input = sys.stdin.readline
    N = int(input())
    S = input().strip()

    ans = 0
    for i in range(N - 2):
        if S[i] == '#' and S[i + 1] == '.' and S[i + 2] == '#':
            ans += 1

    print(ans)

if __name__ == "__main__":
    main()
```
