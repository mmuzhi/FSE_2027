from typing import List

class Solution:
    def isZeroArray(self, nums: List[int], queries: List[List[int]]) -> bool:
        n = len(nums)
        diff = [0] * (n + 1)
        for l, r in queries:
            diff[l] += 1
            diff[r + 1] -= 1
        cover = 0
        for i in range(n):
            cover += diff[i]
            if cover < nums[i]:
                return False
        return True

```python
from typing import List

class Solution:
    def isZeroArray(self, nums: List[int], queries: List[List[int]]) -> bool:
        n = len(nums)
        diff = [0] * (n + 1)

        for l, r in queries:
            diff[l] += 1
            diff[r + 1] -= 1

        covered = 0
        for i in range(n):
            covered += diff[i]
            if covered < nums[i]:
                return False

        return True
```
