=== ANSWER ===
```python
class Solution:
    def findValidPair(self, s: str) -> str:
        cnt = [0] * 10
        for ch in s:
            cnt[int(ch)] += 1

        for i in range(len(s) - 1):
            a = int(s[i])
            b = int(s[i + 1])

            if a == b:
                continue

            if cnt[a] == a and cnt[b] == b:
                return s[i:i + 2]

        return ""
```
