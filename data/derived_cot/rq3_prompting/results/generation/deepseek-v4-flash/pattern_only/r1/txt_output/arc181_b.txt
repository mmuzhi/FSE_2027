def primitive_period(s):
    n = len(s)
    pi = [0]*n
    for i in range(1,n):

import sys

def primitive_period(s):
    n = len(s)
    pi = [0] * n
    for i in range(1, n):
        j = pi[i - 1]
        while j > 0 and s[i] != s[j]:
            j = pi[j - 1]
        if s[i] == s[j]:
            j += 1
        pi[i] = j
    p = n - pi[-1]
    if n % p == 0:
        return p
    return n

def solve():
    input = sys.stdin.buffer.readline
    t_line = input()
    if not t_line:
        return
    t = int(t_line)
    ans = []
    for _ in range(t):
        S = input().strip()
        X = input().strip()
        Y = input().strip()
        s_len = len(S)
        c0x = X.count(b'0')
        c1x = len(X) - c0x
        c0y = Y.count(b'0')
        c1y = len(Y) - c0y
        if c0x == c0y and c1x == c1y:
            ans.append("Yes")
            continue
        d0 = c0x - c0y
        d1 = c1x - c1y
        if d1 == 0:
            ans.append("No")
            continue
        num = -s_len * d0
        if num % abs(d1) != 0:
            ans.append("No")
            continue
        r = num // d1
        if r < 0:
            ans.append("No")
            continue
        p = primitive_period(S)
        if r % p == 0:
            ans.append("Yes")
        else:
            ans.append("No")
    sys.stdout.write("\n".join(ans))

if __name__ == "__main__":
    solve()

import sys

def primitive_period(s):
    n = len(s)
    pi = [0] * n
    for i in range(1, n):
        j = pi[i - 1]
        while j > 0 and s[i] != s[j]:
            j = pi[j - 1]
        if s[i] == s[j]:
            j += 1
        pi[i] = j
    p = n - pi[-1]
    if n % p == 0:
        return p
    return n

def solve():
    input = sys.stdin.buffer.readline
    t = int(input())
    ans = []
    for _ in range(t):
        S = input().strip()
        X = input().strip()
        Y = input().strip()

        c0x = X.count(b'0')
        c1x = len(X) - c0x
        c0y = Y.count(b'0')

            continue
