=== ANSWER ===
```python
import sys

def main():
    data = list(map(int, sys.stdin.read().split()))
    N = data[0]
    P = data[1:1 + N]

    ans = []
    for score in P:
        rank = 1 + sum(1 for other in P if other > score)
        ans.append(str(rank))

    print("\n".join(ans))

if __name__ == "__main__":
    main()
```
