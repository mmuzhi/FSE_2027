import sys

def main():
    S = sys.stdin.read().strip().split()
    
    safe_rows = sum(1 for row in S if '#' not in row)
    safe_cols = sum(1 for col in zip(*S) if '#' not in col)
    
    print(safe_rows * safe_cols)

if __name__ == "__main__":
    main()
