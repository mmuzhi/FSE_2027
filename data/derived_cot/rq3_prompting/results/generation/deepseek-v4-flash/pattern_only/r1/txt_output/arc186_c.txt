import sys, heapq

def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    idx = 0
    T = data[idx]; idx += 1
    out = []
    for _ in range(T):
        N = data[idx]; M = data[idx+1]; idx += 2
        boxes = []
        for _ in range(N):
            V = data[idx]; P = data[idx+1]; idx += 2
            if V >= 2:
                d = V - 1
                b = 1 - P
                a = V - P
                boxes.append((d, b, a))
        boxes.sort(key=lambda x: (x[0], -x[1]))
        n = len(boxes)
        K = M - 1
        NEG = -10**30
        U = [NEG] * (n + 1)
        if K == 0:
            U = [0] * (n + 1)
        else:
            heap = []
            ssum = 0
            # U[n] already NEG
            for i in range(n-1, -1, -1):
                b = boxes[i][1]
                heapq.heappush(heap, b)
                ssum += b
                if len(heap) > K:
                    ssum -= heapq.heappop(heap)
                if len(heap) == K:
                    U[i] = ssum
        pref = 0
        ans = 0
        for s in range(n + 1):
            if s > 0:
                a = boxes[s-1][2]
                if a > 0:
                    pref += a
            if U[s] != NEG:
                val = pref + U[s]
                if val > ans:
                    ans = val
        out.append(str(ans))
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    solve()

```python
import sys
from heapq import heappush, heappop

def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    idx = 0
    T = data[idx]; idx += 1
    out = []
    NEG = -10**30
    for _ in range(T):
        N = data[idx]; M = data[idx+1]; idx += 2
        boxes = []
        for _ in range(N):
            V = data[idx]; P = data[idx+1]; idx += 2
            if V >= 2:
                d = V - 1
                b = 1 - P
                a = V - P
                boxes.append((d, b, a))
        boxes.sort(key=lambda x: (x[0], -x[1]))
        n = len(boxes)
        K = M - 1
        U = [NEG] * (n + 1)
        if K == 0:
            U = [0] * (n + 1)
        else:
            heap = []
            ssum = 0
            for i in range(n - 1, -1, -1):
                b = boxes[i][1]
                heappush(heap, b)
                ssum += b
                if len(heap) > K:
                    ssum -= heappop(heap)
                if len(heap) == K:
                    U[i] = ssum
        pref = 0
        ans = 0
        for s in range(n + 1):
            if s > 0:
                a = boxes[s - 1][2]
                if a > 0:
                    pref += a
            if U[s] != NEG:
                val = pref + U[s]
                if val > ans:
                    ans = val
        out.append(str(ans))
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    solve()
```

```python
import sys
from heapq import heappush, heappop

def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    idx = 0
    T = data[idx]
    idx += 1

    NEG = -10**30
    out = []

    for _ in range(T):
        N = data[idx]
        M = data[idx + 1]
        idx += 2

        boxes = []
        for _ in range(N):
            V = data[idx]
            P = data[idx + 1]
            idx += 2

            # V = 1 boxes are never useful: they give no remaining buffer
            # and their net profit is <= 0.
            if V >= 2:
                d = V - 1          # remaining capacity after first ball
                b = 1 - P          # net profit if this box is never consumed
                a = V - P          # net profit if this box is fully consumed
                boxes.append((d, b, a))

        # Sort by remaining capacity d.
        # For equal d, larger b (smaller P) is better.
        boxes.sort(key=lambda x: (x[0], -x[1]))

        n = len(boxes)
        K = M - 1  # number of boxes that remain unconsumed at the end

        # U[s] = sum of largest K values of b in boxes[s:]
        U = [NEG] * (n + 1)
        if K == 0:
            U = [0] * (n + 1)
        else:
            heap = []
            ssum = 0
            for i in range(n - 1, -1, -1):
                b = boxes[i][1]
                heappush(heap, b)
                ssum += b
                if len(heap) > K:
                    ssum -= heappop(heap)
                if len(heap) == K:
                    U[i] = ssum

        pref = 0
        ans = 0

        for s in range(n + 1):
            if s > 0:
                a = boxes[s - 1][2]
                if a > 0:
                    pref += a

            if U[s] != NEG:
                val = pref + U[s]
                if val > ans:
                    ans = val

        out.append(str(ans))

    sys.stdout.write("\n".join(out))


if __name__ == "__main__":
    solve()
```
