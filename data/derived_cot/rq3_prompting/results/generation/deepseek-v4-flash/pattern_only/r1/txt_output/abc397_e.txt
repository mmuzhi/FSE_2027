You are given a tree with NK vertices. The vertices are numbered 1,2,\dots,NK, and the i-th edge (i=1,2,\dots,NK-1) connects vertices u_i and v_i bidirectionally.
Determine whether this tree can be decomposed into N paths, each of length K. More precisely, determine whether there exists an N \times K matrix P satisfying the following:

- P_{1,1}, \dots, P_{1,K}, P_{2,1}, \dots, P_{N,K} is a permutation of 1,2,\dots,NK.
- For each i=1,2,\dots,N and j=1,2,\dots,K-1, there is an edge connecting vertices P_{i,j} and P_{i,j+1}.

Input

Output

Constraints


- 1 \leq N
- 1 \leq K
- NK \leq 2 \times 10^5
- 1 \leq u_i < v_i \leq NK
- The given graph is a tree.
- All input values are integers.

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    N = int(next(it))
    K = int(next(it))
    V = N * K

        adj[u].append(v)

        order.append(v)
        for to in adj[v]:
            if to == parent[v]:
                continue
            if parent[to] == 0:
                parent[to] = v
                stack.append(to)

        for to in adj[v]:

```python
import sys

def solve():
    it = iter(map(int, sys.stdin.buffer.read().split()))
    try:
        N = next(it)
    except StopIteration:
        return
    K = next(it)

    V = N * K
    adj = [[] for _ in range(V + 1)]

    for _ in range(V - 1):
        u = next(it)
        v = next(it)
        adj[u].append(v)
        adj[v].append(u)

    parent = [0] * (V + 1)
    parent[1] = -1
    order = []
    stack = [1]

    while stack:
        v = stack.pop()
        order.append(v)
        for to in adj[v]:
            if to == parent[v]:
                continue
            if parent[to] == 0:
                parent[to] = v
                stack.append(to)

    r = [0] * (V + 1)

    for v in reversed(order):
        total = 1
        for to in adj[v]:
            if to == parent[v]:
                continue
            total += r[to]
        r[v] = total % K

    for v in order:
        cnt = 0
        for to in adj[v]:
            if to == parent[v]:
                continue
            if r[to] != 0:
                cnt += 1

        if r[v] != 0:
            if cnt > 1:
                print("No")
                return
        else:
            if cnt > 2:
                print("No")
                return

    print("Yes")

if __name__ == "__main__":
    solve()
```
