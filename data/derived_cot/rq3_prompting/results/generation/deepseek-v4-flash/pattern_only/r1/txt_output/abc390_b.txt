import sys

def main():
    data = list(map(int, sys.stdin.read().split()))
    N = data[0]
    A = data[1:1 + N]

    for i in range(1, N - 1):
        if A[i] * A[i] != A[i - 1] * A[i + 1]:
            print("No")
            return

    print("Yes")

if __name__ == "__main__":
    main()
