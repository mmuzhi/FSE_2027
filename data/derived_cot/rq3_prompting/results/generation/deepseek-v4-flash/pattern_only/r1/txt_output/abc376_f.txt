You are holding a ring with both hands.
This ring consists of N\ (N \geq 3) parts numbered 1,2,\dots,N, where parts i and i+1 (1 \leq i \leq N-1) are adjacent, and parts 1 and N are also adjacent.
Initially, your left hand is holding part 1, and your right hand is holding part 2.
In one operation, you can do the following:

- Move one of your hands to an adjacent part of the part it is currently holding. However, you can do this only if the other hand is not on the destination part.

The following figure shows the initial state and examples of operations that can and cannot be made from there. The number written on each part of the ring represents the part number, and the circles labeled L and R represent your left and right hands, respectively.

You need to follow Q instructions given to you in order.
The i-th (1 \leq i \leq Q) instruction is represented by a character H_i and an integer T_i, meaning the following:

- Perform some number of operations (possibly zero) so that your left hand (if H_i is L) or your right hand (if H_i is R) is holding part T_i.
  Here, you may move the other hand not specified by H_i.

Under the settings and constraints of this problem, it can be proved that any instructions are achievable.
Find the minimum total number of operations required to follow all the instructions.

Input

The Input is given from Standard Input in the following format:
N Q
H_1 T_1
H_2 T_2
\vdots
H_Q T_Q

Output

Print the minimum total number of operations required to follow all the instructions.

Constraints


- 3\leq N \leq 3000
- 1\leq Q \leq 3000
- H_i is L or R.
- 1 \leq T_i \leq N
- N, Q, and T_i are integers.

Sample Input 1

6 3
R 4
L 5
R 5

Sample Output 1

6

            v = val - idx

            pref[i] = best
        suff = [0]*m

            idx = l + i

        if q < l:

        if q > r:
            return self.pref_minus[-1] + q
        k = q - l
        left = self.pref_minus[k] + q
        right = self.suff_plus[k] - q
        return left if left < right else right

import sys

def solve():
    input = sys.stdin.readline
    N, Q = map(int, input().split())
    instructions = []
    for _ in range(Q):
        h, t = input().split()
        instructions.append((h, int(t) - 1))

    INF = 10**18
    dp = [INF] * N
    dp[1] = 0  # initial: L=0, R=1, using dummy previous specified hand L at 0

    prev_spec = 'L'
    p = 0
    shifts = [-N, 0, N]
    range_N = range(N)

    def build_pref_suff(l, vals):
        m = len(vals)
        pref = [0] * m
        best = INF
        for i, val in enumerate(vals):
            idx = l + i
            v = val - idx
            if v < best:
                best = v
            pref[i] = best

        suff = [0] * m
        best = INF
        for i in range(m - 1, -1, -1):
            idx = l + i
            v = vals[i] + idx
            if v < best:
                best = v
            suff[i] = best

        return pref, suff, pref[-1], suff[0]

    for h, t in instructions:
        new_dp = [INF] * N

        # other_lift[y] is the lift of the "other" hand in the final state
        other_lift = [0] * N
        if h == 'L':
            for y in range_N:
                if y == t:
                    continue
                other_lift[y] = y if y > t else y + N
        else:
            for y in range_N:
                if y == t:
                    continue
                other_lift[y] = t if y < t else t + N

        if prev_spec == 'L':
            # Previous state: L = p, R = x.
            # R's lift relative to L is X = x if x > p else x + N.
            l = p + 1
            r = p + N - 1
            vals = [dp[X % N] for X in range(l, r + 1)]
            pref, suff, pref_last, suff0 = build_pref_suff(l, vals)

            for sn in shifts:
                if h == 'L':
                    add_const = abs(t + sn - p)
                    for y in range_N:
                        if y == t:
                            continue
                        b = other_lift[y] + sn
                        if b < l:
                            qv = suff0 - b
                        elif b > r:
                            qv = pref_last + b
                        else:
                            k = b - l
                            left = pref[k] + b
                            right = suff[k] - b
                            qv = left if left < right else right
                        val = add_const + qv
                        if val < new_dp[y]:
                            new_dp[y] = val
                else:
                    for y in range_N:
                        if y == t:
                            continue
                        term1 = abs(y + sn - p)
                        b = other_lift[y] + sn
                        if b < l:
                            qv = suff0 - b
                        elif b > r:
                            qv = pref_last + b
                        else:
                            k = b - l
                            left = pref[k] + b
                            right = suff[k] - b
                            qv = left if left < right else right
                        val = term1 + qv
                        if val < new_dp[y]:
                            new_dp[y] = val

        else:
            # Previous state: R = p, L = x.
            # R's lift is p if x < p, and p + N if x > p.
            intervals = []
            if p > 0:
                l0 = 0
                r0 = p - 1
                vals = dp[0:p]
                pref, suff, pref_last, suff0 = build_pref_suff(l0, vals)
                intervals.append((l0, r0, pref, suff, pref_last, suff0, p))
            if p < N - 1:
                l0 = p + 1
                r0 = N - 1
                vals = dp[p + 1:N]
                pref, suff, pref_last, suff0 = build_pref_suff(l0, vals)
                intervals.append((l0, r0, pref, suff, pref_last, suff0, p + N))

            for sn in shifts:
                if h == 'R':
                    for y in range_N:
                        if y == t:
                            continue
                        a = y + sn
                        ol = other_lift[y] + sn
                        best_val = INF
                        for l0, r0, pref, suff, pref_last, suff0, const in intervals:
                            if a < l0:
                                qv = suff0 - a
                            elif a > r0:
                                qv = pref_last + a
                            else:
                                k = a - l0
                                left = pref[k] + a
                                right = suff[k] - a
                                qv = left if left < right else right
                            v = qv + abs(ol - const)
                            if v < best_val:
                                best_val = v
                        if best_val < new_dp[y]:
                            new_dp[y] = best_val
                else:
                    a_const = t + sn
                    qv_list = []
                    for l0, r0, pref, suff, pref_last, suff0, const in intervals:
                        a = a_const
                        if a < l0:
                            qv = suff0 - a
                        elif a > r0:
                            qv = pref_last + a
                        else:
                            k = a - l0
                            left = pref[k] + a
                            right = suff[k] - a
                            qv = left if left < right else right
                        qv_list.append((qv, const))

                    for y in range_N:
                        if y == t:
                            continue
                        ol = other_lift[y] + sn
                        best_val = INF
                        for qv, const in qv_list:
                            v = qv + abs(ol - const)
                            if v < best_val:
                                best_val = v
                        if best_val < new_dp[y]:
                            new_dp[y] = best_val

        dp = new_dp
        prev_spec = h
        p = t

    print(min(dp))


if __name__ == "__main__":
    solve()
