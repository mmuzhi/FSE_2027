def build_trans():
    trans0 = []
    trans1 = []
    for r in range(16):
        t0 = []
        t1 = []

        # d = 0
        ns = ((r & 1) << 1) | (((r >> 2) & 1) << 3)
        if ns:
            t0.append(ns)
            t1.append(ns)

        # d = 1
        # s=0
        if r:
            t0.append(r)
        # s=1
        b0 = r & 1
        b2 = (r >> 2) & 1
        b1 = (r & 1) | ((r >> 1) & 1)
        b3 = ((r >> 2) & 1) | ((r >> 3) & 1)
        ns = b0 | (b1 << 1) | (b2 << 2) | (b3 << 3)
        if ns:
            t1.append(ns)

        # d = 2
        # s=0
        ns = ((r >> 1) & 1) | (((r >> 3) & 1) << 2)
        if ns:
            t0.append(ns)
        # s=1

def build_trans():
    trans0 = []
    trans1 = []
    for r in range(16):
        t0 = []

        ns = ((r & 1) << 1) | (((r >> 2) & 1) << 3)

            t1.append(ns)

            t0.append(r)
        b0 = r & 1
        b2 = (r >> 2) & 1
        b1 = b0 | ((r >> 1) & 1)
        b3 = b2 | ((r >> 3) & 1)
        ns = b0 | (b1 << 1) | (b2 << 2) | (b3 << 3)
        if ns:
            t1.append(ns)

        # d = 2
        ns = ((r >> 1) & 1) | (((r >> 3) & 1) << 2)
        if ns:
            t0.append(ns)
        b0 = (r & 1) | ((r >> 1) & 1)

        b1 = (r >> 1) & 1
        b3 = (r >> 3) & 1
        ns = b0 | (b1 << 1) | (b2 << 2) | (b3 << 3)
        if ns:
            t1.append(ns)

        # d = 3
        ns = ((r >> 1) & 1) | (((r >> 3) & 1) << 2)
        if ns:
            t1.append(ns)

        trans0.append(tuple(t0))
        trans1.append(tuple(t1))

    reachable = [False] * 16

    stack = [9]
    while stack:
        r = stack.pop()
        for tr in (trans0, trans1):
            for ns in tr[r]:
                if not reachable[ns]:
                    reachable[ns] = True
                    stack.append(ns)

    active0 = [(r, trans0[r]) for r in range(16) if reachable[r]]
    active1 = [(r, trans1[r]) for r in range(16) if reachable[r]]
    return active0, active1

import sys

MOD = 998244353


def build_trans():
    trans0 = []
    trans1 = []

    for r in range(16):
        t0 = []
        t1 = []

        # d = 0
        ns = ((r & 1) << 1) | (((r >> 2) & 1) << 3)
        if ns:
            t0.append(ns)
            t1.append(ns)

        # d = 1
        if r:
            t0.append(r)
        b0 = r & 1
        b2 = (r >> 2) & 1
        b1 = b0 | ((r >> 1) & 1)
        b3 = b2 | ((r >> 3) & 1)
        ns = b0 | (b1 << 1) | (b2 << 2) | (b3 << 3)
        if ns:
            t1.append(ns)

        # d = 2

    active0 = [(r, trans0[r]) for r in range(16) if reachable[r]]
