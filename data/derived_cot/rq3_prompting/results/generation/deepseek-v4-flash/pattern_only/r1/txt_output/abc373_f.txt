You will be given a competitive programming problem.
There are N types of items. The i-th type of item has a weight of w_i and a value of v_i. Each type has 10^{10} items available.
Takahashi is going to choose some items and put them into a bag with capacity W. He wants to maximize the value of the selected items while avoiding choosing too many items of the same type. Hence, he defines the happiness of choosing k_i items of type i as k_i v_i - k_i^2. He wants to choose items to maximize the total happiness over all types while keeping the total weight at most W. Calculate the maximum total happiness he can achieve.

Input

The input is given from Standard Input in the following format:
N W
w_1 v_1
w_2 v_2
\vdots
w_N v_N

Output

Print the answer.

Constraints


- 1 \leq N \leq 3000
- 1 \leq W \leq 3000
- 1 \leq w_i \leq W
- 1 \leq v_i \leq 10^9
- All input values are integers.

Sample Input 1

2 10
3 4
3 2

Sample Output 1

5

By choosing 2 items of type 1 and 1 item of type 2, the total happiness can be 5, which is optimal.
Here, the happiness for type 1 is 2 \times 4 - 2^2 = 4, and the happiness for type 2 is 1 \times 2 - 1^2 = 1.
The total weight is 9, which is within the capacity 10.

def solve():
    read N,W
    dp = [0]*(W+1)
    slopes = []
    intercepts = []
    for each item:
        w = int; a = int
        limit = W - w + 1
        if limit > w:
            limit = w
        for r in range(limit):
            M = (W - r) // w
            if M == 1:
                idx = r + w
                val = dp[r] + a - 1
                if val > dp[idx]:
                    dp[idx] = val
                continue
            # M >= 2
            size = 0
            head = 0
            # add line j=0
            if size == len(slopes):
                slopes.append(0)
                intercepts.append(dp[r])
            else:
                slopes[size] = 0
                intercepts[size] = dp[r]
            size += 1
            for m in range(1, M+1):
                idx = r + m*w
                g = dp[idx]
                s = m + m
                b = g - a*m - m*m
                # insert
                while size - head >= 2:
                    s1 = slopes[size-2]; b1 = intercepts[size-2]
                    s2 = slopes[size-1]; b2 = intercepts[size-1]
                    if (b2 - b) * (s2 - s1) <= (b1 - b2) * (s - s2):
                        size -= 1
                    else:
                        break
                if size == len(slopes):
                    slopes.append(s)
                    intercepts.append(b)
                else:
                    slopes[size] = s
                    intercepts[size] = b
                size += 1
                # query
                x = m
                while size - head >= 2:
                    if slopes[head]*x + intercepts[head] <= slopes[head+1]*x + intercepts[head+1]:
                        head += 1
                    else:
                        break
                best = slopes[head]*x + intercepts[head]
                dp[idx] = a*m - m*m + best
    print(dp[W])

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    N = data[0]; W = data[1]
    dp = [0]*(W+1)
    slopes = []
    intercepts = []
    p = 2
    for _ in range(N):
        w = data[p]; v = data[p+1]; p += 2
        limit = W - w + 1
        if limit > w:
            limit = w
        for r in range(limit):
            M = (W - r) // w
            if M == 1:
                idx = r + w
                val = dp[r] + v - 1
                if val > dp[idx]:
                    dp[idx] = val
                continue
            size = 0
            head = 0
            if size == len(slopes):
                slopes.append(0)
                intercepts.append(dp[r])
            else:
                slopes[size] = 0
                intercepts[size] = dp[r]
            size += 1
            idx = r + w
            for m in range(1, M + 1):
                g = dp[idx]
                am = v * m
                mm = m * m
                s = m + m
                b = g - am - mm

                while size - head >= 2:
                    s1 = slopes[size - 2]
                    b1 = intercepts[size - 2]
                    s2 = slopes[size - 1]
                    b2 = intercepts[size - 1]
                    if (b2 - b) * (s2 - s1) <= (b1 - b2) * (s - s2):
                        size -= 1
                    else:
                        break

                if size == len(slopes):
                    slopes.append(s)
                    intercepts.append(b)
                else:
                    slopes[size] = s
                    intercepts[size] = b
                size += 1

                x = m
                while size - head >= 2:
                    if slopes[head] * x + intercepts[head] <= slopes[head + 1] * x + intercepts[head + 1]:

                    else:
                        break

```python
import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    N = data[0]
    W = data[1]
    dp = [0] * (W + 1)

    # Reusable arrays for convex hull trick
    slopes = [0] * (W + 1)
    intercepts = [0] * (W + 1)

    p = 2
    for _ in range(N):
        w = data[p]
        v = data[p + 1]
        p += 2

        limit = W - w + 1
        if limit > w:
            limit = w

        for r in range(limit):
            M = (W - r) // w

            # Only k=0 or k=1 is possible.
            if M == 1:
                idx = r + w
                val = dp[r] + v - 1
                if val > dp[idx]:
                    dp[idx] = val
                continue

            # Convex hull trick for this residue class.
            size = 1
            head = 0
            slopes[0] = 0
            intercepts[0] = dp[r]

            idx = r + w
            for m in range(1, M + 1):
                g = dp[idx]
                am = v * m
                mm = m * m
                s = m + m
                b = g - am - mm

                # Add line: y = s*x + b
                while size - head >= 2:
                    s1 = slopes[size - 2]
                    b1 = intercepts[size - 2]
                    s2 = slopes[size - 1]
                    b2 = intercepts[size - 1]

                    if (b2 - b) * (s2 - s1) <= (b1 - b2) * (s - s2):
                        size -= 1
                    else:
                        break

                slopes[size] = s
                intercepts[size] = b
                size += 1

                # Query at x = m
                while size - head >= 2:
                    if slopes[head] * m + intercepts[head] <= slopes[head + 1] * m + intercepts[head + 1]:
                        head += 1
                    else:
                        break

                best = slopes[head] * m + intercepts[head]
                dp[idx] = am - mm + best
                idx += w

    print(dp[W])

if __name__ == "__main__":
    main()
```
