```python
import sys

def main():
    data = list(map(int, sys.stdin.read().split()))
    N, M = data[0], data[1]
    A = set(data[2:2 + M])

    ans = [i for i in range(1, N + 1) if i not in A]

    print(len(ans))
    print(*ans)

if __name__ == "__main__":
    main()
```
