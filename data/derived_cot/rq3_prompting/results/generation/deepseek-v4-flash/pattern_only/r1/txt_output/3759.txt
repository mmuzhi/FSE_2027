```python
from typing import List
import heapq

class Solution:
    def findMaxSum(self, nums1: List[int], nums2: List[int], k: int) -> List[int]:
        n = len(nums1)
        ans = [0] * n
        order = sorted(range(n), key=nums1.__getitem__)
        heap = []
        total = 0
        i = 0
        while i < n:
            value = nums1[order[i]]
            j = i
            while j < n and nums1[order[j]] == value:
                j += 1

            for p in range(i, j):
                ans[order[p]] = total

            for p in range(i, j):
                x = nums2[order[p]]
                if len(heap) < k:
                    heapq.heappush(heap, x)
                    total += x
                elif x > heap[0]:
                    old = heapq.heapreplace(heap, x)
                    total += x - old

            i = j

        return ans
```

```python
from typing import List
import heapq

class Solution:
    def findMaxSum(self, nums1: List[int], nums2: List[int], k: int) -> List[int]:
        if k == 0:
            return [0] * len(nums1)
        
        n = len(nums1)
        ans = [0] * n
        order = sorted(range(n), key=nums1.__getitem__)
        
        heap = []
        total = 0
        i = 0
        
        while i < n:
            value = nums1[order[i]]
            j = i
            while j < n and nums1[order[j]] == value:
                j += 1
            
            for p in range(i, j):
                ans[order[p]] = total
            
            for p in range(i, j):
                x = nums2[order[p]]
                if len(heap) < k:
                    heapq.heappush(heap, x)
                    total += x
                elif x > heap[0]:
                    old = heapq.heapreplace(heap, x)
                    total += x - old
            
            i = j
        
        return ans
```

```python
from typing import List
import heapq

class Solution:
    def findMaxSum(self, nums1: List[int], nums2: List[int], k: int) -> List[int]:
        if k == 0:
            return [0] * len(nums1)

        n = len(nums1)
        ans = [0] * n
        order = sorted(range(n), key=nums1.__getitem__)

        heap = []
        total = 0
        i = 0

        while i < n:
            value = nums1[order[i]]
            j = i
            while j < n and nums1[order[j]] == value:
                j += 1

            # Answer for this group uses only strictly smaller nums1 values.
            for p in range(i, j):
                ans[order[p]] = total

            # Add this group for future larger nums1 values.
            for p in range(i, j):
                x = nums2[order[p]]
                if len(heap) < k:
                    heapq.heappush(heap, x)
                    total += x
                elif x > heap[0]:
                    old = heapq.heapreplace(heap, x)
                    total += x - old

            i = j

        return ans
```
