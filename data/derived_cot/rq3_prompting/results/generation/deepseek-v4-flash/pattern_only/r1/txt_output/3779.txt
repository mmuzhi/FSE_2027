```python
from typing import List

class Solution:
    def maxWeight(self, pizzas: List[int]) -> int:
        pizzas.sort()
        n = len(pizzas)
        days = n // 4

        odd_days = (days + 1) // 2
        even_days = days // 2

        ans = 0

        # Odd days: gain the largest pizza of the group.
        for i in range(odd_days):
            ans += pizzas[n - 1 - i]

        # Even days: gain the second-largest pizza of the group.
        idx = n - odd_days - 2
        for _ in range(even_days):
            ans += pizzas[idx]
            idx -= 2

        return ans
```
