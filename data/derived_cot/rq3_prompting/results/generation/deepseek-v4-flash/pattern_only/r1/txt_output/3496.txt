class Solution:
    def minNumberOfSeconds(self, mountainHeight: int, workerTimes: List[int]) -> int:

                lo = mid + 1
        return lo

```python
from typing import List
from math import isqrt

class Solution:
    def minNumberOfSeconds(self, mountainHeight: int, workerTimes: List[int]) -> int:
        def can_finish(limit: int) -> bool:
            total = 0
            for w in workerTimes:
                q = (2 * limit) // w
                x = (isqrt(1 + 4 * q) - 1) // 2
                total += x
                if total >= mountainHeight:
                    return True
            return False

        lo = 0
        hi = min(workerTimes) * mountainHeight * (mountainHeight + 1) // 2

        while lo < hi:
            mid = (lo + hi) // 2
            if can_finish(mid):
                hi = mid
            else:
                lo = mid + 1

        return lo
```
