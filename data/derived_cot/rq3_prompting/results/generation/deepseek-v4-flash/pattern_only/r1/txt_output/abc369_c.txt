```python
import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    N = data[0]
    A = data[1:]

    ans = N

    if N >= 2:
        prev_diff = A[1] - A[0]
        run = 1

        for i in range(2, N):
            cur_diff = A[i] - A[i - 1]
            if cur_diff == prev_diff:
                run += 1
            else:
                ans += run * (run + 1) // 2
                prev_diff = cur_diff
                run = 1

        ans += run * (run + 1) // 2

    print(ans)

if __name__ == "__main__":
    main()
```
