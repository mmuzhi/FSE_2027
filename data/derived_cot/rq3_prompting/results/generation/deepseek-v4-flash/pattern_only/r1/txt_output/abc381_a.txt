```python
import sys

def main():
    data = sys.stdin.read().split()
    N = int(data[0])
    S = data[1]

    if N % 2 == 0:
        print("No")
        return

    k = (N + 1) // 2
    expected = '1' * (k - 1) + '/' + '2' * (k - 1)

    if S == expected:
        print("Yes")
    else:
        print("No")

if __name__ == "__main__":
    main()
```
