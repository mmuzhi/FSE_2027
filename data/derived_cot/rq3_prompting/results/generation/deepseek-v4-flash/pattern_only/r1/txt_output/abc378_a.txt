=== ANSWER ===
```python
import sys

def main():
    A = list(map(int, sys.stdin.read().split()))
    cnt = [0] * 5
    for a in A:
        cnt[a] += 1

    ans = sum(c // 2 for c in cnt)
    print(ans)

if __name__ == "__main__":
    main()
```
