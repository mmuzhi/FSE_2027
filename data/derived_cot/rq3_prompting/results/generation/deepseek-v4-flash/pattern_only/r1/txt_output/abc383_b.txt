```python
import sys

def main():
    data = sys.stdin.read().split()
    if not data:
        return

    H = int(data[0])
    W = int(data[1])
    D = int(data[2])
    grid = data[3:3 + H]

    floors = []
    for i in range(H):
        for j in range(W):
            if grid[i][j] == '.':
                floors.append((i, j))

    n = len(floors)
    ans = 0

    for i in range(n):
        r1, c1 = floors[i]
        for j in range(i + 1, n):
            r2, c2 = floors[j]
            cnt = 0

            for r, c in floors:
                d1 = abs(r - r1) + abs(c - c1)
                d2 = abs(r - r2) + abs(c - c2)
                if d1 <= D or d2 <= D:
                    cnt += 1

            if cnt > ans:
                ans = cnt
                if ans == n:
                    print(ans)
                    return

    print(ans)

if __name__ == "__main__":
    main()
```
