```python
from typing import List

class Solution:
    def sortMatrix(self, grid: List[List[int]]) -> List[List[int]]:
        n = len(grid)

        for d in range(-n + 1, n):
            diag = []
            for i in range(max(0, d), min(n, n + d)):
                diag.append(grid[i][i - d])

            if d >= 0:
                diag.sort(reverse=True)
            else:
                diag.sort()

            k = 0
            for i in range(max(0, d), min(n, n + d)):
                grid[i][i - d] = diag[k]
                k += 1

        return grid
```
