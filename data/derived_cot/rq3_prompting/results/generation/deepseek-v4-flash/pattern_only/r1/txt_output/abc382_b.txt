=== ANSWER ===
```python
import sys

def main():
    data = sys.stdin.read().split()
    N = int(data[0])
    D = int(data[1])
    S = data[2]

    ans = list(S)
    removed = 0

    for i in range(N - 1, -1, -1):
        if ans[i] == '@':
            ans[i] = '.'
            removed += 1
            if removed == D:
                break

    print(''.join(ans))

if __name__ == "__main__":
    main()
```
