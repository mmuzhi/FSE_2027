```
from typing import List
from collections import deque

class Solution:
    def countNonDecreasingSubarrays(self, nums: List[int], k: int) -> int:
        n = len(nums)
        dq = deque()
        B = 0
        total = 0
        R = n - 1
        ans = 0

        for L in range(n - 1, -1, -1):
            x = nums[L]
            removed = 0
            while dq and dq[0][1] <= x:
                pos, val = dq.popleft()
                nxt = dq[0][0] if dq else R + 1
                removed += val * (nxt - pos)
            nxt = dq[0][0] if dq else R + 1
            B = B - removed + x * (nxt - L)
            dq.appendleft((L, x))
            total += x

            while B - total > k:
                total -= nums[R]
                pos, val = dq[-1]
                B -= val
                if pos == R:
                    dq.pop()
                R -= 1

            ans += R - L + 1

        return ans
```

```python
from typing import List
from collections import deque

class Solution:
    def countNonDecreasingSubarrays(self, nums: List[int], k: int) -> int:
```
```python
        n = len(nums)
        dq = deque()
        pref_max_sum = 0
        window_sum = 0
        R = n - 1
        ans = 0

        for L in range(n - 1, -1, -1):
            x = nums[L]

            removed = 0
            while dq and dq[0][1] <= x:
                pos, val = dq.popleft()
                nxt = dq[0][0] if dq else R + 1
                removed += val * (nxt - pos)

            nxt = dq[0][0] if dq else R + 1
            pref_max_sum = pref_max_sum - removed + x * (nxt - L)
            dq.appendleft((L, x))
            window_sum += x

            while pref_max_sum - window_sum > k:
                window_sum -= nums[R]
                pos, val = dq[-1]
                pref_max_sum -= val
                if pos == R:
                    dq.pop()
                R -= 1

            ans += R - L + 1

        return ans
```
