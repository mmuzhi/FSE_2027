import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n = data[0]
    a = data[1:1 + n]

    last = {}
    ans = n + 1

    for i, x in enumerate(a, 1):
        if x in last:
            ans = min(ans, i - last[x] + 1)
        last[x] = i

    print(-1 if ans == n + 1 else ans)

if __name__ == "__main__":
    main()
