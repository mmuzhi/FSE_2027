You can perform the following operation on A any number of times (possibly zero):

- First, choose an integer i satisfying 1 \leq i \leq N, and flip the value of A_i (if the original value is 0, change it to 1; if it is 1, change it to 0).

import sys
from bisect import bisect_left

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    it = iter(data)
    N = next(it)
    A = [next(it) for _ in range(N)]
    B = [next(it) for _ in range(N)]
    C = [next(it) for _ in range(N)]
    R=[]; S=[]; T=[]
    for a,b,c in zip(A,B,C):
        if a==1 and b==0:
            R.append(c)
        elif a==0 and b==1:
            S.append(c)
        elif a==1 and b==1:
            T.append(c)
    R_desc = sorted(R, reverse=True)
    S_asc = sorted(S)
    T_asc = sorted(T)
    C_T = sum(T)
    C_R = sum(R)
    r = len(R); s = len(S)
    # F0
    off_total = 0
    if r:
        S0 = C_R + C_T
        off_total = r * S0
        for idx, c in enumerate(R_desc):
            off_total -= (r - idx) * c
    on_total = 0
    if s:
        on_total = s * C_T
        for idx, c in enumerate(S_asc):
            on_total += (s - idx) * c
    F0 = off_total + on_total
    # L for T
    RS_asc = sorted(R + S)
    pref = [0]*(len(RS_asc)+1)
    for i, v in enumerate(RS_asc):
        pref[i+1] = pref[i] + v
    L = []
    for c in T_asc:
        idx = bisect_left(RS_asc, c)
        cnt_less = idx
        sum_less = pref[idx]
        L.append(2*C_T + sum_less - c * (cnt_less + 1))
    t = len(T_asc)
    sufL = [0]*(t+1)
    sufC = [0]*(t+1)
    sufCj = [0]*(t+1)
    for k in range(t-1, -1, -1):
        sufL[k] = sufL[k+1] + L[k]
        sufC[k] = sufC[k+1] + T_asc[k]
        sufCj[k] = sufCj[k+1] + T_asc[k] * k
    best_G = 0
    for k in range(t):
        pair = sufCj[k] - k * sufC[k]
        g = sufL[k] - 2 * pair

import sys
from bisect import bisect_left

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    it = iter(data)

    N = next(it)
    A = [next(it) for _ in range(N)]
    B = [next(it) for _ in range(N)]
    C = [next(it) for _ in range(N)]

    R = []  # 1 -> 0
    S = []  # 0 -> 1
    T = []  # 1 -> 1, can be temporarily turned off/on

    for a, b, c in zip(A, B, C):
        if a == 1 and b == 0:
            R.append(c)
        elif a == 0 and b == 1:
            S.append(c)
        elif a == 1 and b == 1:
            T.append(c)

    R_desc = sorted(R, reverse=True)
    S_asc = sorted(S)
    T_asc = sorted(T)

    C_T = sum(T)
    C_R = sum(R)
    r = len(R)
    s = len(S)

    # Cost when no 1->1 bit is cycled
    off_total = 0
    if r > 0:
        S0 = C_R + C_T
        off_total = r * S0
        for i, c in enumerate(R_desc):
            off_total -= (r - i) * c

    on_total = 0
    if s > 0:
        on_total = s * C_T
        for i, c in enumerate(S_asc):
            on_total += (s - i) * c

    F0 = off_total + on_total

    # L_i for each T bit
    RS_asc = sorted(R + S)
    pref = [0] * (len(RS_asc) + 1)
    for i, v in enumerate(RS_asc):
        pref[i + 1] = pref[i] + v

    L = []
    for c in T_asc:
        idx = bisect_left(RS_asc, c)  # R/S with C < c
        cnt_less = idx
        sum_less = pref[idx]
        L.append(2 * C_T + sum_less - c * (cnt_less + 1))

    # Optimal cycled set is a suffix of T sorted by C.
    t = len(T_asc)
    sufL = [0] * (t + 1)
    sufC = [0] * (t + 1)
    sufCj = [0] * (t + 1)

    for i in range(t - 1, -1, -1):
        sufL[i] = sufL[i + 1] + L[i]
        sufC[i] = sufC[i + 1] + T_asc[i]
        sufCj[i] = sufCj[i + 1] + T_asc[i] * i

    best_G = 0  # empty set
    for k in range(t):
        pair_sum = sufCj[k] - k * sufC[k]
        g = sufL[k] - 2 * pair_sum
        if g < best_G:
            best_G = g

    print(F0 + best_G)

if __name__ == "__main__":
    main()
