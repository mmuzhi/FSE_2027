```python
import sys

def main():
    S = sys.stdin.readline().strip()
    T = sys.stdin.readline().strip()

    if S == T:
        print(0)
        return

    for i in range(min(len(S), len(T))):
        if S[i] != T[i]:
            print(i + 1)
            return

    print(min(len(S), len(T)) + 1)

if __name__ == "__main__":
    main()
```
