```python
import sys

def main():
    S = sys.stdin.readline().strip()
    cnt = [0] * 26
    pos_sum = [0] * 26
    ans = 0

    for idx, ch in enumerate(S, start=1):
        c = ord(ch) - ord('A')
        ans += cnt[c] * (idx - 1) - pos_sum[c]
        cnt[c] += 1
        pos_sum[c] += idx

    print(ans)

if __name__ == "__main__":
    main()
```
