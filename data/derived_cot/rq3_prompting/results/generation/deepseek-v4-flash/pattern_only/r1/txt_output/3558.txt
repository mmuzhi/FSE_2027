```python
from typing import List
from collections import deque

class Solution:
    def findSafeWalk(self, grid: List[List[int]], health: int) -> bool:
        m, n = len(grid), len(grid[0])
        INF = 10**9
        dist = [[INF] * n for _ in range(m)]
        dist[0][0] = grid[0][0]
        dq = deque([(0, 0)])
        dirs = [(1,0),(-1,0),(0,1),(0,-1)]
        while dq:
            i, j = dq.popleft()
            for di, dj in dirs:
                ni, nj = i + di, j + dj
                if 0 <= ni < m and 0 <= nj < n:
                    nd = dist[i][j] + grid[ni][nj]
                    if nd < dist[ni][nj]:
                        dist[ni][nj] = nd
                        if grid[ni][nj] == 0:
                            dq.appendleft((ni, nj))
                        else:
                            dq.append((ni, nj))
        return dist[m-1][n-1] < health
```

```
from collections import deque

class Solution:
    def findSafeWalk(self, grid: List[List[int]], health: int) -> bool:
        m, n = len(grid), len(grid[0])
        INF = 10**9
        dist = [[INF] * n for _ in range(m)]
```

```
        if start_cost >= health:

        dist[0][0] = start_cost
        dq = deque([(start_cost, 0, 0)])
```

```python
        return dist[m - 1][n - 1] < health
```

```python
from typing import List
from collections import deque

class Solution:
    def findSafeWalk(self, grid: List[List[int]], health: int) -> bool:
        m, n = len(grid), len(grid[0])
        INF = 10**9

        dist = [[INF] * n for _ in range(m)]
        start_cost = grid[0][0]

        if start_cost >= health:
            return False

        dist[0][0] = start_cost
        dq = deque([(start_cost, 0, 0)])
        dirs = [(1, 0), (-1, 0), (0, 1), (0, -1)]

        while dq:
            cost, i, j = dq.popleft()

            if cost != dist[i][j]:
                continue

            for di, dj in dirs:
                ni, nj = i + di, j + dj

                if 0 <= ni < m and 0 <= nj < n:
                    new_cost = cost + grid[ni][nj]

                    if new_cost < dist[ni][nj] and new_cost < health:
                        dist[ni][nj] = new_cost

                        if grid[ni][nj] == 0:
                            dq.appendleft((new_cost, ni, nj))
                        else:
                            dq.append((new_cost, ni, nj))

        return dist[m - 1][n - 1] < health
```
