=== ANSWER ===
```python
from typing import List
from math import gcd

class Solution:
    def maxScore(self, nums: List[int]) -> int:
        def lcm(a: int, b: int) -> int:
            return a // gcd(a, b) * b

        def score(arr: List[int]) -> int:
            g = 0
            l = 1
            for x in arr:
                g = gcd(g, x)
                l = lcm(l, x)
            return g * l

        ans = score(nums)
        n = len(nums)

        for i in range(n):
            g = 0
            l = 1
            for j, x in enumerate(nums):
                if j == i:
                    continue
                g = gcd(g, x)
                l = lcm(l, x)
            ans = max(ans, g * l)

        return ans
```
