```python
import sys

def main():
    input = sys.stdin.buffer.readline
    N, M = map(int, input().split())

    INF = 10**18
    minR = [INF] * (M + 2)

    for _ in range(N):
        L, R = map(int, input().split())
        if R < minR[L]:
            minR[L] = R

    bad = 0
    cur_min_R = INF

    for l in range(M, 0, -1):
        if minR[l] < cur_min_R:
            cur_min_R = minR[l]

        if cur_min_R != INF:
            bad += M - cur_min_R + 1

    total = M * (M + 1) // 2
    print(total - bad)

if __name__ == "__main__":
    main()
```
