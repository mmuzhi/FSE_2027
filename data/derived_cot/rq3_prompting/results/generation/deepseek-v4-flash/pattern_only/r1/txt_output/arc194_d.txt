import sys

MOD = 998244353

def main():
    data = sys.stdin.read().split()
    if not data:
        return
    N = int(data[0])
    S = data[1].strip()

    children = [[]]
    stack = [0]
    for ch in S:
        if ch == '(':
            v = len(children)
            children.append([])
            parent = stack[-1]
            children[parent].append(v)
            stack.append(v)
        else:
            stack.pop()

    M = len(children) - 1

    max_deg = max(len(ch) for ch in children)
    fact = [1] * (max_deg + 1)
    for i in range(1, max_deg + 1):
        fact[i] = fact[i-1] * i % MOD
    invfact = [1] * (max_deg + 1)
    invfact[max_deg] = pow(fact[max_deg], MOD - 2, MOD)
    for i in range(max_deg, 0, -1):
        invfact[i-1] = invfact[i] * i % MOD

    type_id = [0] * (M + 1)
    type_map = {}
    ans = 1

    for v in range(M, -1, -1):
        child_types = sorted(type_id[c] for c in children[v])
        key = tuple(child_types)
        if key not in type_map:
            type_map[key] = len(type_map) + 1
        type_id[v] = type_map[key]

        d = len(child_types)
        if d == 0:
            continue
        contrib = fact[d]
        i = 0
        while i < d:
            j = i + 1
            while j < d and child_types[j] == child_types[i]:
                j += 1
            contrib = contrib * invfact[j - i] % MOD
            i = j
        ans = ans * contrib % MOD

    print(ans)

if __name__ == "__main__":
    main()

import sys

MOD = 998244353

def main():
    data = sys.stdin.read().split()
    if not data:
        return

    N = int(data[0])
    S = data[1]

    # Build rooted ordered tree. Node 0 is the implicit root.
    children = [[]]
    stack = [0]

    for ch in S:
        if ch == '(':
            v = len(children)
            children.append([])
            parent = stack[-1]
            children[parent].append(v)
            stack.append(v)
        else:
            stack.pop()

    M = len(children) - 1

    max_deg = max(len(c) for c in children)

    fact = [1] * (max_deg + 1)
    for i in range(1, max_deg + 1):
        fact[i] = fact[i - 1] * i % MOD

    invfact = [1] * (max_deg + 1)
    invfact[max_deg] = pow(fact[max_deg], MOD - 2, MOD)
    for i in range(max_deg, 0, -1):
        invfact[i - 1] = invfact[i] * i % MOD

    # Canonical IDs for unordered rooted subtree isomorphism.
    type_id = [0] * (M + 1)
    type_map = {}

    ans = 1

    # Children have larger IDs, so reverse order is postorder.
    for v in range(M, -1, -1):
        child_types = sorted(type_id[c] for c in children[v])
        key = tuple(child_types)

        if key not in type_map:
            type_map[key] = len(type_map) + 1
        type_id[v] = type_map[key]

        d = len(child_types)
        if d == 0:
            continue

        # Number of ways to order children:
        # d! / product(multiplicity of each unordered child type)!
        contrib = fact[d]
        i = 0
        while i < d:
            j = i + 1
            while j < d and child_types[j] == child_types[i]:
                j += 1
            contrib = contrib * invfact[j - i] % MOD
            i = j

        ans = ans * contrib % MOD

    print(ans)
