import sys

def main():
    N, P = map(int, sys.stdin.readline().split())
    mod = P
    E = N // 2
    D = N * (N - 1) // 2
    S = N + 1
    stride_e = E + 1

    # binom up to N
    C = [[0]*(N+1) for _ in range(N+1)]
    for n in range(N+1):
        C[n][0] = C[n][n] = 1
        for k in range(1, n):
            C[n][k] = (C[n-1][k-1] + C[n-1][k]) % mod

    # factorials for interpolation
    fact = [1]*(D+2)
    for i in range(1, D+2):
        fact[i] = fact[i-1] * i % mod
    invfact = [1]*(D+2)
    invfact[D+1] = pow(fact[D+1], mod-2, mod)
    for i in range(D+1, 0, -1):
        invfact[i-1] = invfact[i] * i % mod

    # Q(x) = prod_{j=0}^D (x-j)
    Q = [1]
    for j in range(D+1):
        negj = (-j) % mod
        new = [0]*(len(Q)+1)
        for k, coef in enumerate(Q):
            if coef:
                new[k] = (new[k] + coef * negj) % mod
                new[k+1] = (new[k+1] + coef) % mod
        Q = new

    denom_inv = [0]*(D+1)
    for i in range(D+1):
        di = invfact[i] * invfact[D-i] % mod
        if (D - i) & 1:
            di = (-di) % mod
        denom_inv[i] = di

    # bit lists for active masks
    bit_lists = [[] for _ in range(1 << (E+1))]
    for mask in range(1 << (E+1)):
        bit_lists[mask] = [i for i in range(E+1) if mask & (1 << i)]

    # base indices
    base_s_a = [[0]*S for _ in range(S)]
    act_base = [[0]*S for _ in range(S)]
    for s in range(S):
        for a in range(S):
            act_base[s][a] = (s * S + a) * 2
            base_s_a[s][a] = act_base[s][a] * stride_e

    dp_len = S * S * 2 * stride_e
    vals = []

    for x in range(D+1):
        base = (1 + x) % mod
        pow_base = [1]*(N+1)
        for a in range(1, N+1):

        u = [0]*(N+1)

        u_pow = [[1]*(N+1) for _ in range(N+1)]

            ua = u[a]
            pw = 1
            for b in range(1, N+1):
                pw = pw * ua % mod
                u_pow[a][b] = pw

        base_comb = [1]*(N+1)
        base_pow = 1
        for b in range(1, N+1):
            base_comb[b] = base_comb[b-1] * base_pow % mod
            base_pow = base_pow * base % mod

        comb_base = [[0]*(N+1) for _ in range(N)]
        for rem in range(1, N):
            cb = comb_base[rem]
            for b in range(1, rem+1):
                cb[b] = C[rem][b] * base_comb[b] % mod

        trans = [[None]*(N+1) for _ in range(N)]
        for rem in range(1, N):
            max_a = N - rem
            cb = comb_base[rem]
            for a in range(1, max_a+1):
                arr = [0]*(rem+1)
                up_a = u_pow[a]
                for b in range(1, rem+1):
                    arr[b] = cb[b] * up_a[b] % mod
                trans[rem][a] = arr

        active = [0]*(S*S*2)
        dp[base_s_a[1][1] + 1] = 1  # p=0, e=1
        active[act_base[1][1]] = 1 << 1

        for s in range(1, N):

            for a in range(1, s+1):

                for p in range(2):
                    mask = active[act_base[s][a] + p]
                    if not mask:
                        continue
                    np = 1 - p
                    dst_parity_offset = np * stride_e
                    src_base = base_s_a[s][a] + p * stride_e
                    if np == 0:

                            if val == 0:

                            max_b = rem

                                    ns = s + b

        for a in range(1, N+1):

            for p in range(2):

        factor = vals[i] * denom_inv[i] % mod
        if factor:

                if r[k]:
                    coeffs[k] = (coeffs[k] + factor * r[k]) % mod

    out = ' '.join(str(coeffs[m] % mod) for m in range(N-1, D+1))
    sys.stdout.write(out + '\n')

import sys

def main():
    N, P = map(int, sys.stdin.readline().split())
    mod = P
    E = N // 2
    D = N * (N - 1) // 2
    S = N + 1
    stride_e = E + 1

    # binomial coefficients modulo P for N
    C = [[0] * (N + 1) for _ in range(N + 1)]
    for n in range(N + 1):
        C[n][0] = C[n][n] = 1
        for k in range(1, n):
            C[n][k] = (C[n - 1][k - 1] + C[n - 1][k]) % mod

    # factorials for interpolation
    fact = [1] * (D + 2)
    for i in range(1, D + 2):
        fact[i] = fact[i - 1] * i % mod
    invfact = [1] * (D + 2)
    invfact[D + 1] = pow(fact[D + 1], mod - 2, mod)
    for i in range(D + 1, 0, -1):
        invfact[i - 1] = invfact[i] * i % mod

    # Q(x) = product_{j=0}^D (x - j)
    Q = [1]
    for j in range(D + 1):
        negj = (-j) % mod
        nxt = [0] * (len(Q) + 1)
        for k, coef in enumerate(Q):
            if coef:
                nxt[k] = (nxt[k] + coef * negj) % mod
                nxt[k + 1] = (nxt[k + 1] + coef) % mod
        Q = nxt

    denom_inv = [0] * (D + 1)
    for i in range(D + 1):
        v = invfact[i] * invfact[D - i] % mod
        if (D - i) & 1:
            v = (-v) % mod
        denom_inv[i] = v

    # bit lists for active even-count masks
    bit_lists = [[] for _ in range(1 << (E + 1))]
    for mask in range(1 << (E + 1)):
        bit_lists[mask] = [i for i in range(E + 1) if mask & (1 << i)]

    # flattened indices
    base_s_a = [[0] * S for _ in range(S)]
    act_base = [[0] * S for _ in range(S)]
    for s in range(S):
        for a in range(S):
            act_base[s][a] = (s * S + a) * 2
            base_s_a[s][a] = act_base[s][a] * stride_e

    dp_len = S * S * 2 * stride_e
    vals = []

    for x in range(D + 1):
        base = (1 + x) % mod

        pow_base = [1] * (N + 1)
        for a in range(1, N + 1):
            pow_base[a] = pow_base[a - 1] * base % mod

        u = [0] * (N + 1)
        for a in range(1, N + 1):
            u[a] = (pow_base[a] - 1) % mod

        u_pow = [[1] * (N + 1) for _ in range(N + 1)]
        for a in range(1, N + 1):
            pw = 1
            for b in range(1, N + 1):
                pw = pw * u[a] % mod
                u_pow[a][b] = pw

        base_comb = [1] * (N + 1)
        base_pow = 1
        for b in range(1, N + 1):
            base_comb[b] = base_comb[b - 1] * base_pow % mod
            base_pow = base_pow * base % mod

        comb_base = [[0] * (N + 1) for _ in range(N)]
        for rem in range(1, N):
            cb = comb_base[rem]
            for b in range(1, rem + 1):
                cb[b] = C[rem][b] * base_comb[b] % mod

        trans = [[None] * (N + 1) for _ in range(N)]
        for rem in range(1, N):
            max_a = N - rem
            cb = comb_base[rem]
            for a in range(1, max_a + 1):
                arr = [0] * (rem + 1)
                up = u_pow[a]
                for b in range(1, rem + 1):
                    arr[b] = cb[b] * up[b] % mod
                trans[rem][a] = arr

        dp = [0] * dp_len
        active = [0] * (S * S * 2)

        # root layer: size 1, even layer, even-count = 1
        dp[base_s_a[1][1] + 1] = 1
        active[act_base[1][1]] = 1 << 1

        for s in range(1, N):
            rem = N - s
            for a in range(1, s + 1):
                trans_a = trans[rem][a]
                for p in range(2):
                    mask = active[act_base[s][a] + p]
                    if not mask:
                        continue

                    np = 1 - p
                    dst_parity_offset = np * stride_e
                    src_base = base_s_a[s][a] + p * stride_e

                    if np == 0:
                        # new layer is even: even-count increases by b
                        for e in bit_lists[mask]:
                            val = dp[src_base + e]
                            if val == 0:
                                continue
                            max_b = rem
                            lim = E - e
                            if max_b > lim:
                                max_b = lim
                            for b in range(1, max_b + 1):
                                coef = trans_a[b]
                                if coef:
                                    add = val * coef % mod
                                    ns = s + b
                                    idx = base_s_a[ns][b] + dst_parity_offset + (e + b)
                                    if dp[idx] == 0:
                                        active[act_base[ns][b] + np] |= 1 << (e + b)
                                    nv = dp[idx] + add
                                    if nv >= mod:
                                        nv -= mod
                                    dp[idx] = nv
                    else:
                        # new layer is odd: even-count unchanged
                        for e in bit_lists[mask]:
                            val = dp[src_base + e]
                            if val == 0:
                                continue
                            for b in range(1, rem + 1):
                                coef = trans_a[b]
                                if coef:
                                    add = val * coef % mod
                                    ns = s + b
                                    idx = base_s_a[ns][b] + dst_parity_offset + e
                                    if dp[idx] == 0:
                                        active[act_base[ns][b] + np] |= 1 << e
                                    nv = dp[idx] + add
                                    if nv >= mod:
                                        nv -= mod
                                    dp[idx] = nv

        fx = 0
        for a in range(1, N + 1):
            base_n = base_s_a[N][a]
            for p in range(2):
                fx += dp[base_n + p * stride_e + E]
                if fx >= mod:
                    fx -= mod
        vals.append(fx)

    # interpolate coefficients from values at x = 0,1,...,D
    coeffs = [0] * (D + 1)
    for i in range(D + 1):
        r = [0] * (D + 1)
        r[D] = Q[D + 1]
        for k in range(D - 1, -1, -1):
            r[k] = (Q[k + 1] + i * r[k + 1]) % mod

        factor = vals[i] * denom_inv[i] % mod
        if factor:
            for k in range(D + 1):
                if r[k]:
                    coeffs[k] = (coeffs[k] + factor * r[k]) % mod

    ans = ' '.join(str(coeffs[m] % mod) for m in range(N - 1, D + 1))
    sys.stdout.write(ans + '\n')


if __name__ == "__main__":
    main()
