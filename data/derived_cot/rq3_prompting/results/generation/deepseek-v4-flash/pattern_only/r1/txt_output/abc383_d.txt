```python
import sys
import math
from bisect import bisect_right

def sieve(n):
    if n < 2:
        return []
    is_prime = bytearray(b'\x01') * (n + 1)
    is_prime[0] = 0
    is_prime[1] = 0
    r = math.isqrt(n)
    for i in range(2, r + 1):
        if is_prime[i]:
            start = i * i
            is_prime[start:n + 1:i] = b'\x00' * (((n - start) // i) + 1)
    return [i for i in range(2, n + 1) if is_prime[i]]

def main():
    data = sys.stdin.buffer.read().strip()
    if not data:
        return
    N = int(data)
    M = math.isqrt(N)
    primes = sieve(M)

    ans = 0
    for p in primes:
        if p ** 8 > N:
            break
        ans += 1

    limit = math.isqrt(M)
    for p in primes:
        if p > limit:
            break
        q_max = M // p
        if q_max > p:
            ans += bisect_right(primes, q_max) - bisect_right(primes, p)

    print(ans)

if __name__ == "__main__":
    main()
```

```python
import sys
import math
from bisect import bisect_right


def sieve(n):
    if n < 2:
        return []
    is_prime = bytearray(b'\x01') * (n + 1)
    is_prime[0] = 0
    is_prime[1] = 0

    r = math.isqrt(n)
    for i in range(2, r + 1):
        if is_prime[i]:
            start = i * i
            is_prime[start:n + 1:i] = b'\x00' * (((n - start) // i) + 1)

    return [i for i in range(2, n + 1) if is_prime[i]]


def main():
    data = sys.stdin.buffer.read().strip()
    if not data:
        return

    N = int(data)
    M = math.isqrt(N)
    primes = sieve(M)

    ans = 0

    # Form: p^8
    for p in primes:
        if p ** 8 > N:
            break
        ans += 1

    # Form: p^2 * q^2, where p and q are distinct primes and p*q <= sqrt(N)
    limit = math.isqrt(M)
    for p in primes:
        if p > limit:
            break
        q_max = M // p
        if q_max > p:
            ans += bisect_right(primes, q_max) - bisect_right(primes, p)

    print(ans)


if __name__ == "__main__":
    main()
```
