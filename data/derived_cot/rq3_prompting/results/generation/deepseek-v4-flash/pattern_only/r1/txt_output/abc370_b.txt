```python
import sys

def main():
    data = list(map(int, sys.stdin.read().split()))
    N = data[0]

    A = [[0] * (N + 1) for _ in range(N + 1)]

    idx = 1
    for i in range(1, N + 1):
        for j in range(1, i + 1):
            A[i][j] = data[idx]
            idx += 1

    cur = 1
    for j in range(1, N + 1):
        if cur >= j:
            cur = A[cur][j]
        else:
            cur = A[j][cur]

    print(cur)

if __name__ == "__main__":
    main()
```
