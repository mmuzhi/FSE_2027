```python
import sys

def main():
    data = sys.stdin.read().split()
    if not data:
        return

    it = iter(data)
    N = int(next(it))
    M = int(next(it))

    has_male = [False] * (N + 1)
    ans = []

    for _ in range(M):
        a = int(next(it))
        b = next(it)

        if b == 'M' and not has_male[a]:
            has_male[a] = True
            ans.append("Yes")
        else:
            ans.append("No")

    print("\n".join(ans))

if __name__ == "__main__":
    main()
```
