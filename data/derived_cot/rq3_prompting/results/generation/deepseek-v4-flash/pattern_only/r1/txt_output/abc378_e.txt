import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    N, M = data[0], data[1]
    A = data[2:2 + N]

    bit = [0] * (M + 1)

    def add(pos, val):
        i = pos + 1
        while i <= M:
            bit[i] += val
            i += i & -i

    def prefix_sum(pos):
        s = 0
        i = pos + 1
        while i > 0:
            s += bit[i]
            i -= i & -i
        return s

    # Prefix residue 0 is initially present.
    add(0, 1)

    pref = 0
    total_sum = 0
    ans = 0

    for r, a in enumerate(A, 1):
        pref = (pref + a) % M

        leq = prefix_sum(pref)
        ans += r * pref - total_sum + M * (r - leq)

        add(pref, 1)
        total_sum += pref

    print(ans)

if __name__ == "__main__":
    main()
