import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return

    N = int(data[0])
    X = int(data[1])
    Y = int(data[2])
    S = data[3]
    T = data[4]

    # If the block length is larger than N, no operation can be performed.
    if X + Y > N:
        sys.stdout.write("Yes\n" if S == T else "No\n")
        return

    ONE = ord('1')

    # Store residues of positions of '1' modulo X and '0' modulo Y in S.
    ones_s = []
    zeros_s = []
    for i, c in enumerate(S):
        if c == ONE:
            ones_s.append(i % X)
        else:
            zeros_s.append(i % Y)

    oi = 0
    zi = 0
    lo = len(ones_s)
    lz = len(zeros_s)
    ok = True

    # Check the same residues in T.
    for i, c in enumerate(T):
        if c == ONE:
            if oi >= lo or (i % X) != ones_s[oi]:
                ok = False
                break
            oi += 1
        else:
            if zi >= lz or (i % Y) != zeros_s[zi]:
                ok = False
                break
            zi += 1

    if ok and oi == lo and zi == lz:
        sys.stdout.write("Yes\n")
    else:
        sys.stdout.write("No\n")

if __name__ == "__main__":
    solve()
