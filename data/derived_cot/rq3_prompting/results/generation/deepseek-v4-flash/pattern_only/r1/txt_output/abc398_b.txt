```python
import sys
from collections import Counter

def main():
    A = list(map(int, sys.stdin.read().split()))
    cnt = Counter(A)
    counts = sorted(cnt.values(), reverse=True)

    if len(counts) >= 2 and counts[0] >= 3 and counts[1] >= 2:
        print("Yes")
    else:
        print("No")

if __name__ == "__main__":
    main()
```
