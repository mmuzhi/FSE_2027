class Solution:
    def smallestNumber(self, n: int, t: int) -> int:
        def digit_product_divisible(x: int) -> bool:
            product = 1
            for ch in str(x):
                product *= int(ch)
            return product % t == 0

        ans = n
        while True:
            if digit_product_divisible(ans):
                return ans
            ans += 1
