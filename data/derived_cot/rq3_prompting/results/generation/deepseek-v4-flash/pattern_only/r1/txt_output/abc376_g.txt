import sys
from heapq import heappush, heappop, heapify

MOD = 998244353

class Entry:
    __slots__ = ('w','s','id')
    def __init__(self, w, s, id):
        self.w = w
        self.s = s
        self.id = id
    def __lt__(self, other):
        left = self.w * other.s
        right = other.w * self.s
        if left != right:
            return left > right
        return self.id < other.id

def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    it = iter(data)
    T = next(it)
    out = []
    for _ in range(T):
        N = next(it)

class Entry:
    __slots__ = ("w", "s", "id")

    def __init__(self, w, s, id):
        self.w = w
        self.s = s
        self.id = id

    def __lt__(self, other):
        left = self.w * other.s
        right = other.w * self.s
        if left != right:
            return left > right
