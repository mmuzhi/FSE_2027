import sys, bisect
def main():
    input = sys.stdin.buffer.readline
    N,W = map(int,input().split())
    cols=[[] for _ in range(W)]
    for i in range(1,N+1):
        x,y=map(int,input().split())
        cols[x-1].append((y,i))
    R = min(len(cols[i]) for i in range(W))  # W>=1
    ymax = [0]*R
    rank = [0]*(N+1)
    for c in range(W):
        lst = cols[c]
        lst.sort()
        # assign ranks
        # If R>0, update ymax for first R
        # We can loop enumerate
        for idx, (y,id) in enumerate(lst, start=1):
            rank[id] = idx
            if idx <= R and y > ymax[idx-1]:
                ymax[idx-1] = y
    tau=[0]
    for r in range(1,R+1):
        m = ymax[r-1] - 1
        tau.append(max(tau[-1], m) + 1)
    Q=int(input())
    out=[]
    for _ in range(Q):
        T,A=map(int,input().split())
        k = bisect.bisect_right(tau, T) - 1
        if rank[A] > k:
            out.append("Yes")
        else:
            out.append("No")
    print("\n".join(out))

```python
import sys
from bisect import bisect_right

def main():
    input = sys.stdin.buffer.readline
    N, W = map(int, input().split())
    cols = [[] for _ in range(W)]
    for i in range(1, N+1):
        x, y = map(int, input().split())
        cols[x-1].append((y, i))

    R = min(len(c) for c in cols)
    ymax = [0] * R
    rank = [0] * (N + 1)

    for c in range(W):
        cols[c].sort()
        for idx, (y, id_) in enumerate(cols[c], 1):
            rank[id_] = idx
            if idx <= R and y > ymax[idx-1]:
                ymax[idx-1] = y

    tau = [0]
    for r in range(1, R+1):
        m = ymax[r-1] - 1
        tau.append(max(tau[-1], m) + 1)

    Q = int(input())
    out = []
    for _ in range(Q):
        T, A = map(int, input().split())
        k = bisect_right(tau, T) - 1
        out.append("Yes" if rank[A] > k else "No")
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    main()
```

```python
import sys
from bisect import bisect_right

def main():
    input = sys.stdin.buffer.readline

    N, W = map(int, input().split())
    cols = [[] for _ in range(W)]

    for i in range(1, N + 1):
        x, y = map(int, input().split())
        cols[x - 1].append((y, i))

    R = min(len(c) for c in cols)

    ymax = [0] * R
    rank = [0] * (N + 1)

    for c in range(W):
        cols[c].sort()
        for idx, (y, block_id) in enumerate(cols[c], 1):
            rank[block_id] = idx
            if idx <= R and y > ymax[idx - 1]:
                ymax[idx - 1] = y

    # tau[r] = time when the r-th removal happens
    tau = [0]
    for r in range(1, R + 1):
        need = ymax[r - 1] - 1
        tau.append(max(tau[-1], need) + 1)

    Q = int(input())
    ans = []

    for _ in range(Q):
        T, A = map(int, input().split())
        removed = bisect_right(tau, T) - 1
        if rank[A] > removed:
            ans.append("Yes")
        else:
            ans.append("No")

    print("\n".join(ans))

if __name__ == "__main__":
    main()
```
