```python
from typing import List
from collections import Counter

class Solution:
    def getLargestOutlier(self, nums: List[int]) -> int:
        total = sum(nums)
        cnt = Counter(nums)
        ans = -10**18
        for x in nums:
            rem = total - x
            if rem % 2 != 0:
                continue
            s = rem // 2
            if cnt[s] > (1 if s == x else 0):
                ans = max(ans, x)
        return ans
```

```
    def getLargestOutlier(self, nums: List[int]) -> int:
```
