=== ANSWER ===
```python
from typing import List
from itertools import permutations

class Solution:
    def totalNumbers(self, digits: List[int]) -> int:
        result = set()
        
        for a, b, c in permutations(digits, 3):
            if a != 0 and c % 2 == 0:
                result.add(a * 100 + b * 10 + c)
                
        return len(result)
```
