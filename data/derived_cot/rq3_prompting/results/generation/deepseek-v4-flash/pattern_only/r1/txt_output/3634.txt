=== ANSWER ===
```python
class Solution:
    def calculateScore(self, s: str) -> int:
        stacks = [[] for _ in range(26)]
        score = 0
        base = ord('a')

        for i, ch in enumerate(s):
            idx = ord(ch) - base
            mirror = 25 - idx

            if stacks[mirror]:
                score += i - stacks[mirror].pop()
            else:
                stacks[idx].append(i)

        return score
```
