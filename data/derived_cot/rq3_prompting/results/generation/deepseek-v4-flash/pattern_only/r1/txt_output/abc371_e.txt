=== ANSWER ===
```python
import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    N = data[0]
    A = data[1:]

    last = [0] * (N + 1)
    ans = 0

    for i, a in enumerate(A, start=1):
        prev = last[a]
        ans += (i - prev) * (N - i + 1)
        last[a] = i

    print(ans)

if __name__ == "__main__":
    main()
```
