```
def main():
    import sys
    input = sys.stdin.buffer.readline
    N = int(input())
    A = [input().strip() for _ in range(N)]
    B = [bytearray(N) for _ in range(N)]
    last = N - 1
    rng = range(N)
    for r in rng:
        row = B[r]
        lr = last - r
        for c in rng:
            a = r
            if lr < a:
                a = lr
            if c < a:
                a = c
            lc = last - c
            if lc < a:
                a = lc
            k = (a + 1) & 3
            if k == 0:
                row[c] = A[r][c]
            elif k == 1:
                row[c] = A[lc][r]
            elif k == 2:
                row[c] = A[lr][lc]
            else:
                row[c] = A[c][lr]
    sys.stdout.buffer.write(b'\n'.join(B))
```

```
        bottom = N - 1 - t
        left = t
```

```
def main():
    input = sys.stdin.buffer.readline
    N = int(input())

    for t in range(N // 2):

        bottom = N - 1 - t
        left = t
        right = N - 1 - t
        L = bottom - top + 1

        idx = L

            idx += 1
```

```
        idx += L - 1

        for r in range(bottom - 1, top, -1):
            S[idx] = A[r][left]
            idx += 1

        k = (t + 1) & 3
        shift = k * (L - 1)
        if shift:
            S = S[-shift:] + S[:-shift]

        B[top][left:right + 1] = S[0:L]

            idx += 1

            idx += 1

if __name__ == "__main__":
```

```python
import sys

def main():
    input = sys.stdin.buffer.readline
    N = int(input())
    A = [input().strip() for _ in range(N)]
    B = [bytearray(N) for _ in range(N)]

    for t in range(N // 2):
        top = t
        bottom = N - 1 - t
        left = t
        right = N - 1 - t
        L = bottom - top + 1

        # Build the ring sequence in clockwise order:
        # top row, right column, bottom row reversed, left column reversed.
        S = bytearray(4 * L - 4)
        S[0:L] = A[top][left:right + 1]

        idx = L
        for r in range(top + 1, bottom + 1):
            S[idx] = A[r][right]
            idx += 1

        S[idx:idx + L - 1] = A[bottom][left:right][::-1]
        idx += L - 1

        for r in range(bottom - 1, top, -1):
            S[idx] = A[r][left]
            idx += 1

        # This layer is rotated d = t + 1 times by 90 degrees clockwise.
        k = (t + 1) & 3
        shift = k * (L - 1)
        if shift:
            S = S[-shift:] + S[:-shift]

        # Write the rotated ring back to the output grid.
        B[top][left:right + 1] = S[0:L]

        idx = L
        for r in range(top + 1, bottom + 1):
            B[r][right] = S[idx]
            idx += 1

        B[bottom][left:right] = S[2 * L - 1:3 * L - 2][::-1]

        idx = 3 * L - 2
        for r in range(bottom - 1, top, -1):
            B[r][left] = S[idx]
            idx += 1

    sys.stdout.buffer.write(b'\n'.join(B))

if __name__ == "__main__":
    main()
```
