=== ANSWER ===
```python
import sys

def main():
    X = int(sys.stdin.readline().strip())
    total = 2025
    count = 0
    for i in range(1, 10):
        for j in range(1, 10):
            if i * j == X:
                count += 1
    print(total - X * count)

if __name__ == "__main__":
    main()
```
