You will be given a competitive programming problem.
The AtCoder company office is represented as a grid of H rows and W columns. Let (i, j) denote the cell at the i-th row from the top and j-th column from the left.
The state of each cell is represented by a character S_{i,j}. If S_{i,j} is #, that cell has a wall; if S_{i,j} is ., that cell is a floor; if S_{i,j} is H, that cell has a humidifier placed on a floor cell.

Find the number of humidified floor cells.

Input

The input is given from Standard Input in the following format:

Print the answer.

- 1 \leq W \leq 1000
- 0 \leq D \leq H\times W
- S_{i,j} is #, ., or H. (1 \leq i \leq H, 1 \leq j \leq W)
- All input numbers are integers.

import sys

def main():
    input = sys.stdin.buffer.readline
    H, W, D = map(int, input().split())
    grid = [input().strip() for _ in range(H)]

```python
    WALL = ord('#')
    HUMID = ord('H')

    visited = bytearray(H * W)
    q = []

    for r, row in enumerate(grid):
        base = r * W
        for c, ch in enumerate(row):
            if ch == HUMID:
                idx = base + c
                visited[idx] = 1
                q.append(idx)

    if not q or D == 0:
        print(len(q))
        return

    head = 0

    for _ in range(D):
        if head == len(q):
            break

        end = len(q)

        while head < end:
            idx = q[head]
            head += 1

            r = idx // W
            c = idx - r * W

            if r > 0:
                ni = idx - W
                if not visited[ni] and grid[r - 1][c] != WALL:
                    visited[ni] = 1
                    q.append(ni)

            if r + 1 < H:
                ni = idx + W
                if not visited[ni] and grid[r + 1][c] != WALL:
                    visited[ni] = 1
                    q.append(ni)

            if c > 0:
                ni = idx - 1
                if not visited[ni] and grid[r][c - 1] != WALL:
                    visited[ni] = 1
                    q.append(ni)

            if c + 1 < W:
                ni = idx + 1
                if not visited[ni] and grid[r][c + 1] != WALL:
                    visited[ni] = 1
                    q.append(ni)

    print(len(q))

if __name__ == "__main__":
    main()
```
