class Solution:
    def hasSpecialSubstring(self, s: str, k: int) -> bool:
        n = len(s)
        for i in range(n - k + 1):
            ch = s[i]
            if all(s[j] == ch for j in range(i, i + k)):

```python
class Solution:
    def hasSpecialSubstring(self, s: str, k: int) -> bool:
        n = len(s)

        for i in range(n - k + 1):
            ch = s[i]

            if s[i:i + k] != ch * k:
                continue

            before_ok = (i == 0) or (s[i - 1] != ch)
            after_ok = (i + k == n) or (s[i + k] != ch)

            if before_ok and after_ok:
                return True

        return False
```
