=== ANSWER ===
```python
import sys

def main():
    data = sys.stdin.read().splitlines()
    N = int(data[0])
    strings = data[1:1+N]
    strings.sort(key=len)
    print(''.join(strings))

if __name__ == "__main__":
    main()
```
