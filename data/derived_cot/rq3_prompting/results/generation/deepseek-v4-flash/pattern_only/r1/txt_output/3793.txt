```python
class Solution:
    def longestPalindrome(self, s: str, t: str) -> int:
        def all_substrings(x: str):
            res = [""]
            for i in range(len(x)):
                for j in range(i + 1, len(x) + 1):
                    res.append(x[i:j])
            return res

        ss = all_substrings(s)
        tt = all_substrings(t)

        ans = 0
        for a in ss:
            for b in tt:
                c = a + b
                if c == c[::-1]:
                    ans = max(ans, len(c))

        return ans
```
