class Solution:
    def longestPalindrome(self, s: str, t: str) -> int:
        n, m = len(s), len(t)
        subs_s = [""]
        for i in range(n):
            for j in range(i + 1, n + 1):
                subs_s.append(s[i:j])
        subs_t = [""]

        best = 0

        return best

class Solution:
    def longestPalindrome(self, s: str, t: str) -> int:
        n, m = len(s), len(t)
        subs_s = {""}
        for i in range(n):
            for j in range(i + 1, n + 1):
                subs_s.add(s[i:j])
        subs_t = {""}
        for i in range(m):
