def solve():
    import sys
    sys.setrecursionlimit(10000)
    data = sys.stdin.buffer.read().split()
    n, m, l = map(int, data[:3])
    vals = [int(x) for x in data[3:3+n+m+l]]
    nn = n + m + l
    v = vals
    # masks
    maskT0 = (1 << n) - 1
    maskA0 = ((1 << m) - 1) << n
    full = (1 << nn) - 1
    
    less = [0]*nn
    eqless = [0]*nn
    for i in range(nn):
        lm = 0
        for j in range(nn):
            if v[j] < v[i]:
                lm |= (1 << j)
            elif v[j] == v[i] and j < i:
                eqless[i] |= (1 << j)
        less[i] = lm
    
    # p3sum[mask] = sum of 3^i for bits i in mask
    p3 = [3**i for i in range(nn)]
    p3sum = [0]*(1 << nn)
    for mask in range(1, 1 << nn):
        low = mask & (-mask)
        p3sum[mask] = p3sum[mask ^ low] + p3[low.bit_length()-1]
    
    memo = bytearray(3**nn)  # 0 unknown, 1 win, 2 lose
    
    # recursive
    def f(h, o):
        # returns True if mover (hand h) wins; opponent hand o
        if h == 0:
            return False
        id_ = p3sum[h] + 2*p3sum[o]
        r = memo[id_]
        if r:
            return r == 1
        t = full & ~h & ~o
        # explore moves
        res = 2  # assume lose
        pm = h
        while pm:
            b = pm & (-pm)
            pm ^= b
            i = b.bit_length() - 1
            if h & eqless[i]:
                continue  # duplicate value in hand; skip
            nh = h ^ b  # hand after playing card i
            # option: no take -> child (mover=o, opp=nh)
            cid = p3sum[o] + 2*p3sum[nh]
            cr = memo[cid]
            if cr == 0:
                # need recursion: child mover is opponent with hand o, opp hand nh
                win_child = f(o, nh)
                cr = 1 if win_child else 2
            if cr == 2:
                res = 1
                break
            # takes
            tm = t & less[i]
            while tm:
                tb = tm & (-tm)
                tm ^= tb
                j = tb.bit_length() - 1
                if t & eqless[j]:
                    continue
                nh2 = nh | tb
                cid2 = p3sum[o] + 2*p3sum[nh2]
                cr2 = memo[cid2]
                if cr2 == 0:
                    win_child2 = f(o, nh2)
                    cr2 = 1 if win_child2 else 2
                if cr2 == 2:
                    res = 1
                    break
            if res == 1:
                break
        memo[id_] = res
        return res == 1
    
    ans = f(maskT0, maskA0)
    print("Takahashi" if ans else "Aoki")
