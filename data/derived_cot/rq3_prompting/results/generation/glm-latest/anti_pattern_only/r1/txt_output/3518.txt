```python
class Solution:
    def maxScore(self, a: List[int], b: List[int]) -> int:
        a0, a1, a2, a3 = a
        NEG = -(1 << 62)
        # d_k = best score picking k elements (matched with a[0..k-1]) so far
        d1 = d2 = d3 = d4 = NEG
        for x in b:
            # update in descending order so each x is used at most once per level
            d4 = max(d4, d3 + a3 * x)
            d3 = max(d3, d2 + a2 * x)
            d2 = max(d2, d1 + a1 * x)
            d1 = max(d1, a0 * x)
        return d4
```
