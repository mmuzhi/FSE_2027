An array a is lexicographically smaller than an array b if in the first position where a and b differ, array a has an element that is less than the corresponding element in b.
If the first min(a.length, b.length) elements do not differ, then the shorter array is the lexicographically smaller one.

```python
from bisect import bisect_left
from typing import List

class Solution:
    def maximumWeight(self, intervals: List[List[int]]) -> List[int]:
        n = len(intervals)
        # Sort interval indices by right endpoint (ties by left endpoint, then index - any consistent order works)
        order = sorted(range(n), key=lambda i: (intervals[i][1], intervals[i][0], i))
        
        R = [intervals[i][1] for i in order]
        L = [intervals[i][0] for i in order]
        Wt = [intervals[i][2] for i in order]
        Idx = order
        
        # dp[k][i] = (best_weight, best_tuple) using exactly k intervals among first i (sorted), or None
        dp = [None] * 5
        dp[0] = [(0, ())] * (n + 1)
        for k in range(1, 5):
            dp[k] = [None] * (n + 1)
        
        for i in range(1, n + 1):
            li = L[i - 1]
            p = bisect_left(R, li)
            w = Wt[i - 1]
            idx = Idx[i - 1]
            for k in range(1, 5):
                cur = dp[k][i - 1]
                base = dp[k - 1][p]
                if base is not None:
                    bw, bt = base
                    nw = bw + w
                    # insert idx into sorted tuple bt (length k-1 <= 3)
                    pos = bisect_left(bt, idx)
                    nt = bt[:pos] + (idx,) + bt[pos:]
                    if cur is None or nw > cur[0] or (nw == cur[0] and nt < cur[1]):
                        cur = (nw, nt)
                dp[k][i] = cur
        
        best_w = -1
        best_t = None
        for k in range(5):
            v = dp[k][n]
            if v is None:
                continue
            wv, tv = v
            if wv > best_w or (wv == best_w and (best_t is None or tv < best_t)):
                best_w = wv
                best_t = tv
        
        return list(best_t) if best_t is not None else []
    ```

from bisect import bisect_left
from typing import List

class Solution:
    def maximumWeight(self, intervals: List[List[int]]) -> List[int]:
        n = len(intervals)
        
        # Sort interval indices by right endpoint.
        order = sorted(range(n), key=lambda i: intervals[i][1])
        R = [intervals[i][1] for i in order]
        L = [intervals[i][0] for i in order]
        W = [intervals[i][2] for i in order]
        Idx = order
        
        # P[j]: number of intervals (in sorted order) with right endpoint < L[j],
        # i.e., the prefix length of intervals that can precede sorted-interval j.
        P = [bisect_left(R, L[j]) for j in range(n)]
        
        # dpw[k][i]: max total weight using exactly k intervals among the first i
        #            (sorted by right endpoint); -1 if impossible.
        # dpt[k][i]: lexicographically smallest sorted tuple of original indices
        #            achieving dpw[k][i].
        dpw = [[-1] * (n + 1) for _ in range(5)]
        dpt = [[None] * (n + 1) for _ in range(5)]
        dpw[0] = [0] * (n + 1)
        dpt[0] = [()] * (n + 1)
        
        for k in range(1, 5):
            wk = dpw[k]
            tk = dpt[k]
            wprev = dpw[k - 1]
            tprev = dpt[k - 1]
            for i in range(1, n + 1):
                j = i - 1
                cw = wk[i - 1]
                ct = tk[i - 1]
                bw = wprev[P[j]]
                if bw >= 0:
                    nw = bw + W[j]
                    bt = tprev[P[j]]
                    idx = Idx[j]
                    if k == 1:
                        nt = (idx,)
                    elif k == 2:
                        a = bt[0]
                        nt = (idx, a) if idx < a else (a, idx)
                    else:
                        pos = bisect_left(bt, idx)
                        nt = bt[:pos] + (idx,) + bt[pos:]
                    if cw < 0 or nw > cw or (nw == cw and nt < ct):
                        cw = nw
                        ct = nt
                wk[i] = cw
                tk[i] = ct
        
        best_w = -1
        best_t = None
        for k in range(5):
            wv = dpw[k][n]
            if wv < 0:
                continue
            tv = dpt[k][n]
            if wv > best_w or (wv == best_w and (best_t is None or tv < best_t)):
                best_w = wv
                best_t = tv
        
        return list(best_t)
