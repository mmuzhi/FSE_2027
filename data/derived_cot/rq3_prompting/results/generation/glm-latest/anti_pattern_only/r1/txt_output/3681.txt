    def maxRectangleArea(self, points: List[List[int]]) -> int:
        n = len(points)
        best = -1
        
        for combo in combinations(range(n), 4):

            ys = set(p[1] for p in corners)

            if len(xs) != 2 or len(ys) != 2:
                continue

            y1, y2 = sorted(ys)

            valid = True

                if j in combo:
                    continue
                px, py = points[j]
                if x1 <= px <= x2 and y1 <= py <= y2:
                    valid = False
                    break
