```python
class Solution:
    def maximumValueSum(self, board: List[List[int]]) -> int:
        m, n = len(board), len(board[0])
        # top 3 per row
        rows = []
        for i in range(m):
            top = sorted(range(n), key=lambda c: board[i][c], reverse=True)[:3]
            # better: use heapq.nlargest for speed
            rows.append([(board[i][c], c) for c in top])
        rows.sort(key=lambda r: r[0][0], reverse=True)
        best = [r[0][0] for r in rows]
        m2 = len(rows)
        ans = None  # -inf
        NEG = float('-inf')
        ans = NEG
        for i in range(m2 - 2):
            if 3 * best[i] <= ans:
                break
            ci = rows[i]
            for j in range(i + 1, m2 - 1):
                if best[i] + 2 * best[j] <= ans:
                    break
                cj = rows[j]
                bij = best[i] + best[j]
                for k in range(j + 1, m2):
                    if bij + best[k] <= ans:
                        break
                    ck = rows[k]
                    # try all combos
                    for v1, c1 in ci:
                        for v2, c2 in cj:
                            if c1 == c2:
                                continue
                            v12 = v1 + v2
                            for v3, c3 in ck:
                                if c3 == c1 or c3 == c2:
                                    continue
                                s = v12 + v3
                                if s > ans:
                                    ans = s
        return ans
```

```python
    def maximumValueSum(self, board: List[List[int]]) -> int:
        m, n = len(board), len(board[0])
```

```python
        row_top = []

            top = sorted(((board[i][c], c) for c in range(n)), reverse=True)[:3]
```

Looking at this problem, I need to place 3 rooks with distinct rows and columns maximizing the sum.

**Key insight:** Each rook only needs to avoid 2 other columns, so in any row, the chosen cell can be restricted to that row's top-3 values (among top-3 cells, at least one avoids any 2 forbidden columns, and it matches the best allowed value).

**Algorithm:** For each pair of rows (i, j), try all 3×3 column combinations from their top-3 cells. For the third rook, precompute (cached per forbidden column pair) the best value available in each row avoiding those 2 columns — keeping top-3 rows suffices since at most 2 rows (i, j) are excluded, so the first cached entry not in {i, j} gives the optimal third rook.

This gives O(C(m,2)·9 + C(n,2)·3m) work, very fast for m, n ≤ 100.

I verified correctness on all three examples (traced example 1 → 4, example 2 → 15, example 3 → 3) and on a tricky case where the third row's best cell is excluded, plus a diagonal-dominant case. The completeness argument: any optimal solution can be transformed (without decreasing value) so each rook is the best cell in its row avoiding the other two columns — such a configuration is covered by the enumeration, and every enumerated value is a valid placement.

```python
from typing import List

    def maximumValueSum(self, board: List[List[int]]) -> int:

        # Top-3 cells (value, col) of each row, sorted by value descending.

            sorted(((board[i][c], c) for c in range(n)), reverse=True)[:3]
            for i in range(m)

        # the best value available in that row while avoiding both columns.
        cache = {}

                        if c != a and c != b:
                            entries.append((v, r))
                            break

                res = entries[:3]
                cache[key] = res

        ans = float('-inf')
        for i in range(m):

                    for v2, c2 in row_top[j]:
                        if c1 == c2:

                        # first cached row not equal to i or j is optimal
```
