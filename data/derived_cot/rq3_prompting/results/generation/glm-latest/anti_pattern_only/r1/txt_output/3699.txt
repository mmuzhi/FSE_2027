```python
class Solution:
    def numberOfSubsequences(self, nums: List[int]) -> int:
        n = len(nums)
        cnt = defaultdict(int)
        ans = 0
        for r in range(n):
            q = r - 2

            if q >= 2:
                nq = nums[q]
```

```python
from math import gcd
from typing import List

class Solution:
    def numberOfSubsequences(self, nums: List[int]) -> int:
        n = len(nums)
        cnt = {}
        ans = 0
        _gcd = gcd
        for r in range(n):
            q = r - 2
            if q >= 2:
                b = nums[q]
                for p in range(q - 1):
                    a = nums[p]
                    g = _gcd(a, b)
                    k = (a // g) * 1024 + (b // g)
                    if k in cnt:
                        cnt[k] += 1
                    else:
                        cnt[k] = 1
            b = nums[r]
            for s in range(r + 2, n):
                a = nums[s]
                g = _gcd(a, b)
                k = (a // g) * 1024 + (b // g)
                ans += cnt.get(k, 0)
        return ans
```

=== ANSWER ===
Looking at this problem:

- We need quadruples (p, q, r, s) with gaps ≥ 2 between consecutive indices, where `nums[p] * nums[r] == nums[q] * nums[s]`, i.e. `nums[p]/nums[q] == nums[s]/nums[r]`.
- Key insight: a **left pair** (p, q) with `q - p ≥ 2` only needs to satisfy `q ≤ r - 2` to combine with any **right pair** (r, s) with `s - r ≥ 2`. So iterate `r` from left to right, maintaining a counter of reduced fractions `nums[p]/nums[q]` over all left pairs with `q ≤ r - 2` (when `r` increases, add the new pairs with `q = r - 2`). Then for each `s ≥ r + 2`, add the count of the matching fraction `nums[s]/nums[r]`.

This is O(n²) total (~10⁶ operations for n = 1000). Fractions are encoded as `(num//g) << 10 | (den//g)` with `g = gcd`, stored in a flat array for speed.

I verified this against both examples (outputs 1 and 3) and several hand-crafted cases including cross-value fraction matches (e.g., 4/2 matching 6/3).

```python
from math import gcd
from typing import List


class Solution:
    def numberOfSubsequences(self, nums: List[int]) -> int:
        n = len(nums)
        cnt = [0] * (1 << 20)  # key = (num // g) << 10 | (den // g), values <= 1000
        _gcd = gcd
        ans = 0
        for r in range(n):
            # Add all left pairs (p, q) with q = r - 2 (so q <= r - 2), p <= q - 2.
            # Store reduced fraction nums[p] / nums[q].
            q = r - 2
            if q >= 2:
                b = nums[q]
                for a in nums[:q - 1]:
                    g = _gcd(a, b)
                    cnt[((a // g) << 10) | (b // g)] += 1
            # For each right pair (r, s) with s >= r + 2, count left pairs whose
            # fraction nums[p]/nums[q] equals nums[s]/nums[r]
            # (equivalent to nums[p]*nums[r] == nums[q]*nums[s]).
            b = nums[r]
            for a in nums[r + 2:]:
                g = _gcd(a, b)
                ans += cnt[((a // g) << 10) | (b // g)]
        return ans
```
