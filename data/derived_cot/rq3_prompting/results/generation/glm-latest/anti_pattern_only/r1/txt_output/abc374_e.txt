The manufacturing of a certain product requires N processes numbered 1,2,\dots,N.
For each process i, there are two types of machines S_i and T_i available for purchase to handle it.

import sys

def solve():
    input_data = sys.stdin.buffer.read().split()
    n = int(input_data[0]); X = int(input_data[1])
    machines = []
    idx = 2
    for i in range(n):
        a = int(input_data[idx]); p = int(input_data[idx+1]); b = int(input_data[idx+2]); q = int(input_data[idx+3])
        idx += 4
        machines.append((a, p, b, q))
    
    def min_cost(a, p, b, q, w):
        if w <= 0:
            return 0
        best = float('inf')
        # x small
        for x in range(0, 101):
            rem = w - x * a
            if rem <= 0:
                cost = x * p
            else:
                y = (rem + b - 1) // b
                cost = x * p + y * q
            if cost < best:
                best = cost
            if rem <= 0:
                break  # further x only increases cost
        for y in range(0, 101):
            rem = w - y * b
            if rem <= 0:
                cost = y * q
            else:
                x = (rem + a - 1) // a
                cost = x * p + y * q
            if cost < best:
                best = cost
            if rem <= 0:
                break
        return best
    
    def feasible(w):
        total = 0
        for (a, p, b, q) in machines:
            total += min_cost(a, p, b, q, w)
            if total > X:
                return False
        return total <= X
    
    lo, hi = 0, 100 * X + 1
    while hi - lo > 1:
        mid = (lo + hi) // 2
        if feasible(mid):
            lo = mid
        else:
            hi = mid
    print(lo)

import sys
from math import gcd

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0]); X = int(data[1])
    procs = []
    idx = 2
    for _ in range(n):
        a = int(data[idx]); p = int(data[idx+1])
        b = int(data[idx+2]); q = int(data[idx+3])
        idx += 4
        g = gcd(a, b)
        procs.append((a, p, b, q, b // g, a // g))

    INF = float('inf')

    def min_cost(a, p, b, q, bx, by, w):
        best = INF
        for x in range(bx):
            rem = w - x * a
            if rem <= 0:
                c = x * p
                return c if c < best else best
            c = x * p + ((rem + b - 1) // b) * q
            if c < best:
                best = c
        for y in range(by):
            rem = w - y * b
            if rem <= 0:
                c = y * q
                return c if c < best else best
            c = y * q + ((rem + a - 1) // a) * p
            if c < best:
                best = c
        return best

    def feasible(w):
        total = 0
        for (a, p, b, q, bx, by) in procs:
            total += min_cost(a, p, b, q, bx, by, w)
            if total > X:
                return False
        return True

    lo, hi = 0, 100 * X + 1
    while hi - lo > 1:
        mid = (lo + hi) // 2
        if feasible(mid):
            lo = mid
        else:
            hi = mid
    print(lo)

main()

import sys
from math import gcd

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0]); X = int(data[1])
    procs = []
    idx = 2
    for _ in range(n):
        a = int(data[idx]); p = int(data[idx + 1])
        b = int(data[idx + 2]); q = int(data[idx + 3])
        idx += 4
        g = gcd(a, b)
        procs.append((a, p, b, q, b // g, a // g))

    INF = float('inf')

    def feasible(w):
        total = 0
        for a, p, b, q, bx, by in procs:
            best = INF
            # enumerate count of S machines x in [0, bx), fill with T
            for x in range(bx):
                rem = w - x * a
                if rem <= 0:
                    c = x * p
                    if c < best:
                        best = c
                    break
                c = x * p + ((rem + b - 1) // b) * q
                if c < best:
                    best = c
            # enumerate count of T machines y in [0, by), fill with S
            for y in range(by):
                rem = w - y * b
                if rem <= 0:
                    c = y * q
                    if c < best:
                        best = c
                    break
                c = y * q + ((rem + a - 1) // a) * p
                if c < best:
                    best = c
            total += best
            if total > X:
                return False
        return True

    lo, hi = 0, 100 * X + 1
    while hi - lo > 1:
        mid = (lo + hi) // 2
        if feasible(mid):
            lo = mid
        else:
            hi = mid
    print(lo)

main()

Looking at this problem:

- **Binary search on the answer W** (monotone feasibility: if W is achievable, any smaller value is too).
- **For each process**, given W, compute the minimum cost to buy machines (x of S, y of T) with x·A + y·B ≥ W. This is an unbounded integer knapsack.
- **Key bound:** Let g = gcd(A,B), L = lcm(A,B). Swapping (remove L/A = B/g units of S, add L/B = A/g units of T) preserves throughput exactly. Depending on which direction is cheaper, there always exists an optimal solution with x < B/g ≤ 100 or y < A/g ≤ 100. So enumerating x ∈ [0, B/g) with y = ceil((W−xA)/B), and symmetrically y ∈ [0, A/g), guarantees finding the optimum (≤ 200 candidates per process).

Additionally, once x·A ≥ W (pure S suffices), larger x only cost more, so we can break early — provably safe since any optimal solution then has x ≤ that point (otherwise pure-S would be cheaper, contradicting optimality).

**Bounds:** cost per unit capacity ≥ 1/100 yen (P≥1, A≤100), so answer ≤ 100X ≤ 10⁹. Binary search on [0, 100X+1].
