import sys
import numpy as np
from collections import Counter

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    A = list(map(int, data[1:1+n]))
    B = list(map(int, data[1+n:1+2*n]))
    FA = [x for x in A if x != -1]
    FB = [x for x in B if x != -1]
    nA, nB = len(FA), len(FB)
    K = nA + nB - n
    if K <= 0:
        print("Yes")
        return
    maxA = max(FA); maxB = max(FB)
    L = max(maxA, maxB)
    cntA = Counter(FA); cntB = Counter(FB)
    X = np.array(sorted(cntA.keys()), dtype=np.int64)
    cx = np.array([cntA[x] for x in X], dtype=np.int64)
    V = np.array(sorted(cntB.keys()), dtype=np.int64)
    cv = np.array([cntB[v] for v in V], dtype=np.int64)
    # sums matrix
    S = X[:, None] + V[None, :]
    W = np.minimum(cx[:, None], cv[None, :])
    S = S.ravel()
    W = W.ravel()
    mask = S >= L
    S = S[mask]
    W = W[mask]
    ans = False
    if S.size:
        # aggregate weights by S
        order = np.argsort(S, kind='stable')
        Ss = S[order]
        Ws = W[order]
        # boundaries
        boundary = np.empty(Ss.size, dtype=bool)
        boundary[0] = True
        np.not_equal(Ss[1:], Ss[:-1], out=boundary[1:])
        idx = np.flatnonzero(boundary)
        # segment sums
        csum = np.concatenate(([0], np.cumsum(Ws)))
        totals = csum[idx[1:]] - csum[idx[:-1]] if idx.size > 1 else np.array([csum[-1]])
        # hmm handle single segment
        if idx.size == 1:
            totals = np.array([csum[-1] - csum[0]])
        if (totals >= K).any():
            ans = True
    print("Yes" if ans else "No")

main()
