You will be given a competitive programming problem.
You are given a sequence A of non-negative integers of length N, and an integer K. It is guaranteed that the binomial coefficient \dbinom{N}{K} is at most 10^6.
When choosing K distinct elements from A, find the maximum possible value of the XOR of the K chosen elements.
That is, find \underset{1\leq i_1\lt i_2\lt \ldots\lt i_K\leq N}{\max} A_{i_1}\oplus A_{i_2}\oplus \ldots \oplus A_{i_K}.

About XOR

For non-negative integers A,B, the XOR A \oplus B is defined as follows:


- In the binary representation of A \oplus B, the bit corresponding to 2^k (k \ge 0) is 1 if and only if exactly one of the bits corresponding to 2^k in A and B is 1, and is 0 otherwise.


For example, 3 \oplus 5 = 6 (in binary notation: 011 \oplus 101 = 110).
In general, the XOR of K integers p_1, \dots, p_k is defined as (\cdots((p_1 \oplus p_2) \oplus p_3) \oplus \cdots \oplus p_k). It can be proved that it does not depend on the order of p_1, \dots, p_k.

Input

The input is given from Standard Input in the following format:
N K
A_1 A_2 \ldots A_N

Output

Print the answer.

Constraints


- 1\leq K\leq N\leq 2\times 10^5
- 0\leq A_i<2^{60}
- \dbinom{N}{K}\leq 10^6
- All input values are integers.

Sample Input 1

4 2
3 2 6 4

Sample Output 1

7

Here are six ways to choose two distinct elements from (3,2,6,4).

- (3,2): The XOR is 3\oplus 2 = 1.
- (3,6): The XOR is 3\oplus 6 = 5.
- (3,4): The XOR is 3\oplus 4 = 7.
- (2,6): The XOR is 2\oplus 6 = 4.
- (2,4): The XOR is 2\oplus 4 = 6.
- (6,4): The XOR is 6\oplus 4 = 2.

Hence, the maximum possible value is 7.

```python
import sys
import numpy as np

def main():
    data = sys.stdin.buffer.read().split()
    N = int(data[0])
    K = int(data[1])
    A = np.array(list(map(int, data[2:2 + N])), dtype=np.int64)
    S = int(np.bitwise_xor.reduce(A))

    m = min(K, N - K)

    # Enumerate XOR values of all m-subsets (by index) of A.
    last = np.full(1, -1, dtype=np.int64)   # max index in current subset, -1 for empty
    vals = np.zeros(1, dtype=np.int64)      # XOR of current subset
    for _ in range(m):
        counts = (N - 1) - last             # number of possible next indices
        total = int(counts.sum())
        cum = np.cumsum(counts)
        starts = np.concatenate(([0], cum))  # starts[i] = start position of row i's children
        row_of_child = np.repeat(np.arange(len(counts), dtype=np.int64), counts)
        idx_within = np.arange(total, dtype=np.int64) - starts[row_of_child]
        new_last = last[row_of_child] + 1 + idx_within
        new_vals = vals[row_of_child] ^ A[new_last]
        last = new_last
        vals = new_vals

    if K <= N - K:
        ans = int(vals.max())
    else:
        ans = int((vals ^ S).max())
    print(ans)

main()
```

```python
import sys

def solve_numpy(N, K, A_list, S, m, direct):
    import numpy as np
    A = np.array(A_list, dtype=np.int64)
    last = np.full(1, -1, dtype=np.int64)   # largest index in subset; -1 = empty
    vals = np.zeros(1, dtype=np.int64)      # XOR of subset
    for _ in range(m):
        counts = (N - 1) - last             # how many larger indices can be appended
        total = int(counts.sum())
        starts = np.concatenate((np.zeros(1, dtype=np.int64), np.cumsum(counts)))
        row = np.repeat(np.arange(len(counts), dtype=np.int64), counts)
        idx = np.arange(total, dtype=np.int64) - starts[row]
        new_last = last[row] + idx + 1
        new_vals = vals[row] ^ A[new_last]
        last, vals = new_last, new_vals
    if direct:
        return int(vals.max())
    return int((vals ^ S).max())

def solve_pure(N, K, A_list, S, m, direct):
    from itertools import combinations
    from functools import reduce
    from operator import xor
    if direct:
        return max(reduce(xor, comb) for comb in combinations(A_list, m))
    return max(S ^ reduce(xor, comb) for comb in combinations(A_list, m))

def main():
    data = sys.stdin.buffer.read().split()
    N = int(data[0]); K = int(data[1])
    A_list = list(map(int, data[2:2 + N]))
    S = 0
    for a in A_list:
        S ^= a
    if K == N:
        print(S)
        return
    m = min(K, N - K)
    direct = (K <= N - K)
    try:
        ans = solve_numpy(N, K, A_list, S, m, direct)
    except ImportError:
        ans = solve_pure(N, K, A_list, S, m, direct)
    print(ans)

main()
```

```python
import sys


def solve_numpy(N, A_list, S, m, direct):
    import numpy as np
    A = np.array(A_list, dtype=np.int64)
    # states: subsets of current size, represented by (last index, xor value)
    last = np.full(1, -1, dtype=np.int64)
    vals = np.zeros(1, dtype=np.int64)
    for _ in range(m):
        counts = (N - 1) - last                # number of appendable larger indices
        total = int(counts.sum())
        starts = np.concatenate((np.zeros(1, dtype=np.int64), np.cumsum(counts)))
        row = np.repeat(np.arange(len(counts), dtype=np.int64), counts)
        idx = np.arange(total, dtype=np.int64) - starts[row]
        new_last = last[row] + idx + 1
        new_vals = vals[row] ^ A[new_last]
        last, vals = new_last, new_vals
    if direct:
        return int(vals.max())
    return int((vals ^ S).max())


def solve_pure(A_list, S, m, direct):
    from itertools import combinations
    from functools import reduce
    from operator import xor
    if direct:
        return max(reduce(xor, comb) for comb in combinations(A_list, m))
    return max(S ^ reduce(xor, comb) for comb in combinations(A_list, m))


def main():
    data = sys.stdin.buffer.read().split()
    N = int(data[0])
    K = int(data[1])
    A_list = list(map(int, data[2:2 + N]))

    S = 0
    for a in A_list:
        S ^= a

    if K == N:
        print(S)
        return

    m = min(K, N - K)          # m >= 1, m <= N/2, C(N, m) = C(N, K) <= 10**6
    direct = (K <= N - K)      # if not direct, chosen = all XOR excluded, i.e. S ^ xor(excluded)

    try:
        ans = solve_numpy(N, A_list, S, m, direct)
    except ImportError:
        ans = solve_pure(A_list, S, m, direct)
    print(ans)


main()
```
