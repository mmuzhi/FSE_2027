```python
class Solution:
    def minimumPossibleSum(self, n: int, target: int) -> int:
        MOD = 10**9 + 7
        S = target // 2
        if n <= S:
            return (n * (n + 1) // 2) % MOD
        r = n - S
        total = S * (S + 1) // 2 + r * target + r * (r - 1) // 2
        return total % MOD
```

```python
class Solution:
    def minimumPossibleSum(self, n: int, target: int) -> int:
        MOD = 10**9 + 7
        # Numbers 1..S can all be taken safely (S = target // 2),
        # since each x <= S has complement target - x > S (or is target/2 itself).
        S = target // 2
        if n <= S:
            # Just take 1..n
            return (n * (n + 1) // 2) % MOD
        # Take 1..S, then the remaining r = n - S numbers must be >= target
        # (at most S numbers can lie in [1, target-1] due to pairing constraint),
        # so take target, target+1, ..., target+r-1.
        r = n - S
        total = S * (S + 1) // 2 + r * target + r * (r - 1) // 2
        return total % MOD
```
