You will be given a competitive programming problem.
You are given a directed graph with N vertices and M edges.
The i-th edge (1 \leq i \leq M) is a directed edge from vertex u _ i to vertex v _ i.
Initially, you are at vertex 1. You want to repeat the following operations until you reach vertex N:

- Perform one of the two operations below:
- Move along a directed edge from your current vertex. This incurs a cost of 1. More precisely, if you are at vertex v, choose a vertex u such that there is a directed edge from v to u, and move to vertex u.
- Reverse the direction of all edges. This incurs a cost of X. More precisely, if and only if there was a directed edge from v to u immediately before this operation, there is a directed edge from u to v immediately after this operation.



It is guaranteed that, for the given graph, you can reach vertex N from vertex 1 by repeating these operations.

    import heapq
    adj = [[] for _ in range(2*N)]
    for u, v in edges:
        u -= 1; v -= 1
        adj[u].append(v)
        adj[N+v].append(N+u)
    INF = float('inf')
    dist = [INF]*(2*N)
    dist[0] = 0
    h = [(0, 0)]
    push = heapq.heappush; pop = heapq.heappop
    t0 = N-1; t1 = 2*N-1
    while h:
        d, s = pop(h)

        nd = d + X
        if nd < dist[f]:
            dist[f] = nd
            push(h, (nd, f))

import sys

def main():
    data = sys.stdin.buffer.read().split()
    N = int(data[0]); M = int(data[1]); X = int(data[2])

    # ---- Fast path: scipy Dijkstra on the 2-layer graph ----
    try:
        import numpy as np
        from scipy.sparse import csr_matrix
        from scipy.sparse.csgraph import dijkstra

        arr = np.fromiter(map(int, data[3:3 + 2 * M]), dtype=np.int64, count=2 * M)
        us = arr[0::2] - 1
        vs = arr[1::2] - 1
        codes = np.unique(us * N + vs)          # deduplicate edges
        us = codes // N
        vs = codes % N
        E = codes.size

        idx = np.arange(2 * N, dtype=np.int64)
        rows = np.concatenate([us, vs + N, idx])
        cols = np.concatenate([vs, us + N, (idx + N) % (2 * N)])
        datas = np.concatenate([
            np.ones(2 * E, dtype=np.float64),
            np.full(2 * N, float(X), dtype=np.float64),
        ])

        g = csr_matrix((datas, (rows, cols)), shape=(2 * N, 2 * N))
        dist = dijkstra(g, directed=True, indices=0,
                        return_predecessors=False)
        ans = min(dist[N - 1], dist[2 * N - 1])
        print(int(ans) if ans != float('inf') else -1)
        return
    except Exception:
        pass

    # ---- Fallback: pure-Python Dijkstra ----
    import heapq
    adj = [[] for _ in range(2 * N)]
    ptr = 3
    for _ in range(M):
        u = int(data[ptr]) - 1
        v = int(data[ptr + 1]) - 1
        ptr += 2
        adj[u].append(v)             # layer 0: u -> v, cost 1
        adj[N + v].append(N + u)     # layer 1: v -> u, cost 1

    INF = float('inf')
    dist = [INF] * (2 * N)
    dist[0] = 0
    heap = [(0, 0)]
    push = heapq.heappush
    pop = heapq.heappop
    t0 = N - 1        # vertex N, layer 0
    t1 = 2 * N - 1    # vertex N, layer 1

    while heap:
        d, s = pop(heap)
        if d > dist[s]:
            continue
        if s == t0 or s == t1:
            print(d)
            return
        nd = d + 1
        for t in adj[s]:
            if nd < dist[t]:
                dist[t] = nd
                push(heap, (nd, t))
        f = s + N if s < N else s - N     # flip layers at same vertex
        nd = d + X
        if nd < dist[f]:
            dist[f] = nd
            push(heap, (nd, f))

    ans = min(dist[t0], dist[t1])
    print(-1 if ans == INF else ans)

main()

Looking at this problem:

- Model states as (vertex, orientation) — 2N states total. Moving costs 1 within a layer (layer 0 = original edges, layer 1 = reversed edges), flipping orientation at the same vertex costs X.
- Run Dijkstra from (vertex 1, layer 0); answer is min distance to (vertex N, layer 0) or (vertex N, layer 1).

For speed I use scipy's C Dijkstra on the doubled graph (deduplicating input edges so COO→CSR summation of duplicates can't corrupt weights), with a pure-Python heapq Dijkstra fallback if scipy is unavailable. All distances fit exactly in float64 (< 2^53). I verified this model against all four samples (e.g., sample 3 needs 7 flips + 7 moves = 7X+7 = 4294967299; sample 4 gives 21 via 1→3→5, flip, 5→8, 8→14(rev), flip, 14→18, flip, 18→20).

import sys


def main():
    data = sys.stdin.buffer.read().split()
    N = int(data[0])
    M = int(data[1])
    X = int(data[2])

    # ---------- Fast path: scipy sparse Dijkstra on the doubled graph ----------
    try:
        import numpy as np
        from scipy.sparse import csr_matrix
        from scipy.sparse.csgraph import dijkstra

        arr = np.fromiter(map(int, data[3:3 + 2 * M]),
                          dtype=np.int64, count=2 * M)
        us = arr[0::2] - 1
        vs = arr[1::2] - 1
        codes = np.unique(us * N + vs)  # remove duplicate edges
        us = codes // N
        vs = codes % N
        E = codes.size

        idx = np.arange(2 * N, dtype=np.int64)
        rows = np.concatenate([us, vs + N, idx])
        cols = np.concatenate([vs, us + N, (idx + N) % (2 * N)])
        datas = np.concatenate([
            np.ones(2 * E, dtype=np.float64),
            np.full(2 * N, float(X), dtype=np.float64),
        ])

        g = csr_matrix((datas, (rows, cols)), shape=(2 * N, 2 * N))
        dist = dijkstra(g, directed=True, indices=0,
                        return_predecessors=False)
        ans = min(dist[N - 1], dist[2 * N - 1])
        if ans == float('inf'):
            print(-1)
        else:
            print(int(ans))
        return
    except Exception:
        pass

    # ---------- Fallback: pure-Python Dijkstra ----------
    import heapq
    adj = [[] for _ in range(2 * N)]
    ptr = 3
    for _ in range(M):
        u = int(data[ptr]) - 1
        v = int(data[ptr + 1]) - 1
        ptr += 2
        adj[u].append(v)           # layer 0: u -> v, cost 1
        adj[N + v].append(N + u)   # layer 1: v -> u, cost 1

    dist[0] = 0

    pop = heapq.heappop

            return
        nd = d + 1
        for t in adj[s]:
            if nd < dist[t]:
                dist[t] = nd
                push(heap, (nd, t))
        f = s + N if s < N else s - N   # flip orientation, cost X
        nd = d + X
        if nd < dist[f]:
            dist[f] = nd
            push(heap, (nd, f))

    ans = min(dist[t0], dist[t1])
