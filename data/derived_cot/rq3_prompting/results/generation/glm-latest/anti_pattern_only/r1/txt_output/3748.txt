class Solution:
    def sortMatrix(self, grid: List[List[int]]) -> List[List[int]]:
        n = len(grid)
        for d in range(-n + 1, n):
            # Collect diagonal values where i - j == d
            i, j = max(0, d), max(0, d) - d
            vals = []
            while i < n and j < n:
                vals.append(grid[i][j])
                i += 1
                j += 1
            # Sort: non-increasing for bottom-left (d >= 0), non-decreasing for top-right (d < 0)
            vals.sort(reverse=(d >= 0))
            # Write back
            i, j = max(0, d), max(0, d) - d
            for v in vals:
                grid[i][j] = v
                i += 1
                j += 1
        return grid

class Solution:
    def sortMatrix(self, grid: List[List[int]]) -> List[List[int]]:
        n = len(grid)
        for d in range(-n + 1, n):
            # Collect values on the diagonal where i - j == d
            i, j = max(0, d), max(0, d) - d
            vals = []
            while i < n and j < n:
                vals.append(grid[i][j])
                i += 1
                j += 1
            # Bottom-left triangle (d >= 0, incl. main diagonal): non-increasing
            # Top-right triangle (d < 0): non-decreasing
            vals.sort(reverse=(d >= 0))
            # Write sorted values back along the diagonal
            i, j = max(0, d), max(0, d) - d
            for v in vals:
                grid[i][j] = v
                i += 1
                j += 1
        return grid
