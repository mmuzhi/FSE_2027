```python
from typing import List
from bisect import bisect_left

class Solution:
    def maxDistance(self, side: int, points: List[List[int]], k: int) -> int:
        s = side
        C = 4 * s
        n = len(points)
        
        # perimeter parametrization (clockwise from (0,0))
        ts = []
        ap = ts.append
        for x, y in points:
            if y == 0:
                ap(x)
            elif x == s:
                ap(s + y)
            elif y == s:
                ap(3 * s - x)
            else:
                ap(4 * s - y)
        ts.sort()
        
        m = 2 * n
        T = ts + [t + C for t in ts]
        BIG = 1 << 62
        Te = T + [BIG]  # extended with sentinel at index m
        
        def check(d: int) -> bool:
            # nxt[i]: smallest index j with T[j] >= T[i] + d (m if none)
            nxt = [bisect_left(T, t + d) for t in T]
            nxt.append(m)  # sentinel self-loop
            # X[i] = index after (k-1) jumps starting from i (m if runs out)
            X = list(range(n))
            p = nxt
            rem = k - 1
            while rem:
                if rem & 1:
                    X = [p[z] for z in X]
                rem >>= 1
                if rem:
                    p = [p[z] for z in p]
            limit = C - d
            for i in range(n):
                if Te[X[i]] - T[i] <= limit:
                    return True
            return False
        
        lo, hi = 1, C // k
        while lo < hi:
            mid = (lo + hi + 1) // 2
            if check(mid):
                lo = mid
            else:
                hi = mid - 1
        return lo
```

```python
from bisect import bisect_left

class Solution:
    def maxDistance(self, side: int, points: List[List[int]], k: int) -> int:
        s = side
        C = 4 * s
        n = len(points)
        
        # Perimeter position (clockwise from (0,0)):
        # bottom (y=0): t = x ; right (x=s): t = s + y ; top (y=s): t = 3s - x ; left (x=0): t = 4s - y
        ts = []
        for x, y in points:
            if y == 0:
                ts.append(x)
            elif x == s:
                ts.append(s + y)
            elif y == s:
                ts.append(3 * s - x)
            else:
                ts.append(4 * s - y)
        ts.sort()
        
        m = 2 * n
        T = ts + [t + C for t in ts]
        Te = T + [1 << 62]          # extended, sentinel value at index m
        
        def check(d: int) -> bool:
            # nxt[i] = smallest j (in doubled array) with T[j] >= T[i] + d ; m if none
            nxt1 = [bisect_left(ts, t + d) for t in ts]
            nxt = nxt1 + [n + b for b in nxt1]
            nxt.append(m)
            for i in range(n):
                if nxt1[i] == n:
                    nxt[i] = n + bisect_left(ts, ts[i] + d - C)
            
            # X[i] = index reached after (k-1) greedy jumps from i
            X = list(range(n))
            p = nxt
            rem = k - 1
            while rem:
                if rem & 1:
                    X = [p[z] for z in X]
                rem >>= 1
                if rem:
                    p = [p[z] for z in p]
            
            limit = C - d
            for i in range(n):
                if Te[X[i]] - T[i] <= limit:
                    return True
            return False
        
        lo, hi = 1, C // k
        while lo < hi:
            mid = (lo + hi + 1) >> 1
            if check(mid):
                lo = mid
            else:
                hi = mid - 1
        return lo
```
