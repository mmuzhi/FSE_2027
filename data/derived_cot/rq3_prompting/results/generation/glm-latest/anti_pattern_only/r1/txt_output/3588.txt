```python
class Solution:
    def countWinningSequences(self, s: str) -> int:
        MOD = 10**9 + 7
        n = len(s)
        
        # score delta for Bob playing b against Alice's a
        # outcome[a][b]
        # a='F': b='W' -> +1, b='E' -> -1, b='F' -> 0
        # a='W': b='E' -> +1, b='F' -> -1, b='W' -> 0
        # a='E': b='F' -> +1, b='W' -> -1, b='E' -> 0
        
        # creatures indexed 0=F, 1=W, 2=E
        # delta[a][b]
        delta = {
            'F': {'W': 1, 'E': -1, 'F': 0},
            'W': {'E': 1, 'F': -1, 'W': 0},
            'E': {'F': 1, 'W': -1, 'E': 0},
        }
        
        # dp[c][d] : number of sequences ending in creature c with difference d
        # Use dictionaries
        # creatures: 0,1,2
        
        # First round
        a = s[0]
        dp = [dict(), dict(), dict()]
        for c in 'FWE':
            d = delta[a][c]
            ci = 'FWE'.index(c)
            dp[ci][d] = dp[ci].get(d, 0) + 1
        
        for i in range(1, n):
            a = s[i]
            ndp = [dict(), dict(), dict()]
            for prev_c in range(3):
                for d, cnt in dp[prev_c].items():
                    for c in 'FWE':
                        ci = 'FWE'.index(c)
                        if ci == prev_c:
                            continue
                        nd = d + delta[a][c]
                        ndp[ci][nd] = (ndp[ci].get(nd, 0) + cnt) % MOD
            dp = ndp
        
        # Sum over all states with d > 0
        ans = 0
        for c in range(3):
            for d, cnt in dp[c].items():
                if d > 0:
                    ans = (ans + cnt) % MOD
        return ans % MOD
```

```python
class Solution:
    def countWinningSequences(self, s: str) -> int:
        MOD = 10**9 + 7
        n = len(s)
        
        # delta[a][b]: points for Bob when Alice plays a and Bob plays b
        # F beats E, W beats F, E beats W
        delta = {
            'F': {'F': 0, 'W': 1, 'E': -1},
            'W': {'F': -1, 'W': 0, 'E': 1},
            'E': {'F': 1, 'W': -1, 'E': 0},
        }
        
        creatures = 'FWE'
        
        # dp[c]: dict mapping score difference -> count, for sequences ending with creature c
        a0 = s[0]
        dp = [dict(), dict(), dict()]
        for ci, c in enumerate(creatures):
            dp[ci][delta[a0][c]] = 1
        
        for i in range(1, n):
            a = s[i]
            da = delta[a]
            ndp = [dict(), dict(), dict()]
            for prev in range(3):
                if not dp[prev]:
                    continue
                for d, cnt in dp[prev].items():
                    for ci in range(3):
                        if ci == prev:
                            continue
                        nd = d + da[creatures[ci]]
                        ndp_ci = ndp[ci]
                        ndp_ci[nd] = (ndp_ci.get(nd, 0) + cnt) % MOD
            dp = ndp
        
        ans = 0
        for c in range(3):
            for d, cnt in dp[c].items():
                if d > 0:
                    ans = (ans + cnt) % MOD
        return ans % MOD
```

```python
class Solution:
    def countWinningSequences(self, s: str) -> int:
        MOD = 10**9 + 7
        n = len(s)
        # delta[a] = (dF, dW, dE): Bob's score delta vs Alice's move a

            'F': (0, 1, -1),
            'W': (-1, 0, 1),
            'E': (1, -1, 0),

            # target E (2): from F and W

            sFE = [(a + b) % MOD for a, b in zip(f, e)]
            sFW = [(a + b) % MOD for a, b in zip(f, w)]
            ndp = [None, None, None]

                ndp[0] = sWE[:]
```

```python
    def countWinningSequences(self, s: str) -> int:
        MOD = 10**9 + 7
        n = len(s)
        # delta[a] = (dF, dW, dE): Bob's score delta vs Alice's move a
        delta = {
            'F': (0, 1, -1),
            'W': (-1, 0, 1),
            'E': (1, -1, 0),
        }
        # Round 0: window covers scores [-1, 1], arrays of length 3, score s at index s+1
        d0 = delta[s[0]]
        dp = [
            [0, 1, 0],  # F: delta d0[0] at position d0[0]+1
            [0, 0, 0],
            [0, 0, 0],
        ]
        dp[0][d0[0] + 1] = 1
        dp[1][d0[1] + 1] = 1
        dp[2][d0[2] + 1] = 1

            da = delta[s[i]]
            f, w, e = dp
            # sources: target F <- W+E, target W <- F+E, target E <- F+W
            sWE = [(a + b) % MOD for a, b in zip(w, e)]
            sFE = [(a + b) % MOD for a, b in zip(f, e)]

                    return src + [0, 0]

        ans = 0
        for c in range(3):
```

```python
class Solution:
    def countWinningSequences(self, s: str) -> int:
        MOD = 10**9 + 7
        n = len(s)
        
        # Bob's score delta vs Alice's move, indexed by Bob's creature (F, W, E):
        # Alice 'F': F->0, W->+1, E->-1
        # Alice 'W': F->-1, W->0, E->+1
        # Alice 'E': F->+1, W->-1, E->0
        
        # dp arrays: dp[0]=ending F, dp[1]=ending W, dp[2]=ending E.
        # After processing k rounds, arrays have length 2k+1 covering score diffs [-(k... 
        # after round 0: length 3, score s at index s+1.
        a0 = s[0]
        if a0 == 'F':
            d = (0, 1, -1)
        elif a0 == 'W':
            d = (-1, 0, 1)
        else:
            d = (1, -1, 0)
        dp = [[0, 0, 0] for _ in range(3)]
        for c in range(3):
            dp[c][d[c] + 1] = 1
        
        for ch in s[1:]:
            f, w, e = dp
            # pairwise source sums (all creatures except the target)
            sWE = [(a + b) % MOD for a, b in zip(w, e)]  # -> target F
            sFE = [(a + b) % MOD for a, b in zip(f, e)]  # -> target W
            sFW = [(a + b) % MOD for a, b in zip(f, w)]  # -> target E
            if ch == 'F':
                # shifts: F:0, W:+1, E:-1
                dp = [[0] + sWE + [0],
                      [0, 0] + sFE,
                      sFW + [0, 0]]
            elif ch == 'W':
                # shifts: F:-1, W:0, E:+1
                dp = [sWE + [0, 0],
                      [0] + sFE + [0],
                      [0, 0] + sFW]
            else:
                # shifts: F:+1, W:-1, E:0
                dp = [[0, 0] + sWE,
                      sFE + [0, 0],
                      [0] + sFW + [0]]
        
        # Final arrays: length 2n+1, score s at index s+n; count s > 0.
        ans = 0
        start = n + 1
        for arr in dp:
            ans += sum(arr[start:])
        return ans % MOD
```

```python
    def countWinningSequences(self, s: str) -> int:
        MOD = 10**9 + 7
        n = len(s)

        # Bob's score delta vs Alice's move, indexed by Bob's creature (F, W, E):
        #   Alice 'F': F->0, W->+1, E->-1
        #   Alice 'W': F->-1, W->0, E->+1
        #   Alice 'E': F->+1, W->-1, E->0

        # dp[0], dp[1], dp[2]: distributions of score difference (Bob - Alice)
        # for prefixes ending with Bob's creature F, W, E respectively.
        # After processing k+1 rounds, each array has length 2k+3 and
        # score s sits at index s + (k+1).

        a0 = s[0]
```

```python
class Solution:
    def countWinningSequences(self, s: str) -> int:
        MOD = 10**9 + 7
        n = len(s)

        # Bob's score delta (Bob points - Alice points) when Alice plays `a`
        # and Bob plays creature indexed 0='F', 1='W', 2='E':
        #   Alice 'F': F->0, W->+1, E->-1
        #   Alice 'W': F->-1, W->0, E->+1
        #   Alice 'E': F->+1, W->-1, E->0

        # dp[0], dp[1], dp[2]: distribution of the score difference over all
        # valid Bob prefixes ending with creature F / W / E.
        # After processing k rounds each array has length 2k+1 and the entry
        # for score s is at index s + k.

        a0 = s[0]
        if a0 == 'F':
            d = (0, 1, -1)
        elif a0 == 'W':
            d = (-1, 0, 1)
        else:
            d = (1, -1, 0)
        dp = [[0, 0, 0] for _ in range(3)]
        for c in range(3):
            dp[c][d[c] + 1] = 1

        for ch in s[1:]:
            f, w, e = dp
            # Sources for each choice of Bob's next creature:
            # next F <- previous W or E, next W <- previous F or E,
            # next E <- previous F or W.
            sWE = [(a + b) % MOD for a, b in zip(w, e)]
            sFE = [(a + b) % MOD for a, b in zip(f, e)]
            sFW = [(a + b) % MOD for a, b in zip(f, w)]
            if ch == 'F':
                # deltas: F:0, W:+1, E:-1  (shift right by 1 = +1)
                dp = [[0] + sWE + [0],
                      [0, 0] + sFE,
                      sFW + [0, 0]]
            elif ch == 'W':
                # deltas: F:-1, W:0, E:+1
                dp = [sWE + [0, 0],
                      [0] + sFE + [0],
                      [0, 0] + sFW]
            else:
                # deltas: F:+1, W:-1, E:0
                dp = [[0, 0] + sWE,
                      sFE + [0, 0],
                      [0] + sFW + [0]]

        # Final arrays have length 2n+1, score s at index s + n.
        # Bob wins iff the final difference is > 0 (indices >= n+1).
        ans = 0
        for arr in dp:
            ans += sum(arr[n + 1:])
        return ans % MOD
```
