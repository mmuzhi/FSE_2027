import sys

def main():
    data = sys.stdin.buffer.read().split()
    ptr = 0
    T = int(data[ptr]); ptr += 1
    out = []
    for _ in range(T):
        N = int(data[ptr]); ptr += 1
        M = 2 * N
        A = list(map(int, data[ptr:ptr + M]))
        ptr += M
        L = [0] * (N + 1)
        R = [0] * (N + 1)
        for pos in range(M):
            v = A[pos]
            lv = L[v]
            if lv:
                R[v] = pos
            else:
                L[v] = pos
        ans = 0
        last = M - 1
        for v in range(1, N + 1):
            r = R[v]
            l = L[v]
            d = r - l
            if d >= 2:
                if r < last and A[l + 1] == A[r + 1]:
                    ans += 1
                if d >= 4 and A[l + 1] == A[r - 1]:
                    ans += 1
        out.append(str(ans))
