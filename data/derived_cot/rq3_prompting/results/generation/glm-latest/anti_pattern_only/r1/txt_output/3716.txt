```python
import numpy as np
from typing import List

class Solution:
    def longestSubsequence(self, nums: List[int]) -> int:
        V = 300
        S = np.zeros((V, V), dtype=np.int32)
        Sflat = S.reshape(-1)
        vi = np.arange(V, dtype=np.int64)
        base = vi * V
        FLATN = [None] * (V + 1)
        for x in range(1, V + 1):
            FLATN[x] = (base + np.abs(vi + 1 - x)).astype(np.intp)
        ABuf = np.zeros(V, dtype=np.int32)
        BBuf = np.zeros(V, dtype=np.int32)
        T0 = np.zeros(V, dtype=np.int32)
        ONES = np.ones(V, dtype=np.int32)
        acc = np.maximum.accumulate
        npmax = np.maximum
        npadd = np.add
        ans = 1
        for x in nums:
            xi = x - 1
            L = Sflat.take(FLATN[x])
            pm = acc(L)
            smr = acc(L[::-1])
            ABuf[:xi + 1] = pm[xi::-1]
            ABuf[xi + 1:] = 0
            j = V - 1 - xi
            BBuf[:j + 1] = smr[j::-1]
            BBuf[j + 1:] = 0
            npmax(ABuf, BBuf, out=T0)
            npadd(T0, ONES, out=T0)   # T0 becomes Tp
            row = S[xi]
            npmax(row, T0, out=row)
            t = int(T0[0])
            if t > ans:
                ans = t
        return ans
```

```python
class Solution:
```
