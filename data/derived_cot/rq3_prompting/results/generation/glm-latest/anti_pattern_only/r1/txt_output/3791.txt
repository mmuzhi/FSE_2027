You are given two arrays of integers, fruits and baskets, each of length n, where fruits[i] represents the quantity of the i^th type of fruit, and baskets[j] represents the capacity of the j^th basket.

Each fruit type must be placed in the leftmost available basket with a capacity greater than or equal to the quantity of that fruit type.
Each basket can hold only one type of fruit.
If a fruit type cannot be placed in any basket, it remains unplaced.

Return the number of fruit types that remain unplaced after all possible allocations are made.

Input: fruits = [4,2,5], baskets = [3,5,4]

fruits[2] = 1 is placed in baskets[1] = 4.

Since all fruits are successfully placed, we return 0.

class Solution:
    def numOfUnplacedFruits(self, fruits: List[int], baskets: List[int]) -> int:
        n = len(fruits)
        # size = smallest power of 2 >= n
        size = 1
        while size < n:
            size <<= 1
        # segment tree for max, with -1 (or 0, since capacities >= 1) as empty
        NEG = 0  # since baskets[i] >= 1, 0 acts as -inf
        tree = [NEG] * (2 * size)
        for i in range(n):
            tree[size + i] = baskets[i]
        for i in range(size - 1, 0, -1):
            tree[i] = max(tree[2*i], tree[2*i+1])
        
        unplaced = 0
        for f in fruits:
            # find leftmost leaf with value >= f
            if tree[1] < f:
                unplaced += 1
                continue
            # descend
            node = 1
            while node < size:
                if tree[2*node] >= f:
                    node = 2*node
                else:
                    node = 2*node + 1
            # node is leaf, leaf index = node - size

        return unplaced

class Solution:
    def numOfUnplacedFruits(self, fruits: List[int], baskets: List[int]) -> int:

                continue

            node = 1

                if s > m:

from typing import List

class Solution:
    def numOfUnplacedFruits(self, fruits: List[int], baskets: List[int]) -> int:
        n = len(fruits)
        size = 1
        while size < n:
            size <<= 1
        
        # Max segment tree over basket positions.
        # t[i] = max capacity among FREE baskets in that node's range; 0 = none.
        t = [0] * (2 * size)
        t[size:size + n] = baskets
        for i in range(size - 1, 0, -1):
            a = t[i + i]
            b = t[i + i + 1]
            t[i] = a if a > b else b
        
        unplaced = 0
        for f in fruits:
            if t[1] < f:          # no free basket can hold this fruit
                unplaced += 1
                continue
            
            # Descend to the leftmost leaf (basket) with capacity >= f.
            node = 1
            while node < size:
                node += node      # go to left child
                if t[node] < f:   # left subtree can't hold it -> right child
                    node += 1
            
            # Remove this basket (leaf) and update ancestors.
            # v was the max of the path subtree (all siblings seen are < v),
            # m tracks the new max along the path.
            v = t[node]
            t[node] = 0
            m = 0
            while node > 1:
                s = t[node ^ 1]   # sibling value
                if s >= v:        # this ancestor (and all above) keep old value
                    break
                if s > m:
                    m = s
                node >>= 1
                t[node] = m
        
        return unplaced
