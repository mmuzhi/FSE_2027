```python
class Solution:
    def minCost(self, nums: List[int]) -> int:
        n = len(nums)
        if n == 1:
            return nums[0]
        if n == 2:
            return max(nums[0], nums[1])
        
        if n & 1:
            nxt = nums[:]          # dp at i = n: cost = nums[j]
            i = n - 2
        else:
            last = nums[-1]
            nxt = [v if v > last else last for v in nums]  # dp at i = n-1
            i = n - 3
        
        while i >= 3:
            x = nums[i]
            y = nums[i + 1]
            A = x if x > y else y
            di = nxt[i]
            di1 = nxt[i + 1]
            ay = (y if y > 0 else 0)  # not needed
            cur = []
            ap = cur.append
            for j in range(i):
                v = nums[j]
                t1 = A + nxt[j]
                m = v if v > y else y
                t2 = m + di
                if t2 < t1:
                    t1 = t2
                m = v if v > x else x
                t2 = m + di1
                if t2 < t1:
                    t1 = t2
                ap(t1)
            nxt = cur
            i -= 2
        
        a, b, c = nums[0], nums[1], nums[2]
        return min(
            (b if b > c else c) + nxt[0],
            (a if a > c else c) + nxt[1],
            (a if a > b else b) + nxt[2],
        )
```

```python
    def minCost(self, nums: List[int]) -> int:
        n = len(nums)
        if n == 1:
            return nums[0]
        if n == 2:
            return max(nums[0], nums[1])

        if n % 2 == 1:
            # dp layer for i = n (one element [nums[j]] remains): cost = nums[j]
            nxt = nums[:]
            i = n - 2
```

=== ANSWER ===
Looking at this problem:

- The array always has the form `[nums[j]] + nums[i:]` where `nums[j]` is a "carried" element kept from a previous operation, and `i` increases by 2 each operation (i is always odd, starting at 3 after the first op on indices 0,1,2).
- From state `(j, i)` (first three are `nums[j], nums[i], nums[i+1]`), we keep one of them and remove the other two, paying the max of the removed pair.
- Terminal: array of length 2 (`i = n-1`, only when n even) costs `max` of the two; array of length 1 (`i = n`, when n odd) costs that element.

So `dp[j][i] = min( max(nums[i],nums[i+1]) + dp[j][i+2], max(nums[j],nums[i+1]) + dp[i][i+2], max(nums[j],nums[i]) + dp[i+1][i+2] )`, computed layer by layer (odd `i` only), which is O(n²) — fine for n ≤ 1000. The answer combines the first operation on indices (0,1,2) with `dp[j][3]`.

Verified against both examples (12 and 5) and hand-computed brute-force cases for n=5,6,7.

```python
from typing import List

class Solution:
    def minCost(self, nums: List[int]) -> int:
        n = len(nums)
        if n == 1:
            return nums[0]
        if n == 2:
            return max(nums[0], nums[1])

        # nxt[j] = min cost to empty current array [nums[j]] + nums[i:] for current layer i
        if n % 2 == 1:
            # layer i = n: only [nums[j]] remains -> cost nums[j]
            nxt = nums[:]
            i = n - 2
        else:
            # layer i = n - 1: [nums[j], nums[n-1]] remains -> cost max of the two
            last = nums[-1]
            nxt = [max(v, last) for v in nums]
            i = n - 3

        while i >= 3:
            x, y = nums[i], nums[i + 1]   # the two fresh elements among the first three
            keep_cost = max(x, y)         # keep carried j: pay max(x, y)
            di = nxt[i]                   # future cost if we keep index i
            di1 = nxt[i + 1]              # future cost if we keep index i + 1
            nxt = [min(keep_cost + d, max(v, y) + di, max(v, x) + di1)
                   for v, d in zip(nums[:i], nxt)]
            i -= 2

        # first operation on indices 0,1,2: keep one, pay max of other two
        a, b, c = nums[0], nums[1], nums[2]
        return min(max(b, c) + nxt[0],
                   max(a, c) + nxt[1],
                   max(a, b) + nxt[2])
```
