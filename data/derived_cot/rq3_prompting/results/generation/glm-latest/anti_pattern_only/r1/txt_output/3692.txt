```python
def kmp_all(s, t):
    # returns list of start indices where t occurs in s
    if not t:
        return list(range(len(s) + 1))
    # build failure function
    m = len(t)
    fail = [0] * m
    k = 0
    for i in range(1, m):
        while k > 0 and t[i] != t[k]:
            k = fail[k - 1]
        if t[i] == t[k]:
            k += 1
        fail[i] = k
    res = []
    k = 0
    for i, ch in enumerate(s):
        while k > 0 and ch != t[k]:
            k = fail[k - 1]
        if ch == t[k]:
            k += 1
        if k == m:
            res.append(i - m + 1)
            k = fail[k - 1]
    return res
```

```python
        n = len(s)
        A, B, C = p.split('*')

        occC = kmp_all(s, C)
```

```python
class Solution:
    def shortestMatchingSubstring(self, s: str, p: str) -> int:
        n = len(s)
        A, B, C = p.split('*')
        la, lb, lc = len(A), len(B), len(C)
        
        def find_all(t):
            # all start indices of occurrences of t in s (overlapping allowed)
            if not t:
                return None  # handled specially
            res = []
            start = 0
            while True:
                i = s.find(t, start)
                if i == -1:
                    return res
                res.append(i)
                start = i + 1
        
        # endA[i]: latest end (exclusive) of an occurrence of A with end <= i; -1 if none
        # For empty A: endA[i] = i
        if la == 0:
            endA = list(range(n + 1))
        else:
            occ = find_all(A)
            endA = [-1] * (n + 1)
            j = 0
            m = len(occ)
            cur = -1
            for i in range(n + 1):
                while j < m and occ[j] + la <= i:
                    cur = occ[j] + la
                    j += 1
                endA[i] = cur
        
        # startC[i]: earliest start of an occurrence of C with start >= i; n+1 if none
        # For empty C: startC[i] = i
        if lc == 0:
            startC = list(range(n + 1))
        else:
            occ = find_all(C)
            startC = [n + 1] * (n + 1)
            j = len(occ) - 1
            cur = n + 1
            for i in range(n, -1, -1):
                while j >= 0 and occ[j] >= i:
                    cur = occ[j]
                    j -= 1
                startC[i] = cur
        
        best = float('inf')
        
        if lb == 0:
            for b in range(n + 1):
                a2 = endA[b]
                if a2 < 0:
                    continue
                c1 = startC[b]
                if c1 > n:
                    continue
                cand = c1 + lc - a2 + la
                if cand < best:
                    best = cand
        else:
            b1 = s.find(B)
            while b1 != -1:
                a2 = endA[b1]
                if a2 >= 0:
                    c1 = startC[b1 + lb]
                    if c1 <= n:
                        cand = c1 + lc - a2 + la
                        if cand < best:
                            best = cand
                b1 = s.find(B, b1 + 1)
        
        return -1 if best == float('inf') else best
```

```python
    def shortestMatchingSubstring(self, s: str, p: str) -> int:
        n = len(s)
        A, B, C = p.split('*')
        la, lb, lc = len(A), len(B), len(C)

        # --- collect occurrences via KMP ---
        def kmp_starts(t):
            """Return sorted list of start indices of all (overlapping) occurrences of t in s."""
            m = len(t)
            if m == 0:
                return list(range(n + 1))
            if m > n:
                return []
            # failure function
            fail = [0] * m
            k = 0
            for i in range(1, m):
                c = t[i]
                while k and c != t[k]:
                    k = fail[k - 1]
                if c == t[k]:
                    k += 1
                fail[i] = k
            res = []
            k = 0
            for i in range(n):
                c = s[i]
                while k and c != t[k]:
                    k = fail[k - 1]
                if c == t[k]:
                    k += 1
                if k == m:
                    res.append(i - m + 1)
                    k = fail[k - 1]
            return res
        
        occA = kmp_starts(A)
        occC = kmp_starts(C)
        # For B, we can also just get all occurrences:
        occB = kmp_starts(B)

                run = endA[i]
            else:
                endA[i] = run

                run = startC[i]

            c1 = startC[b1 + lb]
```

```python
class Solution:
    def shortestMatchingSubstring(self, s: str, p: str) -> int:
        n = len(s)
        A, B, C = p.split('*')
        la, lb, lc = len(A), len(B), len(C)

        def kmp_starts(t):
            """All start positions (overlapping) where t occurs in s, sorted ascending."""
            m = len(t)
            if m == 0:
                return list(range(n + 1))
            if m > n:
                return []
            fail = [0] * m
            k = 0
            for i in range(1, m):
                c = t[i]
                while k and c != t[k]:
                    k = fail[k - 1]
                if c == t[k]:
                    k += 1
                fail[i] = k
            res = []
            k = 0
            ap = res.append
            for i in range(n):
                c = s[i]
                while k and c != t[k]:
                    k = fail[k - 1]
                if c == t[k]:
                    k += 1
                    if k == m:
                        ap(i - m + 1)
                        k = fail[k - 1]
            return res

        occA = kmp_starts(A)
        occB = kmp_starts(B)
        occC = kmp_starts(C)

        # endA[i]: latest end position (exclusive) of an A-occurrence with end <= i, else -1
        endA = [-1] * (n + 1)
        for st in occA:
            endA[st + la] = st + la
        run = -1
        for i in range(n + 1):
            v = endA[i]
            if v != -1:
                run = v
            else:
                endA[i] = run

        # startC[i]: earliest start position of a C-occurrence with start >= i, else n+1
        startC = [n + 1] * (n + 1)
        for st in occC:
            startC[st] = st
        run = n + 1
        for i in range(n, -1, -1):
            v = startC[i]
            if v != n + 1:
                run = v
            else:
                startC[i] = run

            a2 = endA[b1]
            if a2 < 0:
                continue
            c1 = startC[b1 + lb]
            if c1 > n:
                continue
```

```python
from itertools import accumulate


class Solution:
    def shortestMatchingSubstring(self, s: str, p: str) -> int:
        n = len(s)
        A, B, C = p.split('*')
        la, lb, lc = len(A), len(B), len(C)

        def kmp_starts(t):
            """All start positions (overlapping, ascending) where t occurs in s.
            An empty t matches at every position 0..n."""
            m = len(t)
            if m == 0:
                return list(range(n + 1))
            if m > n:
                return []
            # KMP failure function
            fail = [0] * m
            k = 0
            for i in range(1, m):
                c = t[i]
                while k and c != t[k]:
                    k = fail[k - 1]
                if c == t[k]:
                    k += 1
                fail[i] = k
            # scan s
            res = []
            append = res.append
            k = 0
            for i, c in enumerate(s):
                while k and c != t[k]:
                    k = fail[k - 1]
                if c == t[k]:
                    k += 1
                    if k == m:
                        append(i - m + 1)
                        k = fail[k - 1]
            return res

        occA = kmp_starts(A)
        occB = kmp_starts(B)
        occC = kmp_starts(C)

        # endA[i]: latest position e <= i such that A matches s[e-la:e]; -1 if none.
        endA = [-1] * (n + 1)
        for st in occA:
            endA[st + la] = st + la
        endA = list(accumulate(endA, max))

        # startC[i]: earliest position st >= i such that C matches s[st:st+lc]; n+1 if none.
        startC = [n + 1] * (n + 1)
        for st in occC:
            startC[st] = st
        startC = list(accumulate(reversed(startC), min))[::-1]

        best = n + 1
        for b1 in occB:
            a2 = endA[b1]
            if a2 < 0:
                continue
            c1 = startC[b1 + lb]
            if c1 > n:
                continue
            cand = c1 + lc - a2 + la
            if cand < best:
                best = cand

        return best if best <= n else -1
```

```python
from itertools import accumulate


class Solution:
    def shortestMatchingSubstring(self, s: str, p: str) -> int:
        n = len(s)
        A, B, C = p.split('*')
        la, lb, lc = len(A), len(B), len(C)

        def kmp_starts(t):
            # all start positions (overlapping) where t occurs in s, ascending;
            # empty t matches at every position 0..n
            m = len(t)
            if m == 0:
                return list(range(n + 1))
            if m == 1:
                return [i for i, ch in enumerate(s) if ch == t]
            if m > n:
                return []
            # failure function
            fail = [0] * m
            k = 0
            for i in range(1, m):
                c = t[i]
                while k and c != t[k]:
                    k = fail[k - 1]
                if c == t[k]:
                    k += 1
                fail[i] = k
            # scan
            res = []
            append = res.append
            k = 0
            for i, c in enumerate(s):
                while k and c != t[k]:
                    k = fail[k - 1]
                if c == t[k]:
                    k += 1
                    if k == m:
                        append(i - m + 1)
                        k = fail[k - 1]
            return res

        occA = kmp_starts(A)
        occB = kmp_starts(B)
        occC = kmp_starts(C)

        # endA[i] = latest end e <= i with A matching s[e-la:e]; -1 if none
        endA = [-1] * (n + 1)
        for st in occA:
            endA[st + la] = st + la
        endA = list(accumulate(endA, max))

        # startC[i] = earliest start st >= i with C matching s[st:st+lc]; n+1 if none
        startC = [n + 1] * (n + 1)
        for st in occC:
            startC[st] = st
        startC = list(accumulate(reversed(startC), min))[::-1]

        best = n + 1
        for b1 in occB:
            a2 = endA[b1]
            if a2 < 0:
                continue
            c1 = startC[b1 + lb]
            if c1 > n:
                continue
            cand = c1 + lc - a2 + la
            if cand < best:
                best = cand

        return best if best <= n else -1
```
