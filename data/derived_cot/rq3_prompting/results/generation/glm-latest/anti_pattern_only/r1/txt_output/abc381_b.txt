def main():

from collections import Counter

def main():
    S = input().strip()
    n = len(S)
    # Condition 1: even length
    if n % 2 != 0:
        print("No")
        return
    # Condition 2: each adjacent pair (S[2i], S[2i+1]) has equal characters
    for i in range(0, n, 2):
        if S[i] != S[i + 1]:
            print("No")
            return
    # Condition 3: every character that appears does so exactly twice
    cnt = Counter(S)
    if all(v == 2 for v in cnt.values()):
        print("Yes")
    else:
        print("No")

main()
