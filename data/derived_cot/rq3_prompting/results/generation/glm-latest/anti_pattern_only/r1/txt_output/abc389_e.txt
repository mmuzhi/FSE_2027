import sys
from math import isqrt

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    m = int(data[1])
    ps = list(map(int, data[2:2 + n]))

    HI = 4 * 10**18

    try:
        import numpy as np
        P = np.array(ps, dtype=np.int64)
        caps = np.array([isqrt(m // p) for p in ps], dtype=np.int64)
        SHIFT = 20
        MASK = (1 << SHIFT) - 1

        def ok(x):
            q = x // P
            j = (q + 1) >> 1
            if bool(np.any(j > caps)):
                return False
            t = j * j
            t *= P
            hi_part = int((t >> SHIFT).sum())
            lo_part = int((t & MASK).sum())
            return (hi_part << SHIFT) + lo_part <= m

        lo, hi = 0, HI
        while hi - lo > 1:
            mid = (lo + hi) >> 1
            if ok(mid):
                lo = mid
            else:
                hi = mid

        x = lo
        q = x // P
        j = (q + 1) >> 1
        t = j * j * P
        S = (int((t >> SHIFT).sum()) << SHIFT) + int((t & MASK).sum())
        g = int(j.sum())
        R = m - S
        x1 = lo + 1
        mask = ((x1 % P) == 0) & (((x1 // P) & 1) == 1)
        c = int(mask.sum())
        extra = min(R // x1, c)
        print(g + extra)

    except ImportError:
        # Pure Python fallback
        ps.sort(reverse=True)

        def ok(x):
            s = 0
            for p in ps:
                j = ((x // p) + 1) >> 1
                s += j * j * p
