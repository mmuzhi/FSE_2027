```python
class Solution:
    def minOperations(self, nums: List[int]) -> int:
        # smallest prime factor sieve up to max(nums)
        maxv = max(nums)
        if maxv < 2:
            spf = [0, 0]
        else:
            spf = list(range(maxv + 1))
            i = 2
            while i * i <= maxv:
                if spf[i] == i:  # i is prime
                    for j in range(i * i, maxv + 1, i):
                        if spf[j] == j:
                            spf[j] = i
                i += 1
        ...
```

```python
class Solution:
    def minOperations(self, nums: List[int]) -> int:
        maxv = max(nums)

                    ans += 1
                    L = p
                else:
                    return -1
        return ans
```

```python
from typing import List
from math import isqrt

class Solution:
    def minOperations(self, nums: List[int]) -> int:
        maxv = max(nums)
        # smallest-prime-factor table via downward slice sieve
        spf = list(range(maxv + 1))
        for i in range(isqrt(maxv), 1, -1):
            start = i * i
            if start <= maxv:
                cnt = (maxv - start) // i + 1
                spf[start::i] = [i] * cnt
        
        ans = 0
        L = nums[-1]
        for i in range(len(nums) - 2, -1, -1):
            x = nums[i]
            if x <= L:
                L = x
            else:
                p = spf[x]
                if p <= L:
                    ans += 1
                    L = p
                else:
                    return -1
        return ans
```

Looking at this problem:

- The greatest proper divisor of `x` is `x / spf(x)` where `spf(x)` is the smallest prime factor. So **one operation turns `x` into `spf(x)`** (e.g., 25→5, 12→2, 6→2). Primes and 1 never change (greatest proper divisor of a prime is 1).
- Thus each element has only two reachable values: `x` (cost 0) or `spf(x)` (cost 1).
- Sweep right to left keeping a `limit` (final value chosen for the right neighbor). If `x <= limit`, keep it (cost 0, and it maximizes the limit for the elements to the left — an exchange argument shows reducing when not forced is never beneficial). Otherwise we must reduce to `spf(x)`; if even that exceeds `limit`, it's impossible (`-1`).

An SPF sieve up to `10^6` (built with fast slice assignments, iterating downward so the smallest factor is written last) gives all needed factors.

Verified against all examples: `[25,7]→1`, `[7,7,6]→-1` (7 is prime, 6→2 still breaks order), `[1,1,1,1]→0`.

```python
from math import isqrt
from typing import List


class Solution:
    def minOperations(self, nums: List[int]) -> int:
        n = len(nums)
        # Already non-decreasing -> nothing to do.
        if all(nums[i] <= nums[i + 1] for i in range(n - 1)):
            return 0

        maxv = max(nums)

        # spf[v] = smallest prime factor of v (spf[p] = p for primes, spf[1] = 1).
        # Downward sieve: smaller factors are written last, so they win.
        spf = list(range(maxv + 1))
        for i in range(isqrt(maxv), 1, -1):
            start = i * i
            if start <= maxv:
                cnt = (maxv - start) // i + 1
                spf[start::i] = [i] * cnt

        # One operation replaces x by spf(x); primes and 1 never change.
        # Sweep right to left keeping each final value as large as possible:
        # keep x when allowed (0 ops), otherwise fall back to spf(x) (1 op).
        ops = 0
        limit = nums[-1]
        for i in range(n - 2, -1, -1):
            x = nums[i]
            if x <= limit:
                limit = x
            else:
                p = spf[x]
                if p > limit:
                    return -1
                ops += 1
                limit = p
        return ops
```
