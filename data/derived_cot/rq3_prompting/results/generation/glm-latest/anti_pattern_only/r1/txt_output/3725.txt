class Solution:
    def minMaxSubarraySum(self, nums: List[int], k: int) -> int:
        def sum_of_mins(arr, k):
            n = len(arr)
            prev_less = [-1] * n  # nearest j < i with arr[j] < arr[i]
            next_le = [n] * n     # nearest j > i with arr[j] <= arr[i]
            
            stack = []
            for i in range(n):
                while stack and arr[stack[-1]] >= arr[i]:
                    stack.pop()
                prev_less[i] = stack[-1] if stack else -1
                stack.append(i)
            
            stack = []
            for i in range(n):
                while stack and arr[stack[-1]] >= arr[i]:
                    next_le[stack.pop()] = i
                stack.append(i)
            
            T = k - 1
            def g(t):
                return (t + 1) * (t + 2) // 2 if t >= 0 else 0
            
            total = 0
            for i in range(n):
                A = i - prev_less[i]
                B = next_le[i] - i
                cnt = g(T) - g(T - A) - g(T - B) + g(T - A - B)
                total += arr[i] * cnt
            return total
        
        mins_sum = sum_of_mins(nums, k)
        maxs_sum = -sum_of_mins([-x for x in nums], k)
        return mins_sum + maxs_sum

        def sum_of_mins(arr):
            n = len(arr)
            prev_less = [-1] * n   # nearest j < i with arr[j] < arr[i]
            next_le = [n] * n      # nearest j > i with arr[j] <= arr[i]

                stack.append(i)

Looking at this problem, I need to compute the sum of (max + min) over all subarrays of length ≤ k. This equals (sum of subarray minimums with length ≤ k) + (sum of subarray maximums with length ≤ k), where the max sum can be computed as the negated min sum of the negated array.

**Key idea:** For each index `i`, count subarrays where `i` is the *rightmost* occurrence of the minimum (using strict `<` on the left boundary, `≤` on the right boundary to avoid double counting ties). Let `A = i - prev_less[i]` (left extension choices) and `B = next_le[i] - i` (right extension choices). A subarray corresponds to picking `x ∈ [1,A]` (left part size) and `y ∈ [1,B]` (right part size) with total length `x + y - 1 ≤ k`, i.e., `x + y ≤ k + 1`.

With `u = x-1 ∈ [0, A-1]`, `v = y-1 ∈ [0, B-1]`, constraint `u + v ≤ T = k-1`. By inclusion-exclusion:

`count = g(T) - g(T-A) - g(T-B) + g(T-A-B)` where `g(t) = (t+1)(t+2)/2` for `t ≥ 0` else `0`.

I verified this against both examples (getting 20 and -6), plus tie cases like `[2,2]`, `[2,2,1,2]`, and `[2,1,2]` — all match brute force.

class Solution:

        def sum_of_mins(arr):
            n = len(arr)
            prev_less = [-1] * n   # nearest j < i with arr[j] < arr[i] (strict)
            next_le = [n] * n      # nearest j > i with arr[j] <= arr[i] (non-strict)

                while stack and arr[stack[-1]] >= ai:
                    stack.pop()
                if stack:
                    prev_less[i] = stack[-1]
                stack.append(i)

            stack = []
            for i in range(n):
                ai = arr[i]
                while stack and arr[stack[-1]] >= ai:
                    next_le[stack.pop()] = i
                stack.append(i)

            def g(t):
                return (t + 1) * (t + 2) // 2 if t >= 0 else 0

            T = k - 1
            total = 0
            for i in range(n):
                A = i - prev_less[i]
                B = next_le[i] - i
                cnt = g(T) - g(T - A) - g(T - B) + g(T - A - B)
