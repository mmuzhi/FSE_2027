import sys

def main():
    data = sys.stdin.read().split()
    n = int(data[0])
    s = data[1]
    # c_k = k * d_k for k = 1..n
    # sigma_j = prefix sum
    # coefficient of 10^p is sigma_{n-p}, p = 0..n-1
    digits = [ord(ch) - 48 for ch in s]
    # sigma[j] for j = 1..n
    # We'll build the coefficient list from least significant: coeffs[0] = sigma_n, ..., coeffs[n-1] = sigma_1
    # Use accumulate
    from itertools import accumulate
    sig = list(accumulate((k+1) * d for k, d in enumerate(digits)))
    # sig[i] = sigma_{i+1}, i = 0..n-1
    # coeffs[p] = sigma_{n-p} = sig[n-p-1]
    coeffs = sig[::-1]  # coeffs[0] = sig[n-1] = sigma_n, coeffs[n-1] = sig[0] = sigma_1. ✓
    
    # normalize
    res = []
    carry = 0
    for v in coeffs:
        t = v + carry
        res.append(t % 10)
        carry = t // 10
    while carry:
        res.append(carry % 10)
        carry //= 10
    # res is least significant first; reverse for output
    res.reverse()
    print(''.join(map(str, res)))

main()

Looking at this problem:

**Key insight:** Digit $d_k$ at position $k$ (1-indexed) contributes $d_k \cdot 10^{j-k}$ to $f(i,j)$ whenever $i \le k \le j$. There are $k$ choices of $i$, so its total contribution is $k \cdot d_k \cdot \sum_{j=k}^{N} 10^{j-k} = k \cdot d_k \cdot R(N-k+1)$ where $R(m)$ is the repunit with $m$ ones.

So the answer is $\sum_k k\,d_k\, R(N-k+1)$. Since $R(m) = \sum_{p=0}^{m-1}10^p$, the coefficient of $10^p$ in the answer equals $\sum_{k \le N-p} k\,d_k = \sigma_{N-p}$ (prefix sums of $k \cdot d_k$). These coefficients are at most $\approx 1.8\times 10^{11}$, so I can normalize the decimal representation with a simple carry loop — all $O(N)$ small-integer operations, avoiding slow big-integer↔string quadratic conversions for the up-to-$2\times10^5$-digit answer.
