import sys

def solve():
    data = sys.stdin.read().split()
    S = data[1]
    T = data[2]

    n, m = len(S), len(T)
    if abs(n - m) > 1:
        print("No")
        return
    if S == T:
        print("Yes")
        return
    if n == m:
        diff = 0
        for a, b in zip(S, T):
            if a != b:
                diff += 1
                if diff > 1:
                    break
        print("Yes" if diff <= 1 else "No")
        return
    if n == m + 1:
        # delete one char from S
        i = 0
        while i < m and S[i] == T[i]:
            i += 1
        if S[i + 1:] == T[i:]:
            print("Yes")
        else:
            print("No")
        return
    if m == n + 1:
        # insert one char into S (delete one char from T)
        i = 0
        while i < n and S[i] == T[i]:
            i += 1
        if T[i + 1:] == S[i:]:
            print("Yes")
        else:
            print("No")
        return

solve()
