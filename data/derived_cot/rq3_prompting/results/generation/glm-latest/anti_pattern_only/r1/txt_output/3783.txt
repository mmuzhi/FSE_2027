```python
from typing import List

class Solution:
    def permute(self, n: int, k: int) -> List[int]:
        # factorials
        fact = [1] * (n + 1)
        for i in range(1, n + 1):
            fact[i] = fact[i-1] * i
        
        # f(o, e, next_odd): number of ways to fill remaining positions
        # with o unused odds, e unused evens, next position requires parity
        # next_odd (True = odd required).
        def f(o, e, next_odd):
            if next_odd:
                # pattern O,E,O,E,...: need o == e or o == e+1
                if o == e or o == e + 1:
                    return fact[o] * fact[e]
                return 0
            else:
                if e == o or e == o + 1:
                    return fact[o] * fact[e]
                return 0
        
        used = [False] * (n + 1)
        result = []
        o = (n + 1) // 2  # total odds
        e = n // 2        # total evens
        prev_parity = -1  # -1 means no previous
        
        for i in range(n):
            chosen = False
            for v in range(1, n + 1):
                if used[v]:
                    continue
                p = v & 1
                if i > 0 and p == prev_parity:
                    continue
                new_o = o - (1 if p == 1 else 0)
                new_e = e - (1 if p == 0 else 0)
                if i == n - 1:
                    # last position; after placing, nothing remains.
                    # count = 1 (f(0,0,anything) == 1)
                    cnt = 1
                else:
                    cnt = f(new_o, new_e, next_odd=(p == 0))
                if k > cnt:
                    k -= cnt
                else:
                    # choose v
                    used[v] = True
                    result.append(v)
                    o, e = new_o, new_e
                    prev_parity = p
                    chosen = True
                    break
            if not chosen:
                return []
        
        return result
```

Looking at this problem:

- Adjacent elements must alternate parity, so once the parity of the first element is fixed, the entire parity pattern is determined.
- State for counting completions: (unused odds, unused evens, parity required next). The count is `fact[o] * fact[e]` if the counts match the alternating pattern requirement (starting parity must be the majority or equal), else 0.
- Build the answer greedily: at each position try candidate values in increasing order, count completions for each, skip blocks by subtracting from k, otherwise commit.

Verified against all three examples (n=4,k=6 → [3,4,1,2]; n=3,k=2 → [3,2,1]; n=2,k=3 → []).
