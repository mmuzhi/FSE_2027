class Solution:
    def countPairs(self, nums: List[int]) -> int:
        def gen(v):
            s = str(v)
            res = {v}
            n = len(s)
            for a in range(n):
                for b in range(a + 1, n):
                    lst = list(s)
                    lst[a], lst[b] = lst[b], lst[a]
                    res.add(int(''.join(lst)))
            return res
        
        cache = {}
        sets = []
        for v in nums:
            if v not in cache:
                cache[v] = gen(v)
            sets.append(cache[v])
        
        count = 0
        n = len(nums)
        for i in range(n):
            for j in range(i + 1, n):
                if nums[j] in sets[i] or nums[i] in sets[j]:
                    count += 1
        return count
```
