from typing import List
from collections import deque


class Solution:
    def maxMoves(self, kx: int, ky: int, positions: List[List[int]]) -> int:
        n = len(positions)
        pts = [(p[0], p[1]) for p in positions]
        pts.append((kx, ky))
        m = n + 1

        kmoves = ((1, 2), (2, 1), (-1, 2), (-2, 1), (1, -2), (2, -1), (-1, -2), (-2, -1))

        # D[s][t] = minimum knight moves between point s and point t
        D = []
        for sx, sy in pts:
            dist = [[-1] * 50 for _ in range(50)]
            dist[sx][sy] = 0
            q = deque([(sx, sy)])
            while q:
                x, y = q.popleft()
                nd = dist[x][y] + 1
                for dx, dy in kmoves:
                    nx, ny = x + dx, y + dy
                    if 0 <= nx < 50 and 0 <= ny < 50 and dist[nx][ny] < 0:
                        dist[nx][ny] = nd
                        q.append((nx, ny))
            D.append([dist[tx][ty] for tx, ty in pts])

        full = (1 << n) - 1
        # dp[mask][last]: optimal total moves for the rest of the game,
        # given captured set = mask and knight standing on point `last`
        # (last == n means nothing captured yet, knight at start).
        dp = [[0] * m for _ in range(full + 1)]

        for mask in range(full - 1, -1, -1):
            pairs = [(i, dp[mask | (1 << i)][i]) for i in range(n) if not (mask >> i) & 1]
            alice = (bin(mask).count('1') & 1) == 0
            row = dp[mask]
            if mask:
                for last in range(n):
                    if (mask >> last) & 1:
                        Dl = D[last]
                        if alice:
                            row[last] = max([Dl[i] + v for i, v in pairs])
                        else:
                            row[last] = min([Dl[i] + v for i, v in pairs])
            else:
                Dl = D[n]
                if alice:
                    row[n] = max([Dl[i] + v for i, v in pairs])
                else:
                    row[n] = min([Dl[i] + v for i, v in pairs])

        return dp[0][n]
