def main():
    data = sys.stdin.buffer.read().split()
    N = int(data[0]); Q = int(data[1])
    Ks = [int(x) for x in data[2:2 + Q]]
    NN = N * N

    # dp[i][j]: bitmask of achievable sums M using groups totaling exactly i rows, j columns
    dp = [[0] * (N + 1) for _ in range(N + 1)]
    dp[0][0] = 1
    for i in range(2, N + 1):
        dpi = dp[i]
        for j in range(2, N + 1):
            v = 0
            # last group (r, c), 2 <= r <= i, 2 <= c <= j
            # dp[i-r][j-c] nonzero only if (i-r, j-c) == (0,0) or both >= 2
            # case r == i, c == j:
            v = 1 << (i * j)
            for r in range(2, i - 1):  # r from 2 to i-2 (so i-r >= 2)
                src = dp[i - r]
                for c in range(2, j - 1):  # c from 2 to j-2
                    s = src[j - c]
                    if s:
                        v |= s << (r * c)
            dpi[j] = v
    
    achievable = 1  # M = 0 (empty)
    for i in range(2, N + 1):
        for j in range(2, N + 1):
            achievable |= dp[i][j]
    
    out = []
    for K in Ks:
        M = NN - K
        if 0 <= M <= NN and (achievable >> M) & 1:
            out.append("Yes")
        else:
            out.append("No")
