def main():
    import sys
    data = sys.stdin.read().split()
    n = int(data[0])
    c = int(data[1])
    ts = list(map(int, data[2:2+n]))

    last = -10**9  # effectively negative infinity
    for t in ts:
        if t - last >= c:
            count += 1
            last = t
    print(count)

main()
