def main():

    n = int(data[0])
    a = list(map(int, data[1:1 + n]))
    # Check A[i+1] / A[i] == A[1] / A[0] for all i using cross multiplication
    # (all values positive, so no division-by-zero issues; exact integer math)
    for i in range(n - 1):
        if a[i + 1] * a[0] != a[i] * a[1]:
