class Solution:
    def lenOfVDiagonal(self, grid: List[List[int]]) -> int:
        n = len(grid)
        m = len(grid[0])
        dirs = [(1,1),(1,-1),(-1,-1),(-1,1)]
        nxt = {1:2, 0:2, 2:0}
        
        # dp1[d]: after turn
        dp1 = []
        for d,(dr,dc) in enumerate(dirs):
            table = [[1]*m for _ in range(n)]
            rng_r = range(n-1,-1,-1) if dr==1 else range(n)
            rng_c = range(m-1,-1,-1) if dc==1 else range(m)
            for r in rng_r:
                for c in rng_c:
                    nr, nc = r+dr, c+dc
                    if 0<=nr<n and 0<=nc<m and grid[nr][nc]==nxt[grid[r][c]]:
                        table[r][c] = 1 + table[nr][nc]
                    # else stays 1
            dp1.append(table)
        
        # dp0[d]: before turn
        dp0 = []
        for d,(dr,dc) in enumerate(dirs):
            d2 = (d+1)%4
            dr2, dc2 = dirs[d2]
            t2 = dp1[d2]
            table = [[1]*m for _ in range(n)]
            rng_r = range(n-1,-1,-1) if dr==1 else range(n)
            rng_c = range(m-1,-1,-1) if dc==1 else range(m)
            for r in rng_r:
                for c in rng_c:
                    exp = nxt[grid[r][c]]
                    best = 0
                    nr, nc = r+dr, c+dc
                    if 0<=nr<n and 0<=nc<m and grid[nr][nc]==exp:
                        best = table[nr][nc]  # straight, no turn yet
                    nr2, nc2 = r+dr2, c+dc2
                    if 0<=nr2<n and 0<=nc2<m and grid[nr2][nc2]==exp:
                        v = t2[nr2][nc2]
                        if v > best:
                            best = v
                    table[r][c] = 1 + best
            dp0.append(table)
        
        ans = 0
        for r in range(n):
            row = grid[r]
            for c in range(m):
                if row[c] == 1:
                    for d in range(4):
                        v = dp0[d][r][c]
                        if v > ans:
                            ans = v
        return ans

        H = n + 2

        g = [3] * (W * H)
        for r in range(n):
            row = grid[r]
            base = (r + 1) * W + 1
            g[base:base + m] = row

        deltas = [dr * W + dc for dr, dc in dirs]

class Solution:
    def lenOfVDiagonal(self, grid: List[List[int]]) -> int:
        n = len(grid)
        m = len(grid[0])
        W = m + 2
        total = (n + 2) * W
        
        # Flat padded grid; pad value 3 never matches an expected value (0 or 2).
        g = [3] * total
        for r in range(n):
            base = (r + 1) * W + 1
            g[base:base + m] = grid[r]
        
        ones = [i for i, v in enumerate(g) if v == 1]
        if not ones:
            return 0
        
        # next expected value in the sequence, indexed by current value
        nxt = (2, 2, 0, -1)
        E = [nxt[v] for v in g]
        
        dirs = ((1, 1), (1, -1), (-1, -1), (-1, 1))
        
        # dp1[d][i]: longest valid continuation starting at cell i, moving in
        # direction d, with the turn already used (straight only).
        t1s = []
        for dr, dc in dirs:
            t = [1] * total
            for r in (range(n, 0, -1) if dr == 1 else range(1, n + 1)):
                base = r * W
                nb = base + dr * W + 1 + dc     # neighbor slice start
                e_row = E[base + 1: base + m + 1]
                g_row = g[nb: nb + m]
                t_row = t[nb: nb + m]
                t[base + 1: base + m + 1] = [
                    tn + 1 if gn == e else 1
                    for e, gn, tn in zip(e_row, g_row, t_row)
                ]
            t1s.append(t)
        
        ans = 0
        for d in range(4):
            dr, dc = dirs[d]
            dr2, dc2 = dirs[(d + 1) & 3]
            t2 = t1s[(d + 1) & 3]
            t = [1] * total
            for r in (range(n, 0, -1) if dr == 1 else range(1, n + 1)):
                base = r * W
                nb = base + dr * W + 1 + dc
                nb2 = base + dr2 * W + 1 + dc2
                e_row = E[base + 1: base + m + 1]

                t_s = t[nb: nb + m]

                t_w = t2[nb2: nb2 + m]

                    for e, gs, ts, gw, tw in zip(e_row, g_s, t_s, g_w, t_w)
                ]
            cand = max(map(t.__getitem__, ones))
            if cand > ans:
                ans = cand
        return ans

from typing import List

class Solution:
    def lenOfVDiagonal(self, grid: List[List[int]]) -> int:
        n = len(grid)
        m = len(grid[0])
        W = m + 2
        total = (n + 2) * W

        # Flat padded grid; pad value 3 never matches an expected value (0 or 2).
        g = [3] * total
        for r in range(n):
            base = (r + 1) * W + 1
            g[base:base + m] = grid[r]

        ones = [i for i, v in enumerate(g) if v == 1]
        if not ones:
            return 0

        nxt = (2, 2, 0, -1)   # next expected value indexed by current value
        E = [nxt[v] for v in g]

        dirs = ((1, 1), (1, -1), (-1, -1), (-1, 1))  # clockwise order

        # dp1[d][i]: max length of valid continuation starting at cell i,
        # heading in direction d, with the turn already used.
        t1s = []
        for dr, dc in dirs:
            t = [1] * total
            for r in (range(n, 0, -1) if dr == 1 else range(1, n + 1)):
                base = r * W
                nb = base + dr * W + 1 + dc
                e_row = E[base + 1: base + m + 1]
                g_row = g[nb: nb + m]
                t_row = t[nb: nb + m]
                t[base + 1: base + m + 1] = [
                    tn + 1 if gn == e else 1
                    for e, gn, tn in zip(e_row, g_row, t_row)
                ]
            t1s.append(t)

        ans = 0
        for d in range(4):
            dr, dc = dirs[d]
            dr2, dc2 = dirs[(d + 1) & 3]
            t2 = t1s[(d + 1) & 3]
            t = [1] * total
            for r in (range(n, 0, -1) if dr == 1 else range(1, n + 1)):
                base = r * W
                nb = base + dr * W + 1 + dc
                nb2 = base + dr2 * W + 1 + dc2
                e_row = E[base + 1: base + m + 1]
                g_s = g[nb: nb + m]
                t_s = t[nb: nb + m]
                g_w = g[nb2: nb2 + m]
                t_w = t2[nb2: nb2 + m]
                t[base + 1: base + m + 1] = [
                    1 + max(ts if gs == e else 0, tw if gw == e else 0)
                    for e, gs, ts, gw, tw in zip(e_row, g_s, t_s, g_w, t_w)
                ]
            cand = max(map(t.__getitem__, ones))
            if cand > ans:
                ans = cand
        return ans

from typing import List

class Solution:
    def lenOfVDiagonal(self, grid: List[List[int]]) -> int:
        n = len(grid)
        m = len(grid[0])
        W = m + 2
        total = (n + 2) * W

        # Flat padded grid (pad value 3 never matches an expected value 0/2).
        g = [3] * total
        for r in range(n):
            base = (r + 1) * W + 1
            g[base:base + m] = grid[r]

        ones = [i for i, v in enumerate(g) if v == 1]
        if not ones:
            return 0

        # next expected sequence value, indexed by current cell value
        nxt = (2, 2, 0, -1)
        E = [nxt[v] for v in g]

        # diagonal directions in clockwise order; (d+1)&3 is the clockwise turn of d
        dirs = ((1, 1), (1, -1), (-1, -1), (-1, 1))

        # dp1[d][i]: best continuation length starting at cell i heading d,
        # turn already used (straight moves only). Cell i itself is counted.
        t1s = []
        for dr, dc in dirs:
            t = [1] * total
            for r in (range(n, 0, -1) if dr == 1 else range(1, n + 1)):
                base = r * W
                nb = base + dr * W + 1 + dc
                e_row = E[base + 1: base + m + 1]
                g_row = g[nb: nb + m]
                t_row = t[nb: nb + m]
                t[base + 1: base + m + 1] = [
                    tn + 1 if gn == e else 1
                    for e, gn, tn in zip(e_row, g_row, t_row)
                ]
            t1s.append(t)

        ans = 0
        for d in range(4):
            dr, dc = dirs[d]
            dr2, dc2 = dirs[(d + 1) & 3]
            t2 = t1s[(d + 1) & 3]
            t = [1] * total
            for r in (range(n, 0, -1) if dr == 1 else range(1, n + 1)):
                base = r * W
                nb = base + dr * W + 1 + dc     # straight neighbor slice start
                nb2 = base + dr2 * W + 1 + dc2  # turned neighbor slice start
                e_row = E[base + 1: base + m + 1]
                g_s = g[nb: nb + m]
                t_s = t[nb: nb + m]
                g_w = g[nb2: nb2 + m]
                t_w = t2[nb2: nb2 + m]
                t[base + 1: base + m + 1] = [
                    1 + max(ts if gs == e else 0, tw if gw == e else 0)
                    for e, gs, ts, gw, tw in zip(e_row, g_s, t_s, g_w, t_w)
                ]
            cand = max(map(t.__getitem__, ones))
            if cand > ans:
                ans = cand
        return ans

from typing import List


class Solution:
    def lenOfVDiagonal(self, grid: List[List[int]]) -> int:
        n = len(grid)
        m = len(grid[0])
        W = m + 2
        total = (n + 2) * W

        # Flat padded grid; pad value 3 never equals an expected value (0 or 2).
        g = [3] * total
        for r in range(n):
            base = (r + 1) * W + 1
            g[base:base + m] = grid[r]

        # All possible starting cells (value 1). Pads can never be 1.
        ones = [i for i, v in enumerate(g) if v == 1]
        if not ones:
            return 0

        # Expected next value in the sequence 1,2,0,2,0,... indexed by current value.
        nxt = (2, 2, 0, -1)
        E = [nxt[v] for v in g]

        # Diagonal directions in clockwise order: SE, SW, NW, NE.
        # The clockwise 90-degree turn of direction d is (d + 1) & 3.
        dirs = ((1, 1), (1, -1), (-1, -1), (-1, 1))

        # dp1[d][i]: longest valid continuation starting at cell i, heading in
        # direction d, with the single allowed turn already used (straight only).
        t1s = []
        for dr, dc in dirs:
            t = [1] * total
            for r in (range(n, 0, -1) if dr == 1 else range(1, n + 1)):
                base = r * W
                nb = base + dr * W + 1 + dc  # neighbor cells (row r+dr, cols 1+dc..m+dc)
                e_row = E[base + 1: base + m + 1]
                g_row = g[nb: nb + m]
                t_row = t[nb: nb + m]
                t[base + 1: base + m + 1] = [
                    tn + 1 if gn == e else 1
                    for e, gn, tn in zip(e_row, g_row, t_row)
                ]
            t1s.append(t)

        ans = 0
        for d in range(4):
            dr, dc = dirs[d]
            d2 = (d + 1) & 3
            dr2, dc2 = dirs[d2]
            t2 = t1s[d2]
            # dp0[d][i]: longest valid continuation starting at cell i, heading
            # in direction d, turn not used yet.
            t = [1] * total
            for r in (range(n, 0, -1) if dr == 1 else range(1, n + 1)):
                base = r * W
                nb = base + dr * W + 1 + dc    # straight neighbor
                nb2 = base + dr2 * W + 1 + dc2  # neighbor after clockwise turn
                e_row = E[base + 1: base + m + 1]
                g_s = g[nb: nb + m]
                t_s = t[nb: nb + m]
                g_w = g[nb2: nb2 + m]
                t_w = t2[nb2: nb2 + m]
                t[base + 1: base + m + 1] = [
                    1 + max(ts if gs == e else 0, tw if gw == e else 0)
                    for e, gs, ts, gw, tw in zip(e_row, g_s, t_s, g_w, t_w)
                ]
            cand = max(map(t.__getitem__, ones))
            if cand > ans:
                ans = cand
        return ans
