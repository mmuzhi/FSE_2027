```python
import heapq
from typing import List

class Solution:
    def minOperations(self, nums: List[int], x: int, k: int) -> int:
        n = len(nums)
        # sliding window median costs
        # cost[i]: cost to equalize nums[i..i+x-1]
        num_windows = n - x + 1
        cost = [0] * num_windows
        
        LOW = []  # max-heap (negated)
        HIGH = []  # min-heap
        deadLow = {}
        deadHigh = {}
        lowValid = 0
        highValid = 0
        sumLow = 0
        sumHigh = 0
        
        def cleanLow():
            while LOW and deadLow.get(-LOW[0], 0) > 0:
                v = -heapq.heappop(LOW)
                deadLow[v] -= 1
                if deadLow[v] == 0:
                    del deadLow[v]
        
        def cleanHigh():
            while HIGH and deadHigh.get(HIGH[0], 0) > 0:
                v = heapq.heappop(HIGH)
                deadHigh[v] -= 1
                if deadHigh[v] == 0:
                    del deadHigh[v]
        
        def rebalance():
            nonlocal lowValid, highValid, sumLow, sumHigh
            if lowValid > highValid + 1:
                cleanLow()
                v = -heapq.heappop(LOW)
                heapq.heappush(HIGH, v)
                lowValid -= 1
                highValid += 1
                sumLow -= v
                sumHigh += v
            elif highValid > lowValid:
                cleanHigh()
                v = heapq.heappop(HIGH)
                heapq.heappush(LOW, -v)
                highValid -= 1
                lowValid += 1
                sumHigh -= v
                sumLow += v
        
        def insert(v):
            nonlocal lowValid, highValid, sumLow, sumHigh
            cleanLow()
            if lowValid == 0 or v <= -LOW[0]:
                heapq.heappush(LOW, -v)
                lowValid += 1
                sumLow += v
            else:
                heapq.heappush(HIGH, v)
                highValid += 1
                sumHigh += v
            rebalance()
        
        def remove(v):
            nonlocal lowValid, highValid, sumLow, sumHigh
            cleanLow()
            if lowValid > 0 and v <= -LOW[0]:
                deadLow[v] = deadLow.get(v, 0) + 1
                lowValid -= 1
                sumLow -= v
            else:
                deadHigh[v] = deadHigh.get(v, 0) + 1
                highValid -= 1
                sumHigh -= v
            rebalance()
        
        # initial window [0, x-1]
        for i in range(x):
            insert(nums[i])
        
        INF = float('inf')
        for start in range(num_windows):
            # compute cost for current window [start, start+x-1]
            cleanLow()
            m = -LOW[0]  # lowValid >= 1 since window size >= 2... actually >=1
            cost[start] = (sumHigh - highValid * m) + (lowValid * m - sumLow)
            # slide: remove nums[start], insert nums[start+x] if exists
            if start + x < n:
                remove(nums[start])
                insert(nums[start + x])
        
        # DP
        # dp[j][i]: min cost selecting j subarrays, last starts at i
        # optimize space: keep previous dp array
        prev = cost[:]  # j = 1
        for j in range(2, k + 1):
            cur = [INF] * num_windows
            # prefix min of prev
            # cur[i] = cost[i] + min(prev[0..i-x])
            # compute prefix min array
            pm = [0] * num_windows
            pm[0] = prev[0]
            for i in range(1, num_windows):
                pm[i] = prev[i] if prev[i] < pm[i-1] else pm[i-1]
            for i in range(x, num_windows):
                if pm[i - x] < INF:
                    cur[i] = cost[i] + pm[i - x]
            prev = cur
        
        return min(prev)
```

```python
import heapq
from itertools import accumulate
from typing import List

class Solution:
    def minOperations(self, nums: List[int], x: int, k: int) -> int:
        n = len(nums)
        W = n - x + 1

        LOW = []      # max-heap (negated values): lower half (valid part)
        HIGH = []     # min-heap: upper half (valid part)
        deadLow = {}  # pending deletions in LOW by value
        deadHigh = {} # pending deletions in HIGH by value
        state = [0, 0, 0, 0]  # lowCnt, highCnt, sumLow, sumHigh  -- or use nonlocal

        lowCnt = 0; highCnt = 0; sumLow = 0; sumHigh = 0

        def _clean_low():
            while LOW:
                t = -LOW[0]
                d = deadLow.get(t, 0)
                if d:
                    heapq.heappop(LOW)
                    if d == 1:
                        del deadLow[t]
                    else:
                        deadLow[t] = d - 1
                else:
                    break

        def _clean_high():
            while HIGH:
                t = HIGH[0]
                d = deadHigh.get(t, 0)
                if d:
                    heapq.heappop(HIGH)
                    if d == 1:
                        del deadHigh[t]
                    else:
                        deadHigh[t] = d - 1
                else:
                    break

        def insert(v):
            nonlocal lowCnt, highCnt, sumLow, sumHigh
            _clean_low()
            if lowCnt == 0 or v <= -LOW[0]:
                heapq.heappush(LOW, -v)
                lowCnt += 1
                sumLow += v
            else:
                heapq.heappush(HIGH, v)
                highCnt += 1
                sumHigh += v
            _rebalance()

        def remove(v):
            nonlocal lowCnt, highCnt, sumLow, sumHigh
            _clean_low()
            if lowCnt > 0 and v <= -LOW[0]:
                deadLow[v] = deadLow.get(v, 0) + 1
                lowCnt -= 1
                sumLow -= v
            else:
                deadHigh[v] = deadHigh.get(v, 0) + 1
                highCnt -= 1
                sumHigh -= v
            _rebalance()

        def _rebalance():
            nonlocal lowCnt, highCnt, sumLow, sumHigh
            if lowCnt > highCnt + 1:
                _clean_low()
                t = -heapq.heappop(LOW)
                heapq.heappush(HIGH, t)
                lowCnt -= 1
                highCnt += 1
                sumLow -= t
                sumHigh += t
```

```python
import heapq
from itertools import accumulate
from typing import List

class Solution:
    def minOperations(self, nums: List[int], x: int, k: int) -> int:
        n = len(nums)
        W = n - x + 1

        LOW = []       # max-heap (negated): lower half
        HIGH = []      # min-heap: upper half
        deadLow = {}   # pending deletions in LOW (by value)
        deadHigh = {}  # pending deletions in HIGH (by value)
        lowCnt = 0
        highCnt = 0
        sumLow = 0
        sumHigh = 0

        def cleanLow():
            while LOW:
                v = -LOW[0]
                d = deadLow.get(v, 0)
                if not d:
                    break
                heapq.heappop(LOW)
                if d == 1:
                    del deadLow[v]
                else:
                    deadLow[v] = d - 1

        def cleanHigh():
            while HIGH:
                v = HIGH[0]
                d = deadHigh.get(v, 0)
                if not d:
                    break
                heapq.heappop(HIGH)
                if d == 1:
                    del deadHigh[v]
                else:
                    deadHigh[v] = d - 1

        def rebalance():
            nonlocal lowCnt, highCnt, sumLow, sumHigh
            if lowCnt > highCnt + 1:
                cleanLow()
                v = -heapq.heappop(LOW)
                heapq.heappush(HIGH, v)
                lowCnt -= 1
                highCnt += 1
                sumLow -= v
                sumHigh += v
            elif highCnt > lowCnt:
                cleanHigh()
                v = heapq.heappop(HIGH)
                heapq.heappush(LOW, -v)
                highCnt -= 1
                lowCnt += 1
                sumHigh -= v
                sumLow += v

                lowCnt += 1
                sumLow += v
            else:

                highCnt += 1
                sumHigh += v

            cleanLow()

                sumLow -= v
```

=== ANSWER ===
Looking at this problem:

1. **Cost per window**: For a subarray of size `x` starting at `i`, the minimal operations to make all elements equal is achieved by targeting the median — cost = Σ|nums[j] − median|. Compute this for all windows via a sliding-window median using two heaps with lazy deletion (tracking valid counts and sums of each half).

2. **DP**: Choose `k` window starts `i₁ < i₂ < ...` with `i_{m+1} ≥ i_m + x`, minimizing total cost. `dp[j][i] = cost[i] + min(dp[j-1][i' ≤ i-x])`, solved with prefix minima (`k ≤ 15`, so O(nk)).

The two-heap scheme maintains: physical heap = logical multiset ⊎ dead marks, all lower-half elements ≤ all upper-half elements, lower size ∈ {upper, upper+1}. Removal marks the value dead on the side determined by comparing with the current median (proven to be the side actually containing it), and tops are cleaned before any read. I verified the machinery by hand on both examples and several duplicate-heavy traces (costs `[11,0,0,3,4]` → answer 3; costs `[7,5,6,4,4,3,7]` → answer 8).

```python
import heapq
from itertools import accumulate
from typing import List


class Solution:
    def minOperations(self, nums: List[int], x: int, k: int) -> int:
        n = len(nums)
        W = n - x + 1

        LOW = []       # max-heap (stored negated): lower half
        HIGH = []      # min-heap: upper half
        deadLow = {}   # pending deletions inside LOW, by value
        deadHigh = {}  # pending deletions inside HIGH, by value
        lowCnt = 0
        highCnt = 0
        sumLow = 0
        sumHigh = 0

        def cleanLow():
            while LOW:
                v = -LOW[0]
                d = deadLow.get(v)
                if not d:
                    break
                heapq.heappop(LOW)
                if d == 1:
                    del deadLow[v]
                else:
                    deadLow[v] = d - 1

        def cleanHigh():
            while HIGH:
                v = HIGH[0]
                d = deadHigh.get(v)
                if not d:
                    break
                heapq.heappop(HIGH)
                if d == 1:
                    del deadHigh[v]
                else:
                    deadHigh[v] = d - 1

        def rebalance():
            nonlocal lowCnt, highCnt, sumLow, sumHigh
            if lowCnt > highCnt + 1:
                cleanLow()
                v = -heapq.heappop(LOW)
                heapq.heappush(HIGH, v)
                lowCnt -= 1
                highCnt += 1
                sumLow -= v
                sumHigh += v
            elif highCnt > lowCnt:
                cleanHigh()
                v = heapq.heappop(HIGH)
                heapq.heappush(LOW, -v)
                highCnt -= 1
                lowCnt += 1
                sumHigh -= v
                sumLow += v

        def insert(v):
            nonlocal lowCnt, highCnt, sumLow, sumHigh
            cleanLow()
            if lowCnt == 0 or v <= -LOW[0]:
                heapq.heappush(LOW, -v)
                lowCnt += 1
                sumLow += v
            else:
                heapq.heappush(HIGH, v)
                highCnt += 1
                sumHigh += v
            rebalance()

        def remove(v):
            nonlocal lowCnt, highCnt, sumLow, sumHigh
            cleanLow()
            if lowCnt > 0 and v <= -LOW[0]:
                deadLow[v] = deadLow.get(v, 0) + 1
                lowCnt -= 1
                sumLow -= v
            else:
                deadHigh[v] = deadHigh.get(v, 0) + 1
                highCnt -= 1
                sumHigh -= v
            rebalance()

        for i in range(x):
            insert(nums[i])

        cost = [0] * W
        for s in range(W):
            cleanLow()
            m = -LOW[0]
            cost[s] = (sumHigh - highCnt * m) + (lowCnt * m - sumLow)
            if s + x < n:
                remove(nums[s])
                insert(nums[s + x])

        INF = 1 << 62
        prev = cost
        for _ in range(k - 1):
            pm = list(accumulate(prev, min))
            prev = [INF] * x + [c + p for c, p in zip(cost[x:], pm)]
        return min(prev)
```
