import heapq
import sys

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    a = list(map(int, data[1:1+n]))
    
    if n % 2 == 0:
        a.sort()
        half = n // 2
        ans = sum(a[half:]) - sum(a[:half])
        print(ans)
        return
    
    # odd n: compute g[2k] for prefixes and h[2k] for suffixes
    # g[k] for k = 0..(n-1)/2: value on prefix of length 2k
    m = (n - 1) // 2
    
    # prefix: iterate elements a[0..2m-1]; after processing 2k elements record g_k
    # maintain top heap (min-heap of top elements) and bot heap (max-heap, store negatives)
    g = [0] * (m + 1)
    top_heap = []  # min-heap
    bot_heap = []  # max-heap via negatives
    sum_top = 0
    sum_bot = 0
    idx = 0
    for k in range(1, m + 1):
        for _ in range(2):
            z = a[idx]; idx += 1
            if top_heap and z >= top_heap[0]:
                heapq.heappush(top_heap, z)
                sum_top += z
            else:
                heapq.heappush(bot_heap, -z)
                sum_bot += z
        # rebalance: want len(top) == len(bot) == k
        if len(top_heap) > k:
            # move len(top)-k smallest of top to bot
            while len(top_heap) > k:
                v = heapq.heappop(top_heap)
                sum_top -= v
                sum_bot += v
                heapq.heappush(bot_heap, -v)
        elif len(top_heap) < k:
            while len(top_heap) < k:
                v = -heapq.heappop(bot_heap)
                sum_bot -= v
                sum_top += v
                heapq.heappush(top_heap, v)
        g[k] = sum_top - sum_bot
    
    # suffix: elements a[n-2m .. n-1] processed from the right
    h = [0] * (m + 1)
    top_heap = []
    bot_heap = []
    sum_top = 0
    sum_bot = 0
    idx = n - 1
    for k in range(1, m + 1):
        for _ in range(2):
            z = a[idx]; idx -= 1
            if top_heap and z >= top_heap[0]:
                heapq.heappush(top_heap, z)
                sum_top += z
            else:
                heapq.heappush(bot_heap, -z)
                sum_bot += z
        if len(top_heap) > k:
            while len(top_heap) > k:
                v = heapq.heappop(top_heap)
                sum_top -= v
                sum_bot += v
                heapq.heappush(bot_heap, -v)
        elif len(top_heap) < k:
            while len(top_heap) < k:
                v = -heapq.heappop(bot_heap)
                sum_bot -= v
                sum_top += v
                heapq.heappush(top_heap, v)
        h[k] = sum_top - sum_bot
    
    # answer: max over u odd (1-indexed): g[(u-1)/2] + h[(n-u)/2]
    # u = 2j+1, j = 0..m: g[j] + h[m - j]
    ans = max(g[j] + h[m - j] for j in range(m + 1))
    print(ans)

main()

import sys, heapq

def solve():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    a = list(map(int, data[1:1+n]))
    
    if n % 2 == 0:
        a.sort()
        half = n >> 1
        print(sum(a[half:]) - sum(a[:half]))
        return
    
    m = (n - 1) >> 1
    
    # prefix values g[k], k=0..m
    g = [0] * (m + 1)
    top = []   # min-heap (values)
    bot = []   # max-heap (negated values)
    st = 0     # sum top
    sb = 0     # sum bot
    idx = 0
    heappush = heapq.heappush
    heappop = heapq.heappop
    for k in range(1, m + 1):
        for _ in range(2):
            z = a[idx]; idx += 1
            if top and z >= top[0]:
                heappush(top, z); st += z
            else:
                heappush(bot, -z); sb += z
        lt = len(top)
        if lt > k:
            while len(top) > k:
                v = heappop(top); st -= v; sb += v
                heappush(bot, -v)
        elif lt < k:
            while len(top) < k:
                v = -heappop(bot); sb -= v; st += v
                heappush(top, v)
        g[k] = st - sb
    
    # suffix values h[k]
    h = [0] * (m + 1)
    top = []; bot = []
    st = 0; sb = 0
    idx = n - 1
    for k in range(1, m + 1):
        for _ in range(2):
            z = a[idx]; idx -= 1
            if top and z >= top[0]:
                heappush(top, z); st += z
            else:
                heappush(bot, -z); sb += z

                v = heappop(top); st -= v; sb += v

        elif lt < k:
            while len(top) < k:
                v = -heappop(bot); sb -= v; st += v
                heappush(top, v)
        h[k] = st - sb
    
    ans = max(g[j] + h[m - j] for j in range(m + 1))
    print(ans)

import heapq

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    a = list(map(int, data[1:1 + n]))

    if n % 2 == 0:
        a.sort()
        half = n >> 1
        print(sum(a[half:]) - sum(a[:half]))
        return

    heappush = heapq.heappush
    heappop = heapq.heappop

    def compute(arr):
        res = [0] * (m + 1)
        top = []  # min-heap: top half elements
        bot = []  # max-heap (negated): bottom half elements
        st = 0
        sb = 0
        i = 0

                z = arr[i]
                i += 1
                if top and z >= top[0]:
                    heappush(top, z)
                    st += z
                else:
                    heappush(bot, -z)
                    sb += z

                st -= v
                sb += v
                heappush(bot, -v)
            while len(top) < k:
                v = -heappop(bot)
                sb -= v
                st += v
                heappush(top, v)
            res[k] = st - sb
        return res

    g = compute(a)           # g[k]: first 2k elements
    h = compute(a[::-1])     # h[k]: last 2k elements

    ans = max(g[j] + h[m - j] for j in range(m + 1))

import sys
import heapq

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    a = list(map(int, data[1:1 + n]))

    if n % 2 == 0:
        a.sort()
        half = n >> 1
        print(sum(a[half:]) - sum(a[:half]))
        return

    m = (n - 1) >> 1
    heappush = heapq.heappush
    heappop = heapq.heappop

    def compute(arr):
        res = [0] * (m + 1)
        top = []   # min-heap: current k largest
        bot = []   # max-heap (negated): current k smallest
        st = 0     # sum of top
        sb = 0     # sum of bottom
        i = 0
        for k in range(1, m + 1):
            for _ in range(2):
                z = arr[i]
                i += 1
                if top and z >= top[0]:
                    heappush(top, z)
                    st += z
                else:
                    heappush(bot, -z)
                    sb += z
            while len(top) > k:
                v = heappop(top)
                st -= v
                sb += v
                heappush(bot, -v)
            while len(top) < k:
                v = -heappop(bot)
                sb -= v
                st += v
                heappush(top, v)
            res[k] = st - sb
        return res

    g = compute(a)           # g[k]: best score pairing first 2k elements
    h = compute(a[::-1])     # h[k]: best score pairing last 2k elements
    print(max(g[j] + h[m - j] for j in range(m + 1)))

main()

=== ANSWER ===
Looking at this problem:

**Key insight**: Each operation's score |a−b| = max−min. So any sequence of operations corresponds to a signing: +1 on each pair's larger element, −1 on the smaller. The pairing induced by the removal process is exactly a **non-crossing matching** (pairs nested or disjoint), and if one element remains (odd N), it is not strictly inside any pair.

Conversely, **any** balanced sign string (equal + and −) is realizable: repeatedly find two *adjacent* opposite signs (must exist in a non-empty balanced string) and remove them — a valid operation sequence. If every + value ≥ every − value, each such operation scores exactly (+value)−(−value).

- **Even N**: answer = (sum of top N/2) − (sum of bottom N/2), since any balanced signing is achievable, and any process induces a balanced signing.
- **Odd N**: remaining element at odd position u; pairs stay strictly left/right of u, each side perfectly matched with balanced signs. So answer = max over odd u of g(u−1) + h(N−u), where g(L) = top-half sum − bottom-half sum of prefix of length L (similarly h for suffix). Computed for all even prefix/suffix lengths with two heaps (top-k / bottom-k with rebalancing).

Verified against all samples and several hand brute-forced cases (e.g., [1,1,5,5,5,5]→8, N=5 cases, N=7 cases).
