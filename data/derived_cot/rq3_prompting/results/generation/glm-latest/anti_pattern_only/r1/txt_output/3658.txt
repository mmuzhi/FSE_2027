        n = len(nums)
        
        # Max diff between adjacent known elements
        D0 = 0
        for i in range(n - 1):
            if nums[i] != -1 and nums[i + 1] != -1:
                d = abs(nums[i] - nums[i + 1])
                if d > D0:
                    D0 = d
        
        # Collect maximal runs of -1s
        runs = []
        idx = 0
        while idx < n:
            if nums[idx] == -1:
                start = idx
                while idx < n and nums[idx] == -1:
                    idx += 1
                runs.append((start, idx - 1))
            else:
                idx += 1
