import sys
import numpy as np

def main():
    data = sys.stdin.buffer.read().split()
    N = int(data[0]); M = int(data[1])
    B = [int(x) for x in data[2:2+N]]
    MOD = 998244353

    # Precompute avec and bvec for each distinct value? Values: -1 or 1..M.
    # avec(v): length M+1: free: M-k for k=0..M; fixed v: 1 for k<v.
    # bvec(v): length M: free: k for k=0..M-1; fixed v: 1 for k>=v.
    
    # We can cache by value.
    avec_cache = {}
    def get_avec(v):
        if v not in avec_cache:
            if v == -1:
                arr = np.arange(M, -1, -1, dtype=np.int64)
            else:
                arr = np.zeros(M+1, dtype=np.int64)
                if v > 0:
                    arr[:v] = 1
            avec_cache[v] = arr
        return avec_cache[v]
    
    bvec_cache = {}
    def get_bvec(v):
