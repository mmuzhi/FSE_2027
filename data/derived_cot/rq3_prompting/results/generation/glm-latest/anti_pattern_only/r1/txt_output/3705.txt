```python
class Solution:
    def largestInteger(self, nums: List[int], k: int) -> int:
        n = len(nums)
        count = {}  # value -> number of distinct subarrays containing it
        for i in range(n - k + 1):
            window = set(nums[i:i+k])
            for x in window:
                count[x] = count.get(x, 0) + 1
        ans = -1
        for x, c in count.items():
            if c == 1:
                ans = max(ans, x)
        return ans
```

```python
class Solution:
    def largestInteger(self, nums: List[int], k: int) -> int:
        n = len(nums)
        count = {}
        # Count how many distinct size-k subarrays contain each value
        for i in range(n - k + 1):
            window = set(nums[i:i + k])
            for x in window:
                count[x] = count.get(x, 0) + 1
        # Find the largest value contained in exactly one subarray
        ans = -1
        for x, c in count.items():
            if c == 1 and x > ans:
                ans = x
        return ans
```
