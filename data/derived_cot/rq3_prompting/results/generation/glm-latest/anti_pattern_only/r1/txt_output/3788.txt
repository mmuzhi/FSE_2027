```python
from typing import List

class Solution:
    def maxSum(self, nums: List[int]) -> int:
        # Since we may delete any elements before choosing the subarray,
        # any subsequence of nums can be made into a contiguous subarray
        # (keep exactly those elements, delete the rest).
        # So we only need to choose a set of distinct values maximizing the sum:
        # take every distinct positive value once; if none exist, take the
        # single largest element (the subarray must be non-empty).
        positives = {x for x in nums if x > 0}
        if positives:
            return sum(positives)
        return max(nums)
```
