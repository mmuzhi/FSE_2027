import sys
from bisect import bisect_left

def main():
    data = sys.stdin.buffer.read().split()
    ptr = 0
    N = int(data[ptr]); ptr += 1
    X = int(data[ptr]) - 1; ptr += 1
    A = data[ptr:ptr+N]; ptr += N
    B = data[ptr:ptr+N]; ptr += N
    P = data[ptr:ptr+N]; ptr += N
    Q = data[ptr:ptr+N]; ptr += N
    
    A = [int(x) for x in A]
    B = [int(x) for x in B]
    P = [int(x) - 1 for x in P]
    Q = [int(x) - 1 for x in Q]
    
    Pinv = [0] * N
    for i in range(N):
        Pinv[P[i]] = i
    Qinv = [0] * N
    for i in range(N):
        Qinv[Q[i]] = i
    
    pdist = [-1] * N
    pdist[X] = 0
    cur = Pinv[X]
    d = 1
    while cur != X:
        pdist[cur] = d
        d += 1
        cur = Pinv[cur]
    
    qdist = [-1] * N
    qdist[X] = 0
    cur = Qinv[X]
    d = 1
    while cur != X:
        qdist[cur] = d
        d += 1
        cur = Qinv[cur]
    
    dR = 0
    dB = 0
    for i in range(N):
        if A[i]:
            if pdist[i] < 0:
                sys.stdout.write("-1\n")
                return
            if pdist[i] > dR:
                dR = pdist[i]
        if B[i]:
            if qdist[i] < 0:
                sys.stdout.write("-1\n")
                return
            if qdist[i] > dB:
                dB = qdist[i]
    
    pairs = []
    for i in range(N):
        p = pdist[i]
        if 1 <= p <= dR:
            q = qdist[i]
            if 1 <= q <= dB:
                pairs.append((p, q))
    
    pairs.sort()
    tails = []
    for _, q in pairs:
        pos = bisect_left(tails, q)
        if pos == len(tails):
            tails.append(q)
        else:
            tails[pos] = q
    
    sys.stdout.write(str(dR + dB - len(tails)) + "\n")

main()
