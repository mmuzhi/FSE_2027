```python
import sys
from collections import deque

def main():
    input_data = sys.stdin.read().split()
    idx = 0
    N = int(input_data[idx]); idx+=1
    M = int(input_data[idx]); idx+=1
    K = int(input_data[idx]); idx+=1
    edges = []
    for _ in range(M):
        u = int(input_data[idx])-1; idx+=1
        v = int(input_data[idx])-1; idx+=1
        edges.append((u,v))
    s = 0
    t = N-1
    
    INF = 10**9
    
    # Dinic
    class Dinic:
        def __init__(self, n):
            self.n = n
            self.graph = [[] for _ in range(n)]
        def add_edge(self, u, v, cap):
            self.graph[u].append([v, cap, len(self.graph[v])])
            self.graph[v].append([u, 0, len(self.graph[u])-1])
        def max_flow(self, s, t):
            flow = 0
            while True:
                # BFS level
                level = [-1]*self.n
                level[s] = 0
                q = deque([s])
                while q:
                    u = q.popleft()
                    for e in self.graph[u]:
                        if e[1] > 0 and level[e[0]] == -1:
                            level[e[0]] = level[u]+1
                            q.append(e[0])
                if level[t] == -1:
                    return flow
                it = [0]*self.n
                # DFS
                def dfs(u, f):
                    if u == t:
                        return f
                    while it[u] < len(self.graph[u]):
                        e = self.graph[u][it[u]]
                        v = e[0]
                        if e[1] > 0 and level[v] == level[u]+1:
                            ret = dfs(v, min(f, e[1]))
                            if ret > 0:
                                e[1] -= ret
                                self.graph[v][e[2]][1] += ret
                                return ret
                        it[u] += 1
                    return 0
                while True:
                    f = dfs(s, INF)
                    if f == 0:
                        break
                    flow += f
            return flow
    
    def compute_f(d):
        # returns min cut value (min #down edges) for level count d
        # nodes: (v, i) for i in 1..d -> id = v*d + (i-1); source = N*d, sink = N*d+1
        if d == 0:
            return 0
        n_nodes = N*d + 2
        S = N*d
        T = N*d + 1
        din = Dinic(n_nodes)
        def node(v, i):  # i in 1..d
            return v*d + (i-1)
        # force x_{s,d} = 1
        din.add_edge(S, node(s, d), INF)
        # force x_{t,1} = 0
        din.add_edge(node(t, 1), T, INF)
        # chains
        for v in range(N):
            for i in range(1, d):
                din.add_edge(node(v, i+1), node(v, i), INF)
        # implications and penalties
        for (u, v) in edges:
            for i in range(1, d):
                din.add_edge(node(u, i+1), node(v, i), INF)
            for i in range(1, d+1):
                din.add_edge(node(u, i), node(v, i), 1)
        return din.max_flow(S, T)
    
    ans = 0
    for d in range(1, N):  # d up to N-1
        if compute_f(d) <= K:
            ans = d
        else:
            break
    print(ans)

main()
```
