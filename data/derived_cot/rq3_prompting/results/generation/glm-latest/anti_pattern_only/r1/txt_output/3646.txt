```python
class Solution:
    def sumOfGoodSubsequences(self, nums: List[int]) -> int:
        MOD = 10**9 + 7
        # count[v]: number of good subsequences ending with value v
        # total[v]: sum of elements of good subsequences ending with value v
        count = [0] * 100002
        total = [0] * 100002
        ans = 0
        for x in nums:
            # subsequences ending at previous elements with value x-1 (if exists)
            c_low = count[x - 1] if x > 0 else 0
            t_low = total[x - 1] if x > 0 else 0
            # subsequences ending at previous elements with value x+1
            c_high = count[x + 1]
            t_high = total[x + 1]
            # new subsequences ending at this occurrence of x:
            # [x] itself, plus every subsequence ending in x-1 or x+1 extended by x
            nc = (1 + c_low + c_high) % MOD
            ns = (x * nc + t_low + t_high) % MOD
            count[x] = (count[x] + nc) % MOD
            total[x] = (total[x] + ns) % MOD
            ans = (ans + ns) % MOD
        return ans
```
