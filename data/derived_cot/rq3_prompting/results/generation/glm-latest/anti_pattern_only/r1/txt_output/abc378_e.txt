def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    m = int(data[1])

    # prefix sums mod m
    s = [0] * (n + 1)
    cur = 0
    for i in range(n):
        cur = (cur + a[i]) % m
        s[i+1] = cur

    first = 0

        first += s[k] * (2*k - n)
    
    # inversions via BIT
    # BIT over values 0..m-1
    tree = [0] * (m + 1)
    
    def update(pos):
        # pos in [0, m-1], shift to 1-indexed
        pos += 1
        while pos <= m:
            tree[pos] += 1
            pos += pos & (-pos)
    
    def query(pos):
        # sum of counts for values [0, pos] (0-indexed values)
        pos += 1
        res = 0
        while pos > 0:
            res += tree[pos]
            pos -= pos & (-pos)
        return res
    
    inv = 0

        inv += j - le
        update(s[j])
    
    ans = first + m * inv

    cur = 0
    idx = 2

        cur = (cur + int(input_data[idx])) % m
        idx += 1
        s[i] = cur
    
    # first term: sum s[k] * (2k - n)
    first = 0
    for k in range(n+1):
        first += s[k] * (2*k - n)
    
    # inversions with BIT
    tree = [0]*(m+1)
    inv = 0
    for j in range(n+1):
        v = s[j]
        # query count of values <= v: prefix sum up to v+1 (1-indexed pos v+1)
        pos = v + 1
        res = 0
        while pos > 0:
            res += tree[pos]
            pos -= pos & (-pos)
        inv += j - res
        # update at pos v+1
        pos = v + 1
        while pos <= m:
            tree[pos] += 1
            pos += pos & (-pos)
    
    print(first + m * inv)

main()

    n = len(s)

import sys

def count_inversions_bit(s, m):
    """Count pairs i<j with s[i] > s[j] using a Fenwick tree. s values in [0, m)."""
    tree = [0] * (m + 1)
    non_inv = 0
    for v in s:
        pos = v + 1
        while pos:
            non_inv += tree[pos]
            pos &= pos - 1
        pos = v + 1
        while pos <= m:
            tree[pos] += 1
            pos += pos & -pos
    total = (len(s)) * (len(s) - 1) // 2
    return total - non_inv

def count_inversions_np(s_list, m):
    import numpy as np
    s = np.array(s_list, dtype=np.int64)
    n = s.shape[0]
    size = 1
    while size < n:
        size <<= 1
    a = np.full(size, m, dtype=np.int64)  # sentinel m > any real value
    a[:n] = s
    K = m + 1
    inv = 0
    w = 1
    while w < size:
        half = size // 2  # not needed
        num_pairs = size // (2 * w)
        a2 = a.reshape(num_pairs, 2 * w)
        off = np.arange(num_pairs, dtype=np.int64).reshape(-1, 1) * K
        Lg = (a2[:, :w] + off).ravel()
        Rg = (a2[:, w:] + off).ravel()
        pos = np.searchsorted(Lg, Rg, side='right')
        row_starts = np.repeat(np.arange(num_pairs, dtype=np.int64), w)
        inv += int((w - (pos - row_starts)).sum())
        a2.sort(axis=1)
        w <<= 1
    return inv

    # s: list of values in [0, m); count pairs i<j with s[i] > s[j]

    non_inv = 0

import sys


def count_inversions_bit(s, m):
    """Count pairs i < j with s[i] > s[j] (values in [0, m)) via Fenwick tree."""
    tree = [0] * (m + 1)
    non_inv = 0
    for v in s:
        pos = v + 1
        while pos:
            non_inv += tree[pos]
            pos &= pos - 1
        pos = v + 1
        while pos <= m:
            tree[pos] += 1
            pos += pos & -pos
    cnt = len(s)
    total = cnt * (cnt - 1) // 2
    return total - non_inv


def count_inversions_np(s, m):
    """Same as above, vectorized bottom-up merge sort with numpy."""
    import numpy as np
    arr = np.array(s, dtype=np.int64)
    n = arr.shape[0]
    size = 1
    while size < n:
        size <<= 1
    a = np.full(size, m, dtype=np.int64)  # sentinel m > every real value
    a[:n] = arr
    K = m + 1
    inv = 0
    w = 1
    while w < size:
        num_pairs = size // (2 * w)
        a2 = a.reshape(num_pairs, 2 * w)
        off = np.arange(num_pairs, dtype=np.int64).reshape(-1, 1) * K
        Lg = (a2[:, :w] + off).ravel()   # globally sorted
        Rg = (a2[:, w:] + off).ravel()
        pos = np.searchsorted(Lg, Rg, side='right')
        row_starts = np.repeat(np.arange(num_pairs, dtype=np.int64), w)
        inv += int((w - (pos - row_starts)).sum())
        a2.sort(axis=1)
        w <<= 1
    return inv


def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    m = int(data[1])

    # prefix sums modulo m: s[0..n]
    s = [0] * (n + 1)
    cur = 0
    i = 1
    for x in map(int, data[2:2 + n]):
        cur += x
        if cur >= m:
            cur %= m
        s[i] = cur
        i += 1

    # answer = sum_{i<j} (s[j] - s[i]) + M * #inversions
    #        = sum_k s[k]*(2k - n) + M * inv
    weighted = 0
    total_v = 0
    for k, v in enumerate(s):
        weighted += k * v
        total_v += v
    first = 2 * weighted - n * total_v

    try:
        inv = count_inversions_np(s, m)
    except Exception:
        inv = count_inversions_bit(s, m)

    print(first + m * inv)


main()
