def failure(t):

import sys

def find_a_kmp(S, n):
    R = S[::-1]
    t = R + b"\x00" + S
    m = len(t)
    f = [0] * m
    k = 0
    for i in range(1, m):
        c = t[i]
        while k and c != t[k]:
            k = f[k - 1]
        if c == t[k]:
            k += 1
        f[i] = k
    return n - f[m - 1]

def build_pows(np, base, mod, n):
    pw = np.ones(n, dtype=np.int64)
    if n <= 1:
        return pw
    b = base % mod
    pw[1] = b
    c = 2
    while c < n:
        pw[c] = pw[c - 1] * b % mod
        seg = min(2 * c, n)
        if seg > c + 1:
            pw[c + 1:seg] = pw[1:seg - c] * pw[c] % mod
        c = seg
    return pw

def find_a_numpy(S, n):
    import numpy as np
    R = S[::-1]
    arrS = np.frombuffer(S, dtype=np.uint8).astype(np.int64)
    arrR = np.frombuffer(R, dtype=np.uint8).astype(np.int64)
    mods = (2147483647, 2147483629)
    import random
    bases = (random.randrange(300, 2147483647 - 300), random.randrange(300, 2147483629 - 300))

        invb = pow(base, mod - 2, mod)

def find_a_kmp(S, n):
    t = S[::-1] + b"\x00" + S
    m = len(t)
    f = [0] * m
    k = 0
    for i, c in enumerate(t[1:], 1):
        while k and c != t[k]:
            k = f[k - 1]
        if c == t[k]:
            k += 1
        f[i] = k
    return n - f[m - 1]
