import sys
from collections import deque

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0]); m = int(data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    edges_into_1 = []
    for _ in range(m):
        a = int(data[idx]); b = int(data[idx+1]); idx += 2
        adj[a].append(b)
        if b == 1:
            edges_into_1.append(a)
    dist = [-1] * (n + 1)
    dist[1] = 0
    q = deque([1])
    while q:
        u = q.popleft()
        for v in adj[u]:
            if dist[v] == -1:
                dist[v] = dist[u] + 1
                q.append(v)
    ans = -1
    for a in edges_into_1:
        d = dist[a]
        if d != -1:
            if ans == -1 or d + 1 < ans:
                ans = d + 1
    print(ans)

main()
