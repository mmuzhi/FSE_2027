```python
from heapq import heappush, heappop, heapreplace

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    K = int(data[1])
```

```python
import sys
from heapq import heappush, heappop, heapreplace

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    K = int(data[1])
    A = sorted(map(int, data[2:2 + n]), reverse=True)
    B = sorted(map(int, data[2 + n:2 + 2 * n]), reverse=True)
    C = sorted(map(int, data[2 + 2 * n:2 + 3 * n]), reverse=True)
    del data

    a0 = A[0]
    b0 = B[0]
    c0 = C[0]
```
```python
    nm1 = n - 1
    DA = [A[t] - A[t + 1] for t in range(nm1)]

    ab0 = a0 * b0
    sab0 = a0 + b0

    heap = [(-(c0 * sab0 + ab0), 0, 0, 0)]

            if not j:
                k1 = k + 1
                if k1 < n:

    print(-nv)
```

Looking at this problem:

- V(i,j,k) = A_iB_j + B_jC_k + C_kA_i is non-increasing in each index when A, B, C are sorted descending (e.g., V(i+1,j,k)−V(i,j,k) = (A_{i+1}−A_i)(B_j+C_k) ≤ 0).
- Since K ≤ 5×10⁵, use a max-heap (priority queue) starting from (0,0,0), popping K times. Use a **spanning tree** over states so each state is pushed exactly once (no visited set): popping (i,j,k) pushes (i+1,j,k) always; (i,j+1,k) only if i==0; (i,j,k+1) only if i==0 and j==0. Since each state's unique parent has value ≥ its own, and children are pushed when the parent pops, the standard argument shows pops occur in non-increasing value order, so the K-th pop is the answer.

Optimizations: since the rule-1 child (i+1,j,k) has value V − (A[i]−A[i+1])(B[j]+C[k]), precompute D[i]=A[i]−A[i+1] and use `heapreplace` (pop root + push child in one sift) after peeking `heap[0]`.

I verified this exactly reproduces sample 1 (pops: 44, 38, 36, 34, 31 → answer 31) and handles samples 2 and 3 (all-equal values work; big integers fine).

```python
import sys
import gc
from heapq import heappush, heappop, heapreplace

def main():
    gc.disable()
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    K = int(data[1])
    A = sorted(map(int, data[2:2 + n]), reverse=True)
    B = sorted(map(int, data[2 + n:2 + 2 * n]), reverse=True)
    C = sorted(map(int, data[2 + 2 * n:2 + 3 * n]), reverse=True)
    del data

    a0 = A[0]
    nm1 = n - 1
    DA = [A[t] - A[t + 1] for t in range(nm1)]

    b0 = B[0]
    c0 = C[0]
    ab0 = a0 * b0
    sab0 = a0 + b0

    heap = [(-(c0 * sab0 + ab0), 0, 0, 0)]
    push = heappush
    pop = heappop
    replace = heapreplace

    for _ in range(K):
        nv, i, j, k = heap[0]
        if i < nm1:
            replace(heap, (nv + DA[i] * (B[j] + C[k]), i + 1, j, k))
        else:
            pop(heap)
        if not i:
            j1 = j + 1
            if j1 < n:
                c = C[k]
                push(heap, (-(B[j1] * (a0 + c) + a0 * c), 0, j1, k))
            if not j:
                k1 = k + 1
                if k1 < n:
                    push(heap, (-(C[k1] * sab0 + ab0), 0, 0, k1))

    print(-nv)

main()
```
