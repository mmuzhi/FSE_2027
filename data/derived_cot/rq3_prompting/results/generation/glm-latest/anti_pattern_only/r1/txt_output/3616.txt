```python
class Solution:
    def countValidSelections(self, nums: List[int]) -> int:
        # Key insight: starting at a zero position i and moving right,
        # the process strictly alternates: decrement one element on the
        # right side of i, then one on the left side, then right, etc.
        # (positions between i and the current "wall" of positives are
        # always zeros, so we always bounce across i).
        #
        # Let L = sum(nums[0..i-1]), R = sum(nums[i+1..n-1]).
        # Moving right first (decrement sequence R, L, R, L, ...) ends
        # with everything zero iff R == L or R == L + 1.
        # Moving left first (L, R, L, R, ...) is valid iff L == R or L == R + 1.
        #
        # So each zero position contributes:
        #   2 if L == R, 1 if |L - R| == 1, else 0.
        total = sum(nums)
        left = 0
        ans = 0
        for x in nums:
            if x == 0:
                right = total - left  # nums[i] == 0, so this is the suffix sum
                d = right - left
                if d == 0:
                    ans += 2
                elif d == 1 or d == -1:
                    ans += 1
            left += x
        return ans
```
