```python
import sys
import numpy as np

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    s = np.array(data[1:1+n], dtype=np.int64)
    M = 10**6
    f = np.zeros(M + 1, dtype=np.float64)
    f[s] = 1.0
    # convolution size
    L = 1
    while L < 2 * M + 1:
        L <<= 1
    # L = 2^21 = 2097152 ≥ 2000001
    F = np.fft.rfft(f, L)
    c = np.fft.irfft(F * F, L)
    # c[k] ≈ number of ordered pairs summing to k
    c = np.rint(c).astype(np.int64)
    # answer
    ans = int(np.sum((c[2 * s] - 1) // 2))
    print(ans)

main()
```

=== ANSWER ===
Looking at this problem:

- A fine triplet (A, B, C) satisfies A + C = 2B (arithmetic progression), with all three in S.
- For each B ∈ S, the number of fine triplets with middle element B equals the number of unordered pairs {A, C} ⊂ S with A + C = 2B, A ≠ C. If c[k] = number of *ordered* pairs (A,C) ∈ S² with A + C = k, then for B ∈ S: c[2B] = 2·(valid pairs) + 1 (the +1 from (B,B)). So the count for B is (c[2B] − 1)/2.
- c is the self-convolution of the 0/1 indicator array of S over values [0, 10⁶], computable via FFT of size 2²¹.

**Precision check:** counts are ≤ 10⁶; FFT double-precision error for this size/magnitude is ~10⁻⁴ at worst (values ≤ 10⁶, intermediate products ≤ 10¹², well within double range), so rounding to nearest integer is safe by several orders of magnitude.

**Verification on sample 1:** S = {1,2,3,5,8}: c[4]=3 → B=2 gives 1; c[6]=3 → B=3 gives 1; c[10]=3 → B=5 gives 1; others 0. Total = 3 ✓. Samples 2 and 3 also verified by hand enumeration (5 and 10).

Answer can be ~2.5×10¹¹ (when S = {1..10⁶}), fits in int64.

```python
import sys
import numpy as np

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    s = np.fromiter(map(int, data[1:1 + n]), dtype=np.int64, count=n)
    L = 1 << 21  # 2097152 >= 2*10^6 + 1 : linear convolution fits without wraparound
    f = np.zeros(L, dtype=np.float64)
    f[s] = 1.0
    F = np.fft.rfft(f)
    del f
    G = F * F
    del F
    c = np.fft.irfft(G, n=L)  # c[k] = # ordered pairs (A,C) in S with A+C=k
    del G
    # For each B in S: fine triplets with middle B = (c[2B] - 1) / 2
    vals = np.rint(c[2 * s]).astype(np.int64)
    ans = int(((vals - 1) // 2).sum())
    print(ans)

main()
```
