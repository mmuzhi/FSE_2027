    def minMaxWeight(self, n: int, edges: List[List[int]], threshold: int) -> int:
        # Build reversed adjacency sorted by weight
        from collections import deque
        # radj[b].append((w, a)) for edge a->b w  (reversed: b->a)

        for a, b, w in edges:
            radj[b].append((w, a))
        for lst in radj:
            lst.sort()
        
        def ok(W):

            visited[0] = True
            dq = deque([0])
            cnt = 1
            while dq:

                for w, v in radj[u]:

                        break
                    if not visited[v]:
                        visited[v] = True
                        cnt += 1
                        dq.append(v)

        while lo < hi:

            else:

import heapq
class Solution:
    def minMaxWeight(self, n, edges, threshold):
        radj = [[] for _ in range(n)]
        for a, b, w in edges:
            radj[b].append((w, a))
        dist = [float('inf')] * n
        dist[0] = 0
        heap = [(0, 0)]
        while heap:
            d, u = heapq.heappop(heap)
            if d > dist[u]:
                continue
            for w, v in radj[u]:
                nd = max(d, w)
                if nd < dist[v]:
                    dist[v] = nd
                    heapq.heappush(heap, (nd, v))
        ans = max(dist)
        return ans if ans != float('inf') else -1
```

from collections import deque
from typing import List

class Solution:
    def minMaxWeight(self, n: int, edges: List[List[int]], threshold: int) -> int:
        # Reversed adjacency: radj[u] = list of (w, v) meaning reversed edge u -> v with weight w
        radj = [[] for _ in range(n)]
        for a, b, w in edges:
            radj[b].append((w, a))
        for lst in radj:
            lst.sort()
        
        def reachable(W: int) -> bool:
            visited = bytearray(n)
            visited[0] = 1
            cnt = 1
            dq = deque([0])
            while dq:
                u = dq.popleft()
                for w, v in radj[u]:
                    if w > W:
                        break
                    if not visited[v]:
                        visited[v] = 1
                        cnt += 1
                        if cnt == n:
                            return True
                        dq.append(v)
            return cnt == n
        
        weights = sorted({e[2] for e in edges})
        if not reachable(weights[-1]):
            return -1
        lo, hi = 0, len(weights) - 1
        while lo < hi:
            mid = (lo + hi) // 2
            if reachable(weights[mid]):
                hi = mid
            else:
                lo = mid + 1
        return weights[lo]
```

class Solution:
    def minMaxWeight(self, n: int, edges: List[List[int]], threshold: int) -> int:
        # Reverse the graph: an edge a->b (w) becomes b->a (w).
        # "Every node can reach 0" <=> "0 reaches every node" in the reversed graph.

```python
import heapq
from typing import List

class Solution:
    def minMaxWeight(self, n: int, edges: List[List[int]], threshold: int) -> int:
        # Reverse every edge (a -> b, w) to (b -> a, w):
        # "node 0 is reachable from every node" <=> "0 reaches every node" in the
        # reversed graph.
        #
        # Since threshold >= 1, the out-degree limit is never a restriction:
        # whenever every node can reach 0 using edges of weight <= W, we can keep
        # a spanning in-arborescence (each non-root node keeps exactly one
        # outgoing edge, towards 0), so every out-degree is <= 1 <= threshold.
        #
        # Hence the answer is the smallest W such that every node reaches 0 with
        # edges of weight <= W, i.e. the maximum over nodes of the minimum
        # bottleneck path to 0. Compute it with a bottleneck-Dijkstra from 0 in
        # the reversed graph.
        radj = [[] for _ in range(n)]
        for a, b, w in edges:
            radj[b].append((w, a))

        INF = float('inf')
        best = [INF] * n
        best[0] = 0
        heap = [(0, 0)]
        heappush = heapq.heappush
        heappop = heapq.heappop
        while heap:
            d, u = heappop(heap)
            if d > best[u]:
                continue
            for w, v in radj[u]:
                nd = w if w > d else d
                if nd < best[v]:
                    best[v] = nd
                    heappush(heap, (nd, v))

        ans = max(best)
        return -1 if ans == INF else ans
```
