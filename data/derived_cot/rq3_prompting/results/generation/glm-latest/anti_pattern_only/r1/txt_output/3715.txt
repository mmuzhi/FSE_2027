        C = [c[2] for c in coins]
        tot = [(R[i] - L[i] + 1) * C[i] for i in range(n)]
        pre = [0] * (n + 1)
        for i in range(n):
            pre[i+1] = pre[i] + tot[i]
        
        ans = 0

class Solution:
    def maximumCoins(self, coins: List[List[int]], k: int) -> int:
        coins.sort()
        n = len(coins)
        L = [c[0] for c in coins]
        R = [c[1] for c in coins]
        C = [c[2] for c in coins]
        tot = [(R[i] - L[i] + 1) * C[i] for i in range(n)]
        pre = [0] * (n + 1)
        acc = 0
        for i in range(n):
            acc += tot[i]
            pre[i + 1] = acc

        ans = 0

        # Candidate windows starting at a segment's left end
        m = 0
        for i in range(n):
            b = L[i] + k - 1
            if m < i:
                m = i
            while m + 1 < n and L[m + 1] <= b:
                m += 1
            cur = (min(R[i], b) - L[i] + 1) * C[i]
            if m > i:
                cur += pre[m] - pre[i + 1]
                cur += (min(R[m], b) - L[m] + 1) * C[m]
            if cur > ans:
                ans = cur

        # Candidate windows ending at a segment's right end
        p = 0
        for i in range(n):
            a = R[i] - k + 1
            while p < n and R[p] < a:
                p += 1
            if p == i:

                cur = (R[p] - max(L[p], a) + 1) * C[p] + (pre[i] - pre[p + 1]) + tot[i]
            if cur > ans:
                ans = cur

        return ans

=== ANSWER ===
Looking at this problem:

- Segments are non-overlapping; sort them.
- Key insight (exchange argument): an optimal window of length `k` can be shifted so that either its **left edge coincides with some segment's `l`** or its **right edge coincides with some segment's `r`** (if both edges were strictly inside segments, shifting toward the richer side never decreases the sum; a formal proof via the smallest maximizing start position also works). So we only need to evaluate `2n` candidate windows.
- For windows starting at each `l_i`, use a two-pointer `m` = last segment starting at or before `l_i + k - 1`, plus prefix sums of full-segment totals, adding partial overlaps at both ends.
- Symmetrically for windows ending at each `r_i`, pointer `p` = first segment ending at or after `r_i - k + 1`.

Both passes are O(n) after sorting. Verified against both examples (outputs 10 and 6) and several hand-crafted cases (gaps, huge k, k=1, mixed coin values).

```python
from typing import List

class Solution:
    def maximumCoins(self, coins: List[List[int]], k: int) -> int:
        coins.sort()
        n = len(coins)
        L = [c[0] for c in coins]
        R = [c[1] for c in coins]
        C = [c[2] for c in coins]
        tot = [(R[i] - L[i] + 1) * C[i] for i in range(n)]
        pre = [0] * (n + 1)
        acc = 0
        for i in range(n):
            acc += tot[i]
            pre[i + 1] = acc

        ans = 0

        # Candidate windows starting at a segment's left end: [L[i], L[i]+k-1]
        m = 0
        for i in range(n):
            b = L[i] + k - 1
            if m < i:
                m = i
            while m + 1 < n and L[m + 1] <= b:
                m += 1
            cur = (min(R[i], b) - L[i] + 1) * C[i]          # part of segment i
            if m > i:
                cur += pre[m] - pre[i + 1]                   # segments i+1..m-1 fully covered
                cur += (min(R[m], b) - L[m] + 1) * C[m]     # part of segment m
            if cur > ans:
                ans = cur

        # Candidate windows ending at a segment's right end: [R[i]-k+1, R[i]]
        p = 0
        for i in range(n):
            a = R[i] - k + 1
            while p < n and R[p] < a:
                p += 1
            # p <= i always holds since R[i] >= a
            if p == i:
                cur = (R[i] - max(L[i], a) + 1) * C[i]
            else:
                cur = (R[p] - max(L[p], a) + 1) * C[p]       # part of segment p
                cur += pre[i] - pre[p + 1]                   # segments p+1..i-1 fully covered
                cur += tot[i]                                # segment i fully covered
            if cur > ans:
                ans = cur

        return ans
```
