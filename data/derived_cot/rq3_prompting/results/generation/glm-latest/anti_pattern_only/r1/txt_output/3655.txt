import heapq

class Solution:
    def minOperations(self, n: int, m: int) -> int:
        def is_prime(x):
            if x < 2:
                return False
            if x < 4:
                return True
            if x % 2 == 0:
                return False
            i = 3
            while i * i <= x:
                if x % i == 0:
                    return False
                i += 2
            return True
        
        if is_prime(n) or is_prime(m):
            return -1
        
        s = str(n)
        d = len(s)
        lo = 10 ** (d - 1)
        hi = 10 ** d - 1
        
        # Dijkstra
        INF = float('inf')
        dist = {}
        dist[n] = n
        pq = [(n, n)]
        while pq:
            cost, v = heapq.heappop(pq)
            if cost > dist.get(v, INF):
                continue
            if v == m:
                return cost
            vs = str(v)
            for i in range(d):
                digit = int(vs[i])
                for delta in (-1, 1):
                    nd = digit + delta
                    if nd < 0 or nd > 9:
                        continue
                    if i == 0 and nd == 0:
                        continue
                    nv = int(vs[:i] + str(nd) + vs[i+1:])
                    if is_prime(nv):
                        continue
                    ncost = cost + nv
                    if ncost < dist.get(nv, INF):
                        dist[nv] = ncost
                        heapq.heappush(pq, (ncost, nv))
        return -1

class Solution:
    def minOperations(self, n: int, m: int) -> int:
        # Sieve up to 10000
        LIMIT = 10000
        is_comp = [False] * LIMIT
        # We'll build is_prime array
        is_prime = [False, False] + [True] * (LIMIT - 2)
        for i in range(2, int(LIMIT ** 0.5) + 1):
            if is_prime[i]:
                for j in range(i * i, LIMIT, i):
                    is_prime[j] = False
        
        if is_prime[n] or is_prime[m]:
            return -1
        
        d = len(str(n))
        pow10 = [1] * (d + 1)
        for i in range(1, d + 1):
            pow10[i] = pow10[i-1] * 10
        
        # Digits from least significant: position p in [0, d-1], value digit = (v // pow10[p]) % 10
        # Leading digit is position d-1.
        
        dist = {n: n}
        pq = [(n, n)]
        while pq:
            cost, v = heapq.heappop(pq)
            if v == m:
                return cost
            if cost > dist.get(v, float('inf')):
                continue
            for p in range(d):
                digit = (v // pow10[p]) % 10
                for delta in (-1, 1):
                    nd = digit + delta
                    if nd < 0 or nd > 9:
                        continue
                    if p == d - 1 and nd == 0:
                        continue  # no leading zero

                    if is_prime[nv]:
                        continue
                    ncost = cost + nv
                    if ncost < dist.get(nv, float('inf')):
                        dist[nv] = ncost
                        heapq.heappush(pq, (ncost, nv))
        return -1

import heapq

class Solution:
    def minOperations(self, n: int, m: int) -> int:
        N = 10000
        composite = bytearray(N)
        for i in range(2, 100):
            if not composite[i]:
                for j in range(i * i, N, i):
                    composite[j] = 1

        def is_prime(x: int) -> bool:
            return x >= 2 and not composite[x]

        # n must be non-prime at the start; m must be non-prime at the end.
        if is_prime(n) or is_prime(m):
            return -1

        if n == m:
            return n

        d = len(str(n))
        p10 = [10 ** p for p in range(d)]

        # Dijkstra: cost of a path = initial value + value after each operation.
        dist = {n: n}
        heap = [(n, n)]
        while heap:
            cost, v = heapq.heappop(heap)
            if cost > dist[v]:
                continue
            if v == m:
                return cost
            for p in range(d):
                digit = (v // p10[p]) % 10
                for nd in (digit - 1, digit + 1):
                    if nd < 0 or nd > 9:
                        continue
                    if p == d - 1 and nd == 0:
                        continue  # never drop the leading digit to 0
                    nv = v + (nd - digit) * p10[p]
                    if is_prime(nv):
                        continue
                    ncost = cost + nv
                    if ncost < dist.get(nv, float('inf')):
                        dist[nv] = ncost
                        heapq.heappush(heap, (ncost, nv))
        return -1

```python
import heapq

class Solution:
    def minOperations(self, n: int, m: int) -> int:
        # Sieve of Eratosthenes for all possible values (< 10^4).
        N = 10000
        composite = bytearray(N)
        for i in range(2, 100):
            if not composite[i]:
                composite[i * i::i] = b'\x01' * len(range(i * i, N, i))

        def is_prime(x: int) -> bool:
            return x >= 2 and not composite[x]

        # The number is never allowed to be prime, including start and end.
        if is_prime(n) or is_prime(m):
            return -1

        # No operations needed: cost is just the value n already takes.
        if n == m:
            return n

        d = len(str(n))
        p10 = [10 ** p for p in range(d)]

        # Dijkstra where moving onto a value costs that value,
        # and the starting value n is part of the cost.
        dist = {n: n}
        heap = [(n, n)]
        while heap:
            cost, v = heapq.heappop(heap)
            if cost > dist[v]:
                continue
            if v == m:
                return cost
            for p in range(d):
                digit = (v // p10[p]) % 10
                for nd in (digit - 1, digit + 1):
                    if nd < 0 or nd > 9:
                        continue
                    if p == d - 1 and nd == 0:
                        continue  # keep the same number of digits
                    nv = v + (nd - digit) * p10[p]
                    if is_prime(nv):
                        continue
                    ncost = cost + nv
                    if ncost < dist.get(nv, float('inf')):
                        dist[nv] = ncost
                        heapq.heappush(heap, (ncost, nv))
        return -1
```

```python
import heapq

class Solution:
    def minOperations(self, n: int, m: int) -> int:
        # Sieve of Eratosthenes for all values that can appear (< 10^4).
        N = 10000
        composite = bytearray(N)
        for i in range(2, 100):
            if not composite[i]:
                for j in range(i * i, N, i):
                    composite[j] = 1

        def is_prime(x: int) -> bool:
            return x >= 2 and not composite[x]

        # n must be non-prime at the start, m non-prime at the end.
        if is_prime(n) or is_prime(m):
            return -1

        # Already equal: the only value n ever takes is n itself.
        if n == m:
            return n

        d = len(str(n))
        p10 = [10 ** p for p in range(d)]

        # Dijkstra: landing on a value costs that value; the starting
        # value n is counted in the cost as well.
        dist = {n: n}
        heap = [(n, n)]
        while heap:
            cost, v = heapq.heappop(heap)
            if cost > dist[v]:
                continue
            if v == m:
                return cost
            for p in range(d):
                digit = (v // p10[p]) % 10
                for nd in (digit - 1, digit + 1):
                    if nd < 0 or nd > 9:
                        continue
                    if p == d - 1 and nd == 0:
                        continue  # must keep the same number of digits
                    nv = v + (nd - digit) * p10[p]
                    if is_prime(nv):
                        continue
                    ncost = cost + nv
                    if ncost < dist.get(nv, float('inf')):
                        dist[nv] = ncost
                        heapq.heappush(heap, (ncost, nv))
        return -1
```

Looking at this problem:

- The cost includes the **initial value** of `n` plus the value after every operation (example: 10+20+21+22+12 = 85).
- `n` must be non-prime at the start, `m` non-prime at the end, and every intermediate value non-prime.
- States are numbers with the **same digit count** as `n` (the leading digit can never drop to 0, since the number must keep `d` digits — dropping below would be a dead end anyway, as digit count can never grow back).
- Each move to a value `v'` costs `v'`, so this is a shortest-path problem with positive weights → **Dijkstra**, over at most 9000 states.
