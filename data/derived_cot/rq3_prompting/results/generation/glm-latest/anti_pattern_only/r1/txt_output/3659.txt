```python
class Solution:
    def countPathsWithXorValue(self, grid: List[List[int]], k: int) -> int:
        MOD = 10**9 + 7
        m, n = len(grid), len(grid[0])
        # dp[j][x] = number of paths reaching cell (current_row, j) with XOR value x
        dp = [[0] * 16 for _ in range(n)]
        dp[0][grid[0][0]] = 1
        
        # First row: can only come from the left
        for j in range(1, n):
            v = grid[0][j]
            prev = dp[j - 1]
            cur = [0] * 16
            for x in range(16):
                if prev[x]:
                    cur[x ^ v] = (cur[x ^ v] + prev[x]) % MOD
            dp[j] = cur
        
        for i in range(1, m):
            # First column: can only come from above
            v = grid[i][0]
            new = [0] * 16
            for x in range(16):
                if dp[0][x]:
                    new[x ^ v] = (new[x ^ v] + dp[0][x]) % MOD
            dp[0] = new
            
            for j in range(1, n):
                v = grid[i][j]
                top = dp[j]      # still holds previous row's value
                left = dp[j - 1] # already updated to current row
                cur = [0] * 16
                for x in range(16):
                    c = top[x] + left[x]
                    if c:
                        cur[x ^ v] = (cur[x ^ v] + c) % MOD
                dp[j] = cur
        
        return dp[n - 1][k] % MOD
```
