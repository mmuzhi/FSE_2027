```python
class Solution:
    def maxSubarrays(self, n: int, conflictingPairs: List[List[int]]) -> int:
        from collections import defaultdict
        # group pairs by hi (max endpoint)
        by_hi = [[] for _ in range(n + 1)]  # list of lo values, with pair index
        m = len(conflictingPairs)
        gain = [0] * m
        for i, (a, b) in enumerate(conflictingPairs):
            if a > b:
                a, b = b, a
            by_hi[b].append((a, i))
        
        base = 0
        v1, c1, id1, v2 = 0, 0, -1, 0
        for r in range(1, n + 1):
            for (a, i) in by_hi[r]:
                if a > v1:
                    v2 = v1
                    v1 = a
                    c1 = 1
                    id1 = i
                elif a == v1:
                    c1 += 1
                elif a > v2:
                    v2 = a
                # else ignore
            if c1 == 1:
                gain[id1] += v1 - v2
            base += r - v1
        return base + max(gain)
```

```python
class Solution:
    def maxSubarrays(self, n: int, conflictingPairs: List[List[int]]) -> int:
        m = len(conflictingPairs)
        by_hi = [[] for _ in range(n + 2)]
        for i, p in enumerate(conflictingPairs):
            a, b = p[0], p[1]
            if a > b:
                a, b = b, a
            by_hi[b].append((a, i))

                elif a > v2:
                    v2 = a
            if c1 == 1:
```

```python
    def maxSubarrays(self, n: int, conflictingPairs: List[List[int]]) -> int:
```

Looking at this problem:

- Value `v` sits at position `v`, so a pair `(lo, hi)` (with `lo < hi`) makes subarray `[l, r]` bad iff `l <= lo` and `r >= hi`.
- For each right endpoint `r`, valid left endpoints are `l > M(r)` where `M(r)` = max `lo` over pairs with `hi <= r`. Base count = `sum(r - M(r))`.
- For removal gain: at each `r`, if the max `lo` is achieved by a unique pair `i`, removing `i` gains `v1 - v2` at that `r` (where `v2` is the best `lo` among the rest). If tied, removing any single pair gains 0 there. Accumulate per-pair gains and add the max gain to the base.

I verified this against both examples (yields 9 and 12) and several hand-checked edge cases (single pair → `n(n+1)/2`, duplicate pairs → zero gain, ties at max → zero gain).

Complexity: O(n + m) time and space, which fits n ≤ 10^5, m ≤ 2·10^5.

```python
from typing import List
```
