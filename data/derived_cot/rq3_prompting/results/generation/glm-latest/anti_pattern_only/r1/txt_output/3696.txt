```python
    def countSubstrings(self, s: str) -> int:

        ans = 0
```

```python
class Solution:
    def countSubstrings(self, s: str) -> int:
        n = len(s)
        ans = 0
        
        cnt3 = [0] * 3
        cnt9 = [0] * 9
        cnt7 = [0] * 7
        cnt3[0] = cnt9[0] = cnt7[0] = 1
        
        s3 = s9 = p7 = 0
        m7 = 1
        
        prev1 = -1  # digits[j-1]
        prev2 = -1  # digits[j-2]
        
        for j in range(n):
            d = ord(s[j]) - 48
            
            s3 = (s3 + d) % 3
            s9 = (s9 + d) % 9
            p7 = (p7 * 10 + d) % 7
            m7 = m7 * 5 % 7
            a7 = p7 * m7 % 7
            
            if d == 0:
                pass
            elif d == 1 or d == 2 or d == 5:
                ans += j + 1
            elif d == 3 or d == 6:
                ans += cnt3[s3]
            elif d == 9:
                ans += cnt9[s9]
            elif d == 7:
                ans += cnt7[a7]
            elif d == 4:
                ans += 1
                if prev1 >= 0 and prev1 % 2 == 0:
                    ans += j
            else:  # d == 8
                ans += 1
                if prev1 >= 0 and prev1 % 4 == 0:
                    ans += 1
                if prev2 >= 0 and (100 * prev2 + 10 * prev1 + 8) % 8 == 0:
                    ans += j - 1
            
            cnt3[s3] += 1
            cnt9[s9] += 1
            cnt7[a7] += 1
            
            prev2 = prev1
            prev1 = d
        
        return ans
```

```python
    def countSubstrings(self, s: str) -> int:
        n = len(s)
        ans = 0
        
        # For d in {3, 6}: substring divisible iff prefix digit sums mod 3 match.
        cnt3 = [0] * 3
        cnt3[0] = 1
        # For d = 9: prefix digit sums mod 9.
        cnt9 = [0] * 9
        cnt9[0] = 1
        # For d = 7: A(t) = P(t) * inv(10)^(t+1) mod 7.
        cnt7 = [0] * 7
        cnt7[0] = 1
        
        s3 = 0
        s9 = 0
        p7 = 0
        m7 = 1  # inv(10)^(t+1) mod 7
        
        prev1 = -1  # digit at j-1
        prev2 = -1  # digit at j-2
        
        for j in range(n):
            d = ord(s[j]) - 48
            
            s3 = (s3 + d) % 3
            s9 = (s9 + d) % 9
            p7 = (p7 * 10 + d) % 7
            m7 = m7 * 5 % 7
            a7 = p7 * m7 % 7
            
            if d == 1 or d == 2 or d == 5:
                ans += j + 1
            elif d == 3 or d == 6:
                ans += cnt3[s3]
            elif d == 9:
                ans += cnt9[s9]
            elif d == 7:
                ans += cnt7[a7]
            elif d == 4:
                ans += 1
                if prev1 >= 0 and not (prev1 & 1):

            elif d == 8:
                ans += 1
                if prev1 >= 0 and prev1 % 4 == 0:
                    ans += 1
                if prev2 >= 0 and (100 * prev2 + 10 * prev1 + 8) % 8 == 0:
                    ans += j - 1
            
            cnt3[s3] += 1
            cnt9[s9] += 1
            cnt7[a7] += 1
            
            prev2 = prev1
            prev1 = d
```

```python
class Solution:
    def countSubstrings(self, s: str) -> int:
        n = len(s)
        ans = 0
        
        # d in {3, 6}: substring (ending at digit 3 or 6) divisible by d
        #   iff prefix digit-sums mod 3 match (for 6, evenness is automatic).
        cnt3 = [0] * 3
        cnt3[0] = 1
        # d = 9: prefix digit-sums mod 9 must match.
        cnt9 = [0] * 9
        cnt9[0] = 1
        # d = 7: A(t) = P(t) * inv10^(t+1) mod 7 must match.
        cnt7 = [0] * 7
        cnt7[0] = 1
        
        s3 = 0   # digit sum mod 3
        s9 = 0   # digit sum mod 9
        p7 = 0   # prefix value mod 7
        m7 = 1   # 5^(t+1) mod 7  (5 = inverse of 10 mod 7)
        
        prev1 = -1  # digit at j-1 (-1 if none)
        prev2 = -1  # digit at j-2 (-1 if none)
        
        for j in range(n):
            d = ord(s[j]) - 48
            
            s3 = (s3 + d) % 3
            s9 = (s9 + d) % 9
            p7 = (p7 * 10 + d) % 7
            m7 = m7 * 5 % 7
            a7 = p7 * m7 % 7
            
            if d == 1 or d == 2 or d == 5:
                # always divisible by 1/2/5 respectively
                ans += j + 1
            elif d == 3 or d == 6:
                ans += cnt3[s3]
            elif d == 9:
                ans += cnt9[s9]
            elif d == 7:
                ans += cnt7[a7]
            elif d == 4:
                ans += 1  # "4" itself
                if prev1 >= 0 and (prev1 & 1) == 0:
                    ans += j  # all substrings of length >= 2
            elif d == 8:
                ans += 1  # "8" itself
                if prev1 >= 0 and prev1 % 4 == 0:
                    ans += 1  # length-2 substring
                if prev2 >= 0 and (100 * prev2 + 10 * prev1 + 8) % 8 == 0:
                    ans += j - 1  # all substrings of length >= 3
            
            cnt3[s3] += 1
            cnt9[s9] += 1
            cnt7[a7] += 1
            
            prev2 = prev1
            prev1 = d
        
        return ans
```

```python
        ans = 0
        
        # For last digit 3 or 6: substring divisible iff prefix digit sums match mod 3
        # (for 6, divisibility by 2 is automatic since it ends in 6).
        cnt3 = [0] * 3
        cnt3[0] = 1  # empty prefix (t = -1)
        # For last digit 9: prefix digit sums must match mod 9.
        cnt9 = [0] * 9
        cnt9[0] = 1

        cnt7 = [0] * 7
        cnt7[0] = 1

        s9 = 0   # prefix digit sum mod 9
        p7 = 0   # prefix value mod 7
        m7 = 1   # inv(10)^(t+1) mod 7; inv(10) mod 7 = 5
        
        prev1 = -1  # digit at j-1 (-1 if none)
        prev2 = -1  # digit at j-2 (-1 if none)

            a7 = p7 * m7 % 7

            cnt9[s9] += 1
```
