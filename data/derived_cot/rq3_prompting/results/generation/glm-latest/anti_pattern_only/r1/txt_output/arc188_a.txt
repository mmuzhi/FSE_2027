            new = np.zeros((4, M, M, M), dtype=np.int64)
            if ch == '?':
                tot = dp.sum(axis=0)
                new[0, 1:, :, :] = (tot - dp[0])[:-1, :, :]
                new[1, :, 1:, :] = (tot - dp[1])[:, :-1, :]
                new[2, :, :, 1:] = (tot - dp[2])[:, :, :-1]
                new[3] = tot - dp[3]
            else:
                c = 1 if ch == 'A' else (2 if ch == 'B' else 3)
                for v in range(4):
                    vp = v ^ c
                    if vp == 0:
                        new[0, 1:, :, :] += dp[v, :-1, :, :]
                    elif vp == 1:
                        new[1, :, 1:, :] += dp[v, :, :-1, :]
                    elif vp == 2:
                        new[2, :, :, 1:] += dp[v, :, :, :-1]
                    else:
                        new[3] += dp[v]
            dp = new % MOD
        rng = np.arange(M, dtype=np.int64)
        a = rng[:, None, None]
        b = rng[None, :, None]
        cc = rng[None, None, :]
        n3 = (N + 1) - a - b - cc

        pc = pc + np.where(n3 >= 0, n3 * (n3 - 1) // 2, np.int64(-(1 << 60)))
