```python
        INF = float('inf')

        dp[0][0] = 0
        for x in nums:
            ndp = [[INF] * (op2 + 1) for _ in range(op1 + 1)]
            for a in range(op1 + 1):
                for b in range(op2 + 1):

                    if cur == INF:
                        continue

                    if cur < ndp[a][b]:
                        ndp[a][b] = cur

                        if v < ndp[a+1][b]:

                        if v < ndp[a][b+1]:
```

```python
from typing import List

class Solution:
    def minArraySum(self, nums: List[int], k: int, op1: int, op2: int) -> int:
        INF = float('inf')
        dp = [[INF] * (op2 + 1) for _ in range(op1 + 1)]
        dp[0][0] = 0
        for x in nums:
            ndp = [[INF] * (op2 + 1) for _ in range(op1 + 1)]
            h = (x + 1) // 2
            both_vals = []
            if h >= k:
                both_vals.append(h - k)
            if x >= k:
                both_vals.append((x - k + 1) // 2)
            both_best = min(both_vals) if both_vals else INF
            for a in range(op1 + 1):
                row = dp[a]
                for b in range(op2 + 1):
                    cur = row[b]
                    if cur == INF:
                        continue
                    # skip
                    if cur + x < ndp[a][b]:
                        ndp[a][b] = cur + x
                    # op1 only
                    if a < op1 and cur + h < ndp[a + 1][b]:
                        ndp[a + 1][b] = cur + h
                    # op2 only
                    if b < op2 and x >= k and cur + x - k < ndp[a][b + 1]:
                        ndp[a][b + 1] = cur + x - k
                    # both
                    if a < op1 and b < op2 and both_best != INF and cur + both_best < ndp[a + 1][b + 1]:
                        ndp[a + 1][b + 1] = cur + both_best
            dp = ndp
        return min(v for row in dp for v in row)
```

Looking at this problem:

- Each element can receive: nothing, op1 only (`ceil(x/2)`), op2 only (`x-k` if `x>=k`), or both operations (in either order — if op1 first: `ceil(x/2)-k` valid when `ceil(x/2)>=k`; if op2 first: `ceil((x-k)/2)` valid when `x>=k`; take the min of valid options).
- With `n ≤ 100` and `op1, op2 ≤ n`, a DP over `(index, used_op1, used_op2)` works: ~100×101×101 ≈ 10⁶ states with O(1) transitions.

DP state `dp[a][b]` = minimum sum of processed prefix using `a` op1's and `b` op2's. Verified against both examples (Example 2 traces to 3, Example 1 to 23).

Key details:
- Skip transition must add `x` to the sum.
- For "both", consider both orderings since their validity conditions differ (e.g., `x=6, k=4`: only op2-then-op1 is valid, giving 1; `x=7, k=4`: op1-then-op2 gives 0 which is better).

```python
from typing import List

class Solution:
    def minArraySum(self, nums: List[int], k: int, op1: int, op2: int) -> int:
        INF = float('inf')
        dp = [[INF] * (op2 + 1) for _ in range(op1 + 1)]
        dp[0][0] = 0
        for x in nums:
            ndp = [[INF] * (op2 + 1) for _ in range(op1 + 1)]
            h = (x + 1) // 2
            both_vals = []
            if h >= k:
                both_vals.append(h - k)          # op1 then op2
            if x >= k:
                both_vals.append((x - k + 1) // 2)  # op2 then op1
            both_best = min(both_vals) if both_vals else INF
            for a in range(op1 + 1):
                row = dp[a]
                for b in range(op2 + 1):
                    cur = row[b]
                    if cur == INF:
                        continue
                    # skip this element
                    if cur + x < ndp[a][b]:
                        ndp[a][b] = cur + x
                    # op1 only
                    if a < op1 and cur + h < ndp[a + 1][b]:
                        ndp[a + 1][b] = cur + h
                    # op2 only
                    if b < op2 and x >= k and cur + x - k < ndp[a][b + 1]:
                        ndp[a][b + 1] = cur + x - k
                    # both operations on this element
                    if a < op1 and b < op2 and both_best != INF \
                            and cur + both_best < ndp[a + 1][b + 1]:
                        ndp[a + 1][b + 1] = cur + both_best
            dp = ndp
        return min(v for row in dp for v in row)
```
