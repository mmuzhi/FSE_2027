```python
class Solution:
    def minZeroArray(self, nums: List[int], queries: List[List[int]]) -> int:
        n = len(nums)
        q = len(queries)
        
        def can(k):
            # coverage sums using first k queries via difference array
            diff = [0] * (n + 1)
            for i in range(k):
                l, r, v = queries[i]
                diff[l] += v
                diff[r + 1] -= v
            cur = 0
            for i in range(n):
                cur += diff[i]
                if cur < nums[i]:
                    return False
            return True
        
        # binary search minimal k in [0, q]
        lo, hi = 0, q
        if not can(q):
            return -1
        # find minimal k with can(k)
        while lo < hi:
            mid = (lo + hi) // 2
            if can(mid):
                hi = mid
            else:
                lo = mid + 1
        return lo
```

```python
from typing import List

class Solution:
    def minZeroArray(self, nums: List[int], queries: List[List[int]]) -> int:
        n, q = len(nums), len(queries)

        def feasible(k: int) -> bool:
            # Max total decrement available at each index using first k queries
            diff = [0] * (n + 1)
            for l, r, v in queries[:k]:
                diff[l] += v
                diff[r + 1] -= v
            cover = 0
            for i in range(n):
                cover += diff[i]
                if cover < nums[i]:
                    return False
            return True

        # Impossible even with all queries
        if not feasible(q):
            return -1

        # Binary search for the minimal k (feasibility is monotonic in k)
        lo, hi = 0, q
        while lo < hi:
            mid = (lo + hi) // 2
            if feasible(mid):
                hi = mid
            else:
                lo = mid + 1
        return lo
```
