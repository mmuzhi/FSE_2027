```python
import sys

def main():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx+=1
    m = int(data[idx]); idx+=1
    k = int(data[idx]); idx+=1
    edges = []
    for _ in range(m):
        u = int(data[idx]); v = int(data[idx+1]); w = int(data[idx+2]); idx += 3
        edges.append((w, u, v))
    A = data[idx:idx+k]; idx += k
    B = data[idx:idx+k]; idx += k
    cntA = [0]*(n+1)
    cntB = [0]*(n+1)
    for x in A:
        cntA[int(x)] += 1
    for x in B:
        cntB[int(x)] += 1
    edges.sort()
    parent = list(range(n+1))
    size = [1]*(n+1)
    # cntA, cntB indexed by root
    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x
    ans = 0
    for w, u, v in edges:
        ru, rv = find(u), find(v)
        if ru == rv: continue
        # union by size
        if size[ru] < size[rv]: ru, rv = rv, ru
        parent[rv] = ru
        size[ru] += size[rv]
        a1, b1 = cntA[ru], cntB[ru]
        a2, b2 = cntA[rv], cntB[rv]
        a, b = a1+a2, b1+b2
        matched = min(a, b) - min(a1, b1) - min(a2, b2)
        ans += w * matched
        cntA[ru] = a
        cntB[ru] = b
    print(ans)

main()
```

```python
import sys

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0]); m = int(data[1]); k = int(data[2])
    
    base = 3
    ev = list(map(int, data[base:base + 3*m]))
    us = ev[0::3]; vs = ev[1::3]; ws = ev[2::3]
    base += 3*m
    
    cntA = [0]*(n+1)
    cntB = [0]*(n+1)
    for x in data[base:base+k]:
        cntA[int(x)] += 1
    base += k
    for x in data[base:base+k]:
        cntB[int(x)] += 1
    
    edges = sorted(zip(ws, us, vs))
    
    parent = list(range(n+1))
    sz = [1]*(n+1)
    
    ans = 0
    for w, u, v in edges:
        x = u
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        ru = x
        x = v
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        rv = x
        if ru == rv:
            continue
        if sz[ru] < sz[rv]:
            ru, rv = rv, ru
        parent[rv] = ru
        sz[ru] += sz[rv]
        a1 = cntA[ru]; b1 = cntB[ru]
        a2 = cntA[rv]; b2 = cntB[rv]
        a = a1 + a2; b = b1 + b2
        matched = (a if a < b else b) - (a1 if a1 < b1 else b1) - (a2 if a2 < b2 else b2)
        ans += w * matched
        cntA[ru] = a
        cntB[ru] = b
    
    print(ans)
```

Looking at this problem:

- f(x,y) is the bottleneck (minimax) distance, which equals the weight at which x and y first become connected in Kruskal's union process — i.e., the weight of the LCA in the Kruskal dendrogram (merge tree).
- Cost of a matching = Σ w(LCA(aᵢ, bσ(i))). Decomposing w(LCA) telescopically: Cost = K·w(root) − Σ_{u≠root} (w(parent(u))−w(u))·P(u), where P(u) = number of matched pairs fully inside subtree(u). Coefficients are ≥ 0 since merge weights are non-decreasing upward.
- For any matching, P(u) ≤ min(#A-entries, #B-entries) in subtree(u). The greedy that, at every union of two components, matches as many leftover A's with leftover B's as possible (each such cross pair costs exactly the current edge weight), achieves P(u) = min(A(u), B(u)) for **all** u simultaneously — hence it is optimal.
- Concretely, maintaining total counts (a, b) per DSU component, when merging components with counts (a1,b1) and (a2,b2) via edge weight w, the number of newly matched pairs is min(a1+a2, b1+b2) − min(a1,b1) − min(a2,b2), each costing w.

Verified on both samples (gives 8 and 3), plus several hand-crafted cases including duplicates within A/B.

Complexity: O(M log M) for sorting + near-linear DSU.
