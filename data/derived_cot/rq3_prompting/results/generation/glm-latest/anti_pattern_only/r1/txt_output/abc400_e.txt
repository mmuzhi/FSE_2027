A positive integer N is a 400 number if and only if it satisfies both of the following two conditions:

- N has exactly 2 distinct prime factors.
- For each prime factor p of N, p divides N an even number of times. More formally, the maximum non-negative integer k such that p^k divides N is even.

import sys
from math import isqrt
from bisect import bisect_right
from itertools import compress

def main():
    data = sys.stdin.buffer.read().split()
    Q = int(data[0])

    LIMIT = 10 ** 6
    half = LIMIT // 2  # largest possible larger prime q is LIMIT // 2

    # --- sieve primes up to `half` ---
    sieve = bytearray([1]) * (half + 1)
    sieve[0] = 0
    sieve[1] = 0
    for i in range(2, isqrt(half) + 1):
        if sieve[i]:
            sieve[i * i :: i] = bytearray(len(range(i * i, half + 1, i)))
    primes = list(compress(range(half + 1), sieve))

    # m = p^a * q^b with primes p < q and a, b >= 1
    valid = []
    ap = valid.append
    for pi, p in enumerate(primes):
        qmin = primes[pi + 1]
        if p * qmin > LIMIT:

        pa = p

            hi = bisect_right(primes, LIMIT // pa)
            for q in primes[pi + 1 : hi]:
                v = pa * q
                while v <= LIMIT:
                    ap(v)
                    v *= q
            pa *= p

    # --- answer queries ---

    isq = isqrt
    for t in data[1 : Q + 1]:
        s = isq(int(t))
        m = valid[bis(valid, s) - 1]
        ra(m * m)
    sys.stdout.write('\n'.join(map(str, res)))
    sys.stdout.write('\n')

import sys
from math import isqrt
from bisect import bisect_right
from itertools import compress

def main():
    data = sys.stdin.buffer.read().split()
    Q = int(data[0])

    LIMIT = 10 ** 6
    half = LIMIT // 2

    # Sieve of Eratosthenes up to `half` (the larger prime q satisfies q <= LIMIT/2)
    sieve = bytearray([1]) * (half + 1)
    sieve[0] = 0
    sieve[1] = 0
    for i in range(2, isqrt(half) + 1):
        if sieve[i]:
            sieve[i * i :: i] = bytearray(len(range(i * i, half + 1, i)))
    primes = list(compress(range(half + 1), sieve))

    # All m <= LIMIT of the form p^a * q^b with primes p < q, a,b >= 1,
    # i.e. exactly the integers with two distinct prime factors.
    valid = []
    ap = valid.append
    for pi, p in enumerate(primes):
        qmin = primes[pi + 1]
        if p * qmin > LIMIT:
            break
        pa = p
        while pa * qmin <= LIMIT:
            hi = bisect_right(primes, LIMIT // pa)
            for q in primes[pi + 1 : hi]:
                v = pa * q
                ap(v)
                v *= q
                while v <= LIMIT:
                    ap(v)
                    v *= q
            pa *= p

    valid.sort()

    res = []
    ra = res.append
    bis = bisect_right
    isq = isqrt
    for A in map(int, data[1 : Q + 1]):
        s = isq(A)
        m = valid[bis(valid, s) - 1]
        ra(m * m)
    sys.stdout.write('\n'.join(map(str, res)))
    sys.stdout.write('\n')

main()
