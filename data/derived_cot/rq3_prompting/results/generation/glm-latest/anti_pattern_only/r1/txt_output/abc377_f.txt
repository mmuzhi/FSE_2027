```python
import sys
import numpy as np

def main():
    data = sys.stdin.buffer.read().split()
    N = int(data[0]); M = int(data[1])
    vals = np.array(data[2:2 + 2 * M], dtype=np.int64)
    A = vals[0::2].copy()
    B = vals[1::2].copy()

    R = np.unique(A)
    C = np.unique(B)
    D1 = np.unique(A + B)
    D2 = np.unique(A - B)
    r = int(R.size)
    c = int(C.size)

    base = (N - r) * (N - c)

    # All pair sums r + c and pair diffs r - c, sorted
    sums = (R[:, None] + C[None, :]).ravel()
    sums.sort()
    diffs = (R[:, None] - C[None, :]).ravel()
    diffs.sort()

    # ---- f(s) for s in D1 ----
    S = D1
    L = np.maximum(1, S - N)
    U = np.minimum(N, S - 1)
    total = U - L + 1
    # RowBad: |R ∩ [s-N, s-1]|, ColBad: |C ∩ [s-N, s-1]|
    rowbad = np.searchsorted(R, S - 1, side='right') - np.searchsorted(R, S - N, side='left')
    colbad = np.searchsorted(C, S - 1, side='right') - np.searchsorted(C, S - N, side='left')
    # BothBad: #{(r,c) ∈ R×C : r + c = s}
    bothbad = np.searchsorted(sums, S, side='right') - np.searchsorted(sums, S, side='left')
    f_vals = total - rowbad - colbad + bothbad
    sum_f = int(f_vals.sum())

    # ---- g(d) for d in D2 ----
    D = D2
    L2 = np.maximum(1, D + 1)
    U2 = np.minimum(N, D + N)
    total2 = U2 - L2 + 1
    rowbad2 = np.searchsorted(R, D + N, side='right') - np.searchsorted(R, D + 1, side='left')

    g_vals = total2 - rowbad2 - colbad2 + bothbad2
```
