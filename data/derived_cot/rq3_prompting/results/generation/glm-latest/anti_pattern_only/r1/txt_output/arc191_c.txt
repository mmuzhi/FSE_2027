def solve(n, fac):
    if n == 1:
        return "1 1"
    e2 = fac.get(2, 0)
    odd = sorted(p for p in fac if p != 2)  # any order
    comps = []  # (a, m)
    if e2 == 1 and odd:
        # merge into first odd prime
        p = odd[0]; e = fac[p]
        m = p ** (e + 1)
        a = m - (1 + p)   # -(1+p) mod m
        comps.append((a, m))
        start = 1
    else:
        start = 0
        if e2 >= 1:
            m = 1 << (e2 + 2)
            comps.append((3, m))
    for p in odd[start:]:
        e = fac[p]
        m = p ** (e + 1)
        comps.append((1 + p, m))
