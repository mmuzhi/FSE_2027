The array will become [3, 1, 0, 0], which is not a Zero Array.

```python
class Solution:
    def isZeroArray(self, nums: List[int], queries: List[List[int]]) -> bool:
```
```python
        n = len(nums)
        diff = [0] * (n + 1)
        for l, r in queries:
            diff[l] += 1
            diff[r + 1] -= 1
        cur = 0
        for i in range(n):
            cur += diff[i]
            if nums[i] > cur:
                return False
        return True
```
