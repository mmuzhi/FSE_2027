```python
class Solution:
    def largestPalindrome(self, n: int, k: int) -> str:
        from math import gcd
        m = (n + 1) // 2
        # pow10 mod k up to n-1
        # weights w[i] for i in 0..m-1
        # w[i] = (10^i + 10^(n-1-i)) % k, or 10^i % k if i == n-1-i
        pow10 = [0] * n
        p = 1 % k
        pow10[0] = p
        for e in range(1, n):
            p = p * 10 % k
            pow10[e] = p
        w = [0] * m
        for i in range(m):
            j = n - 1 - i
            if j == i:
                w[i] = pow10[i]
            else:
                w[i] = (pow10[i] + pow10[j]) % k
        # suffix gcd g[i] = gcd(k, w[i..m-1]); g[m] = k
        g = [0] * (m + 1)
        g[m] = k
        for i in range(m - 1, -1, -1):
            g[i] = gcd(g[i + 1], w[i])
        # greedy
        r = 0
        half = []
        for i in range(m):
            wi = w[i]
            gn = g[i + 1]
            lo = 1 if i == 0 else 0
            for d in range(9, lo - 1, -1):
                nr = (r + d * wi) % k
                if nr % gn == 0:
                    r = nr
                    half.append(d)
                    break
        s = ''.join(map(str, half))
        if n % 2 == 1:
            result = s + s[:-1][::-1]
        else:
            result = s + s[::-1]
        return result
```

```python
class Solution:
    def largestPalindrome(self, n: int, k: int) -> str:
        from math import gcd
        m = (n + 1) // 2
        
        # powers of 10 mod k
        pow10 = [0] * n
        p = 1 % k
        pow10[0] = p
        for e in range(1, n):
            p = (p * 10) % k
            pow10[e] = p
        
        # weight of each half-position
        w = [0] * m
        for i in range(m):
            j = n - 1 - i
            if j == i:
                w[i] = pow10[i]
            else:
                w[i] = (pow10[i] + pow10[j]) % k
        
        # suffix gcd
        g = [k] * (m + 1)
        for i in range(m - 1, -1, -1):
            g[i] = gcd(g[i + 1], w[i])
        
        r = 0
        half = []
        for i in range(m):
            wi = w[i]
            gn = g[i + 1]
            base = r
            lo = 1 if i == 0 else 0
            chosen = -1
            for d in range(9, lo - 1, -1):
                if (base + d * wi) % gn == 0:
                    chosen = d
                    break
            half.append(chosen)
            r = (r + chosen * wi) % k
        
        s = ''.join(map(str, half))
        if n & 1:
            return s + s[-2::-1]
        return s + s[::-1]
```

```python
class Solution:
    def largestPalindrome(self, n: int, k: int) -> str:
        from math import gcd
        m = (n + 1) // 2
        
        # powers of 10 modulo k
        pow10 = [0] * n
        p = 1 % k
        pow10[0] = p
        for e in range(1, n):
            p = p * 10 % k
            pow10[e] = p
        
        # weight of each half-position i (0-indexed from most significant)
        w = [0] * m
        for i in range(m):
            j = n - 1 - i
            w[i] = pow10[i] if j == i else (pow10[i] + pow10[j]) % k
        
        # g[i] = gcd(k, w[i], ..., w[m-1]); g[m] = k
        g = [k] * (m + 1)
        for i in range(m - 1, -1, -1):
            g[i] = gcd(g[i + 1], w[i])
        
        r = 0
        half = []
        push = half.append
        for i in range(m):
            wi = w[i]
            gn = g[i + 1]
            for d in range(9, -1, -1):
                if (r + d * wi) % gn == 0:
                    break
            push(d)
            r = (r + d * wi) % k
        
        s = ''.join(map(str, half))
        if n & 1:
            return s + s[:-1][::-1]
        return s + s[::-1]
```

```python
class Solution:
    def largestPalindrome(self, n: int, k: int) -> str:
        from math import gcd
        m = (n + 1) // 2

        # pow10[e] = 10^e mod k
        pow10 = [0] * n
        p = 1 % k
        pow10[0] = p
        for e in range(1, n):
            p = p * 10 % k
            pow10[e] = p

        # w[i]: weight of half-position i (its total contribution coefficient mod k)
        w = [0] * m
        for i in range(m):
            j = n - 1 - i
            w[i] = pow10[i] if j == i else (pow10[i] + pow10[j]) % k

        # g[i] = gcd(k, w[i..m-1]); g[m] = k
        g = [k] * (m + 1)
        for i in range(m - 1, -1, -1):
            g[i] = gcd(g[i + 1], w[i])

        r = 0
        half = []
        for i in range(m):
            wi = w[i]
            gn = g[i + 1]
            d = 9
            while (r + d * wi) % gn:
                d -= 1
            half.append(d)
            r = (r + d * wi) % k

        s = ''.join(map(str, half))
        return s + s[:-1][::-1] if n & 1 else s + s[::-1]
```

```python
from math import gcd

class Solution:
    def largestPalindrome(self, n: int, k: int) -> str:
        m = (n + 1) // 2

        # pow10[e] = 10^e mod k
        pow10 = [0] * n
        p = 1 % k
        pow10[0] = p
        for e in range(1, n):
            p = p * 10 % k
            pow10[e] = p

        # w[i] = total coefficient (mod k) of half-digit i in the number
        w = [0] * m
        for i in range(m):
            j = n - 1 - i
            w[i] = pow10[i] if j == i else (pow10[i] + pow10[j]) % k

        # g[i] = gcd(k, w[i..m-1]); g[m] = k
        g = [k] * (m + 1)
        for i in range(m - 1, -1, -1):
            g[i] = gcd(g[i + 1], w[i])

        r = 0
        half = []
        for i in range(m):
            wi = w[i]
            gn = g[i + 1]
            for d in range(9, -1, -1):
                if (r + d * wi) % gn == 0:
                    break
            half.append(d)
            r = (r + d * wi) % k

        s = ''.join(map(str, half))
        if n & 1:
            return s + s[:-1][::-1]
        return s + s[::-1]
```

Looking at this problem:

- The palindrome is fully determined by its first half (first `ceil(n/2)` digits). A half-digit at position `i` (0-indexed from the most significant side) contributes `d_i * (10^i + 10^(n-1-i))` mod `k` (or just `10^i` if it's the middle digit).
- Greedily choose digits from 9 down, left to right. A prefix remainder `r` can be completed to `0 mod k` iff the remaining positions can produce `-r mod k`. Since `k ≤ 9`, digits 0..9 cover every residue mod `k`, so each remaining position can contribute any multiple of `gcd(w_j, k)`, meaning the achievable set is exactly the multiples of `g = gcd(k, w_{i+1},...,w_{m-1})`. So the check is simply `r' % g == 0`.
- This greedy always succeeds (one can show a valid digit exists at every step via the invariant `r % g[i] == 0`), and lexicographically maximal half gives the maximal palindrome.

I verified this against all three examples (595, 8, 89898) and several hand-checked cases (k=7 n=4 → 9779, k=8 n=6 → 888888, k=6 n=6 → 897798, etc.). The feasibility check reduces to `gcd` chains, and positions where multiple digit trials are needed are rare (the suffix gcd only decreases a few times total), so runtime is ~O(n).
