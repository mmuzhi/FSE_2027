```python
class Solution:
    def maxCollectedFruits(self, fruits: List[List[int]]) -> int:
        n = len(fruits)
        NEG = float('-inf')
        
        def best(g):
            # DP for a child that moves down one row per step with column changes in {-1,0,1},
            # starting at column n-1 (row 0) and ending at column n-1 (row n-1).
            # Cells on the main diagonal (j == k) contribute 0 (already collected by child 1).
            prev = [NEG] * n
            prev[n - 1] = g[0][n - 1]
            for k in range(1, n):
                padded = [NEG] + prev + [NEG]
                m = list(map(max, padded, padded[1:], padded[2:]))
                row = g[k]
                prev = m[:k + 1] + [x + y for x, y in zip(m[k + 1:], row[k + 1:])]
            return prev[n - 1]
        
        diag = sum(fruits[i][i] for i in range(n))
        trans = list(zip(*fruits))
        return diag + best(fruits) + best(trans)
```

```python
from typing import List

class Solution:
    def maxCollectedFruits(self, fruits: List[List[int]]) -> int:
        n = len(fruits)
        if n == 1:
            return fruits[0][0]
        NEG = float('-inf')

        def best(g):
            # DP for the child that advances one row per step (column changes by -1/0/+1),
            # starting at (0, n-1) and ending at (n-1, n-1).
            # Main-diagonal cells add 0 (the first child collects them).
            prev = [NEG] * n
            prev[n - 1] = g[0][n - 1]
            for k in range(1, n):
                padded = [NEG] + prev + [NEG]
                m = list(map(max, padded, padded[1:], padded[2:]))
                row = g[k]
                prev = m[:k + 1] + [x + y for x, y in zip(m[k + 1:], row[k + 1:])]
            return prev[n - 1]

        diag = sum(fruits[i][i] for i in range(n))
        trans = list(zip(*fruits))
        return diag + best(fruits) + best(trans)
```

```python
    def maxCollectedFruits(self, fruits: List[List[int]]) -> int:
        n = len(fruits)
        NEG = -(1 << 60)

        def best(g):
            # DP over rows k = 0..n-1; at row k the child is at (k, j) with j in the
            # valid band [max(k, n-1-k), n-1]; j changes by -1/0/+1 per row.
            # Diagonal cells (j == k) contribute 0 (child 1 collects them).
            prev = [NEG] * n
            prev[n - 1] = g[0][n - 1]
            for k in range(1, n):

                prev = [NEG] * nlo + vals
            return prev[n - 1]

        diag = sum(fruits[i][i] for i in range(n))
        trans = list(zip(*fruits))
```

```python
from typing import List

class Solution:
    def maxCollectedFruits(self, fruits: List[List[int]]) -> int:
        n = len(fruits)
        if n == 1:
            return fruits[0][0]
        NEG = -(1 << 60)

        def best(g):
            # Child that advances one row per step (column changes by -1/0/+1),
            # from (0, n-1) to (n-1, n-1). Valid columns at row k form the band
            # [max(k, n-1-k), n-1]. Diagonal cells add 0 (child 1 takes them).
            prev = [NEG] * n
            prev[n - 1] = g[0][n - 1]
            for k in range(1, n):
                nlo = k if k >= n - 1 - k else n - 1 - k
                padded = prev[nlo - 1:] + [NEG]
                w = list(map(max, padded, padded[1:], padded[2:]))
                row = g[k]
                if nlo == k:
                    vals = w[:1] + [x + y for x, y in zip(w[1:], row[k + 1:])]
                else:
                    vals = [x + y for x, y in zip(w, row[nlo:])]
                prev = [NEG] * nlo + vals
            return prev[n - 1]

        diag = sum(fruits[i][i] for i in range(n))
        trans = list(zip(*fruits))
        return diag + best(fruits) + best(trans)
```
