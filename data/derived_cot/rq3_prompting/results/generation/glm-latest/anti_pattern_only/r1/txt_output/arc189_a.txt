import sys

def main():
    input_data = sys.stdin.buffer.read().split()
    N = int(input_data[0])
    A = list(map(int, input_data[1:1+N]))
    MOD = 998244353
    
    if A[0] != 1:
        print(0)
        return
    
    # compute runs
    runs = []
    cur = A[0]
    cnt = 0
    for a in A:
        if a == cur:
            cnt += 1
        else:
            runs.append(cnt)
            cur = a
            cnt = 1
    runs.append(cnt)
    
    if any(r % 2 == 0 for r in runs):
        print(0)
        return
    
    t = [(r - 1) // 2 for r in runs]
    T = sum(t)
    
    # precompute factorials up to N
    maxn = N + 1
    fact = [1] * maxn
    for i in range(1, maxn):
        fact[i] = fact[i-1] * i % MOD
    inv_fact = [1] * maxn
    inv_fact[maxn-1] = pow(fact[maxn-1], MOD-2, MOD)
    for i in range(maxn-1, 0, -1):
        inv_fact[i-1] = inv_fact[i] * i % MOD
    
    inv2 = pow(2, MOD-2, MOD)
    ans = fact[T]
    for tk in t:
        ans = ans * fact[2*tk] % MOD
        ans = ans * inv_fact[tk] % MOD * inv_fact[tk] % MOD
    ans = ans * pow(inv2, T, MOD) % MOD
    print(ans)

main()
