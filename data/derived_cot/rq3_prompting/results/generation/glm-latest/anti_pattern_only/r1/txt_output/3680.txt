class Solution:
    def countComponents(self, nums, threshold):
        T = threshold
        present = [False] * (T + 1)
        big = 0
        for v in nums:
            if v > T:
                big += 1
            else:
                present[v] = True
        # DSU over 0..T (index 0 unused)
        parent = list(range(T + 1))
        # iterative find with path halving

    def countComponents(self, nums: List[int], threshold: int) -> int:
        T = threshold
        present = bytearray(T + 1)
        big = 0

            if v > T:
                big += 1
            else:
                present[v] = 1

                roots.add(a)

```python
class Solution:
    def countComponents(self, nums: List[int], threshold: int) -> int:
        T = threshold
        present = bytearray(T + 1)
        big = 0
        for v in nums:
            if v > T:
                big += 1
            else:
                present[v] = 1
        
        if present[1]:
            # 1 is connected to every value <= threshold (lcm(1, y) = y)
            return big + 1
        
        parent = list(range(T + 1))
        sz = [1] * (T + 1)
        p = parent
        half = T >> 1
        
        for x in range(2, T + 1):
            if not present[x]:
                continue
            # find root of x with path halving
            a = x
            while p[a] != a:
                p[a] = p[p[a]]
                a = p[a]
            rx = a
            rs = sz[rx]
            start = (half // x + 1) * x
            for m in range(start, T + 1, x):
                b = m
                while p[b] != b:
                    p[b] = p[p[b]]
                    b = p[b]
                if b != rx:
                    bs = sz[b]
                    if rs >= bs:
                        p[b] = rx
                        sz[rx] = rs = rs + bs
                    else:
                        p[rx] = b
                        sz[b] = bs + rs
                        rx = b
                        rs = bs
        
        roots = set()
        for x in range(1, T + 1):
            if present[x]:
                a = x
                while p[a] != a:
                    p[a] = p[p[a]]
                    a = p[a]
                roots.add(a)
        return big + len(roots)
```

    def countComponents(self, nums: List[int], threshold: int) -> int:
        T = threshold
        present = bytearray(T + 1)
        big = 0
        for v in nums:
            if v > T:
                big += 1
            else:
                present[v] = 1
        
        if present[1]:
            # lcm(1, y) = y <= threshold for every present y -> all present form one component
            return big + 1
        
        p = list(range(T + 1))

                # find(m)

                        rs += bs
                        sz[rx] = rs

                        p[rx] = b
                        rs += bs
