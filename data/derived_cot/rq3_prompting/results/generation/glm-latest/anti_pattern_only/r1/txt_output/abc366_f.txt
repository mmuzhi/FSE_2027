import sys
from collections import Counter

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0]); k = int(data[1])
    cnt = Counter()
    idx = 2
    for i in range(n):
        a = int(data[idx]); b = int(data[idx+1]); idx += 2
        cnt[(a,b)] += 1
    # sort distinct functions by key (a-1)/b ascending (innermost first)
    distinct = sorted(cnt.keys(), key=lambda p: (p[0]-1)/p[1])
    NEG = -1  # sentinel for -inf; values are always >= 1... use None or -1
    dp = [-1]*(k+1)  # -1 = impossible
    dp[0] = 1
    for (a, b), m in sorted(cnt.items(), key=lambda kv: (kv[0][0]-1)/kv[0][1]):
        m = min(m, k)
        # precompute powers and sums
        # for t in 1..m: a^t, and b*(a^t-1)/(a-1) or t*b
        # transition descending c
        # new dp[c+t] = max(dp[c+t], a^t * dp[c] + add_t)
        powers = [1]*(m+1)
        for t in range(1, m+1):
            powers[t] = powers[t-1]*a
        if a == 1:
            adds = [b*t for t in range(m+1)]
        else:
            adds = [b*(powers[t]-1)//(a-1) for t in range(m+1)]
        # gather updates
        # iterate c descending
        for c in range(k-1, -1, -1):
            x = dp[c]
            if x < 0:
                continue
            maxt = m if m < k-c else k-c
            for t in range(1, maxt+1):
                v = powers[t]*x + adds[t]
                if v > dp[c+t]:
                    dp[c+t] = v
    print(dp[k])

main()

from collections import Counter

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    k = int(data[1])
    vals = list(map(int, data[2:2 + 2 * n]))
    cnt = Counter(zip(vals[0::2], vals[1::2]))

Looking at this problem:

- Composing functions applied innermost-first: value built as x ← a·x + b starting from 1.
- **Exchange argument**: for two adjacent functions i (outer), j (inner), keeping i outer is better iff b_j(a_i−1) ≥ b_i(a_j−1), i.e. iff key(i) = (a_i−1)/b_i ≥ key(j). So for any fixed set, sorting by key descending from outermost is optimal (ties are value-equal).
- Therefore the optimum is a subsequence of all functions sorted by key **ascending**, applied in that order (innermost first). Since all functions are increasing, a DP keeping the maximum current value per (position, count) is exact: `dp[c] = max value using c functions so far`.
- Since A,B ≤ 50, there are ≤ 2500 distinct functions; handle duplicates by allowing up to min(count, K) copies per distinct function in one transition (x ← aᵗx + b·(aᵗ−1)/(a−1)).

Verified on sample 1 (→26) and sample 2 (the optimal 48(48(49+44)+40)+31 = 216223 follows key order 47/31 > 47/40 > 48/44 and is found by the DP).
