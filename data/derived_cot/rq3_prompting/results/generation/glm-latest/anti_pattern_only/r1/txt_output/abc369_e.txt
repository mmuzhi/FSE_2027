There are N islands and M bidirectional bridges connecting two islands. The islands and bridges are numbered 1, 2, \ldots, N and 1, 2, \ldots, M, respectively.
Bridge i connects islands U_i and V_i, and the time it takes to cross it in either direction is T_i.
No bridge connects an island to itself, but it is possible for two islands to be directly connected by more than one bridge.
One can travel between any two islands using some bridges.
You are given Q queries, so answer each of them. The i-th query is as follows:

You are given K_i distinct bridges: bridges B_{i,1}, B_{i,2}, \ldots, B_{i,K_i}.
Find the minimum time required to travel from island 1 to island N using each of these bridges at least once.
Only consider the time spent crossing bridges.
You can cross the given bridges in any order and in any direction.

import numpy as np
from collections import defaultdict

def main():
    data = sys.stdin.buffer.read().split()
    vals = list(map(int, data))
    del data
    pos = 0
    N = vals[pos]; pos += 1
    M = vals[pos]; pos += 1
    
    ev = vals[pos:pos + 3 * M]
    pos += 3 * M
    U = [x - 1 for x in ev[0::3]]
    V = [x - 1 for x in ev[1::3]]
    T = ev[2::3]
    
    INF = 1 << 60
    
    # Build initial distance matrix (min over parallel edges)
    D = np.full((N, N), INF, dtype=np.int64)
    np.fill_diagonal(D, 0)
    
    Uarr = np.array(U, dtype=np.int64)
    Varr = np.array(V, dtype=np.int64)
    Tarr = np.array(T, dtype=np.int64)
    
    key = np.concatenate([Uarr * N + Varr, Varr * N + Uarr])
    keyT = np.concatenate([Tarr, Tarr])
    order = np.argsort(key)
    ks = key[order]
    ts = keyT[order]
    m = ks.shape[0]
    change = np.empty(m, dtype=bool)
    change[0] = True
    if m > 1:
        np.not_equal(ks[1:], ks[:-1], out=change[1:])
    starts = np.flatnonzero(change)
    mins = np.minimum.reduceat(ts, starts)
    D.reshape(-1)[ks[starts]] = mins
    
    # Floyd-Warshall
    tmp = np.empty((N, N), dtype=np.int64)
    for k in range(N):
        np.add(D[:, k, None], D[k, None, :], out=tmp)
        np.minimum(D, tmp, out=D)
    
    # Queries
    Q = vals[pos]; pos += 1
    qK = [0] * Q
    qB = [None] * Q
    for qi in range(Q):
        K = vals[pos]; pos += 1
        qK[qi] = K
        qB[qi] = vals[pos:pos + K]
        pos += K
    
    groups = defaultdict(list)
    for qi in range(Q):
        groups[qK[qi]].append(qi)
    
    answers = [0] * Q
    s = 0
    tnode = N - 1
    Ds = D[s]
    Dt = D[tnode]
    
    for K, qids in groups.items():
        Qg = len(qids)
        Elist = []
        Tlist = []
        for qi in qids:
            rowE = []
            for b in qB[qi]:
                b -= 1
                rowE.append(U[b])
                rowE.append(V[b])
                Tlist.append(T[b])
            Elist.append(rowE)
        E = np.array(Elist, dtype=np.int64)
        Tm = np.array(Tlist, dtype=np.int64).reshape(Qg, K)
        
        uidx = E[:, 0::2]
        vidx = E[:, 1::2]
        Du = D[uidx[:, :, None], E[:, None, :]]
        Dv = D[vidx[:, :, None], E[:, None, :]]
        Dstart = Ds[E]
        Dend = Dt[E]

        twoK = 2 * K

        dp[:, 0, :] = Dstart

            dpm = dp[:, mask, :]
            for i in range(K):
                bit = 1 << i
                if mask & bit:
                    continue
                nm = mask | bit
                base = dpm + Tm[:, i, None]
                c1 = (base + Du[:, i, :]).min(axis=1)
                c2 = (base + Dv[:, i, :]).min(axis=1)
