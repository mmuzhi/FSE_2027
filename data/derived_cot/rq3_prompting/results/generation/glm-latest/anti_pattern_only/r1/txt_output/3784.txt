class Solution:
    def longestCommonPrefix(self, words: List[str], k: int) -> List[int]:
        n = len(words)
        if n - 1 < k:
            return [0] * n
        
        # Build trie
        child = [{}]           # child[u]: char -> node id
        cnt = [0]              # cnt[u]: number of words with this prefix
        dep = [0]              # depth of node
        end_node = [0] * n
        
        for i, w in enumerate(words):
            cur = 0
            cnt[0] += 1
            for c in w:
                d = child[cur]
                nxt = d.get(c, -1)
                if nxt < 0:
                    nxt = len(child)
                    d[c] = nxt
                    child.append({})
                    cnt.append(0)
                    dep.append(dep[cur] + 1)
                cur = nxt
                cnt[cur] += 1
            end_node[i] = cur
        
        T = len(child)
        
        # G: max depth with cnt >= k+1
        G = 0
        for v in range(T):
            if cnt[v] > k and dep[v] > G:
                G = dep[v]
        
        # f[v]: max depth of node with cnt == k in subtree(v) (including v)
        f = [-1] * T
        for v in range(T - 1, -1, -1):
            best = dep[v] if cnt[v] == k else -1
            for c in child[v].values():
                if f[c] > best:
                    best = f[c]
            f[v] = best
        
        # g[v]: max depth of cnt==k node off the root->v path (side nodes)
        # bestChildF[v]: max f among children of v
        g = [-1] * T
        bestChildF = [-1] * T
        for u in range(T):
            kids = child[u]
            if not kids:
                continue
            # top-2 f among kids
            m1 = -1
            m2 = -1
            b = -1
            dup = False
            for c in kids.values():
                fv = f[c]
                if fv > m1:
                    m2 = m1
                    m1 = fv
                    b = c
                    dup = False
                elif fv == m1:
                    dup = True
                elif fv > m2:
                    m2 = fv
            bestChildF[u] = m1
            gu = g[u]
            for c in kids.values():
                if dup or c != b:
                    side = m1
                else:
                    side = m2
                gc = gu if gu > side else side
                g[c] = gc

            v = g[L]

            b = v if v > w2 else w2

        return ans

```python
class Solution:
    def longestCommonPrefix(self, words: List[str], k: int) -> List[int]:
        n = len(words)
        if n <= k:
            return [0] * n
        
        child = [{}]
        cnt = [0]
        dep = [0]
        end_node = [0] * n
        
        for i in range(n):
            w = words[i]
            cur = 0
            cnt[0] += 1
            for c in w:
                d = child[cur]
                nxt = d.get(c)
                if nxt is None:
                    nxt = len(child)
                    d[c] = nxt
                    child.append({})
                    cnt.append(0)
                    dep.append(dep[cur] + 1)
                cur = nxt
                cnt[cur] += 1
            end_node[i] = cur
        
        T = len(child)
        
        # G: deepest node with cnt >= k+1 (safe under any single removal)
        G = 0
        for v in range(T):
            if cnt[v] > k and dep[v] > G:
                G = dep[v]
        
        # f[v]: deepest node with cnt == k within subtree(v), incl. v; -1 if none
        f = [-1] * T
        for v in range(T - 1, -1, -1):
            b = dep[v] if cnt[v] == k else -1
            for c in child[v].values():
                fc = f[c]
                if fc > b:
                    b = fc
            f[v] = b
        
        # g[v]: deepest cnt==k node strictly aside of the root->v path; -1 if none
        g = [-1] * T
        bestChildF = [-1] * T
        for u in range(T):
            kids = child[u]
            if not kids:
                continue
            m1 = -1
            m2 = -1
            bnode = -1
            dup = False
            for c in kids.values():
                fc = f[c]
                if fc > m1:
                    m2 = m1
                    m1 = fc
                    bnode = c
                    dup = False
                elif fc == m1:
                    dup = True
                elif fc > m2:
                    m2 = fc
            bestChildF[u] = m1
            gu = g[u]
            for c in kids.values():
                side = m1 if (dup or c != bnode) else m2
                if gu > side:
                    side = gu
                g[c] = side
        
        ans = [0] * n
        for i in range(n):
            L = end_node[i]
            b = g[L]
            bc = bestChildF[L]
            if bc > b:
                b = bc
            ans[i] = b if b > G else G
        return ans
```

```python
class Solution:
    def longestCommonPrefix(self, words: List[str], k: int) -> List[int]:
        n = len(words)
        if n <= k:
            return [0] * n
        
        # Build trie.
        child = [{}]   # child[u]: char -> node id
        cnt = [0]      # cnt[u]: number of words having this node as prefix
        dep = [0]      # dep[u]: depth (prefix length)
        end_node = [0] * n
        
        for i in range(n):
            cur = 0
            cnt[0] += 1
            for c in words[i]:
                d = child[cur]
                nxt = d.get(c)
                if nxt is None:
                    nxt = len(child)
                    d[c] = nxt
                    child.append({})
                    cnt.append(0)
                    dep.append(dep[cur] + 1)
                cur = nxt
                cnt[cur] += 1
            end_node[i] = cur
        
        T = len(child)
        
        # G: deepest depth reachable by >= k+1 words (immune to any single removal).
        # f[v]: deepest node with cnt == k inside subtree(v) (v included), else -1.
        G = 0
        f = [-1] * T
        for v in range(T - 1, -1, -1):
            cv = cnt[v]
            if cv > k and dep[v] > G:
                G = dep[v]
            b = dep[v] if cv == k else -1
            for c in child[v].values():
                fc = f[c]
                if fc > b:
                    b = fc
            f[v] = b
        
        # g[v]: deepest cnt==k node lying strictly aside of the path root->v, else -1.
        # bestChildF[u]: max f among children of u (deepest cnt==k node strictly below u).
        g = [-1] * T
        bestChildF = [-1] * T
        for u in range(T):
            kids = child[u]
            if not kids:
                continue
            klist = list(kids.values())
            m1 = -1
            m2 = -1
            bnode = -1
            dup = False
            for c in klist:
                fc = f[c]
                if fc > m1:
                    m2 = m1
                    m1 = fc
                    bnode = c
                    dup = False
                elif fc == m1:
                    dup = True
                elif fc > m2:
                    m2 = fc
            bestChildF[u] = m1
            gu = g[u]
            for c in klist:
                side = m1 if (dup or c != bnode) else m2
                if gu > side:
                    side = gu
                g[c] = side
        
        ans = [0] * n
        for i in range(n):
            L = end_node[i]
            b = g[L]
            bc = bestChildF[L]
            if bc > b:
                b = bc
            ans[i] = b if b > G else G
        return ans
```

class Solution:
    def longestCommonPrefix(self, words: List[str], k: int) -> List[int]:
        n = len(words)
        if n <= k:
            return [0] * n

        # ---- Build trie ----
        child = [{}]      # child[u]: char -> node id
        cnt = [0]         # cnt[u]: number of words having node u's string as prefix
        dep = [0]         # dep[u]: depth of node (prefix length)
        end_node = [0] * n

        for i in range(n):
            cur = 0
            cnt[0] += 1
            for c in words[i]:
                d = child[cur]
                nxt = d.get(c)
                if nxt is None:
                    nxt = len(child)
                    d[c] = nxt
                    child.append({})
                    cnt.append(0)
                    dep.append(dep[cur] + 1)
                cur = nxt
                cnt[cur] += 1
            end_node[i] = cur

        T = len(child)

        # G: deepest depth with count >= k+1 (valid for every removal).
        # f[v]: deepest depth of a node with count == k in subtree(v), else -1.
        G = 0
        f = [-1] * T
        for v in range(T - 1, -1, -1):
            cv = cnt[v]
            dv = dep[v]
            if cv > k and dv > G:
                G = dv
            b = dv if cv == k else -1
            for c in child[v].values():
                fc = f[c]
                if fc > b:
                    b = fc
            f[v] = b

        # g[v]: deepest count==k node strictly aside of the root->v path, else -1.
        # bestChildF[u]: deepest count==k node strictly below u, else -1.
        g = [-1] * T
        bestChildF = [-1] * T
        for u in range(T):
            klist = list(child[u].values())
            if not klist:
                continue
            m1 = -1
            m2 = -1
            bnode = -1
            dup = False
            for c in klist:
                fc = f[c]
                if fc > m1:
                    m2 = m1
                    m1 = fc
                    bnode = c
                    dup = False
                elif fc == m1:
                    dup = True
                elif fc > m2:
                    m2 = fc
            bestChildF[u] = m1
            gu = g[u]
            for c in klist:
                side = m2 if (not dup and c == bnode) else m1
                if gu > side:
                    side = gu
                g[c] = side

        ans = [0] * n
        for i in range(n):
            L = end_node[i]
            b = g[L]
            bc = bestChildF[L]
            if bc > b:
                b = bc
            ans[i] = b if b > G else G
        return ans

            b = g[L]
            bc = bestChildF[L]
            if bc > b:
                b = bc
            ans[i] = b if b > G else G
        return ans

Looking at this problem:

- The answer for removal `i` is the max depth `d` such that some prefix of length `d` is shared by ≥ k of the remaining words. Build a trie with counts.
- Nodes with `cnt >= k+1` are safe under **any** single removal → global value `G` (max such depth).
- Nodes with `cnt == k` ("k-nodes") are only usable by words whose trie path does **not** contain that node (on-path nodes lose one count). So `answer[i] = max(G, B_i)` where `B_i` = deepest k-node not an ancestor-or-self of word `i`'s end node.
- `B_i` splits into: deepest k-node in a sibling subtree hanging off the root→end-node path (computed top-down via `g`, using top-2 of subtree-max `f` among children), plus deepest k-node strictly below the end node (`bestChildF`).

All verified against both examples and many hand-crafted cases (duplicates, prefixes of each other, k=1, k=n-1, asymmetric branches). Complexity is O(total length).
