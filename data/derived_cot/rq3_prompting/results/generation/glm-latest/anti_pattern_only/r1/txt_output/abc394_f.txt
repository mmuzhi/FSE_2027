```python
import sys

def main():
    input_data = sys.stdin.buffer.read().split()
    n = int(input_data[0])
    if n == 1:
        print(-1)
        return
    # adjacency
    adj = [[] for _ in range(n+1)]
    idx = 1
    for _ in range(n-1):
        a = int(input_data[idx]); b = int(input_data[idx+1]); idx += 2
        adj[a].append(b)
        adj[b].append(a)
    
    # BFS from root 1 to get parent and order
    parent = [0]*(n+1)
    order = []
    visited = [False]*(n+1)
    from collections import deque
    dq = deque([1])
    visited[1] = True
    parent[1] = 0
    while dq:
        u = dq.popleft()
        order.append(u)
        for w in adj[u]:
            if not visited[w]:
                visited[w] = True
                parent[w] = u
                dq.append(w)
    
    h = [0]*(n+1)
    best = -1  # max a
    # process in reverse BFS order
    for u in reversed(order):
        # children of u: neighbors except parent
        ch = [w for w in adj[u] if w != parent[u]]
        k = len(ch)
        if k >= 4:
            # top-4 h values
            vals = sorted((h[c] for c in ch), reverse=True)[:4]
            cand = 1 + sum(vals)
            if cand > best:
                best = cand
        if k >= 1:
            m = max(h[c] for c in ch)
            if m >= 1 and m > best:
                best = m
        if k >= 3:
            vals = sorted((h[c] for c in ch), reverse=True)[:3]
            h[u] = 1 + sum(vals)
        # else h[u] = 0
    print(3*best + 2 if best >= 1 else -1)

main()
```

```python
def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
```
    
```python
    adj = [[] for _ in range(n + 1)]
```
```python
    ptr = 1
```

```python
        for w in adj[u]:

                order.append(w)
```

```python
import sys

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    
    adj = [[] for _ in range(n + 1)]
    idx = 1
    for _ in range(n - 1):
        a = int(data[idx])
        b = int(data[idx + 1])
        idx += 2
        adj[a].append(b)
        adj[b].append(a)
    
    # BFS from vertex 1: parents and BFS order
    parent = [0] * (n + 1)
    parent[1] = -1
    order = [1]
    head = 0
    while head < len(order):
        u = order[head]
        head += 1
        for w in adj[u]:
            if parent[w] == 0:
                parent[w] = u
                order.append(w)
    
    # h[v]: max number of degree-4 vertices in a "branch" rooted at v
    # (connected subgraph inside v's subtree, containing v, where v has
    #  an upward edge to its parent and 0 or 3 downward edges)
    h = [0] * (n + 1)
    best = 0  # max number of degree-4 vertices in a whole alkane
    
    for u in reversed(order):
        p = parent[u]
        au = adj[u]
        if p > 0:
            vals = [h[w] for w in au if w != p]
        else:
            vals = [h[w] for w in au]
        if not vals:
            continue
        vals.sort(reverse=True)
        v0 = vals[0]
        # u as topmost vertex with degree 1: u + best branch at one child
        if v0 >= 1 and v0 > best:
            best = v0
        k = len(vals)
        if k >= 3:
            # u as degree-4 vertex inside a branch (3 children + parent)
            s = 1 + v0 + vals[1] + vals[2]
            h[u] = s
            if k >= 4:
                # u as topmost vertex with degree 4 (4 children)
                c4 = s + vals[3]
                if c4 > best:
                    best = c4
    
    print(3 * best + 2 if best >= 1 else -1)

main()
```

```python
import sys

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])

    adj = [[] for _ in range(n + 1)]
    idx = 1
    for _ in range(n - 1):
        a = int(data[idx])
        b = int(data[idx + 1])
        idx += 2
        adj[a].append(b)
        adj[b].append(a)

    # BFS from vertex 1 to compute parents and a top-down order
    parent = [0] * (n + 1)
    parent[1] = -1
    order = [1]
    head = 0
    while head < len(order):
        u = order[head]
        head += 1
        for w in adj[u]:
            if parent[w] == 0:
                parent[w] = u
                order.append(w)

    # h[v] = max # of degree-4 vertices in a branch rooted at v:
    # a connected subgraph of v's subtree containing v, where v keeps
    # an edge to its parent and uses exactly 0 or 3 edges to children.
    h = [0] * (n + 1)
    best = 0  # max # of degree-4 vertices over all alkanes

    for u in reversed(order):
        p = parent[u]
        au = adj[u]
        if p > 0:
            vals = [h[w] for w in au if w != p]
        else:
            vals = [h[w] for w in au]
        if not vals:
            continue
        vals.sort(reverse=True)
        v0 = vals[0]
        # u as the topmost vertex of the alkane with degree 1:
        # u plus the best branch hanging from one of its children
        if v0 >= 1 and v0 > best:
            best = v0
        k = len(vals)
        if k >= 3:
            # u is a degree-4 vertex inside a branch (parent + 3 children)
            s = 1 + v0 + vals[1] + vals[2]
            h[u] = s
            if k >= 4:
                # u as the topmost vertex of the alkane with degree 4
                c4 = s + vals[3]
                if c4 > best:
                    best = c4

    print(3 * best + 2 if best >= 1 else -1)

main()
```

Looking at this problem:

- An alkane is a tree with all degrees in {1,4}, at least one degree-4 vertex. If it has `a` degree-4 vertices, then counting degrees gives exactly `V = 3a + 2`. So maximizing vertices = maximizing `a`.
- A subgraph that is a tree must be connected, hence a connected subtree of T. Root T at vertex 1. The topmost vertex uses exactly 1 or 4 edges, all to its children.

- degree 4 (needs ≥4 children): `1 + sum of 4 largest h of children`;
- degree 1 (needs ≥1 child): `max h of children`, valid only if ≥ 1 (child must itself be degree-4, otherwise the subgraph is a single edge with no degree-4 vertex).
