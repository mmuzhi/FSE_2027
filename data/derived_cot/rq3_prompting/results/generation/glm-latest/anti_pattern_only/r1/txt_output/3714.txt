```python
from typing import List

class Solution:
    def minMaxSums(self, nums: List[int], k: int) -> int:
        MOD = 1_000_000_007
        n = len(nums)
        if n == 0 or k <= 0:
            return 0

        a = sorted(nums)

        # Modular inverses of 1..n (used to update C(m, k-1) incrementally).
        inv = [0] * (n + 1)
        inv[1] = 1
        for i in range(2, n + 1):
            inv[i] = (MOD - MOD // i) * inv[MOD % i] % MOD

        # S[m] = sum_{s=0}^{k-1} C(m, s) (mod MOD), computed via the recurrence
        #   S[m] = 2*S[m-1] - C(m-1, k-1),  S[0] = 1
        # (derived from Pascal's rule: sum_{s<=k-1} C(m,s) =
        #  S(m-1) + (S(m-1) - C(m-1, k-1))).
        j = k - 1
        S = [0] * n
        S[0] = 1
        binom = 1 if j == 0 else 0  # C(0, j)
        for m in range(1, n):
            S[m] = (2 * S[m - 1] - binom) % MOD  # binom currently = C(m-1, j)
            # Update binom to C(m, j)
            if m < j:
                binom = 0
            elif m == j:
                binom = 1
            else:
                binom = binom * m % MOD * inv[m - j] % MOD

        # In sorted (ascending) order, a[i] is the maximum of exactly S(i)
        # subsequences of size <= k (companions chosen from the i elements
        # before it), and the minimum of exactly S(n-1-i) subsequences
        # (companions chosen from the elements after it).
        total = 0
        for v, s_lo, s_hi in zip(a, S, S[::-1]):
            total += v * (s_lo + s_hi)
        return total % MOD
```
