```python
import sys, math

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    px, py = 0, 0
    dists = []
    idx = 1
    for i in range(n):
        x = int(data[idx]); y = int(data[idx+1]); idx += 2
        dists.append(math.hypot(x - px, y - py))
        px, py = x, y
    dists.append(math.hypot(px, py))
    print(f"{math.fsum(dists):.10f}")

main()
```
