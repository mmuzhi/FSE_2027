```python
class Solution:
    def resultsArray(self, nums: List[int], k: int) -> List[int]:
        n = len(nums)
        if k == 1:
            return nums[:]
        # run[i] = length of consecutive ascending run ending at index i
        run = [1] * n
        for i in range(1, n):
            if nums[i] == nums[i-1] + 1:
                run[i] = run[i-1] + 1
        return [nums[i + k - 1] if run[i + k - 1] >= k else -1 for i in range(n - k + 1)]
```

```python
from typing import List

class Solution:
    def resultsArray(self, nums: List[int], k: int) -> List[int]:
        n = len(nums)
        # run[i] = length of the strictly consecutive ascending run ending at index i
        run = [1] * n
        for i in range(1, n):
            if nums[i] == nums[i - 1] + 1:
                run[i] = run[i - 1] + 1
        
        # Window [i, i+k-1] is valid iff the run ending at i+k-1 covers the whole window,
        # in which case its maximum (and power) is nums[i+k-1].
        return [nums[i + k - 1] if run[i + k - 1] >= k else -1 for i in range(n - k + 1)]
```
