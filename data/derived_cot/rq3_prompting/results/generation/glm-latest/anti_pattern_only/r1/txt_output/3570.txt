    def countOfSubstrings(self, word: str, k: int) -> int:
        def atLeast(m):
            vowels = {'a','e','i','o','u'}
            cnt = {}
            consonants = 0
            l = 0
            total = 0
            n = len(word)

                        consonants -= 1
                    l += 1

class Solution:
    def countOfSubstrings(self, word: str, k: int) -> int:
        def at_least(m: int) -> int:
            # Counts substrings containing all 5 vowels and at least m consonants.
            vowels = {'a', 'e', 'i', 'o', 'u'}
            cnt = {}
            consonants = 0
            left = 0
            total = 0
            n = len(word)
            for r, ch in enumerate(word):
                if ch in vowels:
                    cnt[ch] = cnt.get(ch, 0) + 1
                else:
                    consonants += 1
                # While window [left, r] is valid, every substring starting at
                # `left` and ending at r..n-1 is also valid (supersets).
                while consonants >= m and len(cnt) == 5:
                    total += n - r
                    c = word[left]
                    if c in vowels:
                        cnt[c] -= 1
                        if cnt[c] == 0:
                            del cnt[c]
                    else:
                        consonants -= 1
                    left += 1
            return total

        # exactly k = (at least k) - (at least k+1)
        return at_least(k) - at_least(k + 1)
