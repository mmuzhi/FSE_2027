class Solution:
    def countKReducibleNumbers(self, s: str, k: int) -> int:
        MOD = 10**9 + 7
        L = len(s)
        
        # Pascal's triangle up to L
        C = [[0] * (L + 1) for _ in range(L + 1)]
        for i in range(L + 1):
            C[i][0] = 1
            for j in range(1, i + 1):
                C[i][j] = (C[i - 1][j - 1] + C[i - 1][j]) % MOD
        
        # steps[c]: min ops to reduce the integer c (1 <= c <= L) to 1
        steps = [0] * (L + 1)
        for c in range(2, L + 1):
            steps[c] = 1 + steps[bin(c).count('1')]
        
        ans = 0
        ones_before = 0
        for i, ch in enumerate(s):
            if ch == '1':
                rem = L - 1 - i
                rowC = C[rem]
                for extra in range(rem + 1):
                    c = ones_before + extra
                    if c == 0:
                        continue  # only the number 0, excluded
                    if c == 1 or 1 + steps[c] <= k:
                        ans = (ans + rowC[extra]) % MOD
                ones_before += 1
        
        return ans % MOD
