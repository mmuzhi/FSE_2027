```python
class Solution:
    def answerString(self, word: str, numFriends: int) -> str:
        if numFriends == 1:
            return word
        
        n = len(word)
        # Any piece of a split into numFriends parts has length at most n - numFriends + 1,
        # and every substring of that length bound is achievable as some piece.
        max_len = n - numFriends + 1
        
        # The lexicographically largest piece must start with the largest character,
        # since a single occurrence of that character (length 1 <= max_len) beats
        # any string starting with a smaller character.
        top = max(word)
        
        best = ""
        for i in range(n):
            if word[i] == top:
                # For a fixed start i, extending the substring (staying within max_len)
                # only makes it lexicographically larger, so take the longest allowed.
                cand = word[i:i + max_len]
                if cand > best:
                    best = cand
        
        return best
```
