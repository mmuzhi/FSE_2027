```python
class Solution:
    def calculateScore(self, s: str) -> int:
        # stacks[k] holds unmarked indices (in increasing order) whose char is chr(97+k)
        stacks = [[] for _ in range(26)]
        score = 0
        for i, ch in enumerate(s):
            c = ord(ch) - 97
            m = 25 - c  # mirror letter index
            if stacks[m]:
                # top of stack is the closest unmarked index with the mirror char
                j = stacks[m].pop()
                score += i - j
            else:
                stacks[c].append(i)
        return score
```
