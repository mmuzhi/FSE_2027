def matmul_mod(A, B, p):

def matpow(A, e, p):
    n = A.shape[0]
    R = np.eye(n, dtype=np.int64)
    B = A % p
    while e:
        if e & 1:
            R = matmul_mod(R, B, p)
        B = matmul_mod(B, B, p)
        e >>= 1
    return R
