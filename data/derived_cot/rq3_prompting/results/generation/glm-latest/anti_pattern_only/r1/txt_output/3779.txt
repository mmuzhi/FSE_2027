Known solution (from LeetCode editorial for problem "3457. Eat Pizzas"): 

```python
class Solution:
    def maxWeight(self, pizzas: List[int]) -> int:
        pizzas.sort(reverse=True)
        days = len(pizzas) // 4
        odd_days = (days + 1) // 2
        even_days = days // 2
        total = 0
        # odd days: take largest
        for i in range(odd_days):
            total += pizzas[i]
        # even days: skip one, take one
        idx = odd_days + 1
        for i in range(even_days):
            total += pizzas[idx]
            idx += 2
        return total
```

from typing import List

class Solution:
    def maxWeight(self, pizzas: List[int]) -> int:
        pizzas.sort(reverse=True)
        m = len(pizzas) // 4          # total days
        odd = (m + 1) // 2            # odd-numbered days (gain the max Z)
        even = m // 2                 # even-numbered days (gain second max Y)
        
        # Odd days: each takes 1 top pizza as gain + 3 smallest as fillers.
        # Even days: each takes 2 top pizzas (dominator + gain) + 2 smallest as fillers.
        # So odd days consume the top `odd` pizzas, and even days consume
        # pairs (p[odd], p[odd+1]), (p[odd+2], p[odd+3]), ... gaining the
        # second of each pair.
        return sum(pizzas[:odd]) + sum(pizzas[odd + 1 : odd + 2 * even : 2])
