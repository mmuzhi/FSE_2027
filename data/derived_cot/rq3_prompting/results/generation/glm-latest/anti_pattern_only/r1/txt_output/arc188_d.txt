def main():
    data = sys.stdin.buffer.read().split()

    Q = [0] * (N + 1)
    for u in range(N):
        Q[pairA[u]] = u
    
    nxt = [-1] * N
    indeg = [0] * N
    slot_used = [False] * (N + 1)
    for u in range(N):
        if B[u] != -1:
            v = (B[u] + 1) // 2
            if slot_used[v]:
                print(0)
                return
            slot_used[v] = True
            d = Q[v]
            nxt[d] = u
            indeg[u] += 1
    
    color = [A[u] & 1 for u in range(N)]
    
    # detect cycles in functional graph
    state = [0] * N
    in_cycle = [False] * N
    for u in range(N):
        if state[u]:
            continue
        path = []
        pos = {}
        cur = u
        while cur != -1 and state[cur] == 0:
            pos[cur] = len(path)
            path.append(cur)
            state[cur] = 1
            cur = nxt[cur]
        if cur != -1 and state[cur] == 1:
            cyc = path[pos[cur]:]
            c0 = color[cyc[0]]
            mono = True
            for x in cyc:
                if color[x] != c0:
                    mono = False
                    break
            if mono:
                print(0)
                return
            for x in cyc:
                in_cycle[x] = True
        for x in path:
            state[x] = 2
    
    a_cnt = b_cnt = c_cnt = 0
    
    # path blocks: starts = nxt defined, indeg 0, not in cycle
    for u in range(N):
        if nxt[u] != -1 and indeg[u] == 0 and not in_cycle[u]:
            nodes = []
            cur = u
            while cur != -1:
                nodes.append(cur)
                cur = nxt[cur]
            has1 = any(color[x] == 1 for x in nodes)
            has0 = any(color[x] == 0 for x in nodes)
            if has1 and has0:
                c_cnt += 1
            elif has1:
                a_cnt += 1
            else:
                b_cnt += 1
    
    # free singletons
    for u in range(N):
        if nxt[u] == -1 and indeg[u] == 0 and not in_cycle[u]:
            if color[u] == 1:
                a_cnt += 1
            else:
                b_cnt += 1

    fact = [1] * (K + 1)
