```python
class Solution:
    def maximumSubarrayXor(self, nums: List[int], queries: List[List[int]]) -> List[int]:
        n = len(nums)
        # F_L[i] = xor score of nums[i..i+L-1]
        # G_L[i] = max score over subarrays of nums[i..i+L-1]
        F = list(nums)
        G = list(nums)
        # store G per length
        Gs = [None, G]  # Gs[L] for L=1
        for L in range(2, n + 1):
            F = [x ^ y for x, y in zip(F, F[1:])]
            Gp = G  # G_{L-1}, length n-L+2
            G = [max(a, b, c) for a, b, c in zip(Gp, Gp[1:], F)]
            Gs.append(G)
        # answer queries
        out = []
        for l, r in queries:
            out.append(Gs[r - l + 1][l])
        return out
```

=== ANSWER ===
**Approach:**
- Let `f(i, j)` = XOR score of `nums[i..j]`. By linearity over GF(2), `f(i, j) = f(i, j-1) XOR f(i+1, j)` (Pascal-triangle-style DP).
- Let `g(l, r)` = max score over all subarrays inside `nums[l..r]`. Any subarray either lies in `[l+1, r]`, or in `[l, r-1]`, or is exactly `[l, r]`, so `g(l, r) = max(g(l+1, r), g(l, r-1), f(l, r))`.
- Compute both DP tables diagonal-by-diagonal (by length), storing `g` only for lengths that appear in queries, then answer each query in O(1). Total O(n² + q).

Verified against both examples (e.g., `f` for `[8,4,32,16]` = 60, `g(0,2)` for `[2,8,4]` = 12, etc.).

```python
from operator import xor
from typing import List


class Solution:
    def maximumSubarrayXor(self, nums: List[int], queries: List[List[int]]) -> List[int]:
        n = len(nums)
        if not queries:
            return []

        need = set(r - l + 1 for l, r in queries)
        maxlen = max(need)

        # F[i] = xor score of nums[i .. i+L-1] for current length L
        # G[i] = max xor score over all subarrays of nums[i .. i+L-1]
        F = list(nums)
        G = list(nums)

        store = {}
        if 1 in need:
            store[1] = G

        for L in range(2, maxlen + 1):
            # f(i, i+L-1) = f(i, i+L-2) ^ f(i+1, i+L-1)
            F = list(map(xor, F, F[1:]))
            # g(i, i+L-1) = max(g(i+1, i+L-1), g(i, i+L-2), f(i, i+L-1))
            G = list(map(max, G, G[1:], F))
            if L in need:
                store[L] = G

        return [store[r - l + 1][l] for l, r in queries]
```
