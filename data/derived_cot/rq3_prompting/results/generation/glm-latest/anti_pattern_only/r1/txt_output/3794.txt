from typing import List

class Solution:
    def minTime(self, skill: List[int], mana: List[int]) -> int:
        n = len(skill)
        m = len(mana)

        # prefix[i] = sum of skill[0..i-1]
        prefix = [0] * (n + 1)
        for i in range(n):
            prefix[i + 1] = prefix[i] + skill[i]

        # For j >= 1:  s_j = s_{j-1} + max_{i} ( prefix[i+1]*mana[j-1] - prefix[i]*mana[j] )
        # Model as upper envelope of lines: y_i(x) = prefix[i+1]*x - prefix[i]
        # (slope strictly increasing, intercept strictly decreasing)
        # Query x = mana[j-1]/mana[j] represented by numerator a, denominator b.

        hs = []  # hull slopes  (prefix[i+1])
        hc = []  # hull intercepts (-prefix[i])
        for i in range(n):
            s = prefix[i + 1]
            c = -prefix[i]
            while len(hs) >= 2 and \
                  (hc[-2] - c) * (hs[-1] - hs[-2]) <= (hc[-2] - hc[-1]) * (s - hs[-2]):
                hs.pop()
                hc.pop()
            hs.append(s)
            hc.append(c)

        k = len(hs)
        total = 0
        for j in range(1, m):
            a = mana[j - 1]  # numerator
            b = mana[j]      # denominator
            lo, hi = 0, k - 1
            while lo < hi:
                mid = (lo + hi) >> 1
                # x >= X_mid  <=>  a*(hs[mid+1]-hs[mid]) >= b*(hc[mid]-hc[mid+1])
                if a * (hs[mid + 1] - hs[mid]) >= b * (hc[mid] - hc[mid + 1]):
                    lo = mid + 1
                else:
                    hi = mid
            total += a * hs[lo] + b * hc[lo]

        return total + prefix[n] * mana[m - 1]

from typing import List

class Solution:
    def minTime(self, skill: List[int], mana: List[int]) -> int:
        n, m = len(skill), len(mana)

        # prefix[i] = skill[0] + ... + skill[i-1]
        prefix = [0] * (n + 1)
        for i, s in enumerate(skill):
            prefix[i + 1] = prefix[i] + s

        # If potion j enters wizard 0 at time s_j, it reaches wizard i at
        # s_j + prefix[i]*mana[j] and leaves the lab at s_j + prefix[n]*mana[j].
        # Wizard i is free for potion j only after finishing potion j-1:
        #   s_j + prefix[i]*mana[j] >= s_{j-1} + prefix[i+1]*mana[j-1]
        # => s_j >= s_{j-1} + max_i( prefix[i+1]*mana[j-1] - prefix[i]*mana[j] )
        # The max is the upper envelope of lines y = prefix[i+1]*x - prefix[i]
        # evaluated at x = mana[j-1]/mana[j].

        # Build upper hull; slopes strictly increase, intercepts strictly decrease.
        hs, hc = [], []
        for i in range(n):
            s, c = prefix[i + 1], -prefix[i]
            while len(hs) >= 2 and \
                    (hc[-2] - c) * (hs[-1] - hs[-2]) <= (hc[-2] - hc[-1]) * (s - hs[-2]):
                hs.pop()
                hc.pop()

            hc.append(c)

        k = len(hs)
        finish = 0
        for j in range(1, m):
            a, b = mana[j - 1], mana[j]  # query point x = a / b
            lo, hi = 0, k - 1

                if a * (hs[mid + 1] - hs[mid]) >= b * (hc[mid] - hc[mid + 1]):

from typing import List

class Solution:
    def minTime(self, skill: List[int], mana: List[int]) -> int:
        n, m = len(skill), len(mana)

        # prefix[i] = skill[0] + ... + skill[i-1]
        prefix = [0] * (n + 1)
        for i in range(n):
            prefix[i + 1] = prefix[i] + skill[i]

        # Potion j enters at s_j; it reaches wizard i at s_j + prefix[i]*mana[j].
        # Wizard i must finish potion j-1 first:
        #   s_j >= s_{j-1} + max_i( prefix[i+1]*mana[j-1] - prefix[i]*mana[j] )
        # The max is the upper envelope of lines y = prefix[i+1]*x - prefix[i]
        # at x = mana[j-1] / mana[j]  (times mana[j]).

        # Upper hull: slopes strictly increasing, intercepts strictly decreasing.
        hs, hc = [], []
        for i in range(n):
            s, c = prefix[i + 1], -prefix[i]
            while len(hs) >= 2 and (hc[-2] - c) * (hs[-1] - hs[-2]) <= (hc[-2] - hc[-1]) * (s - hs[-2]):
                hs.pop()
                hc.pop()
            hs.append(s)
            hc.append(c)

        k = len(hs)
        t = 0
        for j in range(1, m):
            a, b = mana[j - 1], mana[j]
            lo, hi = 0, k - 1
            while lo < hi:
                mid = (lo + hi) >> 1
                if a * (hs[mid + 1] - hs[mid]) >= b * (hc[mid] - hc[mid + 1]):
                    lo = mid + 1
                else:
                    hi = mid
            t += a * hs[lo] + b * hc[lo]

        return t + prefix[n] * mana[m - 1]

Looking at this problem:

- If potion `j` enters wizard 0 at time `s_j`, it reaches wizard `i` at `s_j + prefix[i]*mana[j]` (no waiting allowed), finishing at `s_j + prefix[n]*mana[j]`.
- Wizard `i` must finish potion `j-1` before starting potion `j`, giving:
  `s_j >= s_{j-1} + max_i( prefix[i+1]*mana[j-1] - prefix[i]*mana[j] )`
- Since constraints form a simple chain with constants independent of `s`, greedy earliest starts are optimal. The max is the upper envelope of lines `y = prefix[i+1]*x - prefix[i]` evaluated at `x = mana[j-1]/mana[j]` (scaled by `mana[j]`), solvable with a convex hull trick using exact integer arithmetic (slopes strictly increase, intercepts strictly decrease).

I verified this against all three examples (52, 54, 86 start times for example 1, giving 110) and several hand-simulated cases including hull-collapse edge cases (all equal skills).

        # The max is the upper envelope of lines y = prefix[i+1]*x - prefix[i]
        # evaluated at x = mana[j-1] / mana[j] (value scaled by mana[j]).
