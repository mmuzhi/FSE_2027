import sys
from math import atan2

def main():
    input_data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(input_data[idx]); idx += 1
    # points: list of (x, y, color, id) color 0 = P, 1 = Q
    pts = []
    for i in range(n):
        a = int(input_data[idx]); b = int(input_data[idx+1]); idx += 2
        pts.append((a, b, 0, i))
    for j in range(n):
        c = int(input_data[idx]); d = int(input_data[idx+1]); idx += 2
        pts.append((c, d, 1, j))
    
    R = [0] * n  # R[p_index] = q_index
    
    # iterative stack of lists of point tuples
    stack = [pts]
    while stack:
        S = stack.pop()
        if not S:
            continue
        if len(S) == 2:
            # one P and one Q
            if S[0][2] == 0:
                R[S[0][3]] = S[1][3]
            else:
                R[S[1][3]] = S[0][3]
            continue
        # pivot: lexicographic min (y, x)

import sys
from math import atan2

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    ptr = 1
    pts = []
    for i in range(n):
        a = int(data[ptr]); b = int(data[ptr+1]); ptr += 2
        pts.append((a, b, 0, i))
    for j in range(n):
        c = int(data[ptr]); d = int(data[ptr+1]); ptr += 2
        pts.append((c, d, 1, j))

    R = [-1] * n
    stack = [pts]
    while stack:
        S = stack.pop()
        if not S:
            continue
        # pivot: lexicographic min (y, x)
        p = S[0]
        for t in S:
            if (t[1], t[0]) < (p[1], p[0]):
                p = t
        px, py, pc, pid = p
        others = [(x, y, c, i) for (x, y, c, i) in S if not (x == px and y == py and c == pc and i == pid)]
        others.sort(key=lambda t: atan2(t[1] - py, t[0] - px))
        f = 0
        kstar = 0
        for k in range(len(others)):

            if f == 0:
                kstar = k + 1
        q = others[kstar]
        if pc == 0:

        else:
            R[q[3]] = pid
        if kstar > 0:
            stack.append(others[:kstar])
        if kstar + 1 < len(others):
            stack.append(others[kstar+1:])

    print(' '.join(str(r + 1) for r in R))

import sys
from math import atan2

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    ptr = 1
    pts = []
    for i in range(n):
        a = int(data[ptr]); b = int(data[ptr + 1]); ptr += 2
        pts.append((a, b, 0, i))
    for j in range(n):
        c = int(data[ptr]); d = int(data[ptr + 1]); ptr += 2
        pts.append((c, d, 1, j))

    R = [-1] * n
    stack = [pts]
    while stack:
        S = stack.pop()
        if not S:
            continue
        p = min(S, key=lambda t: (t[1], t[0]))
        px, py, pc, pid = p
        others = [t for t in S if t != p]
        others.sort(key=lambda t: atan2(t[1] - py, t[0] - px))
        f = 0
        kstar = 0
        for k, t in enumerate(others):
            f += 1 if t[2] else -1
            if f == 0:
                kstar = k + 1
        q = others[kstar]
        if pc == 0:
            R[pid] = q[3]
        else:
            R[q[3]] = pid
        if kstar:
            stack.append(others[:kstar])
        if kstar + 1 < len(others):
            stack.append(others[kstar + 1:])

    sys.stdout.write(' '.join(str(r + 1) for r in R) + '\n')

main()

Looking at this problem:

- We need a perfect matching between P-points and Q-points with pairwise non-intersecting segments. Such a matching always exists (classical result), so we never print -1.

**Constructive algorithm (exact, integer-safe):** Recursively, take the pivot p = lexicographically smallest (y, x) point of the current balanced set (equal #P and #Q). All other points then lie at polar angles in [0, π) around p, all distinct (no three collinear). Sort them by angle. Scanning in angular order, track f = #Q − #P; let k* be the last prefix length where f = 0. Then the next point q = others[k*] has the opposite color of p, the prefix is balanced, and the suffix is balanced. Match p with q. Points before q (smaller angle) lie strictly on one side of line(p,q), points after strictly on the other side, so recursing on prefix and suffix yields segments that never cross each other or segment p–q (open half-planes are convex).

Verified by hand on Sample 1 (produces valid `1 2 3`) and traced through Sample 2 producing a valid permutation. Complexity is at most O(N² log N), trivial for N = 300. `atan2` is safe here since distinct integer directions differ by ≥ 2·10⁻⁸ radians, far above floating point error.
