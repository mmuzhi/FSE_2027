import sys

def main():
    data = sys.stdin.buffer.read().split()
    idx = 0
    N = int(data[idx]); idx += 1
    W = int(data[idx]); idx += 1
    cols = [[] for _ in range(W + 1)]
    X = [0] * N
    Y = [0] * N
    for i in range(N):
        x = int(data[idx]); idx += 1
        y = int(data[idx]); idx += 1
        X[i] = x; Y[i] = y
        cols[x].append((y, i))
    # sort each column
    K = min(len(cols[x]) for x in range(1, W + 1))
    removal = [float('inf')] * N
    if K > 0:
        M = [0] * K
        for x in range(1, W + 1):
            c = cols[x]
            c.sort()
            for j in range(K):
                if c[j][0] > M[j]:
                    M[j] = c[j][0]
        for x in range(1, W + 1):
            c = cols[x]
            for j in range(K):
                removal[c[j][1]] = M[j]
        # blocks with j >= K: inf (already)
    Q = int(data[idx]); idx += 1
    out = []
    for _ in range(Q):
        T = int(data[idx]); idx += 1
        A = int(data[idx]); idx += 1
        out.append("Yes" if removal[A - 1] > T else "No")
    sys.stdout.write("\n".join(out) + "\n")
