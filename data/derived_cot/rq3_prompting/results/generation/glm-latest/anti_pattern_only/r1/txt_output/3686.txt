```python
import numpy as np
class Solution:
    def beautifulSplits(self, nums):
        n = len(nums)
        if n < 3: return 0
        arr = np.array(nums, dtype=np.int8)  # or int32
        # Z-array
        Z = self.z_function(nums)
        A = 0
        for i in range(1, (n-1)//2 + 1):
            if Z[i] >= i:
                A += n - 2*i
        total = A
        okZ = np.array([Z[i] >= i for i in range(n)], dtype=bool)  # okZ[i]
        for d in range(1, (n-1)//2 + 1):
            m = (arr[:-d] == arr[d:])  # length n-d
            # prefix sums
            P = np.zeros(n-d+1, dtype=np.int32); np.cumsum(m, out=P[1:])  # hmm dtype
            N = n - 2*d
            if N < 1: break
            # W[i] = P[i+d] - P[i] == d for i in [0, N]
            W = (P[d:N+1+d] - P[:N+1]) == d  # boolean, length N+1
            # count i in [1, N] with W[i] and (i > d or not okZ[i])
            idx = np.arange(1, N+1)
            mask = (idx > d) | ~okZ[1:N+1]
            total += np.count_nonzero(W[1:] & mask)
        return total
```

```python
class Solution:
    def beautifulSplits(self, nums: List[int]) -> int:
        n = len(nums)
        if n < 3:
            return 0
        # Z-array
        z = [0] * n
        z[0] = n
        l = r = 0
        for i in range(1, n):
            zi = 0
            if i < r:
                zi = min(r - i, z[i - l])
            while i + zi < n and nums[zi] == nums[i + zi]:
                zi += 1
            z[i] = zi
            if i + zi > r:
                l, r = i, i + zi
        # |A|
        A = 0
        for i in range(1, (n - 1) // 2 + 1):
            if z[i] >= i:
                A += n - 2 * i
        # Zlt bytes
        zlt = bytearray(n)
        for i in range(1, n):
            if z[i] < i:
                zlt[i] = 1
        zltB = bytes(zlt)
        
        S = int.from_bytes(bytes(nums), 'big')
        
        # popcount
        try:
            (0).bit_count
            def popcount(v): return v.bit_count()
        except AttributeError:
            def popcount(v): return bin(v).count('1')
        
        TBL = bytes([1] + [0] * 255)  # 0 -> 1, else 0
        
        total = A
        for d in range(1, (n - 1) // 2 + 1):
```
