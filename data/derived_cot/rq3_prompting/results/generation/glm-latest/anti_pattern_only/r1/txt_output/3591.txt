from typing import List

class Solution:
    def shiftDistance(self, s: str, t: str, nextCost: List[int], previousCost: List[int]) -> int:
        # Doubled nextCost for circular prefix sums (forward shifts)
        nc2 = nextCost * 2
        # Reverse previousCost: a backward step from index j pays previousCost[j],
        # which maps to index (25 - j) in the reversed array, making the walk increasing.
        pcr2 = [previousCost[25 - j] for j in range(26)] * 2
        
        pn = [0] * 53
        for k in range(52):
            pn[k + 1] = pn[k] + nc2[k]
        
        pr = [0] * 53
        for k in range(52):
            pr[k + 1] = pr[k] + pcr2[k]
        
        total = 0
        for cs, ct in zip(s, t):
            a = ord(cs) - 97
            b = ord(ct) - 97
            if a == b:
                continue
            # Forward: d steps, paying nextCost at a, a+1, ..., a+d-1 (mod 26)
            d = (b - a) % 26
            fwd = pn[a + d] - pn[a]
            # Backward: e steps, paying previousCost at a, a-1, ..., b+1 (mod 26)
            e = (a - b) % 26
            start = 25 - a  # in [0, 25]
            bwd = pr[start + e] - pr[start]
            total += min(fwd, bwd)
        return total
