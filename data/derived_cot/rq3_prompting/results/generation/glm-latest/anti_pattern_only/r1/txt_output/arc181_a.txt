import sys

def main():
    data = sys.stdin.buffer.read().split()
    t = int(data[0])
    pos = 1
    out = []
    for _ in range(t):
        n = int(data[pos]); pos += 1
        end = pos + n
        vals = list(map(int, data[pos:end]))
        pos = end
        is_id = True
        one_op = False
        mx = 0
        for i, v in enumerate(vals, 1):
            if v == i:
                if mx < i:
                    one_op = True
            else:
                is_id = False
            if v > mx:
                mx = v
        if is_id:
            out.append('0')
        elif vals[0] == n and vals[-1] == 1:
            out.append('3')
        elif one_op:
            out.append('1')
        else:
            out.append('2')
    sys.stdout.write('\n'.join(out) + '\n')
