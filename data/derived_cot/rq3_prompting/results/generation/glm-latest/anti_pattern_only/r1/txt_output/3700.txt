class Solution:
    def subsequencesWithMiddleMode(self, nums: List[int]) -> int:
        MOD = 10**9 + 7
        n = len(nums)
        # coordinate compression
        sorted_vals = sorted(set(nums))
        comp = {v: i for i, v in enumerate(sorted_vals)}
        m = len(sorted_vals)
        a = [comp[x] for x in nums]
        
        # counts
        total = [0] * m
        for x in a:
            total[x] += 1
        
        l = [0] * m
        r = total[:]  # r[u] = count of u in nums[j..n-1] initially; we'll remove nums[j] at start of each iteration
        
        # aggregates
        A = 0  # sum l_u * r_u
        B = 0  # sum l_u * r_u^2
        Cc = 0  # sum r_u * l_u^2
        Sl2 = 0  # sum C(l_u, 2)
        Sr2 = 0  # sum C(r_u, 2)
        for u in range(m):
            Sr2 += r[u] * (r[u] - 1) // 2
        
        def comb2(x):
            return x * (x - 1) // 2
        
        ans = 0
        for j in range(n):
            v = a[j]
            # remove nums[j] from r
            rv = r[v]
            # update aggregates for r[v]: rv -> rv-1
            A -= l[v]
            B -= l[v] * (2 * rv - 1)
            Cc -= l[v] * l[v]
            Sr2 -= (rv - 1)
            r[v] = rv - 1
            rv = rv - 1  # now rv = count of v strictly right of j
            
            lv = l[v]
            left_total = j
            right_total = n - 1 - j
            L_nonv = left_total - lv
            R_nonv = right_total - rv
            
            T = comb2(left_total) * comb2(right_total)
            W0 = comb2(L_nonv) * comb2(R_nonv)
            W1 = lv * L_nonv * comb2(R_nonv) + comb2(L_nonv) * rv * R_nonv
            
            # D_left, D_right

            D_right = comb2(R_nonv) - (Sr2 - comb2(rv))

            S2p = Cc - rv * lv * lv

            Sl2 += lv_old

        return ans % MOD

from typing import List

class Solution:
    def subsequencesWithMiddleMode(self, nums: List[int]) -> int:
        MOD = 10**9 + 7
        n = len(nums)
        
        # coordinate compression
        sorted_vals = sorted(set(nums))
        comp = {v: i for i, v in enumerate(sorted_vals)}
        m = len(sorted_vals)
        a = [comp[x] for x in nums]
        
        r = [0] * m
        for x in a:
            r[x] += 1
        l = [0] * m
        
        # aggregates
        A = 0   # sum_u l[u]*r[u]
        B = 0   # sum_u l[u]*r[u]^2
        Cc = 0  # sum_u r[u]*l[u]^2
        Sl2 = 0 # sum_u C(l[u],2)
        Sr2 = 0 # sum_u C(r[u],2)
        for u in range(m):
            Sr2 += r[u] * (r[u] - 1) // 2
        
        def c2(x):
            return x * (x - 1) // 2
        
        ans = 0
        for j in range(n):
            v = a[j]
            # remove a[j] from r
            old_r = r[v]
            A -= l[v]
            B -= l[v] * (2 * old_r - 1)
            Cc -= l[v] * l[v]
            Sr2 -= old_r - 1
            r[v] = old_r - 1
            rv = old_r - 1
            
            lv = l[v]
            left_total = j
            right_total = n - 1 - j
            Lnv = left_total - lv
            Rnv = right_total - rv
            
            T = c2(left_total) * c2(right_total)
            W0 = c2(Lnv) * c2(Rnv)
            W1 = lv * Lnv * c2(Rnv) + c2(Lnv) * rv * Rnv
            
            Dl = c2(Lnv) - (Sl2 - c2(lv))
            Dr = c2(Rnv) - (Sr2 - c2(rv))
            
            S1 = A - lv * rv
            S2 = B - lv * rv * rv
            S2p = Cc - rv * lv * lv
            
            V1L = lv * (Lnv * Dr - Rnv * S1 + S2)
            V1R = rv * (Rnv * Dl - Lnv * S1 + S2p)
            
            ans = (ans + T - W0 - W1 + V1L + V1R) % MOD
            
            # add a[j] to l
            A += rv
            B += rv * rv
            Cc += rv * (2 * lv + 1)
            Sl2 += lv
            l[v] = lv + 1
        
        return ans % MOD

from typing import List

class Solution:
    def subsequencesWithMiddleMode(self, nums: List[int]) -> int:
        MOD = 10**9 + 7
        n = len(nums)
        
        # Coordinate compression
        sorted_vals = sorted(set(nums))
        comp = {val: i for i, val in enumerate(sorted_vals)}
        m = len(sorted_vals)
        a = [comp[x] for x in nums]
        
        r = [0] * m
        for x in a:
            r[x] += 1
        l = [0] * m
        
        # Aggregates over all values u:
        A = 0    # sum l[u] * r[u]
        B = 0    # sum l[u] * r[u]^2
        Cc = 0   # sum r[u] * l[u]^2
        Sl2 = 0  # sum C(l[u], 2)
        Sr2 = 0  # sum C(r[u], 2)
        for u in range(m):
            Sr2 += r[u] * (r[u] - 1) // 2
        
        def c2(x):
            return x * (x - 1) // 2
        
        ans = 0
        for j in range(n):
            v = a[j]
            # Remove a[j] from r (r[v]: old_r -> old_r - 1)
            old_r = r[v]
            A -= l[v]
            B -= l[v] * (2 * old_r - 1)
            Cc -= l[v] * l[v]
            Sr2 -= old_r - 1
            r[v] = old_r - 1
            rv = old_r - 1
            
            lv = l[v]
            left_total = j
            right_total = n - 1 - j
            Lnv = left_total - lv
            Rnv = right_total - rv
            
            T = c2(left_total) * c2(right_total)
            W0 = c2(Lnv) * c2(Rnv)
            W1 = lv * Lnv * c2(Rnv) + c2(Lnv) * rv * Rnv
            
            Dl = c2(Lnv) - (Sl2 - c2(lv))
            Dr = c2(Rnv) - (Sr2 - c2(rv))
            
            S1 = A - lv * rv
            S2 = B - lv * rv * rv
            S2p = Cc - rv * lv * lv
            
            V1L = lv * (Lnv * Dr - Rnv * S1 + S2)
            V1R = rv * (Rnv * Dl - Lnv * S1 + S2p)
            
            ans = (ans + T - W0 - W1 + V1L + V1R) % MOD
            
            # Add a[j] to l (l[v]: lv -> lv + 1)
            A += rv
            B += rv * rv
            Cc += rv * (2 * lv + 1)
            Sl2 += lv
            l[v] = lv + 1
        
        return ans % MOD
