```python
from typing import List
import math

class Solution:
    def minNumberOfSeconds(self, mountainHeight: int, workerTimes: List[int]) -> int:
        times = sorted(workerTimes)
        target = mountainHeight
        
        def feasible(T: int) -> bool:
            total = 0
            for t in times:
                m = T // t
                if m == 0:
                    # remaining workers have larger t, so also m == 0
                    break
                x = (math.isqrt(8 * m + 1) - 1) // 2
                total += x
                if total >= target:
                    return True
            return False
        
        lo = 1
        hi = times[0] * mountainHeight * (mountainHeight + 1) // 2
        while lo < hi:
            mid = (lo + hi) // 2
            if feasible(mid):
                hi = mid
            else:
                lo = mid + 1
        return lo
```

=== ANSWER ===
Looking at this problem:

- **Key insight**: Binary search on the answer T. For a given time T, worker i can reduce x units where x is the largest integer with `workerTimes[i] * x*(x+1)/2 <= T`. Sum these maximum reductions across all workers; T is feasible if the sum ≥ mountainHeight.
- **Finding max x**: With `m = T // t`, we need max x with `x*(x+1) <= 2m`, which solves exactly to `x = (isqrt(8m+1) - 1) // 2` (since `x(x+1) <= 2m ⟺ (2x+1)² <= 8m+1`).
- **Upper bound**: The fastest worker doing everything alone: `min(workerTimes) * h(h+1)/2`.

Verified against all three examples (T=3 for example 1, T=12 for example 2, T=15 for example 3, with T-1 infeasible in each case).

```python
from typing import List
import math

class Solution:
    def minNumberOfSeconds(self, mountainHeight: int, workerTimes: List[int]) -> int:
        times = sorted(workerTimes)  # ascending: largest contributions first, early exit
        target = mountainHeight

        def feasible(T: int) -> bool:
            total = 0
            for t in times:
                m = T // t
                if m == 0:
                    # sorted ascending -> all remaining workers also can't do anything
                    break
                # max x with x*(x+1) <= 2*m
                x = (math.isqrt(8 * m + 1) - 1) // 2
                total += x
                if total >= target:
                    return True
            return False

        lo = 1
        hi = times[0] * mountainHeight * (mountainHeight + 1) // 2
        while lo < hi:
            mid = (lo + hi) // 2
            if feasible(mid):
                hi = mid
            else:
                lo = mid + 1
        return lo
```
