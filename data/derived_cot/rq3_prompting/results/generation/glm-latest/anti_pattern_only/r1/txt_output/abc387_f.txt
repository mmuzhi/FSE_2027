import sys
import numpy as np

def main():
    data = sys.stdin.buffer.read().split()
    N = int(data[0]); M = int(data[1])
    A = [0] * (N + 1)
    for i in range(1, N + 1):
        A[i] = int(data[1 + i])
    
    MOD = 998244353
    
    indeg = [0] * (N + 1)
    for i in range(1, N + 1):
        indeg[A[i]] += 1
    
    # prod[i][v]: accumulated T_i(v); index 0 = 0, indices 1..M
    prod = np.zeros((N + 1, M + 1), dtype=np.int64)
    prod[:, 1:] = 1
    
    stack = [i for i in range(1, N + 1) if indeg[i] == 0]
    removed = bytearray(N + 1)
    
    while stack:
        j = stack.pop()
        removed[j] = 1
        G = np.cumsum(prod[j]) % MOD
        p = A[j]
        prod[p] *= G
        prod[p] %= MOD
        indeg[p] -= 1
        if indeg[p] == 0:
            stack.append(p)
    
    visited = bytearray(N + 1)
    ans = 1
    for c in range(1, N + 1):
        if removed[c] or visited[c]:
            continue
        visited[c] = 1
        comp = prod[c].copy()
        d = A[c]
        while d != c:
            visited[d] = 1
            comp *= prod[d]
            comp %= MOD
            d = A[d]
        total = int(comp[1:].sum()) % MOD
        ans = ans * total % MOD
    
    sys.stdout.write(str(ans) + "\n")

main()

import sys

    MOD = 998244353

    indeg = [0] * (N + 1)
    for i in range(1, N + 1):
        indeg[A[i]] += 1

    try:
        import numpy as np
    except Exception:
        np = None

    # prod[i][v] = accumulated T_i(v): number of assignments of the subtree
    # rooted at i (within its tree) with x_i = v, as a product of children's
    # prefix sums. Index 0 is kept at 0. Initially "empty product" = 1.

Looking at this problem:

- The functional graph (i → A_i) splits into components, each with exactly one cycle. On a cycle, all values must be equal (chain of ≤ around the cycle).
- Each cycle node is the root of a tree; within a tree, child ≤ parent. If T_r(v) = #assignments of the subtree rooted at r with x_r = v, then T_r(v) = Π over children c of G_c(v), where G_c(v) = Σ_{w≤v} T_c(w) (prefix sum).
- Component answer = Σ_{v=1..M} Π_{c on cycle} T_c(v); total answer = product over components.

I peel tree nodes (Kahn-style: a node is processed once all children are), maintaining a length-(M+1) value array per node (index 0 = 0). Cycle nodes are never peeled; afterwards I traverse each cycle, multiply the rows pointwise, and sum over v. This is O(N·M), vectorized with numpy (values stay < 2^63 since (MOD-1)² < 2^63 and cumsums of ≤2026 terms < 2^41). A pure-Python fallback is included.

Verified against all three samples (6, 2025, 10010).
