import sys
from collections import Counter

def main():
    input_data = sys.stdin.buffer.read().split()
    n = int(input_data[0])
    a = list(map(int, input_data[1:1+n]))
    cnt = Counter(a)
    best_val = -1
    best_idx = -1
    for i, v in enumerate(a):
        if cnt[v] == 1 and v > best_val:
            best_val = v
            best_idx = i + 1
    print(best_idx)

main()
