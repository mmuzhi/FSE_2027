```python
class Solution:
    def minValidStrings(self, words: List[str], target: str) -> int:
        n = len(target)
        d = [0] * n
        for w in words:
            m = len(w)
            # build failure function
            fail = [0] * m
            k = 0
            for i in range(1, m):
                c = w[i]
                while k and w[k] != c:
                    k = fail[k - 1]
                if w[k] == c:
                    k += 1
                fail[i] = k
            # scan target
            k = 0
            if k < m and w[0] == target[0]:  # handle j=0 inside loop; just do loop

                    k = fail[k - 1]
                if k < m and w[k] == c:
                    k += 1
                if k > d[j]:
                    d[j] = k
        # dp

        dp[0] = 0
        for m in range(1, n + 1):
```

```python
class Solution:
    def minValidStrings(self, words: List[str], target: str) -> int:
        n = len(target)
        d = [0] * n  # d[j] = length of longest prefix of any word that is a suffix of target[:j+1]
        
        # Remove words that are prefixes of other words (redundant).
        ws = sorted(set(words))
        ws = [w for i, w in enumerate(ws) if i + 1 == len(ws) or not ws[i + 1].startswith(w)]
        
        for w in ws:
            p = w + '#'  # sentinel: never matches target chars
            lp = len(p)
            fl = [0] * lp
            k = 0
            for i in range(1, lp):
                c = p[i]
                while k and p[k] != c:
                    k = fl[k - 1]
                if p[k] == c:
                    k += 1
                fl[i] = k
```

```python
from typing import List

class Solution:
    def minValidStrings(self, words: List[str], target: str) -> int:
        n = len(target)
        # d[j] = length of the longest prefix of any word that is a suffix of target[:j+1]
        d = [0] * n

        # Drop words that are prefixes of other words (they add no new valid prefixes).
        ws = sorted(set(words))
        ws = [w for i, w in enumerate(ws)
              if i + 1 == len(ws) or not ws[i + 1].startswith(w)]

        for w in ws:
            p = w + '#'          # sentinel never matches a lowercase letter
            lp = len(p)
            fl = [0] * lp        # KMP failure function of p
            k = 0
            for i in range(1, lp):
                c = p[i]
                while k and p[k] != c:
                    k = fl[k - 1]
                if p[k] == c:
                    k += 1
                fl[i] = k
            # run KMP automaton over target; state k = longest prefix of w
            # that is a suffix of target[:j+1]
            k = 0
            for j in range(n):
                c = target[j]
                while k and p[k] != c:
                    k = fl[k - 1]
                if p[k] == c:
                    k += 1
                if k > d[j]:
                    d[j] = k

        # dp[i] = min pieces covering target[:i]; dp is non-decreasing, hence the
        # optimal last piece ending at i-1 is the longest one (length d[i-1]).
        INF = float('inf')
        dp = [INF] * (n + 1)
        dp[0] = 0
        for i in range(1, n + 1):
            L = d[i - 1]
            if L:
                v = dp[i - L] + 1
                if v < dp[i]:
                    dp[i] = v
        return dp[n] if dp[n] < INF else -1
```

```python
from typing import List

class Solution:
    def minValidStrings(self, words: List[str], target: str) -> int:
        n = len(target)
        # d[j] = length of the longest prefix of any word that is a suffix of target[:j+1]
        d = [0] * n

        # Remove words that are prefixes of other words (no new valid prefixes).
        ws = sorted(set(words))
        ws = [w for i, w in enumerate(ws)
              if i + 1 == len(ws) or not ws[i + 1].startswith(w)]

        for w in ws:
            p = w + '#'  # sentinel char never appears in target
            lp = len(p)
            fl = [0] * lp
            k = 0
            for i in range(1, lp):
                c = p[i]
                while k and p[k] != c:
                    k = fl[k - 1]
                if p[k] == c:
                    k += 1
                fl[i] = k

            k = 0
            for j, c in enumerate(target):
                while k and p[k] != c:
                    k = fl[k - 1]
                if p[k] == c:
                    k += 1
                if k > d[j]:
                    d[j] = k

        # dp[i] = min number of valid pieces covering target[:i].
        # dp is non-decreasing in i, so the best last piece ending at i-1
        # is the longest valid one, of length d[i-1].
        INF = float('inf')
        dp = [INF] * (n + 1)
        dp[0] = 0
        for i in range(1, n + 1):
            L = d[i - 1]
            if L:
                v = dp[i - L] + 1
                if v < dp[i]:
                    dp[i] = v
        return dp[n] if dp[n] < INF else -1
```

=== ANSWER ===
Looking at this problem:

- A piece is valid iff it's a prefix of some word. For each end position `j`, let `d[j]` = length of the longest word-prefix that is a suffix of `target[:j+1]` (computable per word via the KMP automaton, taking max across words — using a sentinel char so the state never exceeds the word length).
- Let `dp[i]` = min pieces covering `target[:i]`. Since a truncated valid piece is still valid, `dp` is **non-decreasing**. Therefore `dp[i] = dp[i - d[i-1]] + 1` — the optimal last piece ending at `i-1` is always the *longest* valid one (its start index is smallest, hence has minimal `dp`). This avoids needing per-start max-match lengths entirely.

Complexity: O(sum|w| + K·n) ≈ 6·10⁵ steps. Verified against all three examples (outputs 3, 2, -1) and edge cases (unreachable positions give `d[j]=0` → -1).
