def batch(tables, A, B):
    L = B - A + 1
    _, ex = np.frexp(L.astype(np.float64))
    kk = ex.astype(np.intp) - 1
    res = np.empty(len(A), dtype=np.int64)
    kmax = int(kk.max())
    for k in range(kmax + 1):
        sel = np.nonzero(kk == k)[0]
        if sel.size:
            shift = (1 << k) - 1
            res[sel] = np.maximum(tables[k][A[sel]], tables[k][B[sel] - shift])
    return res

```python
class Solution:
    def maxSubarraySum(self, nums: List[int]) -> int:
        import numpy as np
        from collections import defaultdict
        
        n = len(nums)
        
        # prefix sums
        S = [0] * (n + 1)
        s = 0
        for i in range(n):
            s += nums[i]
            S[i + 1] = s
        
        # Kadane (non-empty), no deletion
        K = cur = nums[0]
        for i in range(1, n):
            v = nums[i]
            cur = cur + v if cur > 0 else v
            if cur > K:
                K = cur
        
        # group occurrence positions by value
        pos = defaultdict(list)
        for i, v in enumerate(nums):
            pos[v].append(i)
        
        # build non-empty blocks (gaps between consecutive occurrences), grouped by value
        Ls = []
        Rs = []
        gstarts = []  # (start, end) in Ls/Rs for groups with >= 2 blocks
        for ps in pos.values():
            start = len(Ls)
            prev = -1
            for p in ps:
                if p > prev + 1:
                    Ls.append(prev + 1)
                    Rs.append(p - 1)
                prev = p
            if prev < n - 1:
                Ls.append(prev + 1)
                Rs.append(n - 1)
            end = len(Ls)
            if end - start >= 2:
                gstarts.append((start, end))
        
        if not gstarts:
            return K
        
        m = len(Ls)
        
        # sparse tables over S[0..n] for range max / min
        Sn = np.array(S, dtype=np.int64)
        M = n + 1
        LOG = M.bit_length()
        tmax = [Sn]
        tmin = [Sn]
        for k in range(1, LOG):
            half = 1 << (k - 1)
            p = tmax[-1]
            tmax.append(np.maximum(p[:len(p) - half], p[half:]))
            p = tmin[-1]
            tmin.append(np.minimum(p[:len(p) - half], p[half:]))
        
        LA = np.array(Ls, dtype=np.int64)
        RA = np.array(Rs, dtype=np.int64)
        
        def batch(tables, A, B):
            # max/min over Sn[A..B] inclusive (tables built with np.max or np.min accordingly)
            L = B - A + 1
            kk = np.frexp(L.astype(np.float64))[1].astype(np.int64) - 1
            res = np.empty(A.shape[0], dtype=np.int64)
            for k in range(int(kk.max()) + 1):
                sel = np.nonzero(kk == k)[0]
                if sel.size:
                    shift = (1 << k) - 1
                    res[sel] = np.maximum(tables[k][A[sel]], tables[k][B[sel] - shift]) if tables is tmax else np.minimum(tables[k][A[sel]], tables[k][B[sel] - shift])
            return res
        
        # per block: pre = max non-empty prefix sum, suf = max non-empty suffix sum, tot = total
        qmax = batch(tmax, LA + 1, RA + 1)          # max S[l+1 .. r+1]
        qmin = batch(tmin, LA, RA)                  # min S[l .. r]
        Sl = Sn[LA]
        Sr1 = Sn[RA + 1]
        pre = (qmax - Sl).tolist()
        suf = (Sr1 - qmin).tolist()
        tot = (Sr1 - Sl).tolist()
        
        # per group scan: cross-block max subarray
        NEG = -(1 << 62)
        best = NEG
        for gs, ge in gstarts:
            cum = tot[gs]
            ba = suf[gs] - cum
            for idx in range(gs + 1, ge):
                c = ba + cum + pre[idx]
                if c > best:
                    best = c
                cum += tot[idx]
                a = suf[idx] - cum
                if a > ba:
                    ba = a
        
        return K if K > best else best
```
