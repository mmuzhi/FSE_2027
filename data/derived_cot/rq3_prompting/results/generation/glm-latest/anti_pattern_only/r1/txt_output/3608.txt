from math import gcd
from typing import List

class Solution:
    def subsequencePairCount(self, nums: List[int]) -> int:
        MOD = 10**9 + 7
        n = len(nums)
        V = max(nums)
        
        # cnt[m] = number of elements divisible by m
        cnt = [0] * (V + 1)
        for m in range(1, V + 1):
            c = 0
            for x in nums:
                if x % m == 0:
                    c += 1
            cnt[m] = c
        # This is O(V·n) = 200·200 = 40k, fine.
        
        # powers
        pow2 = [1] * (n + 1)
        pow3 = [1] * (n + 1)
        for i in range(1, n + 1):
            pow2[i] = pow2[i-1] * 2 % MOD
            pow3[i] = pow3[i-1] * 3 % MOD
        
        # Möbius function up to V
        mu = [0] * (V + 1)
        mu[1] = 1
        # sieve
        is_comp = [False] * (V + 1)
        primes = []
        mu = [0]*(V+1)
        mu[1] = 1
        for i in range(2, V + 1):
            if not is_comp[i]:
                primes.append(i)
                mu[i] = -1
            for p in primes:
                if i * p > V:
                    break
                is_comp[i*p] = True
                if i % p == 0:
                    mu[i*p] = 0
                    break
                else:
                    mu[i*p] = -mu[i]
        
        # K(u,v) table
        # K[u][v]
        K = [[0] * (V + 1) for _ in range(V + 1)]
        for u in range(1, V + 1):
            cu = cnt[u]
            for v in range(1, V + 1):
                cv = cnt[v]
                l = u // gcd(u, v) * v  # lcm
                if l <= V:
                    a = cnt[l]
                else:
                    a = 0
                b = cu - a
                c = cv - a
                K[u][v] = (pow3[a] * pow2[b + c] - pow2[a + c] - pow2[a + b] + 1) % MOD
        
        # Answer = sum_g sum_{u=g*j<=V} sum_{v=g*k<=V} mu[j]*mu[k]*K[u][v]
        ans = 0
        for g in range(1, V + 1):
            s = 0
            j = 1
            while g * j <= V:
                u = g * j
                if mu[j] != 0:
                    k = 1
                    while g * k <= V:
                        if mu[k] != 0:
                            s += mu[j] * mu[k] * K[u][g*k]
                        k += 1
                j += 1
            ans = (ans + s) % MOD
        return ans % MOD

from math import gcd
from typing import List

class Solution:
    def subsequencePairCount(self, nums: List[int]) -> int:
        MOD = 10**9 + 7
        n = len(nums)
        V = max(nums)
        
        # cnt[m] = number of elements divisible by m
        cnt = [0] * (V + 1)
        for m in range(1, V + 1):
            cnt[m] = sum(1 for x in nums if x % m == 0)
        
        # powers of 2 and 3
        pow2 = [1] * (n + 1)
        pow3 = [1] * (n + 1)
        for i in range(1, n + 1):
            pow2[i] = pow2[i - 1] * 2 % MOD
            pow3[i] = pow3[i - 1] * 3 % MOD
        
        # Möbius function via linear sieve
        mu = [0] * (V + 1)
        mu[1] = 1
        if V >= 2:
            is_comp = [False] * (V + 1)
            primes = []
            for i in range(2, V + 1):
                if not is_comp[i]:
                    primes.append(i)
                    mu[i] = -1
                for p in primes:
                    ip = i * p
                    if ip > V:
                        break
                    is_comp[ip] = True
                    if i % p == 0:
                        mu[ip] = 0
                        break
                    else:
                        mu[ip] = -mu[i]
        
        # K[u][v] = number of ordered pairs (A, B) of disjoint non-empty index sets
        # with A ⊆ M_u and B ⊆ M_v, mod MOD
        K = [[0] * (V + 1) for _ in range(V + 1)]
        for u in range(1, V + 1):
            cu = cnt[u]
            for v in range(1, V + 1):
                cv = cnt[v]
                g = gcd(u, v)
                l = u // g * v
                a = cnt[l] if l <= V else 0
                b = cu - a
                c = cv - a
                K[u][v] = (pow3[a] * pow2[b + c] - pow2[a + c] - pow2[a + b] + 1) % MOD
        
        # answer = sum over g of E(g,g), with
        # E(g,g) = sum_{u=g*j<=V} sum_{v=g*k<=V} mu[j]*mu[k]*K[u][v]
        ans = 0
        for g in range(1, V + 1):
            s = 0
            j = 1
            while True:
                u = g * j
                if u > V:
                    break
                mj = mu[j]
                if mj:
                    Ku = K[u]
                    k = 1
                    while True:
                        v = g * k
                        if v > V:
                            break
                        mk = mu[k]
                        if mk:
                            s += mj * mk * Ku[v]
                        k += 1
                j += 1
            ans = (ans + s) % MOD
        return ans % MOD

from math import gcd
from typing import List

class Solution:
    def subsequencePairCount(self, nums: List[int]) -> int:
        MOD = 10 ** 9 + 7
        n = len(nums)
        V = max(nums)

        # cnt[m] = number of elements of nums divisible by m
        cnt = [0] * (V + 1)
        for m in range(1, V + 1):
            c = 0
            for x in nums:
                if x % m == 0:
                    c += 1
            cnt[m] = c

        # powers of 2 and 3 modulo MOD
        pow2 = [1] * (n + 1)
        pow3 = [1] * (n + 1)
        for i in range(1, n + 1):
            pow2[i] = pow2[i - 1] * 2 % MOD
            pow3[i] = pow3[i - 1] * 3 % MOD

        # Möbius function up to V
        mu = [0] * (V + 1)
        mu[1] = 1
        for i in range(1, V + 1):
            mi = mu[i]
            for j in range(2 * i, V + 1, i):
                mu[j] -= mi

        # K[u][v] = number of ordered pairs (A, B) of disjoint, non-empty index sets
        # with A ⊆ M_u, B ⊆ M_v (M_u = indices of elements divisible by u), mod MOD.
        K = [[0] * (V + 1) for _ in range(V + 1)]
        for u in range(1, V + 1):
            cu = cnt[u]
            Ku = K[u]
            for v in range(1, V + 1):
                cv = cnt[v]
                g = gcd(u, v)
                l = (u // g) * v
                a = cnt[l] if l <= V else 0
                b = cu - a
                c = cv - a
                Ku[v] = (pow3[a] * pow2[b + c] - pow2[a + c] - pow2[a + b] + 1) % MOD

        # answer = Σ_g E(g,g) where E(g,g) = Σ_{u=g·j ≤ V} Σ_{v=g·k ≤ V} μ(j)μ(k) K[u][v]
        ans = 0
        for g in range(1, V + 1):
            s = 0
            j = 1
            while g * j <= V:
                mj = mu[j]
                if mj:
                    Ku = K[g * j]
                    k = 1
                    while g * k <= V:
                        mk = mu[k]
                        if mk:
                            s += mj * mk * Ku[g * k]
                        k += 1
                j += 1
            ans = (ans + s) % MOD
        return ans
```

from math import gcd
from typing import List

class Solution:
    def subsequencePairCount(self, nums: List[int]) -> int:
        MOD = 10 ** 9 + 7
        n = len(nums)
        V = max(nums)

        # cnt[m] = number of elements divisible by m
        cnt = [0] * (V + 1)
        for m in range(1, V + 1):
            c = 0
            for x in nums:
                if x % m == 0:
                    c += 1
            cnt[m] = c

        pow2 = [1] * (n + 1)
        pow3 = [1] * (n + 1)
        for i in range(1, n + 1):
            pow2[i] = pow2[i - 1] * 2 % MOD
            pow3[i] = pow3[i - 1] * 3 % MOD

        # Möbius function
        mu = [0] * (V + 1)
        mu[1] = 1
        for i in range(1, V + 1):
            mi = mu[i]
            if mi:
                for j in range(2 * i, V + 1, i):
                    mu[j] -= mi

        # K[u][v]: number of ordered pairs (A,B) of disjoint non-empty index sets
        # with A ⊆ M_u and B ⊆ M_v, modulo MOD
        K = [[0] * (V + 1) for _ in range(V + 1)]
        for u in range(1, V + 1):
            cu = cnt[u]
            Ku = K[u]
            for v in range(1, V + 1):
                g = gcd(u, v)
                l = (u // g) * v
                a = cnt[l] if l <= V else 0
                b = cu - a

from math import gcd
from typing import List


class Solution:
    def subsequencePairCount(self, nums: List[int]) -> int:
        MOD = 10 ** 9 + 7
        n = len(nums)
        V = max(nums)

        # cnt[m] = number of elements of nums divisible by m
        cnt = [0] * (V + 1)
        for m in range(1, V + 1):
            c = 0
            for x in nums:
                if x % m == 0:
                    c += 1
            cnt[m] = c

        # powers of 2 and 3 modulo MOD (up to n)
        pow2 = [1] * (n + 1)
        pow3 = [1] * (n + 1)
        for i in range(1, n + 1):
            pow2[i] = pow2[i - 1] * 2 % MOD
            pow3[i] = pow3[i - 1] * 3 % MOD

        # Möbius function up to V
        mu = [0] * (V + 1)
        mu[1] = 1
        for i in range(1, V + 1):
            mi = mu[i]
            if mi:
                for j in range(i + i, V + 1, i):
                    mu[j] -= mi

        # K[u][v] = number of ordered pairs (A, B) of disjoint non-empty index
        # sets with A ⊆ M_u and B ⊆ M_v, taken modulo MOD.
        K = [[0] * (V + 1) for _ in range(V + 1)]
        for u in range(1, V + 1):
            cu = cnt[u]
            Ku = K[u]
            for v in range(1, V + 1):
                g = gcd(u, v)
                l = u // g * v
                a = cnt[l] if l <= V else 0
                b = cu - a
                c = cnt[v] - a
                Ku[v] = (pow3[a] * pow2[b + c] - pow2[a + c] - pow2[a + b] + 1) % MOD

        # answer = Σ_g E(g,g) with
        # E(g,g) = Σ_{u=g·j≤V} Σ_{v=g·k≤V} μ(j)·μ(k)·K[u][v]
        ans = 0
        for g in range(1, V + 1):
            lim = V // g
            s = 0
            for j in range(1, lim + 1):
                mj = mu[j]
                if mj:
                    Ku = K[g * j]
                    for k in range(1, lim + 1):
                        mk = mu[k]
                        if mk:
                            s += mj * mk * Ku[g * k]
            ans = (ans + s) % MOD
        return ans
```
