- A_i\neq A_j\ (1\leq i<j\leq N)

def main():

    n = int(data[0])

            v = s

                cycle_id[v] = cid
                pos_in_cycle[v] = len(cyc)

        cyc = cycles[cid]

import sys
from math import gcd

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    P = [int(x) - 1 for x in data[1:1 + n]]
    A = list(map(int, data[1 + n:1 + 2 * n]))

    # Decompose P into cycles.
    # Each cycle list starts at its minimum element; cycles appear in
    # increasing order of their minimum elements.
    visited = bytearray(n)
    cycles = []
    for s in range(n):
        if not visited[s]:
            cyc = []
            v = s
            while not visited[v]:
                visited[v] = 1
                cyc.append(v)
                v = P[v]
            cycles.append(cyc)

    # Greedy: maintain k ≡ r (mod g).
    r, g = 0, 1
    getA = A.__getitem__

    for cyc in cycles:
        L = len(cyc)
        if L == 1:
            continue
        G = gcd(g, L)
        if G == L:
            continue  # rotation already forced
        base = r % G
        sub = cyc[base::G]
        best_pos = min(sub, key=getA)
        best_d = base + G * sub.index(best_pos)
        # combine k ≡ r (mod g) with k ≡ best_d (mod L)
        Lp = L // G
        gp = g // G
        inv = pow(gp % Lp, -1, Lp)
        t = ((best_d - r) // G % Lp) * inv % Lp
        r += g * t
        g *= Lp

    # Build answer for k = r.
    ans = A[:]
    for cyc in cycles:
        L = len(cyc)
        rot = r % L
        if rot:
            S = cyc[rot:] + cyc[:rot]
            for dst, src in zip(cyc, S):
                ans[dst] = A[src]

    sys.stdout.write(' '.join(map(str, ans)) + '\n')

main()

import sys
from math import gcd

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    P = [int(x) - 1 for x in data[1:1 + n]]
    A = list(map(int, data[1 + n:1 + 2 * n]))

    # Decompose P into cycles.
    # Each cycle list starts at its minimum element; cycles are stored
    # in increasing order of their minimum elements.
    visited = bytearray(n)
    cycles = []
    for s in range(n):
        if not visited[s]:
            cyc = []
            v = s
            while not visited[v]:
                visited[v] = 1
                cyc.append(v)
                v = P[v]
            cycles.append(cyc)

    # Greedy: maintain the set of optimal k as a single residue class
    # k ≡ r (mod g).  Process cycles in order of their minimum element
    # (i.e., in the order positions 1..N first hit them).
    r, g = 0, 1
    getA = A.__getitem__

    for cyc in cycles:
        L = len(cyc)
        if L == 1:
            continue
        G = gcd(g % L, L)
        if G == L:
            # k mod L is already forced to r mod L
            continue
        base = r % G
        sub = cyc[base::G]  # positions cyc[d] for d ≡ base (mod G), d < L
        best_pos = min(sub, key=getA)
        best_d = base + G * sub.index(best_pos)
        # merge k ≡ r (mod g) with k ≡ best_d (mod L)
        Lp = L // G
        gp = g // G
        inv = pow(gp % Lp, -1, Lp)
        t = ((best_d - r) // G % Lp) * inv % Lp
        r += g * t
        g *= Lp

    # Build the answer for k = r:  ans[cyc[j]] = A[cyc[(j + r) mod L]]
    ans = A[:]
    rot_cache = {}
    for cyc in cycles:
        L = len(cyc)
        rot = rot_cache.get(L)
        if rot is None:
            rot = r % L
            rot_cache[L] = rot
        if rot:
            S = cyc[rot:] + cyc[:rot]
            for dst, src in zip(cyc, S):
                ans[dst] = A[src]

    sys.stdout.write(' '.join(map(str, ans)) + '\n')
