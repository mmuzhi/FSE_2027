import sys

def main():
    data = sys.stdin.read().split()
    idx = 0
    q = int(data[idx]); idx += 1
    stack = [0] * 100  # bottom ... top (top is the end of the list)
    out = []
    for _ in range(q):
        c = data[idx]; idx += 1
        if c == '1':
            x = int(data[idx]); idx += 1
            stack.append(x)
        else:
            out.append(str(stack.pop()))
    sys.stdout.write('\n'.join(out) + '\n')

main()
