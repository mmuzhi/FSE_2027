class Solution:
    def validSequence(self, word1: str, word2: str) -> List[int]:
        n, m = len(word1), len(word2)
        # suff[j]: max start index in word1 to exactly match word2[j:]; -1 if impossible; suff[m] = n
        suff = [-1] * (m + 1)
        suff[m] = n
        cur = n
        for j in range(m - 1, -1, -1):
            p = word1.rfind(word2[j], 0, cur)
            if p == -1:
                break  # all suff[j'], j' <= j remain -1
            suff[j] = p
            cur = p
        
        ans = []
        pos = 0
        used = False
        for j in range(m):
            if pos >= n:
                return []
            c = word2[j]
            if not used:
                if word1[pos] == c:
                    ans.append(pos)
                    pos += 1
                elif pos + 1 <= suff[j + 1]:
                    ans.append(pos)
                    pos += 1
                    used = True
                else:
                    g = word1.find(c, pos)
                    if g == -1:
                        return []
                    ans.append(g)
                    pos = g + 1
            else:
                g = word1.find(c, pos)
                if g == -1 or g + 1 > suff[j + 1]:
                    # invariant says this shouldn't happen if feasible; but guard anyway
                    return []
                ans.append(g)
                pos = g + 1
        return ans
