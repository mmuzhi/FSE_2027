Hmm, let me recall the exact Al.Cash scheme for "range add, range max":

```
void apply(int x, int v) {
    tree[x] += v;
    if (x < size) lazy[x] += v;
}
void pull(int x) {  # x is a leaf index (>= size); update ancestors
    while (x > 1) {
        x >>= 1;
        tree[x] = max(tree[2x], tree[2x+1]) + lazy[x];
    }
}
void add(int l, int r, int v) {  # [l, r)
    l += size; r += size;
    ll_, rr_ = l, r-1
    while (l < r) {
        if (l & 1): apply(l, v); l += 1
        if (r & 1): r -= 1; apply(r, v)
        l >>= 1; r >>= 1
    }
    pull(ll_); pull(rr_)
}
```
