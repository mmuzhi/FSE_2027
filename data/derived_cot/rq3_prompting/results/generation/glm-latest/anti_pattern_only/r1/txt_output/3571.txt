```python
class Solution:
    def maxPathLength(self, coordinates: List[List[int]], k: int) -> int:
        xk, yk = coordinates[k]
        
        lower = [(x, y) for x, y in coordinates if x < xk and y < yk]
        upper = [(x, y) for x, y in coordinates if x > xk and y > yk]
        
        def lis(points):
            # longest strictly increasing chain (both coords)
            points.sort(key=lambda p: (p[0], -p[1]))
            tails = []
            for _, y in points:
                idx = bisect_left(tails, y)
                if idx == len(tails):
                    tails.append(y)
                else:
                    tails[idx] = y
            return len(tails)
        
        return lis(lower) + 1 + lis(upper)
```

```python
from bisect import bisect_left
from typing import List

class Solution:
    def maxPathLength(self, coordinates: List[List[int]], k: int) -> int:
        xk, yk = coordinates[k]
        
        # Points that can appear before coordinates[k] in the path
        lower = [(x, y) for x, y in coordinates if x < xk and y < yk]
        # Points that can appear after coordinates[k] in the path
        upper = [(x, y) for x, y in coordinates if x > xk and y > yk]
        
        def lis(points):
            # Longest chain with both coordinates strictly increasing.
            # Sort by x ascending; for equal x, y descending so that
            # points with the same x can't both appear in a strictly
            # increasing y-subsequence. Then it's a strict LIS on y.
            points.sort(key=lambda p: (p[0], -p[1]))
            tails = []
            for _, y in points:
                idx = bisect_left(tails, y)
                if idx == len(tails):
                    tails.append(y)
                else:
                    tails[idx] = y
            return len(tails)
        
        return lis(lower) + 1 + lis(upper)
```
