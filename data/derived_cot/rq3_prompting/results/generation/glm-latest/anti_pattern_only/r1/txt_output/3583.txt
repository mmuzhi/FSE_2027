```python
class Solution:
    def gcdValues(self, nums: List[int], queries: List[int]) -> List[int]:
        from bisect import bisect_right
        from itertools import accumulate

        maxV = max(nums)
        cnt = [0] * (maxV + 1)
        for x in nums:
            cnt[x] += 1

        # exact[g] = number of pairs whose gcd is exactly g
        exact = [0] * (maxV + 1)
        for g in range(maxV, 0, -1):
            c = sum(cnt[g::g])  # elements divisible by g
            if c > 1:
                # pairs whose gcd is a multiple of g, minus those with larger gcd
                e = c * (c - 1) // 2 - sum(exact[2 * g::g])
                exact[g] = e

        # prefix[g] = number of pairs with gcd value <= g
        prefix = list(accumulate(exact))

        # for query q (0-based), find smallest g with prefix[g] > q
        return [bisect_right(prefix, q) for q in queries]
```
