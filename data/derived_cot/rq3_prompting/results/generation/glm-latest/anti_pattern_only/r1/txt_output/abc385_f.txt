import sys

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    if n == 1:
        sys.stdout.write("-1\n")
        return
    # parse
    vals = list(map(int, data[1:1+2*n]))
    xs = vals[0::2]
    hs = vals[1::2]
    bn = hs[0]*xs[1] - hs[1]*xs[0]
    bd = xs[1] - xs[0]
    for i in range(1, n-1):
        num = hs[i]*xs[i+1] - hs[i+1]*xs[i]
        den = xs[i+1] - xs[i]
        if num * bd > bn * den:
            bn = num
            bd = den
    if bn < 0:
        sys.stdout.write("-1\n")
        return
    # exact decimal with 18 fractional digits, rounded
    ip = bn // bd
    rem = bn - ip*bd
    d = rem * 10**18
    q = d // bd
    r = d - q*bd
    if 2*r >= bd:
        q += 1
    if q >= 10**18:
        q -= 10**18
        ip += 1
    sys.stdout.write(f"{ip}.{q:018d}\n")

main()

    if 2 * r >= bd:
        q += 1
        if q == 10 ** 18:
            q = 0
            ip += 1
    print("%d.%018d" % (ip, q))
