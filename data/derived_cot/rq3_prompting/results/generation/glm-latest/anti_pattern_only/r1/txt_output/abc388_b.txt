=== ANSWER ===
```python
import sys

def main():
    data = sys.stdin.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    d = int(data[idx]); idx += 1
    snakes = []
    for _ in range(n):
        t = int(data[idx]); idx += 1
        l = int(data[idx]); idx += 1
        snakes.append((t, l))
    out = []
    for k in range(1, d + 1):
        best = max(t * (l + k) for t, l in snakes)
        out.append(str(best))
    sys.stdout.write('\n'.join(out) + '\n')

main()
```
