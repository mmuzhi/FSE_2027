```python
class Solution:
    def hasIncreasingSubarrays(self, nums: List[int], k: int) -> bool:
        n = len(nums)
        inc = [1] * n
        for i in range(n - 2, -1, -1):
            if nums[i] < nums[i + 1]:
                inc[i] = inc[i + 1] + 1
        for a in range(n - 2 * k + 1):
            if inc[a] >= k and inc[a + k] >= k:
                return True
        return False
```

```python
class Solution:
    def hasIncreasingSubarrays(self, nums: List[int], k: int) -> bool:
        n = len(nums)
        # inc[i] = length of longest strictly increasing run starting at i
        inc = [1] * n
        for i in range(n - 2, -1, -1):
            if nums[i] < nums[i + 1]:
                inc[i] = inc[i + 1] + 1
        
        # Try every start a of the first subarray; second starts at a + k
        for a in range(n - 2 * k + 1):
            if inc[a] >= k and inc[a + k] >= k:
                return True
        return False
```
