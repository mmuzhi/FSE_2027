class Solution:
    def longestSpecialPath(self, edges: List[List[int]], nums: List[int]) -> List[int]:
        n = len(nums)
        adj = [[] for _ in range(n)]
        for u, v, w in edges:
            adj[u].append((v, w))
            adj[v].append((u, w))
        
        max_val = max(nums) + 1 if nums else 1
        last_occ = [-1] * (max_val)
        
        path_dist = [0]  # path_dist[i] = dist from root to node at depth i
        # path_nodes not really needed except for... we don't need node ids on path actually. We need depth only, which is len(path_dist)-1.
        
        best_len = -1
        best_cnt = 0
        
        start = 0
        
        # stack entries: (node, parent, w, is_enter, saved_start, saved_last)
        stack = [(0, -1, 0, True, 0, -1)]
        while stack:
            node, parent, w, is_enter, sv_start, sv_last = stack.pop()
            if not is_enter:
                # exit: restore
                last_occ[nums[node]] = sv_last
                start = sv_start
                path_dist.pop()
                continue
            d = len(path_dist)  # depth of node; also index in path
            # enter
            val = nums[node]
            p_last = last_occ[val]
            old_start = start
            new_start = old_start if old_start > p_last + 1 else p_last + 1
            # hmm, p_last+1 where p_last=-1 gives 0.
            start = new_start
            last_occ[val] = d
            path_dist.append((path_dist[d-1] + w) if d > 0 else 0)
            
            cur_len = path_dist[d] - path_dist[new_start]
            cur_cnt = d - new_start + 1
            if cur_len > best_len:
                best_len = cur_len
                best_cnt = cur_cnt
            elif cur_len == best_len and cur_cnt < best_cnt:
                best_cnt = cur_cnt
            
            # push exit
            stack.append((node, parent, w, False, old_start, p_last))
            for v2, w2 in adj[node]:
                if v2 != parent:
                    stack.append((v2, node, w2, True, 0, -1))
        
        return [best_len, best_cnt]

from typing import List

class Solution:
    def longestSpecialPath(self, edges: List[List[int]], nums: List[int]) -> List[int]:
        n = len(nums)
        adj = [[] for _ in range(n)]
        for u, v, w in edges:
            adj[u].append((v, w))
            adj[v].append((u, w))
        
        last_occ = [-1] * (max(nums) + 1)
        
        path_dist = []   # path_dist[i] = distance from root to node at depth i
        start = 0        # window left boundary (a depth)
        best_len, best_cnt = -1, -1
        
        # stack entries: (True, node, parent, w) for enter,
        #                 (False, node, saved_start, saved_last_occ) for exit
        stack = [(True, 0, -1, 0)]
        while stack:
            item = stack.pop()
            if item[0]:
                _, node, parent, w = item
                d = len(path_dist)

                last_occ[val] = d
                if d:
                    path_dist.append(path_dist[-1] + w)
                else:
                    path_dist.append(0)
                
                cur_len = path_dist[d] - path_dist[start]
                cur_cnt = d - start + 1
                if cur_len > best_len or (cur_len == best_len and cur_cnt < best_cnt):
                    best_len, best_cnt = cur_len, cur_cnt
                
                stack.append((False, node, old_start, prev))
                for nxt, nw in adj[node]:

            else:
                _, node, old_start, prev = item

                start = old_start

```python
from typing import List

class Solution:
    def longestSpecialPath(self, edges: List[List[int]], nums: List[int]) -> List[int]:
        n = len(nums)
        adj = [[] for _ in range(n)]
        for u, v, w in edges:
            adj[u].append((v, w))
            adj[v].append((u, w))
        
        # last occurrence (depth index) of each value on the current root path
        last_occ = [-1] * (max(nums) + 1)
        
        path_dist = []   # path_dist[i] = distance from root to the node at depth i
        start = 0        # left boundary (depth) of the unique-value window
        best_len, best_cnt = -1, -1
        
        # stack entries: (True, node, parent, weight) for "enter",
        #                 (False, node, saved_start, saved_last_occ) for "exit"
        stack = [(True, 0, -1, 0)]
        while stack:
            item = stack.pop()
            if item[0]:
                _, node, parent, w = item
                d = len(path_dist)              # depth of current node
                val = nums[node]
                prev = last_occ[val]            # previous occurrence of this value
                old_start = start
                if prev >= start:               # duplicate inside current window
                    start = prev + 1
                last_occ[val] = d
                if d:
                    path_dist.append(path_dist[-1] + w)
                else:
                    path_dist.append(0)
                
                # maximal all-unique window ending at this node is [start, d]
                cur_len = path_dist[d] - path_dist[start]
                cur_cnt = d - start + 1
                if cur_len > best_len or (cur_len == best_len and cur_cnt < best_cnt):
                    best_len, best_cnt = cur_len, cur_cnt
                
                # schedule state restoration, then descend
                stack.append((False, node, old_start, prev))
                for nxt, nw in adj[node]:
                    if nxt != parent:
                        stack.append((True, nxt, node, nw))
            else:
                _, node, old_start, prev = item
                last_occ[nums[node]] = prev     # restore previous occurrence
                start = old_start               # restore window start
                path_dist.pop()
        
        return [best_len, best_cnt]
```
