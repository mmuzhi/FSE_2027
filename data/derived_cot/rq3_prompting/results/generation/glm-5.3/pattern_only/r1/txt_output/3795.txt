```python
class Solution:
    def minZeroArray(self, nums: List[int], queries: List[List[int]]) -> int:
        n = len(nums)
        if all(x == 0 for x in nums):
            return 0
        
        def feasible(k):
            for i in range(n):
                target = nums[i]
                if target == 0:
                    continue
                mask = 1
                limit = (1 << (target + 1)) - 1
                for l, r, val in queries[:k]:
                    if l <= i <= r:
                        mask |= (mask << val)
                        mask &= limit
                        if mask >> target & 1:
                            break
                if not (mask >> target & 1):
                    return False
            return True
        
        lo, hi = 0, len(queries)
        if not feasible(hi):
            return -1
        # binary search minimal k with feasible(k)
        while lo < hi:
            mid = (lo + hi) // 2
            if feasible(mid):
                hi = mid
            else:
                lo = mid + 1
        return lo
```

```python
class Solution:
    def minZeroArray(self, nums: List[int], queries: List[List[int]]) -> int:
        n = len(nums)
        if all(x == 0 for x in nums):
            return 0

        def feasible(k: int) -> bool:
            # For each index, check if some subset of the vals of the first k
            # queries covering it sums exactly to nums[i] (bitset subset-sum).
            for i in range(n):
                target = nums[i]
                if target == 0:
                    continue
                mask = 1                      # bit j set => sum j achievable
                limit = (1 << (target + 1)) - 1
                for l, r, val in queries[:k]:
                    if l <= i <= r:
                        mask |= mask << val   # add item with weight val
                        mask &= limit         # sums > target can never recover
                        if (mask >> target) & 1:
                            break
                if not ((mask >> target) & 1):
                    return False
            return True

        lo, hi = 0, len(queries)
        if not feasible(hi):
            return -1
        while lo < hi:
            mid = (lo + hi) // 2
            if feasible(mid):
                hi = mid
            else:
                lo = mid + 1
        return lo
```
