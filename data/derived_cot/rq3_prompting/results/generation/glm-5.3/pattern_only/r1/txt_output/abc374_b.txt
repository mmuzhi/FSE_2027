=== ANSWER ===
```python
S = input().strip()
T = input().strip()

n = min(len(S), len(T))
ans = 0
for i in range(n):
    if S[i] != T[i]:
        ans = i + 1
        break

if ans == 0:
    if len(S) != len(T):
        ans = n + 1
    else:
        ans = 0

print(ans)
```
