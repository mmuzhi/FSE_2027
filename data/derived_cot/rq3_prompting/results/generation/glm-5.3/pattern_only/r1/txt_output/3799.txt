from collections import Counter
from typing import List

class Solution:
    def totalNumbers(self, digits: List[int]) -> int:
        available = Counter(digits)
        count = 0
        for n in range(100, 999, 2):
            d = Counter([n // 100, (n // 10) % 10, n % 10])

                count += 1
        return count

- **Committed approach:** Iterate `n` from 100 to 998 step 2 (this guarantees 3 digits, even last digit, and no leading zero). For each `n`, check if its digit counts are covered by the `Counter` of the input array. Distinctness is automatic since each `n` is visited once.
