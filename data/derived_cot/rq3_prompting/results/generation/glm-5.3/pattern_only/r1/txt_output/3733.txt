import numpy as np

class Solution:
    def lenOfVDiagonal(self, grid: List[List[int]]) -> int:
        n, m = len(grid), len(grid[0])
        # padded grid, sentinel -1
        g = np.full((n + 2, m + 2), -1, dtype=np.int32)
        g[1:n+1, 1:m+1] = np.asarray(grid, dtype=np.int32)
        
        dirs = [(1, 1), (1, -1), (-1, 1), (-1, -1)]
        
        NEG = np.int32(-10**6)
        
        # S[d]: run arrays
        S = {}
        for (dr, dc) in dirs:
            Sd = np.zeros((n + 2, m + 2), dtype=np.int32)
            rng = range(n, 0, -1) if dr == 1 else range(1, n + 1)
            for r in rng:
                cur = g[r, 1:m+1]
                nxtg = g[r+dr, 1+dc:m+1+dc]
                nxtS = Sd[r+dr, 1+dc:m+1+dc]
                good = (cur == 0) | (cur == 2)
                cont = nxtg == (2 - cur)
                # careful: 2 - cur when cur == -1 → 3, no match; cur == 1 → 1, no match (nxtg never 1... could be 1! nxtg==1 vs 2-cur=1 → match, but good is False then)
                vals = np.where(good & cont, 1 + nxtS, np.where(good, 1, 0))
                Sd[r, 1:m+1] = vals
            S[(dr, dc)] = Sd
        
        best = 0
        mask1 = (g[1:n+1, 1:m+1] == 1)
        if mask1.any():
            best = 1
        
        for (dr, dc) in dirs:
            dr2, dc2 = dc, -dr  # clockwise
            S2 = S[(dr2, dc2)]
            P0 = np.zeros((n + 2, m + 2), dtype=np.int32)
            P1 = np.zeros((n + 2, m + 2), dtype=np.int32)
            rng = range(n, 0, -1) if dr == 1 else range(1, n + 1)
            for r in rng:
                gnext = g[r+dr, 1+dc:m+1+dc]
                gturn = g[r+dc2, 1+dr2:m+1+dr2]  # X + d2 = (r + dc, c - dr) → column shift -dr = dr2
                P1next = P1[r+dr, 1+dc:m+1+dc]
                P0next = P0[r+dr, 1+dc:m+1+dc]
                St = S2[r+dc2, 1+dr2:m+1+dr2]
                # p = 0 (req 2)
                c0 = np.maximum(np.where(gnext == 2, 1 + P1next, NEG),
                                np.where(gturn == 2, 1 + St, NEG))
                P0[r, 1:m+1] = np.maximum(c0, 0)
                # p = 1 (req 0)
                c1 = np.maximum(np.where(gnext == 0, 1 + P0next, NEG),
                                np.where(gturn == 0, 1 + St, NEG))
                P1[r, 1:m+1] = np.maximum(c1, 0)
            vals = np.where(mask1, P0[1:n+1, 1:m+1], -1)
            v = int(vals.max())
            if v >= 0 and 1 + v > best:
                best = 1 + v
        
        return best

```python
import numpy as np
from typing import List

class Solution:
    def lenOfVDiagonal(self, grid: List[List[int]]) -> int:
        n, m = len(grid), len(grid[0])
        g = np.full((n + 2, m + 2), -1, dtype=np.int32)
        g[1:n+1, 1:m+1] = np.asarray(grid, dtype=np.int32)

        dirs = [(1, 1), (1, -1), (-1, 1), (-1, -1)]

        # S[(dr,dc)][r][c]: length of maximal run starting at (r,c) (inclusive)
        # in direction (dr,dc) whose values alternate starting with grid value itself
        # (only meaningful when grid value in {0,2})
        S = {}
        for (dr, dc) in dirs:
            Sd = np.zeros((n + 2, m + 2), dtype=np.int32)
            rows = range(n, 0, -1) if dr == 1 else range(1, n + 1)
            for r in rows:
                cur = g[r, 1:m+1]
                ng = g[r+dr, 1+dc:m+1+dc]
                ns = Sd[r+dr, 1+dc:m+1+dc]
                good = (cur == 0) | (cur == 2)
                cont = ng == (2 - cur)
                Sd[r, 1:m+1] = np.where(good & cont, ns + 1, np.where(good, 1, 0))
            S[(dr, dc)] = Sd

        interior = (g[1:n+1, 1:m+1] == 1)
        best = 1 if interior.any() else 0
        NEG = -1

        for (dr, dc) in dirs:
            d2 = (dc, -dr)          # clockwise turn
            S2 = S[d2]
            P0 = np.zeros((n + 2, m + 2), dtype=np.int32)  # even position
            P1 = np.zeros((n + 2, m + 2), dtype=np.int32)  # odd position
            rows = range(n, 0, -1) if dr == 1 else range(1, n + 1)
            for r in rows:
                gnext = g[r+dr, 1+dc:m+1+dc]
                p1next = P1[r+dr, 1+dc:m+1+dc]
                p0next = P0[r+dr, 1+dc:m+1+dc]
                gturn = g[r+dc, 1-dr:m+1-dr]
                sturn = S2[r+dc, 1-dr:m+1-dr]
                # even position q: next needed value 2
                a = np.where(gnext == 2, p1next + 1, NEG)
                b = np.where(gturn == 2, sturn, NEG)
                P0[r, 1:m+1] = np.maximum(np.maximum(a, b), 0)
                # odd position q: next needed value 0
                a = np.where(gnext == 0, p0next + 1, NEG)
                b = np.where(gturn == 0, sturn, NEG)
                P1[r, 1:m+1] = np.maximum(np.maximum(a, b), 0)
            vals = np.where(interior, P0[1:n+1, 1:m+1], NEG)
            v = int(vals.max())
            if 1 + v > best:
                best = 1 + v
        return best
```

        dirs = [(1, 1), (1, -1), (-1, 1), (-1, -1)]

        # maximal chain starting at (r,c) going in (dr,dc) whose values alternate
        # beginning with the cell's own value (meaningful only when value is 0 or 2).
