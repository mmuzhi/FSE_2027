class Solution:
    def maxRemovals(self, source: str, pattern: str, targetIndices: List[int]) -> int:
        n, m = len(source), len(pattern)
        is_target = [False] * n
        for idx in targetIndices:
            is_target[idx] = True
        INF = float('inf')
        dp = [0] + [INF] * m
        for i in range(n):

            cost = 1 if is_target[i] else 0

import numpy as np

class Solution:
    def maxRemovals(self, source: str, pattern: str, targetIndices: List[int]) -> int:
        n, m = len(source), len(pattern)
        cost = np.zeros(n, dtype=np.int64)
        if targetIndices:
            cost[np.array(targetIndices)] = 1
        pat = np.frombuffer(pattern.encode(), dtype=np.uint8)
        src = np.frombuffer(source.encode(), dtype=np.uint8)
        INF = np.int64(1 << 30)
        dp = np.full(m + 1, INF, dtype=np.int64)
        dp[0] = 0
        for i in range(n):
            c = src[i]
            mask = pat == c  # length m, mask[j-1] corresponds to pattern position j
            # positions j in 1..m where pattern[j-1]==c
            cand = dp[:m] + cost[i]
            new_vals = np.where(mask, np.minimum(dp[1:], cand), dp[1:])
            dp[1:] = new_vals
        return len(targetIndices) - int(dp[m])
```

from typing import List

class Solution:
    def maxRemovals(self, source: str, pattern: str, targetIndices: List[int]) -> int:
        n, m = len(source), len(pattern)
        T = len(targetIndices)
        cost = [0] * n
        for idx in targetIndices:
            cost[idx] = 1

        try:
            import numpy as np
        except Exception:
            np = None

        if np is not None:
            pat = np.frombuffer(pattern.encode(), dtype=np.uint8)
            BIG = 1 << 40
            dp = np.full(m + 1, BIG, dtype=np.int64)
            dp[0] = 0
            tab = {}
            for ch in set(source):
                if ch in pattern:
                    mask = (pat == ord(ch))
                    tab[ch] = (~mask,)
            for i in range(n):
                ch = source[i]
                entry = tab.get(ch)
                if entry is None:
                    continue
                nmask = entry[0]
                cand = dp[:m] + cost[i]
                cand[nmask] = BIG
                np.minimum(dp[1:], cand, out=dp[1:])
            return T - int(dp[m])

        # pure python fallback
        INF = float('inf')
        ml = {}
        for j in range(m, 0, -1):
            ml.setdefault(pattern[j - 1], []).append(j)
        dp = [0] + [INF] * m
        for i in range(n):

            if not lst:
                continue

                for j in lst:

                    if v < dp[j]:
                        dp[j] = v
            else:
                for j in lst:
                    v = dp[j - 1]
                    if v < dp[j]:
                        dp[j] = v
        return T - dp[m]

from typing import List

try:
    import numpy as np
except Exception:
    np = None


class Solution:
    def maxRemovals(self, source: str, pattern: str, targetIndices: List[int]) -> int:
        n, m = len(source), len(pattern)
        T = len(targetIndices)
        cost = [0] * n
        for idx in targetIndices:
            cost[idx] = 1

        if np is not None:
            pat = np.frombuffer(pattern.encode(), dtype=np.uint8)
            BIG = 1 << 40
            dp = np.full(m + 1, BIG, dtype=np.int64)
            dp[0] = 0
            # per-character invalid-position masks for chars appearing in both
            tab = {}
            for ch in set(source):
                if ch in pattern:
                    tab[ch] = ~(pat == ord(ch))
            for i in range(n):
                nmask = tab.get(source[i])
                if nmask is None:
                    continue
                cand = dp[:m] + cost[i]
                cand[nmask] = BIG
                np.minimum(dp[1:], cand, out=dp[1:])
            return T - int(dp[m])

        # Pure Python fallback: O(n*m) 1D DP.
        INF = float('inf')
        ml = {}
        for j in range(m, 0, -1):
            ml.setdefault(pattern[j - 1], []).append(j)
        dp = [0] + [INF] * m
        for i in range(n):
            lst = ml.get(source[i])
            if not lst:
                continue
            if cost[i]:
                for j in lst:
                    v = dp[j - 1] + 1
                    if v < dp[j]:
                        dp[j] = v
            else:
                for j in lst:
                    v = dp[j - 1]
                    if v < dp[j]:
                        dp[j] = v
        return T - dp[m]
```

class Solution:
    def maxRemovals(self, source: str, pattern: str, targetIndices: List[int]) -> int:
        n, m = len(source), len(pattern)
        T = len(targetIndices)

        # cost[i] = 1 if index i is a removable (target) index
        cost = [0] * n
        for idx in targetIndices:
            cost[idx] = 1

        # dp[j] = minimum number of target indices used by an embedding of
        # pattern[:j] into the source prefix processed so far.
        # answer = T - dp[m]

        if np is not None:
            pat = np.frombuffer(pattern.encode(), dtype=np.uint8)
            BIG = 1 << 40
            dp = np.full(m + 1, BIG, dtype=np.int64)
            dp[0] = 0
            # invalid-position masks per character (where pattern char != c)
            tab = {}
            for ch in set(source):
                if ch in pattern:

            for i in range(n):
                nmask = tab.get(source[i])
                if nmask is None:
                    continue
                cand = dp[:m] + cost[i]          # dp_old[j-1] + cost
                cand[nmask] = BIG                # forbid non-matching j
                np.minimum(dp[1:], cand, out=dp[1:])
            return T - int(dp[m])

        # Pure Python fallback (same DP, in-place, j processed descending)
        INF = float('inf')
        ml = {}
        for j in range(m, 0, -1):
            ml.setdefault(pattern[j - 1], []).append(j)
        dp = [0] + [INF] * m
        for i in range(n):
            lst = ml.get(source[i])
            if not lst:
                continue
            if cost[i]:
                for j in lst:
                    v = dp[j - 1] + 1
                    if v < dp[j]:
                        dp[j] = v
            else:
                for j in lst:
                    v = dp[j - 1]
                    if v < dp[j]:
                        dp[j] = v
        return T - dp[m]

Looking at this problem:

**Key reduction:** Since removals don't shift indices, a set S ⊆ targetIndices is removable one-by-one iff the final string (source minus S) still contains `pattern` as a subsequence — because at every intermediate step the surviving embedding is intact. So the answer equals `|targetIndices| − (minimum number of target indices used by any embedding of pattern into source)`: given an optimal embedding E, all target indices not in E can be removed; conversely any removable set S is avoided by some embedding.

**DP:** `dp[j]` = min target-hit count embedding `pattern[:j]` into the source prefix processed so far. Transition per source char at i: `dp[j] = min(dp[j], dp[j-1] + cost[i])` when `source[i] == pattern[j-1]`. In-place 1D update with j descending (or vectorized with numpy using old values). O(n·m) ≤ 9·10⁶.

Verified on all four examples by hand trace (e.g. Ex1: min hits = 2 → 3−2 = 1 ✓; Ex4: min hits = 2 → 4−2 = 2 ✓; Ex3: only embedding uses all targets → 0 ✓). Critical implementation details: descending j in the pure-Python in-place update (prevents reusing one source char for two pattern positions), and computing `cand` from old `dp` before writing in the numpy path.
