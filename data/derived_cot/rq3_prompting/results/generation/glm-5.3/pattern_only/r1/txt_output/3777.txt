You are given an integer array nums and two integers, k and limit. Your task is to find a non-empty subsequence of nums that:

Has an alternating sum equal to k.
Maximizes the product of all its numbers without the product exceeding limit.

Return the product of the numbers in such a subsequence. If no subsequence satisfies the requirements, return -1.
The alternating sum of a 0-indexed array is defined as the sum of the elements at even indices minus the sum of the elements at odd indices.
 
Example 1:

Input: nums = [1,2,3], k = 2, limit = 10
Output: 6
Explanation:
The subsequences with an alternating sum of 2 are:

[1, 2, 3]

Alternating Sum: 1 - 2 + 3 = 2
Product: 1 * 2 * 3 = 6


[2]

Alternating Sum: 2
Product: 2



The maximum product within the limit is 6.

Example 2:

Input: nums = [0,2,3], k = -5, limit = 12
Output: -1
Explanation:
A subsequence with an alternating sum of exactly -5 does not exist.

Example 3:

Input: nums = [2,2,3,3], k = 0, limit = 9
Output: 9
Explanation:
The subsequences with an alternating sum of 0 are:

[2, 2]

Alternating Sum: 2 - 2 = 0
Product: 2 * 2 = 4


[3, 3]

Alternating Sum: 3 - 3 = 0
Product: 3 * 3 = 9


[2, 2, 3, 3]

Alternating Sum: 2 - 2 + 3 - 3 = 0
Product: 2 * 2 * 3 * 3 = 36



The subsequence [2, 2, 3, 3] has the greatest product with an alternating sum equal to k, but 36 > 9. The next greatest product is 9, which is within the limit.

```python
import numpy as np
from bisect import bisect_right
from typing import List

class Solution:
    def maxProduct(self, nums: List[int], k: int, limit: int) -> int:
        T = sum(nums)
        if k > T or k < -T:
            return -1
        
        # 12-smooth numbers up to limit (all achievable positive products)
        def is_smooth(x):
            for p in (2, 3, 5, 7, 11):
                while x % p == 0:
                    x //= p
            return x == 1
        
        PRODS = [x for x in range(1, limit + 1) if is_smooth(x)]
        idx = {v: i for i, v in enumerate(PRODS)}
        nE = len(PRODS)
        OVER, ZERO, START, DUMMY = nE, nE + 1, nE + 2, nE + 3
        Pdim = nE + 4
        
        S = 2 * T + 1
        OFF = T
        
        # per-value tables
        tables = {}
        for v in range(1, 13):
            INV = np.full(nE, DUMMY, dtype=np.intp)
            for j, val in enumerate(PRODS):
                if val % v == 0:
                    INV[j] = idx[val // v]
            so = bisect_right(PRODS, limit // v)
            tables[v] = (INV, so)
        
        A = np.zeros((S, Pdim), dtype=bool)  # even length so far
        B = np.zeros((S, Pdim), dtype=bool)  # odd length so far
        A[OFF, START] = True
        
        for v in nums:
            srcA = A.copy()
            srcA[:, START] = False
            srcB = B.copy()
            srcB[:, START] = False
            if v == 0:
                cB = np.zeros_like(B)
                cB[:, ZERO] = srcA.any(axis=1)
                cA = np.zeros_like(A)
                cA[:, ZERO] = srcB.any(axis=1)
            else:
                INV, so = tables[v]
                # take from even-length (sign +), becomes odd length
                mA = np.zeros_like(srcA)
                mA[:, :nE] = srcA[:, INV]
                mA[:, OVER] = srcA[:, OVER] | srcA[:, so:nE].any(axis=1)
                mA[:, ZERO] = srcA[:, ZERO]
                cB = np.zeros_like(B)
                cB[v:, :] = mA[:-v, :] if v < S else mA[:0, :]
                # take from odd-length (sign -), becomes even length
                mB = np.zeros_like(srcB)
                mB[:, :nE] = srcB[:, INV]
                mB[:, OVER] = srcB[:, OVER] | srcB[:, so:nE].any(axis=1)
                mB[:, ZERO] = srcB[:, ZERO]
                cA = np.zeros_like(A)
                cA[:S - v, :] = mB[v:, :] if v < S else mB[:0, :]
            A |= cA
            B |= cB
            # taking v as the single first element
            if v == 0:
                B[OFF, ZERO] = True
            elif v <= limit:
                B[OFF + v, idx[v]] = True
            else:
                B[OFF + v, OVER] = True
        
        row = OFF + k
        best = -1
        for arr in (A, B):
            r = arr[row]
            cols = np.nonzero(r[:nE])[0]
            if cols.size:
                best = max(best, int(PRODS[cols[-1]]))
            if r[ZERO]:
                best = max(best, 0)
        return best
```

```python
import numpy as np
from bisect import bisect_right
from typing import List

class Solution:
    def maxProduct(self, nums: List[int], k: int, limit: int) -> int:
        T = sum(nums)
        if k > T or -k > T:
            return -1

        def is_smooth(x: int) -> bool:
            for p in (2, 3, 5, 7, 11):
                while x % p == 0:
                    x //= p
            return x == 1

        PRODS = [x for x in range(1, limit + 1) if is_smooth(x)]
        idx = {v: i for i, v in enumerate(PRODS)}
        nE = len(PRODS)
        OVER, ZERO, START, DUMMY = nE, nE + 1, nE + 2, nE + 3
        Pdim = nE + 4

        S = 2 * T + 1
        OFF = T

        tables = {}
        for v in range(1, 13):
            INV = np.full(nE, DUMMY, dtype=np.intp)
            for j, val in enumerate(PRODS):
                if val % v == 0:
                    INV[j] = idx[val // v]
            so = bisect_right(PRODS, limit // v)
            tables[v] = (INV, so)

        A = np.zeros((S, Pdim), dtype=bool)
        B = np.zeros((S, Pdim), dtype=bool)
        A[OFF, START] = True

        for v in nums:
            srcA = A.copy()
            srcA[OFF, START] = False
            srcB = B.copy()
            if v == 0:
                cB = np.zeros_like(B)
                cB[:, ZERO] = srcA.any(axis=1)
                cA = np.zeros_like(A)
                cA[:, ZERO] = srcB.any(axis=1)
            else:
                INV, so = tables[v]
                mA = np.zeros_like(srcA)
                mA[:, :nE] = srcA[:, INV]
                mA[:, OVER] = srcA[:, OVER] | srcA[:, so:nE].any(axis=1)
                mA[:, ZERO] = srcA[:, ZERO]
                cB = np.zeros_like(B)
                cB[v:, :] = mA[:S - v, :]
                mB = np.zeros_like(srcB)
                mB[:, :nE] = srcB[:, INV]
                mB[:, OVER] = srcB[:, OVER] | srcB[:, so:nE].any(axis=1)
                mB[:, ZERO] = srcB[:, ZERO]
                cA = np.zeros_like(A)
                cA[:S - v, :] = mB[v:, :]
            A |= cA
            B |= cB
            if v == 0:
                B[OFF, ZERO] = True
            elif v <= limit:
                B[OFF + v, idx[v]] = True
            else:
                B[OFF + v, OVER] = True

        row = OFF + k
        best = -1
        for arr in (A, B):
            r = arr[row]
            cols = np.flatnonzero(r[:nE])
            if cols.size:
                best = max(best, int(PRODS[cols[-1]]))
            if r[ZERO]:
                best = max(best, 0)
        return best
```

```python
import numpy as np
from bisect import bisect_right
from typing import List


class Solution:
    def maxProduct(self, nums: List[int], k: int, limit: int) -> int:
        T = sum(nums)
        # alternating sum of any subsequence lies in [-T, T]
        if k > T or -k > T:
            return -1

        # all achievable positive products are 12-smooth numbers <= limit
        def is_smooth(x: int) -> bool:
            for p in (2, 3, 5, 7, 11):
                while x % p == 0:
                    x //= p
            return x == 1

        PRODS = [x for x in range(1, limit + 1) if is_smooth(x)]
        idx = {v: i for i, v in enumerate(PRODS)}
        nE = len(PRODS)
        OVER, ZERO, START, DUMMY = nE, nE + 1, nE + 2, nE + 3  # product > limit, product 0, empty, always-false
        Pdim = nE + 4

        S = 2 * T + 1          # sum offsets, sum s -> row s + OFF
        OFF = T

        # per-value tables: target exact column j' gathers source column INV[j']; so = first col with value > limit//v
        tables = {}
        for v in range(1, 13):
            INV = np.full(nE, DUMMY, dtype=np.intp)
            for j, val in enumerate(PRODS):
                if val % v == 0:
                    INV[j] = idx[val // v]
            so = bisect_right(PRODS, limit // v)
            tables[v] = (INV, so)

        # A: current subsequence length even (next sign +), B: odd (next sign -)
        A = np.zeros((S, Pdim), dtype=bool)
        B = np.zeros((S, Pdim), dtype=bool)
        A[OFF, START] = True   # empty subsequence marker (excluded from transforms and from the answer)

        for v in nums:
            srcA = A.copy()
            srcA[OFF, START] = False
            srcB = B.copy()
            if v == 0:
                # product -> 0, sum unchanged, parity flips
                cB = np.zeros_like(B)
                cB[:, ZERO] = srcA.any(axis=1)
                cA = np.zeros_like(A)
                cA[:, ZERO] = srcB.any(axis=1)
            else:
                INV, so = tables[v]
                # take from even length: sign +, becomes odd length
                mA = np.zeros_like(srcA)
                mA[:, :nE] = srcA[:, INV]
                mA[:, OVER] = srcA[:, OVER] | srcA[:, so:nE].any(axis=1)
                mA[:, ZERO] = srcA[:, ZERO]
                cB = np.zeros_like(B)
                cB[v:, :] = mA[:S - v, :]
                # take from odd length: sign -, becomes even length
                mB = np.zeros_like(srcB)
                mB[:, :nE] = srcB[:, INV]
                mB[:, OVER] = srcB[:, OVER] | srcB[:, so:nE].any(axis=1)
                mB[:, ZERO] = srcB[:, ZERO]
                cA = np.zeros_like(A)
                cA[:S - v, :] = mB[v:, :]
            A |= cA
            B |= cB
            # taking v as the very first element
            if v == 0:
                B[OFF, ZERO] = True
            elif v <= limit:
                B[OFF + v, idx[v]] = True
            else:
                B[OFF + v, OVER] = True

        row = OFF + k
        best = -1
        for arr in (A, B):
            r = arr[row]
            cols = np.flatnonzero(r[:nE])
            if cols.size:
                best = max(best, int(PRODS[cols[-1]]))
            if r[ZERO]:
                best = max(best, 0)
        return best
```
