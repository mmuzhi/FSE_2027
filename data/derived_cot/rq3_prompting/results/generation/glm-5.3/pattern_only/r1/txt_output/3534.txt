```python
class Solution:
    def countPairs(self, nums: List[int]) -> int:
        def variants(x):
            s = str(x)
            res = {x}
            n = len(s)
            for a in range(n):
                for b in range(a + 1, n):
                    t = list(s)
                    t[a], t[b] = t[b], t[a]
                    res.add(int(''.join(t)))
            return res
        
        v = [variants(x) for x in nums]
        count = 0
        n = len(nums)
        for i in range(n):
            for j in range(i + 1, n):
                if nums[j] in v[i]:
                    count += 1
        return count
```
