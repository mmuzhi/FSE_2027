```python
class Solution:
    def minMaxSubarraySum(self, nums: List[int], k: int) -> int:
        n = len(nums)

        def solve(cmp):
            # cmp(a, b) True if a should be popped when b arrives
            stack = []  # (val, cnt)
            total = 0
            cur = 0  # sum of min/max over subarrays ending at current r
            for r in range(n):
                if r >= k:
                    # expire start s = r - k, at bottom
                    stack[0] = (stack[0][0], stack[0][1] - 1)
                    cur -= stack[0][0]
                    if stack[0][1] == 0:
                        stack.pop(0)
                v = nums[r]
                acc = 0
                while stack and cmp(stack[-1][0], v):
                    val, cnt = stack.pop()
                    cur -= val * cnt
                    acc += cnt
                stack.append((v, acc + 1))
                cur += v * (acc + 1)
                total += cur
            return total

        return solve(lambda a, b: a >= b) + solve(lambda a, b: a <= b)
```

```python
class Solution:
    def minMaxSubarraySum(self, nums: List[int], k: int) -> int:
        from collections import deque

        def solve(pop_cond):
            stack = deque()  # (val, cnt) groups of start positions, bottom = oldest
            cur = 0
            total = 0
            for r, v in enumerate(nums):

                    acc += cnt
                stack.append((v, acc + 1))
                cur += v * (acc + 1)
                total += cur
            return total

        return solve(lambda a, b: a >= b) + solve(lambda a, b: a <= b)
```

Looking at this problem, I need the sum of (min+max) over all subarrays of length ≤ k.

**Approach:** Two independent passes — sum of subarray minimums and sum of subarray maximums, each restricted to length ≤ k.

**Key mechanism (per pass):** Slide the right endpoint `r`, maintaining a monotonic stack of `(val, cnt)` partitioning start positions `s ∈ [r-k+1, r]` into consecutive groups with constant min/max (bottom = oldest/longest window, smallest min). Maintain `cur` = sum of mins over subarrays ending at `r`:
- **Expire:** when `r ≥ k`, start `s = r-k` leaves; it's always covered by the bottom entry (it's the longest subarray). Decrement its count, subtract its value, pop if empty.
- **Add:** pop entries dominated by `nums[r]` (val ≥ v for min), accumulate counts, push `(v, acc+1)`, update `cur`.

Each start position is counted exactly once per endpoint, so no tie-breaking hazards. O(n) amortized.

**Verification (Example 1, mins):** r=0: cur=1; r=1: push (2,1), cur=3; r=2: expire s=0 → subtract 1, push (3,1), cur=5. Min total = 1+3+5 = 9 ✓ (1+2+3+1+2). Max total = 11 ✓. Sum = 20 ✓.

**Verification (Example 2, mins):** r=0: cur=1; r=1: pop (1,1), push (-3,2), cur=-6; r=2: expire bottom (-3,2)→(-3,1), cur=-3, push (1,1), cur=-2. Total = 1-6-2 = -7 ✓. Max total = 1 ✓. Sum = -6 ✓.

```python
class Solution:
    def minMaxSubarraySum(self, nums: List[int], k: int) -> int:
        from collections import deque

        def solve(pop_cond):
            stack = deque()  # (val, cnt) groups of start positions; bottom = oldest
            cur = 0          # sum of min (or max) of subarrays ending at r, length <= k
            total = 0
            for r, v in enumerate(nums):
                # expire start position s = r - k (covered by bottom entry)
                if r >= k:
                    v0, c0 = stack.popleft()
                    cur -= v0
                    if c0 > 1:
                        stack.appendleft((v0, c0 - 1))
                # add nums[r]: merge dominated groups
                acc = 0
                while stack and pop_cond(stack[-1][0], v):
                    val, cnt = stack.pop()
                    cur -= val * cnt
                    acc += cnt
                stack.append((v, acc + 1))
                cur += v * (acc + 1)
                total += cur
            return total

        # min pass: pop while top >= v ; max pass: pop while top <= v
        return solve(lambda a, b: a >= b) + solve(lambda a, b: a <= b)
```
