import sys
from math import isqrt

def main():
    R = int(sys.stdin.buffer.read().split()[0])
    D = 4 * R * R
    isq = isqrt
    total = 0
    first = -1
    for t in range(1, 2 * R, 2):
        # t = 2a+1; need largest b with (2b+1)^2 <= D - t^2
        S = isq(D - t * t)
        v = ((S - 1) >> 1 << 1) + 1  # = 2*b_max + 1
        total += v
        if first < 0:
