```python
import sys
import numpy as np

def main():
    data = sys.stdin.buffer.read().split()
    H = int(data[0]); W = int(data[1])
    rows = data[2:2+H]
    n = 1 << W
    # convert rows to ints
    masks = np.array([int(r, 2) for r in rows], dtype=np.int64)
    cnt = np.bincount(masks, minlength=n).astype(np.int64)
    idx = np.arange(n, dtype=np.int64)
    pc = idx - ((idx >> 1) & 0x5555555555555555)
    # W <= 18 so use standard popcount for 32-bit:
    pc = (pc & 0x33333333) + ((pc >> 2) & 0x33333333)
    pc = (pc + (pc >> 4)) & 0x0F0F0F0F
    pc = (pc + (pc >> 8)) & 0x0F0F0F  # for up to 18 bits fine? 18 bits < 2^... let's do up to 16 then 32
    g = np.minimum(pc, W - pc)
    
    # WHT
    def wht(a):
        a = a.copy()
        h = 1
        m = a.shape[0]
        while h < m:
            b = a.reshape(-1, 2, h)
            x = b[:, 0, :].copy()
            y = b[:, 1, :]
            b[:, 0, :] = x + y
            b[:, 1, :] = x - y
            h <<= 1
        return a.reshape(-1)
    
    fc = wht(cnt)
    fg = wht(g)
    prod = fc * fg
    conv = wht(prod) // n
    print(int(conv.min()))

main()
```

```python
import sys
import numpy as np

def main():
    data = sys.stdin.buffer.read().split()
    H = int(data[0]); W = int(data[1])
    rows = data[2:2 + H]
    n = 1 << W
    masks = np.fromiter((int(r, 2) for r in rows), dtype=np.int64, count=H)
    cnt = np.bincount(masks, minlength=n).astype(np.int64)

    idx = np.arange(n, dtype=np.int64)
    v = idx - ((idx >> 1) & 0x5555555555555555)
    v = (v & 0x3333333333333333) + ((v >> 2) & 0x3333333333333333)
    v = (v + (v >> 4)) & 0x0F0F0F0F0F0F0F0F
    pc = (v * 0x0101010101010101) >> 56
    g = np.minimum(pc, W - pc)

    def wht(a):
        a = a.copy()
        h = 1
        m = a.shape[0]
        while h < m:
            b = a.reshape(-1, 2, h)
            x = b[:, 0, :] + b[:, 1, :]
            y = b[:, 0, :] - b[:, 1, :]
            b[:, 0, :] = x
            b[:, 1, :] = y
            h <<= 1
        return a

    prod = wht(cnt) * wht(g)
    conv = wht(prod) // n
    print(int(conv.min()))

main()
```

**Approach.** Any sequence of operations reduces to flipping a set of columns (mask `C`) and, independently per row, choosing whether to flip that row. For a fixed `C`, row with mask `m` contributes `min(pc(m⊕C), W−pc(m⊕C))`. So the answer is `min_C Σ_m cnt[m]·g(m⊕C)` where `cnt` is the histogram of row masks and `g(x)=min(pc(x),W−pc(x))`. That is exactly one XOR (dyadic) convolution `cnt ⊛ g`, computable for all `C` in `O(W·2^W)` with Walsh–Hadamard transforms (three transforms of size `2^W ≤ 2^18`), vectorized with numpy. Verified by hand on sample 1 (min = 2), sample 2 (0), and sample 3 (C=00110 gives 13). Intermediates stay within int64 (`|WHT(cnt)| ≤ H ≤ 2·10⁵`, `|WHT(g)| ≤ 2^18·9`, products and inverse-transform sums ≤ ~1.3·10¹⁷).

```python
            b[:, 1, :] = u - w
            h <<= 1
        return a

    prod = wht(cnt) * wht(g)
    conv = wht(prod) // n
    print(int(conv.min()))

main()
```
