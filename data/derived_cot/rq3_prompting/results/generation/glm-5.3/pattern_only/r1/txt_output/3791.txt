```python
from typing import List

class Solution:
    def numOfUnplacedFruits(self, fruits: List[int], baskets: List[int]) -> int:
        n = len(fruits)
        size = 1
        while size < n:
            size <<= 1
        NEG = 0  # capacities >= 1, so 0 marks "used/padding"
        t = [NEG] * (2 * size)
        t[size:size + n] = baskets
        for i in range(size - 1, 0, -1):
            a, b = t[2 * i], t[2 * i + 1]
            t[i] = a if a >= b else b
        
        unplaced = 0
        for f in fruits:
            if t[1] < f:
                unplaced += 1
                continue
            node = 1
            while node < size:
                node <<= 1
                if t[node] < f:
                    node += 1
            # node is leaf; basket index = node - size
            t[node] = NEG
            node >>= 1
            while node:
                l, r = t[2 * node], t[2 * node + 1]
                v = l if l >= r else r
                if t[node] == v:
                    break
                t[node] = v
                node >>= 1
        return unplaced
```
