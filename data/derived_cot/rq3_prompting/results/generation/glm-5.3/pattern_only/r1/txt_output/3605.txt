class Solution:

        ans = []
        for p in nums:
            if p == 2:

            t = 0

```python
class Solution:
    def minBitwiseArray(self, nums: List[int]) -> List[int]:
        # Key facts:
        # - x | (x+1) is always odd (one of two consecutive ints is odd),
        #   so nums[i] == 2 has no solution -> -1.
        # - If x ends with k trailing 1-bits (bit k = 0), then
        #   x | (x+1) = (prefix+1)*2^(k+1) - 1, i.e. a value whose low
        #   s = k+1 bits are all 1. Inverting for minimal x given p:
        #   choose s as large as possible (s = number of trailing 1s t of p),
        #   yielding x = p - 2^(t-1).
        ans = []
        for p in nums:
            if p == 2:
                ans.append(-1)
                continue
            t = 0
            while (p >> t) & 1:
                t += 1
            ans.append(p - (1 << (t - 1)))
        return ans
```
