```python
import sys
from bisect import bisect_right
import numpy as np

def axis_counts(coord_list, n, d):
    s = sorted(coord_list)
    P = [0]*(n+1)
    for i, v in enumerate(s):
        P[i+1] = P[i] + v
    total = P[n]
    k_med = (n-1)//2
    m = s[k_med]
    
    def f(x):
        k = bisect_right(s, x)
        return x*k - P[k] + (total - P[k]) - x*(n-k)

        return None  # no valid values

            hi = mid
```

```python
import sys
from bisect import bisect_right
import numpy as np

def axis_fvals(coord_list, n, d):
    s = sorted(coord_list)
    P = [0] * (n + 1)
    for i in range(n):
        P[i+1] = P[i] + s[i]
    total = P[n]
    m = s[(n - 1) // 2]

    def f(x):
        k = bisect_right(s, x)
        return x * k - P[k] + (total - P[k]) - x * (n - k)

    if f(m) > d:
        return None

    lo, hi = s[0] - d - 1, m
    while hi - lo > 1:
        mid = (lo + hi) // 2
        if f(mid) <= d:
            hi = mid
        else:
            lo = mid
    L = hi

    lo, hi = m, s[-1] + d + 1
    while hi - lo > 1:
        mid = (lo + hi) // 2
        if f(mid) <= d:
            lo = mid
        else:
            hi = mid
    R = lo

    sarr = np.array(s, dtype=np.int64)
    Parr = np.array(P, dtype=np.int64)
    xs = np.arange(L, R + 1, dtype=np.int64)
    ks = np.searchsorted(sarr, xs, side='right')
    fvals = xs * ks - Parr[ks] + (total - Parr[ks]) - xs * (n - ks)
    return fvals
```

```python
import sys
from bisect import bisect_right
import numpy as np

def axis_fvals(coord_list, n, d):
    s = sorted(coord_list)
    P = [0] * (n + 1)
    acc = 0
    for i in range(n):
        acc += s[i]
        P[i + 1] = acc
    total = P[n]
    m = s[(n - 1) // 2]

    def f(x):
        k = bisect_right(s, x)
        return x * k - P[k] + (total - P[k]) - x * (n - k)

    if f(m) > d:
        return None

    lo, hi = s[0] - d - 1, m
    while hi - lo > 1:
        mid = (lo + hi) // 2
        if f(mid) <= d:
            hi = mid
        else:
            lo = mid
    L = hi

    lo, hi = m, s[-1] + d + 1
    while hi - lo > 1:
        mid = (lo + hi) // 2
        if f(mid) <= d:
            lo = mid
        else:
            hi = mid
    R = lo

    sarr = np.array(s, dtype=np.int64)
    Parr = np.array(P, dtype=np.int64)
    xs = np.arange(L, R + 1, dtype=np.int64)
    ks = np.searchsorted(sarr, xs, side='right')
    fvals = xs * ks - Parr[ks] + (total - Parr[ks]) - xs * (n - ks)
    return fvals


def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0]); d = int(data[1])
    vals = np.fromiter(map(int, data[2:2 + 2 * n]), dtype=np.int64, count=2 * n)
    xs = vals[0::2].tolist()
    ys = vals[1::2].tolist()

    fx = axis_fvals(xs, n, d)
    if fx is None:
        print(0)
        return
    fy = axis_fvals(ys, n, d)
    if fy is None:
        print(0)
        return

    cntF = np.bincount(fx, minlength=d + 1)
    cntG = np.bincount(fy, minlength=d + 1)
    prefG = np.cumsum(cntG)
    ans = int(np.dot(cntF, prefG[::-1].copy()))
    print(ans)


main()
```

**Analysis.** The sum splits: `f(x) + g(y) ≤ D` where `f(x)=Σ|x−xᵢ|`, `g(y)=Σ|y−yᵢ|`. Both are convex piecewise-linear, so `{x : f(x) ≤ D}` is an interval found by binary search around a median (its size is O(D) worst case, N=1). Evaluate `f` over that interval vectorized via sorted coordinates + prefix sums, bucket-count values `0..D` for both axes, then answer = `Σ_v cntF[v]·(#y with g(y) ≤ D−v)` using prefix sums.

```python
import sys
from bisect import bisect_right
import numpy as np


def axis_fvals(coord_list, n, d):
    s = sorted(coord_list)
    P = [0] * (n + 1)
    acc = 0
    for i in range(n):
        acc += s[i]
        P[i + 1] = acc
    total = P[n]
    m = s[(n - 1) // 2]

    def f(x):
        k = bisect_right(s, x)
        return x * k - P[k] + (total - P[k]) - x * (n - k)

    if f(m) > d:
        return None

    lo, hi = s[0] - d - 1, m
    while hi - lo > 1:
        mid = (lo + hi) // 2
        if f(mid) <= d:
            hi = mid
        else:
            lo = mid
    L = hi

    lo, hi = m, s[-1] + d + 1
    while hi - lo > 1:
        mid = (lo + hi) // 2
        if f(mid) <= d:
            lo = mid
        else:
            hi = mid
    R = lo

    sarr = np.array(s, dtype=np.int64)
    Parr = np.array(P, dtype=np.int64)
    xs = np.arange(L, R + 1, dtype=np.int64)
    ks = np.searchsorted(sarr, xs, side='right')
    fvals = xs * ks - Parr[ks] + (total - Parr[ks]) - xs * (n - ks)
    return fvals
```
