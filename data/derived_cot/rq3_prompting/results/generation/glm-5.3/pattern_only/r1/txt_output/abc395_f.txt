def solve():
    import sys
    data = sys.stdin.buffer.read().split()

import sys
import numpy as np

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    x = int(data[1])
    vals = np.fromiter(map(int, data[2:2 + 2 * n]), dtype=np.int64, count=2 * n)
    U = vals[0::2]
    D = vals[1::2]
    total = int(U.sum() + D.sum())
    hi = int((U + D).min())
    idx = np.arange(n, dtype=np.int64)
    xidx = x * idx

    def feasible(H):
        L = np.maximum(H - D, 0)
        R = np.minimum(U, H)
        A = np.maximum.accumulate(L + xidx)
        B = np.minimum.accumulate(R - xidx)

    lo = 0
    while lo < hi:
        mid = (lo + hi + 1) // 2
        if feasible(mid):
            lo = mid
        else:
            hi = mid - 1
    print(total - n * lo)

main()

import sys
import numpy as np

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    x = int(data[1])
    vals = np.fromiter(map(int, data[2:2 + 2 * n]), dtype=np.int64, count=2 * n)

    xidx = x * idx

    def feasible(H):

        R = np.minimum(U, H)
        A = np.maximum.accumulate(L + xidx)
        B = np.minimum.accumulate(R - xidx)
