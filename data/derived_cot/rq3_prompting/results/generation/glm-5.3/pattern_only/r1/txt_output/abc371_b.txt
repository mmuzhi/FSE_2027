import sys

def main():
    data = sys.stdin.read().split()
    n = int(data[0])
    m = int(data[1])
    has_male = [False] * (n + 1)
    out = []
    idx = 2
    for _ in range(m):
        a = int(data[idx])
        b = data[idx + 1]
        idx += 2
        if b == 'M' and not has_male[a]:
            has_male[a] = True
            out.append("Yes")
        else:
            out.append("No")
    sys.stdout.write("\n".join(out) + "\n")

main()
