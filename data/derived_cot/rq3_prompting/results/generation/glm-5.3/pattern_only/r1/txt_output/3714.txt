from itertools import accumulate

class Solution:
    def minMaxSums(self, nums: List[int], k: int) -> int:
        MOD = 10**9 + 7
        n = len(nums)
        nums.sort()
        K = k - 1
        G = [1] * n  # level 0
        for _ in range(K):
            H = [1] + G[:n-1]
            G = [x % MOD for x in accumulate(H)]
        # G[m] = S(m, K)
        ans = 0
        for i in range(n):
            ans += nums[i] * (G[i] + G[n-1-i]) % MOD
            ans %= MOD
        return ans % MOD
```

from itertools import accumulate

class Solution:
    def minMaxSums(self, nums: List[int], k: int) -> int:
        MOD = 10 ** 9 + 7
        n = len(nums)
        nums.sort()
        K = k - 1
        G = [1] * n                      # G[c][m] = sum_{j<=c} C(m, j); level 0
        for _ in range(K):
            H = [1] + G[: n - 1]
            G = [x % MOD for x in accumulate(H)]
        ans = 0
        for i, v in enumerate(nums):
            ans = (ans + v * (G[i] + G[n - 1 - i])) % MOD
        return ans
```

from itertools import accumulate

class Solution:
    def minMaxSums(self, nums: List[int], k: int) -> int:
        MOD = 10 ** 9 + 7
        n = len(nums)
        nums.sort()
        K = k - 1  # max number of extra elements picked alongside nums[i]

        # G[m] = S(m, K) = sum_{j=0}^{min(K, m)} C(m, j)
        # Recurrence over levels c: S(m, c) = S(m-1, c) + S(m-1, c-1), S(0, c) = 1
        # => G_c[m] = 1 + prefix_sum(G_{c-1}[0..m-1]), built via accumulate.
        G = [1] * n  # level 0: S(m, 0) = 1
        for _ in range(K):
            H = [1] + G[: n - 1]
            G = [x % MOD for x in accumulate(H)]

        # For sorted index i (ties broken by position):
        #   as maximum: pick extras only from indices < i  -> S(i, K) = G[i]
        #   as minimum: pick extras only from indices > i  -> S(n-1-i, K) = G[n-1-i]
        ans = 0
        for i, v in enumerate(nums):
            ans = (ans + v * (G[i] + G[n - 1 - i])) % MOD
        return ans
