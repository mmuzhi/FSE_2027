from collections import Counter

class Solution:
    def maxDifference(self, s: str) -> int:
        cnt = Counter(s)
        odd_max = max(f for f in cnt.values() if f % 2 == 1)
        even_min = min(f for f in cnt.values() if f % 2 == 0)
        return odd_max - even_min
```

from collections import Counter

class Solution:
    def maxDifference(self, s: str) -> int:

        odd_max = max(f for f in cnt.values() if f % 2 == 1)
        even_min = min(f for f in cnt.values() if f % 2 == 0)
