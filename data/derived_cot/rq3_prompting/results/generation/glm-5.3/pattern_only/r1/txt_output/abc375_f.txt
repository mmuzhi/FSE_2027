import sys
import numpy as np

def main():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    q = int(data[idx]); idx += 1
    
    roads = []
    for _ in range(m):
        a = int(data[idx]) - 1; idx += 1
        b = int(data[idx]) - 1; idx += 1
        c = int(data[idx]); idx += 1
        roads.append((a, b, c))
    
    queries = []
    for _ in range(q):
        t = int(data[idx]); idx += 1
        if t == 1:
            i = int(data[idx]) - 1; idx += 1
            queries.append((1, i))
        else:
            x = int(data[idx]) - 1; idx += 1
            y = int(data[idx]) - 1; idx += 1
            queries.append((2, x, y))
    
    INF = 1 << 60
    # Determine roads closed at some point
    closed = [False] * m
    for tqq in queries:
        if tqq[0] == 1:
            closed[tqq[1]] = True
    
    dist = np.full((n, n), INF, dtype=np.int64)
    np.fill_diagonal(dist, 0)
    for i, (a, b, c) in enumerate(roads):
        if not closed[i]:
            if c < dist[a, b]:
                dist[a, b] = c
                dist[b, a] = c
    
    # Floyd-Warshall
    for k in range(n):
        dist = np.minimum(dist, dist[:, k:k+1] + dist[k:k+1, :])
    
    THRESH = 1 << 59
    answers = [None] * q
    for qi in range(q - 1, -1, -1):
        tqq = queries[qi]
        if tqq[0] == 2:
            x, y = tqq[1], tqq[2]
            d = int(dist[x, y])

            cand = np.minimum(dist[:, a:a+1] + (c + dist[b:b+1, :]),
                              dist[:, b:b+1] + (c + dist[a:a+1, :]))
            dist = np.minimum(dist, cand)
    
    out = '\n'.join(str(ans) for ans in answers if ans is not None)
    sys.stdout.write(out + '\n' if out else '')

if __name__ == '__main__':
    main()

**Approach (offline, reverse processing):** Since roads only ever close (never reopen) and there are ≤ 300 closures, process queries in reverse. Start from the *final* set of open roads (those never closed), compute all-pairs shortest paths once with Floyd–Warshall (N ≤ 300, vectorized with numpy), then sweep queries backwards: type-2 queries are answered from the current distance matrix in O(1); each type-1 query "re-opens" a road, and adding a single edge updates the whole matrix in O(N²) via `new[u][v] = min(old[u][v], old[u][a]+c+old[b][v], old[u][b]+c+old[a][v])` (valid since positive weights mean a shortest path uses the new edge at most once, decomposing into old-graph shortest paths). Total: one O(N³) Floyd + 300 × O(N²) updates + O(Q) lookups.

Key invariant: distance entries are either exact path lengths (≤ ~3·10¹¹) or exactly INF = 2⁶⁰, so no overflow (max computed value ≈ 2.3·10¹⁸ < int64 max), and `d ≥ 2⁵⁹` means unreachable.
