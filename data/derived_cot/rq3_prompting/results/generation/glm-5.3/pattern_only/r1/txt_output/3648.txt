        n = len(fruits)
        ans = sum(fruits[i][i] for i in range(n))
        
        def best(g):

            NEG = float('-inf')
            # dp over columns j in [lo, n-1] at current row

class Solution:
    def maxCollectedFruits(self, fruits: List[List[int]]) -> int:
        n = len(fruits)
        
        # Child 1: forced to walk the main diagonal.
        ans = sum(fruits[i][i] for i in range(n))
        
        def best(g):
            # Child starting at (0, n-1) moving down with column steps in {-1,0,+1}.
            # At row t, feasible columns are [max(t, n-1-t), n-1].
            # Cells on the main diagonal (j == t) are already collected by child 1.
            NEG = float('-inf')
            dp = [NEG] * n
            dp[n - 1] = g[0][n - 1]
            for t in range(1, n):
                lo = t if t >= n - 1 - t else n - 1 - t
                ndp = [NEG] * n
                for j in range(lo, n):
                    m = dp[j]
                    if j and dp[j - 1] > m:
                        m = dp[j - 1]
                    if j + 1 < n and dp[j + 1] > m:
                        m = dp[j + 1]
                    if m != NEG:
                        ndp[j] = m + (g[t][j] if j > t else 0)
                dp = ndp
            return dp[n - 1]
        
        # Child 2 (from (0, n-1)) on the original grid.
        ans += best(fruits)
        # Child 3 (from (n-1, 0)) is the mirror problem on the transposed grid.
        ans += best([list(row) for row in zip(*fruits)])
        return ans

Looking at this problem:

**Key deductions:**
1. Child 1 (from (0,0)): each move raises `i+j` by 1 or 2, and it must gain `2(n-1)` over exactly `n-1` moves → every move is `(i+1,j+1)`. It walks the main diagonal (fixed contribution).
2. Child 2 (from (0,n-1)): at row `t`, column `j` with `j ≥ n-1-t` (column drops ≤1/row from start) and `j ≥ t` (must climb back to column `n-1` at the end). So it stays **on/above** the main diagonal.
3. Child 3 (from (n-1,0)): by symmetry stays **on/below** the main diagonal (its problem is child 2's problem on the transposed grid).

Therefore the three paths only ever overlap on the main diagonal, which child 1 fully collects. The total = diagonal sum + best "extra" path of child 2 (cells strictly above diagonal) + best "extra" path of child 3 (strictly below diagonal), optimized **independently** via O(n²) DPs over the band `j ∈ [max(t, n-1-t), n-1]` (diagonal cells count 0).

Verified on both examples (34+24+42=100; 2+1+1=4) and hand-checked edge cases (n=2, n=3 zigzags).
