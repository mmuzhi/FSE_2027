```python
class Solution:
    def minCosts(self, cost: List[int]) -> List[int]:
        answer = []
        mn = float('inf')
        for c in cost:
            mn = min(mn, c)
            answer.append(mn)
        return answer
```

class Solution:
    def minCosts(self, cost: List[int]) -> List[int]:
        answer = []
        mn = float('inf')
        for c in cost:
            if c < mn:
                mn = c
            answer.append(mn)
        return answer
