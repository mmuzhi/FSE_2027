```python
class Solution:
    def countNonDecreasingSubarrays(self, nums: List[int], k: int) -> int:
        from collections import deque
        n = len(nums)
        dq = deque()  # blocks [value, length, sum], left to right
        cost = 0
        r = n - 1
        ans = 0
        for l in range(n - 1, -1, -1):
            x = nums[l]
            tot_len, tot_sum = 1, x
            while dq and dq[0][0] <= x:
                v, ln, s = dq.popleft()
                cost += (x - v) * ln
                tot_len += ln
                tot_sum += s
            dq.appendleft((x, tot_len, tot_sum))
            while cost > k and r > l:
                v, ln, s = dq.pop()
                y = nums[r]
                cost -= v - y
                if ln > 1:
                    dq.append((v, ln - 1, s - y))
                r -= 1
            ans += r - l + 1
        return ans
```

```python
from collections import deque
from typing import List


class Solution:
    def countNonDecreasingSubarrays(self, nums: List[int], k: int) -> int:
        # Minimal cost to fix subarray [l..r] = sum_i (prefix_max(l..i) - nums[i]).
        # Iterate l from right to left; maintain window [l, r] where r is the max
        # right endpoint with cost <= k (r is non-increasing as l decreases).
        # Blocks of equal prefix-max values, stored left->right in a deque with
        # strictly increasing values; cost = sum over blocks of (v*len - s).
        n = len(nums)
        dq = deque()          # blocks (value, length, sum_of_elements)
        cost = 0
        r = n - 1
        ans = 0
        for l in range(n - 1, -1, -1):
            # Extend window to the left with nums[l]:
            # all prefix maxima of blocks with value <= x rise to x.
            x = nums[l]
            tot_len, tot_sum = 1, x
            while dq and dq[0][0] <= x:
                v, ln, s = dq.popleft()
                cost += (x - v) * ln
                tot_len += ln
                tot_sum += s
            dq.appendleft((x, tot_len, tot_sum))
            # Shrink from the right while cost exceeds k (never below r == l,
            # where cost is 0 <= k).
            while cost > k and r > l:
                v, ln, s = dq.pop()
                y = nums[r]
                cost -= v - y
                if ln > 1:
                    dq.append((v, ln - 1, s - y))
                r -= 1
            ans += r - l + 1
        return ans
```
