class Solution:
    def solveQueries(self, nums: List[int], queries: List[int]) -> List[int]:
        n = len(nums)
        groups = defaultdict(list)
        for i, v in enumerate(nums):
            groups[v].append(i)
        ans = [-1] * n
        for pos in groups.values():
            k = len(pos)
            if k == 1:
                continue
            for i in range(k):
                g_next = (pos[(i + 1) % k] - pos[i]) % n
                g_prev = (pos[i] - pos[(i - 1) % k]) % n
                ans[pos[i]] = min(g_next, n - g_next, g_prev, n - g_prev)
        return [ans[q] for q in queries]
```
