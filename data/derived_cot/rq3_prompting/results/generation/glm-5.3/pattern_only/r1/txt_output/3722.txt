class Solution:
    def maxSum(self, nums: List[int], k: int, m: int) -> int:
        n = len(nums)
        prefix = [0] * (n + 1)
        s = 0
        for i, x in enumerate(nums):
            s += x
            prefix[i + 1] = s

        NEG = -10**18
        # prev[i] = dp[i][j-1]: best sum of (j-1) subarrays within first i elements
        prev = [0] * (n + 1)  # j = 0 case: 0 subarrays -> sum 0

        for j in range(1, k + 1):
            cur = [NEG] * (n + 1)
            best = NEG
            start = j * m
            for i in range(start, n + 1):
                l = i - m
                # new candidate start index l for the j-th subarray ending at i-1
                c = prev[l] - prefix[l]
                if c > best:
                    best = c
                t = prefix[i] + best          # take subarray [l..i-1]
                p = cur[i - 1]                # skip element i-1
                cur[i] = t if t > p else p
            prev = cur

        return prev[n]
