class Solution:
    def countWinningSequences(self, s: str) -> int:
        MOD = 1_000_000_007
        # delta for Bob's move (F, W, E) against Alice's move
        # F beats E, W beats F, E beats W
        DL = {'F': (0, 1, -1), 'W': (-1, 0, 1), 'E': (1, -1, 0)}
        
        first = DL[s[0]]
        # dp[m] : distribution of (Bob points - Alice points) after rounds so far,
        # given Bob's last move was m. List index j corresponds to diff = j - r
        # after r rounds (list length 2r+1).
        dp = []
        for m in range(3):
            row = [0, 0, 0]
            row[first[m] + 1] = 1
            dp.append(row)
        
        for ch in s[1:]:
            dl = DL[ch]
            a, b, c = dp
            total = [x + y + z for x, y, z in zip(a, b, c)]
            ndp = []
            for m in range(3):
                # sequences whose previous move != m
                comp = [(t - x) % MOD for t, x in zip(total, dp[m])]
                d = dl[m]
                # new index j (diff = j - (r+1)) pulls comp from index j - 1 - d
                if d == 1:
                    ndp.append([0, 0] + comp)
                elif d == -1:
                    ndp.append(comp + [0, 0])
                else:
                    ndp.append([0] + comp + [0])
            dp = ndp
        
        n = len(s)
        # final index j corresponds to diff = j - n; need diff > 0 -> j >= n+1
        return sum(sum(row[n + 1:]) for row in dp) % MOD
