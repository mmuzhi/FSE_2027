```python
class Solution:
    def countGoodIntegers(self, n: int, k: int) -> int:
        from math import factorial
        good = set()
        half_len = (n + 1) // 2
        start = 10 ** (half_len - 1)
        end = 10 ** half_len
        for half in range(start, end):
            s = str(half)
            if n % 2 == 0:
                pal = s + s[::-1]
            else:
                pal = s + s[:-1][::-1]
            num = int(pal)
            if num % k == 0:
                good.add(tuple(sorted(pal)))
        total = 0
        fact = [factorial(i) for i in range(n + 1)]
        for sig in good:
            cnt = [0] * 10
            for ch in sig:
                cnt[ord(ch) - 48] += 1
            denom = 1
            for c in cnt:
                denom *= fact[c]
            total += fact[n - 1] * (n - cnt[0]) // denom
        return total
```

```python
class Solution:
    def countGoodIntegers(self, n: int, k: int) -> int:
        from math import factorial
        fact = [factorial(i) for i in range(n + 1)]
        good = set()
        half_len = (n + 1) // 2
        for half in range(10 ** (half_len - 1), 10 ** half_len):
            s = str(half)
            pal = s + s[::-1] if n % 2 == 0 else s + s[:-1][::-1]
            if int(pal) % k == 0:
                good.add(''.join(sorted(pal)))
        total = 0
        for sig in good:
            cnt = [0] * 10
            for ch in sig:
                cnt[ord(ch) - 48] += 1
            denom = 1
            for c in cnt:
                if c:
                    denom *= fact[c]
            total += fact[n - 1] * (n - cnt[0]) // denom
        return total
```

```python
class Solution:
    def countGoodIntegers(self, n: int, k: int) -> int:
        from math import factorial
        fact = [factorial(i) for i in range(n + 1)]
        good = set()
        half_len = (n + 1) // 2
        for half in range(10 ** (half_len - 1), 10 ** half_len):
            s = str(half)
            pal = s + s[::-1] if n % 2 == 0 else s + s[:-1][::-1]
            if int(pal) % k == 0:
                good.add(''.join(sorted(pal)))
        total = 0

            denom = 1
            for c in cnt:
                denom *= fact[c]
            total += fact[n - 1] * (n - cnt[0]) // denom
```

```python
class Solution:
    def countGoodIntegers(self, n: int, k: int) -> int:
        from math import factorial
        fact = [factorial(i) for i in range(n + 1)]
        
        # Enumerate all n-digit palindromes by their first half (no leading zero),
        # keep digit multiset signature of those divisible by k.
        good = set()
        half_len = (n + 1) // 2
        for half in range(10 ** (half_len - 1), 10 ** half_len):
            s = str(half)
            pal = s + s[::-1] if n % 2 == 0 else s + s[:-1][::-1]
            if int(pal) % k == 0:
                good.add(''.join(sorted(pal)))
        
        # For each good multiset, count distinct n-digit arrangements (no leading zero):
        # n!/prod(cnt!) - (n-1)!*cnt0/prod(cnt!) = (n-1)!*(n - cnt0)/prod(cnt!)
        total = 0
        for sig in good:
            cnt = [0] * 10
            for ch in sig:
                cnt[ord(ch) - 48] += 1
            denom = 1
            for c in cnt:
                if c:
                    denom *= fact[c]
            total += fact[n - 1] * (n - cnt[0]) // denom
        return total
```
