class Solution:
    def shiftDistance(self, s: str, t: str, nextCost: List[int], previousCost: List[int]) -> int:
        # Doubled arrays + prefix sums for O(1) cyclic range sums
        nd = nextCost + nextCost
        pd = previousCost + previousCost
        prefN = [0] * 53
        prefP = [0] * 53
        for i in range(52):
            prefN[i + 1] = prefN[i] + nd[i]
            prefP[i + 1] = prefP[i] + pd[i]

        # Precompute min cost for every (from, to) pair
        table = [[0] * 26 for _ in range(26)]
        for a in range(26):
            for b in range(26):
                d = (b - a) % 26  # forward steps
                fwd = prefN[a + d] - prefN[a]  # nextCost[a..b-1] cyclic
                d2 = (a - b) % 26  # backward steps
                bwd = prefP[b + d2 + 1] - prefP[b + 1]  # prevCost[b+1..a] cyclic
                table[a][b] = fwd if fwd < bwd else bwd

        total = 0
        for cs, ct in zip(s, t):
            total += table[ord(cs) - 97][ord(ct) - 97]
        return total

        # means a removable non-negative cycle), so answer = min(pure forward, pure backward).
        table = [[0] * 26 for _ in range(26)]
        for a in range(26):
            for b in range(26):
                d = (b - a) % 26  # forward steps
                fwd = prefN[a + d] - prefN[a]  # sum nextCost[a..b-1] cyclically
                d2 = (a - b) % 26  # backward steps
                bwd = prefP[b + d2 + 1] - prefP[b + 1]  # sum prevCost[b+1..a] cyclically
                table[a][b] = fwd if fwd < bwd else bwd

        total = 0
        for cs, ct in zip(s, t):
            total += table[ord(cs) - 97][ord(ct) - 97]
        return total
