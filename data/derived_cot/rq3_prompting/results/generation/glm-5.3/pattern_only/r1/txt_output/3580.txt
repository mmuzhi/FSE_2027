class Solution:
    def minStartingIndex(self, s: str, pattern: str) -> int:
        n, m = len(s), len(pattern)

        def z_func(t: str):
            L = len(t)
            z = [0] * L
            l = r = 0
            for i in range(1, L):
                if i < r:
                    z[i] = min(r - i, z[i - l])
                while i + z[i] < L and t[z[i]] == t[i + z[i]]:
                    z[i] += 1
                if i + z[i] > r:
                    l, r = i, i + z[i]
            return z

        # A[i] = length of longest common prefix of s[i:] and pattern
        z1 = z_func(pattern + '#' + s)
        A = z1[m + 1:]  # A[i] for i in 0..n-1, capped at m by the '#'

        # B[e] = length of longest common suffix of s[:e] and pattern (e in 1..n)
        z2 = z_func(pattern[::-1] + '#' + s[::-1])
        B = [0] * (n + 1)
        for e in range(1, n + 1):
            B[e] = z2[m + 1 + n - e]

        # Substring starting at i matches with <= 1 mismatch
        # iff A[i] + B[i+m] >= m - 1
        for i in range(n - m + 1):
            if A[i] + B[i + m] >= m - 1:
                return i
        return -1
