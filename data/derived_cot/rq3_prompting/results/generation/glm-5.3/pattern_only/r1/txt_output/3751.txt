```python
class Solution:
    def maxFrequency(self, nums: List[int], k: int) -> int:
        n = len(nums)
        # prefix count of k
        prek = [0] * (n + 1)
        for i, x in enumerate(nums):
            prek[i + 1] = prek[i] + (1 if x == k else 0)
        base = prek[n]
        
        from collections import defaultdict
        pos = defaultdict(list)
        for i, x in enumerate(nums):
            if x != k:
                pos[x].append(i)
        
        best_gain = 0
        for v, P in pos.items():
            m = len(P)
            cur = 0
            for i in range(m):
                cur += 1
                if cur > best_gain:
                    best_gain = cur
                if i + 1 < m:
                    gap = prek[P[i + 1]] - prek[P[i] + 1]
                    cur -= gap
                    if cur < 0:
                        cur = 0
        return base + best_gain
```

```python
class Solution:
    def maxFrequency(self, nums: List[int], k: int) -> int:
        n = len(nums)
        # prefix count of value k
        prek = [0] * (n + 1)
        for i, x in enumerate(nums):
            prek[i + 1] = prek[i] + (1 if x == k else 0)
        base = prek[n]  # frequency of k without any change

        # positions of every other value
        pos = {}
        for i, x in enumerate(nums):
            if x != k:
                if x in pos:
                    pos[x].append(i)
                else:
                    pos[x] = [i]

        best_gain = 0
        for P in pos.values():
            m = len(P)
            cur = 0
            for i in range(m):
                cur += 1  # picking one more occurrence of v into the window
                if cur > best_gain:
                    best_gain = cur
                if i + 1 < m:
                    # k's strictly between P[i] and P[i+1] are lost
                    cur -= prek[P[i + 1]] - prek[P[i] + 1]
                    if cur < 0:
                        cur = 0
        return base + best_gain
```
