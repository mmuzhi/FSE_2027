```python
        self.large = []  # min-heap

        self.smallSize = 0  # valid count
        self.largeSize = 0
        self.k = k
        self.smallSum = 0
        self.largeSum = 0

            while heap and self.delayed[-heap[0]]:
                self.delayed[-heap[0]] -= 1
                heapq.heappop(heap)
        else:
            while heap and self.delayed[heap[0]]:
                self.delayed[heap[0]] -= 1
                heapq.heappop(heap)

            self.prune(self.small)
```

```python
import heapq
from collections import Counter
from typing import List

class Solution:
    def minOperations(self, nums: List[int], x: int, k: int) -> int:
        n = len(nums)
        m = n - x + 1  # number of windows
        
        # Dual heap sliding median
        small = []  # max-heap (negated values)
        large = []  # min-heap
        delayed = Counter()
        smallSum = 0
        largeSum = 0
        smallSize = 0
        largeSize = 0
        
        heappush = heapq.heappush
        heappop = heapq.heappop
        
        def pruneSmall():
            while small and delayed[-small[0]]:
                delayed[-small[0]] -= 1
                heappop(small)
        
        def pruneLarge():
            while large and delayed[large[0]]:
                delayed[large[0]] -= 1
                heappop(large)
        
        def balance():
            nonlocal smallSize, largeSize, smallSum, largeSum
            if smallSize > largeSize + 1:
                pruneSmall()
                v = -heappop(small)
                smallSize -= 1
                smallSum -= v
                heappush(large, v)
                largeSize += 1
                largeSum += v
                pruneSmall()
            elif smallSize < largeSize:
                pruneLarge()
                v = heappop(large)
                largeSize -= 1
                largeSum -= v
                heappush(small, -v)
                smallSize += 1
                smallSum += v
                pruneLarge()
        
        cost = [0] * m
        for i in range(n):
            v = nums[i]
            # insert v
            if not small or v <= -small[0]:
                heappush(small, -v)
                smallSize += 1
                smallSum += v
            else:
                heappush(large, v)
                largeSize += 1
                largeSum += v
            balance()
            if i >= x:
                # remove nums[i - x]
                w = nums[i - x]
                delayed[w] += 1
                if w <= -small[0]:
                    smallSize -= 1
                    smallSum -= w
                    if w == -small[0]:
                        pruneSmall()
                else:
                    largeSize -= 1
                    largeSum -= w
                    if w == large[0]:
                        pruneLarge()
                balance()
            if i >= x - 1:
                med = -small[0]
                cost[i - x + 1] = smallSize * med - smallSum + largeSum - largeSize * med
        
        # DP
        INF = float('inf')
        prev = [0] * (n + 1)
        for j in range(1, k + 1):
            cur = [INF] * (n + 1)
            base = j * x
            for i in range(base, n + 1):
                c = prev[i - x] + cost[i - x]
                p = cur[i - 1]
                cur[i] = c if c < p else p
            prev = cur
        return prev[n]
```

```python
import heapq
from collections import Counter
from itertools import accumulate
from operator import add
from typing import List

class Solution:
    def minOperations(self, nums: List[int], x: int, k: int) -> int:
        n = len(nums)
        m = n - x + 1
        
        small = []   # max-heap via negation: holds smaller half
        large = []   # min-heap: holds larger half
        delayed = Counter()
        smallSum = [0]
        largeSum = [0]
        smallSize = [0]
        largeSize = [0]
        
        heappush = heapq.heappush
        heappop = heapq.heappop
        
        def pruneSmall():
            s = small
            while s and delayed[-s[0]]:
                delayed[-s[0]] -= 1
                heappop(s)
        
        def pruneLarge():
            l = large
            while l and delayed[l[0]]:
                delayed[l[0]] -= 1
                heappop(l)
        
        def balance():
            if smallSize[0] > largeSize[0] + 1:
                pruneSmall()
                v = -heappop(small)
                smallSize[0] -= 1
                smallSum[0] -= v
                heappush(large, v)
                largeSize[0] += 1
                largeSum[0] += v
                pruneSmall()
            elif smallSize[0] < largeSize[0]:
                pruneLarge()
                v = heappop(large)
                largeSize[0] -= 1
                largeSum[0] -= v
                heappush(small, -v)
                smallSize[0] += 1
                smallSum[0] += v
                pruneLarge()
        
        cost = [0] * m
        for i in range(n):
            v = nums[i]
            if not small or v <= -small[0]:
                heappush(small, -v)
                smallSize[0] += 1
                smallSum[0] += v
            else:
                heappush(large, v)
                largeSize[0] += 1
                largeSum[0] += v
            balance()
            
            if i >= x:
                w = nums[i - x]
                delayed[w] += 1
                if w <= -small[0]:
                    smallSize[0] -= 1
                    smallSum[0] -= w
                    if w == -small[0]:
                        pruneSmall()
                else:
                    largeSize[0] -= 1
                    largeSum[0] -= w
                    if w == large[0]:
                        pruneLarge()
                balance()
            
            if i >= x - 1:
                med = -small[0]
                cost[i - x + 1] = smallSize[0] * med - smallSum[0] + largeSum[0] - largeSize[0] * med
        
        INF = float('inf')
        prev = [0] * (n + 1)
        for j in range(1, k + 1):
            lo = (j - 1) * x
            base = j * x
            cand = map(add, prev[lo:n - x + 1], cost[lo:])
            cur = [INF] * (n + 1)
            cur[base:n + 1] = accumulate(cand, min)
            prev = cur
        return prev[n]
```

```python
import heapq
from collections import Counter
from itertools import accumulate
from operator import add
from typing import List

class Solution:
    def minOperations(self, nums: List[int], x: int, k: int) -> int:
        n = len(nums)
        m = n - x + 1

        small = []      # max-heap (stored negated): smaller half
        large = []      # min-heap: larger half
        delayed = Counter()
        smallSum = 0
        largeSum = 0
        smallSize = 0
        largeSize = 0

        heappush = heapq.heappush
        heappop = heapq.heappop

        def pruneSmall():
            while small and delayed[-small[0]]:
                delayed[-small[0]] -= 1
                heappop(small)

        def pruneLarge():
            while large and delayed[large[0]]:
                delayed[large[0]] -= 1
                heappop(large)

        def balance():
            nonlocal smallSize, largeSize, smallSum, largeSum
            if smallSize > largeSize + 1:
                pruneSmall()
                v = -heappop(small)
                smallSize -= 1
                smallSum -= v
                heappush(large, v)
                largeSize += 1
                largeSum += v
                pruneSmall()
            elif smallSize < largeSize:
                pruneLarge()
                v = heappop(large)
                largeSize -= 1
                largeSum -= v
                heappush(small, -v)
                smallSize += 1
                smallSum += v
                pruneLarge()

        cost = [0] * m
        for i in range(n):
            v = nums[i]
            if not small or v <= -small[0]:
                heappush(small, -v)
                smallSize += 1
                smallSum += v
            else:
                heappush(large, v)
                largeSize += 1
                largeSum += v
            balance()

            if i >= x:
                w = nums[i - x]
                delayed[w] += 1
                if w <= -small[0]:
                    smallSize -= 1
                    smallSum -= w
                    if w == -small[0]:
                        pruneSmall()
                else:
                    largeSize -= 1
                    largeSum -= w
                    if w == large[0]:
                        pruneLarge()
                balance()

            if i >= x - 1:
                med = -small[0]
                cost[i - x + 1] = (smallSize * med - smallSum) + (largeSum - largeSize * med)

        INF = float('inf')
        prev = [0] * (n + 1)
        for j in range(1, k + 1):
            lo = (j - 1) * x
            base = j * x
            cur = [INF] * (n + 1)
            cur[base:n + 1] = accumulate(map(add, prev[lo:n - x + 1], cost[lo:]), min)
            prev = cur
        return prev[n]
```

=== ANSWER ===
Looking at this problem:

1. **Cost per window**: Making all elements in a window of size `x` equal is minimized by the median → cost = Σ|aᵢ − median|. Compute for all `n−x+1` windows with a dual-heap (lazy deletion) sliding median, tracking sums of both halves.

2. **Selection DP**: `dp[j][i]` = min cost to pick `j` non-overlapping windows within first `i` elements: `dp[j][i] = min(dp[j][i−1], dp[j−1][i−x] + cost[i−x])`. Answer `dp[k][n]`. Complexity O(n log n + k·n), fine for n=1e5, k≤15.

Key invariants verified: heap tops are always valid (delayed elements pruned on surfacing); the formula `(smallSize·med − smallSum) + (largeSum − largeSize·med)` gives exact median cost for both parities of `x`. Both examples hand-traced correctly (costs `[11,0,0,3,4]` → 3; `[7,5,6,4,6,3,7]` → 8).

```python
            cur = [INF] * (n + 1)
            cur[base:n + 1] = accumulate(map(add, prev[lo:n - x + 1], cost[lo:]), min)
            prev = cur
        return prev[n]
```
