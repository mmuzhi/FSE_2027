```python
class Solution:
    def countOfSubstrings(self, word: str, k: int) -> int:
        vowels = set('aeiou')

        def at_least(m: int) -> int:
            # counts substrings containing all 5 vowels and >= m consonants
            cnt = {v: 0 for v in vowels}
            cons = 0
            left = 0
            total = 0
            for right, ch in enumerate(word):
                if ch in cnt:
                    cnt[ch] += 1
                else:
                    cons += 1
                # shrink while current window still satisfies the condition;
                # each left position passed through was a valid start for this right
                while left <= right and cons >= m and all(cnt[v] > 0 for v in vowels):
                    c = word[left]
                    if c in cnt:
                        cnt[c] -= 1
                    else:
                        cons -= 1
                    left += 1
                total += left
            return total

        return at_least(k) - at_least(k + 1)
```
