```python
from itertools import accumulate
from operator import xor
from typing import List

class Solution:
    def maximumSubarrayXor(self, nums: List[int], queries: List[List[int]]) -> List[int]:
        n = len(nums)
        q = len(queries)
        ans = [0] * q
        # group queries by l
        queries_by_l = [[] for _ in range(n)]
        for qi, (l, r) in enumerate(queries):
            queries_by_l[l].append((r, qi))
        
        INF_placeholder = 0
        prev_f = [0] * n
        prev_g = [0] * n
        for l in range(n - 1, -1, -1):
            # build f row
            tail = prev_f[l+1:]
            f_row = [0] * l
            f_row.extend(accumulate([nums[l]] + tail, xor))
            # h = elementwise max of f_row and prev_g
            h = list(map(max, f_row, prev_g))
            g_row = list(accumulate(h, max))
            # answer queries
            for r, qi in queries_by_l[l]:
                ans[qi] = g_row[r]
            prev_f = f_row
            prev_g = g_row
        return ans
```

```python
from itertools import accumulate
from operator import xor
from typing import List

class Solution:
    def maximumSubarrayXor(self, nums: List[int], queries: List[List[int]]) -> List[int]:
        n = len(nums)
        ans = [0] * len(queries)
        
        # Group queries by their left endpoint.
        queries_by_l = [[] for _ in range(n)]
        for qi, (l, r) in enumerate(queries):
            queries_by_l[l].append((r, qi))
        
        prev_f = [0] * n  # f row for l+1 (valid from index l+1)
        prev_g = [0] * n  # g row for l+1 (valid from index l+1)
        for l in range(n - 1, -1, -1):
            # f[l][r] = f[l][r-1] ^ f[l+1][r], with f[l][l] = nums[l]
            f_row = [0] * l
            f_row.extend(accumulate(prev_f[l + 1:], xor, initial=nums[l]))
            # g[l][r] = max(f[l][r], g[l][r-1], g[l+1][r]) -> running max of h
            h = list(map(max, f_row, prev_g))
            g_row = list(accumulate(h, max))
            
            for r, qi in queries_by_l[l]:
                ans[qi] = g_row[r]
            
            prev_f, prev_g = f_row, g_row
        
        return ans
```

**Commit:** The XOR score of `nums[l..r]` satisfies `f[l][r] = f[l][r-1] ^ f[l+1][r]` (by the Pascal-triangle-mod-2 structure of the reduction). Then the query answer `g[l][r] = max(f[l][r], g[l][r-1], g[l+1][r])`, which unrolls to a running max of `h[r] = max(f[l][r], g[l+1][r])` along row `l`. So: process rows `l` from `n-1` down to `0`, keeping only previous rows of `f` and `g` (O(n) memory), answering queries grouped by `l` in O(1). Complexity O(n² + q), implemented with C-speed `accumulate` (xor) and `map(max, ...)` + `accumulate` (max).

**Ledger:** invariant — f-row built via running XOR of `prev_f[l+1:]` seeded with `nums[l]`; placeholders `0` at indices `< l` are harmless since all values ≥ 0. Edge cases: `l == r` (running max at index `l` gives `nums[l]`), `n == 1`.
