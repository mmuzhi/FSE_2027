import sys

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    a = list(map(int, data[1:1+n]))
    NEG = float('-inf')
    dp = [NEG, NEG]  # dp[0]: ends even, dp[1]: ends odd
    for t in range(1, n + 1):
        v = a[t - 1]
        q = t & 1
        if q:
            best = dp[1]
            if v > best:
                best = v
            if dp[0] != NEG and dp[0] + v > best:
                best = dp[0] + v
            dp[1] = best
        else:
            if dp[1] != NEG and dp[1] + v > dp[0]:
                dp[0] = dp[1] + v
    ans = dp[n & 1]
    if n % 2 == 0:
        ans = max(ans, 0)
    print(ans)

main()

import sys

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    a = list(map(int, data[1:1 + n]))
    NEG = -(1 << 60)
    # Characterization (proven): the final stack is exactly a subsequence
    # A[j1],...,A[jk] with j1 odd, consecutive indices of opposite parity,
    # and N - jk even (equivalently jk ≡ N mod 2). Empty stack only if N even.
    # dp0: best sum of such an alternating (start-odd) subsequence ending at an even index
    # dp1: best sum ending at an odd index
    dp0 = NEG
    dp1 = NEG
    for i, v in enumerate(a):
        if i % 2 == 0:            # 1-based position i+1 is odd: may start fresh or extend from even
            dp1 = max(dp1, v, dp0 + v)
        else:                     # even position: extend from odd-ending only
            dp0 = max(dp0, dp1 + v)
    if n % 2:
        print(dp1)
    else:
        print(max(dp0, 0))

main()
