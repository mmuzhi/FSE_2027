Given two integers, n and k, an alternating permutation is a permutation of the first n positive integers such that no two adjacent elements are both odd or both even.
Return the k-th alternating permutation sorted in lexicographical order. If there are fewer than k valid alternating permutations, return an empty list.

Input: n = 4, k = 6

```python
from typing import List

class Solution:
    def permute(self, n: int, k: int) -> List[int]:
        fact = [1] * (n + 2)
        for i in range(1, n + 2):
            fact[i] = fact[i - 1] * i
        
        odd_list = list(range(1, n + 1, 2))
        even_list = list(range(2, n + 1, 2))
        
        def completions(o: int, e: int, next_parity: int) -> int:
            # ways to place o odds and e evens in alternating order,
            # where the next position must have parity next_parity (1=odd, 0=even)
            t = o + e
            if t == 0:
                return 1
            if next_parity == 1:
                need_o, need_e = (t + 1) // 2, t // 2
            else:
                need_o, need_e = t // 2, (t + 1) // 2
            if o == need_o and e == need_e:
                return fact[o] * fact[e]
            return 0
        
        res = []
        req = None  # required parity for current position (None = free choice)
        
        for _ in range(n):
            if req is None:
                cand = sorted(odd_list + even_list)
            elif req == 1:
                cand = odd_list
            else:
                cand = even_list
            
            chosen = None
            for c in cand:
                if c % 2 == 1:
                    cnt = completions(len(odd_list) - 1, len(even_list), 0)
                else:
                    cnt = completions(len(odd_list), len(even_list) - 1, 1)
                if k > cnt:
                    k -= cnt
                else:
                    chosen = c
                    break
            
            if chosen is None:
                return []
            
            if chosen % 2 == 1:
                odd_list.remove(chosen)
                req = 0
            else:
                even_list.remove(chosen)
                req = 1
            res.append(chosen)
        
        return res
```

Looking at this problem:

- **Key insight**: Once the first element's parity is chosen, the entire parity pattern is forced (strictly alternating). For odd `n`, the first element must be odd (more odds than evens); for even `n`, either parity works.
- **Counting**: From a state with `o` odds and `e` evens remaining where the next slot needs parity `p`, the number of completions is `o! * e!` if the counts exactly match the forced pattern (`ceil(t/2)` of parity `p`, `floor(t/2)` of the other, `t = o+e`), else `0`. This is because any bijection of remaining odds to odd-slots and evens to even-slots works.
- **Construction**: Standard k-th permutation descent — at each position try candidates in increasing order (position 0 free parity, later positions forced parity), subtract counts until `k` falls within a candidate's block.
- Python big ints handle counts up to `50!²` with no overflow; `k ≤ 10^15` just gets absorbed.

I traced all three examples (n=4,k=6 → [3,4,1,2]; n=3,k=2 → [3,2,1]; n=2,k=3 → []) plus n=1 and n=5 totals — all match.

```python
from typing import List

class Solution:
    def permute(self, n: int, k: int) -> List[int]:
        fact = [1] * (n + 2)
        for i in range(1, n + 2):
            fact[i] = fact[i - 1] * i

        odd_list = list(range(1, n + 1, 2))
        even_list = list(range(2, n + 1, 2))

        def completions(o: int, e: int, next_parity: int) -> int:
            # ways to fill o odds and e evens alternately,
            # next slot must have parity next_parity (1=odd, 0=even)
            t = o + e
            if t == 0:
                return 1
            if next_parity == 1:
                need_o, need_e = (t + 1) // 2, t // 2
            else:
                need_o, need_e = t // 2, (t + 1) // 2
            if o == need_o and e == need_e:
                return fact[o] * fact[e]
            return 0

        res = []
        req = None  # required parity for current position (None = free choice)

        for _ in range(n):
            if req is None:
                cand = sorted(odd_list + even_list)
            elif req == 1:
                cand = odd_list
            else:
                cand = even_list

            chosen = None
            for c in cand:
                if c % 2 == 1:
                    cnt = completions(len(odd_list) - 1, len(even_list), 0)
                else:
                    cnt = completions(len(odd_list), len(even_list) - 1, 1)
                if k > cnt:
                    k -= cnt
                else:
                    chosen = c
                    break

            if chosen is None:
                return []

            if chosen % 2 == 1:
                odd_list.remove(chosen)
                req = 0
            else:
                even_list.remove(chosen)
                req = 1
            res.append(chosen)

        return res
```
