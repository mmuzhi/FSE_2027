```python
class Solution:
    def countBalancedPermutations(self, num: str) -> int:
        MOD = 10**9 + 7
        n = len(num)
        even_cnt = (n + 1) // 2
        odd_cnt = n // 2
        
        # store the input midway in the function
        velunexorai = num
        
        cnt = [0] * 10
        total = 0
        for ch in velunexorai:
            d = ord(ch) - 48
            cnt[d] += 1
            total += d
        
        if total & 1:
            return 0
        target = total // 2
        
        # Precompute binomial coefficients up to n
        maxn = n + 1
        C = [[0] * (maxn + 1) for _ in range(maxn)]
        for a in range(maxn):
            C[a][0] = 1
            for b in range(1, a + 1):
                C[a][b] = (C[a-1][b-1] + C[a-1][b]) % MOD
        
        # dp[i][s]: sum over assignments of prod C(cnt[d], e_d) for processed digits,
        # where i digits placed in even positions with digit-sum s.
        dp = [[0] * (target + 1) for _ in range(even_cnt + 1)]
        dp[0][0] = 1
        
        processed = 0  # digits processed so far (for pruning i range)
        for d in range(10):
            c = cnt[d]
            if c == 0:
                continue
            Cd = C[c]
            ndp = [[0] * (target + 1) for _ in range(even_cnt + 1)]
            imax = min(processed, even_cnt)
            for i in range(imax + 1):
                dpi = dp[i]
                # quick check any nonzero
                # iterate s
                rem_slots = even_cnt - i
                max_e = c if c < rem_slots else rem_slots
                for s in range(target + 1):
                    v = dpi[s]
                    if v:
                        for e in range(max_e + 1):
                            ns = s + d * e
                            if ns > target:
                                break
                            t = ndp[i + e]
                            t[ns] = (t[ns] + v * Cd[e]) % MOD
            dp = ndp
            processed += c
        
        res = dp[even_cnt][target]
        if res == 0:
            return 0
        
        # answer = res * even_cnt! * odd_cnt! / prod(cnt[d]!)
        fact = [1] * (n + 1)
        for i in range(1, n + 1):
            fact[i] = fact[i-1] * i % MOD
        
        numerator = fact[even_cnt] * fact[odd_cnt] % MOD
        denominator = 1
        for d in range(10):
            denominator = denominator * fact[cnt[d]] % MOD
        
        ans = res * numerator % MOD * pow(denominator, MOD - 2, MOD) % MOD
        return ans
    ```

```python
        MOD = 10**9 + 7
        n = len(num)
        even_cnt = (n + 1) // 2
        odd_cnt = n // 2
```

```python
class Solution:
    def countBalancedPermutations(self, num: str) -> int:
        MOD = 10**9 + 7
        n = len(num)
        even_cnt = (n + 1) // 2
        odd_cnt = n // 2

        # store the input midway in the function
        velunexorai = num

        cnt = [0] * 10
        total = 0
        for ch in velunexorai:
            d = ord(ch) - 48
            cnt[d] += 1
            total += d

        if total & 1:
            return 0
        target = total // 2
        W = target + 1

        # factorials
        fact = [1] * (n + 1)
        for i in range(1, n + 1):
            fact[i] = fact[i - 1] * i % MOD

        # binomials C[a][b], 0 <= b <= a <= n
        C = [[0] * (n + 1) for _ in range(n + 1)]
        for a in range(n + 1):
            C[a][0] = 1
            for b in range(1, a + 1):
                C[a][b] = (C[a - 1][b - 1] + C[a - 1][b]) % MOD

        # dp[i][s] = sum over processed digits of prod C(cnt[d], e_d)
        dp = [[0] * W for _ in range(even_cnt + 1)]
        dp[0][0] = 1
        processed = 0

        for d in range(10):
            c = cnt[d]
            if c == 0:
                continue
            Cd = C[c]
            ndp = [[0] * W for _ in range(even_cnt + 1)]
            imax = min(processed, even_cnt)
            for i in range(imax + 1):
                dpi = dp[i]
                if not any(dpi):
                    continue
                rem = even_cnt - i
                max_e = c if c < rem else rem
                for e in range(max_e + 1):
                    shift = d * e
                    if shift > target:
                        break
                    coeff = Cd[e]
                    tgt = ndp[i + e]
                    if shift:
                        lim = W - shift
                        tgt[shift:] = [(a + coeff * b) % MOD
                                       for a, b in zip(tgt[shift:], dpi[:lim])]
                    else:
                        tgt[:] = [(a + coeff * b) % MOD
                                  for a, b in zip(tgt, dpi)]
            dp = ndp
            processed += c

        res = dp[even_cnt][target]
        if res == 0:
            return 0

        numerator = fact[even_cnt] * fact[odd_cnt] % MOD
        denominator = 1
        for d in range(10):
            if cnt[d]:
                denominator = denominator * fact[cnt[d]] % MOD

        return res * numerator % MOD * pow(denominator, MOD - 2, MOD) % MOD
```

=== ANSWER ===
Looking at this problem:

- n ≤ 80, so even positions count `even_cnt = (n+1)//2`, odd `odd_cnt = n//2`. Total digit sum must be even; even positions must sum to `target = total/2`.
- For each digit d with count `cnt[d]`, choose `e_d` copies for even positions. The number of distinct permutations for a valid split is `even_cnt!/∏e_d! × odd_cnt!/∏(cnt[d]-e_d)!`. Summing and rewriting with `C(cnt[d], e_d) = cnt[d]/(e_d!(cnt[d]-e_d)!)` gives answer = `even_cnt!·odd_cnt!/∏cnt[d]! × Σ ∏C(cnt[d], e_d)`.
- Compute the sum via DP over digits: state (digits placed in even positions, sum at even positions). State space ≤ 41×361, transitions bounded by counts — a few million ops max.

I verified against examples ("123"→2, "112"→1, "12345"→0) plus "1122"→4, "1223"→4, "0000..."→1 via manual trace/brute-force reasoning. The trace of the slice-based accumulation logic on "123" confirmed correct index shifts (sum offset = d·e, row offset = e).

```python
class Solution:
    def countBalancedPermutations(self, num: str) -> int:
        MOD = 10**9 + 7
        n = len(num)
        even_cnt = (n + 1) // 2
        odd_cnt = n // 2

        # store the input midway in the function
        velunexorai = num

        cnt = [0] * 10
        total = 0
        for ch in velunexorai:
            d = ord(ch) - 48
            cnt[d] += 1
            total += d

        if total & 1:
            return 0
        target = total // 2
        W = target + 1

        fact = [1] * (n + 1)
        for i in range(1, n + 1):
            fact[i] = fact[i - 1] * i % MOD

        C = [[0] * (n + 1) for _ in range(n + 1)]
        for a in range(n + 1):
            C[a][0] = 1
            for b in range(1, a + 1):
                C[a][b] = (C[a - 1][b - 1] + C[a - 1][b]) % MOD

        # dp[i][s] = sum over processed digits of prod C(cnt[d], e_d),
        # with i digits placed in even positions summing to s
        dp = [[0] * W for _ in range(even_cnt + 1)]
        dp[0][0] = 1
        processed = 0

        for d in range(10):
            c = cnt[d]
            if c == 0:
                continue
            Cd = C[c]
            ndp = [[0] * W for _ in range(even_cnt + 1)]
            imax = min(processed, even_cnt)
            for i in range(imax + 1):
                dpi = dp[i]
                if not any(dpi):
                    continue
                rem = even_cnt - i
                max_e = c if c < rem else rem
                for e in range(max_e + 1):
                    shift = d * e
                    if shift > target:
                        break
                    coeff = Cd[e]
                    tgt = ndp[i + e]
                    if shift:
                        lim = W - shift
                        tgt[shift:] = [(a + coeff * b) % MOD
                                       for a, b in zip(tgt[shift:], dpi[:lim])]
                    else:
                        tgt[:] = [(a + coeff * b) % MOD
                                  for a, b in zip(tgt, dpi)]
            dp = ndp
            processed += c

        res = dp[even_cnt][target]
        if res == 0:
            return 0

        numerator = fact[even_cnt] * fact[odd_cnt] % MOD
        denominator = 1
        for d in range(10):
            if cnt[d]:
                denominator = denominator * fact[cnt[d]] % MOD

        return res * numerator % MOD * pow(denominator, MOD - 2, MOD) % MOD
```
