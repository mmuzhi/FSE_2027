```python
import sys

def main():
    data = sys.stdin.read().split()
    n = int(data[0])
    A = list(map(int, data[1:1+n]))
    if all(a == 1 for a in A):
        print("Yes"); return
    need_alt_check = (n % 4 == 2)
    # DP over b_1..b_n with state (b_j, flag) where flag = seen "00" pair among (b_1,b_2)...(b_{j-1},b_j)
    # reachable: dict/set of (b, f)
    ans = False
    for b1 in (0, 1):
        # states: set of (b, f)
        cur = {(b1, False)}
        for j in range(2, n + 1):
            aj = A[j - 1]
            nxt = set()
            for (b, f) in cur:
                for nb in (0, 1):
                    if b == 1 and nb == 1:
                        continue
                    if aj == 0 and not (b or nb):
                        continue
                    nf = f or (b == 0 and nb == 0)
                    nxt.add((nb, nf))
            cur = nxt
            if not cur:
                break
        for (bn, f) in cur:
            # wrap pair (b_n, b_1)
            if bn == 1 and b1 == 1:
                continue
            if A[0] == 0 and not (bn or b1):
                continue
            if need_alt_check:
                if not (f or (bn == 0 and b1 == 0)):
                    continue
            ans = True
            break
        if ans:
            break
    print("Yes" if ans else "No")

main()
```

import sys

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    A = [int(x) for x in data[1:1+n]]
    if 0 not in A:
        print("Yes")
        return
    need = (n % 4 == 2)

    # build transition tables
    trans = [[0]*16 for _ in range(2)]  # trans[aj][mask] -> next mask
    for aj in (0, 1):
        for mask in range(16):
            nm = 0
            for s in range(4):
                if not (mask >> s) & 1:
                    continue
                b = s >> 1
                f = s & 1
                for nb in (0, 1):
                    if b == 1 and nb == 1:
                        continue
                    if aj == 0 and b == 0 and nb == 0:
                        continue
                    nf = 1 if (f or (b == 0 and nb == 0)) else 0
                    nm |= 1 << (2*nb + nf)
            trans[aj][mask] = nm

    ans = False
    for b1 in (0, 1):
        mask = 1 << (2*b1)  # f = 0
        ok = True
        for j in range(1, n):
            mask = trans[A[j]][mask]
            if mask == 0:
                ok = False
                break

            continue
        for s in range(4):
            if not (mask >> s) & 1:
                continue
            bn = s >> 1
            f = s & 1
            if bn == 1 and b1 == 1:
                continue
            if A[0] == 0 and bn == 0 and b1 == 0:
                continue
            if need and not (f or (bn == 0 and b1 == 0)):
                continue
            ans = True
            break
        if ans:
            break
    print("Yes" if ans else "No")

import sys

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    A = [int(x) for x in data[1:1 + n]]
    if 0 not in A:
        print("Yes")
        return
    need = (n % 4 == 2)

    # state encoding: s = 2*b + f, where b = current b value, f = seen a "00" pair so far
    trans = [[0] * 16 for _ in range(2)]
    for aj in (0, 1):
        for mask in range(16):
            nm = 0
            for s in range(4):
                if not (mask >> s) & 1:
                    continue
                b = s >> 1
                f = s & 1
                for nb in (0, 1):
                    if b == 1 and nb == 1:
                        continue
                    if aj == 0 and b == 0 and nb == 0:
                        continue
                    nf = 1 if (f == 1 or (b == 0 and nb == 0)) else 0
                    nm |= 1 << (2 * nb + nf)
            trans[aj][mask] = nm

    ans = False
    for b1 in (0, 1):
        mask = 1 << (2 * b1)
        dead = False
        for j in range(1, n):
            mask = trans[A[j]][mask]
            if mask == 0:
                dead = True
                break
        if dead:
            continue
        for s in range(4):
            if not (mask >> s) & 1:
                continue
            bn = s >> 1
            f = s & 1
            if bn == 1 and b1 == 1:
                continue
            if A[0] == 0 and bn == 0 and b1 == 0:
                continue
            if need and not (f == 1 or (bn == 0 and b1 == 0)):
                continue
            ans = True
            break
        if ans:
            break
    print("Yes" if ans else "No")

main()

import sys

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    A = [int(x) for x in data[1:1 + n]]
    if 0 not in A:
        print("Yes")
        return
    need = (n % 4 == 2)

    # DP state s = 2*b + f : b = current start-flag, f = whether a "00" adjacent pair seen
    trans = [[0] * 16 for _ in range(2)]
    for aj in (0, 1):
        for mask in range(16):
            nm = 0
            for s in range(4):
                if not (mask >> s) & 1:
                    continue
                b = s >> 1
                f = s & 1
                for nb in (0, 1):
                    if b == 1 and nb == 1:
                        continue
                    if aj == 0 and b == 0 and nb == 0:
                        continue
                    nf = 1 if (f or (b == 0 and nb == 0)) else 0
                    nm |= 1 << (2 * nb + nf)
            trans[aj][mask] = nm

    ans = False
    for b1 in (0, 1):
        mask = 1 << (2 * b1)
        dead = False
        for j in range(1, n):
            mask = trans[A[j]][mask]
            if mask == 0:
                dead = True
                break
        if dead:
            continue
        for s in range(4):
            if not (mask >> s) & 1:
                continue
            bn = s >> 1
            f = s & 1
            if bn == 1 and b1 == 1:
                continue
            if A[0] == 0 and bn == 0 and b1 == 0:
                continue
            if need and not (f or (bn == 0 and b1 == 0)):
                continue
            ans = True
            break
        if ans:
            break
    print("Yes" if ans else "No")

main()

Let T = set of pattern starts. Then:
- T must be an **independent set in the cycle C_N** (adjacent starts conflict: one forces S_{i+1}=R, the other forces S_{i+1}∈{A,C}).
- Every zero position j needs j ∈ T or j-1 ∈ T (a start covers positions i, i+1).
- Additionally, each i∈T forces S_{i+1}=R and S_i≠S_{i+2} (both in {A,C}). Analyzing the inequality graph (edges {i, i+2}): it's only uncolorable when T contains an entire parity class with N/2 odd, i.e., **N ≡ 2 (mod 4) and T is perfectly alternating** (all evens or all odds). Every other independent T is realizable as a string.
