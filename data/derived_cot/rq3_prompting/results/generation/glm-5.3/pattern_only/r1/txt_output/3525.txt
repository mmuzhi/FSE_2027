```python
class Solution:
    def maxEnergyBoost(self, energyDrinkA: List[int], energyDrinkB: List[int]) -> int:
        a_prev, b_prev = energyDrinkA[0], energyDrinkB[0]
        a_prev2, b_prev2 = 0, 0
        for i in range(1, n):
            a_cur = max(a_prev, b_prev2) + energyDrinkA[i]
            b_cur = max(b_prev, a_prev2) + energyDrinkB[i]
            a_prev2, b_prev2, a_prev, b_prev = a_prev, b_prev, a_cur, b_cur
        return max(a_prev, b_prev)
```

```python
class Solution:
    def maxEnergyBoost(self, energyDrinkA: List[int], energyDrinkB: List[int]) -> int:
        # dpA[i] = best total energy if we drink A at hour i
        # dpB[i] = best total energy if we drink B at hour i
        # Transition: dpA[i] = max(dpA[i-1], dpB[i-2]) + A[i]
        #   (stay on A, or come from B two hours ago with one cleansing hour)
        n = len(energyDrinkA)
        a_prev, b_prev = energyDrinkA[0], energyDrinkB[0]   # dp at i-1
        a_prev2, b_prev2 = 0, 0                             # dp at i-2 (i=1: can't switch)
        for i in range(1, n):
            a_cur = max(a_prev, b_prev2) + energyDrinkA[i]
            b_cur = max(b_prev, a_prev2) + energyDrinkB[i]
            a_prev2, b_prev2, a_prev, b_prev = a_prev, b_prev, a_cur, b_cur
        return max(a_prev, b_prev)
```
