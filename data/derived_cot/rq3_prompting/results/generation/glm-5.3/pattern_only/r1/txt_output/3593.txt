def lcm(a, b):
    return a * b // gcd(a, b)

    def maxScore(self, nums: List[int]) -> int:

        # prefix[i] = (gcd, lcm) of nums[0..i-1]
        preg = [0] * (n + 1)
        prel = [1] * (n + 1)

        sufl = [1] * (n + 1)

=== ANSWER ===
```python
from math import gcd
from typing import List

class Solution:
    def maxScore(self, nums: List[int]) -> int:
        n = len(nums)
        # Prefix GCD/LCM: preg[i], prel[i] cover nums[0..i-1]
        preg = [0] * (n + 1)
        prel = [1] * (n + 1)
        for i in range(n):
            preg[i + 1] = gcd(preg[i], nums[i])
            prel[i + 1] = prel[i] * nums[i] // gcd(prel[i], nums[i])
        # Suffix GCD/LCM: sufg[i], sufl[i] cover nums[i..n-1]
        sufg = [0] * (n + 2)
        sufl = [1] * (n + 2)
        for i in range(n - 1, -1, -1):
            sufg[i] = gcd(sufg[i + 1], nums[i])
            sufl[i] = sufl[i + 1] * nums[i] // gcd(sufl[i + 1], nums[i])
        # Score without removal
        best = preg[n] * prel[n]
        # Try removing each element i
        for i in range(n):
            g = gcd(preg[i], sufg[i + 1])
            l = prel[i] * sufl[i + 1] // gcd(prel[i], sufl[i + 1])
            best = max(best, g * l)
        return best
```
