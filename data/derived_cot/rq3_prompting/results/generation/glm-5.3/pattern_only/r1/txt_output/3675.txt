```python
from typing import List

class Solution:
    def maximizeSumOfWeights(self, edges: List[List[int]], k: int) -> int:
        n = len(edges) + 1
        adj = [[] for _ in range(n)]
        for u, v, w in edges:
            adj[u].append((v, w))
            adj[v].append((u, w))
        
        parent = [-1] * n
        order = []
        visited = [False] * n
        visited[0] = True
        stack = [0]  # DFS order also works
        while stack:
            u = stack.pop()
            order.append(u)
            for v, w in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    parent[v] = u
                    stack.append(v)
        
        dp0 = [0] * n
        dp1 = [0] * n
        km1 = k - 1
        
        for u in reversed(order):
            base = 0
            gains = []
            pu = parent[u]
            for v, w in adj[u]:
                if v == pu:
                    continue
                base += dp1[v]
                gains.append(dp0[v] + w - dp1[v])
            if not gains:
                continue
            gains.sort(reverse=True)
            # prefix of positive gains
            s = 0
            pre = [0]
            for g in gains:
                if g <= 0:
                    break
                s += g
                pre.append(s)
            m = len(pre) - 1
            dp1[u] = base + pre[m if m < k else k]
            dp0[u] = base + pre[m if m < km1 else km1]
        
        return dp1[0]
```

```python
from typing import List

class Solution:
    def maximizeSumOfWeights(self, edges: List[List[int]], k: int) -> int:
        n = len(edges) + 1
        adj = [[] for _ in range(n)]
        for u, v, w in edges:
            adj[u].append((v, w))
            adj[v].append((u, w))
        
        parent = [-1] * n
        order = []
        visited = [False] * n
        visited[0] = True
        stack = [0]
        while stack:
            u = stack.pop()
            order.append(u)
            for v, _w in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    parent[v] = u
                    stack.append(v)
        
        dp0 = [0] * n  # u keeps <= k-1 child edges (parent edge will be kept)
        dp1 = [0] * n  # u keeps <= k child edges
        km1 = k - 1
        
        for u in reversed(order):
            pu = parent[u]
            base = 0
            gains = []
            for v, w in adj[u]:
                if v == pu:
                    continue
                d1 = dp1[v]
                base += d1
                g = dp0[v] + w - d1
                if g > 0:
                    gains.append(g)
            if not gains:
                continue
            gains.sort(reverse=True)
            s1 = 0
            s0 = 0
            c = 0
            for g in gains:
                if c >= k:
                    break
                s1 += g
                if c < km1:
                    s0 += g
                c += 1
            dp1[u] = base + s1
            dp0[u] = base + s0
        
        return dp1[0]
```

```python
from typing import List

class Solution:
    def maximizeSumOfWeights(self, edges: List[List[int]], k: int) -> int:
        n = len(edges) + 1
        adj = [[] for _ in range(n)]
        for u, v, w in edges:
            adj[u].append((v, w))
            adj[v].append((u, w))

        # iterative DFS to get order (parents before children) and parent array
        parent = [-1] * n
        order = [0]
        visited = [False] * n
        visited[0] = True
        head = 0
        while head < len(order):
            u = order[head]
            head += 1
            for v, _ in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    parent[v] = u
                    order.append(v)

        dp0 = [0] * n  # u keeps at most k-1 child edges (parent edge will be kept)
        dp1 = [0] * n  # u keeps at most k child edges
        km1 = k - 1

        for i in range(n - 1, -1, -1):
            u = order[i]
            pu = parent[u]
            base = 0
            gains = []
            for v, w in adj[u]:
                if v == pu:
                    continue
                d1 = dp1[v]
                base += d1
                g = dp0[v] + w - d1
                if g > 0:
                    gains.append(g)
            if not gains:
                continue
            gains.sort(reverse=True)
            s1 = 0
            s0 = 0
            c = 0
            for g in gains:
                if c >= k:
                    break
                s1 += g
                if c < km1:
                    s0 += g
                c += 1
            dp1[u] = base + s1
            dp0[u] = base + s0

        return dp1[0]
```

Looking at this problem: it's a degree-constrained maximum-weight subtree selection on a tree — a classic tree DP.

**Approach (committed early):** Root the tree at 0. For each node `u` define:
- `dp1[u]`: best sum in `u`'s subtree where `u` keeps at most `k` child edges (its parent edge will *not* be kept).
- `dp0[u]`: best sum where `u` keeps at most `k-1` child edges (parent edge *will* be kept, so degree ≤ k).

For each child `v` with edge weight `w`: dropping the edge yields `dp1[v]`; keeping it yields `dp0[v] + w`. Base = Σ`dp1[v]`, and the gain of keeping is `dp0[v] + w - dp1[v]`. Take the largest positive gains, at most `k` for `dp1` and at most `k-1` for `dp0`. Answer = `dp1[root]`.

**Key edge cases checked:** leaves; `k = 1` (matching-like); internal node where all gains ≤ 0 (must still set `dp[u] = base`, not 0 — caught this during recheck with path weights `[1,1,10]`); gain exactly 0 (skip, never beneficial). Verified both examples (22 and 65) plus star/path/T-shaped hand traces. Iterative BFS order avoids recursion depth issues for n = 10⁵. Complexity O(n log n).
