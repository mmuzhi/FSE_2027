```python
class Solution:
    def countValidSelections(self, nums: List[int]) -> int:
        total = sum(nums)
        left = 0
        count = 0
        for i, x in enumerate(nums):
            if x == 0:
                right = total - left
                d = abs(left - right)
                if d == 0:
                    count += 2
                elif d == 1:
                    count += 1
            left += x
        return count
```

```python
class Solution:
    def countValidSelections(self, nums: List[int]) -> int:
        # Key invariant: starting at a zero position i, the pointer ping-pongs between
        # the nearest positive on each side, decrementing alternately (one per hit),
        # until one side's sum is exhausted; it then sweeps out the other side.
        # Moving left is valid iff leftSum == rightSum or leftSum == rightSum + 1.
        # Moving right is valid iff rightSum == leftSum or rightSum == leftSum + 1.
        # => diff 0 contributes 2, diff 1 contributes 1.
        total = sum(nums)
        left = 0
        count = 0
        for x in nums:
            if x == 0:
                right = total - left
                d = abs(left - right)
                if d == 0:
                    count += 2
                elif d == 1:
                    count += 1
            left += x
        return count
```
