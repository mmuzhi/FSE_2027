You will be given a competitive programming problem.
An election is being held with N candidates numbered 1, 2, \ldots, N. There are K votes, some of which have been counted so far.
Up until now, candidate i has received A_i votes.
After all ballots are counted, candidate i (1 \leq i \leq N) will be elected if and only if the number of candidates who have received more votes than them is less than M.  There may be multiple candidates elected.
For each candidate, find the minimum number of additional votes they need from the remaining ballots to guarantee their victory regardless of how the other candidates receive votes.
Formally, solve the following problem for each i = 1,2,\ldots,N.
Determine if there is a non-negative integer X not exceeding K - \displaystyle{\sum_{i=1}^{N}} A_i satisfying the following condition.  If it exists, find the minimum possible such integer.

- If candidate i receives X additional votes, then candidate i will always be elected.

Input

The input is given from Standard Input in the following format:
N M K
A_1 A_2 \ldots A_N

Output

Let C_i be the minimum number of additional votes candidate i needs from the remaining ballots to guarantee their victory regardless of how other candidates receive votes. Print C_1, C_2, \ldots, C_N separated by spaces.
If candidate i has already secured their victory, then let C_i = 0. If candidate i cannot secure their victory under any circumstances, then let C_i = -1.

Constraints


- 1 \leq M \leq N \leq 2 \times 10^5
- 1 \leq K \leq 10^{12}
- 0 \leq A_i \leq 10^{12}
- \displaystyle{\sum_{i=1}^{N} A_i} \leq K
- All input values are integers.

Sample Input 1

5 2 16
3 1 4 1 5

Sample Output 1

2 -1 1 -1 0

14 votes have been counted so far, and 2 votes are left.
The C to output is (2, -1, 1, -1, 0).  For example:

- Candidate 1 can secure their victory by obtaining 2 more votes, while not by obtaining 1 more vote.  Thus, C_1 = 2.
- Candidate 2 can never (even if they obtain 2 more votes) secure their victory, so C_2 = -1.

```python
import sys
from bisect import bisect_left

def solve():
    data = sys.stdin.buffer.read().split()
    N, M, K = int(data[0]), int(data[1]), int(data[2])
    A = [int(x) for x in data[3:3+N]]
    R = K - sum(A)
    
    if M == N:
        print(' '.join(['0']*N))
        return
    
    # sorted order with tie-break by index
    order = sorted(range(N), key=lambda i: (-A[i], i))
    pos = [0]*N
    for r, i in enumerate(order):
        pos[i] = r
    Bdesc = [A[i] for i in order]  # descending
    
    ans = [0]*N
    
    # Build helper for a value multiset given descending list vals (take first cnt values)
    # returns (u asc list, Q prefix, psi list, mu-agnostic) — actually psi depends on mu, build separately.
    
    def build(vals_desc):
        u = sorted(vals_desc)
        m = len(u)
        Q = [0]*(m+1)
        for j in range(m):
            Q[j+1] = Q[j] + u[j]
        return u, Q
    
    def make_psi(u, Q, mu):
        m = len(u)
        psi = []
        for j in range(m):
            # psi[j] = (j+mu)*(u[j]-1) + j - Q[j]
            psi.append((j+mu)*(u[j]-1) + j - Q[j])
        return psi
    
    def query_min_S(u, Q, psi, mu, W):
        # minimal S with Phi(S) >= W, Phi(S) = F(S) + mu*S
        idx = bisect_left(psi, W)
        m = len(u)
        j = idx
        denom = j + mu
        num = W - j + Q[j]
        # ceildiv
        S = -(-num // denom) if denom > 0 else None
        if j >= 1:
            lo = u[j-1]
            if S < lo:
                S = lo
        return S
    
    # Case 2: candidates with pos < M. V2 = Bdesc[0..M] (M+1 values), mu=0, W=R+2 constant.
    u2, Q2 = build(Bdesc[:M+1])
    psi2 = make_psi(u2, Q2, 0)
    S0 = query_min_S(u2, Q2, psi2, 0, R+2)
    
    # Case 1: V = Bdesc[0..M-1], mu=1, per-candidate W = R+1+A_i.
    u1, Q1 = build(Bdesc[:M])
    psi1 = make_psi(u1, Q1, 1)
    
    for i in range(N):
        if pos[i] < M:
            X = S0 - A[i]
            if X < 0: X = 0
            ans[i] = X if X <= R else -1
        else:
            W = R + 1 + A[i]
            Smin = query_min_S(u1, Q1, psi1, 1, W)
            X = Smin - A[i]
            if X < 0: X = 0
            ans[i] = X if X <= R else -1
    
    print(' '.join(map(str, ans)))

solve()
```

```python
import sys
from bisect import bisect_left

def main():
    data = sys.stdin.buffer.read().split()
    N = int(data[0]); M = int(data[1]); K = int(data[2])
    A = list(map(int, data[3:3+N]))
    R = K - sum(A)
    
    if M == N:
        sys.stdout.write(' '.join(['0'] * N))
        return
    
    order = sorted(range(N), key=lambda i: (-A[i], i))
    pos = [0] * N
    for r, i in enumerate(order):
        pos[i] = r
    Bdesc = [A[i] for i in order]
    
    def query_min_S(u, Q, psi, mu, W):
        # Phi(S) = (j+mu) S + j - Q[j] on piece j; psi[j] = Phi(u[j]-1), psi[len] = inf
        idx = bisect_left(psi, W)
        j = idx
        denom = j + mu
        num = W - j + Q[j]
        S = -(-num // denom)
        if j >= 1 and S < u[j-1]:
            S = u[j-1]
        return S
    
    # Case 2 setup
    vals2 = sorted(Bdesc[:M+1])
    m2 = M + 1
    Q2 = [0] * (m2 + 1)
    for j in range(m2):
        Q2[j+1] = Q2[j] + vals2[j]
    psi2 = [j * vals2[j] - Q2[j] for j in range(m2)]  # mu=0
    S0 = query_min_S(vals2, Q2, psi2, 0, R + 2)
    
    # Case 1 setup
    vals1 = sorted(Bdesc[:M])
    m1 = M
    Q1 = [0] * (m1 + 1)
    for j in range(m1):
        Q1[j+1] = Q1[j] + vals1[j]
    psi1 = [(j + 1) * (vals1[j] - 1) + j - Q1[j] for j in range(m1)]  # mu=1
    
    out = []
    for i in range(N):
        if pos[i] < M:
            X = S0 - A[i]
        else:
            W = R + 1 + A[i]
            idx = bisect_left(psi1, W)
            num = W - idx + Q1[idx]
            S = -(-num // (idx + 1))
            if idx >= 1 and S < vals1[idx-1]:
                S = vals1[idx-1]
            X = S - A[i]
        if X < 0:
            X = 0
        out.append(str(X) if X <= R else '-1')
    
    sys.stdout.write(' '.join(out))

main()
```

Looking at this problem:

- Candidate i with final score S = A_i + X is guaranteed elected iff the adversary (having R−X votes for others) cannot push M others strictly above S. The cheapest M rivals to push are simply the M largest A_j among j≠i, giving cost F(S) = Σ max(0, S+1−v) over those M values. Condition: F(A_i+X) + X ≥ R+1.

- Sort descending with tie-break. If candidate i is inside the top M (pos < M), its rival set = top M+1 values minus A_i, and the condition collapses to F₂(A_i+X) ≥ R+2 where F₂ is over the fixed top M+1 values (single threshold S₀ for all such candidates). Otherwise rival set = top M values, condition Φ(S) = F₁(S)+S ≥ R+1+A_i.

- Φ is non-decreasing piecewise linear; precompute prefix sums and ψ_j = Φ(u_j − 1) (non-decreasing), then each query is one bisect + O(1) ceil computation, giving minimal S, and X = max(0, S−A_i), −1 if X > R. Verified against both samples and several hand brute-forced cases (including boundary ties).

```python
import sys
from bisect import bisect_left
from itertools import accumulate

def main():
    data = sys.stdin.buffer.read().split()
    N = int(data[0]); M = int(data[1]); K = int(data[2])
    A = list(map(int, data[3:3 + N]))
    R = K - sum(A)

    if M >= N:
        print(' '.join(['0'] * N))
        return

    order = sorted(range(N), key=lambda i: (-A[i], i))
    pos = [0] * N
    for r, i in enumerate(order):
        pos[i] = r
    B = [A[i] for i in order]  # values, descending

    def min_S(u, Q, psi, mu, W):
        # Phi(S) = F(S) + mu*S ; on piece j: (j+mu)*S + j - Q[j] ; psi[j] = Phi(u[j]-1)
        j = bisect_left(psi, W)
        S = (W + Q[j] + mu - 1) // (j + mu)
        if j and S < u[j - 1]:
            S = u[j - 1]
        return S

    # Candidates inside the top M (pos < M):
    # rival set = top(M+1) minus own value -> condition F2(A_i+X) >= R+2
    u2 = sorted(B[:M + 1])
    Q2 = [0] + list(accumulate(u2))
    psi2 = [j * u2[j] - Q2[j] for j in range(M + 1)]          # mu = 0
    S0 = min_S(u2, Q2, psi2, 0, R + 2)

    # Candidates outside the top M (pos >= M):
    # rival set = top M -> condition F1(A_i+X) + X >= R+1
    u1 = sorted(B[:M])
    Q1 = [0] + list(accumulate(u1))
    psi1 = [(j + 1) * u1[j] - 1 - Q1[j] for j in range(M)]    # mu = 1

    res = []
    for i in range(N):
        if pos[i] < M:
            X = S0 - A[i]
        else:
            X = min_S(u1, Q1, psi1, 1, R + 1 + A[i]) - A[i]
        if X < 0:
            X = 0
        res.append(str(X) if X <= R else '-1')

    sys.stdout.write(' '.join(res) + '\n')

main()
```
