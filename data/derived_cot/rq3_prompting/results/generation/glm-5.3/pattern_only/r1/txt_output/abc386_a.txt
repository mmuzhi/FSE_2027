from collections import Counter

def is_full_house(cards):
    c = Counter(cards)
    if len(c) != 2:
        return False
    return sorted(c.values()) == [2, 3]

def main():
    A, B, C, D = map(int, input().split())
    cards = [A, B, C, D]
    for v in range(1, 14):
        if is_full_house(cards + [v]):
            print("Yes")
            return
    print("No")

main()
```
