import sys

def main():
    data = sys.stdin.buffer.read().split()
    S = data[0].decode()
    Q = int(data[1])
    Ks = data[2:2+Q]
    n = len(S)
    orig = list(S)
    swap = [c.swapcase() for c in S]
    out = []
    for kb in Ks:
        K = int(kb)
        i = K - 1
        q, r = divmod(i, n)
        if bin(q).count('1') & 1:
            out.append(swap[r])
        else:
            out.append(orig[r])
    sys.stdout.write(' '.join(out))
