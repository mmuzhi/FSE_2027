    def maxSubstringLength(self, s: str, k: int) -> bool:
        n = len(s)

        letters = list(occ)

        # expand [first,last] until every character occurring inside is fully contained.
        intervals = set()
        for c in letters:

            if l != 0 or r != n - 1:
                intervals.add((l, r))
