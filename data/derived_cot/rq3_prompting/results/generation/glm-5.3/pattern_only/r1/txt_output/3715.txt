from typing import List
import bisect

class Solution:
    def maximumCoins(self, coins: List[List[int]], k: int) -> int:
        coins.sort()
        n = len(coins)
        
        def solve(L, R, C):
            # windows [L[i], L[i]+k-1]
            pre = [0] * (n + 1)
            for i in range(n):
                pre[i + 1] = pre[i] + C[i] * (R[i] - L[i] + 1)
            best = 0
            j = 0
            for i in range(n):
                rend = L[i] + k - 1
                if j < i:
                    j = i
                while j < n and R[j] <= rend:
                    j += 1
                total = pre[j] - pre[i]
                if j < n and L[j] <= rend:
                    total += C[j] * (rend - L[j] + 1)
                if total > best:
                    best = total
            return best
        
        L = [c[0] for c in coins]
        R = [c[1] for c in coins]
        C = [c[2] for c in coins]
        ans = solve(L, R, C)
        # reverse: x -> -x
        L2 = [-x for x in reversed(R)]
        R2 = [-x for x in reversed(L)]
        ans = max(ans, solve(L2, R2, C[::-1]))
        return ans

```python
from typing import List

class Solution:
    def maximumCoins(self, coins: List[List[int]], k: int) -> int:
        coins.sort()
        n = len(coins)
        
        def best_aligned(L, R, C):
            # max over i of coins in window [L[i], L[i] + k - 1]
            pre = [0] * (n + 1)
            for i in range(n):
                pre[i + 1] = pre[i] + C[i] * (R[i] - L[i] + 1)
            best = 0
            j = 0
            for i in range(n):
                rend = L[i] + k - 1
                if j < i:
                    j = i
                while j < n and R[j] <= rend:
                    j += 1
                total = pre[j] - pre[i]
                if j < n and L[j] <= rend:
                    total += C[j] * (rend - L[j] + 1)
                if total > best:
                    best = total
            return best
        
        L = [c[0] for c in coins]
        R = [c[1] for c in coins]
        C = [c[2] for c in coins]
        
        ans = best_aligned(L, R, C)
        # mirror: x -> -x turns "window ending at R[i]" into "window starting at -R[i]"
        L2 = [-x for x in reversed(R)]
        R2 = [-x for x in reversed(L)]
        C2 = C[::-1]
        ans = max(ans, best_aligned(L2, R2, C2))
        return ans
```

```python
from typing import List

class Solution:
    def maximumCoins(self, coins: List[List[int]], k: int) -> int:
        coins.sort()
        n = len(coins)
        
        def best_aligned(L, R, C):
            # max over i of coins collected in window [L[i], L[i] + k - 1]
            pre = [0] * (n + 1)
            for i in range(n):
                pre[i + 1] = pre[i] + C[i] * (R[i] - L[i] + 1)
            best = 0
            j = 0
            for i in range(n):
                rend = L[i] + k - 1
                if j < i:
                    j = i
                while j < n and R[j] <= rend:
                    j += 1
                total = pre[j] - pre[i]
                if j < n and L[j] <= rend:
                    total += C[j] * (rend - L[j] + 1)
                if total > best:
                    best = total
            return best
        
        L = [seg[0] for seg in coins]
        R = [seg[1] for seg in coins]
        C = [seg[2] for seg in coins]
        
        ans = best_aligned(L, R, C)
        # Mirror the line (x -> -x): windows ending at R[i] become windows starting at -R[i]
        L2 = [-x for x in reversed(R)]
        R2 = [-x for x in reversed(L)]
        C2 = C[::-1]
        return max(ans, best_aligned(L2, R2, C2))
```

Looking at this problem:

**Key insight:** The window sum `f(s)` (coins in `[s, s+k-1]`) is piecewise linear in `s`, with slope `coins(s+k) - coins(s)`. Its breakpoints occur where the window's start hits a segment start, or the window's end hits a segment end. Any local-max breakpoint is either start-anchored (window starts at some `l_i`) or end-anchored (window ends at some `r_i`) — the other breakpoint types (`s = r_i+1`, `s = l_i-k`) can only be local maxima when they coincide with an anchored one (e.g., `s = r_i+1` being a max forces `r_i+k = r_j`, making the window end at `r_j`).

**Plan:** Sort segments. Pass 1: for each `i`, compute coins in `[l_i, l_i+k-1]` via prefix sums + two pointers (segments `i..j-1` fully inside, segment `j` partially). Pass 2: mirror the line (`x → -x`) so windows ending at `r_i` become windows starting at `-r_i`, and rerun the same routine. Answer is the max of both passes.

**Verified** against both examples (10 and 6), plus brute-force checks on small multi-segment cases (e.g., `[[2,3,4],[6,8,2],[10,10,7]], k=3 → 9` via window `[8,10]`, found by the end-anchored pass; `[[1,4,3],[6,7,5],[9,12,2]], k=6 → 19` via window `[2,7]`).

Edge cases handled: segment longer than `k` (pointer `j` resets to `i`, partial gives `c_i*k`), gap segments after window end (`L[j] > rend` check), monotonic pointer across iterations.
