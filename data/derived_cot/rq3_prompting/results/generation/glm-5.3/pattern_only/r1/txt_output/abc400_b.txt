def main():
    import sys
    N, M = map(int, sys.stdin.read().split())
    LIMIT = 10**9
    X = 1
    power = 1
    for _ in range(M):
        power *= N
        X += power
        if X > LIMIT:
            print("inf")
            return
    print(X)

main()
```
