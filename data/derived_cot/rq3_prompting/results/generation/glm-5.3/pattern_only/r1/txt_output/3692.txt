The empty substring is the shortest matching substring.

Explanation:
The shortest matching substring of p in s is "adlogi".

Actually, alternative clean approach avoiding explicit occurrence lists: compute, for each part, arrays using Z-function or KMP prefix function on s+sep+part. Let me just write a helper:

def all_occurrences(s, pat):
    # returns sorted list of start indices of pat in s (overlapping)
    if not pat: return None (special)
    # KMP
    m = len(pat)
    # build failure function
    fail = [0]*m
    k = 0
    for i in range(1, m):
        while k and pat[i] != pat[k]: k = fail[k-1]
        if pat[i] == pat[k]: k += 1
        fail[i] = k
    res = []
    k = 0
    for i, ch in enumerate(s):
        while k and ch != pat[k]: k = fail[k-1]
        if ch == pat[k]: k += 1
        if k == m:
            res.append(i - m + 1)
            k = fail[k-1]
    return res

```python
class Solution:
    def shortestMatchingSubstring(self, s: str, p: str) -> int:
        p1, p2, p3 = p.split('*')
        n = len(s)
        INF = float('inf')

        def kmp_starts(pat):
            # all (overlapping) start indices where pat occurs in s
            m = len(pat)
            fail = [0] * m
            k = 0
            for i in range(1, m):
                c = pat[i]
                while k and c != pat[k]:
                    k = fail[k - 1]
                if c == pat[k]:
                    k += 1
                fail[i] = k
            res = []
            k = 0
            for i in range(n):
                c = s[i]
                while k and c != pat[k]:
                    k = fail[k - 1]
                if c == pat[k]:
                    k += 1
                    if k == m:
                        res.append(i - m + 1)
                        k = fail[k - 1]
            return res

        # nxt3[i] = earliest start >= i of an occurrence of p3 in s (i in 0..n)
        if p3:
            nxt3 = [INF] * (n + 2)
            for st in kmp_starts(p3):
                nxt3[st] = st
            for i in range(n - 1, -1, -1):
                if nxt3[i + 1] < nxt3[i]:
                    nxt3[i] = nxt3[i + 1]
        else:
            nxt3 = list(range(n + 1))
            nxt3.append(INF)

        # G[e] = minimal end (exclusive) of a "p2 * p3" match whose p2 starts at index >= e
        G = [INF] * (n + 2)
        if p2:
            L2 = len(p2)
            add = len(p3)
            for st in kmp_starts(p2):
                G[st] = nxt3[st + L2] + add
        else:
            add = len(p3)
            for st in range(n + 1):
                G[st] = nxt3[st] + add
        for i in range(n, -1, -1):
            if G[i + 1] < G[i]:
                G[i] = G[i + 1]

        best = INF
        if p1:
            L1 = len(p1)
            for st in kmp_starts(p1):
                v = G[st + L1]
                if v is not INF and v - st < best:
                    best = v - st
        else:
            for e in range(n + 1):
                v = G[e]
                if v is not INF and v - e < best:
                    best = v - e
        return -1 if best == INF else best
```

```python
class Solution:
    def shortestMatchingSubstring(self, s: str, p: str) -> int:
        p1, p2, p3 = p.split('*')
        n = len(s)
        INF = float('inf')

        def kmp_starts(pat):
            # all (overlapping) start indices of occurrences of pat in s
            m = len(pat)
            fail = [0] * m
            k = 0
            for i in range(1, m):
                c = pat[i]
                while k and c != pat[k]:
                    k = fail[k - 1]
                if c == pat[k]:
                    k += 1
                fail[i] = k
            res = []
            k = 0
            for i, c in enumerate(s):
                while k and c != pat[k]:
                    k = fail[k - 1]
                if c == pat[k]:
                    k += 1
                    if k == m:
                        res.append(i - m + 1)
                        k = fail[k - 1]
            return res

        # nxt3[i] = earliest start (>= i) of an occurrence of p3; i in 0..n
        if p3:
            nxt3 = [INF] * (n + 2)
            for st in kmp_starts(p3):
                nxt3[st] = st
            for i in range(n - 1, -1, -1):
                if nxt3[i + 1] < nxt3[i]:
                    nxt3[i] = nxt3[i + 1]
        else:
            nxt3 = list(range(n + 1))
            nxt3.append(INF)

        # G[e] = min end (exclusive) of "p2 ... p3" with p2 starting at index >= e
        add3 = len(p3)
        G = [INF] * (n + 2)
        if p2:
            L2 = len(p2)
            for st in kmp_starts(p2):
                G[st] = nxt3[st + L2] + add3
        else:
            for st in range(n + 1):
                G[st] = nxt3[st] + add3
        for i in range(n, -1, -1):
            if G[i + 1] < G[i]:
                G[i] = G[i + 1]

        if p1:
            L1 = len(p1)
            best = INF
            for st in kmp_starts(p1):
                cand = G[st + L1] - st
                if cand < best:
                    best = cand
        else:
            best = INF
            for e in range(n + 1):
                cand = G[e] - e
                if cand < best:
                    best = cand
        return -1 if best == INF else best
```

```python
class Solution:
    def shortestMatchingSubstring(self, s: str, p: str) -> int:
        p1, p2, p3 = p.split('*')
        n = len(s)
        INF = float('inf')

        def kmp_starts(pat):
            m = len(pat)
            if m > n:
                return []
            fail = [0] * m
            k = 0
            for i in range(1, m):
                c = pat[i]
                while k and c != pat[k]:
                    k = fail[k - 1]
                if c == pat[k]:
                    k += 1
                fail[i] = k
            res = []
            k = 0
            for i, c in enumerate(s):
                while k and c != pat[k]:
                    k = fail[k - 1]
                if c == pat[k]:
                    k += 1
                    if k == m:
                        res.append(i - m + 1)
                        k = fail[k - 1]
            return res

        # nxt3[i]: earliest start >= i of an occurrence of p3 in s (i in 0..n)
        if p3:
            nxt3 = [INF] * (n + 2)
            for st in kmp_starts(p3):
                nxt3[st] = st
            for i in range(n - 1, -1, -1):
                if nxt3[i + 1] < nxt3[i]:
                    nxt3[i] = nxt3[i + 1]
        else:
            nxt3 = list(range(n + 1))
            nxt3.append(INF)

        # G[e]: minimal end (exclusive) of a match of "p2 * p3" whose p2 starts at index >= e
        add3 = len(p3)
        G = [INF] * (n + 2)
        if p2:
            L2 = len(p2)
            for st in kmp_starts(p2):
                G[st] = nxt3[st + L2] + add3
        else:
            for st in range(n + 1):
                G[st] = nxt3[st] + add3
        for i in range(n, -1, -1):
            if G[i + 1] < G[i]:
                G[i] = G[i + 1]

        if p1:
            L1 = len(p1)
            best = INF
            for st in kmp_starts(p1):
                cand = G[st + L1] - st
                if cand < best:
                    best = cand
        else:
            best = INF
            for e in range(n + 1):
                cand = G[e] - e
                if cand < best:
                    best = cand
        return -1 if best == INF else best
```

```python
class Solution:
    def shortestMatchingSubstring(self, s: str, p: str) -> int:
        p1, p2, p3 = p.split('*')
        n = len(s)
        INF = float('inf')

        def kmp_starts(pat):
            # all (possibly overlapping) start indices of pat in s
            m = len(pat)
            if m > n:
                return []
            fail = [0] * m
            k = 0
            for i in range(1, m):
                c = pat[i]
                while k and c != pat[k]:
                    k = fail[k - 1]
                if c == pat[k]:
                    k += 1
                fail[i] = k
            res = []
            k = 0
            for i, c in enumerate(s):
                while k and c != pat[k]:
                    k = fail[k - 1]
                if c == pat[k]:
                    k += 1
                    if k == m:
                        res.append(i - m + 1)
                        k = fail[k - 1]
            return res

        # nxt3[i] = earliest start >= i of an occurrence of p3 (i in 0..n); INF if none
        if p3:
            nxt3 = [INF] * (n + 2)
            for st in kmp_starts(p3):
                nxt3[st] = st
            for i in range(n - 1, -1, -1):
                if nxt3[i + 1] < nxt3[i]:
                    nxt3[i] = nxt3[i + 1]
        else:
            nxt3 = list(range(n + 1))
            nxt3.append(INF)

        # G[e] = minimal end (exclusive) of "p2 ... p3" with p2 starting at index >= e
        add3 = len(p3)
        G = [INF] * (n + 2)
        if p2:
            L2 = len(p2)

        else:

                cand = G[st + L1] - st

                    best = cand
```
