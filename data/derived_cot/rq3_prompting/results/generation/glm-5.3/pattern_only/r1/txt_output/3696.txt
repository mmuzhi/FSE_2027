```python
class Solution:
    def countSubstrings(self, s: str) -> int:
        ans = 0
        # counters of prefix residues W_{-1..j-1}; W_{-1} = 0 counted once
        cnt3 = [0] * 3; cnt3[0] = 1   # 10 ≡ 1 (mod 3): value ≡ digit sum
        cnt9 = [0] * 9; cnt9[0] = 1   # 10 ≡ 1 (mod 9): value ≡ digit sum
        cnt7 = [0] * 7; cnt7[0] = 1   # use 10^{-k} weighted prefix sums
        p3 = p9 = w7 = 0
        ip = 1  # 10^{-j} mod 7  (== 5^j mod 7)
        prev1 = prev2 = -1  # previous digits for d=4 and d=8 local checks
        for j, ch in enumerate(s):
            c = ord(ch) - 48
            p3 = (p3 + c) % 3
            p9 = (p9 + c) % 9
            w7 = (w7 + c * ip) % 7
            ip = ip * 5 % 7
            if c:
                if c == 1 or c == 2 or c == 5:
                    # divisible by 1 always; ends in 2 -> even; ends in 5 -> div by 5
                    ans += j + 1
                elif c == 3 or c == 6:
                    # mod 3: digit sum; for 6 evenness is automatic (last digit 6)
                    ans += cnt3[p3]
                elif c == 9:
                    ans += cnt9[p9]
                elif c == 7:
                    ans += cnt7[w7]
                elif c == 4:
                    ans += 1  # "4" itself
                    # len>=2: value ≡ 10*prev+4 ≡ 2*prev (mod 4) -> prev even
                    if prev1 >= 0 and prev1 % 2 == 0:
                        ans += j
                else:  # c == 8
                    ans += 1  # "8" itself
                    # len 2: (10*prev+8) ≡ 2*prev (mod 8) -> prev ≡ 0 (mod 4)
                    if prev1 >= 0 and prev1 % 4 == 0:
                        ans += 1
                    # len>=3: (4*a + 2*b) ≡ 0 (mod 8) -> 2*a + b ≡ 0 (mod 4)
                    if prev2 >= 0 and (2 * prev2 + prev1) % 4 == 0:
                        ans += j - 1
            # record prefix residue W_j for future endpoints
            cnt3[p3] += 1
            cnt9[p9] += 1
            cnt7[w7] += 1
            prev2 = prev1
            prev1 = c
        return ans
```
