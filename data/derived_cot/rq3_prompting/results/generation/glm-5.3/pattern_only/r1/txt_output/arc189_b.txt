def main():
    data = sys.stdin.read().split()
    n = int(data[0])
    x = list(map(int, data[1:1+n]))
    gaps = [x[i+1] - x[i] for i in range(n-1)]
    odd = sorted(gaps[0::2])   # indices 0,2,4... (1-based odd)
    even = sorted(gaps[1::2])  # indices 1,3,5... (1-based even)
    # rebuild
    # coefficient of gap j (1-based) is n - j
    total = n * x[0]
    # place odd gaps at 1-based indices 1,3,...: coefficients n-1, n-3, ...
    # place even at 2,4,...: coefficients n-2, n-4,...
    coefs = list(range(n-1, 0, -1))  # coefficient for 1-based gap index j is n-j
    res = []
    io = 0
    ie = 0
    for j in range(1, n):
        if j % 2 == 1:
            res.append((n - j) * odd[io]); io += 1
        else:
            res.append((n - j) * even[ie]); ie += 1
    print(total + sum(res))

main()
