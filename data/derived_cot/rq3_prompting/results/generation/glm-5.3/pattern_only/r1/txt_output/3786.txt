    def longestPalindromicSubsequence(self, s: str, k: int) -> int:
        n = len(s)
        K = k + 1
        zero = [0] * K
        # length 1 layer
        prev1 = [[1] * K for _ in range(n)]
        # length 0 layer (for L=2 interiors): need index up to n
        prev2 = [zero] * (n + 1)
        for L in range(2, n + 1):
            cur = []
            for i in range(n - L + 1):
                j = i + L - 1
                m1 = prev1[i + 1]
                m2 = prev1[i]
                base = [x if x > y else y for x, y in zip(m1, m2)]
                d = ord(s[i]) - ord(s[j])
                if d < 0:
                    d = -d
                c = d if d < 26 - d else 26 - d
                inn = prev2[i + 1]
                if c:
                    cand = [0] * c
                    cand.extend([v + 2 for v in inn[:K - c]])
                else:
                    cand = [v + 2 for v in inn]
                base = [y if y > x else x for x, y in zip(base, cand)]
                cur.append(base)
            prev2 = prev1
            prev1 = cur
        return prev1[0][k]

class Solution:
    def longestPalindromicSubsequence(self, s: str, k: int) -> int:
        n = len(s)
        K = k + 1
        add2 = (2).__add__
        zero = [0] * K
        prev1 = [[1] * K for _ in range(n)]      # length 1
        prev2 = [zero] * (n + 1)                  # length 0
        for L in range(2, n + 1):
            cur = []
            for i in range(n - L + 1):
                d = ord(s[i]) - ord(s[j := i + L - 1])
                if d < 0:
                    d = -d
                c = d if d <= 13 else 26 - d
                inn = prev2[i + 1]
                if c:
                    cand = [0] * c
                    cand += list(map(add2, inn[:K - c]))
                else:
                    cand = list(map(add2, inn))
                cur.append(list(map(max, map(max, prev1[i], prev1[i + 1]), cand)))
            prev2 = prev1
            prev1 = cur
        return prev1[0][k]

class Solution:
    def longestPalindromicSubsequence(self, s: str, k: int) -> int:
        n = len(s)
        K = k + 1
        add2 = (2).__add__
        zero = [0] * K
        # prev1[v] = dp array over budgets for the length-(L-1) interval starting at v
        # prev2[v] = dp array over budgets for the length-(L-2) interval starting at v
        prev1 = [[1] * K for _ in range(n)]   # intervals of length 1
        prev2 = [zero] * (n + 1)              # intervals of length 0 (empty)

        for L in range(2, n + 1):
            cur = []
            for i in range(n - L + 1):
                x = ord(s[i])
                y = ord(s[i + L - 1])
                d = x - y if x > y else y - x
                # minimal cyclic cost to make s[i] and s[j] equal
                c = d if d <= 13 else 26 - d
                if c <= k:
                    inn = prev2[i + 1]  # interior interval dp
                    if c:
                        # matching pair affordable only for budgets >= c
                        cand = [0] * c + list(map(add2, inn[:K - c]))
                    else:
                        cand = list(map(add2, inn))
                    # option: match ends (cost c), or skip either end (cost 0)
                    row = list(map(max, map(max, prev1[i], prev1[i + 1]), cand))
                else:
                    row = list(map(max, prev1[i], prev1[i + 1]))
                cur.append(row)
            prev2 = prev1
            prev1 = cur

        return prev1[0][k]
