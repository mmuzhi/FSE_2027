import sys

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    xs = data[1:1+2*n:2]
    hs = data[2:2+2*n:2]
    X = list(map(int, xs))
    H = list(map(int, hs))
    # process right to left
    stack = []  # indices, stack[0] rightmost
    best_num = None  # numerator
    best_den = 1
    for u in range(n-1, -1, -1):
        xu = X[u]; hu = H[u]
        while len(stack) >= 2:
            v = stack[-1]; w = stack[-2]
            # cross(v-u, w-v) < 0 -> pop
            ax = X[v]-xu; ay = H[v]-hu
            bx = X[w]-X[v]; by = H[w]-H[v]
            if ax*by - ay*bx < 0:
                stack.pop()
            else:
                break
        if stack:
            b = stack[-1]
            num = hu*X[b] - H[b]*xu
            den = X[b]-xu
            if best_num is None or num*best_den > best_num*den:
                best_num = num; best_den = den
        stack.append(u)
    if best_num is None or best_num < 0:
        print(-1)
        return
    # print best_num/best_den with 18 decimals
    ip = best_num // best_den
    rem = best_num - ip*best_den
    digits = []
    for _ in range(18):
        rem *= 10
        d, rem = divmod(rem, best_den)
        digits.append(str(d))
    print(f"{ip}.{''.join(digits)}")

main()

    n = int(data[0])
    X = list(map(int, data[1:1 + 2 * n:2]))
    H = list(map(int, data[2:2 + 2 * n:2]))
    stack = []
    best_num = None
    best_den = 1
    for u in range(n - 1, -1, -1):
        xu = X[u]; hu = H[u]
        while len(stack) >= 2:
            v = stack[-1]; w = stack[-2]
            ax = X[v] - xu; ay = H[v] - hu
            bx = X[w] - X[v]; by = H[w] - H[v]
            if ax * by - ay * bx < 0:
                stack.pop()
            else:
                break
        if stack:
            b = stack[-1]
            num = hu * X[b] - H[b] * xu
            den = X[b] - xu
            if best_num is None or num * best_den > best_num * den:
                best_num = num; best_den = den
        stack.append(u)
    if best_num is None or best_num < 0:
        print(-1)
        return
    ip, rem = divmod(best_num, best_den)
    out = []
    for _ in range(18):
        rem *= 10
        d, rem = divmod(rem, best_den)
        out.append(str(d))
    print(str(ip) + "." + "".join(out))

main()
