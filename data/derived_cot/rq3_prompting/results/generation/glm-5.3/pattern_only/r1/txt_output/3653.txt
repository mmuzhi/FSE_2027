```python
from itertools import accumulate
from typing import List


class Solution:
    def maxSubarraySum(self, nums: List[int], k: int) -> int:
        INF = float('inf')
        min_pref = [INF] * k  # min_pref[r] = min prefix P[i] seen so far with i % k == r
        ans = -INF
        for j, prefix in enumerate(accumulate(nums, initial=0)):
            r = j % k
            cand = prefix - min_pref[r]
            if cand > ans:
                ans = cand
            if prefix < min_pref[r]:
                min_pref[r] = prefix
        return ans
```
