```python
from typing import List
from math import gcd

class Solution:
    def maxLength(self, nums: List[int]) -> int:
        n = len(nums)
        ans = 0
        for i in range(n):
            p = 1   # running product
            g = 0   # running gcd (gcd(0, x) == x)
            l = 1   # running lcm
            for j in range(i, n):
                x = nums[j]
                p *= x
                g = gcd(g, x)
                l = l * x // gcd(l, x)
                if p == l * g:
                    ans = max(ans, j - i + 1)
        return ans
```
