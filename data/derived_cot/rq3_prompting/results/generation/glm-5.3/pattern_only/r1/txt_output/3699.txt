```python
from math import gcd
from typing import List

class Solution:
    def numberOfSubsequences(self, nums: List[int]) -> int:
        n = len(nums)
        # cnt[k] = number of left pairs (p, q) with q <= r-2 (so far) whose
        # reduced fraction nums[p]/nums[q] encodes to integer key k.
        cnt = [0] * (1001 * 1001)
        ans = 0
        _gcd = gcd

        for r in range(n - 2):
            # Before scanning r, newly usable left pairs are those ending at q = r-2
            # (they satisfy q <= r-2, i.e. r - q > 1). Each such pair is inserted exactly once.
            q = r - 2
            if q >= 2:
                vq = nums[q]
                for a in nums[:q - 1]:  # p in [0, q-2] so q - p > 1
                    g = _gcd(a, vq)
                    cnt[(a // g) * 1001 + vq // g] += 1

            # Right pairs (r, s) with s >= r + 2 (s - r > 1).
            # Condition nums[p]*nums[r] == nums[q]*nums[s]  <=>  nums[p]/nums[q] == nums[s]/nums[r],
            # i.e. reduced fraction of (nums[s], nums[r]) must match a left pair's reduced fraction.
            vr = nums[r]
            for b in nums[r + 2:]:
                g = _gcd(b, vr)
                ans += cnt[(b // g) * 1001 + vr // g]

        return ans
```
