```python
class Solution:
    def maxScore(self, grid: List[List[int]]) -> int:
        m, n = len(grid), len(grid[0])
        
        # For each distinct value, collect the rows that contain it.
        # (Multiple occurrences within one row collapse, since at most one
        # cell per row can be selected anyway.)
        value_rows = {}
        for r in range(m):
            for v in set(grid[r]):
                value_rows.setdefault(v, []).append(r)
        
        NEG = float('-inf')
        size = 1 << m
        # dp[mask] = best sum using distinct processed values assigned to
        # exactly the rows in mask (each row used at most once).
        dp = [NEG] * size
        dp[0] = 0
        
        for v, rows in value_rows.items():
            new_dp = dp[:]  # snapshot so each value is used at most once
            for mask in range(size):
                if dp[mask] == NEG:
                    continue
                for r in rows:
                    bit = 1 << r
                    if not (mask & bit):
                        nm = mask | bit
                        cand = dp[mask] + v
                        if cand > new_dp[nm]:
                            new_dp[nm] = cand
            dp = new_dp
        
        # Non-empty selection required -> exclude dp[0]
        return max(dp[1:])
```
