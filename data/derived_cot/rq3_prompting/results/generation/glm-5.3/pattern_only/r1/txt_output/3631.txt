class Solution:
    def countKReducibleNumbers(self, s: str, k: int) -> int:
        MOD = 10**9 + 7
        L = len(s)
        
        # binomials
        # C[m][j] for m up to L
        C = [[0] * (L + 1) for _ in range(L + 1)]
        for m in range(L + 1):
            C[m][0] = 1
            for j in range(1, m + 1):
                C[m][j] = (C[m-1][j-1] + C[m-1][j]) % MOD
        
        # h(c): min ops to reduce c to 1
        from functools import lru_cache
        @lru_cache(None)
        def h(c):
            if c == 1:
                return 0
            return 1 + h(bin(c).count('1'))
        
        ans = 0
        ones = 0  # ones in s[0..i-1]
        for i, ch in enumerate(s):
            if ch == '1':
                free = L - 1 - i
                # numbers matching prefix, 0 here, free suffix
                # popcount = ones + j, j in [0, free]
                base_c = ones
                for j in range(free + 1):
                    c = base_c + j
                    if c >= 1 and 1 + h(c) <= k:
                        ans = (ans + C[free][j]) % MOD
            ones += 1 if ch == '1' else 0
        
        return ans % MOD

```python
class Solution:
    def countKReducibleNumbers(self, s: str, k: int) -> int:
        MOD = 10**9 + 7
        L = len(s)
        
        # Binomial coefficients mod MOD via Pascal's triangle
        C = [[0] * (L + 1) for _ in range(L + 1)]
        for m in range(L + 1):
            C[m][0] = 1
            row = C[m]
            prev = C[m - 1] if m > 0 else None
            for j in range(1, m + 1):
                row[j] = (prev[j - 1] + prev[j]) % MOD
        
        # ok[c]: whether a number (>=2) with popcount c is k-reducible
        # i.e., 1 + h(c) <= k, where h(c) = steps to reduce c to 1.
        ok = [False] * (L + 1)
        for c in range(1, L + 1):
            t, steps = c, 0
            while t != 1:
                t = bin(t).count('1')
                steps += 1
            ok[c] = (1 + steps) <= k
        
        ans = 0
        ones = 0
        for i, ch in enumerate(s):
            if ch == '1':
                free = L - 1 - i
                crow = C[free]
                for j in range(free + 1):
                    c = ones + j
                    if c >= 1 and ok[c]:
                        ans = (ans + crow[j]) % MOD
                ones += 1
        
        return ans
```

        L = len(s)

            row = C[m]
            row[0] = 1
            for j in range(1, m + 1):
                row[j] = (prev[j - 1] + prev[j]) % MOD
