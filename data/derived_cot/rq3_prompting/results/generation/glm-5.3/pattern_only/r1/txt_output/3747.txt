```python
class Solution:
    def maxAdjacentDistance(self, nums: List[int]) -> int:
        n = len(nums)
        # Start with the wrap-around pair (last, first)
        ans = abs(nums[-1] - nums[0])
        # All consecutive pairs
        for i in range(n - 1):
            ans = max(ans, abs(nums[i] - nums[i + 1]))
        return ans
```
