For a positive rational number x, define f(x) as follows:

We call a sequence S=(S_1,S_2,\dots,S_N) of positive integers of length N a good sequence if it satisfies all of the following conditions:

```python
import sys
import numpy as np

def main():
    input_data = sys.stdin.buffer.read().split()
    N = int(input_data[0])
    A = list(map(int, input_data[1:1 + N - 1]))
    MOD = 998244353

    # sieve SPF up to 1000
    MAXV = 1001
    spf = list(range(MAXV))
    i = 2
    while i * i < MAXV:
        if spf[i] == i:
            for j in range(i * i, MAXV, i):
                if spf[j] == j:
                    spf[j] = i
        i += 1

    # per-prime exponent arrays
    from collections import defaultdict
    exps = defaultdict(lambda: [0] * (N - 1))
    for i, v in enumerate(A):
        while v > 1:
            q = spf[v]
            e = 0
            while v % q == 0:
                v //= q
                e += 1
            exps[q][i] = e

    ans = 1
    for p, a_list in exps.items():
        a = np.array(a_list, dtype=np.int64)
        H = int(a.sum())
        if H == 0:
            continue
        cuts = np.nonzero(a)[0]
        K = len(cuts)
        # block lengths
        lens = []
        prev = -1
        for idx in cuts:
            idx = int(idx)
            lens.append(idx - prev)
            prev = idx
        lens.append((N - 1) - prev)
        # steps
        steps = [int(a[int(idx)]) for idx in cuts]
        # distinct lens -> Tab
        distinct = sorted(set(lens))
        d_index = {l: j for j, l in enumerate(distinct)}
        D = len(distinct)
        W = H + 1
        bases = np.array([pow(p, l, MOD) for l in distinct], dtype=np.int64).reshape(D, 1)
        Tab = np.ones((D, 1), dtype=np.int64)
        cur = bases
        width = 1
        while width < W:
            take = min(width, W - width)
            Tab = np.concatenate([Tab, Tab[:, :take] * cur % MOD], axis=1)
            cur = cur * cur % MOD
            width += take
        # DP
        Arr = np.zeros((2, W), dtype=np.int64)
        w0 = Tab[d_index[lens[0]]]
        Arr[0] = w0
        Arr[1, :W - 1] = w0[:W - 1]
        # Arr[1, H] stays 0
        for j in range(1, K + 1):
            c = steps[j - 1]
            w = Tab[d_index[lens[j]]]
            new = np.zeros((2, W), dtype=np.int64)
            new[:, c:] += Arr[:, :W - c]
            new[:, :W - c] += Arr[:, c:]
            new[1, H] = 0
            new *= w
            new %= MOD
            Arr = new
        U_H = int(Arr[0].sum() % MOD)
        U_Hm1 = int(Arr[1].sum() % MOD)
        F = (U_H - pow(p, N, MOD) * U_Hm1) % MOD
        ans = ans * F % MOD

    print(ans % MOD)

main()
```

```python
import sys
from collections import defaultdict
import numpy as np

def main():
    data = sys.stdin.buffer.read().split()
    N = int(data[0])
    A = [int(x) for x in data[1:N]]
    MOD = 998244353

    MAXV = 1001
    spf = list(range(MAXV))
    i = 2
    while i * i < MAXV:
        if spf[i] == i:
            for j in range(i * i, MAXV, i):
                if spf[j] == j:
                    spf[j] = i
        i += 1

    M = N - 1
    exps = defaultdict(lambda: [0] * M)
    for i, v in enumerate(A):
        while v > 1:
            q = spf[v]
            e = 0
            while v % q == 0:
                v //= q
                e += 1
            exps[q][i] = e

    ans = 1
    for p, a_list in exps.items():
        a = np.array(a_list, dtype=np.int64)
        H = int(a.sum())
        cuts = np.nonzero(a)[0]
        K = len(cuts)
        lens = []
        prev = -1
        for idx in cuts:
            idx = int(idx)
            lens.append(idx - prev)
            prev = idx
        lens.append(M - prev)
        steps = [int(a[int(idx)]) for idx in cuts]

        distinct = sorted(set(lens))
        didx = {l: j for j, l in enumerate(distinct)}
        D = len(distinct)
        W = H + 1
        cur = np.array([pow(p, l, MOD) for l in distinct], dtype=np.int64).reshape(D, 1)
        Tab = np.ones((D, 1), dtype=np.int64)
        width = 1
        while width < W:
            take = min(width, W - width)
            Tab = np.concatenate([Tab, Tab[:, :take] * cur % MOD], axis=1)
            cur = cur * cur % MOD
            width += take

        Arr = np.zeros((2, W), dtype=np.int64)
        w0 = Tab[didx[lens[0]]]
        Arr[0] = w0
        if W >= 2:
            Arr[1, :W - 1] = w0[:W - 1]
        else:
            Arr[1, 0] = w0[0]
        # note W >= 2 always here (H >= 1)

        for j in range(1, K + 1):
            c = steps[j - 1]
            new = np.zeros((2, W), dtype=np.int64)
```

```python
import sys
from collections import defaultdict
import numpy as np

def main():
    data = sys.stdin.buffer.read().split()
    N = int(data[0])
    A = [int(x) for x in data[1:N]]
    MOD = 998244353

    MAXV = 1001
    spf = list(range(MAXV))
    i = 2
    while i * i < MAXV:
        if spf[i] == i:
            for j in range(i * i, MAXV, i):
                if spf[j] == j:
                    spf[j] = i
        i += 1

    M = N - 1
    exps = defaultdict(lambda: [0] * M)
    for i, v in enumerate(A):
        while v > 1:
            q = spf[v]
            e = 0
            while v % q == 0:
                v //= q
                e += 1
            exps[q][i] = e

    ans = 1
    for p, a_list in exps.items():
        a = np.array(a_list, dtype=np.int64)
        H = int(a.sum())
        if H == 0:
            continue
        cuts = np.nonzero(a)[0]
        K = len(cuts)
        lens = []
        prev = -1
        for idx in cuts:
            ii = int(idx)
            lens.append(ii - prev)
            prev = ii
        lens.append(M - prev)
        steps = [int(a[int(idx)]) for idx in cuts]

        distinct = sorted(set(lens))
        didx = {}
        for j, l in enumerate(distinct):
            didx[l] = j
        D = len(distinct)
        W = H + 1
        cur = np.array([pow(p, l, MOD) for l in distinct], dtype=np.int64).reshape(D, 1)
        Tab = np.ones((D, 1), dtype=np.int64)
        width = 1
        while width < W:
            take = min(width, W - width)
            Tab = np.concatenate([Tab, Tab[:, :take] * cur % MOD], axis=1)
            cur = cur * cur % MOD
            width += take

        Arr = np.zeros((2, W), dtype=np.int64)
        w0 = Tab[didx[lens[0]]]
        Arr[0] = w0
        Arr[1, :W - 1] = w0[:W - 1]

        for j in range(1, K + 1):
            c = steps[j - 1]
            new = np.zeros((2, W), dtype=np.int64)
            new[:, c:] += Arr[:, :W - c]
            new[:, :W - c] += Arr[:, c:]
            new[1, H] = 0
            new *= Tab[didx[lens[j]]]
            new %= MOD
            Arr = new

        U_H = int(Arr[0].sum()) % MOD
        U_H1 = int(Arr[1].sum()) % MOD
        F = (U_H - pow(p, N, MOD) * U_H1) % MOD
        ans = ans * F % MOD

    print(ans)

main()
```

Looking at this problem:

- For each prime $p$, writing $e_i = v_p(S_i)$, the constraint $f(S_i/S_{i+1})=A_i$ becomes $|e_i - e_{i+1}| = v_p(A_i)$, and $\gcd=1$ means $\min_i e_i = 0$. The answer factorizes as $\prod_p F_p$ where $F_p$ sums $p^{\sum e_i}$ over valid exponent walks touching 0.
- Using the bijection (walks in $[0,H]$) = (min-0 walks) ⊔ (walks in $[1,H]$), and shifting the latter down by 1 (which multiplies weight by $p^N$): $F_p = U(H) - p^N U(H-1)$, where $H=\sum_i v_p(A_i)$ (max height of any min-0 walk) and $U(H)$ sums weights over walks capped in $[0,H]$.
- Compress positions into blocks separated by positive steps (a zero step forces equal exponents); block $j$ of length $\ell_j$ contributes weight vector $p^{\ell_j h}$. Run a vectorized DP over heights $0..H$ (two rows: caps $H$ and $H-1$), with numpy. Total block count $\le 4N+168$ and total element work $\lesssim 10^8$, fast enough.

Verified by hand against samples 1 and 2, and small brute-force cases like $A=[2,2]$ ($F=22$), $A=[1,2]$ ($F=6$), $A=[1,1,1]$ ($F=200$) — all match.
