```python
class Solution:
    def maxWeight(self, pizzas: List[int]) -> int:
        pizzas.sort(reverse=True)
        m = len(pizzas) // 4
        odd = (m + 1) // 2
        even = m // 2
        ans = sum(pizzas[:odd])
        idx = odd
        for _ in range(even):
            ans += pizzas[idx + 1]
            idx += 2
        return ans
```

```python
    def maxWeight(self, pizzas: List[int]) -> int:
        pizzas.sort(reverse=True)
        m = len(pizzas) // 4
        odd = (m + 1) // 2   # odd days gain the group max
        even = m // 2        # even days gain the group second max
        # Top `odd` pizzas are gained on odd days (each + 3 fillers).
        # Next 2*even pizzas paired adjacently: gain the smaller of each pair.
        return sum(pizzas[:odd]) + sum(pizzas[odd + 1: odd + 1 + 2 * even: 2])
```
