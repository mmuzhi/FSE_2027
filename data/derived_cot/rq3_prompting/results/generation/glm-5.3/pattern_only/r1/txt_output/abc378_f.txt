        print(0)
        return

    deg = [0] * (n + 1)
    idx = 1

        u = int(data[idx]); v = int(data[idx+1]); idx += 2
        adj[u].append(v)
        adj[v].append(u)
        deg[u] += 1
        deg[v] += 1

    typ = [0] * (n + 1)

        order.append(u)
        for w in adj[u]:
            if not visited[w]:
                visited[w] = True
                parent[w] = u
                q.append(w)
    
    d = [0] * (n + 1)

    # process in reverse order
    for u in reversed(order):
        tu = typ[u]

        s = 0

import sys
from collections import deque

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    adj = [[] for _ in range(n + 1)]
    deg = [0] * (n + 1)
    idx = 1
    for _ in range(n - 1):
        u = int(data[idx]); v = int(data[idx + 1]); idx += 2
        adj[u].append(v)
        adj[v].append(u)
        deg[u] += 1
        deg[v] += 1

    # type: 0 bad, 1 = A (deg 2), 2 = B (deg 3)
    typ = [0] * (n + 1)
    for i in range(1, n + 1):
        if deg[i] == 2:
            typ[i] = 1
        elif deg[i] == 3:
            typ[i] = 2

    parent = [0] * (n + 1)
    order = []
    visited = [False] * (n + 1)
    dq = deque([1])
    visited[1] = True
    while dq:
        u = dq.popleft()
        order.append(u)
        for w in adj[u]:
            if not visited[w]:
                visited[w] = True
                parent[w] = u
                dq.append(w)

    d = [0] * (n + 1)
    total = 0
    for u in reversed(order):
        tu = typ[u]
        s = 0
        ssq = 0
        c1 = 0
        for w in adj[u]:
            if parent[w] == u:
                tw = typ[w]
                if tw == 2:
                    h = d[w]
                    c1 += h
                elif tw == 1:
                    h = 1
                else:
                    h = 0
                s += h
                ssq += h * h
        d[u] = (1 if tu == 1 else 0) + s
        if tu == 1:
            total += c1
        elif tu == 2:
            total += (s * s - ssq) >> 1

    print(total)

main()

import sys

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    adj = [[] for _ in range(n + 1)]
    deg = [0] * (n + 1)
    idx = 1
    for _ in range(n - 1):
        u = int(data[idx]); v = int(data[idx + 1]); idx += 2
        adj[u].append(v)
        adj[v].append(u)
        deg[u] += 1
        deg[v] += 1

    # typ: 1 = tree-degree 2 (must be an endpoint of the added edge),
    #      2 = tree-degree 3 (may be an internal cycle vertex),
    #      0 = anything else (cannot lie on the cycle)
    typ = [0] * (n + 1)
    for i in range(1, n + 1):
        if deg[i] == 2:
            typ[i] = 1
        elif deg[i] == 3:
            typ[i] = 2

    # BFS from vertex 1: order with parents before children
    parent = [0] * (n + 1)
    order = [1]
    head = 0
    while head < len(order):
        u = order[head]
        head += 1
        pu = parent[u]
        for w in adj[u]:
            if w != pu:
                parent[w] = u
                order.append(w)

    # d[u]: number of deg-2 vertices x in subtree(u) such that every vertex
    # strictly between u and x has tree-degree 3.
    d = [0] * (n + 1)
    total = 0
    for u in reversed(order):
        tu = typ[u]
        s = 0      # sum of h over children
        ssq = 0    # sum of h^2 over children
        c1 = 0     # sum of d over deg-3 children (for pairs starting at u)
        for w in adj[u]:
            if parent[w] == u:
                tw = typ[w]
                if tw == 2:
                    h = d[w]
                    c1 += h
                    s += h
                    ssq += h * h
                elif tw == 1:
                    s += 1
                    ssq += 1
        if tu == 1:
            d[u] = s + 1
            total += c1
        elif tu == 2:
            d[u] = s
            total += (s * s - ssq) // 2
        else:
            d[u] = s

    print(total)

Looking at this problem:

- Adding edge (a,b) creates a cycle = tree path a→b plus the new edge. Cycle vertices must all have degree 3 in the final graph: internal path vertices keep tree degree (must be exactly 3), endpoints gain 1 (must have tree degree 2). The graph must be simple, so (a,b) must not be an existing edge, i.e. the path needs at least one internal vertex.

So: count unordered pairs of degree-2 vertices (type A) whose tree path has all internal vertices of degree exactly 3 (type B), with ≥1 internal vertex (vertices of other degrees are forbidden on the path entirely).

**DP on rooted tree** (count each pair at its LCA): define `d(v)` = number of A-vertices in subtree(v) with only B vertices strictly between v and them. For a child `c`, its contribution `h(c) = d(c)` if c is B, `1` if c is A, `0` otherwise. Then `d(v) = [v is A] + Σ h(c)`.

- **LCA = endpoint** (v is A, other vertex below through a B child): add `[v is A] · Σ_{B children} d(c)` — each such pair has ≥1 internal (the B child).
- **LCA strictly inside** (v must be B, endpoints in different child subtrees): add `[v is B] · (S² − Σh²)/2` where S = Σh(c).

Verified against all three samples (1, 0, 6) by hand-tracing the DP, plus several adversarial cases (adjacent A–A pairs excluded, bad-degree vertices blocking paths, branching B vertices, A-vertex between two B's invalidating longer paths). Complexity O(N), iterative BFS avoids recursion limits.
