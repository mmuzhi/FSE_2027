def main():
    data = sys.stdin.buffer.read().split()
    ptr = 0
    n = int(data[ptr]); m = int(data[ptr+1]); k = int(data[ptr+2]); ptr += 3

    for i in range(m):
        us[i] = int(data[ptr]); vs[i] = int(data[ptr+1]); ws[i] = int(data[ptr+2]); ptr += 3
    ca = [0]*(n+1)
    cb = [0]*(n+1)

        ca[int(data[i])] += 1
    ptr += k
    for i in range(ptr, ptr+k):
        cb[int(data[i])] += 1
    ptr += k
    order = sorted(range(m), key=lambda i: ws[i])

    sz = [1]*(n+1)

Each merge event with weight w increases this total by Δ = min(a_X+a_Y, b_X+b_Y) − min(a_X,b_X) − min(a_Y,b_Y), and newly matched cross pairs have f exactly w (they were disconnected below w).
- Lower bound: any permutation has at most g(w) pairs with f ≤ w, so sorted f-values satisfy v_i ≥ min{w : g(w) ≥ i}, giving sum ≥ Σ Δ_k·w_k. The greedy (match as soon as components merge) achieves this exactly, so the answer is Σ over merges of Δ·w.

Verified on sample 1: merges give 1×1 + 1×2 + 1×5 = 8 ✓. Sample 2: 1×1 + 1×2 = 3 ✓.

Implementation: DSU with union by size and path halving, counts of A and B per component, O(M log M).

```python
import sys

def main():
    data = sys.stdin.buffer.read().split()
    ptr = 0
    n = int(data[ptr]); m = int(data[ptr+1]); k = int(data[ptr+2]); ptr += 3
    edges = []
    for _ in range(m):
        u = int(data[ptr]); v = int(data[ptr+1]); w = int(data[ptr+2]); ptr += 3
        edges.append((w, u, v))
    ca = [0]*(n+1)
    cb = [0]*(n+1)
    for i in range(k):
        ca[int(data[ptr+i])] += 1
    ptr += k
    for i in range(k):
        cb[int(data[ptr+i])] += 1
    edges.sort()
    parent = list(range(n+1))
    sz = [1]*(n+1)
    ans = 0
    def find(x):
        p = parent
        while p[x] != x:
            p[x] = p[p[x]]
            x = p[x]
        return x
    for w, u, v in edges:
        ru = find(u); rv = find(v)
        if ru == rv:
            continue
        if sz[ru] < sz[rv]:
            ru, rv = rv, ru
        parent[rv] = ru
        sz[ru] += sz[rv]
        before = min(ca[ru], cb[ru]) + min(ca[rv], cb[rv])
        ca[ru] += ca[rv]; cb[ru] += cb[rv]
        after = min(ca[ru], cb[ru])
        ans += (after - before) * w
    print(ans)

main()
