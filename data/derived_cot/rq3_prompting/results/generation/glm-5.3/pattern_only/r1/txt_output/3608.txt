            for a, ma in lst:

                for b, mb in lst:
                    cb = cnt[b]
                    l = a // gcd(a, b) * b
                    t = cnt[l] if l <= maxV else 0
                    m = (pw3[t] * pw2[ca - t] % MOD * pw2[cb - t]
                         - pw2[ca] - pw2[cb] + 1) % MOD
                    ans = (ans + ma * mb * m) % MOD
        return ans
```
