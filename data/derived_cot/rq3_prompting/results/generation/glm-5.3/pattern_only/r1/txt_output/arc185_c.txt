        Qc = np.fft.irfft(F * F, n)
        Qi = np.rint(Qc[:X + 1]).astype(np.int64)
        sidx = np.arange(0, X + 1, 2)
        diag = np.zeros(X + 1, dtype=np.int64)
        diag[sidx] = pres[sidx >> 1]
        R = Qi - diag
        R[R < 0] = 0

        s = X - Da
        Rs = R[s]
        t2 = X - 2 * Da
        inr = t2 >= 1
        t2c = np.where(inr, t2, 0)
        containsA = inr & (t2 != Da) & pres[t2c]
        cnta = cnt[Da]
        scanOK = (Rs >= 3) | ((Rs == 2) & (~containsA | (cnta >= 2)))
        idxs = np.flatnonzero(scanOK)
        if idxs.size:
            a = int(Da[idxs[0]])
            sv = X - a
            lo = max(1, sv - X)
            hi = (sv - 1) // 2
            bs = np.arange(lo, hi + 1, dtype=np.int64)
            cs = sv - bs
            mask = pres[bs] & pres[cs]
            cand = bs[mask]
            if cnt[a] < 2:
                cand = cand[(cand != a) & (cs[mask] != a)]
            if cand.size:
                b = int(cand[0])
                ans = (a, b, sv - b)
        if ans is None:
            evm = (s % 2) == 0
            Dae = Da[evm]
            if Dae.size:
                te = s[evm] >> 1
                ok = pres[te] & (((te != Dae) & (cnt[te] >= 2)) | ((te == Dae) & (cnt[Dae] >= 3)))
                j = np.flatnonzero(ok)
                if j.size:
                    a = int(Dae[j[0]])
                    t = int(te[j[0]])
                    ans = (a, t, t)

    if ans is None:
        sys.stdout.write("-1\n")
        return
    a, b, c = ans
    res = []
    take = {}
    for v in (a, b, c):
        k = take.get(v, 0)
        if k == 0:
            res.append(int(p1[v]))
        elif k == 1:
            res.append(int(p2[v]))
        else:
            res.append(int(p3[v]))
        take[v] = k + 1
    res.sort()
    sys.stdout.write(f"{res[0]} {res[1]} {res[2]}\n")

main()
