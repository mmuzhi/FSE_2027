class Solution:
    def maxMoves(self, kx: int, ky: int, positions: List[List[int]]) -> int:
        from collections import deque
        from functools import lru_cache

        n = len(positions)
        N = 50
        moves = ((1, 2), (2, 1), (-1, 2), (-2, 1), (1, -2), (2, -1), (-1, -2), (-2, -1))

        def bfs(sx, sy):
            dist = [[-1] * N for _ in range(N)]
            dist[sx][sy] = 0
            q = deque([(sx, sy)])
            while q:
                x, y = q.popleft()
                d = dist[x][y] + 1
                for dx, dy in moves:
                    nx, ny = x + dx, y + dy
                    if 0 <= nx < N and 0 <= ny < N and dist[nx][ny] == -1:
                        dist[nx][ny] = d
                        q.append((nx, ny))
            return dist

        sources = [(p[0], p[1]) for p in positions] + [(kx, ky)]
        m = len(sources)
        D = [[0] * m for _ in range(m)]
        for i, (sx, sy) in enumerate(sources):
            dist = bfs(sx, sy)
            for j, (px, py) in enumerate(sources):
                D[i][j] = dist[px][py]

        @lru_cache(maxsize=None)
        def dp(mask, last):
            if mask == 0:
                return 0
            captured = n - bin(mask).count('1')
            if captured % 2 == 0:  # Alice's turn: maximize
                best = -1
                for i in range(n):
                    if mask >> i & 1:
                        v = D[last][i] + dp(mask ^ (1 << i), i)
                        if v > best:
                            best = v
                return best
            else:  # Bob's turn: minimize
                best = float('inf')
                for i in range(n):
                    if mask >> i & 1:
                        v = D[last][i] + dp(mask ^ (1 << i), i)
                        if v < best:
                            best = v
                return best

        return dp((1 << n) - 1, n)
