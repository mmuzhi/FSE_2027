```python
from typing import List

class Solution:
    def minTime(self, skill: List[int], mana: List[int]) -> int:
        n = len(skill)
        # Prefix sums of skill: S[i] = skill[0] + ... + skill[i-1]
        S = [0] * (n + 1)
        for i in range(n):
            S[i + 1] = S[i] + skill[i]

        # If potion j starts at wizard 0 at time t[j], wizard i starts it at t[j] + mana[j]*S[i]
        # and finishes at t[j] + mana[j]*S[i+1]. Constraints between consecutive potions:
        #   t[j] >= t[j-1] + max_i( mana[j-1]*S[i+1] - mana[j]*S[i] )
        # The max term depends only on mana values, so greedy minimal t[j] is optimal.
        # delta_j = max_i (a*y_i - b*x_i) with a=mana[j-1], b=mana[j], points (x_i,y_i)=(S[i],S[i+1]).
        # Maximizing a*y - b*x over the point set -> upper convex hull + binary search on slope b/a.

        # Build upper hull (points already sorted by strictly increasing x since skill>=1)
        hx, hy = [], []
        for i in range(n):
            x, y = S[i], S[i + 1]
            while len(hx) >= 2 and (hy[-1] - hy[-2]) * (x - hx[-1]) <= (y - hy[-1]) * (hx[-1] - hx[-2]):
                hx.pop()
                hy.pop()
            hx.append(x)
            hy.append(y)
        k = len(hx)
        # Edge slopes of the hull (strictly decreasing left to right)
        dxy = [(hy[t + 1] - hy[t], hx[t + 1] - hx[t]) for t in range(k - 1)]

        total = 0
        for j in range(1, len(mana)):
            a = mana[j - 1]
            b = mana[j]
            # Find smallest hull vertex t that is the peak of y - (b/a)*x:
            # vertex t is peak iff t is last, or slope(t -> t+1) <= b/a, i.e. dy*a <= dx*b.
            lo, hi = 0, k - 1
            while lo < hi:
                mid = (lo + hi) // 2
                dy, dx = dxy[mid]
                if dy * a <= dx * b:
                    hi = mid
                else:
                    lo = mid + 1
            total += a * hy[lo] - b * hx[lo]

        return total + mana[-1] * S[n]
```
