```python
class Solution:
    def buttonWithLongestTime(self, events: List[List[int]]) -> int:
        best_idx = events[0][0]
        best_time = events[0][1]
        for i in range(1, len(events)):
            t = events[i][1] - events[i-1][1]
            if t > best_time or (t == best_time and events[i][0] < best_idx):
                best_time = t
                best_idx = events[i][0]
        return best_idx
```
