class Solution:
    def maxPathLength(self, coordinates: List[List[int]], k: int) -> int:
        import bisect
        x0, y0 = coordinates[k]
        
        def lis(points):
            # points: list of (x, y), compute longest strictly increasing chain
            points.sort(key=lambda p: (p[0], -p[1]))
            tails = []
            for x, y in points:
                idx = bisect.bisect_left(tails, y)
                if idx == len(tails):
                    tails.append(y)
                else:
                    tails[idx] = y
            return len(tails)
        
        left = [p for p in coordinates if p[0] < x0 and p[1] < y0]
        right = [p for p in coordinates if p[0] > x0 and p[1] > y0]
        
        return lis(left) + lis(right) + 1

```python
class Solution:
    def maxPathLength(self, coordinates: List[List[int]], k: int) -> int:
        from bisect import bisect_left

        x0, y0 = coordinates[k]

        def lis_len(points):
            # Longest chain with strictly increasing x and y.
            # Sort by (x asc, y desc) so equal-x points can't both be picked,
            # then take strictly increasing LIS on y.
            pts = sorted(points, key=lambda p: (p[0], -p[1]))
            tails = []
            for _, y in pts:
                idx = bisect_left(tails, y)
                if idx == len(tails):
                    tails.append(y)
                else:
                    tails[idx] = y
            return len(tails)

        # Points usable strictly before k in a path, and strictly after k.
        left = [(p[0], p[1]) for p in coordinates if p[0] < x0 and p[1] < y0]
        right = [(p[0], p[1]) for p in coordinates if p[0] > x0 and p[1] > y0]

        # (best chain ending at k, counting k) + (best chain starting at k, counting k) - 1
        return lis_len(left) + lis_len(right) + 1
```
