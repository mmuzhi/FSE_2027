```python
class Solution:
    def maxSubarrays(self, n: int, conflictingPairs: List[List[int]]) -> int:
        m = len(conflictingPairs)
        # bucket pairs by min endpoint
        pairs_at = [[] for _ in range(n + 1)]  # pairs_at[a] = list of b
        for i, (x, y) in enumerate(conflictingPairs):
            a, b = (x, y) if x < y else (y, x)
            pairs_at[a].append((b, i))
        
        gain = [0] * m
        total = 0
        INF = n + 1
        v1, c1, idx1, v2 = INF, 0, -1, INF
        
        for l in range(n, 0, -1):
            for b, i in pairs_at[l]:
                if b < v1:
                    v2 = v1
                    v1 = b
                    c1 = 1
                    idx1 = i
                elif b == v1:
                    c1 += 1
                else:
                    if b < v2:
                        v2 = b
            # compute f(l)
            if c1 == 0:
                total += INF - l
            else:
                total += v1 - l
                if c1 == 1:
                    gain[idx1] += v2 - v1
        
        best_gain = max(gain) if m > 0 else 0
        return total + best_gain
    ```

```python
from typing import List

class Solution:
    def maxSubarrays(self, n: int, conflictingPairs: List[List[int]]) -> int:
        m = len(conflictingPairs)
        pairs_at = [[] for _ in range(n + 1)]
        for i, pr in enumerate(conflictingPairs):
            x, y = pr
            if x > y:
                x, y = y, x
            pairs_at[x].append((y, i))
        
        gain = [0] * m
        total = 0
        INF = n + 1
        v1 = INF  # smallest b among active
        c1 = 0    # count of active pairs with b == v1
        idx1 = -1 # pair index of the unique min (valid when c1 == 1)
        v2 = INF  # smallest b among active with b > v1
        
        for l in range(n, 0, -1):
            for b, i in pairs_at[l]:
                if b < v1:
                    v2 = v1
                    v1 = b
                    c1 = 1
                    idx1 = i
                elif b == v1:
                    c1 += 1
                elif b < v2:
                    v2 = b
            if c1 == 0:
                total += INF - l
            else:
                total += v1 - l
                if c1 == 1:
                    gain[idx1] += v2 - v1
        
        return total + max(gain)
```

```python
class Solution:
    def maxSubarrays(self, n: int, conflictingPairs: List[List[int]]) -> int:
        m = len(conflictingPairs)
        pairs_at = [[] for _ in range(n + 1)]
        for i, (x, y) in enumerate(conflictingPairs):
            if x > y:
                x, y = y, x
            pairs_at[x].append((y, i))
        
        gain = [0] * m
        total = 0
        INF = n + 1
        v1, c1, idx1, v2 = INF, 0, -1, INF

                if b < v1:
                    v2 = v1
                    v1 = b
                    c1 = 1
                    idx1 = i
                elif b == v1:
                    c1 += 1
                elif b < v2:
                    v2 = b
            if c1 == 0:
                total += INF - l
            else:
                total += v1 - l
                if c1 == 1:
                    gain[idx1] += v2 - v1
        
        return total + (max(gain) if gain else 0)
```

Looking at this problem:

**Key insight:** A subarray `[l, r]` is bad for a normalized pair `(a, b)` (a < b) iff `l <= a` and `r >= b`. So for each left endpoint `l`, the maximal valid `r` is `f(l) - 1` where `f(l) = min{b : pair (a,b) with a >= l}` (or `n+1` if none). Good subarrays = `Σ (f(l) - l)`.

**Removal gain:** Removing pair `p` only changes `f(l)` where `p` is the *unique* active pair achieving the minimum `b`; then new `f(l)` is the second-smallest `b`. Sweep `l` from `n` down to `1`, adding pairs with `a == l`, maintaining `(v1, c1, idx1, v2)`: smallest `b`, its count, the index of the unique minimizer, and second-smallest distinct `b`. When `c1 == 1`, credit `gain[idx1] += v2 - v1`.

**Verification on examples:**
- Example 1: base total = 6, gain for removing (2,3) = 3 → 9 ✓
- Example 2: base total = 9, gain for removing (1,2) = 3 → 12 ✓
- Edge cases checked: duplicate pairs (c1 ≥ 2, no credit), single pair (v2 = n+1, full gain), reversed pair order (normalized).

Complexity: O(n + m).

```python
from typing import List

class Solution:
    def maxSubarrays(self, n: int, conflictingPairs: List[List[int]]) -> int:
        m = len(conflictingPairs)
        # Bucket normalized pairs by their smaller endpoint a.
        pairs_at = [[] for _ in range(n + 1)]
        for i, (x, y) in enumerate(conflictingPairs):
            if x > y:
                x, y = y, x
            pairs_at[x].append((y, i))

        gain = [0] * m
        total = 0
        INF = n + 1
        v1, c1, idx1, v2 = INF, 0, -1, INF

        for l in range(n, 0, -1):
            # Activate all pairs with a == l.
            for b, i in pairs_at[l]:
                if b < v1:
                    v2 = v1
                    v1 = b
                    c1 = 1
                    idx1 = i
                elif b == v1:
                    c1 += 1
                elif b < v2:
                    v2 = b
            # Count good right endpoints for this left endpoint.
            if c1 == 0:
                total += INF - l
            else:
                total += v1 - l
                if c1 == 1:
                    # Removing the unique minimizer pair raises f(l) to v2.
                    gain[idx1] += v2 - v1

        return total + (max(gain) if gain else 0)
```
