```python
class Solution:
    def countPathsWithXorValue(self, grid: List[List[int]], k: int) -> int:
        MOD = 10**9 + 7
        m, n = len(grid), len(grid[0])
        # dp[j][x]: number of paths reaching column j of the current row
        # whose XOR value (including that cell) equals x
        dp = [[0] * 16 for _ in range(n)]
        dp[0][grid[0][0]] = 1

        # Build first row: only moves to the right
        for j in range(1, n):
            v = grid[0][j]
            prev = dp[j - 1]
            cur = [0] * 16
            for x in range(16):
                if prev[x]:
                    cur[x ^ v] = prev[x]
            dp[j] = cur

        # Remaining rows
        for i in range(1, m):
            # Column 0: only moves down
            v = grid[i][0]
            top = dp[0]
            cur = [0] * 16
            for x in range(16):
                if top[x]:
                    cur[x ^ v] = top[x]
            dp[0] = cur

            for j in range(1, n):
                v = grid[i][j]
                left = dp[j - 1]  # current row, already updated
                up = dp[j]        # previous row, not yet updated
                cur = [0] * 16
                for x in range(16):
                    c = left[x] + up[x]
                    if c:
                        cur[x ^ v] = c % MOD
                dp[j] = cur

        return dp[n - 1][k] % MOD
```
