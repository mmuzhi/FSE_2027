import sys

def main():
    input_data = sys.stdin.buffer.read().split()
    N = int(input_data[0])
    A = list(map(int, input_data[1:1+N]))
    MOD = 998244353
    
    # runs
    if A[0] != 1:
        print(0)
        return
    runs = []
    cur = A[0]
    cnt = 0
    for a in A:
        if a == cur:
            cnt += 1
        else:
            runs.append(cnt)
            cur = a
            cnt = 1
    runs.append(cnt)
    
    if any(r % 2 == 0 for r in runs):
        print(0)
        return
    
    m = len(runs)
    T = (N - m) // 2
    
    # precompute factorials up to N
    maxn = N + 1
    fact = [1] * (maxn)
    for i in range(1, maxn):
        fact[i] = fact[i-1] * i % MOD
    inv_fact = [1] * maxn
    inv_fact[maxn-1] = pow(fact[maxn-1], MOD-2, MOD)
    for i in range(maxn-1, 0, -1):
        inv_fact[i-1] = inv_fact[i] * i % MOD
    
    inv2 = (MOD + 1) // 2
    # answer = T! * inv2^T * prod(fact[a_j - 1] * invfact[t_j]^2)
    ans = fact[T] * pow(inv2, T, MOD) % MOD
    for r in runs:
        t = (r - 1) // 2
        ans = ans * fact[r-1] % MOD
        ans = ans * inv_fact[t] % MOD
        ans = ans * inv_fact[t] % MOD
    print(ans)

main()

import sys

def main():
    data = sys.stdin.buffer.read().split()
    N = int(data[0])
    A = list(map(int, data[1:1+N]))
    MOD = 998244353
    
    if A[0] != 1:
        print(0)
        return
    
    # compute run lengths
    runs = []
    cnt = 1
    for i in range(1, N):
        if A[i] == A[i-1]:
            cnt += 1
        else:
            runs.append(cnt)
            cnt = 1
    runs.append(cnt)
    
    for r in runs:
        if r % 2 == 0:
            print(0)
            return
    
    m = len(runs)
    T = (N - m) // 2
    
    # factorials up to N
    fact = [1] * (N + 1)
    for i in range(1, N + 1):
        fact[i] = fact[i-1] * i % MOD
    inv_fact = [1] * (N + 1)
    inv_fact[N] = pow(fact[N], MOD - 2, MOD)
    for i in range(N, 0, -1):
        inv_fact[i-1] = inv_fact[i] * i % MOD
    
    inv2 = (MOD + 1) // 2
    ans = fact[T] * pow(inv2, T, MOD) % MOD
    for r in runs:
        t = (r - 1) >> 1
        ans = ans * fact[r-1] % MOD
        ans = ans * inv_fact[t] % MOD
        ans = ans * inv_fact[t] % MOD
    print(ans)

main()

**Key observations:**
1. An operation requires all cells strictly between `l` and `r` to differ from `X_l = X_r`. Since values are 0/1 and runs alternate, this means `l` is the last cell of its run, `r` the first cell of a run exactly two runs later — the operation merges **three consecutive runs** into one. Every state is a run-length composition of N, alternating in value starting with 1 (cell 1 never changes, so `A_1` must be 1).

2. Each operation reduces run count by exactly 2, so the sequence length is fixed: `T = (N - m)/2` where `m` = number of runs of `A`.

3. Reverse the process: splits of a part `s` into `(x,y,z)`, and each target run's cell range (region) is independent. A region of size `a_j` needs `(a_j-1)/2` splits, so `a_j` must be odd. Total count = (interleavings) × ∏(per-region split sequences): `T!/∏t_j! × ∏ f(a_j)` where `t_j=(a_j-1)/2`.

So answer = `T! · 2^{-T} · ∏ (a_j-1)!/(t_j!)²`.
