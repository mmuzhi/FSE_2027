import sys
import numpy as np

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    grid = np.frombuffer(b''.join(data[1:1+n]), dtype=np.uint8).reshape(n, n)
    # careful: data[1:1+n] are the rows as bytes objects
    A = grid.copy()  # writable
    B = A.copy()  # result
    
    for k in range(1, n//2 + 1):
        lo = k - 1
        hi = n - k
        s = hi - lo + 1
        r = k % 4
        if r == 0:
            continue
        T = A[lo, lo:hi+1]
        Rseg = A[lo+1:hi+1, hi]
        Brev = A[hi, lo:hi][::-1]
        Lrev = A[lo+1:hi, lo][::-1]
        seq = np.concatenate([T, Rseg, Brev, Lrev])
        seq = np.roll(seq, r * (s - 1))
        m1 = s          # end of T
        m2 = m1 + (s-1) # end of Rseg
        m3 = m2 + (s-1) # end of Brev
        B[lo, lo:hi+1] = seq[:m1]
        B[lo+1:hi+1, hi] = seq[m1:m2]
        B[hi, lo:hi] = seq[m2:m3][::-1]
        B[lo+1:hi, lo] = seq[m3:][::-1]
    
    sys.stdout.buffer.write(b'\n'.join(map(bytes, B)) + b'\n')

main()

def main():

**Implementation**: For each ring `k`, extract the clockwise sequence starting at top-left corner: top row (length `s`), right column without top corner (`s-1`), bottom row right→left without right corner (`s-1`), left column bottom→top without corners (`s-2`), total `4(s-1)`. Reading always from the original array and writing to a result array avoids interference.
