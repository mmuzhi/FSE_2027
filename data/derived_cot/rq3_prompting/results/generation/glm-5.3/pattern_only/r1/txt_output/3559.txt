You are given an array of strings words and a string target.
A string x is called valid if x is a prefix of any string in words.

Input: words = ["abc","aaaaa","bcdef"], target = "aabcdabc"

Explanation:
The target string can be formed by concatenating:

Prefix of length 2 of words[1], i.e. "aa".
Prefix of length 3 of words[2], i.e. "bcd".
Prefix of length 3 of words[0], i.e. "abc".

Input: words = ["abababab","ab"], target = "ababaababa"
Output: 2
Explanation:
The target string can be formed by concatenating:

Prefix of length 5 of words[0], i.e. "ababa".

class Solution:
    def minValidStrings(self, words: List[str], target: str) -> int:
        from collections import deque

        # Build Aho-Corasick trie of all words.
        # Every trie node (root-to-node path) is a "valid string" (a prefix of some word).
        children = [{}]   # children[node] : dict char -> node id
        dep = [0]         # depth of node = length of the prefix it spells
        for w in words:
            cur = 0
            for ch in w:
                nxt = children[cur].get(ch)
                if nxt is None:
                    nxt = len(children)
                    children[cur][ch] = nxt
                    children.append({})
                    dep.append(dep[cur] + 1)
                cur = nxt

        # Fail links via BFS
        fail = [0] * len(children)
        q = deque(children[0].values())
        while q:
            u = q.popleft()
            for ch, v in children[u].items():
                f = fail[u]
                while f and ch not in children[f]:
                    f = fail[f]
                fail[v] = children[f].get(ch, 0)
                q.append(v)

        # Key fact: dp is non-decreasing (a prefix of a valid string is valid),
        # so dp[i] = dp[i - L_i] + 1 where L_i is the longest suffix of target[:i]
        # that is a prefix of some word (= depth of AC state after reading i chars).
        INF = float('inf')
        n = len(target)
        dp = [INF] * (n + 1)
        dp[0] = 0
        state = 0
        for i in range(1, n + 1):
            ch = target[i - 1]
            while state and ch not in children[state]:
                state = fail[state]
            state = children[state].get(ch, 0)
            L = dep[state]
            if L:
                cand = dp[i - L] + 1
                if cand < dp[i]:
                    dp[i] = cand

        return dp[n] if dp[n] < INF else -1
