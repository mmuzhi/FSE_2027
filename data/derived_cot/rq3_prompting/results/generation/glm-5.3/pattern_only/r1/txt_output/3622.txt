```python
class Solution:
    def maxFrequency(self, nums: List[int], k: int, numOperations: int) -> int:
        MAXV = 10**5
        limit = MAXV + k  # max useful T
        size = limit + 2
        cnt = [0] * size
        diff = [0] * (size + 1)
        for x in nums:
            cnt[x] += 1
            lo = max(0, x - k)
            hi = x + k  # ≤ limit
            diff[lo] += 1
            diff[hi + 1] -= 1
        ans = 0
        cur = 0
        for T in range(size):
            cur += diff[T]
            # cur = window count at T
            base = cnt[T] if T <= MAXV else 0
            ans = max(ans, base + min(numOperations, cur - base))
        return ans
```

```python
class Solution:
    def maxFrequency(self, nums: List[int], k: int, numOperations: int) -> int:
        MAXV = 10**5
        limit = MAXV + k          # largest useful target value
        size = limit + 2
        cnt = [0] * size          # cnt[v] = elements equal to v
        diff = [0] * size         # difference array for window coverage

        for x in nums:
            cnt[x] += 1
            lo = max(0, x - k)
            hi = x + k            # element x covers targets [x-k, x+k]
            diff[lo] += 1
            if hi + 1 < size:
                diff[hi + 1] -= 1

        ans = 0
        cur = 0
        for T in range(size):
            cur += diff[T]        # cur = number of elements in [T-k, T+k]
            base = cnt[T]         # already equal to T, no operation needed
            ans = max(ans, base + min(numOperations, cur - base))
        return ans
```
