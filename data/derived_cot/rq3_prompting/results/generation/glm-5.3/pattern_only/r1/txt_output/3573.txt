```python
class Solution:
    def validSubstringCount(self, word1: str, word2: str) -> int:
        need = [0] * 26
        for ch in word2:
            need[ord(ch) - 97] += 1
        missing = len(word2)
        ans = 0
        left = 0
        for right, ch in enumerate(word1):
            i = ord(ch) - 97
            need[i] -= 1
            if need[i] >= 0:
                missing -= 1
            while missing == 0:
                j = ord(word1[left]) - 97
                need[j] += 1
                if need[j] > 0:
                    missing += 1
                left += 1
            ans += left
        return ans
```
