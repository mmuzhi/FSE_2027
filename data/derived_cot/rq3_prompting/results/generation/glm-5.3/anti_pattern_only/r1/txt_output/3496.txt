from math import isqrt
from typing import List

class Solution:
    def minNumberOfSeconds(self, mountainHeight: int, workerTimes: List[int]) -> int:
        def units(t: int, w: int) -> int:
            v = (2 * t) // w
            x = (isqrt(4 * v + 1) - 1) // 2
            return x
        
        def feasible(T: int) -> bool:
            total = 0
            for w in workerTimes:
                total += units(T, w)
                if total >= mountainHeight:
                    return True
            return False
        
        lo, hi = 0, min(workerTimes) * mountainHeight * (mountainHeight + 1) // 2
        # answer is in [lo, hi]; hi feasible
        while lo < hi:
            mid = (lo + hi) // 2
            if feasible(mid):
                hi = mid
            else:
                lo = mid + 1
        return lo

from math import isqrt
from typing import List

class Solution:
    def minNumberOfSeconds(self, mountainHeight: int, workerTimes: List[int]) -> int:
        # For a worker with time w, units reducible by time T is the largest x
        # with w * x * (x+1) / 2 <= T, i.e. x*(x+1) <= floor(2T/w).
        def feasible(T: int) -> bool:
            total = 0
            for w in workerTimes:
                v = (2 * T) // w
                # largest x with x*(x+1) <= v
                x = (isqrt(4 * v + 1) - 1) // 2
                total += x
                if total >= mountainHeight:
                    return True
            return False
        
        # Upper bound: the fastest worker doing everything alone suffices.
        lo = 0
        hi = min(workerTimes) * mountainHeight * (mountainHeight + 1) // 2
        while lo < hi:
            mid = (lo + hi) // 2
            if feasible(mid):
                hi = mid
            else:
                lo = mid + 1
        return lo
