There are N mochi (rice cakes), arranged in ascending order of size.
The size of the i-th mochi (1\leq i\leq N) is A_i.
Given two mochi A and B, with sizes a and b respectively, you can make one kagamimochi (a stacked rice cake) by placing mochi A on top of mochi B if and only if a is at most half of b.
You are given Q integer pairs. Let (L_i, R_i) be the i-th pair (1\leq i\leq Q), and solve the following problem for each i:

Using only the R_i - L_i + 1 mochi from the L_i-th to the R_i-th, how many kagamimochi can you make simultaneously?  
More precisely, find the maximum non-negative integer K such that:

- Out of the R_i - L_i + 1 mochi from the L_i-th to the R_i-th, choose 2K mochi and form K pairs. For each pair, place one mochi on top of the other, to make K kagamimochi.

- A_i \leq A_{i+1} \ (1 \leq i < N)
- 1 \leq Q \leq 2 \times 10^5
- 1 \leq L_i < R_i \leq N \ (1 \leq i \leq Q)
- All input values are integers.

```python
import sys
import numpy as np

def main():
    data = sys.stdin.buffer.read().split()
    idx = 0
    N = int(data[idx]); idx += 1
    A = np.array(list(map(int, data[idx:idx+N])), dtype=np.int64)
    idx += N
    Q = int(data[idx]); idx += 1
    qs = np.array(list(map(int, data[idx:idx+2*Q])), dtype=np.int64).reshape(Q, 2) if Q > 0 else np.zeros((0,2), dtype=np.int64)
    
    # g[p] = nxt(p) - p, where nxt(p) = first index q with A[q] >= 2*A[p]
    nxt = np.searchsorted(A, 2 * A, side='left')
    g = (nxt - np.arange(N, dtype=np.int64)).astype(np.int32)
    
    LOGN = N.bit_length() - 1
    sp = np.zeros((LOGN + 1, N), dtype=np.int32)
    sp[0] = g
    for j in range(1, LOGN + 1):
        step = 1 << (j - 1)
        length = N - (1 << j) + 1
        sp[j, :length] = np.maximum(sp[j-1, :length], sp[j-1, step:step+length])
    
    log_table = np.zeros(N + 1, dtype=np.int64)
    j = 0
    while (1 << j) <= N:
        lo_i = 1 << j
        hi_i = min(1 << (j + 1), N + 1)
        log_table[lo_i:hi_i] = j
        j += 1
    
    if Q == 0:
        return
    
    L = qs[:, 0]
    R = qs[:, 1]
    L0 = L - 1
    M = R - L + 1
    lo = np.zeros(Q, dtype=np.int64)
    hi = (M >> 1).astype(np.int64)
    
    while True:
        act = np.nonzero(lo < hi)[0]
        if act.size == 0:
            break
        l = lo[act]
        h = hi[act]
        m = (l + h + 1) >> 1
        s = L0[act]
        e = s + m - 1
        jj = log_table[m]
        p2 = np.left_shift(np.int64(1), jj)
        v1 = sp[jj, s]
        v2 = sp[jj, e - p2 + 1]
        rmax = np.maximum(v1, v2)
        ok = rmax <= (M[act] - m)
        lo[act] = np.where(ok, m, l)
        hi[act] = np.where(ok, h, m - 1)
    
    out = '\n'.join(map(str, lo.tolist()))
    sys.stdout.write(out + '\n')

main()
```

```python
    log_table = np.zeros(N + 1, dtype=np.int64)
```

=== ANSWER ===
Looking at this problem:

**Key insight**: If K pairs can be made in [L,R], then using the K smallest mochi as tops and K largest as bottoms with sorted matching also works (exchange argument). So K is feasible iff `2K ≤ R-L+1` and for all i in [0,K): `2*A[L+i] ≤ A[R-K+1+i]`.

Defining `nxt(i)` = first position q with `A[q] ≥ 2*A[i]` and `g(i) = nxt(i) - i`, the condition becomes `max g over [L, L+K-1] ≤ (R-L+1) - K`, which is monotone in K (dropping a pair keeps feasibility). So binary search K per query with a sparse table for range max of g → O((N+Q) log N). I vectorize the binary search across all queries with numpy.

Verified against both samples (e.g., query (1,11): max g[0..4]=3 ≤ 6 → 5 ✓; query (7,11): max g[6..7]=4 ≤ 3 fails, g[6]=4 ≤ 4 → 1 ✓).

```python
import sys
import numpy as np


def main():
    data = sys.stdin.buffer.read().split()
    pos = 0
    N = int(data[pos]); pos += 1
    A = np.array(list(map(int, data[pos:pos + N])), dtype=np.int64)
    pos += N
    Q = int(data[pos]); pos += 1
    qs = np.array(list(map(int, data[pos:pos + 2 * Q])), dtype=np.int64).reshape(Q, 2)

    # nxt[p]: smallest index q with A[q] >= 2*A[p] (N if none)
    nxt = np.searchsorted(A, 2 * A, side='left')
    g = (nxt - np.arange(N, dtype=np.int64)).astype(np.int32)

    # sparse table for range-max of g
    LOGN = N.bit_length() - 1
    sp = np.zeros((LOGN + 1, N), dtype=np.int32)
    sp[0] = g
    for j in range(1, LOGN + 1):
        step = 1 << (j - 1)
        length = N - (1 << j) + 1
        sp[j, :length] = np.maximum(sp[j - 1, :length], sp[j - 1, step:step + length])

    # log_table[x] = floor(log2(x)) for x in [1, N]
    log_table = np.zeros(N + 1, dtype=np.int64)
    j = 0
    while (1 << j) <= N:
        log_table[1 << j:min(1 << (j + 1), N + 1)] = j
        j += 1

    L0 = qs[:, 0] - 1            # 0-indexed start of query range
    M = qs[:, 1] - qs[:, 0] + 1  # range length
    lo = np.zeros(Q, dtype=np.int64)
    hi = (M >> 1).astype(np.int64)

    while True:
        act = np.nonzero(lo < hi)[0]
        if act.size == 0:
            break
        l = lo[act]
        h = hi[act]
        m = (l + h + 1) >> 1
        s = L0[act]
        jj = log_table[m]
        p2 = np.left_shift(1, jj)
        e2 = s + m - p2
        rmax = np.maximum(sp[jj, s], sp[jj, e2])
        ok = rmax <= (M[act] - m)
        lo[act] = np.where(ok, m, l)
        hi[act] = np.where(ok, h, m - 1)

    print('\n'.join(map(str, lo.tolist())))


main()
```
