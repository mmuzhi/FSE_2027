                    i += i & (-i)

                res = 0
                while i > 0:
                    if tree[i] > res:
                        res = tree[i]
                    i -= i & (-i)

```python
from typing import List
from bisect import bisect_left

class Solution:
    def maxPathLength(self, coordinates: List[List[int]], k: int) -> int:
        xk, yk = coordinates[k]

        def longest_chain(points):
            # points: list of (x, y) tuples
            # returns the max number of points that form a chain
            # with strictly increasing x and y
            pts = sorted((x, -y) for x, y in points)
            tops = []
            for _, negy in pts:
                y = -negy
                idx = bisect_left(tops, y)
                if idx == len(tops):
                    tops.append(y)
                else:
                    tops[idx] = y
            return len(tops)

        below = [(x, y) for x, y in coordinates if x < xk and y < yk]
        above = [(x, y) for x, y in coordinates if x > xk and y > yk]
        return 1 + longest_chain(below) + longest_chain(above)
```

```python
from typing import List
from bisect import bisect_left

class Solution:
    def maxPathLength(self, coordinates: List[List[int]], k: int) -> int:
        xk, yk = coordinates[k]

        def longest_chain(points):
            # Max length of a chain with strictly increasing x and y.
            pts = sorted((x, -y) for x, y in points)
            tops = []
            for _, negy in pts:
                y = -negy
                i = bisect_left(tops, y)
                if i == len(tops):
                    tops.append(y)
                else:
                    tops[i] = y
            return len(tops)

        below = [(x, y) for x, y in coordinates if x < xk and y < yk]
        above = [(x, y) for x, y in coordinates if x > xk and y > yk]

        return 1 + longest_chain(below) + longest_chain(above)
```

Looking at this problem:

- Any increasing path containing point `k` splits at `k` into a prefix (points with `x < x_k` and `y < y_k`) and a suffix (points with `x > x_k` and `y > y_k`), which are completely independent.
- So the answer = (longest chain among points strictly dominated by `k`) + 1 + (longest chain among points strictly dominating `k`).
- Longest chain with both coordinates strictly increasing: sort by `(x asc, y desc)` — this reduces to a strict 1D LIS on the `y` values (the `y desc` tiebreak prevents chaining points with equal `x`, and strict LIS via `bisect_left` prevents chaining equal `y`). Use patience piles for O(n log n).

Verification on Example 1: point (2,2); below = {(0,0)} → chain 1; above = {(5,3)} → chain 1; answer = 1+1+1 = 3 ✓. Example 2: point (5,6); below = {(2,1)} → 1; above = {} → 0; answer = 2 ✓.
