```python
from collections import deque

def main():
    data = sys.stdin.buffer.read().split()
    N = int(data[0]); M = int(data[1])
    A = [int(x) - 1 for x in data[2:2 + N]]
    MOD = 998244353

    try:
        import numpy as np
        use_np = True
    except Exception:
        use_np = False

    if use_np:
        LEAF = np.arange(1, M + 1, dtype=np.int64)
        def vec_mul(a, b):
            return (a * b) % MOD
        def vec_psum(q):
            return np.cumsum(q) % MOD
        def vec_total(r):
            return int(r.sum()) % MOD
    else:
        LEAF = list(range(1, M + 1))
        def vec_mul(a, b):
            return [x * y % MOD for x, y in zip(a, b)]
        def vec_psum(q):
            return [x % MOD for x in accumulate(q)]
        def vec_total(r):
            return sum(r) % MOD

    # Peel indegree-0 nodes to find cycle nodes
    indeg = [0] * N
    for v in A:
        indeg[v] += 1
    stack = [i for i in range(N) if indeg[i] == 0]
    on_cycle = [True] * N
    while stack:
        u = stack.pop()
        on_cycle[u] = False
        v = A[u]
        indeg[v] -= 1
        if indeg[v] == 0:
            stack.append(v)

    # Tree children (reverse edges, excluding cycle nodes)
    children = [[] for _ in range(N)]
    for i in range(N):
        if not on_cycle[i]:
            children[A[i]].append(i)

    # BFS from cycle nodes; reversed order processes children before parents
    order = []
    dq = deque(i for i in range(N) if on_cycle[i])
    while dq:
        u = dq.popleft()
        order.append(u)
        dq.extend(children[u])

    vecs = [None] * N
    for u in reversed(order):
        if on_cycle[u]:
            continue
        ch = children[u]
        if not ch:
            vecs[u] = LEAF
        else:
            q = vecs[ch[0]]
            for w in ch[1:]:
                q = vec_mul(q, vecs[w])
            vecs[u] = vec_psum(q)
            for w in ch:
                vecs[w] = None  # free memory

                v = A[v]

            s = vec_total(r)
        else:
            s = M % MOD
        ans = ans * s % MOD
    print(ans % MOD)
```

```python
import sys
from itertools import accumulate
from collections import deque

def main():
    data = sys.stdin.buffer.read().split()
    N = int(data[0]); M = int(data[1])
    A = [int(x) - 1 for x in data[2:2 + N]]
    MOD = 998244353

    try:
        import numpy as np
    except Exception:
        np = None

    if np is not None:
        LEAF = np.arange(1, M + 1, dtype=np.int64)

        def vec_mul(a, b):
            return (a * b) % MOD

        def vec_psum(q):
            return np.cumsum(q) % MOD

        def vec_total(r):
            return int(r.sum()) % MOD
    else:
        LEAF = list(range(1, M + 1))

        def vec_mul(a, b, _m=MOD):
            return [x * y % _m for x, y in zip(a, b)]

        def vec_psum(q, _m=MOD):
            return [x % _m for x in accumulate(q)]

        def vec_total(r, _m=MOD):
            return sum(r) % _m

    # ---- find cycle nodes: peel nodes with indegree 0 ----
    indeg = [0] * N
    for v in A:
        indeg[v] += 1
    stack = [i for i in range(N) if indeg[i] == 0]
    on_cycle = [True] * N
    while stack:
        u = stack.pop()
        on_cycle[u] = False
        v = A[u]
        indeg[v] -= 1
        if indeg[v] == 0:
            stack.append(v)

    # ---- tree children lists (reverse edges, non-cycle nodes only) ----
    children = [[] for _ in range(N)]
    for i in range(N):
        if not on_cycle[i]:
            children[A[i]].append(i)

    # ---- BFS from cycle nodes; reversed => children processed before parents ----
    order = []
    dq = deque(i for i in range(N) if on_cycle[i])
    while dq:
        u = dq.popleft()
        order.append(u)
        dq.extend(children[u])

    # ---- compute vectors: P_u(v) = #assignments in subtree of u with x_u <= v ----
    vecs = [None] * N
    for u in reversed(order):
        if on_cycle[u]:
            continue
        ch = children[u]
        if not ch:
            vecs[u] = LEAF
        else:
            q = vecs[ch[0]]
            for w in ch[1:]:
                q = vec_mul(q, vecs[w])
            vecs[u] = vec_psum(q)
            for w in ch:
                vecs[w] = None  # free

    # ---- label cycle components ----
    comp_id = [-1] * N
    ncomp = 0
    for u in range(N):
        if on_cycle[u] and comp_id[u] < 0:
            v = u
            while True:
                comp_id[v] = ncomp
                v = A[v]
                if v == u:
                    break
            ncomp += 1

    # ---- subtrees hanging directly off each cycle ----
    comp_children = [[] for _ in range(ncomp)]
    for w in range(N):
        if not on_cycle[w] and on_cycle[A[w]]:
            comp_children[comp_id[A[w]]].append(w)

    # ---- combine ----
    ans = 1
    for c in range(ncomp):
        lst = comp_children[c]
        if lst:
            r = vecs[lst[0]]
            for w in lst[1:]:
                r = vec_mul(r, vecs[w])
            s = vec_total(r)
        else:
            s = M % MOD
        ans = ans * s % MOD

    print(ans % MOD)

main()
```

=== ANSWER ===
Looking at this problem:

- The edges i → A_i form a functional graph. Around any cycle, the inequalities force all cycle values equal. Non-cycle nodes form trees hanging off cycle nodes, where each child's value ≤ its parent's value.
- Components are independent: answer = product over components of Σ_{v=1..M} (ways for trees given cycle value v).
- For a tree node u define P_u(v) = # assignments in u's subtree with x_u ≤ v. Then P_u(v) = Σ_{t=1..v} Π_{child w} P_w(t). Leaves: P(v)=v.
- All values needed only at v ∈ [1..M], so keep length-M value vectors: pointwise products + prefix sums → O(N·M) total (≈4M ops), which easily fits limits. Verified against all three samples (e.g. sample 3 gives 5 · 2002 = 10010).

Cycle nodes are found by peeling indegree-0 nodes; trees are processed children-before-parents via reversed BFS from cycle nodes.

```python
import sys
from itertools import accumulate
from collections import deque

def main():
    data = sys.stdin.buffer.read().split()
    N = int(data[0]); M = int(data[1])
    A = [int(x) - 1 for x in data[2:2 + N]]
    MOD = 998244353

    try:
        import numpy as np
    except Exception:
        np = None

    if np is not None:
        LEAF = np.arange(1, M + 1, dtype=np.int64)
        def vec_mul(a, b):
            return (a * b) % MOD
        def vec_psum(q):
            return np.cumsum(q) % MOD
        def vec_total(r):
            return int(r.sum()) % MOD
    else:
        LEAF = list(range(1, M + 1))
        def vec_mul(a, b, _m=MOD):
            return [x * y % _m for x, y in zip(a, b)]
        def vec_psum(q, _m=MOD):
            return [x % _m for x in accumulate(q)]
        def vec_total(r, _m=MOD):
            return sum(r) % _m

    # find cycle nodes by peeling indegree-0 nodes
    indeg = [0] * N
    for v in A:
        indeg[v] += 1
    stack = [i for i in range(N) if indeg[i] == 0]
    on_cycle = [True] * N
    while stack:
        u = stack.pop()
        on_cycle[u] = False
        v = A[u]
        indeg[v] -= 1
        if indeg[v] == 0:
            stack.append(v)

    # tree children (reverse edges, non-cycle nodes only)
    children = [[] for _ in range(N)]
    for i in range(N):
        if not on_cycle[i]:
            children[A[i]].append(i)

    # BFS from cycle nodes; reversed order => children before parents
    order = []
    dq = deque(i for i in range(N) if on_cycle[i])
    while dq:
        u = dq.popleft()
        order.append(u)
        dq.extend(children[u])

    vecs = [None] * N
    for u in reversed(order):
        if on_cycle[u]:
            continue
        ch = children[u]
        if not ch:
            vecs[u] = LEAF
        else:
            q = vecs[ch[0]]
            for w in ch[1:]:
                q = vec_mul(q, vecs[w])
            vecs[u] = vec_psum(q)
            for w in ch:
                vecs[w] = None  # free memory

    # label cycle components
    comp_id = [-1] * N
    ncomp = 0
    for u in range(N):
        if on_cycle[u] and comp_id[u] < 0:
            v = u
            while True:
                comp_id[v] = ncomp
                v = A[v]
                if v == u:
                    break
            ncomp += 1

    # subtrees hanging directly off each cycle
    comp_children = [[] for _ in range(ncomp)]
    for w in range(N):
        if not on_cycle[w] and on_cycle[A[w]]:
            comp_children[comp_id[A[w]]].append(w)

    ans = 1
    for c in range(ncomp):
        lst = comp_children[c]
        if lst:
            r = vecs[lst[0]]
            for w in lst[1:]:
                r = vec_mul(r, vecs[w])
            s = vec_total(r)
        else:
            s = M % MOD
        ans = ans * s % MOD

    print(ans % MOD)

main()
```
