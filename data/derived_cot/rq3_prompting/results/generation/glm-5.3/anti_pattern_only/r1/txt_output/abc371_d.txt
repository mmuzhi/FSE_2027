import sys
from bisect import bisect_left, bisect_right
from itertools import accumulate

def main():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    X = data[idx:idx+n]
    X = list(map(int, X))
    idx += n
    P = list(map(int, data[idx:idx+n]))

    q = int(data[idx]); idx += 1
