```python
from typing import List

class Solution:
    def shiftDistance(self, s: str, t: str, nextCost: List[int], previousCost: List[int]) -> int:
        # Prefix sums: nextPrefix[i] = nextCost[0] + ... + nextCost[i-1]
        nextPrefix = [0] * 27
        prevPrefix = [0] * 27
        for i in range(26):
            nextPrefix[i + 1] = nextPrefix[i] + nextCost[i]
            prevPrefix[i + 1] = prevPrefix[i] + previousCost[i]
        
        totalNext = nextPrefix[26]
        totalPrev = prevPrefix[26]
        
        result = 0
        for cs, cd in zip(s, t):
            c = ord(cs) - ord('a')
            d = ord(cd) - ord('a')
            
            # Forward: steps from c to d going next
            steps_f = (d - c) % 26
            if steps_f == 0:
                cost_f = 0
            else:
                end = c + steps_f
                if end <= 26:
                    cost_f = nextPrefix[end] - nextPrefix[c]
                else:
                    cost_f = (totalNext - nextPrefix[c]) + nextPrefix[end - 26]
            
            # Backward: steps from c to d going previous
            steps_b = (c - d) % 26
            if steps_b == 0:
                cost_b = 0
            else:
                start_min = c - steps_b + 1  # smallest starting character index (may be negative)
                if start_min >= 0:
                    cost_b = prevPrefix[c + 1] - prevPrefix[start_min]
                else:
                    # wrapped: characters [0, c] and [start_min + 26, 25]
                    cost_b = (prevPrefix[c + 1] - prevPrefix[0]) + (totalPrev - prevPrefix[start_min + 26])
            
            result += min(cost_f, cost_b)
        
        return result
```

```python
from typing import List

class Solution:
    def shiftDistance(self, s: str, t: str, nextCost: List[int], previousCost: List[int]) -> int:
        nextPrefix = [0] * 27
        prevPrefix = [0] * 27
        for i in range(26):
            nextPrefix[i + 1] = nextPrefix[i] + nextCost[i]
            prevPrefix[i + 1] = prevPrefix[i] + previousCost[i]
        
        totalNext = nextPrefix[26]
        totalPrev = prevPrefix[26]
        
        result = 0
        for cs, cd in zip(s, t):
            c = ord(cs) - 97
            d = ord(cd) - 97
            
            # Forward cost: c -> d via next shifts
            steps_f = (d - c) % 26
            if steps_f:
                end = c + steps_f
                if end <= 26:
                    cost_f = nextPrefix[end] - nextPrefix[c]
                else:
                    cost_f = (totalNext - nextPrefix[c]) + nextPrefix[end - 26]
            else:
                cost_f = 0
            
            # Backward cost: c -> d via previous shifts
            steps_b = (c - d) % 26
            if steps_b:
                lo = c - steps_b + 1
                if lo >= 0:
                    cost_b = prevPrefix[c + 1] - prevPrefix[lo]
                else:
                    cost_b = prevPrefix[c + 1] + (totalPrev - prevPrefix[lo + 26])
            else:
                cost_b = 0
            
            result += cost_f if cost_f < cost_b else cost_b
        
        return result
```

Looking at this problem:

- For each character pair (c, d), we can either shift forward `(d-c) mod 26` times (paying `nextCost` of each character visited as the *starting* character) or shift backward `(c-d) mod 26` times (paying `previousCost` of each starting character).
- Use prefix sums over the 26-letter cost arrays to compute the cost of any forward or backward path (handling wraparound) in O(1).
- Forward path from `c` with `k` steps covers `nextCost[c..c+k-1]` (mod 26). Backward path covers `previousCost` for characters `c, c-1, ..., c-k+1` (mod 26), i.e., a contiguous range `[c-k+1, c]` which may wrap past 0 into `[c-k+27, 25]`.

I verified against both examples (e.g., example 1: 'a'→'b' backward costs `previousCost[0] + previousCost[2..25] = 1`, forward costs 100, giving total 2; example 2 gives 9+10+1+11=31).

```python
from typing import List

class Solution:
    def shiftDistance(self, s: str, t: str, nextCost: List[int], previousCost: List[int]) -> int:

            prevPrefix[i + 1] = prevPrefix[i] + previousCost[i]

        totalPrev = prevPrefix[26]

                    cost_f = nextPrefix[end] - nextPrefix[c]
                else:
```
