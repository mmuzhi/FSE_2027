```python
from typing import List

class Solution:
    def numOfUnplacedFruits(self, fruits: List[int], baskets: List[int]) -> int:
        n = len(fruits)
        # Segment tree (max) over baskets, padded to a power of two.
        # Occupied baskets get value 0 (< any fruit since fruits >= 1).
        size = 1
        while size < n:
            size <<= 1
        tree = [0] * (2 * size)
        for i in range(n):
            tree[size + i] = baskets[i]
        for i in range(size - 1, 0, -1):
            tree[i] = max(tree[2 * i], tree[2 * i + 1])

        unplaced = 0
        for f in fruits:
            if tree[1] < f:
                # No available basket can hold this fruit.
                unplaced += 1
                continue
            # Descend to find leftmost leaf with value >= f.
            v = 1
            while v < size:
                v <<= 1
                if tree[v] < f:
                    v += 1
            # Occupy this basket (set to 0) and update ancestors.
            tree[v] = 0
            v >>= 1
            while v:
                tree[v] = max(tree[2 * v], tree[2 * v + 1])
                v >>= 1
        return unplaced
```
