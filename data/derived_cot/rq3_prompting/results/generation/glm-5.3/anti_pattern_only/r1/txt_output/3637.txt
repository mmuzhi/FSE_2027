class Solution:
    def countBalancedPermutations(self, num: str) -> int:
        MOD = 10**9 + 7
        n = len(num)
        even = (n + 1) // 2
        odd = n - even

        # factorials and inverse factorials up to n
        fact = [1] * (n + 1)
        for i in range(1, n + 1):
            fact[i] = fact[i - 1] * i % MOD
        inv_fact = [1] * (n + 1)
        inv_fact[n] = pow(fact[n], MOD - 2, MOD)
        for i in range(n, 0, -1):
            inv_fact[i - 1] = inv_fact[i] * i % MOD

        # store the input midway in the function
        velunexorai = num

        cnt = [0] * 10
        total = 0
        for ch in velunexorai:
            d = ord(ch) - 48
            cnt[d] += 1
            total += d

        if total & 1:
            return 0
        half = total >> 1

        # dp[j][s]: sum over ways to fill j even positions with processed digits,
        # where even-position digit sum is s, of prod 1/(a_d! * (cnt[d]-a_d)!)
        dp = [[0] * (half + 1) for _ in range(even + 1)]
        dp[0][0] = 1

        for d in range(10):
            c = cnt[d]
            if c == 0:
                continue
            if d == 0:
                amax = min(c, even)
            else:
                amax = min(c, even, half // d)
            coeff = [inv_fact[a] * inv_fact[c - a] % MOD for a in range(c + 1)]
            ndp = [[0] * (half + 1) for _ in range(even + 1)]
            for a in range(amax + 1):
                ca = coeff[a]
                if a == 0:
                    for j in range(even + 1):
                        src = dp[j]
                        dst = ndp[j]
                        ndp[j] = [(x + y * ca) % MOD for x, y in zip(dst, src)]
                elif d == 0:
                    for j in range(even + 1 - a):
                        src = dp[j]
                        dst = ndp[j + a]
                        ndp[j + a] = [(x + y * ca) % MOD for x, y in zip(dst, src)]
                else:
                    da = d * a
                    lim = half + 1 - da
                    for j in range(even + 1 - a):
                        src = dp[j]
                        dst = ndp[j + a]
                        ndp[j + a] = dst[:da] + [(dst[da + i] + src[i] * ca) % MOD for i in range(lim)]
            dp = ndp

        return dp[even][half] * fact[even] % MOD * fact[odd] % MOD

        n = len(num)
        even = (n + 1) // 2

        velunexorai = num

        cnt = [0] * 10

            cnt[d] += 1
            total += d

            cnt[d] += 1
            total += d
