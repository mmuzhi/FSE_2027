class Solution:
    def maximumValueSum(self, board: List[List[int]]) -> int:
        m, n = len(board), len(board[0])
        # top 3 (value, col) per row
        top = []
        for i in range(m):
            row = board[i]
            # get top 3 by value
            cands = sorted(range(n), key=lambda c: row[c], reverse=True)[:3]
            top.append([(row[c], c) for c in cands])
        
        best = -inf
        for i, j, k in combinations(range(m), 3):
            # brute force 27 combos
            for vi, ci in top[i]:
                for vj, cj in top[j]:
                    if cj == ci: continue
                    for vk, ck in top[k]:
                        if ck == ci or ck == cj: continue
                        s = vi + vj + vk
                        if s > best: best = s
        return best
```

        top3row = []
        for i in range(m):
            row = board[i]
            idx = sorted(range(n), key=row.__getitem__, reverse=True)[:3]
            top3row.append([(row[c], c) for c in idx])
        
        # top 3 per column: list of (val, row)
        top3col = []
        for c in range(n):
            col = [board[r][c] for r in range(m)]
            idx = sorted(range(m), key=col.__getitem__, reverse=True)[:3]
            top3col.append([(col[r], r) for r in idx])
        
        best = NEG
        for i in range(m):
            ti = top3row[i]
            for j in range(i+1, m):
                tj = top3row[j]

from itertools import combinations
from typing import List

class Solution:
    def maximumValueSum(self, board: List[List[int]]) -> int:
        m, n = len(board), len(board[0])
        
        # Top 3 cells per row: list of (value, col)
        top3row = []
        for row in board:
            idx = sorted(range(n), key=row.__getitem__, reverse=True)[:3]
            top3row.append([(row[c], c) for c in idx])
        
        # Top 3 cells per column: list of (value, row)
        cols = list(zip(*board))
        top3col = []
        for col in cols:
            idx = sorted(range(m), key=col.__getitem__, reverse=True)[:3]
            top3col.append([(col[r], r) for r in idx])
        
        best = float('-inf')
        
        for i in range(m):
            ti = top3row[i]
            for j in range(i + 1, m):
                tj = top3row[j]
                # g[c]: best value in column c using rows != i, j; collect (g, c)
                gc = []
                ap = gc.append
                for c in range(n):
                    t = top3col[c]
                    v0, r0 = t[0]
                    if r0 != i and r0 != j:
                        ap((v0, c))
                    else:
                        v1, r1 = t[1]
                        if r1 != i and r1 != j:
                            ap((v1, c))
                        else:
                            v2, r2 = t[2]
                            ap((v2, c))
                gc.sort(reverse=True)
                g0v, g0c = gc[0]
                g1v, g1c = gc[1]
                g2v, g2c = gc[2]
                for vi, ci in ti:
                    for vj, cj in tj:
                        if ci == cj:
                            continue
                        s2 = vi + vj
                        if g0c != ci and g0c != cj:
                            t = s2 + g0v
                        elif g1c != ci and g1c != cj:
                            t = s2 + g1v
                        else:
                            t = s2 + g2v
                        if t > best:
                            best = t
        return best
```

```python
class Solution:
    def maximumValueSum(self, board: List[List[int]]) -> int:
        m, n = len(board), len(board[0])

        # Top-3 cells (by value) in each row: list of (value, col)
        top3row = []
        for row in board:
            idx = sorted(range(n), key=row.__getitem__, reverse=True)[:3]
            top3row.append([(row[c], c) for c in idx])

        # Top-3 cells (by value) in each column: list of (value, row)
        top3col = []
        for c in range(n):
            col = [board[r][c] for r in range(m)]
            idx = sorted(range(m), key=col.__getitem__, reverse=True)[:3]
            top3col.append([(col[r], r) for r in idx])

        best = float('-inf')

        for i in range(m):
            ti = top3row[i]
            for j in range(i + 1, m):
                tj = top3row[j]
                # For every column c, best value available in column c
                # when rows i and j are forbidden.
                gc = []
                for c in range(n):
                    (v0, r0), (v1, r1), (v2, r2) = top3col[c]
                    if r0 != i and r0 != j:
                        gc.append((v0, c))
                    elif r1 != i and r1 != j:
                        gc.append((v1, c))
                    else:
                        gc.append((v2, c))
                gc.sort(reverse=True)
                (g0v, g0c), (g1v, g1c), (g2v, g2c) = gc[0], gc[1], gc[2]

                for vi, ci in ti:
                    for vj, cj in tj:
                        if ci == cj:
                            continue
                        s2 = vi + vj
                        if g0c != ci and g0c != cj:
                            t = s2 + g0v
                        elif g1c != ci and g1c != cj:
                            t = s2 + g1v
                        else:
                            t = s2 + g2v
                        if t > best:
                            best = t
        return best
```

```python
from typing import List

class Solution:
    def maximumValueSum(self, board: List[List[int]]) -> int:
        m, n = len(board), len(board[0])

        # Top-3 cells (by value) in each row: entries (value, col)
        top3row = []
        for row in board:
            idx = sorted(range(n), key=row.__getitem__, reverse=True)[:3]
            top3row.append([(row[c], c) for c in idx])

        # Top-3 cells (by value) in each column: entries (value, row)
        top3col = []
        for c in range(n):
            col = [board[r][c] for r in range(m)]
            idx = sorted(range(m), key=col.__getitem__, reverse=True)[:3]
            top3col.append([(col[r], r) for r in idx])

        best = float('-inf')

        for i in range(m):
            ti = top3row[i]
            for j in range(i + 1, m):
                tj = top3row[j]
                # g[c] = best value in column c among rows != i, j
                gc = []
                for c in range(n):
                    t = top3col[c]
                    v0, r0 = t[0]
                    if r0 != i and r0 != j:
                        gc.append((v0, c))
                    else:
                        v1, r1 = t[1]
                        if r1 != i and r1 != j:
                            gc.append((v1, c))
                        else:
                            v2, _ = t[2]
                            gc.append((v2, c))
                gc.sort(reverse=True)
                (g0v, g0c), (g1v, g1c), (g2v, g2c) = gc[0], gc[1], gc[2]

                for vi, ci in ti:
                    for vj, cj in tj:
                        if ci == cj:
                            continue
                        s2 = vi + vj
                        if g0c != ci and g0c != cj:
                            total = s2 + g0v
                        elif g1c != ci and g1c != cj:
                            total = s2 + g1v
                        else:
                            total = s2 + g2v
                        if total > best:
                            best = total
        return best
```

class Solution:
    def maximumValueSum(self, board: List[List[int]]) -> int:
        m, n = len(board), len(board[0])

        # Top-3 cells (by value) in each row: entries (value, col)
        top3row = []
        for row in board:
            idx = sorted(range(n), key=row.__getitem__, reverse=True)[:3]
            top3row.append([(row[c], c) for c in idx])

        # Top-3 cells (by value) in each column: entries (value, row)
        top3col = []
        for c in range(n):
            col = [board[r][c] for r in range(m)]
            idx = sorted(range(m), key=col.__getitem__, reverse=True)[:3]
            top3col.append([(col[r], r) for r in idx])

        rowmax = [t[0][0] for t in top3row]
        rm_top = sorted(range(m), key=lambda r: rowmax[r], reverse=True)[:3]

        best = float('-inf')

        for i in range(m):
            ti = top3row[i]
            for j in range(i + 1, m):
                tj = top3row[j]
                # upper bound prune
                ub = rowmax[i] + rowmax[j]
                for r in rm_top:
                    if r != i and r != j:
                        ub += rowmax[r]
                        break
                if ub <= best:
                    continue
                # g[c] = best value in column c among rows != i, j
                gc = []
                for c, t in enumerate(top3col):
                    v0, r0 = t[0]
                    if r0 != i and r0 != j:
                        gc.append((v0, c))
                    else:
                        v1, r1 = t[1]
                        if r1 != i and r1 != j:
                            gc.append((v1, c))

                            v2, _r2 = t[2]

                        elif g1c != ci and g1c != cj:
                            total = s2 + g1v
                        else:
                            total = s2 + g2v
                        if total > best:
                            best = total
        return best

=== ANSWER ===
Looking at this problem: I need 3 rooks on distinct rows and distinct columns maximizing sum.

**Key insight (exchange argument):** In an optimal solution, the rook in any given row can be assumed to be among the top-3 valued cells of that row (since only 2 other columns are blocked, at least one top-3 cell is available with value ≥ the chosen one). Similarly, for a "third" rook, given 2 excluded rows, the best value in a column lies among that column's top-3 cells.

**Algorithm:** Enumerate every pair of rows (i, j). For the two rooks in these rows, try their top-3 cells (≤9 combos). For the third rook, compute `g[c]` = best value in column c among rows ≠ i, j (using top-3 per column). The best third rook for any 2 blocked columns is among the top-3 columns by `g` value. This is O(m²·n) ≈ 500K core operations.

Correctness: any optimal triple (rows i,j,k) can be transformed (without decreasing value) so the row-i and row-j rooks are in their rows' top-3, and the row-k rook's value is ≤ max over allowed columns of g(c), which my enumeration captures. Every enumerated candidate is feasible, so the maximum equals the optimum.

I verified this against all three examples (including the all-ties case and negative-value cases) plus several hand-constructed edge cases (shared top columns, forced non-top columns, m=3).
