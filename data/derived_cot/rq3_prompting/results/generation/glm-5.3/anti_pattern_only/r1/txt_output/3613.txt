class Solution:
    def maxAmount(self, initialCurrency: str, pairs1: List[List[str]], rates1: List[float], pairs2: List[List[str]], rates2: List[float]) -> float:
        # Collect all currencies
        currencies = {initialCurrency}
        for a, b in pairs1:
            currencies.add(a)
            currencies.add(b)
        for a, b in pairs2:
            currencies.add(a)
            currencies.add(b)
        
        # Day 1: best1[X] = max amount of X starting from 1.0 initialCurrency
        best1 = {initialCurrency: 1.0}
        for _ in range(len(currencies)):
            changed = False
            for (a, b), r in zip(pairs1, rates1):
                if a in best1:
                    v = best1[a] * r
                    if v > best1.get(b, 0.0):
                        best1[b] = v
                        changed = True
                if b in best1:
                    v = best1[b] / r
                    if v > best1.get(a, 0.0):
                        best1[a] = v
                        changed = True
            if not changed:
                break
        
        # Day 2 (reversed): best2[X] = max initialCurrency from 1.0 of X
        # Equivalent to max product from initialCurrency in reversed graph.
        best2 = {initialCurrency: 1.0}
        for _ in range(len(currencies)):
            changed = False
            for (a, b), r in zip(pairs2, rates2):
                # G^R edges: B->A with mult r, A->B with mult 1/r
                if b in best2:
                    v = best2[b] * r
                    if v > best2.get(a, 0.0):
                        best2[a] = v
                        changed = True
                if a in best2:
                    v = best2[a] / r
                    if v > best2.get(b, 0.0):
                        best2[b] = v
                        changed = True
            if not changed:
                break
        
        ans = 0.0
        for c in currencies:
            ans = max(ans, best1.get(c, 0.0) * best2.get(c, 0.0))
        return ans

from typing import List

class Solution:
    def maxAmount(self, initialCurrency: str, pairs1: List[List[str]], rates1: List[float], pairs2: List[List[str]], rates2: List[float]) -> float:
        currencies = {initialCurrency}
        for a, b in pairs1:
            currencies.add(a)
            currencies.add(b)
        for a, b in pairs2:
            currencies.add(a)
            currencies.add(b)
        
        n_iters = len(currencies) + 1
        
        # Day 1: max amount of each currency starting from 1.0 initialCurrency
        best1 = {initialCurrency: 1.0}
        for _ in range(n_iters):
            changed = False
            for (a, b), r in zip(pairs1, rates1):
                if a in best1:
                    v = best1[a] * r
                    if v > best1.get(b, 0.0):
                        best1[b] = v
                        changed = True
                if b in best1:
                    v = best1[b] / r
                    if v > best1.get(a, 0.0):
                        best1[a] = v
                        changed = True
            if not changed:
                break
        
        # Day 2 (reversed): best2[X] = max initialCurrency obtainable from 1.0 of X
        # = max product path from initialCurrency to X in the reversed day-2 graph.
        best2 = {initialCurrency: 1.0}
        for _ in range(n_iters):
            changed = False
            for (a, b), r in zip(pairs2, rates2):
                if b in best2:
                    v = best2[b] * r
                    if v > best2.get(a, 0.0):
                        best2[a] = v
                        changed = True
                if a in best2:
                    v = best2[a] / r
                    if v > best2.get(b, 0.0):
                        best2[b] = v
                        changed = True
            if not changed:
                break
        
        ans = 0.0
        for c in currencies:
            ans = max(ans, best1.get(c, 0.0) * best2.get(c, 0.0))
        return ans

Looking at this problem:

- **Day 1**: compute `best1[X]` = max amount of currency X obtainable from 1.0 of initialCurrency (max-product path via Bellman-Ford, since graphs have no profitable cycles).
- **Day 2**: compute `best2[X]` = max amount of initialCurrency obtainable from 1.0 of X. This equals the max-product path from initialCurrency to X in the **reversed** day-2 graph (reversing each directed edge keeps its multiplier, preserving path products).
- Answer = max over currencies X of `best1[X] * best2[X]` (day-2 value is linear in the amount held, so holding the max of the best intermediate currency is optimal).
