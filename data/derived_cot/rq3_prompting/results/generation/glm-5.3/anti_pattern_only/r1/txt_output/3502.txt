```python
class Solution:
    def numberOfSubstrings(self, s: str, k: int) -> int:
        n = len(s)
        total = n * (n + 1) // 2
        cnt = [0] * 26
        over = 0  # number of chars with cnt >= k
        l = 0
        invalid = 0
        for r, ch in enumerate(s):
            i = ord(ch) - 97
            cnt[i] += 1
            if cnt[i] == k:
                over += 1
            while over > 0:
                j = ord(s[l]) - 97
                cnt[j] -= 1
                if cnt[j] == k - 1:
                    over -= 1
                l += 1
            invalid += r - l + 1
        return total - invalid
```

```python
class Solution:
    def numberOfSubstrings(self, s: str, k: int) -> int:
        n = len(s)
        total = n * (n + 1) // 2
        
        # Count substrings where every character appears at most k-1 times
        # (the complement of what we want), using a sliding window.
        cnt = [0] * 26
        over = 0  # number of characters whose count is >= k in current window
        l = 0
        invalid = 0
        
        for r, ch in enumerate(s):
            i = ord(ch) - 97
            cnt[i] += 1
            if cnt[i] == k:
                over += 1
            
            while over > 0:
                j = ord(s[l]) - 97
                cnt[j] -= 1
                if cnt[j] == k - 1:
                    over -= 1
                l += 1
            
            # window [l, r] has all counts <= k-1
            invalid += r - l + 1
        
        return total - invalid
```
