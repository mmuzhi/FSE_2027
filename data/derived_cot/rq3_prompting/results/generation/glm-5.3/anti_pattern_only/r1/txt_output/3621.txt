class Solution:
    def minOperations(self, nums: List[int], k: int) -> int:
        if min(nums) < k:

        return len(set(x for x in nums if x > k))

=== ANSWER ===
```python
class Solution:
    def minOperations(self, nums: List[int], k: int) -> int:
        # Values can only decrease, so anything below k makes it impossible
        if min(nums) < k:
            return -1
        # Each operation can eliminate at most one distinct value > k,
        # and we can always eliminate them one by one (largest first),
        # so the answer is the count of distinct values greater than k.
        return len(set(x for x in nums if x > k))
```
