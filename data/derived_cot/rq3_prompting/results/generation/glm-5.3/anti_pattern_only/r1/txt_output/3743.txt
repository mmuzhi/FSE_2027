    def maxFreeTime(self, eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
        n = len(startTime)

        for i in range(n):

            prev = endTime[i]
        gaps.append(eventTime - prev)

        window = sum(gaps[:k+1])
        ans = window
        for i in range(k+1, len(gaps)):
            window += gaps[i] - gaps[i - k - 1]
            ans = max(ans, window)
        return ans

from typing import List

class Solution:
    def maxFreeTime(self, eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
        n = len(startTime)
        # Build the n+1 free-time gaps: before first meeting, between meetings, after last meeting
        gaps = [0] * (n + 1)
        prev = 0
        for i in range(n):
            gaps[i] = startTime[i] - prev
            prev = endTime[i]
        gaps[n] = eventTime - prev
        
        # With at most k reschedules, we can merge at most k+1 consecutive gaps:
        # move the k meetings lying between them so they hug one side of the window.
        # Sliding window of size k+1 over the gaps.
        w = k + 1
        window = sum(gaps[:w])
        ans = window
        for i in range(w, n + 1):
            window += gaps[i] - gaps[i - w]
            if window > ans:
                ans = window
        return ans
