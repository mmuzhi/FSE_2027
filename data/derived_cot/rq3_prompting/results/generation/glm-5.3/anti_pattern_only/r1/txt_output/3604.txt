```python
class Solution:
    def numberOfWays(self, n: int, x: int, y: int) -> int:
        MOD = 10**9 + 7
        # Stirling numbers second kind: S(n, k)
        m = min(n, x)
        dp = [0] * (m + 1)
        dp[0] = 1
        for i in range(1, n + 1):
            ndp = [0] * (m + 1)
            for j in range(1, m + 1):
                ndp[j] = (j * dp[j] + dp[j - 1]) % MOD
            dp = ndp
        # factorials
        maxf = 1001
        fact = [1] * (maxf)
        for i in range(1, maxf):
            fact[i] = fact[i-1] * i % MOD
        # C(x, k) = fact[x] / (fact[k]*fact[x-k])
        ans = 0
        yp = 1
        for k in range(1, m + 1):
            yp = yp * y % MOD
            c = fact[x] * pow(fact[k], MOD-2, MOD) % MOD * pow(fact[x-k], MOD-2, MOD) % MOD
            ans = (ans + c * dp[k] % MOD * fact[k] % MOD * yp) % MOD
        return ans % MOD
```

```python
class Solution:
    def numberOfWays(self, n: int, x: int, y: int) -> int:
        MOD = 10**9 + 7
        m = min(n, x)
        
        # dp[k] = Stirling numbers of the second kind S(i, k):
        # ways to partition i performers into k non-empty unlabeled groups
        dp = [0] * (m + 1)
        dp[0] = 1
        for i in range(1, n + 1):
            ndp = [0] * (m + 1)
            for j in range(1, m + 1):
                ndp[j] = (j * dp[j] + dp[j - 1]) % MOD
            dp = ndp
        
        # Precompute factorials
        maxf = 1001
        fact = [1] * maxf
        for i in range(1, maxf):
            fact[i] = fact[i - 1] * i % MOD
        
        ans = 0
        yp = 1  # y^k
        for k in range(1, m + 1):
            yp = yp * y % MOD
            # C(x, k) * k! = x! / (x-k)!  (falling factorial: ordered stage choice)
            ff = fact[x] * pow(fact[x - k], MOD - 2, MOD) % MOD
            ans = (ans + ff * dp[k] % MOD * yp) % MOD
        
        return ans % MOD
```
