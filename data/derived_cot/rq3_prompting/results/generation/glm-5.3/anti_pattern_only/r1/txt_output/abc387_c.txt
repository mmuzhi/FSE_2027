def count_snake(n):
    if n < 10:
        return 0
    s = str(n)
    k = len(s)
    d = [int(ch) for ch in s]
    total = 0
    # numbers with fewer digits (m digits, 2 <= m <= k-1)
    for m in range(2, k):
        total += sum(c ** (m - 1) for c in range(1, 10))
    # k-digit numbers with first digit < d[0]
    for c in range(1, d[0]):
        total += c ** (k - 1)
    # k-digit numbers matching prefix
    for i in range(1, k):
        if i > 1 and d[i-1] >= d[0]:
            break
        choices = min(d[i], d[0])
        total += choices * (d[0] ** (k - 1 - i))
    # N itself
    if all(d[j] < d[0] for j in range(1, k)):
        total += 1
    return total

L, R = map(int, input().split())
print(count_snake(R) - count_snake(L - 1))

Looking at this problem:

- A Snake number with first digit `c` and `m` total digits has each of the remaining `m-1` digits freely chosen from `0..c-1`, giving `c^(m-1)` numbers.
- To count Snake numbers ≤ N: count all Snake numbers with fewer digits than N; count k-digit Snake numbers with first digit < N's first digit; then for k-digit numbers sharing N's prefix, at each position i (while the prefix still satisfies the Snake condition, i.e., all previous digits < first digit), choose a smaller digit at position i (bounded by both `d[i]` and `d[0]`), leaving the rest free (`d[0]^(k-1-i)` options); finally check N itself.
