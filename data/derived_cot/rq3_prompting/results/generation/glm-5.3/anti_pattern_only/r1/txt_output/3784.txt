```python
    def longestCommonPrefix(self, words: List[str], k: int) -> List[int]:
        n = len(words)
        if n - 1 < k:
            return [0] * n
        
        # Build trie
        # children: list of dict
        children = [{}]
        cnt = [0]
        
        for w in words:
            cur = 0
            for ch in w:
                nxt = children[cur].get(ch)
                if nxt is None:
                    nxt = len(children)
                    children[cur][ch] = nxt
                    children.append({})
                    cnt.append(0)
                cur = nxt
                cnt[cur] += 1
        
        m_nodes = len(children)
```

```python
        n = len(words)
        if n - 1 < k:

        # Build trie

                    children[cur][ch] = nxt
                    children.append({})
                    cnt.append(0)
                    depth.append(depth[cur] + 1)

                cnt[cur] += 1

                    a1 = c

                    b2 = fc

            maxf2[v] = b2
```

```python
from typing import List

class Solution:
    def longestCommonPrefix(self, words: List[str], k: int) -> List[int]:
        n = len(words)
        if n - 1 < k:
            return [0] * n
        
        # ---- build trie ----
        children = [{}]
        cnt = [0]
        depth = [0]
        
        for w in words:
            cur = 0
            for ch in w:
                d = children[cur]
                nxt = d.get(ch)
                if nxt is None:
                    nxt = len(children)
                    d[ch] = nxt
                    children.append({})
                    cnt.append(0)
                    depth.append(depth[cur] + 1)
                cur = nxt
                cnt[cur] += 1
        
        m = len(children)
        f = [0] * m       # deepest node with cnt>=k in subtree(v) (0 if none)
        maxf1 = [0] * m   # best f among children
        maxf2 = [0] * m   # second best f among children
        argf = [-1] * m   # child achieving maxf1
        
        for v in range(m - 1, -1, -1):
            b1 = 0; b2 = 0; a1 = -1
            for c in children[v].values():
                fc = f[c]
                if fc > b1:
                    b2 = b1; b1 = fc; a1 = c
                elif fc > b2:
                    b2 = fc
            maxf1[v] = b1; maxf2[v] = b2; argf[v] = a1
            own = depth[v] if cnt[v] >= k else 0
            f[v] = b1 if b1 > own else own
        
        kp1 = k + 1
        ans = []
        for w in words:
            cur = 0
            best_off = 0
            best_on = 0
            for idx, ch in enumerate(w, 1):
                child = children[cur][ch]
                cand = maxf2[cur] if child == argf[cur] else maxf1[cur]
                if cand > best_off:
                    best_off = cand
                cur = child
                if cnt[cur] >= kp1:
                    best_on = idx
            cand = maxf1[cur]
            if cand > best_off:
                best_off = cand
```

```python
    def longestCommonPrefix(self, words: List[str], k: int) -> List[int]:
        n = len(words)
        if n - 1 < k:
            return [0] * n

        # Build trie: children[v] maps char -> child node id
        children = [{}]
        cnt = [0]
        depth = [0]

        for w in words:
            cur = 0
            for ch in w:
                d = children[cur]
                nxt = d.get(ch)
                if nxt is None:
                    nxt = len(children)
                    d[ch] = nxt
                    children.append({})
                    cnt.append(0)
                    depth.append(depth[cur] + 1)
                cur = nxt
                cnt[cur] += 1

        m = len(children)
        f = [0] * m        # deepest depth (cnt>=k node) within subtree of v
        maxf1 = [0] * m    # best child f
        maxf2 = [0] * m    # second best child f
        argf = [-1] * m    # child id achieving maxf1

        for v in range(m - 1, -1, -1):
            b1 = 0
            b2 = 0
            a1 = -1
            for c in children[v].values():
                fc = f[c]
                if fc > b1:
                    b2 = b1
                    b1 = fc
                    a1 = c
                elif fc > b2:
                    b2 = fc
            maxf1[v] = b1
            maxf2[v] = b2
            argf[v] = a1
            own = depth[v] if cnt[v] >= k else 0
            f[v] = b1 if b1 > own else own

        kp1 = k + 1
        res = []
        for w in words:
            cur = 0
            best_off = 0
            best_on = 0

                child = children[cur][ch]
                cand = maxf2[cur] if child == argf[cur] else maxf1[cur]

            res.append(best_off if best_off > best_on else best_on)
```

=== ANSWER ===
Looking at this problem:

- The answer for index `i` is the maximum length `L` such that some prefix of length `L` occurs at least `k` times among the remaining words.
- Build a trie with pass-through counts. Removing word `i` only decrements counts along word `i`'s trie path. So the answer is the max of:
  - **best_on**: deepest node *on* word `i`'s path with `cnt >= k+1` (still ≥ k after removal),
  - **best_off**: deepest node *off* the path with `cnt >= k` (unaffected by removal). Off-path nodes live in sibling subtrees hanging off the path, so precompute for each node `f(v)` = deepest qualifying node in its subtree, plus top-2 child `f` values so we can skip the followed child in O(1) per step.

Verified against both examples and edge cases (duplicates, prefix-words, `k = n`).

```python
from typing import List

class Solution:
    def longestCommonPrefix(self, words: List[str], k: int) -> List[int]:
        n = len(words)
        if n - 1 < k:
            return [0] * n

        # Build trie: children[v] maps char -> child node id
        children = [{}]
        cnt = [0]
        depth = [0]

        for w in words:
            cur = 0
            for ch in w:
                d = children[cur]
                nxt = d.get(ch)
                if nxt is None:
                    nxt = len(children)
                    d[ch] = nxt
                    children.append({})
                    cnt.append(0)
                    depth.append(depth[cur] + 1)
                cur = nxt
                cnt[cur] += 1

        m = len(children)
        f = [0] * m        # deepest node with cnt>=k in subtree(v), 0 if none
        maxf1 = [0] * m    # best child f
        maxf2 = [0] * m    # second best child f
        argf = [-1] * m    # child id achieving maxf1

        # children have larger ids than parents -> reverse order works
        for v in range(m - 1, -1, -1):
            b1 = 0
            b2 = 0
            a1 = -1
            for c in children[v].values():
                fc = f[c]
                if fc > b1:
                    b2 = b1
                    b1 = fc
                    a1 = c
                elif fc > b2:
                    b2 = fc
            maxf1[v] = b1
            maxf2[v] = b2
            argf[v] = a1
            own = depth[v] if cnt[v] >= k else 0
            f[v] = b1 if b1 > own else own

        kp1 = k + 1
        res = []
        for w in words:
            cur = 0
            best_off = 0
            best_on = 0
            for idx, ch in enumerate(w, 1):
                child = children[cur][ch]
                # best f among siblings (children of cur except the one we follow)
                cand = maxf2[cur] if child == argf[cur] else maxf1[cur]
                if cand > best_off:
                    best_off = cand
                cur = child
                if cnt[cur] >= kp1:
                    best_on = idx
            # all children of the final node are off-path
            if maxf1[cur] > best_off:
                best_off = maxf1[cur]
            res.append(best_off if best_off > best_on else best_on)
        return res
```
