```python
class Solution:
    def minimumPairRemoval(self, nums: List[int]) -> int:
        arr = list(nums)
        ops = 0
        # Array is non-decreasing when no adjacent inversion exists.
        # Since n <= 50, at most n-1 merges can occur, so direct
        # simulation is sufficient.
        while any(arr[i] > arr[i + 1] for i in range(len(arr) - 1)):
            # Find the leftmost adjacent pair with the minimum sum.
            best_i = 0
            best_sum = arr[0] + arr[1]
            for i in range(1, len(arr) - 1):
                s = arr[i] + arr[i + 1]
                if s < best_sum:  # strict '<' keeps the leftmost pair on ties
                    best_sum = s
                    best_i = i
            # Replace that pair with its sum.
            arr[best_i:best_i + 2] = [best_sum]
            ops += 1
        return ops
```
