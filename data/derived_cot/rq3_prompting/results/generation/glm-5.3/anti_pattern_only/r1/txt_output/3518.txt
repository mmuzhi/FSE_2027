```python
class Solution:
    def maxScore(self, a: List[int], b: List[int]) -> int:
        NEG = -(10**18)
        # dp[k] = best score having chosen k elements of a so far
        dp = [0, NEG, NEG, NEG, NEG]
        for x in b:
            # update higher k first so x is used at most once per step
            for k in (4, 3, 2, 1):
                v = dp[k - 1] + a[k - 1] * x
                if v > dp[k]:
                    dp[k] = v
        return dp[4]
```
