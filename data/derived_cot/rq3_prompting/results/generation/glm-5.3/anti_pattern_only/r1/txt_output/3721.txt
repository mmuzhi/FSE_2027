class Solution:
    def countMentions(self, numberOfUsers: int, events: List[List[str]]) -> List[int]:
        # Sort by timestamp; at equal timestamps, status changes (OFFLINE)
        # must be processed before MESSAGE events.
        sorted_events = sorted(
            events,
            key=lambda e: (int(e[1]), 0 if e[0] == "OFFLINE" else 1)
        )
        
        mentions = [0] * numberOfUsers
        # offline_until[u]: time at which user u comes back online.
        # User is online at time t iff offline_until[u] <= t.
        offline_until = [0] * numberOfUsers
        
        for etype, ts_str, payload in sorted_events:
            t = int(ts_str)
            if etype == "OFFLINE":
                u = int(payload)
                offline_until[u] = t + 60
            else:  # MESSAGE
                for token in payload.split():
                    if token == "ALL":
                        for u in range(numberOfUsers):
                            mentions[u] += 1
                    elif token == "HERE":
                        for u in range(numberOfUsers):
                            if offline_until[u] <= t:
                                mentions[u] += 1
                    else:
                        # token of the form "id<number>"
                        u = int(token[2:])
                        mentions[u] += 1
        
        return mentions
