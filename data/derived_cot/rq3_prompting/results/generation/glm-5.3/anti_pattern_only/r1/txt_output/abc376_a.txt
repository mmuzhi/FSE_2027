def main():
    import sys
    data = sys.stdin.read().split()
    n = int(data[0])
    c = int(data[1])
    ts = list(map(int, data[2:2+n]))
    
    count = 0
    last = None  # last time candy was received
    for t in ts:
        if last is None or t - last >= c:
            count += 1
            last = t
    print(count)

main()
