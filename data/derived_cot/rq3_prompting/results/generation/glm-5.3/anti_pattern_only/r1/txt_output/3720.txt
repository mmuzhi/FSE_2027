```python
import heapq
from collections import defaultdict

class Solution:
    def minMaxWeight(self, n: int, edges: List[List[int]], threshold: int) -> int:
        # Build reversed adjacency: rev[b].append((a, w)) means in reversed graph b -> a with weight w
        rev = [[] for _ in range(n)]
        for a, b, w in edges:
            rev[b].append((a, w))
        INF = float('inf')
        dist = [INF] * n
        dist[0] = 0
        pq = [(0, 0)]  # (bottleneck, node)
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist[u]:
                continue
            for v, w in rev[u]:
                nd = d if w <= d else w  # nd = max(d, w)
                if nd < dist[v]:
                    dist[v] = nd
                    heapq.heappush(pq, (nd, v))
        ans = max(dist)
        return -1 if ans == INF else ans
```

```python
import heapq
from typing import List

class Solution:
    def minMaxWeight(self, n: int, edges: List[List[int]], threshold: int) -> int:
        # Reverse graph: for original edge a->b (w), reversed edge b->a (w).
        # We need: from node 0 in reversed graph, reach all nodes,
        # minimizing the max edge weight (bottleneck).
        # Dijkstra-like minimax.
        radj = [[] for _ in range(n)]
        for a, b, w in edges:
            radj[b].append((w, a))
        INF = float('inf')
        dist = [INF] * n
        dist[0] = 0
        heap = [(0, 0)]
        while heap:
            d, u = heapq.heappop(heap)
            if d > dist[u]:
                continue
            for w, v in radj[u]:
                nd = w if w > d else d
                if nd < dist[v]:
                    dist[v] = nd
                    heapq.heappush(heap, (nd, v))
        ans = max(dist)
        return -1 if ans == INF else ans
```

```python
import heapq
from typing import List

class Solution:
    def minMaxWeight(self, n: int, edges: List[List[int]], threshold: int) -> int:
        # Reverse every edge: original a->b becomes b->a.
        # "Every node reaches 0 in the original graph" <=> "0 reaches every node in the reversed graph".
        # For any feasible kept subgraph, following a spanning in-tree toward 0 gives
        # every node (except 0) out-degree exactly 1, which satisfies the threshold
        # constraint since threshold >= 1. Hence the threshold never binds, and the
        # answer is the minimum W such that 0 reaches all nodes using edges of weight <= W,
        # i.e., the maximum over nodes of the bottleneck (minimax) distance from 0 in the reversed graph.
        radj = [[] for _ in range(n)]
        for a, b, w in edges:
            radj[b].append((w, a))
        
        INF = float('inf')
        dist = [INF] * n
        dist[0] = 0
        heap = [(0, 0)]
        while heap:
            d, u = heapq.heappop(heap)
            if d > dist[u]:
                continue
            for w, v in radj[u]:
                nd = w if w > d else d
                if nd < dist[v]:
                    dist[v] = nd
                    heapq.heappush(heap, (nd, v))
        
        ans = max(dist)
        return -1 if ans == INF else ans
```

```python
import heapq
from typing import List

class Solution:
    def minMaxWeight(self, n: int, edges: List[List[int]], threshold: int) -> int:
        # Reverse all edges: "every node reaches 0" <=> "0 reaches every node" in reversed graph.
        # The out-degree constraint is always satisfiable alongside reachability:
        # take a spanning in-tree toward 0, where each node != 0 has exactly one
        # outgoing edge (and node 0 has none) — out-degree <= 1 <= threshold.
        # So the answer is the smallest W such that 0 reaches all nodes with edges of
        # weight <= W = max over nodes of the minimax (bottleneck) distance from 0
        # in the reversed graph.
        radj = [[] for _ in range(n)]
        for a, b, w in edges:
            radj[b].append((w, a))
        
        INF = float('inf')
        dist = [INF] * n
        dist[0] = 0
        heap = [(0, 0)]
        pop = heapq.heappop
        push = heapq.heappush
        while heap:
            d, u = pop(heap)
            if d > dist[u]:
                continue
            for w, v in radj[u]:
                nd = w if w > d else d
                if nd < dist[v]:
                    dist[v] = nd
                    push(heap, (nd, v))
        
        ans = max(dist)
        return -1 if ans == INF else ans
```

```python
import heapq
from typing import List

class Solution:
    def minMaxWeight(self, n: int, edges: List[List[int]], threshold: int) -> int:
        # Since threshold >= 1, the out-degree condition never blocks feasibility:
        # whenever every node can reach node 0 using edges of weight <= W, we can keep
        # a spanning in-tree toward 0 in which every node except 0 has exactly one
        # outgoing edge (node 0 has none), i.e. out-degree <= 1 <= threshold.
        #
        # So the task reduces to: find the minimum W such that every node reaches
        # node 0 using only edges of weight <= W.
        #
        # Reverse every edge; then we need node 0 to reach every node, and the answer
        # is the maximum over nodes of the minimax (bottleneck) distance from 0,
        # computed with a Dijkstra-like algorithm.

        if threshold <= 0:
            return 0 if n <= 1 else -1

        radj = [[] for _ in range(n)]
        for a, b, w in edges:
            radj[b].append((w, a))

        INF = float('inf')
        dist = [INF] * n
        dist[0] = 0
        heap = [(0, 0)]
        heappop = heapq.heappop
        heappush = heapq.heappush
        while heap:
            d, u = heappop(heap)
            if d > dist[u]:
                continue
            for w, v in radj[u]:
                nd = w if w > d else d
                if nd < dist[v]:
                    dist[v] = nd
                    heappush(heap, (nd, v))

        ans = max(dist)
        return -1 if ans == INF else ans
```

=== ANSWER ===
Looking at this problem, the key insight is that **the out-degree constraint never blocks feasibility when `threshold >= 1`**: if every node can reach node 0 using edges of weight ≤ W, we can keep a spanning in-tree oriented toward node 0 — in that tree every node except 0 has exactly one outgoing edge (node 0 has none), so all out-degrees are ≤ 1 ≤ threshold. Conversely, any valid graph with max weight W gives every node a path to 0 using only weights ≤ W.

So the answer is simply the smallest W such that every node reaches node 0 using edges of weight ≤ W. Reversing all edges, this becomes: node 0 must reach every node, and the answer equals the **maximum bottleneck (minimax) distance** from node 0 in the reversed graph — computable with a Dijkstra-like sweep where the combined cost of an edge is `max(dist[u], w)`.

Verification on examples:
- Ex1: bottlenecks [0,1,1,1,1] → 1 ✓
- Ex2: node 0 has no incoming edges → unreachable → -1 ✓
- Ex3: bottlenecks [0,2,2,2,1] → 2 ✓
- Ex4: nodes 2,3 cannot reach 0 → -1 ✓

```python
        # has exactly one outgoing edge and node 0 has none (out-degree <= 1 <= threshold).

        # so the answer is the smallest W such that all nodes reach node 0 with weight <= W.
        #
        # Reverse all edges: "every node reaches 0" <=> "0 reaches every node".
        # The answer is the maximum minimax (bottleneck) distance from 0 in the reversed graph.
```
