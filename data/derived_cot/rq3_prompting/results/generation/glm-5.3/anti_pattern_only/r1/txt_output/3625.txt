=== ANSWER ===
```python
class Solution:
    def canAliceWin(self, n: int) -> bool:
        stones = 10          # amount the current player must remove
        remaining = n        # stones left in the pile
        is_alice = True      # whose turn it is right now
        
        # The player to move keeps playing while they can take exactly `stones`
        while stones > 0 and remaining >= stones:
            remaining -= stones
            stones -= 1
            is_alice = not is_alice
        
        # Loop exited: the player whose turn it is now cannot make a move and loses
        return not is_alice
```
