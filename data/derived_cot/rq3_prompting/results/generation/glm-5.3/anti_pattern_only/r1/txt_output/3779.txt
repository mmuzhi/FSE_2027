```python
class Solution:
    def maxWeight(self, pizzas: List[int]) -> int:
        pizzas.sort(reverse=True)
        k = len(pizzas) // 4          # total days
        odd = (k + 1) // 2            # odd-numbered days (gain the max Z)
        even = k // 2                 # even-numbered days (gain the second max Y)
        
        # Odd days: take the largest `odd` pizzas, each as the max of its group
        # (paired with 3 smallest fillers).
        total = sum(pizzas[:odd])
        
        # Even days: pair up the next 2*even pizzas consecutively;
        # the larger of each pair is the untouched Z, the smaller is gained Y.
        # Gains are at indices odd+1, odd+3, ..., odd+2*even-1.
        total += sum(pizzas[odd + 1 : odd + 2 * even : 2])
        
        return total
```
