class Solution:
    def findAnswer(self, parent: List[int], s: str) -> List[bool]:
        n = len(parent)
        children = [[] for _ in range(n)]
        for i in range(1, n):
            children[parent[i]].append(i)
        
        # BFS order (parents before children)
        order = [0]
        idx = 0
        while idx < len(order):
            v = order[idx]
            idx += 1
            order.extend(children[v])
        
        # subtree sizes
        size = [1] * n
        for v in reversed(order):
            p = parent[v]
            if p != -1:
                size[p] += size[v]
        
        # start positions of each subtree block in postorder string
        start = [0] * n
        for v in order:
            b = start[v]
            for c in children[v]:
                start[c] = b
                b += size[c]
        
        # postorder string P: node v's char is at start[v] + size[v] - 1
        P = [0] * n
        for v in range(n):
            P[start[v] + size[v] - 1] = ord(s[v])
        
        # Manacher
        # d1[i]: max k with P[i-k+1..i+k-1] palindrome (odd, length 2k-1)
        d1 = [0] * n
        l, r = 0, -1
        for i in range(n):
            k = 1 if i > r else min(d1[l + r - i], r - i + 1)
            while i - k >= 0 and i + k < n and P[i - k] == P[i + k]:
                k += 1
            d1[i] = k
            if i + k - 1 > r:
                l = i - k + 1
                r = i + k - 1
        
        # d2[i]: max k with P[i-k..i+k-1] palindrome (even, length 2k)
        d2 = [0] * n
        l, r = 0, -1
        for i in range(n):
            k = 0 if i > r else min(d2[l + r - i + 1], r - i + 1)
            while i + k < n and i - k - 1 >= 0 and P[i + k] == P[i - k - 1]:
                k += 1
            d2[i] = k
            if i + k - 1 > r:
                l = i - k
                r = i + k - 1

            else:
                c = lo + (m >> 1)

class Solution:
    def findAnswer(self, parent: List[int], s: str) -> List[bool]:
        n = len(parent)
        children = [[] for _ in range(n)]
        for i in range(1, n):
            children[parent[i]].append(i)

        # BFS order: parents before children
        order = [0]
        idx = 0
        while idx < len(order):
            order.extend(children[order[idx]])
            idx += 1

        # subtree sizes
        size = [1] * n
        for v in reversed(order):
            p = parent[v]
            if p >= 0:
                size[p] += size[v]

        # start position of each subtree's block in the postorder string
        start = [0] * n
        for v in order:
            b = start[v]
            for c in children[v]:
                start[c] = b
                b += size[c]

        # postorder string as list of char codes
        P = [0] * n
        for v in range(n):
            P[start[v] + size[v] - 1] = ord(s[v])

        # Manacher: d1[i] = max k such that P[i-k+1..i+k-1] is a palindrome
        d1 = [1] * n
        l, r = 0, -1
        for i in range(n):
            if i > r:
                k = 1
            else:
                k = d1[l + r - i]
                t = r - i + 1
                if k > t:
                    k = t
            while i - k >= 0 and i + k < n and P[i - k] == P[i + k]:
                k += 1
            d1[i] = k
            if i + k - 1 > r:
                l = i - k + 1
                r = i + k - 1

        # Manacher: d2[i] = max k such that P[i-k..i+k-1] is a palindrome
        d2 = [0] * n
        l, r = 0, -1
        for i in range(n):
            if i > r:
                k = 0
            else:
                k = d2[l + r - i + 1]
                t = r - i + 1
                if k > t:
                    k = t
            while i + k < n and i - k - 1 >= 0 and P[i + k] == P[i - k - 1]:
                k += 1
            d2[i] = k
            if i + k - 1 > r:
                l = i - k
                r = i + k - 1

        ans = [False] * n
        for v in range(n):
            lo = start[v]
            m = size[v]
            half = m >> 1
            c = lo + half
            if m & 1:
                ans[v] = (d1[c] << 1) - 1 >= m
            else:
                ans[v] = (d2[c] << 1) >= m
        return ans

        idx = 0

            idx += 1

        size = [1] * n
        for v in reversed(order):

            if p >= 0:
                size[p] += size[v]

        start = [0] * n
        for v in order:
            b = start[v]
            for c in children[v]:
                start[c] = b
                b += size[c]

        # full postorder string P; node v is the last char of its block
        P = [0] * n

            P[start[v] + size[v] - 1] = ord(s[v])

        # Manacher (odd): d1[i] = max k with P[i-k+1 .. i+k-1] palindrome
        d1 = [1] * n
        l, r = 0, -1
        for i in range(n):

                if k > lim:

                k += 1
            d1[i] = k
            if i + k - 1 > r:

                r = i + k - 1

Looking at this problem:

- `dfs(i)` produces the **postorder string of the subtree rooted at i** (children in increasing order, then node's own char).
- Every subtree's postorder is a **contiguous block** of the full postorder string `P` of the whole tree. Node `v` is the last character of its block.
- So compute sizes, block start positions, build `P`, then answer "is `P[l..r]` a palindrome" for each node using **Manacher's algorithm** (deterministic, O(n)): a substring is a palindrome iff the maximal palindrome at its center covers it.

I verified this against both examples (e.g., example 1 gives `P = "abaaba"`, node 0 is the full string → true; node 2 is `"ab"` → false) and against tricky cases like `"a"+"aa"+"a" = "aaaa"` where block lengths are asymmetric but the string is still a palindrome (Manacher handles content correctly, unlike naive block-symmetry checks).

Key implementation points:
- Children lists built in increasing index order automatically.
- BFS order gives parents-before-children; reversed for size accumulation, forward for start positions.
- `d1[i]` = max k with `P[i-k+1..i+k-1]` palindrome (odd), `d2[i]` = max k with `P[i-k..i+k-1]` palindrome (even). Query: odd length m needs `2*d1[c]-1 >= m`; even needs `2*d2[c] >= m` at center `c = l + m/2`.
