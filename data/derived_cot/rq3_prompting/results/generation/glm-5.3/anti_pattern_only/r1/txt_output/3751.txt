class Solution:
    def maxFrequency(self, nums: List[int], k: int) -> int:
        total_k = 0
        dec = 0
        # anchor[v], anchor_dec[v] for v in 1..50, v != k
        anchor = [0] * 51
        anchor_dec = [0] * 51
        best = [0] * 51

                dec += 1
                total_k += 1
            else:
                # real = max(0, anchor[u] - (dec - anchor_dec[u])) + 1

                if r < 0:
                    r = 0
                r += 1
                if r > best[u]:
                    best[u] = r
                anchor[u] = r
                anchor_dec[u] = dec
        
        return total_k + max(best)

class Solution:
    def maxFrequency(self, nums: List[int], k: int) -> int:
        total_k = 0
        dec = 0
        anchor = [0] * 51
        anchor_dec = [0] * 51
        best = [0] * 51

                dec += 1
                total_k += 1
            else:

                if r < 0:
                    r = 0
                r += 1
                if r > best[u]:
                    best[u] = r
                anchor[u] = r
                anchor_dec[u] = dec
        
        return total_k + max(best)

=== ANSWER ===
Looking at this problem:

- After adding `x` to subarray `[i..j]`, the count of `k` equals: (k's outside the subarray) + (elements inside equal to `k - x`).
- So for each candidate value `v = k - x`, the gain over the baseline (count of `k`, achievable with `x = 0`) is `max over subarrays of (count of v in subarray) − (count of k in subarray)` — a Kadane problem on the array `d` where `d[t] = +1` if `nums[t] == v`, `-1` if `nums[t] == k`, else `0`.

Since a `k` position decrements the running Kadane value for **every** `v`, I handle all 49 candidate values in one pass using a lazy global decrement counter: for value `v`, store `anchor[v]` (its running value when last touched) and `anchor_dec[v]` (decrement count at that time). Its true running value is `max(0, anchor[v] - (dec - anchor_dec[v]))`. New maxima only occur at `+1` steps (positions of `v`), so `best` is updated only there. This gives O(n) total time.

Verified against both examples (outputs 2 and 4) and several hand-traced edge cases (all elements = k, no k present, interleaved k's where clamping at 0 matters).

```python
from typing import List

class Solution:
    def maxFrequency(self, nums: List[int], k: int) -> int:
        total_k = 0   # total count of k (baseline via x = 0)
        dec = 0       # number of k's seen so far (global decrement for all values)
        
        # For each value v (1..50, v != k):
        #   real running Kadane value = max(0, anchor[v] - (dec - anchor_dec[v]))
        #   updated (+1) only at positions where nums[t] == v
        anchor = [0] * 51
        anchor_dec = [0] * 51
        best = [0] * 51  # best[v] = max subarray sum of (+1 at v, -1 at k), floored at 0
        
        for u in nums:
            if u == k:
                dec += 1
                total_k += 1
            else:
                # running value before this +1 step
                r = anchor[u] - dec + anchor_dec[u]
                if r < 0:
                    r = 0
                r += 1
                if r > best[u]:
                    best[u] = r
                anchor[u] = r
                anchor_dec[u] = dec
        
        return total_k + max(best)
```
