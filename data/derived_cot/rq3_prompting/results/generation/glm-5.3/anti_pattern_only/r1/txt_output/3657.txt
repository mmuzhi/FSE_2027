from typing import List

class Solution:
    def checkValidCuts(self, n: int, rectangles: List[List[int]]) -> bool:
        def can_cut(intervals):
            intervals.sort()
            blocks = 0
            cur_end = -1
            for s, e in intervals:
                if s < cur_end:
                    if e > cur_end:
                        cur_end = e
                else:
                    blocks += 1
                    cur_end = e
                    if blocks >= 3:
                        return True
            return False
        
        return (can_cut([(r[0], r[2]) for r in rectangles]) or
                can_cut([(r[1], r[3]) for r in rectangles]))
