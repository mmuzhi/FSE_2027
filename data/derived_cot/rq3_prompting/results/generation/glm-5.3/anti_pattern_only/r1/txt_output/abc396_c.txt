```python
import sys

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0]); m = int(data[1])
    B = list(map(int, data[2:2+n]))
    W = list(map(int, data[2+n:2+n+m]))
    B.sort(reverse=True)
    W.sort(reverse=True)
    
    # prefix sums of B
    pb = [0]*(n+1)
    for i in range(n):
        pb[i+1] = pb[i] + B[i]
    
    # prefix sums of W and prefix max (including 0 at index 0)
    pw = [0]*(m+1)
    pm = [0]*(m+1)
    cur = 0
    for j in range(m):
        pw[j+1] = pw[j] + W[j]
        cur = max(cur, pw[j+1])
        pm[j+1] = cur
    
    ans = 0
    for i in range(n+1):
        j = min(i, m)
        ans = max(ans, pb[i] + pm[j])
    print(ans)

main()
```

=== ANSWER ===
Looking at this problem:

- Sort black values descending, sort white values descending.
- If we choose exactly `i` blacks, the best choice is the top `i` blacks (prefix sum `PB[i]`). If we choose `j` whites, best is top `j` whites (prefix sum `PW[j]`), with constraint `j ≤ min(i, M)`.
- For each `i` (number of blacks from 0 to N), the best white contribution is the maximum prefix sum of whites over lengths `0..min(i,M)` (taking 0 whites is allowed, so include 0 as a floor).
- Answer is max over `i` of `PB[i] + prefixMax(PW)[min(i,M)]`.

Verification with samples:
- Sample 1: PB=[0,8,13,16,15], PW=[0,3,1,-3], pm=[0,3,3,3] → max is 16+3=19 ✓
- Sample 2: PB=[0,5,3,-2,-12], PW=[0,8,12,13] → max is 3+12=15 ✓
- Sample 3: all PB negative even with best whites → 0 ✓

This runs in O((N+M) log(N+M)) which is fine for 2×10^5.

```python
import sys
from itertools import accumulate

def main():
    data = sys.stdin.buffer.read().split()
    n, m = int(data[0]), int(data[1])
    B = sorted(map(int, data[2:2 + n]), reverse=True)
    W = sorted(map(int, data[2 + n:2 + n + m]), reverse=True)

    pb = [0] + list(accumulate(B))
    pw = [0] + list(accumulate(W))

    # running maximum of white prefix sums (pm[0] = 0 means choosing no whites)
    pm = list(accumulate(pw, max))

    ans = 0
    for i in range(n + 1):
        cand = pb[i] + pm[min(i, m)]
        if cand > ans:
            ans = cand
    print(ans)

main()
```
