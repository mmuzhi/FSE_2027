import sys

def main():
    input_data = sys.stdin.buffer.read().split()
    N = int(input_data[0])
    A = list(map(int, input_data[1:1+N]))
    MOD = 998244353
    if A[0] != 1:

        return

    B = []
    cur = A[0]
    cnt = 0
    for a in A:
        if a == cur:
            cnt += 1
        else:
            B.append(cnt)
            cur = a
            cnt = 1
    B.append(cnt)
    m = len(B)

    if any(b % 2 == 0 for b in B):
        print(0)
        return

    # precompute factorials up to N
    maxn = N + 10
    fact = [1] * (maxn)
    for i in range(1, maxn):
        fact[i] = fact[i-1] * i % MOD
    inv_fact = [1] * maxn
    inv_fact[maxn-1] = pow(fact[maxn-1], MOD-2, MOD)
    for i in range(maxn-1, 0, -1):
        inv_fact[i-1] = inv_fact[i] * i % MOD
    inv2 = pow(2, MOD-2, MOD)

        t = b >> 1

        c = fact[2*t] * inv_fact[t] % MOD * inv_fact[t] % MOD
        ans = ans * c % MOD
    ans = ans * pow(inv2, I, MOD) % MOD

def main():
    data = sys.stdin.buffer.read().split()
    N = int(data[0])
