```python
class Solution:
    def resultsArray(self, nums: List[int], k: int) -> List[int]:
        n = len(nums)
        cnt = [1] * n
        for i in range(1, n):
            if nums[i] == nums[i-1] + 1:
                cnt[i] = cnt[i-1] + 1
        res = []
        for j in range(k - 1, n):
            if cnt[j] >= k:
                res.append(nums[j])
            else:
                res.append(-1)
        return res
```

```python
class Solution:
    def resultsArray(self, nums: List[int], k: int) -> List[int]:
        n = len(nums)
        # cnt[i] = length of consecutive ascending run ending at index i
        cnt = [1] * n
        for i in range(1, n):
            if nums[i] == nums[i - 1] + 1:
                cnt[i] = cnt[i - 1] + 1
        
        res = []
        # window ending at j (starting at j - k + 1) is valid iff cnt[j] >= k
        for j in range(k - 1, n):
            if cnt[j] >= k:
                res.append(nums[j])
            else:
                res.append(-1)
        return res
```
