import numpy as np

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    a = np.array(data[1:1 + n], dtype=np.int64)

    asum = int(a.sum())
    answer = (n + 1) * asum

    max_a = int(a.max())
    kmax = (2 * max_a).bit_length() - 1

    for k in range(1, kmax + 1):
        m = 1 << k
        b = a % m
        half = m >> 1

        if m <= (1 << 20):
            cnt = np.bincount(b, minlength=m)
            t = m - b
            t[b == 0] = 0
            vals = cnt[t]
        else:
            u, counts = np.unique(b, return_counts=True)
            t = (m - b) % m
            pos = np.searchsorted(u, t)
            pos = np.minimum(pos, len(u) - 1)
            vals = np.where(u[pos] == t, counts[pos], 0)

        s = int((a * vals).sum())
        s += int(a[b == 0].sum())
        s += int(a[b == half].sum())
        answer -= s // m

    print(answer)

main()

- $f(x) = x \cdot 2^{-v_2(x)}$, and using $2^{-v} = 1 - \sum_{k=1}^{v} 2^{-k}$:
$$\sum_{i\le j} f(A_i{+}A_j) = \underbrace{(N{+}1)\sum A_i}_{\sum_{i\le j}(A_i+A_j)} - \sum_{k\ge 1}\frac{S_k}{2^k}$$
where $S_k = \sum_{i\le j,\ 2^k \mid A_i+A_j}(A_i+A_j)$.

$$S_k = \sum_i A_i\cdot \text{cnt}[(-B_i)\bmod 2^k] + \text{sum}[0] + \text{sum}[2^{k-1}]$$
