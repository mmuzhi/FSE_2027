    n = int(data[0])
    h = [int(x) for x in data[1:1 + n]]

    for j in range(n):

        while stack and h[stack[-1]] < hj:

        hi = j - 1

            diff[lo] += 1
            diff[hi + 1] -= 1
        stack.append(j)
