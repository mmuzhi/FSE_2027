import sys

def main():
    data = sys.stdin.buffer.read().split()
    S = data[0].decode()
    Q = int(data[1])
    N = len(S)
    out = []
    for j in range(Q):
        K = int(data[2 + j])
        i = K - 1
        a, b = divmod(i, N)
        c = S[b]
        if bin(a).count('1') & 1:
            c = c.lower() if c.isupper() else c.upper()
        out.append(c)
    print(' '.join(out))

main()

            c = c.swapcase()
        res.append(c)
    sys.stdout.write(' '.join(res) + '\n')
