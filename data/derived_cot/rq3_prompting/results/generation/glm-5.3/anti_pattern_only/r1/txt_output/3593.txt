=== ANSWER ===
```python
from math import gcd
from functools import reduce
from typing import List

class Solution:
    def maxScore(self, nums: List[int]) -> int:
        def lcm(a: int, b: int) -> int:
            return a * b // gcd(a, b)

        def score(arr: List[int]) -> int:
            if not arr:
                return 0
            g = reduce(gcd, arr)
            l = reduce(lcm, arr)
            return g * l

        best = score(nums)
        n = len(nums)
        for i in range(n):
            rest = nums[:i] + nums[i + 1:]
            best = max(best, score(rest))
        return best
```
