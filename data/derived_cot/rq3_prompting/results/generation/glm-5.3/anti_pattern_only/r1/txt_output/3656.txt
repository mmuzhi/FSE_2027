class Solution:
    def minimumOperations(self, nums: List[int]) -> int:
        # Find the longest suffix of nums that contains all distinct elements
        # by scanning from the end backwards until we hit a duplicate.
        seen = set()
        i = len(nums) - 1
        while i >= 0 and nums[i] not in seen:
            seen.add(nums[i])
            i -= 1
        
        # nums[i+1:] is the longest distinct suffix.
        # We must remove the first (i+1) elements; each operation removes 3.
        # So the minimum number of operations is ceil((i+1) / 3).
        return (i + 1 + 2) // 3
