```python
from fractions import Fraction

class Solution:
    def minDamage(self, power: int, damage: List[int], health: List[int]) -> int:
        n = len(damage)
        # time to kill each enemy
        enemies = []
        for i in range(n):
            t = (health[i] + power - 1) // power
            enemies.append((t, damage[i]))
        # sort by damage/time descending, using cross-multiplication comparator
        # equivalently: t_i * d_j < t_j * d_i means i first
        import functools
        def cmp(a, b):
            # a = (t_a, d_a), b = (t_b, d_b)
            # a before b if t_a * d_b < t_b * d_a
            left = a[0] * b[1]
            right = b[0] * a[1]
            if left < right:
                return -1
            elif left > right:
                return 1
            return 0
        enemies.sort(key=functools.cmp_to_key(cmp))
        total_dps = sum(damage)
        ans = 0
        for t, d in enemies:
            ans += t * total_dps
            total_dps -= d
        return ans
```

```python
import functools

class Solution:
    def minDamage(self, power: int, damage: List[int], health: List[int]) -> int:
        n = len(damage)
        times = [(health[i] + power - 1) // power for i in range(n)]
        idx = list(range(n))
        def cmp(i, j):
            # i before j if times[i]*damage[j] < times[j]*damage[i]
            a = times[i] * damage[j]
            b = times[j] * damage[i]
            if a < b:
                return -1
            if a > b:
                return 1
            return 0
        idx.sort(key=functools.cmp_to_key(cmp))
        total_dps = sum(damage)
        ans = 0
        for i in idx:
            ans += times[i] * total_dps
            total_dps -= damage[i]
        return ans
```

=== ANSWER ===
Looking at this problem:

- Each enemy needs `t_i = ceil(health[i] / power)` seconds of attacks to die.
- Total damage = Σ over each second of the DPS of all still-alive enemies. Equivalently, for a fixed kill order, when attacking enemy `i` for `t_i` seconds, all enemies not yet killed (including `i`) deal damage, so damage += `t_i * (sum of dps of remaining enemies)`.

**Exchange argument:** For two enemies `i`, `j`, killing `i` first costs `t_i·d_i + t_i·d_j + t_j·d_j` while `j` first costs `t_j·d_j + t_j·d_i + t_i·d_i`. Killing `i` first is better iff `t_i·d_j < t_j·d_i`, i.e., `d_i/t_i > d_j/t_j`. So sort enemies in decreasing order of damage-per-kill-time (using exact `Fraction` comparison to avoid floating point issues), then accumulate damage.

**Verification with Example 1:** power=4 → times=[1,2,2,2], ratios=[1,1,1.5,2]. Order: 3,2,0,1. Damage: 2·10 + 2·6 + 1·3 + 2·2 = 20+12+3+4 = 39 ✓

**Example 2:** times=[1,2,3,4], ratios=[1,.5,.33,.25]. Order 0,1,2,3: 1·4+2·3+3·2+4·1 = 4+6+6+4 = 20 ✓

**Example 3:** time = ceil(59/8) = 8, damage = 8·40 = 320 ✓

Complexity: O(n log n), well within limits for n = 10^5.

```python
from typing import List
from fractions import Fraction

class Solution:
    def minDamage(self, power: int, damage: List[int], health: List[int]) -> int:
        n = len(damage)
        times = [(health[i] + power - 1) // power for i in range(n)]
        # Kill enemies with highest damage-per-second-of-attack first (exact ratio comparison)
        order = sorted(range(n), key=lambda i: Fraction(damage[i], times[i]), reverse=True)
        total_dps = sum(damage)
        ans = 0
        for i in order:
            ans += times[i] * total_dps
            total_dps -= damage[i]
        return ans
```
