```python
class Solution:
    def subarraySum(self, nums: List[int]) -> int:
        n = len(nums)
        # prefix sums: P[k] = sum of nums[0..k-1]
        P = [0] * (n + 1)
        for k in range(n):
            P[k + 1] = P[k] + nums[k]
        
        total = 0
        for i in range(n):
            start = max(0, i - nums[i])
            total += P[i + 1] - P[start]
        return total
```
