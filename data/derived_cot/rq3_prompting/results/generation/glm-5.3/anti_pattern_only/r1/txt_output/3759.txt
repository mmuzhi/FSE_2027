class Solution:
    def findMaxSum(self, nums1: List[int], nums2: List[int], k: int) -> List[int]:
        n = len(nums1)
        order = sorted(range(n), key=lambda i: nums1[i])
        heap = []  # min-heap of size at most k
        heap_sum = 0
        answer = [0] * n
        i = 0
        while i < n:
            j = i
            val = nums1[order[i]]
            # answer all indices in this group
            while j < n and nums1[order[j]] == val:
                answer[order[j]] = heap_sum
                j += 1
            # add group's nums2 values to heap
            for t in range(i, j):
                v = nums2[order[t]]
                if len(heap) < k:
                    heapq.heappush(heap, v)
                    heap_sum += v
                elif v > heap[0]:
                    heap_sum += v - heapq.heapreplace(heap, v)
            i = j
        return answer

import heapq
from typing import List

class Solution:
    def findMaxSum(self, nums1: List[int], nums2: List[int], k: int) -> List[int]:
        n = len(nums1)
        order = sorted(range(n), key=lambda i: nums1[i])
        
        heap = []        # min-heap holding the k largest nums2 values among strictly smaller nums1 elements
        heap_sum = 0     # running sum of heap contents
        answer = [0] * n
        
        i = 0
        while i < n:
            val = nums1[order[i]]
            j = i
            # All indices with the same nums1 value share the same answer
            # (computed from strictly smaller elements processed so far)
            while j < n and nums1[order[j]] == val:
                answer[order[j]] = heap_sum
                j += 1
            # Now add this group's nums2 values into the top-k heap
            for t in range(i, j):
                v = nums2[order[t]]
                if len(heap) < k:
                    heapq.heappush(heap, v)
                    heap_sum += v
                elif v > heap[0]:
                    heap_sum += v - heapq.heapreplace(heap, v)
            i = j
        
        return answer
