```python
class Solution:
    def maxSum(self, grid: List[List[int]], limits: List[int], k: int) -> int:
        candidates = []
        for row, lim in zip(grid, limits):
            if lim > 0:
                row_sorted = sorted(row, reverse=True)
                candidates.extend(row_sorted[:lim])
        candidates.sort(reverse=True)
        return sum(candidates[:k])
```

```python
class Solution:
    def maxSum(self, grid: List[List[int]], limits: List[int], k: int) -> int:

        for row, lim in zip(grid, limits):
```

=== ANSWER ===
```python
import heapq

class Solution:
    def maxSum(self, grid: List[List[int]], limits: List[int], k: int) -> int:
        # For each row, only its top limits[i] elements can ever be useful,
        # since we take at most limits[i] elements from that row.
        candidates = []
        for row, lim in zip(grid, limits):
            if lim > 0:
                candidates.extend(heapq.nlargest(lim, row))
        # All values are non-negative, so taking the k largest candidates
        # (k <= sum(limits) guarantees enough candidates) is optimal.
        return sum(heapq.nlargest(k, candidates))
```
