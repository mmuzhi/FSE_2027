```python
class Solution:
    def minStartingIndex(self, s: str, pattern: str) -> int:
        n, m = len(s), len(pattern)
        
        def z_function(t):
            L = len(t)
            z = [0] * L
            z[0] = L
            l = r = 0
            for i in range(1, L):
                zi = 0
                if i < r:
                    zi = min(r - i, z[i - l])
                while i + zi < L and t[zi] == t[i + zi]:
                    zi += 1
                z[i] = zi
                if i + zi > r:
                    l = i
                    r = i + zi
            return z
        
        z1 = z_function(pattern + '#' + s)
        z2 = z_function(pattern[::-1] + '#' + s[::-1])
        
        for i in range(n - m + 1):
            if z1[m + 1 + i] + z2[n + 1 - i] >= m - 1:
                return i
        return -1
```

```python
class Solution:
    def minStartingIndex(self, s: str, pattern: str) -> int:
        n, m = len(s), len(pattern)

        def z_function(t):
            L = len(t)
            z = [0] * L
            z[0] = L
            l = r = 0
            for i in range(1, L):
                zi = 0
                if i < r:
                    zi = z[i - l]
                    if zi > r - i:
                        zi = r - i
                while i + zi < L and t[zi] == t[i + zi]:
                    zi += 1
                z[i] = zi
                if i + zi > r:
                    l = i
                    r = i + zi
            return z

        # z1: common prefix lengths of s[i:] with pattern
        z1 = z_function(pattern + '#' + s)
        # z2: common prefix lengths of reversed s with reversed pattern
        z2 = z_function(pattern[::-1] + '#' + s[::-1])

        # For window starting at i:
        #   lcp = z1[m + 1 + i]
        #   lcs = z2[n + 1 - i]
        for i in range(n - m + 1):
            if z1[m + 1 + i] + z2[n + 1 - i] >= m - 1:
                return i
        return -1
```

=== ANSWER ===
Looking at this problem, I need to find the smallest starting index where the window of length `m = len(pattern)` differs from `pattern` in at most 1 position.

**Key insight:** A window starting at `i` has at most 1 mismatch with `pattern` if and only if `lcp(i) + lcs(i) >= m - 1`, where:
- `lcp(i)` = longest common prefix of `s[i:]` and `pattern` (via Z-algorithm on `pattern + '#' + s`)
- `lcs(i)` = longest common suffix of the window and `pattern` (via Z-algorithm on reversed strings)

If the prefix match covers positions `[0, lcp)` and suffix match covers `[m-lcs, m)`, the uncovered gap has size `m - lcp - lcs <= 1`, meaning at most one position needs changing. Conversely, one mismatch at position `p` gives `lcp >= p` and `lcs >= m-1-p`.

For the reversed Z-array: window starting at `i` ends at `i+m-1`; its reversed suffix starts at position `n-i-m` in reversed `s`, so `lcs(i) = z2[m+1 + (n-m-i)]`, which is a reversed slice of `z2`.

I verified against all four examples (e.g., example 1: `lcp[1]=3`, `lcs[1]=2`, `3+2=5>=5` → index 1; example 3: both windows fail → -1). For `m=1`, `need=0` so index 0 is always returned, which is correct since any single character can be changed.

Complexity: O(n + m) time and space.

```python
class Solution:
    def minStartingIndex(self, s: str, pattern: str) -> int:
        n, m = len(s), len(pattern)

        def z_function(t):
            L = len(t)
            z = [0] * L
            z[0] = L
            l = r = 0
            for i in range(1, L):
                zi = 0
                if i < r:
                    zi = z[i - l]
                    rem = r - i
                    if zi > rem:
                        zi = rem
                while i + zi < L and t[zi] == t[i + zi]:
                    zi += 1
                z[i] = zi
                if i + zi > r:
                    l = i
                    r = i + zi
            return z

        # z1[m+1+i] = longest common prefix of s[i:] and pattern
        z1 = z_function(pattern + '#' + s)
        # z2[m+1+j] = longest common prefix of reversed(s)[j:] and reversed(pattern)
        z2 = z_function(pattern[::-1] + '#' + s[::-1])

        # lcp[i]: common prefix of window starting at i and pattern
        lcp = z1[m + 1: n + 2]
        # lcs[i]: common suffix of window starting at i and pattern
        # window i -> reversed-s start index n-m-i -> z2[m+1+(n-m-i)]; slicing + reversing maps it
        lcs = z2[m + 1: n + 2][::-1]

        need = m - 1
        for i in range(n - m + 1):
            if lcp[i] + lcs[i] >= need:
                return i
        return -1
```
