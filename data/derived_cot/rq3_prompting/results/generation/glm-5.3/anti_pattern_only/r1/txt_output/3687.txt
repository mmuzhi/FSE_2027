    def longestSpecialPath(self, edges: List[List[int]], nums: List[int]) -> List[int]:
        n = len(nums)

            adj[u].append((v, l))
            adj[v].append((u, l))
        
        max_val = max(nums)

        stack = [(0, -1, 0, True)]

            if is_enter:

                d = len(path_prefix) - 1  # not used directly

from typing import List

class Solution:
    def longestSpecialPath(self, edges: List[List[int]], nums: List[int]) -> List[int]:
        n = len(nums)
        adj = [[] for _ in range(n)]
        for u, v, w in edges:
            adj[u].append((v, w))
            adj[v].append((u, w))

        prefix = []   # prefix[i] = total edge length from root to path position i

        #   (1, node, parent, edge_len)      -> enter node
        #   (0, val, prev_occ, prev_start)   -> restore state after node's subtree
        stack = [(1, 0, -1, 0)]
        
        while stack:
            tag, a, b, c = stack.pop()
            if tag:
                node, parent, e = a, b, c
                pos = len(prefix)
                prefix.append((prefix[-1] if pos else 0) + e)
                v = nums[node]
                p = last_occ[v]
                old_start = start
                if p >= start:

                last_occ[v] = pos
                cur_len = prefix[pos] - prefix[start]
                cur_cnt = pos - start + 1

=== ANSWER ===
Looking at this problem:

- A special path is a downward (ancestor → descendant) path with all unique node values; its length is the sum of edge lengths.
- DFS from the root maintaining the current root-path. For each node at path position `pos`, the best special path ending there is the maximal window `[start, pos]` where all values are unique — maintained via a "last occurrence" map (classic sliding window along the root path). Since edge lengths are ≥ 1, only the maximal window per end node can tie the global maximum length (sub-windows are strictly shorter), so per node we only need to check that one window for both length and node-count tie-breaking.
- Because values must be unique only within the window, when arriving at a node whose value last occurred at position `p`, the window start moves to `p+1` if `p >= start`. On backtracking, restore the previous `last_occ`, window start, and prefix array (done iteratively with enter/restore events to avoid recursion depth issues).

I verified the logic against both examples (Example 1 yields `[6,2]` via path 2→5, Example 2 yields `[0,1]`), plus edge cases like all-equal values (`[0,1]`) and duplicates across sibling branches (restoration of `last_occ`/`start` is essential there).
