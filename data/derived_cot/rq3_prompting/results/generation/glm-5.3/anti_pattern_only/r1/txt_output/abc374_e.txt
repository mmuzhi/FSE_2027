import sys

def main():
    input_data = sys.stdin.buffer.read().split()
    idx = 0
    N = int(input_data[idx]); idx+=1
    X = int(input_data[idx]); idx+=1
    machines = []
    for _ in range(N):
        A = int(input_data[idx]); idx+=1
        P = int(input_data[idx]); idx+=1
        B = int(input_data[idx]); idx+=1
        Q = int(input_data[idx]); idx+=1
        machines.append((A, P, B, Q))
    
    def min_cost(w, A, P, B, Q):
        # min cost to get capacity >= w
        best = float('inf')
        # cap on s: min(B, ceil(w/A))
        cap_s = (w + A - 1) // A
        if cap_s > B:
            cap_s = B
        for s in range(cap_s + 1):
            rem = w - s * A
            if rem <= 0:
                t = 0
            else:
                t = (rem + B - 1) // B
            c = s * P + t * Q
            if c < best:
                best = c
        cap_t = (w + B - 1) // B
        if cap_t > A:
            cap_t = A
        for t in range(cap_t + 1):
            rem = w - t * B
            if rem <= 0:
                s = 0
            else:
                s = (rem + A - 1) // A
            c = s * P + t * Q
            if c < best:
                best = c
        return best
    
    def feasible(w):
        total = 0

            total += min_cost(w, A, P, B, Q)

def main():

Looking at this problem:

- Binary search on the answer `w` (production capacity). Max possible is 10^9 (X=10^7 machines × 100 capacity), so search range [0, 10^9+1].
- For feasibility of `w`: for each process, compute minimum cost to reach capacity ≥ w using machines (A,P) and (B,Q), sum must be ≤ X.
- Min cost per process: minimize `s*P + t*Q` s.t. `s*A + t*B ≥ w`. Exchange argument: swapping `B` S-machines for `A` T-machines keeps capacity (both give A·B) and changes cost by `A*Q - B*P`. So if `A*Q ≤ B*P`, an optimal solution exists with `s < B` (loop s from 0 to min(B, ⌈w/A⌉)); otherwise optimal has `t < A` (loop t from 0 to min(A, ⌈w/B⌉)). Since A,B ≤ 100, each check is O(200) per process.
