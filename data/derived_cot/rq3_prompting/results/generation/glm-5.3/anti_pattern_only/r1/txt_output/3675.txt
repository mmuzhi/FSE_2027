```python
class Solution:
    def maximizeSumOfWeights(self, edges: List[List[int]], k: int) -> int:
        n = len(edges) + 1
        adj = [[] for _ in range(n)]
        for u, v, w in edges:
            adj[u].append((v, w))
            adj[v].append((u, w))
        
        parent = [-1] * n
        parent_w = [0] * n
        order = []
        visited = [False] * n
        stack = [0]
        visited[0] = True
        while stack:
            u = stack.pop()
            order.append(u)
            for v, w in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    parent[v] = u
                    parent_w[v] = w
                    stack.append(v)
        
        children = [[] for _ in range(n)]
        for v in range(1, n):
            children[parent[v]].append(v)
        
        dp0 = [0] * n
        dp1 = [0] * n
        
        for u in reversed(order):
            if not children[u]:
                continue  # dp0 = dp1 = 0
            base = 0
            gains = []
            for v in children[u]:
                base += dp0[v]
                gains.append(parent_w[v] + dp1[v] - dp0[v])
            if not gains:
                continue
            gains.sort(reverse=True)
            # prefix sums of positive gains
            m = 0
            s = 0
            prefix = []
            for g in gains:
                if g <= 0:
                    break
                s += g
                prefix.append(s)
                m += 1
            a = k if k < m else m
            b = k - 1 if k - 1 < m else m
            dp0[u] = base + (prefix[a - 1] if a > 0 else 0)
            dp1[u] = base + (prefix[b - 1] if b > 0 else 0)
        
        return dp0[0]
```

```python
from typing import List

class Solution:
    def maximizeSumOfWeights(self, edges: List[List[int]], k: int) -> int:
        n = len(edges) + 1
        
        # Build adjacency list
        adj = [[] for _ in range(n)]
        for u, v, w in edges:
            adj[u].append((v, w))
            adj[v].append((u, w))
        
        # Iterative DFS to get order and parents
        parent = [-1] * n
        parent[0] = 0
        pw = [0] * n          # weight of edge to parent
        order = []
        stack = [0]
        while stack:
            u = stack.pop()
            order.append(u)
            for v, w in adj[u]:
                if parent[v] == -1:
                    parent[v] = u
                    pw[v] = w
                    stack.append(v)
        
        # Children lists
        children = [[] for _ in range(n)]
        for v in range(1, n):
            children[parent[v]].append(v)
        
        dp0 = [0] * n   # best sum in subtree, edge to parent NOT kept (can keep up to k child edges)
        dp1 = [0] * n   # best sum in subtree, edge to parent kept (can keep up to k-1 child edges)
        
        for u in reversed(order):
            ch = children[u]
            if not ch:
                continue
            base = 0
            gains = []
            for v in ch:
                d0 = dp0[v]
                base += d0
                g = pw[v] + dp1[v] - d0
                if g > 0:
                    gains.append(g)
            if not gains:
                dp0[u] = base
                dp1[u] = base
                continue
            gains.sort(reverse=True)
            m = len(gains)
            b = k - 1 if k - 1 < m else m   # number taken for dp1
            dp1[u] = base + sum(gains[:b])
            if b < m and b < k:
                dp0[u] = dp1[u] + gains[b]
            else:
                dp0[u] = dp1[u]
        
        return dp0[0]
```

```python
    def maximizeSumOfWeights(self, edges: List[List[int]], k: int) -> int:
        n = len(edges) + 1

        adj = [[] for _ in range(n)]
        for u, v, w in edges:
            adj[u].append((v, w))
            adj[v].append((u, w))

                    stack.append(v)

            for v in ch:
```

=== ANSWER ===
Looking at this problem, I need to select a subset of tree edges maximizing total weight, subject to every node having degree ≤ k.

**Approach: Tree DP.** Root the tree at node 0. For each node `u`, define:
- `dp0[u]`: best sum in `u`'s subtree when the edge to its parent is **dropped** (u may keep up to k child edges)
- `dp1[u]`: best sum in `u`'s subtree when the edge to its parent is **kept** (u may keep up to k−1 child edges)

For each child `v` with edge weight `w`: dropping the child edge yields `dp0[v]`; keeping it yields `w + dp1[v]`. The gain of keeping is `g = w + dp1[v] - dp0[v]`. Since children are independent, for each state I take the top positive gains up to the allowed limit (greedy is exact — it's just picking the best `limit` items by value).

Verified against Example 1 (node 2's gains: keeping [0,2] gives gain −4, so drop it → 22) and Example 2 (all gains positive → 65). Also hand-checked k=1 cases (maximum weight matching) and cases where a parent edge must be dropped.

Using iterative DFS (n up to 1e5, avoids recursion depth issues), processing nodes in reverse DFS order so children are finalized before parents.

```python
from typing import List

class Solution:
    def maximizeSumOfWeights(self, edges: List[List[int]], k: int) -> int:
        n = len(edges) + 1

        adj = [[] for _ in range(n)]
        for u, v, w in edges:
            adj[u].append((v, w))
            adj[v].append((u, w))

        # Iterative DFS from node 0: order (parents before children), parent edge weights
        parent = [-1] * n
        parent[0] = 0
        pw = [0] * n
        order = []
        stack = [0]
        while stack:
            u = stack.pop()
            order.append(u)
            for v, w in adj[u]:
                if parent[v] == -1:
                    parent[v] = u
                    pw[v] = w
                    stack.append(v)

        children = [[] for _ in range(n)]
        for v in range(1, n):
            children[parent[v]].append(v)

        dp0 = [0] * n  # parent edge dropped -> up to k child edges kept
        dp1 = [0] * n  # parent edge kept   -> up to k-1 child edges kept

        for u in reversed(order):
            ch = children[u]
            if not ch:
                continue
            base = 0
            gains = []
            for v in ch:
                d0 = dp0[v]
                base += d0
                g = pw[v] + dp1[v] - d0
                if g > 0:
                    gains.append(g)
            if not gains:
                dp0[u] = base
                dp1[u] = base
                continue
            gains.sort(reverse=True)
            m = len(gains)
            b = k - 1 if k - 1 < m else m  # count allowed for dp1
            dp1[u] = base + sum(gains[:b])
            # dp0 may take one more gain (index b) if limits allow
            if b < m and b < k:
                dp0[u] = dp1[u] + gains[b]
            else:
                dp0[u] = dp1[u]

        return dp0[0]
```
