```python
import numpy as np

class Solution:
    def minTime(self, skill: List[int], mana: List[int]) -> int:
        n = len(skill)
        S = np.zeros(n, dtype=np.int64)
        acc = 0
        for i, s in enumerate(skill):
            acc += s
            S[i] = acc
        Sprev = np.zeros(n, dtype=np.int64)
        Sprev[1:] = S[:-1]
        T = 0
        for j in range(len(mana) - 1):
            a = mana[j]
            b = mana[j+1]
            T += int(np.max(a * S - b * Sprev))
        return T + mana[-1] * int(S[-1])
```

```python
class Solution:
    def minTime(self, skill: List[int], mana: List[int]) -> int:
        n = len(skill)
        m = len(mana)
        
        # Prefix sums: S[i] = skill[0] + ... + skill[i]
        S = []
        acc = 0
        for s in skill:
            acc += s
            S.append(acc)
        total = S[-1]
        
        if m == 1:
            return mana[0] * total
        
        # X[i] = S[i-1] (prefix sum up to i-1), X[0] = 0
        X = [0] * n
        for i in range(1, n):
            X[i] = S[i - 1]
        
        # Build upper convex hull of points (X[i], S[i]), i = 0..n-1.
        # X strictly increasing, S strictly increasing.

        hull = []
        for i in range(n):

            yi = S[i]
            while len(hull) >= 2:
                a1 = hull[-1]
                a2 = hull[-2]
                # vectors: (X[a1]-X[a2], S[a1]-S[a2]) and (xi-X[a2], yi-S[a2])

                if cross >= 0:
                    hull.pop()
                else:
                    break
            hull.append(i)
        
        k = len(hull)

        # = mana[j] * max_i ( S[i] - t * X[i] ), t = mana[j+1]/mana[j]

        # Binary search: largest r such that r == 0 or slope(edge r-1) >= t.
        
        ans = mana[m - 1] * total
```

```python
from typing import List

class Solution:
    def minTime(self, skill: List[int], mana: List[int]) -> int:
        n = len(skill)
        m = len(mana)

        # Prefix sums S[i] = skill[0] + ... + skill[i]
        S = [0] * n
        acc = 0
        for i in range(n):
            acc += skill[i]
            S[i] = acc
        total = S[-1]

        if m == 1:
            return mana[0] * total

        # X[i] = S[i-1] (prefix sum of first i skills), X[0] = 0
        X = [0] * n
        for i in range(1, n):
            X[i] = S[i - 1]

        # For consecutive potions j, j+1:
        #   T_{j+1} = T_j + max_i ( mana[j]*S[i] - mana[j+1]*X[i] )
        # With a = mana[j], b = mana[j+1], t = b/a:
        #   max_i ( a*S[i] - b*X[i] ) = a * max_i ( S[i] - t*X[i] )
        # The inner max over points (X[i], S[i]) is attained on the upper convex hull.
        # Build upper hull (points have strictly increasing x and y).
        hull = []
        Xh = []  # x coords of hull points
        Yh = []  # y coords of hull points
        for i in range(n):
            xi = X[i]
            yi = S[i]
            while len(hull) >= 2:
                x1 = Xh[-1]; y1 = Yh[-1]
                x2 = Xh[-2]; y2 = Yh[-2]
                # pop last hull point if it is not a strict right turn
                if (x1 - x2) * (yi - y2) - (y1 - y2) * (xi - x2) >= 0:
                    hull.pop(); Xh.pop(); Yh.pop()
                else:
                    break
            hull.append(i)
            Xh.append(xi)
            Yh.append(yi)

        k = len(hull)
        # Edge slopes (dy, dx) between consecutive hull points; slopes strictly decreasing.
        # Vertex r is optimal for t in [slope_r, slope_{r-1}] (with conventions).
        EDX = [0] * (k - 1) if k > 1 else []
        EDY = [0] * (k - 1) if k > 1 else []
        for r in range(k - 1):
            EDY[r] = Yh[r + 1] - Yh[r]
            EDX[r] = Xh[r + 1] - Xh[r]

        ans = mana[m - 1] * total
        for j in range(m - 1):
            a = mana[j]
            b = mana[j + 1]
            # largest r in [0, k-1] with r == 0 or slope(edge r-1) >= b/a
            lo = 0
            hi = k - 1
            while lo < hi:
                mid = (lo + hi + 1) >> 1
                e = mid - 1
                if a * EDY[e] >= b * EDX[e]:
                    lo = mid
                else:
                    hi = mid - 1
            iopt = hull[lo]
            ans += a * S[iopt] - b * X[iopt]
        return ans
```

```python
from typing import List

class Solution:
    def minTime(self, skill: List[int], mana: List[int]) -> int:
        n = len(skill)
        m = len(mana)

        # S[i]: prefix sum skill[0..i]; X[i]: prefix sum skill[0..i-1] (X[0] = 0)
        S = [0] * n
        acc = 0
        for i in range(n):
            acc += skill[i]
            S[i] = acc
        total = S[-1]

        if m == 1:
            return mana[0] * total

        X = [0] * n
        for i in range(1, n):
            X[i] = S[i - 1]

        # delta_j = max_i (mana[j]*S[i] - mana[j+1]*X[i]) is the minimal gap
        # T_{j+1} - T_j between start times of consecutive potions at wizard 0.
        # With a=mana[j], b=mana[j+1], t=b/a: equals a * max_i (S[i] - t*X[i]).
        # The max of (y - t*x) over points (X[i], S[i]) is attained on the
        # upper convex hull; vertex r is optimal for t in [slope_r, slope_{r-1}].
        hull_idx = []
        hx = []
        hy = []
        for i in range(n):
            xi = X[i]
            yi = S[i]
            while len(hull_idx) >= 2:
                x1 = hx[-1]; y1 = hy[-1]
                x2 = hx[-2]; y2 = hy[-2]
                if (x1 - x2) * (yi - y2) - (y1 - y2) * (xi - x2) >= 0:
                    hull_idx.pop(); hx.pop(); hy.pop()
                else:
                    break
            hull_idx.append(i)
            hx.append(xi)
            hy.append(yi)

        k = len(hull_idx)
        # edges: e = 0..k-2 between hull points e and e+1; slopes strictly decreasing
        edy = [hy[r + 1] - hy[r] for r in range(k - 1)]
        edx = [hx[r + 1] - hx[r] for r in range(k - 1)]

        ans = mana[m - 1] * total
        for j in range(m - 1):
            a = mana[j]
            b = mana[j + 1]
            # largest r with r == 0 or slope(edge r-1) >= b/a,
            # i.e. a*edy[r-1] >= b*edx[r-1]
            lo, hi = 0, k - 1
            while lo < hi:
                mid = (lo + hi + 1) >> 1
                e = mid - 1
                if a * edy[e] >= b * edx[e]:
                    lo = mid
                else:
                    hi = mid - 1
            iopt = hull_idx[lo]
            ans += a * S[iopt] - b * X[iopt]

        return ans
```

```python
from typing import List

class Solution:
    def minTime(self, skill: List[int], mana: List[int]) -> int:
        n = len(skill)
        m = len(mana)

        # Prefix sums: S[i] = skill[0] + ... + skill[i]
        # X[i] = S[i-1] (prefix of first i skills), X[0] = 0
        S = [0] * n
        acc = 0
        for i in range(n):
            acc += skill[i]
            S[i] = acc
        total = S[-1]

        if m == 1:
            return mana[0] * total

        X = [0] * n
        for i in range(1, n):
            X[i] = S[i - 1]

        # For consecutive potions j and j+1 the start times at wizard 0 satisfy
        #   T[j+1] - T[j] >= max_i( mana[j]*S[i] - mana[j+1]*X[i] )   (= delta_j)
        # and the optimal makespan is mana[m-1]*S[n-1] + sum(delta_j).
        # With a = mana[j], b = mana[j+1], t = b/a:
        #   delta_j = a * max_i( S[i] - t*X[i] )
        # The maximizer of (y - t*x) over points (X[i], S[i]) lies on the upper
        # convex hull of these points (x strictly increasing). On the hull,
        # vertex r is optimal iff slope_r <= t <= slope_{r-1}, where slope_e is
        # the slope of hull edge e (slopes strictly decrease along the hull).
        hull = []   # wizard indices on the upper hull
        hx = []     # their x coords
        hy = []     # their y coords
        for i in range(n):
            xi = X[i]
            yi = S[i]
            while len(hull) >= 2:
                x1 = hx[-1]; y1 = hy[-1]
                x2 = hx[-2]; y2 = hy[-2]
                # remove last point if it is not a strict right (clockwise) turn
                if (x1 - x2) * (yi - y2) - (y1 - y2) * (xi - x2) >= 0:
                    hull.pop(); hx.pop(); hy.pop()
                else:
                    break
            hull.append(i)
            hx.append(xi)
            hy.append(yi)

        k = len(hull)
        # edge slopes as (dy, dx) with dx > 0; strictly decreasing along hull
        edy = [hy[r + 1] - hy[r] for r in range(k - 1)]
        edx = [hx[r + 1] - hx[r] for r in range(k - 1)]

        ans = mana[m - 1] * total
        for j in range(m - 1):
            a = mana[j]
            b = mana[j + 1]
            # largest r in [0, k-1] with r == 0 or slope(edge r-1) >= b/a
            # slope(edge e) >= b/a  <=>  a*edy[e] >= b*edx[e]
            lo, hi = 0, k - 1
            while lo < hi:
                mid = (lo + hi + 1) >> 1
                e = mid - 1
                if a * edy[e] >= b * edx[e]:
                    lo = mid
                else:
                    hi = mid - 1
            iopt = hull[lo]
            ans += a * S[iopt] - b * X[iopt]

        return ans
```

Looking at this problem:

- If potion j starts at wizard 0 at time T_j, wizard i finishes it at T_j + mana[j]·S_i (S_i = prefix sum of skill). The "immediate pass" (no-wait) constraint means the whole schedule is determined by the T_j.
- Feasibility between consecutive potions: for each wizard i, T_{j+1} + mana[j+1]·S_{i-1} ≥ T_j + mana[j]·S_i, so the minimal gap is delta_j = max_i(mana[j]·S_i − mana[j+1]·S_{i-1}), and the answer is mana[m-1]·S_{n-1} + Σ delta_j (these deltas are constants, so the tight schedule is optimal).

Computing each delta_j naively is O(n·m) = 25M. Instead, with a = mana[j], b = mana[j+1], t = b/a: delta_j = a·max_i(S_i − t·X_i) over points (X_i, S_i) where X_i = S_{i-1}. The maximizer of (y − t·x) lies on the upper convex hull of these points; vertex r is optimal exactly when t lies between adjacent edge slopes. Build the hull once (O(n)), then answer each query with an exact integer binary search (a·dy ≥ b·dx), giving O(n + m log n) — fast pure Python, verified against all three examples (110, 5, 21) and several hand-simulated cases.

```python
from typing import List

class Solution:
    def minTime(self, skill: List[int], mana: List[int]) -> int:
        n = len(skill)
        m = len(mana)

        # S[i] = skill[0..i] prefix sum; X[i] = S[i-1] (X[0] = 0)
        S = [0] * n
        acc = 0
        for i in range(n):
            acc += skill[i]
            S[i] = acc
        total = S[-1]

        if m == 1:
            return mana[0] * total

        X = [0] * n
        for i in range(1, n):
            X[i] = S[i - 1]

        # delta_j = max_i(mana[j]*S[i] - mana[j+1]*X[i]); answer = mana[m-1]*total + sum(delta_j)
        # With a=mana[j], b=mana[j+1], t=b/a: delta_j = a * max_i(S[i] - t*X[i]).
        # Maximizer of (y - t*x) over points (X[i], S[i]) is on the upper convex hull;
        # hull vertex r is optimal iff slope_r <= t <= slope_{r-1} (slopes decrease along hull).
        hull = []
        hx = []
        hy = []
        for i in range(n):
            xi = X[i]
            yi = S[i]
            while len(hull) >= 2:
                x1 = hx[-1]; y1 = hy[-1]
                x2 = hx[-2]; y2 = hy[-2]
                # pop last point if not a strict right (clockwise) turn
                if (x1 - x2) * (yi - y2) - (y1 - y2) * (xi - x2) >= 0:
                    hull.pop(); hx.pop(); hy.pop()
                else:
                    break
            hull.append(i)
            hx.append(xi)
            hy.append(yi)

        k = len(hull)
        # edge slopes as (dy, dx), strictly decreasing along the hull
        edy = [hy[r + 1] - hy[r] for r in range(k - 1)]
        edx = [hx[r + 1] - hx[r] for r in range(k - 1)]

        ans = mana[m - 1] * total
        for j in range(m - 1):
            a = mana[j]
            b = mana[j + 1]
            # largest r in [0, k-1] with r == 0 or slope(edge r-1) >= b/a
            # slope(edge e) >= b/a  <=>  a*edy[e] >= b*edx[e]
            lo, hi = 0, k - 1
            while lo < hi:
                mid = (lo + hi + 1) >> 1
                e = mid - 1
                if a * edy[e] >= b * edx[e]:
                    lo = mid
                else:
                    hi = mid - 1
            iopt = hull[lo]
            ans += a * S[iopt] - b * X[iopt]

        return ans
```
