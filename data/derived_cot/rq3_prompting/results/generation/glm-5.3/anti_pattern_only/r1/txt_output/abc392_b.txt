```python
import sys

def main():
    data = sys.stdin.read().split()
    n = int(data[0])
    m = int(data[1])
    a = set(map(int, data[2:2 + m]))
    res = [i for i in range(1, n + 1) if i not in a]
    print(len(res))
    print(' '.join(map(str, res)))

main()
```
