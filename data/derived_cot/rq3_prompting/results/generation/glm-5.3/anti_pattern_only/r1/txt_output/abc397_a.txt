```python
x_str = input().strip()
if '.' in x_str:
    whole, frac = x_str.split('.')
    tenths = int(whole) * 10 + int(frac.ljust(1, '0')[:1])
else:
    tenths = int(x_str) * 10

if tenths >= 380:
    print(1)
elif tenths >= 375:
    print(2)
else:
    print(3)
```
