```python
class Solution:
    def minBitwiseArray(self, nums: List[int]) -> List[int]:
        ans = []
        for p in nums:
            if p % 2 == 0:
                ans.append(-1)
            else:
                t = (p + 1) & -(p + 1)  # 2^trailing_ones
                ans.append(p - (t >> 1))
        return ans
```

```python
class Solution:
    def minBitwiseArray(self, nums: List[int]) -> List[int]:
        ans = []
        for p in nums:
            if p % 2 == 0:
                # x | (x+1) is always odd, so even targets are impossible
                ans.append(-1)
            else:
                # Let t = number of trailing 1-bits in p.
                # Valid x are p with one trailing 1-bit cleared; minimum clears the highest one.
                # 2^t = lowest set bit of (p+1); answer = p - 2^(t-1)
                low = (p + 1) & -(p + 1)
                ans.append(p - (low >> 1))
        return ans
```
