```python
from typing import List

class Solution:
    def minCost(self, nums: List[int]) -> int:
        n = len(nums)
        if n <= 2:
            return max(nums)
        
        # dp[k]: min cost so far, given nums[k] is carried at front and nums[j:] follows
        dp = [max(nums[1], nums[2]), max(nums[0], nums[2]), max(nums[0], nums[1])]
        j = 3
        
        while j <= n - 2:
            x, y = nums[j], nums[j + 1]
            m = x if x > y else y
            # keep carried element k: pay max(x, y)
            ndp = [v + m for v in dp]
            # keep nums[j] (drop carried k and nums[j+1]): pay max(nums[k], y)
            best_j = min(v + (nums[k] if nums[k] > y else y) for k, v in enumerate(dp))
            # keep nums[j+1] (drop carried k and nums[j]): pay max(nums[k], x)

            ndp.append(best_j)
            ndp.append(best_j1)

            j += 2
        
        if j == n - 1:
            last = nums[-1]
            return min(v + (nums[k] if nums[k] > last else last) for k, v in enumerate(dp))
        else:
            return min(v + nums[k] for k, v in enumerate(dp))
```

```python
from typing import List

class Solution:
    def minCost(self, nums: List[int]) -> int:
        n = len(nums)
        if n <= 2:
            return max(nums)
        
        # dp[k] = min cost so far, given nums[k] is carried at the front
        # and nums[j:] is the untouched suffix.
        dp = [max(nums[1], nums[2]),
              max(nums[0], nums[2]),
              max(nums[0], nums[1])]
        j = 3
        
        while j <= n - 2:
            x = nums[j]
            y = nums[j + 1]
            m = x if x > y else y
            # keep the carried element: remove nums[j], nums[j+1], pay m
            ndp = [v + m for v in dp]
            # keep nums[j]: remove carried and nums[j+1], pay max(nums[k], y)
            ndp.append(min(v + (nums[k] if nums[k] > y else y)
                           for k, v in enumerate(dp)))
            # keep nums[j+1]: remove carried and nums[j], pay max(nums[k], x)
            ndp.append(min(v + (nums[k] if nums[k] > x else x)
                           for k, v in enumerate(dp)))
            dp = ndp
            j += 2
        
        if j == n - 1:
            last = nums[n - 1]
            return min(v + (nums[k] if nums[k] > last else last)
                       for k, v in enumerate(dp))
        # j == n: single element nums[k] remains
        return min(v + nums[k] for k, v in enumerate(dp))
```

=== ANSWER ===
Looking at this problem:

- Each operation (while ≥3 elements remain) takes the first three elements `[carried, nums[j], nums[j+1]]`, removes two (paying their max), and keeps one which becomes the new front.
- So the state is fully described by `(k, j)`: `nums[k]` is the carried front element and `nums[j:]` is the untouched suffix. From `(k, j)` we transition to `(k, j+2)`, `(j, j+2)`, or `(j+1, j+2)` with the corresponding costs.
- When 1 element remains (j == n) we pay `nums[k]`; when 2 remain (j == n-1) we pay `max(nums[k], nums[n-1])`.

Since `j` only takes values 3, 5, 7, ... (increasing by 2 per step), a level-by-level DP over the carried index `k` gives an O(n²) solution, which is fine for n ≤ 1000.

I verified against both examples (12 and 5) and several hand-traced cases (e.g., `[1..6]` → 12, `[3,1,2,5,4]` → 9, `[10,1,1,1,1,1,1]` → 13), all matching exhaustive enumeration of keep-choices.

```python
from typing import List

class Solution:
    def minCost(self, nums: List[int]) -> int:
        n = len(nums)
        if n <= 2:
            return max(nums)

        # dp[k] = min cost so far, given nums[k] is carried at the front
        # and nums[j:] is the untouched suffix (level j).
        dp = [max(nums[1], nums[2]),
              max(nums[0], nums[2]),
              max(nums[0], nums[1])]
        j = 3

        while j <= n - 2:
            x = nums[j]
            y = nums[j + 1]
            m = x if x > y else y
            # keep the carried element: remove nums[j], nums[j+1], pay max(x, y)
            ndp = [v + m for v in dp]
            # keep nums[j]: remove carried nums[k] and nums[j+1], pay max(nums[k], y)
            ndp.append(min(v + (nums[k] if nums[k] > y else y)
                           for k, v in enumerate(dp)))
            # keep nums[j+1]: remove carried nums[k] and nums[j], pay max(nums[k], x)
            ndp.append(min(v + (nums[k] if nums[k] > x else x)
                           for k, v in enumerate(dp)))
            dp = ndp
            j += 2

        if j == n - 1:
            # two elements remain: nums[k] and nums[n-1]
            last = nums[n - 1]
            return min(v + (nums[k] if nums[k] > last else last)
                       for k, v in enumerate(dp))
        # j == n: single element nums[k] remains
        return min(v + nums[k] for k, v in enumerate(dp))
```
