```python
    def minMaxSubarraySum(self, nums: List[int], k: int) -> int:
        n = len(nums)
        
        def count_pairs(a, b, k):
            # number of (x, y): 0 <= x < a, 0 <= y < b, x + y <= k - 1
            if a <= 0 or b <= 0:
                return 0
            if k >= a + b - 1:
                return a * b
            # k < a + b - 1
            x_hi = min(a - 1, k - 1)  # last x with k - x >= 1

                return 0
```

```python
class Solution:
    def minMaxSubarraySum(self, nums: List[int], k: int) -> int:
        def count_pairs(a, b):
            # number of (x, y): 0 <= x < a, 0 <= y < b, x + y <= k - 1
            if a <= 0 or b <= 0:
                return 0
            if k >= a + b - 1:
                return a * b
            x_hi = min(a - 1, k - 1)  # last x with positive count
            if x_hi < 0:
                return 0
            t = k - b  # for x <= t, count of y is b
            if t >= x_hi:
                return (x_hi + 1) * b
            if t >= 0:
                n_mid = x_hi - t
                return (t + 1) * b + k * n_mid - (t + 1 + x_hi) * n_mid // 2
            m = x_hi + 1
            return k * m - x_hi * (x_hi + 1) // 2

        def sum_subarray_mins(arr):
            n = len(arr)
            prev = [-1] * n
            nxt = [n] * n

            stack = []
            for i in range(n):
                v = arr[i]
                while stack and arr[stack[-1]] >= v:
                    stack.pop()
                prev[i] = stack[-1] if stack else -1
                stack.append(i)

            stack = []
            for i in range(n - 1, -1, -1):
                v = arr[i]
                while stack and arr[stack[-1]] > v:
                    stack.pop()
                nxt[i] = stack[-1] if stack else n
                stack.append(i)

            total = 0
            for i in range(n):
                total += arr[i] * count_pairs(i - prev[i], nxt[i] - i)
            return total

        return sum_subarray_mins(nums) - sum_subarray_mins([-x for x in nums])
```

```python
from typing import List

class Solution:
    def minMaxSubarraySum(self, nums: List[int], k: int) -> int:
        def count_pairs(a: int, b: int) -> int:
            # Number of pairs (x, y) with 0 <= x < a, 0 <= y < b, x + y <= k - 1.
            if a <= 0 or b <= 0:
                return 0
            if k >= a + b - 1:
                return a * b
            x_hi = min(a - 1, k - 1)  # largest x with a nonzero count of y
            if x_hi < 0:
                return 0
            t = k - b  # for x <= t the count of y is b
            if t >= x_hi:
                return (x_hi + 1) * b
            if t >= 0:
                n_mid = x_hi - t
                return (t + 1) * b + k * n_mid - (t + 1 + x_hi) * n_mid // 2
            m = x_hi + 1
            return k * m - x_hi * (x_hi + 1) // 2

        def sum_subarray_mins(arr: List[int]) -> int:
            n = len(arr)
            prev = [-1] * n  # nearest j < i with arr[j] < arr[i]
            nxt = [n] * n    # nearest j > i with arr[j] <= arr[i]

            stack = []
            for i in range(n):
                v = arr[i]
                while stack and arr[stack[-1]] >= v:
                    stack.pop()
                prev[i] = stack[-1] if stack else -1
                stack.append(i)

            stack = []
            for i in range(n - 1, -1, -1):
                v = arr[i]
                while stack and arr[stack[-1]] > v:
                    stack.pop()
                nxt[i] = stack[-1] if stack else n
                stack.append(i)

            total = 0
            for i in range(n):
                total += arr[i] * count_pairs(i - prev[i], nxt[i] - i)
            return total

        return sum_subarray_mins(nums) - sum_subarray_mins([-x for x in nums])
```

=== ANSWER ===
Looking at this problem:

- Answer = (sum of maximums over all subarrays of length ≤ k) + (sum of minimums over all subarrays of length ≤ k).
- For sum of minimums: for each `i`, find `prev[i]` (nearest index to the left with a strictly smaller value) and `next[i]` (nearest index to the right with a smaller-or-equal value). Then `nums[i]` is the "responsible" minimum for subarrays whose left endpoint `L ∈ (prev[i], i]` and right endpoint `R ∈ [i, next[i])`. With `x = i - L`, `y = R - i`, we need `0 ≤ x < a`, `0 ≤ y < b`, `x + y ≤ k - 1`. This count has a closed form (verified against brute force on multiple cases including duplicates and negative values).
- Sum of maximums = `-` (sum of minimums of the negated array), keeping the same tie-breaking consistency.

I verified this against both examples (20 and -6) plus extra cases with duplicates (`[3,1,3,1,3]`, k=2 → 38) and increasing arrays with various k — all match brute force.
