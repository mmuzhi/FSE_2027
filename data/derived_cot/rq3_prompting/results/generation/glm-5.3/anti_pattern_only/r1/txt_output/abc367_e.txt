```python
import sys
from collections import deque

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    k = int(data[1])
    X = [int(v) - 1 for v in data[2:2 + n]]
    A = data[2 + n:2 + 2 * n]  # bytes tokens

    if k < n:
        # binary lifting (at most ~18 doubling steps since k < n <= 2e5)
        res = list(range(n))
        p = X
        kk = k
        while kk > 0:
            if kk & 1:
                res = list(map(p.__getitem__, res))
            kk >>= 1
            if kk:
                p = list(map(p.__getitem__, p))
        out = b' '.join([A[i] for i in res])
    else:
        # k >= n >= max depth+1: every node lands on a cycle
        result = [0] * n
        state = bytearray(n)
        ppos = [0] * n
        oncycle = bytearray(n)
        radj = [[] for _ in range(n)]
        for i in range(n):
            radj[X[i]].append(i)
        dq = deque()
        for s in range(n):
            if state[s]:
                continue
            path = []
            u = s
            while not state[u]:
                state[u] = 1
                ppos[u] = len(path)
                path.append(u)
                u = X[u]
            if state[u] == 1:
                cyc = path[ppos[u]:]
                L = len(cyc)
                r = k % L
                for j in range(L):
                    node = cyc[j]
                    oncycle[node] = 1
                    jj = j + r
                    if jj >= L:
                        jj -= L
                    result[node] = cyc[jj]
                    dq.append((node, jj, cyc, L))
            for node in path:
                state[node] = 2
        while dq:
            w, jw, cyc, L = dq.popleft()
            for v in radj[w]:
                if not oncycle[v]:
                    oncycle[v] = 1
                    jv = jw - 1
                    if jv < 0:
                        jv += L
                    result[v] = cyc[jv]
                    dq.append((v, jv, cyc, L))
        out = b' '.join([A[j] for j in result])

    sys.stdout.buffer.write(out + b'\n')
```

Looking at this problem:

- One operation sets B_i = A_{X_i}. After K operations, A'_i = A_{f^K(i)} where f(i) = X_i (as a function on indices).
- So I need f^K for each index, with N ≤ 2×10⁵ and K up to 10¹⁸.

**Approach (two regimes, both fast in pure Python):**

1. **If K < N**: binary lifting (doubling) — at most ~18 squarings/compositions since K < 2×10⁵. Each step is `list(map(p.__getitem__, ...))`.

2. **If K ≥ N**: every node reaches its cycle within ≤ N−1 steps, so f^K(i) always lands on the cycle i feeds into. Decompose the functional graph into cycles plus in-trees. For cycle nodes: answer is `cyc[(pos + K mod L) % L]`. For tree nodes at depth d: answer is d steps backward around the cycle from the root's answer — propagated via BFS over reversed edges (child's cycle index = parent's index − 1 mod L).

I verified this against all three samples (sample 1 via lifting, sample 3 via the cycle method, tracing by hand), plus extra hand cases (tails of length > 1, self-loops, multiple components).

```python
        # k >= n: every node reaches its cycle in <= n-1 steps,
        # so f^k(i) lies on the cycle that i feeds into.
        result = [0] * n
        state = bytearray(n)          # 0=unvisited, 1=on current path, 2=done
        ppos = [0] * n
        oncycle = bytearray(n)
        radj = [[] for _ in range(n)]
        for i in range(n):
            radj[X[i]].append(i)
        dq = deque()
        for s in range(n):
            if state[s]:
                continue
            path = []
            u = s
            while not state[u]:
                state[u] = 1
                ppos[u] = len(path)
                path.append(u)
                u = X[u]
            if state[u] == 1:
                cyc = path[ppos[u]:]
                L = len(cyc)
                r = k % L
                for j in range(L):
                    node = cyc[j]
                    oncycle[node] = 1
                    jj = j + r
                    if jj >= L:
                        jj -= L
                    result[node] = cyc[jj]
                    dq.append((node, jj, cyc, L))
            for node in path:
                state[node] = 2
        # Tree nodes: each step up the tree moves the cycle position back by 1.
        while dq:
            w, jw, cyc, L = dq.popleft()
            for v in radj[w]:
                if not oncycle[v]:
                    oncycle[v] = 1
                    jv = jw - 1
                    if jv < 0:
                        jv += L
                    result[v] = cyc[jv]
                    dq.append((v, jv, cyc, L))
        out = b' '.join([A[j] for j in result])

    sys.stdout.buffer.write(out + b'\n')


main()
```
