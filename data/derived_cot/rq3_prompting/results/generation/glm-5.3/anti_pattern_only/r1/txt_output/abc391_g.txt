import sys

def main():
    data = sys.stdin.buffer.read().split()
    N = int(data[0])
    M = int(data[1])
    S = data[2].decode()
    MOD = 998244353

    # Character classes: each distinct char of S (weight 1) + all others (weight 26 - #distinct)
    chars_in_S = sorted(set(S))
    class_list = []
    for c in chars_in_S:
        mask = tuple(1 if S[j] == c else 0 for j in range(N))
        class_list.append((mask, 1))
    class_list.append((tuple([0] * N), 26 - len(chars_in_S)))

    init = tuple([0] * (N + 1))
    state_index = {init: 0}
    state_list = [init]
    compact = []  # compact[i]: list of (next_state, weight)

    i = 0
    while i < len(state_list):
        st = state_list[i]
        acc = {}
        for mask, w in class_list:
            new_row = [0] * (N + 1)
            for j in range(1, N + 1):
                a = st[j]
                b = new_row[j - 1]
                c = st[j - 1] + mask[j - 1]
                m = a if a > b else b
                new_row[j] = m if m > c else c
            nst = tuple(new_row)
            idx = state_index.get(nst)
            if idx is None:
                idx = len(state_list)
                state_index[nst] = idx
                state_list.append(nst)
            acc[idx] = acc.get(idx, 0) + w
        compact.append(list(acc.items()))
        i += 1

    ns = len(state_list)

    # Transpose: targets[j] = list of (source, weight) with transitions into j
    targets = [[] for _ in range(ns)]
    for si, trs in enumerate(compact):
        for j, w in trs:
            targets[j].append((si, w))

    cnt = [0] * ns
    cnt[0] = 1
    for _ in range(M):
        cnt = [sum(cnt[si] * w for si, w in t) % MOD for t in targets]

    ans = [0] * (N + 1)
    for idx in range(ns):
        k = state_list[idx][N]
        ans[k] = (ans[k] + cnt[idx]) % MOD

    sys.stdout.write(' '.join(map(str, ans)) + '\n')

main()

- N ≤ 10, so the LCS DP row of T's prefix against S (values dp[0..N], non-decreasing with steps 0/1) has at most 2^N ≤ 1024 possible states, and this row is a sufficient statistic for future evolution.
- Stream characters of T: for each state (row) and each character class (each distinct char of S, plus "all other letters" which behave identically), compute the next row via the standard LCS row update `new[j] = max(prev[j], new[j-1], prev[j-1] + match)`.
- Count strings by DP over M steps, then bucket final states by dp[N] (the LCS value).
