class Solution:
    def countGoodArrays(self, n: int, m: int, k: int) -> int:
        MOD = 10**9 + 7
        
        # The array splits into n - k blocks of equal consecutive values.
        # Choose which k of the n-1 adjacent pairs are "equal": C(n-1, k) ways.
        # First block: m choices. Each next block must differ: (m-1) choices,
        # and there are (n - k - 1) such blocks after the first.
        # Answer = C(n-1, k) * m * (m-1)^(n-k-1) mod 1e9+7
        
        # Precompute factorials up to n-1
        fact = [1] * n
        for i in range(1, n):
            fact[i] = fact[i - 1] * i % MOD
        
        inv_fact = [1] * n
        inv_fact[n - 1] = pow(fact[n - 1], MOD - 2, MOD)
        for i in range(n - 1, 0, -1):
            inv_fact[i - 1] = inv_fact[i] * i % MOD
        
        def comb(a: int, b: int) -> int:
            if b < 0 or b > a:
                return 0
            return fact[a] * inv_fact[b] % MOD * inv_fact[a - b] % MOD
        
        return comb(n - 1, k) * m % MOD * pow(m - 1, n - k - 1, MOD) % MOD
