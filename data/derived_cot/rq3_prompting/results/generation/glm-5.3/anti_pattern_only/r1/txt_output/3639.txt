```python
class Solution:
    def isZeroArray(self, nums: List[int], queries: List[List[int]]) -> bool:
        n = len(nums)
        diff = [0] * (n + 1)
        for l, r in queries:
            diff[l] += 1
            diff[r + 1] -= 1
        cur = 0
        for i in range(n):
            cur += diff[i]
            if nums[i] > cur:
                return False
        return True
```

```python
class Solution:
    def isZeroArray(self, nums: List[int], queries: List[List[int]]) -> bool:
        n = len(nums)
        diff = [0] * (n + 1)
        
        # Count how many queries cover each index using a difference array
        for l, r in queries:
            diff[l] += 1
            diff[r + 1] -= 1
        
        cur = 0
        for i in range(n):
            cur += diff[i]
            # Each covering query can decrement index i at most once,
            # so coverage count must be at least nums[i]
            if nums[i] > cur:
                return False
        
        return True
```
