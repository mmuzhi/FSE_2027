class Solution:
    def maxDifference(self, s: str) -> int:
        from collections import Counter
        counts = Counter(s)
        max_odd = max(c for c in counts.values() if c % 2 == 1)
        min_even = min(c for c in counts.values() if c % 2 == 0)
        return max_odd - min_even
```
