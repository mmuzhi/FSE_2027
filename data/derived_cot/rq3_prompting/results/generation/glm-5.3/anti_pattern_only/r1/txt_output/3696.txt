```python
class Solution:
    def countSubstrings(self, s: str) -> int:
        n = len(s)
        ans = 0
        # counters
        cnt3 = [0]*3
        cnt9 = [0]*9
        cnt7 = [0]*7
        cnt3[0] = 1
        cnt9[0] = 1
        cnt7[0] = 1
        ps9 = 0  # digit sum mod 9 of prefix
        C = 0    # B(k-1) * 10^{-k} mod 7, k = current index (number of chars processed)
        # inv10 powers mod 7, exponent e (1-based): e%6: 0->1, 1->5, 2->4, 3->6, 4->2, 5->3
        t = [1, 5, 4, 6, 2, 3]
        prev1 = -1  # s[j-1]
        prev2 = -1  # s[j-2]
        for j, ch in enumerate(s):
            c = ord(ch) - 48
            # update prefix states for position j
            ps9 = (ps9 + c) % 9
            C = (C + c * t[(j+1) % 6]) % 7
            if c == 0:
                pass
            elif c == 1 or c == 2 or c == 5:
                ans += j + 1
            elif c == 3:
                ans += cnt3[ps9 % 3]
            elif c == 6:
                ans += cnt3[ps9 % 3]
            elif c == 9:
                ans += cnt9[ps9]
            elif c == 4:
                ans += 1
                if prev1 >= 0 and (prev1 & 1) == 0:
                    ans += j
            elif c == 8:
                ans += 1
                if prev1 >= 0 and (prev1 % 4) == 0:
                    ans += 1
                if prev2 >= 0 and (4*prev2 + 2*prev1) % 8 == 0:
                    ans += j - 1
            else:  # c == 7
                ans += cnt7[C]
            # add current prefix values to counters
            r3 = ps9 % 3
            cnt3[r3] += 1
            cnt9[ps9] += 1
            cnt7[C] += 1
            prev2 = prev1
            prev1 = c
        return ans
    ```

```python
    def countSubstrings(self, s: str) -> int:
        ans = 0
        cnt3 = [0] * 3
```

```python
    def countSubstrings(self, s: str) -> int:

        cnt3 = [0] * 3
        cnt9 = [0] * 9
        cnt7 = [0] * 7
        cnt3[0] = cnt9[0] = cnt7[0] = 1
        ps9 = 0          # digit sum of prefix, mod 9
        C = 0            # B(k-1) * 10^(-k) mod 7 (k = chars processed)
```

=== ANSWER ===
Looking at this problem, I need to count substrings ending at each position `j` whose numeric value is divisible by the digit `d = s[j]` (skipping `d = 0`). I handle each possible last digit with specialized math:

- **d = 1, 2, 5**: value always divisible (ends in 1/2/5) → all `j+1` substrings count.
- **d = 3, 9**: `10 ≡ 1`, so divisibility = digit sum mod 3 (or 9) → prefix-sum counters.
- **d = 6**: last digit 6 makes it even, so just need digit sum ≡ 0 mod 3.
- **d = 4, 8**: only last 2 (or 3) digits matter → local checks on neighbors.
- **d = 7**: since `gcd(10,7)=1`, with `B(k)` = value of first `k` chars, `V(i..j) ≡ 0 (mod 7) ⟺ B(j+1)·10^{-(j+1)} ≡ B(i)·10^{-i} (mod 7)`. Define `C(k) = B(k)·10^{-k} mod 7` with recurrence `C(k+1) = C(k) + s[k]·10^{-(k+1)}` (powers of `10⁻¹ = 5` cycle with period 6). Then substring `(i,j)` counts iff `C(i) = C(j+1)` — counted with a residue counter.

I verified this against all three examples (11, 18, 25) and several brute-force cases (e.g. "2757"→6, "3428"→6, "787"→3, "1757"→7). Single pass, O(n) time, O(1) space.

```python
class Solution:
    def countSubstrings(self, s: str) -> int:
        ans = 0
        cnt3 = [0] * 3   # prefix digit sums mod 3 (incl. empty prefix)
        cnt9 = [0] * 9   # prefix digit sums mod 9
        cnt7 = [0] * 7   # C(k) = value(s[:k]) * 10^(-k) mod 7 (incl. k = 0)
        cnt3[0] = cnt9[0] = cnt7[0] = 1
        ps9 = 0          # digit sum of processed prefix, mod 9
        C = 0            # C(k) where k = number of processed chars
        u = (5, 4, 6, 2, 3, 1)   # 10^{-(j+1)} mod 7, period 6
        prev1 = prev2 = -1       # previous digit / digit before it
        for j, b in enumerate(s.encode()):
            c = b - 48
            ps9 = (ps9 + c) % 9
            C = (C + c * u[j % 6]) % 7
            if c == 1 or c == 2 or c == 5:
                # ends in 1/2/5 -> always divisible by that digit
                ans += j + 1
            elif c == 3 or c == 6:
                # need digit sum of substring ≡ 0 mod 3 (d=6 is auto-even)
                ans += cnt3[ps9 % 3]
            elif c == 9:
                ans += cnt9[ps9]
            elif c == 4:
                ans += 1                      # "4" itself
                if prev1 >= 0 and (prev1 & 1) == 0:
                    ans += j                  # all length>=2: last two digits 10*prev1+4
            elif c == 8:
                ans += 1                      # "8" itself
                if prev1 >= 0 and (prev1 & 3) == 0:
                    ans += 1                  # length-2 substring
                if prev2 >= 0 and (4 * prev2 + 2 * prev1) % 8 == 0:
                    ans += j - 1              # all length>=3: last three digits
            elif c == 7:
                ans += cnt7[C]
            # register current prefix states for future endpoints
            cnt3[ps9 % 3] += 1
            cnt9[ps9] += 1
            cnt7[C] += 1
            prev2 = prev1
            prev1 = c
        return ans
```
