```python
class Solution:
    def calculateScore(self, s: str) -> int:
        stacks = [[] for _ in range(26)]
        score = 0
        for i, ch in enumerate(s):
            c = ord(ch) - 97
            m = 25 - c
            if stacks[m]:
                j = stacks[m].pop()
                score += i - j
            else:
                stacks[c].append(i)
        return score
```

Note mirror index: mirror of letter with index c (0-based) is 25 - c. E.g., 'a' (0) -> 'z' (25). ✓

```python
class Solution:
    def calculateScore(self, s: str) -> int:
        # For each of the 26 letters, keep a stack of unmarked indices
        # (indices that were skipped and are still available for matching).
        # The top of a stack is always the closest (largest) unmarked index
        # with that letter.
        stacks = [[] for _ in range(26)]
        score = 0
        
        for i, ch in enumerate(s):
            c = ord(ch) - 97      # 0-based index of current letter
            m = 25 - c            # mirror letter's index (a<->z, b<->y, ...)
            st = stacks[m]
            if st:
                # Closest unmarked index j with the mirror letter
                j = st.pop()      # marking j (removing it) and i (never pushed)
                score += i - j
            else:
                # No matching mirror to the left; leave i unmarked for future
                stacks[c].append(i)
        
        return score
```
