```python
class Solution:
    def countOfSubstrings(self, word: str, k: int) -> int:
        vowels = set('aeiou')
        n = len(word)
        ans = 0
        for i in range(n):
            vcnt = {}
            cons = 0
            for j in range(i, n):
                ch = word[j]
                if ch in vowels:
                    vcnt[ch] = vcnt.get(ch, 0) + 1
                else:
                    cons += 1
                if cons == k and len(vcnt) == 5:
                    ans += 1
                if cons > k:
                    break  # consonants only increase as j increases; can break early
        return ans
```

Looking at this problem, I'll use the classic "at most" decomposition: count substrings with all 5 vowels and **at most** k consonants, then subtract the count for **at most** k−1 consonants.

For `atMost(k)`, I maintain two monotone pointers per right endpoint `r`:
- `cleft`: minimal `l` such that `[l, r]` has ≤ k consonants (valid `l` must satisfy `l ≥ cleft`).
- `vleft`: maximal `l` such that `[l, r]` contains all 5 vowels (valid `l` must satisfy `l ≤ vleft`).

Then valid substrings ending at `r` number `max(0, vleft - cleft + 1)`. Both pointers only move forward, giving O(n).

I verified this against all three examples (getting 0, 1, and 3 respectively) plus extra cases like `"aeiouq"`, `"qaeiouq"` (k=1 → 2), and `"aeioubaeiou"` (k=1 → 21, matching a manual brute-force count).

```python
class Solution:
    def countOfSubstrings(self, word: str, k: int) -> int:
        return self._at_most(word, k) - self._at_most(word, k - 1)

    def _at_most(self, word: str, k: int) -> int:
        # Number of substrings containing all 5 vowels and at most k consonants.
        if k < 0:
            return 0
        vowels = {'a', 'e', 'i', 'o', 'u'}
        n = len(word)
        total = 0
        cleft, cons = 0, 0   # window [cleft, r] kept with <= k consonants (minimal cleft)
        vleft = 0            # maximal l such that [l, r] contains all 5 vowels
        vcnt = {'a': 0, 'e': 0, 'i': 0, 'o': 0, 'u': 0}
        distinct = 0
        for r in range(n):
            ch = word[r]
            if ch in vowels:
                vcnt[ch] += 1
                if vcnt[ch] == 1:
                    distinct += 1
            else:
                cons += 1
            # Shrink consonant window until it has at most k consonants.
            while cons > k:
                if word[cleft] not in vowels:
                    cons -= 1
                cleft += 1
            # Advance vleft while [vleft+1, r] still contains all 5 vowels.
            while distinct == 5:
                c = word[vleft]
                if c in vowels:
                    if vcnt[c] == 1:
                        break
                    vcnt[c] -= 1
                vleft += 1
            # Valid start positions: l in [cleft, vleft]
            if distinct == 5 and vleft >= cleft:
                total += vleft - cleft + 1
        return total
```
