```python
class Solution:
    def findMaximumScore(self, nums: List[int]) -> int:
        total = 0
        cur_max = nums[0]
        for m in range(len(nums) - 1):
            cur_max = max(cur_max, nums[m])
            total += cur_max
        return total
```

```python
class Solution:
    def findMaximumScore(self, nums: List[int]) -> int:
        # Any path 0 = i0 < i1 < ... < ik = n-1 has score
        # sum over each unit step m -> m+1 of nums[p], where p is the
        # last jump point <= m. This is at most max(nums[0..m]) for each m,
        # and the greedy (jump to next strictly greater value, else to end)
        # achieves exactly that bound. So the answer is
        # sum_{m=0}^{n-2} prefix_max(m).
        total = 0
        cur_max = nums[0]
        for m in range(len(nums) - 1):
            if nums[m] > cur_max:
                cur_max = nums[m]
            total += cur_max
        return total
```
