Note: The empty substring is considered valid.

```python
class Solution:
    def shortestMatchingSubstring(self, s: str, p: str) -> int:
        A, B, C = p.split('*')
        n = len(s)
        INF = float('inf')
        
        def kmp_occurrences(pat):
            # returns sorted list of start indices of pat in s
            m = len(pat)
            if m == 0:
                return []
            if m > n:
                return []
            fail = [0] * m
            k = 0
            for i in range(1, m):
                c = pat[i]
                while k and c != pat[k]:
                    k = fail[k - 1]
                if c == pat[k]:
                    k += 1
                fail[i] = k
            occ = []
            k = 0
            for i in range(n):
                c = s[i]
                while k and c != pat[k]:
                    k = fail[k - 1]
                if c == pat[k]:
                    k += 1
                    if k == m:
                        occ.append(i - m + 1)
                        k = fail[k - 1]
            return occ
        
        # bestA[i] for i in 0..n: max start a with A at a and a + len(A) <= i; -1 if none
        # bestC[i] for i in 0..n: min end e with C at e - len(C) and e - len(C) >= i; INF if none
        
        la, lb, lc = len(A), len(B), len(C)
        
        bestA = [-1] * (n + 1)
        if la == 0:
            for i in range(n + 1):
                bestA[i] = i
        else:
            occA = kmp_occurrences(A)
            ptr = 0
            cur = -1
            na = len(occA)
            for i in range(n + 1):
                limit = i - la
                while ptr < na and occA[ptr] <= limit:
                    cur = occA[ptr]
                    ptr += 1
                bestA[i] = cur
        
        bestC = [INF] * (n + 1)
        if lc == 0:
            for i in range(n + 1):
                bestC[i] = i
        else:
            occC = kmp_occurrences(C)
            nc = len(occC)
            j = nc
            for i in range(n, -1, -1):
                while j > 0 and occC[j - 1] >= i:
                    j -= 1
                if j < nc:
                    bestC[i] = occC[j] + lc
                # else remains INF
        
        ans = INF
        if lb == 0:
            # B occurs at every position 0..n
            for b in range(n + 1):
                a = bestA[b]
                if a < 0:
                    continue
                e = bestC[b]
                if e == INF:
                    continue
                cand = e - a
                if cand < ans:
                    ans = cand
        else:
            occB = kmp_occurrences(B)
            for b in occB:
                a = bestA[b]
                if a < 0:
                    continue
                idx = b + lb
                if idx > n:
                    continue  # can't happen since b + lb <= n for occurrences, but safe
                e = bestC[idx]
                if e == INF:
                    continue
                cand = e - a
                if cand < ans:
                    ans = cand
        
        return -1 if ans == INF else ans
```

```python
class Solution:
    def shortestMatchingSubstring(self, s: str, p: str) -> int:
        A, B, C = p.split('*')
        n = len(s)
        SENT = n + 1

        def find_all(pat):
            m = len(pat)
            if m == 0 or m > n:
                return []
            if m == 1:
                return [i for i, c in enumerate(s) if c == pat]
            pi = [0] * m
            k = 0
            for i in range(1, m):
                c = pat[i]
                while k and c != pat[k]:
                    k = pi[k - 1]
                if c == pat[k]:
                    k += 1
                pi[i] = k
            res = []
            append = res.append
            k = 0
            last = m - 1
            for i, c in enumerate(s):
                while k and c != pat[k]:
                    k = pi[k - 1]
                if c == pat[k]:
                    k += 1
                    if k == m:
                        append(i - last)
                        k = pi[last]
            return res

        la, lb, lc = len(A), len(B), len(C)

        if la == 0:
            bestA = list(range(n + 1))
        else:
            occA = find_all(A)
            bestA = [-1] * (n + 1)
            fill = 0
            cur = -1
            for a in occA:
                e = a + la
                if e > fill:
                    bestA[fill:e] = [cur] * (e - fill)
                    fill = e
                cur = a
            if fill <= n:
                bestA[fill:] = [cur] * (n + 1 - fill)

        if lc == 0:
            bestC = list(range(n + 1))
        else:
            occC = find_all(C)
            bestC = [SENT] * (n + 1)
            for j in range(len(occC) - 1, -1, -1):
                hi = occC[j]
                lo = occC[j - 1] + 1 if j else 0
                bestC[lo:hi + 1] = [hi + lc] * (hi + 1 - lo)

        if lb == 0:
            cands = [bestC[b] - bestA[b] for b in range(n + 1)

        else:
            occB = find_all(B)
```

```python
class Solution:
    def shortestMatchingSubstring(self, s: str, p: str) -> int:
        A, B, C = p.split('*')
        n = len(s)
        SENT = n + 1  # sentinel meaning "no valid occurrence end"

        def find_all(pat):
            """All (overlapping) start indices where pat occurs in s (sorted)."""
            m = len(pat)
            if m == 0 or m > n:
                return []
            if m == 1:
                return [i for i, c in enumerate(s) if c == pat]
            # KMP failure table
            pi = [0] * m
            k = 0
            for i in range(1, m):
                c = pat[i]
                while k and c != pat[k]:
                    k = pi[k - 1]
                if c == pat[k]:
                    k += 1
                pi[i] = k
            res = []
            append = res.append
            k = 0
            last = m - 1
            for i, c in enumerate(s):
                while k and c != pat[k]:
                    k = pi[k - 1]
                if c == pat[k]:
                    k += 1
                    if k == m:
                        append(i - last)
                        k = pi[last]
            return res

        la, lb, lc = len(A), len(B), len(C)

        # bestA[i]: largest start a of an occurrence of A with a + la <= i (else -1)
        if la == 0:
            bestA = list(range(n + 1))
        else:
            occA = find_all(A)
            bestA = [-1] * (n + 1)
            fill = 0
            cur = -1
            for a in occA:
                e = a + la
                if e > fill:
                    bestA[fill:e] = [cur] * (e - fill)
                    fill = e
                cur = a
            bestA[fill:] = [cur] * (n + 1 - fill)

        # bestC[i]: smallest end c+lc of an occurrence of C with c >= i (else SENT)
        if lc == 0:
            bestC = list(range(n + 1))
        else:
            occC = find_all(C)
            bestC = [SENT] * (n + 1)
            for j in range(len(occC) - 1, -1, -1):
                hi = occC[j]
                lo = occC[j - 1] + 1 if j else 0
                bestC[lo:hi + 1] = [hi + lc] * (hi + 1 - lo)

        else:
            occB = find_all(B)
            cands = [bestC[b + lb] - bestA[b] for b in occB
                     if bestA[b] >= 0 and bestC[b + lb] <= n]

        return min(cands) if cands else -1
```

```python
            k = 0
            last = m - 1

                    if k == m:

                        k = pi[last]

                bestC[lo:hi + 1] = [hi + lc] * (hi + 1 - lo)
```

```python
        # Enumerate placements of B (every boundary position if B is empty)
        if lb == 0:
            cands = [bestC[b] - bestA[b] for b in range(n + 1)
                     if bestA[b] >= 0 and bestC[b] <= n]
        else:
            occB = find_all(B)
            cands = [bestC[b + lb] - bestA[b] for b in occB
                     if bestA[b] >= 0 and bestC[b + lb] <= n]

        return min(cands) if cands else -1
```
