class Solution:
    def hasMatch(self, s: str, p: str) -> bool:
        i = p.index('*')
        prefix, suffix = p[:i], p[i+1:]
        pos = s.find(prefix) if prefix else 0

class Solution:
    def hasMatch(self, s: str, p: str) -> bool:
        i = p.index('*')
        prefix, suffix = p[:i], p[i+1:]
        pos = s.find(prefix)
