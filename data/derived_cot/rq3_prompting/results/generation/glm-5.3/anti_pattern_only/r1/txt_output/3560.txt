class Solution:
    def maxMoves(self, kx: int, ky: int, positions: List[List[int]]) -> int:
        from collections import deque
        n = len(positions)
        pts = [(p[0], p[1]) for p in positions] + [(kx, ky)]
        m = n + 1
        N = 50
        
        dist = [[0] * m for _ in range(m)]
        knight_moves = ((1,2),(2,1),(-1,2),(-2,1),(1,-2),(2,-1),(-1,-2),(-2,-1))
        for s in range(m):
            sx, sy = pts[s]
            d = [[-1] * N for _ in range(N)]
            d[sx][sy] = 0
            q = deque([(sx, sy)])
            while q:
                x, y = q.popleft()
                nd = d[x][y] + 1
                for dx, dy in knight_moves:
                    nx, ny = x + dx, y + dy
                    if 0 <= nx < N and 0 <= ny < N and d[nx][ny] < 0:
                        d[nx][ny] = nd
                        q.append((nx, ny))
            for t in range(m):
                dist[s][t] = d[pts[t][0]][pts[t][1]]
        
        full = (1 << n) - 1
        
        # popcount table
        popcnt = [0] * (full + 1)
        for mask in range(1, full + 1):
            popcnt[mask] = popcnt[mask >> 1] + (mask & 1)

        for mask in range(full - 1, -1, -1):
            alice = (popcnt[mask] & 1) == 0

            unset = []

                unset.append((b, b.bit_length() - 1))

                curs.append(n)

- The knight's position after each capture is exactly the captured pawn's square, and the number of moves made in a turn is the shortest knight distance (pawns never block a knight).

- Precompute all pairwise knight distances with BFS from each of the ≤16 relevant squares on the 50×50 board.

            r = full ^ mask
            while r:
                b = r & (-r)
                i = b.bit_length() - 1
                items.append((i, dp[mask | b][i]))
                r ^= b
            row = dp[mask]
            if mask == 0:
                curs = (n,)
            else:
                curs = []
                mm = mask
                while mm:
                    b = mm & (-mm)
                    curs.append(b.bit_length() - 1)
                    mm ^= b
            if alice_turn:
                for cur in curs:
                    dcur = dist[cur]
                    best = -1
                    for i, nv in items:
                        v = dcur[i] + nv
                        if v > best:
                            best = v
                    row[cur] = best
            else:
                for cur in curs:
                    dcur = dist[cur]
                    best = 1 << 30
                    for i, nv in items:
                        v = dcur[i] + nv
                        if v < best:
                            best = v
                    row[cur] = best

        return dp[0][n]
