class Solution:
    def solveQueries(self, nums: List[int], queries: List[int]) -> List[int]:
        n = len(nums)
        pos_map = {}
        for i, v in enumerate(nums):
            pos_map.setdefault(v, []).append(i)
        
        ans = [-1] * n
        for pos in pos_map.values():
            k = len(pos)
            if k == 1:
                continue
            for m in range(k):
                q = pos[m]
                # counterclockwise (decreasing direction) immediate neighbor
                prev = pos[m - 1] - n if m == 0 else pos[m - 1]
                # clockwise immediate neighbor
                nxt = pos[0] + n if m == k - 1 else pos[m + 1]
                ans[q] = min(q - prev, nxt - q)
        
        return [ans[q] for q in queries]

```python
from typing import List
from collections import defaultdict

class Solution:
    def solveQueries(self, nums: List[int], queries: List[int]) -> List[int]:
        n = len(nums)
        
        # Group indices by value (lists are naturally sorted by index)
        pos_map = defaultdict(list)
        for i, v in enumerate(nums):
            pos_map[v].append(i)
        
        # Precompute answer for every index (queries.length <= nums.length)
        ans = [-1] * n
        for pos in pos_map.values():
            k = len(pos)
            if k == 1:
                continue  # unique value -> -1
            
            for m in range(k):
                q = pos[m]
                # Immediate neighbor going counterclockwise (decreasing index, wrapping)
                # When m == 0, wrap to the last occurrence (pos[-1]) shifted by -n
                prev = pos[-1] - n if m == 0 else pos[m - 1]
                # Immediate neighbor going clockwise (increasing index, wrapping)
                # When m == k-1, wrap to the first occurrence shifted by +n
                nxt = pos[0] + n if m == k - 1 else pos[m + 1]
                
                # The minimum circular distance is achieved by one of the two
                # immediate neighbors in the circular ordering of occurrences.
                ans[q] = min(q - prev, nxt - q)
        
        return [ans[q] for q in queries]
```
