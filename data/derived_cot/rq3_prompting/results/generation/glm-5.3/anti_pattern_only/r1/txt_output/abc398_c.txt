```python
import sys
from collections import Counter

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    a = list(map(int, data[1:1+n]))
    cnt = Counter(a)
    best_val = -1
    best_label = -1
    for i in range(n):
        if cnt[a[i]] == 1 and a[i] > best_val:
            best_val = a[i]
            best_label = i + 1
    print(best_label)

main()
```
