- N has exactly 2 distinct prime factors.

import sys
from math import isqrt
from bisect import bisect_right

def main():
    data = sys.stdin.buffer.read().split()
    Q = int(data[0])
    LIMIT = 10 ** 6

    # primes up to LIMIT//2 (larger factor q never exceeds LIMIT//2 since pa>=2)
    PLIM = LIMIT // 2
    sieve = bytearray(b'\x01') * (PLIM + 1)
    sieve[0] = 0
    sieve[1] = 0
    for i in range(2, isqrt(PLIM) + 1):
        if sieve[i]:
            start = i * i
            sieve[start::i] = bytearray(len(range(start, PLIM + 1, i)))
    primes = [i for i in range(PLIM + 1) if sieve[i]]

    good = []
    ap = good.append
    for i, p in enumerate(primes):
        if p * p > LIMIT:
            break
        pa = p
        while True:
            maxq = LIMIT // pa
            hi = bisect_right(primes, maxq)
            if hi <= i + 1:
                break
            for q in primes[i + 1:hi]:
                v = pa * q
                while v <= LIMIT:
                    ap(v)
                    v *= q
            pa *= p

    good.sort()

    out = []
    for j in range(1, Q + 1):
        r = isqrt(int(data[j]))
        idx = bisect_right(good, r) - 1
        n = good[idx]
        out.append(str(n * n))

    sys.stdout.write("\n".join(out) + "\n")

main()

Looking at this problem:

- A 400 number is `p^(2a) * q^(2b)` with distinct primes p, q and a,b ≥ 1 — i.e., a perfect square `n²` where `n` has **exactly 2 distinct prime factors**.
- Since A ≤ 10¹², we need all such `n ≤ 10⁶`. For each query, answer = (largest n ≤ isqrt(A) with ω(n)=2)².

This generates ~280k values quickly. Then each query is an integer sqrt + binary search.
