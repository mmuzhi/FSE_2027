class Solution:
    def minCost(self, n: int, cost: List[List[int]]) -> int:
        m = n // 2
        cL = cost[0]
        cR = cost[n - 1]
        # state values
        s01 = cL[0] + cR[1]
        s02 = cL[0] + cR[2]
        s10 = cL[1] + cR[0]
        s12 = cL[1] + cR[2]
        s20 = cL[2] + cR[0]
        s21 = cL[2] + cR[1]
        for k in range(1, m):
            cL = cost[k]
            cR = cost[n - 1 - k]
            p01 = min(s10, s12, s20)
            p02 = min(s10, s20, s21)
            p10 = min(s01, s02, s21)
            p12 = min(s01, s20, s21)
            p20 = min(s01, s02, s12)

            s10 = p10 + cL[1] + cR[0]
            s12 = p12 + cL[1] + cR[2]
            s20 = p20 + cL[2] + cR[0]
            s21 = p21 + cL[2] + cR[1]

Looking at this problem:

- Pair up house `k` with house `n-1-k` (k = 0..n/2-1). Each pair must have two different colors (equidistant condition).
- Between consecutive pairs, the left houses are adjacent (must differ) and the right houses are adjacent (must differ).
- So DP over pairs with state (color of left house `a`, color of right house `b`), `a != b` → 6 states. Each state has exactly 3 valid predecessor states.

Transitions (predecessors for each new state):
- (0,1) ← min(s10, s12, s20)
- (0,2) ← min(s10, s20, s21)
- (1,0) ← min(s01, s02, s21)
- (1,2) ← min(s01, s20, s21)
- (2,0) ← min(s01, s02, s12)
- (2,1) ← min(s02, s10, s12)

This is O(n) with a tiny constant. Verified against both examples (yields 9 and 18).

from typing import List

class Solution:
    def minCost(self, n: int, cost: List[List[int]]) -> int:
        m = n // 2
        # pairs: (cost[k], cost[n-1-k]) for k = 0..m-1, processed outside-in
        it = zip(cost, reversed(cost[m:]))
        cL, cR = next(it)
        # dp states: s_ab = min cost so far, left house color a, right house color b (a != b)
        s01 = cL[0] + cR[1]
        s02 = cL[0] + cR[2]
        s10 = cL[1] + cR[0]
        s12 = cL[1] + cR[2]
        s20 = cL[2] + cR[0]
        s21 = cL[2] + cR[1]
        for cL, cR in it:
            # shared partial mins
            mA = min(s01, s02)
            mB = min(s10, s12)
            mC = min(s20, s21)
            # best predecessor for each new state
            p01 = min(mB, s20)   # min(s10, s12, s20)
            p02 = min(s10, mC)   # min(s10, s20, s21)
            p10 = min(mA, s21)   # min(s01, s02, s21)
            p12 = min(s01, mC)   # min(s01, s20, s21)
            p20 = min(mA, s12)   # min(s01, s02, s12)
            p21 = min(mB, s02)   # min(s02, s10, s12)
            s01 = p01 + cL[0] + cR[1]
            s02 = p02 + cL[0] + cR[2]
            s10 = p10 + cL[1] + cR[0]
            s12 = p12 + cL[1] + cR[2]
            s20 = p20 + cL[2] + cR[0]
            s21 = p21 + cL[2] + cR[1]
