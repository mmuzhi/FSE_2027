```python
    def maxSubarraySum(self, nums: List[int], k: int) -> int:
        INF = float('inf')

                cand = pref - min_pref[r]

            if pref < min_pref[r]:
```

```python
class Solution:
    def maxSubarraySum(self, nums: List[int], k: int) -> int:
        # Prefix sums: subarray sum = P[j] - P[i], length = j - i.
        # Need (j - i) % k == 0  <=>  i ≡ j (mod k).
        # For each j, subtract the minimum P[i] with the same residue and i < j.
        INF = float('inf')
        min_pref = [INF] * k
        min_pref[0] = 0  # P[0] = 0, residue 0
        pref = 0
        ans = -INF
        for j in range(1, len(nums) + 1):
            pref += nums[j - 1]
            r = j % k
            if min_pref[r] != INF:
                cand = pref - min_pref[r]
                if cand > ans:
                    ans = cand
            if pref < min_pref[r]:
                min_pref[r] = pref
        return ans
```
