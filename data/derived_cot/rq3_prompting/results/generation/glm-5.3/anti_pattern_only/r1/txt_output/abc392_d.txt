There are N dice.

Input

import sys
import numpy as np

def main():
    data = sys.stdin.buffer.read().split()
    ptr = 0
    n = int(data[ptr]); ptr += 1

    all_faces = []

    M = np.zeros((n, V), dtype=np.float64)
    M[rows, cols] = vals
    S = M @ M.T
    karr = np.asarray(ks, dtype=np.float64)
    probs = S / np.outer(karr, karr)
    iu = np.triu_indices(n, k=1)
    return float(probs[iu].max())


def solve_pure(n, counters, ks):
    from operator import mul
    best = 0.0
    for i in range(n):
        di = counters[i]
        ki = ks[i]
        for j in range(i + 1, n):
            dj = counters[j]
            if len(di) <= len(dj):
                si, sj = di, dj
            else:
                si, sj = dj, di
            common = si.keys() & sj.keys()
            if not common:
                continue
            s = sum(map(mul, map(si.__getitem__, common), map(sj.__getitem__, common)))
            p = s / (ki * ks[j])
            if p > best:
                best = p
    return best


def main():
    data = sys.stdin.buffer.read().split()
    ptr = 0
    n = int(data[ptr]); ptr += 1
    ks = []
    counters = []
    for _ in range(n):
        k = int(data[ptr]); ptr += 1
        counters.append(Counter(map(int, data[ptr:ptr + k])))
        ks.append(k)
        ptr += k
    try:
        import numpy as np
    except ImportError:
        np = None
    if np is not None:
        best = solve_numpy(n, counters, ks, np)
    else:
        best = solve_pure(n, counters, ks)
    print("%.15f" % best)


main()
