```python
class Solution:
    def countPathsWithXorValue(self, grid: List[List[int]], k: int) -> int:
        MOD = 10**9 + 7
        m, n = len(grid), len(grid[0])
        
        # dp[j] is a list of 16 counts: number of paths to current cell in column j
        # with each possible XOR value.
        dp = [[0] * 16 for _ in range(n)]
        dp[0][grid[0][0]] = 1
        
        # First row: only reachable from the left
        for j in range(1, n):
            val = grid[0][j]
            prev = dp[j - 1]
            cur = [0] * 16
            for v in range(16):
                if prev[v]:
                    cur[v ^ val] = (cur[v ^ val] + prev[v]) % MOD
            dp[j] = cur
        
        # Remaining rows
        for i in range(1, m):
            # First column: only reachable from above
            val = grid[i][0]
            prev = dp[0]
            cur = [0] * 16
            for v in range(16):
                if prev[v]:
                    cur[v ^ val] = (cur[v ^ val] + prev[v]) % MOD
            dp[0] = cur
            
            for j in range(1, n):
                val = grid[i][j]
                left = dp[j - 1]   # already updated for row i
                up = dp[j]         # still holds row i-1
                cur = [0] * 16
                for v in range(16):
                    c = left[v] + up[v]
                    if c:
                        cur[v ^ val] = (cur[v ^ val] + c) % MOD
                dp[j] = cur
        
        return dp[n - 1][k] % MOD
```
