```python
class Solution:
    def largestPalindrome(self, n: int, k: int) -> str:
        m = (n + 1) // 2
        # pow10 mod k
        p = [1] * (n + 1)
        for j in range(1, n + 1):
            p[j] = p[j-1] * 10 % k
        # weights for positions 1..m (1-indexed)
        w = [0] * (m + 1)
        for i in range(1, m + 1):
            if n % 2 == 1 and i == m:
                w[i] = p[i-1] % k  # = p[(n-1)//2]
            else:
                w[i] = (p[n-i] + p[i-1]) % k
        # trans table: trans[w][mask]
        full_mask = (1 << k) - 1
        # contribs for each w
        trans = [[0] * (1 << k) for _ in range(k)]
        for ww in range(k):
            contribs = set()
            for d in range(10):
                contribs.add(d * ww % k)
            contribs = sorted(contribs)
            for mask in range(1 << k):
                res = 0
                for b in range(k):
                    if (mask >> b) & 1:
                        for c in contribs:
                            res |= 1 << ((b + c) % k)
                trans[ww][mask] = res
        # DP backward: masks[i] for i = m+1 down to 2
        masks = [0] * (m + 2)
        masks[m + 1] = 1
        for i in range(m, 1, -1):
            masks[i] = trans[w[i]][masks[i+1]]
        # Greedy forward
        r = 0
        digits = []
        for i in range(1, m + 1):
            wi = w[i]
            nxt = masks[i + 1]
            lo = 1 if i == 1 else 0
            chosen = -1
            for d in range(9, lo - 1, -1):
                new_r = (r + d * wi) % k
                if (nxt >> ((k - new_r) % k)) & 1:
                    chosen = d
                    r = new_r
                    break
            digits.append(chosen)
        # build palindrome
        if n % 2 == 0:
            full = digits + digits[::-1]
        else:
            full = digits + digits[:-1][::-1]
        return ''.join(map(str, full))
```

```python
class Solution:
    def largestPalindrome(self, n: int, k: int) -> str:
        m = (n + 1) // 2
        
        # powers of 10 modulo k
        p = [1] * (n + 1)
        for j in range(1, n + 1):
            p[j] = p[j - 1] * 10 % k
        
        # w[i]: combined residue weight of free digit i (1-indexed, i = 1..m)
        w = [0] * (m + 1)
        for i in range(1, m + 1):
            if (n & 1) and i == m:
                w[i] = p[i - 1] % k
            else:
                w[i] = (p[n - i] + p[i - 1]) % k
        
        # distinct contributions d*w % k for d in 0..9
        contribs = [sorted({d * ww % k for d in range(10)}) for ww in range(k)]
        
        # lazy transition memo
        memo = {}
        
        def step(ww, mask):
            key = ww * 512 + mask
            v = memo.get(key)
            if v is not None:
                return v
            res = 0
            cs = contribs[ww]
            for b in range(k):
                if (mask >> b) & 1:
                    for c in cs:
                        res |= 1 << ((b + c) % k)
            memo[key] = res
            return res
        
        # masks[i]: bitmask of achievable remainders using free positions i..m
        masks = [0] * (m + 2)
        masks[m + 1] = 1
        for i in range(m, 1, -1):
            masks[i] = step(w[i], masks[i + 1])
        
        # greedy forward
        r = 0
        digits = []
        for i in range(1, m + 1):
            wi = w[i]
            nxt = masks[i + 1]
            lo = 1 if i == 1 else 0
            for d in range(9, lo - 1, -1):
                new_r = (r + d * wi) % k
                if (nxt >> ((k - new_r) % k)) & 1:
                    digits.append(d)
                    r = new_r
                    break
        
        if n & 1:
            full = digits + digits[-2::-1]
        else:
            full = digits + digits[::-1]
        return ''.join(map(str, full))
```

```python
class Solution:
    def largestPalindrome(self, n: int, k: int) -> str:
        m = (n + 1) // 2
        
        # powers of 10 modulo k
        p = [1] * (n + 1)
        for j in range(1, n + 1):
            p[j] = p[j - 1] * 10 % k
        
        # w[i]: residue weight of the i-th free digit (1-indexed, i = 1..m)
        w = [0] * (m + 1)
        odd = n & 1
        for i in range(1, m + 1):
            if odd and i == m:
                w[i] = p[i - 1] % k
            else:
                w[i] = (p[n - i] + p[i - 1]) % k
        
        # distinct contributions d*ww % k over d = 0..9
        contribs = [sorted({d * ww % k for d in range(10)}) for ww in range(k)]
        
        memo = {}
        
        def step(ww, mask):
            key = ww * 512 + mask
            v = memo.get(key)
            if v is not None:
                return v
            res = 0
            for c in contribs[ww]:
                for b in range(k):
                    if (mask >> b) & 1:
                        res |= 1 << ((b + c) % k)
            memo[key] = res
            return res
        
        # masks[i]: bitmask of remainders achievable using free digits i..m
        masks = [0] * (m + 2)
        masks[m + 1] = 1
        for i in range(m, 1, -1):
            masks[i] = step(w[i], masks[i + 1])
        
        # greedy: pick largest digit at each position that still allows completion
        r = 0
        digits = []
        for i in range(1, m + 1):
            wi = w[i]
            nxt = masks[i + 1]
            lo = 1 if i == 1 else 0
            for d in range(9, lo - 1, -1):
                new_r = (r + d * wi) % k
                if (nxt >> ((k - new_r) % k)) & 1:
                    digits.append(d)
                    r = new_r
                    break
        
        half = ''.join(map(str, digits))
        if odd:
            return half + half[-2::-1]
        return half + half[::-1]
```

```python
class Solution:
    def largestPalindrome(self, n: int, k: int) -> str:
        m = (n + 1) // 2
        
        # p[j] = 10^j mod k
        p = [1] * (n + 1)
        for j in range(1, n + 1):
            p[j] = p[j - 1] * 10 % k
        
        # w[i]: residue contribution coefficient of i-th free digit (1-indexed)
        odd = n & 1
        w = [0] * (m + 1)
        for i in range(1, m + 1):
            if odd and i == m:
                w[i] = p[i - 1] % k
            else:
                w[i] = (p[n - i] + p[i - 1]) % k
        
        # distinct values of d*w mod k for d in 0..9, per w value
        contribs = [sorted({d * ww % k for d in range(10)}) for ww in range(k)]
        
        # masks[i]: bitmask over remainders 0..k-1 achievable by digits i..m
        masks = [0] * (m + 2)
        masks[m + 1] = 1  # only remainder 0
        memo = {}
        cget = memo.get
        for i in range(m, 1, -1):
            ww = w[i]
            cur = masks[i + 1]
            key = (ww << 9) | cur
            v = cget(key)
            if v is None:
                v = 0
                cs = contribs[ww]
                for b in range(k):
                    if (cur >> b) & 1:
                        for c in cs:
                            v |= 1 << ((b + c) % k)
                memo[key] = v
            masks[i] = v
        
        # forward greedy
        r = 0
        digits = []
        for i in range(1, m + 1):
            wi = w[i]
            nxt = masks[i + 1]
            lo = 1 if i == 1 else 0
            for d in range(9, lo - 1, -1):
                nr = (r + d * wi) % k
                if (nxt >> ((k - nr) % k)) & 1:
                    digits.append(d)
                    r = nr
                    break
        
        half = ''.join(map(str, digits))
        if odd:
            return half + half[-2::-1]
        return half + half[::-1]
```

```python
class Solution:
    def largestPalindrome(self, n: int, k: int) -> str:
        m = (n + 1) // 2
        
        # p[j] = 10^j mod k
        p = [1] * (n + 1)
        for j in range(1, n + 1):
            p[j] = p[j - 1] * 10 % k
        
        # w[i]: residue weight of the i-th free digit (positions 1..m, 1-indexed)
        odd = n & 1
        w = [0] * (m + 1)
        for i in range(1, m + 1):
            if odd and i == m:
                w[i] = p[i - 1] % k          # middle digit (counted once)
            else:
                w[i] = (p[n - i] + p[i - 1]) % k
        
        # distinct values of (d * ww) mod k for d in 0..9, for each ww
        contribs = [sorted({d * ww % k for d in range(10)}) for ww in range(k)]
        
        # masks[i]: bitmask of remainders reachable by digits i..m (i >= 2)
        masks = [0] * (m + 2)
        masks[m + 1] = 1                      # empty suffix -> remainder 0
        memo = {}
        for i in range(m, 1, -1):
            ww = w[i]
            cur = masks[i + 1]
            key = (ww << 9) | cur
            v = memo.get(key)
            if v is None:
                v = 0
                cs = contribs[ww]
                for b in range(k):
                    if (cur >> b) & 1:
                        for c in cs:
                            v |= 1 << ((b + c) % k)
                memo[key] = v
            masks[i] = v
        
        # greedy from the most significant free digit
        r = 0
        digits = []
        for i in range(1, m + 1):
            wi = w[i]
            nxt = masks[i + 1]
            lo = 1 if i == 1 else 0
            for d in range(9, lo - 1, -1):
                nr = (r + d * wi) % k
                if (nxt >> ((k - nr) % k)) & 1:
                    digits.append(d)
                    r = nr
                    break
        
        half = ''.join(map(str, digits))
        if odd:
            return half + half[:-1][::-1]
        return half + half[::-1]
```

```python
class Solution:
    def largestPalindrome(self, n: int, k: int) -> str:
        m = (n + 1) // 2
        
        # p[j] = 10^j mod k
        p = [1] * (n + 1)
        for j in range(1, n + 1):
            p[j] = p[j - 1] * 10 % k
        
        odd = n & 1
        # w[i]: residue weight of the i-th free digit (1-indexed)
        w = [0] * (m + 1)
        for i in range(1, m + 1):
            if odd and i == m:
                w[i] = p[i - 1] % k  # central digit counted once
            else:
                w[i] = (p[n - i] + p[i - 1]) % k
        
        # distinct values of (d * ww) % k for d in 0..9
        contribs = [sorted({d * ww % k for d in range(10)}) for ww in range(k)]
        
        # masks[i]: bitmask of remainders reachable using free digits i..m
        masks = [0] * (m + 2)
        masks[m + 1] = 1  # empty suffix -> only remainder 0
        memo = {}
        for i in range(m, 1, -1):
            ww = w[i]
            cur = masks[i + 1]
            key = (ww << 9) | cur
            v = memo.get(key)
            if v is None:
                v = 0
                cs = contribs[ww]
                for b in range(k):
                    if (cur >> b) & 1:
                        for c in cs:
                            v |= 1 << ((b + c) % k)
                memo[key] = v
            masks[i] = v
        
        # greedy: largest digit at each free position that keeps a valid completion
        r = 0
        digits = []
        for i in range(1, m + 1):
            wi = w[i]
            nxt = masks[i + 1]
            lo = 1 if i == 1 else 0
            for d in range(9, lo - 1, -1):
                nr = (r + d * wi) % k
                if (nxt >> ((k - nr) % k)) & 1:
                    digits.append(d)
                    r = nr
                    break
        
        half = ''.join(map(str, digits))
        if odd:
            return half + half[:-1][::-1]
        return half + half[::-1]
```

Looking at this problem:

- A palindrome of length n is determined by its first m = ⌈n/2⌉ digits (d₁..d_m), where digit d_i contributes d_i·(10^(n-i) + 10^(i-1)) mod k to the value (single power for the middle digit when n is odd).
- Since k ≤ 9, I do a backward DP over the ≤ 9 remainders (as bitmasks) computing which remainders are achievable by each suffix of free digits, then greedily pick the largest digit at each position (from 9 down, first digit ≥ 1) that still allows completing to total ≡ 0 mod k.

I verified this against all provided examples (595, 8, 89898) and many hand-checked cases (e.g., n=8,k=7 → 99944999; n=4,k=6 → 8778; n=5,k=4 → 88988; n=10,k=8 → 8889999888). A valid palindrome always exists (e.g. "8"·n for k∈{2,4,8}, "d"·n for k=d∈{5,6,7}, "9"·n for k∈{1,3,9}), so the greedy never fails. Complexity is O(n·10), fine for n = 10⁵.
