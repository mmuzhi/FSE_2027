class Solution:
    def maxSubarrays(self, n: int, conflictingPairs: List[List[int]]) -> int:
        m = len(conflictingPairs)
        by_hi = [[] for _ in range(n + 1)]
        for i, p in enumerate(conflictingPairs):
            a, b = p[0], p[1]
            if a > b:
                a, b = b, a
            by_hi[b].append((a + 1, i))
        
        gain = [0] * m
        base = 0
        best1 = 1
        best2 = 1
        idx1 = -1
        
        for r in range(1, n + 1):
            for val, i in by_hi[r]:
                if val > best1:
                    best2 = best1
                    best1 = val
                    idx1 = i
                elif val > best2:
                    best2 = val
            base += r - best1 + 1
            if best1 > best2:
                gain[idx1] += best1 - best2
        
        return base + (max(gain) if m else 0)

from typing import List

class Solution:
    def maxSubarrays(self, n: int, conflictingPairs: List[List[int]]) -> int:
        m = len(conflictingPairs)
        by_hi = [[] for _ in range(n + 1)]
        for i, (a, b) in enumerate(conflictingPairs):
            if a > b:
                a, b = b, a
            by_hi[b].append((a + 1, i))
        
        gain = [0] * m
        base = 0
        best1 = 1  # largest active constraint (min allowed left bound)
        best2 = 1  # second largest
        idx1 = -1  # pair index achieving best1
        
        for r in range(1, n + 1):
            for val, i in by_hi[r]:
                if val > best1:
                    best2 = best1
                    best1 = val
                    idx1 = i
                elif val > best2:
                    best2 = val
            base += r - best1 + 1
            if best1 > best2:
                gain[idx1] += best1 - best2
        
        return base + max(gain)

Looking at this problem:

- For each right endpoint `r`, a subarray `[l, r]` is valid iff `l >= (min(a,b)+1)` for every pair with `max(a,b) <= r`. So valid count for `r` is `r - best1(r) + 1` where `best1(r)` is the max constraint (default 1).
- Removing a pair only helps at positions `r` where that pair's constraint is the **strict unique maximum**. The gain there equals `best1 - best2` (second max, default 1).
- So sweep `r` from 1 to n, maintain top two constraint values and the index of the max holder, accumulate base count and per-pair gains. Answer = base + max gain.
