class Solution:
    def hasIncreasingSubarrays(self, nums: List[int], k: int) -> bool:
        n = len(nums)
        run = [1] * n
        for i in range(1, n):
            if nums[i] > nums[i - 1]:
                run[i] = run[i - 1] + 1
        for a in range(n - 2 * k + 1):
            if run[a + k - 1] >= k and run[a + 2 * k - 1] >= k:
                return True
        return False

class Solution:
    def hasIncreasingSubarrays(self, nums: List[int], k: int) -> bool:
        n = len(nums)
        # run[i] = length of the maximal strictly increasing run ending at index i
        run = [1] * n
        for i in range(1, n):
            if nums[i] > nums[i - 1]:
                run[i] = run[i - 1] + 1
        
        # Need indices a with a+2k-1 < n such that:
        #  - nums[a..a+k-1] strictly increasing  <=> run[a+k-1] >= k
        #  - nums[a+k..a+2k-1] strictly increasing <=> run[a+2k-1] >= k
        for a in range(n - 2 * k + 1):
            if run[a + k - 1] >= k and run[a + 2 * k - 1] >= k:
                return True
        return False
