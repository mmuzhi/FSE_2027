import random
def pollard_brent(n):
    if n % 2 == 0: return 2
    while True:
        y = random.randrange(1, n)
        c = random.randrange(1, n)
        m = 128
        g = r = q = 1
        x = ys = y
        while g == 1:
            x = y
            for _ in range(r):
                y = (y*y + c) % n
            k = 0
            while k < r and g == 1:
                ys = y
                for _ in range(min(m, r - k)):
                    y = (y*y + c) % n
                    q = q * abs(x - y) % n
                g = gcd(q, n)
                k += m
            r <<= 1
        if g == n:
            g = 1
            while g == 1:
                ys = (ys*ys + c) % n
                g = gcd(abs(x - ys), n)
        if g != n:
            return g

def crt(residues, moduli):
    # pairwise coprime
    r0, m0 = residues[0], moduli[0]
    for r1, m1 in zip(residues[1:], moduli[1:]):
        # solve x ≡ r0 mod m0, x ≡ r1 mod m1
        g = ... coprime assumed
        t = (r1 - r0) * pow(m0, -1, m1) % m1
        r0 = r0 + m0 * t

import sys

from math import gcd, isqrt

def main():
    data = sys.stdin.buffer.read().split()
    T = int(data[0])

    # small primes up to 1000
    LIM = 1000
    sieve = bytearray([1]) * (LIM + 1)
    sieve[0] = sieve[1] = 0
    for i in range(2, isqrt(LIM) + 1):
        if sieve[i]:
            sieve[i*i::i] = bytearray(len(range(i*i, LIM + 1, i)))
    SP = [i for i in range(2, LIM + 1) if sieve[i]]

    def is_prime(n):
        # deterministic for n < 3,215,031,751 using bases 2,3,5,7
        for p in (2, 3, 5, 7):
            if n % p == 0:
                return n == p
        d = n - 1
        s = 0
        while not d & 1:
            d >>= 1
            s += 1
        for a in (2, 3, 5, 7):
            x = pow(a, d, n)
            if x == 1 or x == n - 1:
                continue
            ok = False
            for _ in range(s - 1):
                x = x * x % n
                if x == n - 1:
                    ok = True
                    break
            if not ok:
                return False
        return True

    def pollard(n):
        while True:
            c = random.randrange(1, n)
            y = random.randrange(0, n)
            g = r = q = 1
            x = ys = y
            while g == 1:
                x = y

                    ys = y

                        y = (y * y + c) % n
                        q = q * (x - y) % n
                    g = gcd(q, n)
                    k += 128
                r <<= 1

                g = 1

                    ys = (ys * ys + c) % n
                    g = gcd(x - ys, n)
            # g is a divisor; if g == n retry with new (c, y)
            if g != n:
