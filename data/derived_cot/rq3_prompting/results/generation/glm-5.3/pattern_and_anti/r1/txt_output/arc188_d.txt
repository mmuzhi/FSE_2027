```python
import sys

def main():
    input_data = sys.stdin.buffer.read().split()
    MOD = 998244353
    n = int(input_data[0])
    A = list(map(int, input_data[1:1+n]))
    B = list(map(int, input_data[1+n:1+2*n]))
    
    # x
    x = [(A[i] + 1) // 2 for i in range(n)]  # 1-indexed values
    if len(set(x)) != n:
        print(0); return
    
    # gamma per value v (1..n): gamma[v] = A[x^{-1}(v)] - (2v-1)
    gamma = [0] * (n + 1)
    for i in range(n):
        v = x[i]
        gamma[v] = A[i] - (2 * v - 1)  # 0 or 1
    
    forced = [0] * (n + 1)  # 0 = none
    for i in range(n):
        if B[i] != -1:
            v = x[i]
            w = (B[i] + 1) // 2
            forced[v] = w
    
    # find cycles in functional graph
    color = [0] * (n + 1)  # 0 unvisited, 1 in progress, 2 done
    removed = [False] * (n + 1)
    for s in range(1, n + 1):
        if color[s] != 0:
            continue
        path = []
        u = s
        while True:
            if color[u] == 0:
                color[u] = 1
                path.append(u)
                if forced[u] == 0:
                    break
                u = forced[u]
            elif color[u] == 1:
                # found a cycle starting at u; cycle = path from u's position
                idx = path.index(u)
                cycle = path[idx:]
                # check mixed
                g = set(gamma[c] for c in cycle)
                if len(g) == 1:
                    print(0); return
                for c in cycle:
                    removed[c] = True
                break
            else:
                break
        for pnode in path:
            color[pnode] = 2
    # (mark done after processing)
    
    # units: chains among non-removed nodes
    # in_forced
    in_forced = [False] * (n + 1)
    for v in range(1, n + 1):
        w = forced[v]
        if w != 0:
            in_forced[w] = True
    
    a = b = c = 0
    for s in range(1, n + 1):
        if removed[s] or in_forced[s]:
            continue
        # chain start s (could be a free node with forced[s]==0, or chain head)
        # walk
        vals = set()
        u = s
        while True:
            vals.add(gamma[u])
            if forced[u] == 0 or removed[u]:
                break
            u = forced[u]
        if len(vals) == 1:
            if 0 in vals:
                a += 1
            else:
                b += 1
        else:
            c += 1
    
    m = a + b + c
    # factorials
    fact = [1] * (max(n + 1, 3))
    for i in range(1, len(fact)):
        fact[i] = fact[i-1] * i % MOD
    ans = fact[m]
    if m >= 1:
        ans = (ans - (a + b) * fact[m-1]) % MOD
    if m >= 2:
        ans = (ans + a * b % MOD * fact[m-2]) % MOD
    print(ans % MOD)

main()
```

```python
import sys

def main():
    data = sys.stdin.buffer.read().split()
    MOD = 998244353
    n = int(data[0])
    A = [int(t) for t in data[1:1 + n]]
    B = [int(t) for t in data[1 + n:1 + 2 * n]]

    x = [(A[i] + 1) // 2 for i in range(n)]
    if len(set(x)) != n:
        print(0)
        return

    gamma = [0] * (n + 1)
    for i in range(n):
        v = x[i]
        gamma[v] = A[i] - 2 * v + 1  # 0 or 1

    forced = [0] * (n + 1)
    for i in range(n):
        if B[i] != -1:
            forced[x[i]] = (B[i] + 1) // 2

    color = [0] * (n + 1)  # 0 unvisited, 1 on stack, 2 done
    removed = [False] * (n + 1)
    for s in range(1, n + 1):
        if color[s]:
            continue
        path = []
        u = s
        found = -1
        while True:
            if color[u] != 0:
                if color[u] == 1:
                    found = u
                break
            color[u] = 1
            path.append(u)
            if forced[u] == 0:
                break
            u = forced[u]
        if found != -1:
            idx = path.index(found)
            cycle = path[idx:]
            if len(set(gamma[t] for t in cycle)) == 1:
                print(0)
                return
            for t in cycle:
                removed[t] = True
        for t in path:
            color[t] = 2

    in_forced = [False] * (n + 1)
    for v in range(1, n + 1):
        if forced[v]:
            in_forced[forced[v]] = True

    a = b = c = 0
    for s in range(1, n + 1):
        if removed[s] or in_forced[s]:
            continue
        vals = set()
        u = s
        while True:
            vals.add(gamma[u])
            if forced[u] == 0 or removed[forced[u]]:
                break
            u = forced[u]
        if len(vals) == 1:
            if 0 in vals:
                a += 1
            else:
                b += 1
        else:
            c += 1

    m = a + b + c
    fact = [1] * (m + 1)
    for i in range(1, m + 1):
        fact[i] = fact[i - 1] * i % MOD
    ans = fact[m]
    if m >= 1:
        ans = (ans - (a + b) * fact[m - 1]) % MOD
    if m >= 2:
        ans = (ans + a * b % MOD * fact[m - 2]) % MOD
    print(ans % MOD)

main()
```
