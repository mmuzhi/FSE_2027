class Solution:
    def maximumTotalSum(self, maximumHeight: List[int]) -> int:
        arr = sorted(maximumHeight, reverse=True)
        total = 0
        cur = arr[0]
        total += cur
        for h in arr[1:]:
            cur = min(cur - 1, h)
            if cur < 1:
                return -1
            total += cur
        return total
```
