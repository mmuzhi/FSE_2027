```python
class Solution:
    def maxSubarrays(self, n: int, conflictingPairs: List[List[int]]) -> int:
        groups = [[] for _ in range(n + 1)]
        for i, (a, b) in enumerate(conflictingPairs):
            lo, hi = (a, b) if a < b else (b, a)
            groups[hi].append((lo, i))
        
        m = len(conflictingPairs)
        gain = [0] * m
        best1 = (0, -1)
        best2 = (0, -1)
        total = 0
        for r in range(1, n + 1):
            for lo, i in groups[r]:
                if lo > best1[0]:
                    best2 = best1
                    best1 = (lo, i)
                elif lo > best2[0]:
                    best2 = (lo, i)
            if best1[0] > best2[0]:
                gain[best1[1]] += best1[0] - best2[0]
            total += r - best1[0]
        return total + max(gain)
```

```python
class Solution:
    def maxSubarrays(self, n: int, conflictingPairs: List[List[int]]) -> int:
        # Group pairs by their right endpoint (max of the two values)
        groups = [[] for _ in range(n + 1)]
        for i, (a, b) in enumerate(conflictingPairs):
            lo, hi = (a, b) if a < b else (b, a)
            groups[hi].append((lo, i))
        
        m = len(conflictingPairs)
        gain = [0] * m
        
        # Track the two largest left-boundary constraints (min values) among
        # pairs whose max <= current r, along with which pair provides each.
        best1_val, best1_idx = 0, -1
        best2_val, best2_idx = 0, -1
        
        total = 0
        for r in range(1, n + 1):
            for lo, i in groups[r]:
                if lo > best1_val:
                    best2_val, best2_idx = best1_val, best1_idx
                    best1_val, best1_idx = lo, i
                elif lo > best2_val:
                    best2_val, best2_idx = lo, i
            
            # If the top constraint is strictly unique, removing that pair
            # relaxes the left boundary from best1 to best2 for this r.
            if best1_val > best2_val:
                gain[best1_idx] += best1_val - best2_val
            
            total += r - best1_val
        
        return total + max(gain)
```
