```python
import sys

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    a = data[1:1+n]  # keep as bytes, convert to int
    a = list(map(int, a))
    
    pref = [0] * (n + 1)
    seen = [False] * (n + 1)
    cnt = 0
    for i in range(n):
        v = a[i]
        if not seen[v]:
            seen[v] = True
            cnt += 1
        pref[i + 1] = cnt
    
    suf = [0] * (n + 2)
    seen2 = [False] * (n + 1)
    cnt = 0
    for i in range(n - 1, -1, -1):
        v = a[i]
        if not seen2[v]:
            seen2[v] = True
            cnt += 1
        suf[i + 1] = cnt  # suf[j] = distinct count of A_j..A_N (1-indexed)
    
    ans = 0
    for i in range(1, n):
        s = pref[i] + suf[i + 1]
        if s > ans:
            ans = s
    print(ans)

main()
```
