class Solution:
    def numberOfComponents(self, properties: List[List[int]], k: int) -> int:
        n = len(properties)
        parent = list(range(n))

            while parent[x] != x:
                parent[x] = parent[parent[x]]
                x = parent[x]
            return x

            ra, rb = find(a), find(b)
            if ra != rb:
                parent[ra] = rb

            for j in range(i + 1, n):
                if len(sets[i] & sets[j]) >= k:
