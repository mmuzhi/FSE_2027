Alice and Bob are playing a fantasy battle game consisting of n rounds where they summon one of three magical creatures each round: a Fire Dragon, a Water Serpent, or an Earth Golem. In each round, players simultaneously summon their creature and are awarded points as follows:

If one player summons a Fire Dragon and the other summons an Earth Golem, the player who summoned the Fire Dragon is awarded a point.
If one player summons a Water Serpent and the other summons a Fire Dragon, the player who summoned the Water Serpent is awarded a point.

```python
class Solution:
    def countWinningSequences(self, s: str) -> int:
        MOD = 10**9 + 7
        n = len(s)
        beats = {'F': 'E', 'W': 'F', 'E': 'W'}
        moves = 'FWE'
        # dp[last][diff index]
        # offset
        off = n
        # dp arrays of size 2n+1 for each last move
        dp = [[0]*(2*n+1) for _ in range(3)]
        # first round
        a = s[0]
        for li, b in enumerate(moves):
            d = 0
            if beats[b] == a:
                d = 1
            elif beats[a] == b:
                d = -1
            dp[li][d + off] = 1
        for i in range(1, n):
            a = s[i]
            ndp = [[0]*(2*n+1) for _ in range(3)]
            for li, b in enumerate(moves):
                if beats[b] == a:
                    delta = 1
                elif beats[a] == b:
                    delta = -1
                else:
                    delta = 0
                for pj, pb in enumerate(moves):
                    if pj == li:
                        continue
                    src = dp[pj]
                    dst = ndp[li]
                    if delta == 0:
                        for k in range(2*n+1):
                            if src[k]:
                                dst[k] = (dst[k] + src[k]) % MOD
                    elif delta == 1:
                        for k in range(2*n):
                            if src[k]:
                                dst[k+1] = (dst[k+1] + src[k]) % MOD
                    else:
                        for k in range(1, 2*n+1):
                            if src[k]:
                                dst[k-1] = (dst[k-1] + src[k]) % MOD
            dp = ndp
        ans = 0
        for li in range(3):
            for k in range(off+1, 2*n+1):
                ans = (ans + dp[li][k]) % MOD
        return ans
```

```python
class Solution:
    def countWinningSequences(self, s: str) -> int:
        MOD = 10**9 + 7
        n = len(s)
        beats = {'F': 'E', 'W': 'F', 'E': 'W'}
        moves = 'FWE'
        off = n
        size = 2 * n + 1
        dp = [[0] * size for _ in range(3)]
        a = s[0]
        for li in range(3):
            b = moves[li]
            if beats[b] == a:
                delta = 1
            elif beats[a] == b:
                delta = -1
            else:
                delta = 0
            dp[li][off + delta] = 1
        for i in range(1, n):
            a = s[i]
            ndp = []
            # precompute deltas per li
            deltas = []
            for li in range(3):
                b = moves[li]
                if beats[b] == a:
                    deltas.append(1)
                elif beats[a] == b:
                    deltas.append(-1)
                else:
                    deltas.append(0)
            for li in range(3):
                # sum of dp from other two last moves
                j1 = (li + 1) % 3
                j2 = (li + 2) % 3
                total = [(x + y) % MOD for x, y in zip(dp[j1], dp[j2])]
                d = deltas[li]
                if d == 0:
                    ndp.append(total)
                elif d == 1:
                    ndp.append([0] + total[:-1])
                else:
                    ndp.append(total[1:] + [0])
            dp = ndp
        ans = 0
        for li in range(3):
            ans = (ans + sum(dp[li][off + 1:])) % MOD
        return ans % MOD
```

```python
class Solution:
    def countWinningSequences(self, s: str) -> int:
        MOD = 10**9 + 7
        n = len(s)
        beats = {'F': 'E', 'W': 'F', 'E': 'W'}  # key beats value
        moves = 'FWE'
        off = n
        size = 2 * n + 1

        # dp[li][k] = number of Bob sequences of length (rounds processed) ending in move `li`,
        # with score difference (Bob - Alice) = k - off
        dp = [[0] * size for _ in range(3)]
        a = s[0]
        for li in range(3):
            b = moves[li]
            if beats[b] == a:
                delta = 1
            elif beats[a] == b:
                delta = -1
            else:
                delta = 0
            dp[li][off + delta] = 1

        for i in range(1, n):
            a = s[i]
            deltas = []
            for li in range(3):
                b = moves[li]
                if beats[b] == a:
                    deltas.append(1)
                elif beats[a] == b:
                    deltas.append(-1)
                else:
                    deltas.append(0)

            ndp = []
            for li in range(3):
                j1 = (li + 1) % 3
                j2 = (li + 2) % 3
                # sum over previous last moves != current move (no two same in a row)
                total = [(x + y) % MOD for x, y in zip(dp[j1], dp[j2])]
                d = deltas[li]
                if d == 0:
                    ndp.append(total)
                elif d == 1:
                    ndp.append([0] + total[:-1])
                else:
                    ndp.append(total[1:] + [0])
            dp = ndp

        ans = 0
        for li in range(3):
            ans = (ans + sum(dp[li][off + 1:])) % MOD
        return ans
```
