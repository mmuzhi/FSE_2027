=== ANSWER ===
```python
class Solution:
    def minimumOperations(self, nums: List[int]) -> int:
        # Find the longest suffix with all distinct elements.
        # If that suffix starts at index i, we must remove the first i elements,
        # which takes ceil(i / 3) operations.
        seen = set()
        i = len(nums)
        for j in range(len(nums) - 1, -1, -1):
            if nums[j] in seen:
                break
            seen.add(nums[j])
            i = j
        return (i + 2) // 3
```
