```
def batch_floor_sum(n, a, m, bs):
    # returns array F with F[i] = sum_{k=0}^{n-1} floor((a*k + bs[i]) / m)
    B = np.array(bs, dtype=np.int64)
    N = np.full(len(bs), n, dtype=np.int64)
    ANS = np.zeros(len(bs), dtype=np.int64)
    while True:
        if a >= m:
            q = a // m
            if q:
                ANS += (N - 1) * N // 2 * q
                a -= q * m
        qB = B // m
        ANS += N * qB
        B -= qB * m
        Y = a * N + B
        mask = Y >= m
        if not mask.any():
            break
        Nn = Y // m
        Bn = Y - Nn * m
        N = np.where(mask, Nn, 0)
        B = np.where(mask, Bn, 0)
        a, m = m, a
    return ANS
```
