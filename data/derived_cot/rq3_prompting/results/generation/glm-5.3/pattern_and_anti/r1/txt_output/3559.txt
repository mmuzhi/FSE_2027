```python
class Solution:
    def minValidStrings(self, words: List[str], target: str) -> int:
        n = len(target)
        maxLen = [0] * n  # maxLen[j]: longest prefix of any word matching target at j
        
        for w in words:

            for i in range(n):
                v = z[m + 1 + i]
                if v > maxLen[i]:
                    maxLen[i] = v
        
        INF = float('inf')
        dp = [INF] * (n + 1)
        dp[0] = 0
        p = 0
        for i in range(1, n + 1):
            # advance p while invalid: p + maxLen[p] < i
            while p < i and p + maxLen[p] < i:
                p += 1
            if p < i and dp[p] + 1 < dp[i]:
                dp[i] = dp[p] + 1
        
        return dp[n] if dp[n] < INF else -1
```

```python
def zfunc(s):
    m = len(s)
    z = [0] * m
    z[0] = m
    l, r = 0, 0
    for i in range(1, m):
        if i < r:
            z[i] = min(r - i, z[i - l])
        while i + z[i] < m and s[z[i]] == s[i + z[i]]:
            z[i] += 1
        if i + z[i] > r:
            l, r = i, i + z[i]
    return z
```

```python
    def minValidStrings(self, words: List[str], target: str) -> int:
        n = len(target)
        max_len = [0] * n

        def zfunc(s):
```

```python
class Solution:
    def minValidStrings(self, words: List[str], target: str) -> int:
        n = len(target)
        max_len = [0] * n  # max_len[i]: longest prefix of any word matching target starting at i

        def zfunc(s):
            m = len(s)
            z = [0] * m
            l = r = 0
            for i in range(1, m):
                zi = 0
                if i < r:
                    zi = z[i - l]
                    rem = r - i
                    if zi > rem:
                        zi = rem
                while i + zi < m and s[zi] == s[i + zi]:
                    zi += 1
                z[i] = zi
                if i + zi > r:
                    l = i
                    r = i + zi
            return z

        for w in words:
            start = len(w) + 1
            z = zfunc(w + '\x00' + target)
            tail = z[start:]
            max_len = [b if b > a else a for a, b in zip(max_len, tail)]

        INF = float('inf')
        # dp is non-decreasing: truncating the last segment of a decomposition keeps validity,
        # so dp[i] = 1 + dp[j*] where j* is the smallest j < i with j + max_len[j] >= i.
        dp = [INF] * (n + 1)
        dp[0] = 0
        p = 0
        for i in range(1, n + 1):
            while p < i and p + max_len[p] < i:
                p += 1
            if p < i:
                dp[i] = dp[p] + 1

        return dp[n] if dp[n] != INF else -1
```
