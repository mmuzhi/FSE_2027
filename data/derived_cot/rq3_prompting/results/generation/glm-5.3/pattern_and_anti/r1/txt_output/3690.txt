class Solution:
    def minLength(self, s: str, numOps: int) -> int:
        n = len(s)
        # runs
        runs = []
        i = 0
        while i < n:
            j = i
            while j < n and s[j] == s[i]:
                j += 1
            runs.append(j - i)
            i = j

        lo, hi = 1, n
