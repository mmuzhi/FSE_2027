class Solution:
    def minMaxSums(self, nums: List[int], k: int) -> int:
        MOD = 10**9 + 7
        nums.sort()
        n = len(nums)
        K = k - 1
        
        # Precompute inverses 1..n
        inv = [0] * (n + 1)
        if n >= 1:
            inv[1] = 1 if n >= 1 else 0
        for i in range(2, n + 1):
            inv[i] = (MOD - (MOD // i) * inv[MOD % i]) % MOD
        
        # S[m] = sum_{j=0}^{min(m,K)} C(m, j)
        S = [0] * n
        S[0] = 1
        c = 1 if K == 0 else 0  # C(0, K)
        for m in range(1, n):
            S[m] = (2 * S[m-1] - c) % MOD
            # update c to C(m, K)
            if m < K:
                c = 0
            elif m == K:
                c = 1
            else:
                c = c * m % MOD * inv[m - K] % MOD
        
        ans = 0
        for i in range(n):
            ans = (ans + nums[i] * (S[i] + S[n-1-i])) % MOD

class Solution:
    def minMaxSums(self, nums: List[int], k: int) -> int:
        MOD = 10**9 + 7
        nums.sort()
        n = len(nums)
        K = k - 1  # max number of additional elements besides the anchor

        # Precompute modular inverses 1..n
        inv = [0] * (n + 1)
        if n >= 1:
            inv[1] = 1
        for i in range(2, n + 1):
            inv[i] = (MOD - (MOD // i) * inv[MOD % i]) % MOD

        # S[m] = sum_{j=0}^{min(m,K)} C(m, j)  (mod MOD)
        # Recurrence: S[m] = 2*S[m-1] - C(m-1, K)
        S = [0] * n
        S[0] = 1
        c = 1 if K == 0 else 0  # holds C(m-1, K) at start of each iteration
        for m in range(1, n):
            S[m] = (2 * S[m - 1] - c) % MOD
            # update c to C(m, K)
            if m < K:
                c = 0
            elif m == K:
                c = 1
            else:
                c = c * m % MOD * inv[m - K] % MOD

        # Element at sorted index i is the max for S[i] subsequences
        # (choose <=K from the i smaller-index elements),
        # and the min for S[n-1-i] subsequences.
        ans = 0
        for i in range(n):
            ans = (ans + nums[i] * (S[i] + S[n - 1 - i])) % MOD
        return ans
