```python
class Solution:
    def maxScore(self, grid: List[List[int]]) -> int:
        from collections import defaultdict
        rows_with = defaultdict(list)
        for r, row in enumerate(grid):
            for v in set(row):
                rows_with[v].append(r)
        n = len(grid)
        NEG = float('-inf')
        dp = [NEG] * (1 << n)
        dp[0] = 0
        for v, rlist in rows_with.items():
            ndp = dp[:]
            for mask in range(1 << n):
                if dp[mask] == NEG:
                    continue
                for r in rlist:
                    if not (mask >> r) & 1:
                        nm = mask | (1 << r)
                        if dp[mask] + v > ndp[nm]:
                            ndp[nm] = dp[mask] + v
            dp = ndp
        return max(dp)
```

```python
class Solution:
    def maxScore(self, grid: List[List[int]]) -> int:
        from collections import defaultdict
        
        n = len(grid)
        # For each distinct value, list the rows that contain it
        rows_with = defaultdict(list)
        for r, row in enumerate(grid):
            for v in set(row):
                rows_with[v].append(r)
        
        NEG = float('-inf')
        # dp[mask] = max score using exactly the rows in mask
        dp = [NEG] * (1 << n)
        dp[0] = 0
        
        for v, rlist in rows_with.items():
            ndp = dp[:]  # option: skip this value entirely
            for mask in range(1 << n):
                if dp[mask] == NEG:
                    continue
                base = dp[mask] + v
                for r in rlist:
                    if not (mask >> r) & 1:
                        nm = mask | (1 << r)
                        if base > ndp[nm]:
                            ndp[nm] = base
            dp = ndp
        
        # values are positive, so the maximum always selects at least one cell
        return max(dp)
```
