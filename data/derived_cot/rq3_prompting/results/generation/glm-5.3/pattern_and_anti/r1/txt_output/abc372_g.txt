def floor_sum(n, m, a, b):
    # sum_{i=0}^{n-1} floor((a*i+b)/m), m>=1
    ans = 0
    if a < 0:
        a2 = a % m  # a2 in [0, m)
        ans += (n-1)*n//2 * ((a2 - a)//m)
        a = a2
    if b < 0:
        b2 = b % m
        ans += n * ((b2 - b)//m)
        b = b2
    while True:
        if a >= m:
            ans += (n-1)*n//2 * (a//m)
            a %= m
        if b >= m:
            ans += n * (b//m)
            b %= m
        y_max = a*n + b
        if y_max < m: break  # hmm, standard: if a*n+b < m: break
        n = y_max // m
        b = y_max % m
        m, a = a, m
        # careful: new n = (a*n+b)//m, and swap
    return ans

unsigned long long floor_sum_unsigned(n, m, a, b):
  ans = 0
  while True:
    if a >= m:
      ans += n*(n-1)/2 * (a//m)
      a %= m
    if b >= m:
      ans += n * (b//m)
      b %= m
    y_max = a*n + b
    if y_max < m: break
    n = y_max // m
    b = y_max % m
    m, a = a, m
  return ans

```cpp
unsigned long long floor_sum_unsigned(unsigned long long n, unsigned long long m, unsigned long long a, unsigned long long b) {
```

```python
import sys
from math import gcd

def floor_sum(n, m, a, b):
    # sum_{i=0}^{n-1} floor((a*i + b)/m), m >= 1
    ans = 0
    if a < 0:
        a2 = a % m
        ans -= (n * (n - 1) >> 1) * ((a2 - a) // m)
        a = a2
    if b < 0:
        b2 = b % m
        ans -= n * ((b2 - b) // m)
        b = b2
    while True:
        if a >= m:
            ans += (n * (n - 1) >> 1) * (a // m)
            a %= m
        if b >= m:
            ans += n * (b // m)
            b %= m
        y = a * n + b
        if y < m:
            break
        n, b = divmod(y, m)
        m, a = a, m
    return ans
```

```python
def _slope_key(L):
    return (L[1] << 60) // L[2]

def fsum(n, m, a, b):
    # sum_{i=0}^{n-1} (a*i + b) // m
    if n <= 60:
        s = 0
        for i in range(n):
            s += (a * i + b) // m
        return s
    ans = 0
    if a < 0:
        a2 = a % m
        ans -= (n * (n - 1) >> 1) * ((a2 - a) // m)
        a = a2
    if b < 0:
        b2 = b % m
        ans -= n * ((b2 - b) // m)
        b = b2
    while True:
        if a >= m:
            ans += (n * (n - 1) >> 1) * (a // m)
            a %= m
        if b >= m:
            ans += n * (b // m)
            b %= m
        y = a * n + b
        if y < m:
            break
        n, b = divmod(y, m)
        m, a = a, m
    return ans
```
