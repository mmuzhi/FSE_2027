```python
import sys

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0]); p = int(data[1])
    A = [[int(data[2 + i * n + j]) for j in range(n)] for i in range(n)]

    if p == 2:
        B = [[1 if x == 0 else x for x in row] for row in A]
        Bt = list(zip(*B))
        res = [[sum(a * b for a, b in zip(r, c)) % 2 for c in Bt] for r in B]
        sys.stdout.write('\n'.join(' '.join(map(str, r)) for r in res) + '\n')
        return

    SL = 9  # slot bytes: 72 bits; slot values < 100 * p^2 < 2^67

    def mat_mul(X, Y):
        packed = []
        for row in Y:
            barr = bytearray(SL * n)
            for k in range(n):
                x = row[k]
                if x:
                    barr[SL * k:SL * k + SL] = x.to_bytes(SL, 'little')
            packed.append(int.from_bytes(bytes(barr), 'little'))
        out = []
        for xrow in X:
            s = 0
            for a, pr in zip(xrow, packed):
                if a:
                    s += a * pr
            sb = s.to_bytes(SL * n, 'little')
            out.append([int.from_bytes(sb[SL * k:SL * k + SL], 'little') % p for k in range(n)])
        return out

    # M = C^p mod p, C = A
    result = None
    base = [row[:] for row in A]
    e = p
    while e:
        if e & 1:
            result = base if result is None else mat_mul(result, base)
        e >>= 1
        if e:
            base = mat_mul(base, base)
    M = result

    K = 0
    for i in range(n):
        for j in range(n):
            if A[i][j] == 0:
                K += 1
                if i == j:
                    for t in range(n):
                        M[t][j] = (M[t][j] + A[t][j]) % p
                        M[j][t] = (M[j][t] + A[j][t]) % p
                elif p == 3:
                    M[i][j] = (M[i][j] + A[j][i]) % p

    if K & 1:
        M = [[(-x) % p for x in row] for row in M]

    sys.stdout.write('\n'.join(' '.join(map(str, r)) for r in M) + '\n')

main()
```

Looking at this problem:

**Key derivation:** Each entry of $B^p$ is a polynomial of total degree ≤ p in the $x_k$ (values replacing zeros). Since $\sum_{x\in\mathbb{F}_p^*}x^e = -1$ if $(p-1)\mid e$ (including $e=0$), else 0, and degree ≤ p < 2(p−1), only monomials $x^{\emptyset}$ and $x_k^{p-1}$ survive, giving $\sum_B B^p = (-1)^K(C^p + \sum_k [x_k^{p-1}]B^p)$ where $[x_k^{p-1}]B^p=\sum_{j=1}^p E_k^{j-1}CE_k^{p-j}$.

- Diagonal zero $(d,d)$: $E^m=E$ for $m\ge1$, middle term $ECE=C_{dd}E=0$ ⇒ contributes $CE_{dd}+E_{dd}C$.
- Off-diagonal zero: $E^2=0$, so terms need $j\le2$ and $j\ge p-1$: only $p=3$, giving $ECE = C_{cr}E_{rc}$.
- $p=2$: unique $B$ (zeros→1), answer is $B^2 \bmod 2$.

So answer $=(-1)^K\big(C^p + \sum_{\text{diag zeros}}(CE_{dd}+E_{dd}C) + [p{=}3]\sum_{\text{offdiag zeros }(r,c)}C_{cr}E_{rc}\big)$. Verified against all three samples (sample 3 by hand: $C^{13}$ plus diagonal column/row corrections, negated, matches exactly).

Matrix power uses big-integer packing (9-byte slots, values $<100p^2<2^{67}$) for speed in pure Python.

```python
import sys

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0]); p = int(data[1])
    A = [[int(data[2 + i * n + j]) for j in range(n)] for i in range(n)]

    if p == 2:
        B = [[1 if x == 0 else x for x in row] for row in A]
        Bt = list(zip(*B))
        res = [[sum(a * b for a, b in zip(r, c)) % 2 for c in Bt] for r in B]
        sys.stdout.write('\n'.join(' '.join(map(str, r)) for r in res) + '\n')
        return

    SL = 9  # slot bytes (72 bits); slot sums < 100*p^2 < 2^67

    def mat_mul(X, Y):
        packed = []
        for row in Y:
            barr = bytearray(SL * n)
            for k in range(n):
                x = row[k]
                if x:
                    barr[SL * k:SL * k + SL] = x.to_bytes(SL, 'little')
            packed.append(int.from_bytes(bytes(barr), 'little'))
        out = []
        for xrow in X:
            s = 0
            for a, pr in zip(xrow, packed):
                if a:
                    s += a * pr
            sb = s.to_bytes(SL * n, 'little')
            out.append([int.from_bytes(sb[SL * k:SL * k + SL], 'little') % p for k in range(n)])
        return out

    result = None
    base = [row[:] for row in A]
    e = p
    while e:
        if e & 1:
            result = base if result is None else mat_mul(result, base)
        e >>= 1
        if e:
            base = mat_mul(base, base)
    M = result

    K = 0
    for i in range(n):
        for j in range(n):
            if A[i][j] == 0:
                K += 1
                if i == j:
                    for t in range(n):
                        M[t][j] = (M[t][j] + A[t][j]) % p
                        M[j][t] = (M[j][t] + A[j][t]) % p
                elif p == 3:
                    M[i][j] = (M[i][j] + A[j][i]) % p

    if K & 1:
        M = [[(-x) % p for x in row] for row in M]

    sys.stdout.write('\n'.join(' '.join(map(str, r)) for r in M) + '\n')

main()
```
