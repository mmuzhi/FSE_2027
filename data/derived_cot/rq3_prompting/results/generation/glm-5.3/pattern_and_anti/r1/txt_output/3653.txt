```python
class Solution:
    def maxSubarraySum(self, nums: List[int], k: int) -> int:
        INF = float('inf')
        min_pref = [INF] * k
        min_pref[0] = 0
        ans = -INF
        s = 0
        for i, x in enumerate(nums):
            s += x
            r = (i + 1) % k
            if min_pref[r] != INF:
                ans = max(ans, s - min_pref[r])
            if s < min_pref[r]:
                min_pref[r] = s
        return ans
```

```python
class Solution:
    def maxSubarraySum(self, nums: List[int], k: int) -> int:
        INF = float('inf')
        # min_pref[r] = minimum prefix sum P[i] with i % k == r (i < current)
        min_pref = [INF] * k
        min_pref[0] = 0  # P[0] = 0
        ans = -INF
        s = 0
        for i, x in enumerate(nums):
            s += x
            r = (i + 1) % k
            if min_pref[r] != INF:
                ans = max(ans, s - min_pref[r])
            if s < min_pref[r]:
                min_pref[r] = s
        return ans
```
