```python
class Solution:
    def maxAmount(self, initialCurrency: str, pairs1: List[List[str]], rates1: List[float], pairs2: List[List[str]], rates2: List[float]) -> float:
        from collections import deque, defaultdict

        def best_rates(pairs, rates, start):
            g = defaultdict(list)
            for (u, v), r in zip(pairs, rates):
                g[u].append((v, r))
                g[v].append((u, 1.0 / r))
            best = {start: 1.0}
            dq = deque([start])
            while dq:
                cur = dq.popleft()
                for nxt, r in g[cur]:
                    val = best[cur] * r
                    if nxt not in best:
                        best[nxt] = val
                        dq.append(nxt)
            return best

        day1 = best_rates(pairs1, rates1, initialCurrency)
        
        # reverse graph for day 2
        g2 = defaultdict(list)
        for (u, v), r in zip(pairs2, rates2):
            g2[v].append((u, r))     # original u->v becomes v->u
            g2[u].append((v, 1.0 / r))  # original v->u becomes u->v
        back = {initialCurrency: 1.0}
        dq = deque([initialCurrency])
        while dq:
            cur = dq.popleft()
            for nxt, r in g2[cur]:
                val = back[cur] * r
                if nxt not in back:
                    back[nxt] = val
                    dq.append(nxt)
        
        ans = 1.0
        for cur, amt in day1.items():
            if cur in back:
                ans = max(ans, amt * back[cur])
        return ans
```
