```python
class Solution:
    def maxEnergyBoost(self, energyDrinkA: List[int], energyDrinkB: List[int]) -> int:
        # dpA[i] = best total drinking A at hour i
        # dpB[i] = best total drinking B at hour i
        # dpA[i] = A[i] + max(dpA[i-1], dpB[i-2])  (i-2 = one cleansing hour after B)
        # dpB[i] = B[i] + max(dpB[i-1], dpA[i-2])
        prevA, prevB = energyDrinkA[0], energyDrinkB[0]  # dp values at hour i-1
        prevA2 = prevB2 = 0                              # dp values at hour i-2 (sentinel 0, never wins since values >= 1)
        for i in range(1, len(energyDrinkA)):
            curA = energyDrinkA[i] + max(prevA, prevB2)
            curB = energyDrinkB[i] + max(prevB, prevA2)
            prevA2, prevB2 = prevA, prevB
            prevA, prevB = curA, curB
        return max(prevA, prevB)
```
