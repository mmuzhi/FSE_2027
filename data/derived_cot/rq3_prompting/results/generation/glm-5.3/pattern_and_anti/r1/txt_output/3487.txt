You will be given a competitive programming problem.
You are given a string source of size n, a string pattern that is a subsequence of source, and a sorted integer array targetIndices that contains distinct numbers in the range [0, n - 1].
We define an operation as removing a character at an index idx from source such that:

idx is an element of targetIndices.
pattern remains a subsequence of source after removing the character.

Performing an operation does not change the indices of the other characters in source. For example, if you remove 'c' from "acb", the character at index 2 would still be 'b'.
Return the maximum number of operations that can be performed.
 
Example 1:

Input: source = "abbaa", pattern = "aba", targetIndices = [0,1,2]
Output: 1
Explanation:
We can't remove source[0] but we can do either of these two operations:

Remove source[1], so that source becomes "a_baa".
Remove source[2], so that source becomes "ab_aa".


Example 2:

Input: source = "bcda", pattern = "d", targetIndices = [0,3]
Output: 2
Explanation:
We can remove source[0] and source[3] in two operations.

Example 3:

Input: source = "dda", pattern = "dda", targetIndices = [0,1,2]
Output: 0
Explanation:
We can't remove any character from source.

Example 4:

Input: source = "yeyeykyded", pattern = "yeyyd", targetIndices = [0,2,3,4]
Output: 2
Explanation:
We can remove source[2] and source[3] in two operations.

 
Constraints:

1 <= n == source.length <= 3 * 10^3
1 <= pattern.length <= n
1 <= targetIndices.length <= n
targetIndices is sorted in ascending order.
The input is generated such that targetIndices contains distinct elements in the range [0, n - 1].
source and pattern consist only of lowercase English letters.
The input is generated such that pattern appears as a subsequence in source.

```python
import numpy as np

class Solution:
    def maxRemovals(self, source, pattern, targetIndices):
        n, m = len(source), len(pattern)
        INF = m + 1
        dp = np.full(m + 1, INF, dtype=np.int64)
        dp[0] = 0
        is_target = np.zeros(n, dtype=np.int64)
        is_target[targetIndices] = 1  # fancy index assignment with list
        # positions per char: 1-based j indices as numpy arrays
        pos = {}
        for j, c in enumerate(pattern, 1):
            pos.setdefault(c, []).append(j)
        pos = {c: np.array(v) for c, v in pos.items()}
        for i in range(n):
            js = pos.get(source[i])
            if js is not None:
                t = int(is_target[i])
                dp[js] = np.minimum(dp[js], dp[js - 1] + t)
        return len(targetIndices) - int(dp[m])
```

```python
from typing import List
from collections import defaultdict
import numpy as np

class Solution:
    def maxRemovals(self, source: str, pattern: str, targetIndices: List[int]) -> int:
        n, m = len(source), len(pattern)
        INF = m + 1
        dp = np.full(m + 1, INF, dtype=np.int64)
        dp[0] = 0
        is_target = np.zeros(n, dtype=np.int64)
        if targetIndices:
            is_target[np.array(targetIndices)] = 1
        pos = defaultdict(list)
        for j, c in enumerate(pattern, 1):
            pos[c].append(j)
        for c in pos:
            pos[c] = np.array(pos[c], dtype=np.int64)
        for i in range(n):
            js = pos.get(source[i]) if isinstance(pos, defaultdict) else pos.get(source[i])
            if js is not None:
                t = int(is_target[i])
                if t == 0:
                    dp[js] = np.minimum(dp[js], dp[js - 1])
                else:
                    dp[js] = np.minimum(dp[js], dp[js - 1] + 1)
        return len(targetIndices) - int(dp[m])
```

```python
from typing import List
import numpy as np

class Solution:
    def maxRemovals(self, source: str, pattern: str, targetIndices: List[int]) -> int:
        n, m = len(source), len(pattern)
        
        # dp[j] = minimum number of target indices used by an embedding of pattern[0..j)
        # into the prefix of source processed so far.
        INF = m + 1
        dp = np.full(m + 1, INF, dtype=np.int64)
        dp[0] = 0
        
        is_target = np.zeros(n, dtype=np.int64)
        is_target[np.asarray(targetIndices, dtype=np.int64)] = 1
        
        # 1-based positions of each character in pattern
        pos = {}
        for j, c in enumerate(pattern, 1):
            pos.setdefault(c, []).append(j)
        pos = {c: np.asarray(v, dtype=np.int64) for c, v in pos.items()}
        
        for i in range(n):
            js = pos.get(source[i])
            if js is not None:
                t = int(is_target[i])
                # Simultaneous update: dp[j] = min(dp[j], dp[j-1] + t) using old values
                # (NumPy evaluates the whole RHS before scatter assignment).
                dp[js] = np.minimum(dp[js], dp[js - 1] + t)
        
        return len(targetIndices) - int(dp[m])
```
