import numpy as np

def main():
    import sys
    data = sys.stdin.read().split()
    n = int(data[0])
    A = []
    B = []
    idx = 1
    for i in range(n):
        A.append(int(data[idx])); B.append(int(data[idx+1])); idx += 2
    S = sum(B)
    if S % 3 != 0:
        print(-1); return
    T = S // 3
    INF = 10**9  # use int32/int16
    # dp[x][y] = min movers, x,y in [0,T]
    size = T + 1
    dp = np.full((size, size), INF, dtype=np.int32)
    dp[0][0] = 0
    for a, b in zip(A, B):
        ndp = dp + (0 if a == 3 else 1)  # assign to team 3
        if b <= T:
            # assign to team 1: shift x by b
            c1 = np.full((size, size), INF, dtype=np.int32)
            c1[b:, :] = dp[:size-b, :] + (0 if a == 1 else 1)
            ndp = np.minimum(ndp, c1)

- Each person gets a final team; cost 0 if it equals their original team, else 1. Constraint: team 1 sum = T, team 2 sum = T (team 3 then forced to T).     if S % 3 != 0:
        print(-1); return
    T = S // 3
    if max(B) > T:
        print(-1); return
    try:
        import numpy as np
        INF = 1 << 28
        size = T + 1

        for a, b in zip(A, B):
            c1 = 0 if a == 1 else 1

            ndp = dp + c3
            arr1 = np.full((size, size), INF, dtype=np.int64)
            arr1[b:, :] = dp[:size - b, :] + c1

            arr2 = np.full((size, size), INF, dtype=np.int64)
            arr2[:, b:] = dp[:, :size - b] + c2
            ndp = np.minimum(ndp, arr2)
