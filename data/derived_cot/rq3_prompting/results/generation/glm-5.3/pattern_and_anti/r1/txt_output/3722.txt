```python
from itertools import accumulate
from typing import List

class Solution:
    def maxSum(self, nums: List[int], k: int, m: int) -> int:
        n = len(nums)
        NEG = float('-inf')
        prefix = [0] * (n + 1)
        for i, x in enumerate(nums):
            prefix[i + 1] = prefix[i] + x
        
        prev = [0] * (n + 1)  # dp[i][0] = 0
        for _ in range(k):
            g = [a - b for a, b in zip(prev, prefix)]          # dp[l][j-1] - prefix[l]
            run = list(accumulate(g, max))                      # best l' <= l
            h = [p + r for p, r in zip(prefix[m:], run)]        # best ending exactly at t (t = idx + m)
            cur = [NEG] * (n + 1)
            cur[m:] = accumulate(h, max)                        # dp[t][j] = prefix max of h
            prev = cur
        return prev[n]

from itertools import accumulate
from typing import List

class Solution:
    def maxSum(self, nums: List[int], k: int, m: int) -> int:
        n = len(nums)
        NEG = float('-inf')
        prefix = [0] * (n + 1)
        for i, x in enumerate(nums):
            prefix[i + 1] = prefix[i] + x
        
        prev = [0] * (n + 1)  # dp[i][0] = 0 (zero subarrays chosen)
        for _ in range(k):
            g = [a - b for a, b in zip(prev, prefix)]     # dp[l][j-1] - prefix[l]
            run = list(accumulate(g, max))                 # best over l' <= l
            h = [p + r for p, r in zip(prefix[m:], run)]   # best last subarray ending at t = idx + m
            cur = [NEG] * (n + 1)
            cur[m:] = accumulate(h, max)                   # dp[t][j] = prefix-max of h
            prev = cur
        return prev[n]
```

**Commit:** Classic partition DP. Let `dp[i][j]` = max sum picking exactly `j` non-overlapping subarrays (each length ≥ m) within the first `i` elements. Transition: last subarray is `nums[l..i-1]` with `l ≤ i-m`, so `dp[i][j] = max(dp[i-1][j], prefix[i] + max_{l ≤ i-m}(dp[l][j-1] - prefix[l]))`. The inner max over `l` is a running maximum, and `dp[·][j]` itself is a prefix-max of `h[t] = prefix[t] + runmax[t-m]` — both computable with `accumulate(…, max)`. That gives O(n·k) time (≤ ~4·10⁶ C-level/list-comp ops for n=2000), O(n) space.

**Invariant:** exactly-j semantics (needed for the all-negative example 2); invalid states stay `-inf` automatically since `dp[l][j-1] = -inf` for `l < (j-1)·m`.

**Recheck:** Hand-traced both examples — example 1 yields 13 (`[1,2]` + `[3,3,4]`), example 2 yields -10; also verified edge cases n=1, all-negative arrays, and `[1,2,3], k=2, m=1 → 6`. Feasibility is guaranteed by `k ≤ n//m`, so `dp[n][k]` is finite.

from itertools import accumulate
from typing import List

class Solution:
    def maxSum(self, nums: List[int], k: int, m: int) -> int:
        n = len(nums)
        NEG = float('-inf')
        prefix = [0] * (n + 1)
        for i, x in enumerate(nums):
            prefix[i + 1] = prefix[i] + x

        prev = [0] * (n + 1)  # dp[i][0] = 0: zero subarrays chosen
        for _ in range(k):
            # g[l] = dp[l][j-1] - prefix[l]
            g = [a - b for a, b in zip(prev, prefix)]
            # run[l] = max over l' <= l of g[l']
            run = list(accumulate(g, max))
            # h[idx] = best value when the j-th subarray ends at t = idx + m
            h = [p + r for p, r in zip(prefix[m:], run)]
            # dp[t][j] = prefix-max of h up to t; dp[t][j] = -inf for t < m
            cur = [NEG] * (n + 1)
            cur[m:] = accumulate(h, max)
            prev = cur

        return prev[n]
