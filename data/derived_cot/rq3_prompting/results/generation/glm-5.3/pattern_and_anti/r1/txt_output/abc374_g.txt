```python
import sys
from collections import deque

def main():
    input_data = sys.stdin.read().split()
    n = int(input_data[0])
    adj = [[False]*26 for _ in range(26)]
    outdeg = [0]*26
    indeg = [0]*26
    has_edge = [False]*26
    for i in range(1, n+1):
        s = input_data[i]
        u = ord(s[0]) - 65
        v = ord(s[1]) - 65
        if not adj[u][v]:
            adj[u][v] = True
            outdeg[u] += 1
            indeg[v] += 1
            has_edge[u] = True
            has_edge[v] = True
    # names are distinct so no dup anyway
    
    # transitive closure
    reach = [row[:] for row in adj]
    for k in range(26):
        rk = reach[k]
        for i in range(26):
            if reach[i][k]:
                ri = reach[i]
                for j in range(26):
                    if rk[j]:
                        ri[j] = True
    
    # weak components
    comp = [-1]*26
    cid = 0
    for i in range(26):
        if has_edge[i] and comp[i] == -1:
            # BFS undirected
            q = deque([i])
            comp[i] = cid
            while q:
                u = q.popleft()
                for w in range(26):
                    if adj[u][w] or adj[w][u]:
                        if comp[w] == -1:
                            comp[w] = cid
                            q.append(w)
            cid += 1
    
    total = 0
    for c in range(cid):
        verts = [v for v in range(26) if comp[v] == c]
        d = {v: outdeg[v] - indeg[v] for v in verts}
        P = sum(max(0, d[v]) for v in verts)
        # max flow: deficit -> surplus via reach
        # nodes: verts mapped to 0..m-1, source = m, sink = m+1
        idx = {v: k for k, v in enumerate(verts)}
        m = len(verts)
        S, T = m, m+1
        N = m + 2
        INF = 10**9
        cap = [[0]*N for _ in range(N)]
        for v in verts:
            if d[v] < 0:
                cap[S][idx[v]] = -d[v]
            elif d[v] > 0:
                cap[idx[v]][T] = d[v]
        for u in verts:
            if d[u] < 0:
                for w in verts:
                    if d[w] > 0 and reach[u][w]:
                        cap[idx[u]][idx[w]] = INF
        # Edmonds-Karp
        flow = 0
        while True:
            parent = [-1]*N
            parent[S] = S
            q = deque([S])
            while q and parent[T] == -1:
                u = q.popleft()
                cu = cap[u]
                for w in range(N):
                    if cu[w] > 0 and parent[w] == -1:
                        parent[w] = u
                        q.append(w)
                        if w == T:
                            break
            if parent[T] == -1:
                break
            # find bottleneck
            f = INF
            w = T
            while w != S:
                u = parent[w]
                f = min(f, cap[u][w])
                w = u
            w = T
            while w != S:
                u = parent[w]
                cap[u][w] -= f
                cap[w][u] += f
                w = u
            flow += f
        A = flow
        total += max(1, P - A)
    print(total)
```

Looking at this problem:

- Model each used name "XY" as a directed edge X→Y (26 vertices). Each NG string is a walk in this graph (every adjacent pair must be an edge), and all edges must be covered. Walks stay inside weakly connected components.
- Within a component, a set of walks traverses a multiset M ⊇ E (edges may be repeated). The minimum number of trails decomposing M is φ(M) = Σ max(0, out−in), i.e., number of open trails. So we minimize φ over multiplicities m(e) ≥ 1.
- Duplicating a path from a deficit vertex u to a surplus vertex w (u →* w) annihilates one surplus/deficit pair (extra copies are unlimited, so only reachability matters). A matching/charging argument shows no sequence of duplications can beat the maximum matching A between deficit units and surplus units where the deficit vertex reaches the surplus vertex.
- Hence per component the answer is max(1, P − A), where P = Σ max(0, outdeg−indeg) and A is a max flow: source → deficit vertices (cap |d|), reachability edges (cap ∞), surplus vertices → sink (cap d).

Implementation: Floyd–Warshall closure on 26 nodes, weak components via BFS, tiny Edmonds–Karp per component.

```python
import sys
from collections import deque

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    adj = [[False] * 26 for _ in range(26)]
    outd = [0] * 26
    ind = [0] * 26
    has = [False] * 26
    for i in range(1, n + 1):
        s = data[i].decode()
        u = ord(s[0]) - 65
        v = ord(s[1]) - 65
        adj[u][v] = True
        outd[u] += 1
        ind[v] += 1
        has[u] = True
        has[v] = True

    reach = [row[:] for row in adj]
    for k in range(26):
        rk = reach[k]
        for i in range(26):
            if reach[i][k]:
                ri = reach[i]
                for j in range(26):
                    if rk[j]:
                        ri[j] = True

    comp = [-1] * 26
    nc = 0
    for st in range(26):
        if has[st] and comp[st] == -1:
            comp[st] = nc
            dq = deque([st])
            while dq:
                u = dq.popleft()
                for w in range(26):
                    if comp[w] == -1 and (adj[u][w] or adj[w][u]):
                        comp[w] = nc
                        dq.append(w)
            nc += 1

    INF = 10 ** 9
    total = 0
    for c in range(nc):
        verts = [v for v in range(26) if comp[v] == c]
        m = len(verts)
        d = [outd[v] - ind[v] for v in verts]
        P = sum(x for x in d if x > 0)
        S, T = m, m + 1
        N = m + 2
        cap = [[0] * N for _ in range(N)]
        for i in range(m):
            if d[i] < 0:
                cap[S][i] = -d[i]
            elif d[i] > 0:
                cap[i][T] = d[i]
        for i in range(m):
            if d[i] < 0:
                for j in range(m):
                    if d[j] > 0 and reach[verts[i]][verts[j]]:
                        cap[i][j] = INF
        flow = 0
        while True:
            par = [-1] * N
            par[S] = S
            dq = deque([S])
            while dq:
                u = dq.popleft()
                if u == T:
                    break
                row = cap[u]
                for w in range(N):
                    if row[w] > 0 and par[w] == -1:
                        par[w] = u
                        dq.append(w)
            if par[T] == -1:
                break
            f = INF
            w = T
            while w != S:
                f = min(f, cap[par[w]][w])
                w = par[w]
            w = T
            while w != S:
                p = par[w]
                cap[p][w] -= f
                cap[w][p] += f
                w = p
            flow += f
        total += max(1, P - flow)
    print(total)

main()
```
