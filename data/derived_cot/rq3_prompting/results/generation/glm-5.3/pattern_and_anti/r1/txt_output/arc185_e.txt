    # returns array [base_val^0 .. base_val^n] mod MOD
    B = 1 << 10
    small = np.empty(B, np.int64)

    step = x  # base_val^B

    full = (seg[:, None] * small[None, :]) % MOD

import numpy as np

def build_pow(b, n, MOD):
    B = 1024
    small = np.empty(B, dtype=np.int64)
    x = 1
    for i in range(B):
        small[i] = x
        x = x * b % MOD
    step = x
    nseg = n // B + 1
    seg = np.empty(nseg, dtype=np.int64)
    y = 1
    for i in range(nseg):
        seg[i] = y
        y = y * step % MOD
    res = (seg[:, None] * small[None, :]) % MOD
    return res.reshape(-1)[:n + 1]

def main():
    data = sys.stdin.buffer.read().split()
    N = int(data[0])
    MOD = 998244353
    A = np.fromiter(map(int, data[1:1 + N]), dtype=np.int64, count=N)
    V = int(A.max())

    # Euler phi
    phi = np.arange(V + 1, dtype=np.int64)
    if V >= 2:
        sieve = np.ones(V + 1, dtype=bool)
        sieve[:2] = False
        i = 2
        while i * i <= V:
            if sieve[i]:
                sieve[i * i::i] = False
            i += 1
        for pr in np.flatnonzero(sieve).tolist():
            v = phi[pr::pr]
            v -= v // pr

    # divisor table
    ar = np.arange(1, V + 1, dtype=np.int64)
    lens_d = V // ar
    total = int(lens_d.sum())
    seg_starts = np.zeros(V, dtype=np.int64)
    np.cumsum(lens_d[:-1], out=seg_starts[1:])
    d_rep = np.repeat(ar, lens_d)
    inner = np.arange(total, dtype=np.int64) - np.repeat(seg_starts, lens_d)
    marr = d_rep * (inner + 1)
    tau_full = np.bincount(marr, minlength=V + 1)
    order = np.lexsort((d_rep, marr))
    divflat = d_rep[order].astype(np.int32)
    divptr = np.zeros(V + 2, dtype=np.int64)
    np.cumsum(tau_full, out=divptr[1:])
    del ar, lens_d, seg_starts, d_rep, inner, marr, order

    pw = build_pow(2, N, MOD)
    ipw = build_pow((MOD + 1) // 2, N, MOD)

    K = 4096
    W = np.zeros(V + 1, dtype=np.int64)
    ans = np.empty(N, dtype=np.int64)
    Fb = 0
    arangeK_cache = None

    for base in range(0, N, K):
        ab = A[base:base + K]
        Kk = ab.shape[0]
        lens = tau_full[ab]
        offs_el = np.zeros(Kk, dtype=np.int64)
        np.cumsum(lens[:-1], out=offs_el[1:])
        L = int(offs_el[-1] + lens[-1])
        jidx = np.repeat(np.arange(Kk, dtype=np.int64), lens)
        inner2 = np.arange(L, dtype=np.int64) - np.repeat(offs_el, lens)
        didx = divflat[divptr[ab][jidx] + inner2]

        ph = phi[didx]
        t3 = ph * W[didx] % MOD
        Dold = np.add.reduceat(t3, offs_el)

        pwloc = pw[base:base + Kk]
        pw_e = pwloc[jidx]
        w_orig = ph * pw_e % MOD

        order2 = np.argsort(didx, kind='stable')
        sd = didx[order2]
        w_s = w_orig[order2]
        wr_s = pw_e[order2]
        sj = jidx[order2]

        newg = np.empty(L, dtype=bool)
        newg[0] = True
        np.not_equal(sd[1:], sd[:-1], out=newg[1:])
        gid = np.cumsum(newg)
        C = np.cumsum(w_s)
        starts = np.flatnonzero(newg)
        baseC = C[starts] - w_s[starts]
        excl = C - w_s - baseC[gid - 1]
        excl %= MOD

        Din = np.bincount(sj, weights=excl.astype(np.float64), minlength=Kk).astype(np.int64)

        Wsum = np.add.reduceat(wr_s, starts)
        uniq = sd[starts]
        W[uniq] += Wsum
        W[uniq] %= MOD

        D %= MOD

        S = np.cumsum(G)
        S %= MOD

        ans[base:base + Kk] = Fl
        Fb = int(Fl[-1])

    out = '\n'.join(map(str, ans.tolist()))

import sys
import numpy as np

def build_pow(b, n, MOD):
    B = 1024
    small = np.empty(B, dtype=np.int64)
    x = 1
    for i in range(B):
        small[i] = x
        x = x * b % MOD
    step = x
    nseg = n // B + 1
    seg = np.empty(nseg, dtype=np.int64)
    y = 1
    for i in range(nseg):
        seg[i] = y
        y = y * step % MOD
    res = (seg[:, None] * small[None, :]) % MOD
    return res.reshape(-1)[:n + 1]

def main():
    data = sys.stdin.buffer.read().split()
    N = int(data[0])
    MOD = 998244353
    A = np.fromiter(map(int, data[1:1 + N]), dtype=np.int64, count=N)
    V = int(A.max())

    # Euler's totient via sieve
    phi = np.arange(V + 1, dtype=np.int64)
    if V >= 2:
        sieve = np.ones(V + 1, dtype=bool)
        sieve[:2] = False
        i = 2
        while i * i <= V:
            if sieve[i]:
                sieve[i * i::i] = False
            i += 1
        for pr in np.flatnonzero(sieve).tolist():
            v = phi[pr::pr]
            v -= v // pr

    # flat divisor table: for each v in 1..V, sorted list of divisors
    ar = np.arange(1, V + 1, dtype=np.int64)
    lens_d = V // ar
    total = int(lens_d.sum())
    seg_starts = np.zeros(V, dtype=np.int64)
    np.cumsum(lens_d[:-1], out=seg_starts[1:])
    d_rep = np.repeat(ar, lens_d)
    inner0 = np.arange(total, dtype=np.int64) - np.repeat(seg_starts, lens_d)
    marr = d_rep * (inner0 + 1)
    tau_full = np.bincount(marr, minlength=V + 1)
    ordr = np.lexsort((d_rep, marr))
    divflat = d_rep[ordr].astype(np.int32)
    divptr = np.zeros(V + 2, dtype=np.int64)
    np.cumsum(tau_full, out=divptr[1:])
    del ar, lens_d, seg_starts, d_rep, inner0, marr, ordr

    pw = build_pow(2, N, MOD)
    ipw = build_pow((MOD + 1) // 2, N, MOD)

    K = 4096
    W = np.zeros(V + 1, dtype=np.int64)
    ans = np.empty(N, dtype=np.int64)
    Fb = 0

    for base in range(0, N, K):
        ab = A[base:base + K]
        Kk = ab.shape[0]
        lens = tau_full[ab]
        offs_el = np.zeros(Kk, dtype=np.int64)
        np.cumsum(lens[:-1], out=offs_el[1:])
        L = int(offs_el[-1] + lens[-1])
        jidx = np.repeat(np.arange(Kk, dtype=np.int64), lens)
        innerb = np.arange(L, dtype=np.int64) - np.repeat(offs_el, lens)
        didx = divflat[divptr[ab][jidx] + innerb]

        ph = phi[didx]
        t3 = ph * W[didx] % MOD
        Dold = np.add.reduceat(t3, offs_el)

        pwloc = pw[base:base + Kk]
        pw_e = pwloc[jidx]
        w_orig = ph * pw_e % MOD

        ordb = np.argsort(didx, kind='stable')
        sd = didx[ordb]
        w_s = w_orig[ordb]
        wr_s = pw_e[ordb]
        sj = jidx[ordb]

        newg = np.empty(L, dtype=bool)
        newg[0] = True
        np.not_equal(sd[1:], sd[:-1], out=newg[1:])
        gid = np.cumsum(newg)
        C = np.cumsum(w_s)
        starts = np.flatnonzero(newg)
        baseC = C[starts] - w_s[starts]
        excl = C - w_s - baseC[gid - 1]
        excl %= MOD

        Din = np.bincount(sj, weights=excl.astype(np.float64),
                          minlength=Kk).astype(np.int64)

        Wsum = np.add.reduceat(wr_s, starts)
        uniq = sd[starts]
        W[uniq] += Wsum
        W[uniq] %= MOD

        D = Dold + Din
        D %= MOD
        G = D * ipw[1:Kk + 1] % MOD
        S = np.cumsum(G)
        S %= MOD
        Fl = pw[1:Kk + 1] * ((Fb + S) % MOD) % MOD
        ans[base:base + Kk] = Fl
        Fb = int(Fl[-1])

    sys.stdout.write('\n'.join(map(str, ans.tolist())) + '\n')

main()
