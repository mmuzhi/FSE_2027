```python
class Solution:
    def longestSpecialPath(self, edges: List[List[int]], nums: List[int]) -> List[int]:
        n = len(nums)
        adj = [[] for _ in range(n)]
        for u, v, w in edges:
            adj[u].append((v, w))
            adj[v].append((u, w))
        
        best_len = 0
        best_cnt = 1
        last_pos = {}
        path = []
        
        stack = [('e', 0, -1, 0, 0)]
        while stack:
            item = stack.pop()
            if item[0] == 'e':
                _, node, parent, dist, start = item
                v = nums[node]
                old = last_pos.get(v, -1)
                ns = start if old < start else old + 1
                path.append(dist)
                depth = len(path) - 1
                length = dist - path[ns]
                cnt = depth - ns + 1
                if length > best_len or (length == best_len and cnt < best_cnt):
                    best_len = length
                    best_cnt = cnt
                last_pos[v] = depth
                stack.append(('x', v, old))
                for nxt, w in adj[node]:
                    if nxt != parent:
                        stack.append(('e', nxt, node, dist + w, ns))
            else:
                _, v, old = item
                if old < 0:
                    del last_pos[v]
                else:
                    last_pos[v] = old
                path.pop()
        
        return [best_len, best_cnt]
```

```python
class Solution:
    def longestSpecialPath(self, edges: List[List[int]], nums: List[int]) -> List[int]:
        n = len(nums)
        adj = [[] for _ in range(n)]
        for u, v, w in edges:
            adj[u].append((v, w))
            adj[v].append((u, w))

        best_len = 0
        best_cnt = 1
        last_pos = {}   # value -> depth of last occurrence on current root path
        path = []       # path[d] = cumulative distance from root to node at depth d

        # iterative DFS; entries: ('e', node, parent, cumdist, start) / ('x', value, old)
        stack = [('e', 0, -1, 0, 0)]
        while stack:
            item = stack.pop()
            if item[0] == 'e':
                _, node, parent, dist, start = item
                v = nums[node]
                old = last_pos.get(v, -1)
                ns = start if old < start else old + 1  # new minimal window start
                path.append(dist)
                depth = len(path) - 1
                length = dist - path[ns]
                cnt = depth - ns + 1
                if length > best_len or (length == best_len and cnt < best_cnt):
                    best_len = length
                    best_cnt = cnt
                last_pos[v] = depth
                stack.append(('x', v, old))
                for nxt, w in adj[node]:
                    if nxt != parent:
                        stack.append(('e', nxt, node, dist + w, ns))
            else:
                _, v, old = item
                if old < 0:
                    del last_pos[v]
                else:
                    last_pos[v] = old
                path.pop()

        return [best_len, best_cnt]
```
