class Solution:
    def maxValue(self, nums: List[int], k: int) -> int:
        n = len(nums)
        K = k
        FULL = (1 << 128) - 1
        P0 = []
        for b in range(7):
            blk = (1 << (1 << b)) - 1
            step = 1 << (b + 1)
            p = 0
            for st in range(0, 128, step):
                p |= blk << st
            P0.append(p)
        P1 = [FULL ^ p for p in P0]

        def image(M, x):
            r = M
            b = 0
            while x:
                if x & 1:
                    r = ((r & P0[b]) << (1 << b)) | (r & P1[b])
                x >>= 1
                b += 1
            return r

        firstA = [-1] * 128
        dp = [0] * (K + 1)
        dp[0] = 1
        for i in range(n):
            x = nums[i]
            for j in range(min(i + 1, K), 0, -1):
                dp[j] |= image(dp[j - 1], x)
            if i + 1 >= K:
                m = dp[K]
                v = 0
                while m:
                    lb = m & -m
                    val = lb.bit_length() - 1
                    if firstA[val] < 0:
                        firstA[val] = i
                    m ^= lb

        lastB = [-1] * 128
        dp = [0] * (K + 1)
        dp[0] = 1
        for r in range(n):
            orig = n - 1 - r
            x = nums[orig]
            for j in range(min(r + 1, K), 0, -1):
                dp[j] |= image(dp[j - 1], x)
            if r + 1 >= K:
                m = dp[K]
                v = 0
                while m:
                    lb = m & -m
                    val = lb.bit_length() - 1
                    if lastB[val] < 0:
                        lastB[val] = orig
                    m ^= lb

        ans = 0
        for a in range(128):
            if firstA[a] < 0:
                continue
            for b in range(128):
                if lastB[b] < 0:
                    continue
                if firstA[a] < lastB[b] and (a ^ b) > ans:
                    ans = a ^ b
        return ans

Looking at this problem: values < 2^7 (128 possible OR values), n ≤ 400. Plan: for each split point, first k picks come from a prefix, last k from the corresponding suffix. Compute, for every OR value v, the earliest prefix end `firstA[v]` (v achievable with exactly k elements in nums[0..m]) and the latest suffix start `lastB[v]`. A pair (a,b) is feasible iff `firstA[a] < lastB[b]` (sets are monotone in prefix/suffix length). Then answer = max feasible `a^b` over ≤128×128 pairs.

DP over (count of elements, OR value) as a 128-bit bitmask per count. Key trick: mapping set {v} → {v|x} decomposes into per-bit image maps: for bit b, `((M & P0[b]) << 2^b) | (M & P1[b])` where P0[b] marks values lacking bit b. Descending-j knapsack keeps "exactly k" semantics.

I verified both examples by hand: [2,6,7],k=1 → 5; [4,2,5,6,7],k=2 → 2 (firstA: {6:1,5:2,7:3}, lastB: {7:3,5:0,6:0}; pair a=5,b=7 gives 2). Also checked the transform must be a pure image (not union with original), otherwise "exactly k" breaks (traced a counterexample [1,2,3,4],k=2 mentally).

Complexity: ~n·k image ops on small ints, plus 128² pairing — fast for Python.

```python
from typing import List

class Solution:
    def maxValue(self, nums: List[int], k: int) -> int:
        n = len(nums)
        K = k
        FULL = (1 << 128) - 1
        P0 = []
        for b in range(7):
            blk = (1 << (1 << b)) - 1
            step = 1 << (b + 1)
            p = 0
            for st in range(0, 128, step):
                p |= blk << st
            P0.append(p)
        P1 = [FULL ^ p for p in P0]
        SH = [1 << b for b in range(7)]

        def image(M, x):
            # set { v | x : v in M }  (pure image)
            r = M
            b = 0
            while x:
                if x & 1:
                    r = ((r & P0[b]) << SH[b]) | (r & P1[b])
                x >>= 1
                b += 1
            return r

        def solve(order):
            # order: list of (original_index, value) in processing order
            first = [-1] * 128
            dp = [0] * (K + 1)
            dp[0] = 1
            for t, (idx, x) in enumerate(order):
                top = min(t + 1, K)
                for j in range(top, 0, -1):
                    dp[j] |= image(dp[j - 1], x)
                if t + 1 >= K:
                    m = dp[K]
                    while m:
                        lb = m & -m
                        val = lb.bit_length() - 1
                        if first[val] < 0:
                            first[val] = idx
                        m ^= lb
            return first

        fwd = solve(list(enumerate(nums)))                                   # firstA
        bwd = solve([(n - 1 - t, x) for t, x in enumerate(reversed(nums))])  # lastB

        ans = 0
        blist = [(b, s) for b, s in enumerate(bwd) if s >= 0]
        for a, fa in enumerate(fwd):
            if fa < 0:
                continue
            for b, s in blist:
                if fa < s and (a ^ b) > ans:
                    ans = a ^ b
        return ans
```
