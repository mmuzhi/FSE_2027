class Solution:
    def resultsArray(self, nums: List[int], k: int) -> List[int]:
        n = len(nums)
        cnt = [1] * n
        for i in range(1, n):
            if nums[i] == nums[i-1] + 1:
                cnt[i] = cnt[i-1] + 1
        res = []
        for i in range(n - k + 1):
            if cnt[i + k - 1] >= k:
                res.append(nums[i + k - 1])
            else:
                res.append(-1)
        return res

```python
class Solution:
    def resultsArray(self, nums: List[int], k: int) -> List[int]:
        n = len(nums)
        # cnt[i] = length of the strictly consecutive ascending run ending at i
        cnt = [1] * n
        for i in range(1, n):
            if nums[i] == nums[i - 1] + 1:
                cnt[i] = cnt[i - 1] + 1
        
        res = []
        for i in range(n - k + 1):
            if cnt[i + k - 1] >= k:
                # Window is consecutive & ascending, so last element is the max
                res.append(nums[i + k - 1])
            else:
                res.append(-1)
        return res
```
