```python
class Solution:
    def minMaxWeight(self, n: int, edges: List[List[int]], threshold: int) -> int:
        # Reverse edges: rev adjacency from 0
        # weights sorted distinct
        # binary search minimal W such that BFS from 0 over edges with weight <= W reaches all n nodes
        from collections import deque
        
        rev = [[] for _ in range(n)]  # (neighbor, weight)
        weights = set()
        for a, b, w in edges:
            rev[b].append((a, w))  # reversed: b -> a
            weights.add(w)
        ws = sorted(weights)
        
        def ok(W):
            seen = [False] * n
            seen[0] = True
            q = deque([0])
            cnt = 1
            while q:
                u = q.popleft()
                for v, w in rev[u]:
                    if w <= W and not seen[v]:
                        seen[v] = True
                        cnt += 1
                        q.append(v)
            return cnt == n
        
        lo, hi = 0, len(ws) - 1
        ans = -1
        while lo <= hi:
            mid = (lo + hi) // 2
            if ok(ws[mid]):
                ans = ws[mid]
                hi = mid - 1
            else:
                lo = mid + 1
        return ans
```

```python
import heapq
def minMaxWeight(...):
    rev = [[] for _ in range(n)]
    for a, b, w in edges:
        rev[b].append((a, w))
    dist = [inf] * n
    dist[0] = 0
    heap = [(0, 0)]
    while heap:
        d, u = heapq.heappop(heap)
        if d > dist[u]: continue
        for v, w in rev[u]:
            nd = max(d, w)
            if nd < dist[v]:
                dist[v] = nd
                heapq.heappush(heap, (nd, v))
    ans = max(dist)
    return ans if ans < inf else -1
```

```python
class Solution:
    def minMaxWeight(self, n: int, edges: List[List[int]], threshold: int) -> int:
        INF = float('inf')
        # reversed adjacency
        radj = [[] for _ in range(n)]
        for a, b, w in edges:
```

```python
from heapq import heappush, heappop

class Solution:
    def minMaxWeight(self, n: int, edges: List[List[int]], threshold: int) -> int:
        # Since threshold >= 1, a spanning arborescence toward node 0
        # (one outgoing edge per node) always satisfies the degree limit.
        # So the answer is the minimum W such that every node can reach 0
        # using only edges of weight <= W: bottleneck shortest paths from 0
        # in the reversed graph.
        radj = [[] for _ in range(n)]
        for a, b, w in edges:
            radj[b].append((a, w))
        INF = float('inf')
        dist = [INF] * n
        dist[0] = 0
        heap = [(0, 0)]
        while heap:
            d, u = heappop(heap)
            if d > dist[u]:
                continue
            for v, w in radj[u]:
                nd = d if d > w else w
                if nd < dist[v]:
                    dist[v] = nd
                    heappush(heap, (nd, v))
        ans = max(dist)
        return -1 if ans == INF else ans
```

```python
from typing import List
from heapq import heappush, heappop

class Solution:
    def minMaxWeight(self, n: int, edges: List[List[int]], threshold: int) -> int:
        # Key invariant: since threshold >= 1 (given 1 <= threshold <= n-1), the
        # out-degree constraint is always satisfiable: whenever every node can
        # reach node 0 using edges of weight <= W, pick for each node v != 0 the
        # first edge of a hop-shortest path to 0. That yields a spanning
        # arborescence toward 0 with out-degree exactly 1 per node (0 for node 0),
        # which respects any threshold >= 1.
        #
        # Therefore the answer is the minimal W such that all nodes reach node 0
        # using edges of weight <= W, i.e. the bottleneck (minimax) distance from
        # node 0 to every node in the REVERSED graph; -1 if some node is unreachable.
        radj = [[] for _ in range(n)]
        for a, b, w in edges:
            radj[b].append((a, w))  # reverse edge b -> a

        INF = float('inf')
        dist = [INF] * n
        dist[0] = 0
        heap = [(0, 0)]
        while heap:
            d, u = heappop(heap)
            if d > dist[u]:
                continue
            for v, w in radj[u]:
                nd = d if d > w else w  # bottleneck relaxation: max(d, w)
                if nd < dist[v]:
                    dist[v] = nd
                    heappush(heap, (nd, v))

        ans = max(dist)
        return -1 if ans == INF else ans
```
