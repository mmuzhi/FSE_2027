```python
def count_up_to(N):
    if N <= 0:
        return 0
    digits = [int(c) for c in str(N)]
    n = len(digits)
    from functools import lru_cache
    @lru_cache(maxsize=None)
    def dp(pos, tight, started, sm, prod):
        # prod: product of digits so far (1 if none); 0 if any zero digit among started
        if pos == n:
            if not started:
                return 0  # number 0
            return 1 if prod % sm == 0 else 0
        limit = digits[pos] if tight else 9
        total = 0
        for d in range(limit + 1):
            nt = tight and (d == limit)
            if not started and d == 0:
                total += dp(pos + 1, nt, False, 0, 1)
            else:
                total += dp(pos + 1, nt, True, sm + d, prod * d)
        return total
    return dp(0, True, False, 0, 1)
```

```python
class Solution:
    def beautifulNumbers(self, l: int, r: int) -> int:
        def count_up_to(N):
            if N <= 0:
                return 0
            digs = [int(c) for c in str(N)]
            # free[i]: dict (started, sm, pr) -> cnt ; after processing i positions, numbers strictly less than N on prefix
            free = {}
            t_started, t_sm, t_pr = False, 0, 1
            for di in digs:
                nf = {}
                g = nf.get
                # diverge from tight prefix with smaller digit
                for d in range(di):
                    if t_started or d != 0:
                        k = (True, t_sm + d, t_pr * d)
                    else:
                        k = (False, 0, 1)
                    nf[k] = g(k, 0) + 1
                # expand free states with any digit
                for (st, sm, pr), c in free.items():
                    if st:
                        for d in range(10):
                            k = (True, sm + d, pr * d)
                            nf[k] = g(k, 0) + c
                    else:
                        for d in range(10):
                            k = (True, d, d) if d else (False, 0, 1)

                if t_started or di != 0:

            total = 0
            for (st, sm, pr), c in free.items():
                if st and pr % sm == 0:
                    total += c
            if t_started and t_pr % t_sm == 0:
                total += 1
            return total
        return count_up_to(r) - count_up_to(l - 1)
```

```python
class Solution:
    def beautifulNumbers(self, l: int, r: int) -> int:
        def count_up_to(N):
            if N <= 0:
                return 0
            digs = [ord(c) - 48 for c in str(N)]
            # free: dict (started, digit_sum, digit_product) -> count
            # counts prefixes already strictly smaller than N's prefix
            free = {}
            t_started, t_sm, t_pr = False, 0, 1  # tight (matching) prefix state
            for di in digs:
                nf = {}
                # diverge from the tight prefix by placing a smaller digit here
                for d in range(di):
                    if t_started or d:
                        k = (True, t_sm + d, t_pr * d)
                    else:
                        k = (False, 0, 1)
                    nf[k] = nf.get(k, 0) + 1
                # expand already-free states with any digit 0..9
                for (st, sm, pr), c in free.items():
                    if st:
                        for d in range(10):
                            k = (True, sm + d, pr * d)
                            nf[k] = nf.get(k, 0) + c
                    else:
                        for d in range(10):
                            k = (True, d, d) if d else (False, 0, 1)
                            nf[k] = nf.get(k, 0) + c
                free = nf
                # advance the tight prefix with N's own digit
                if t_started or di:
                    t_started, t_sm, t_pr = True, t_sm + di, t_pr * di
            total = 0
            for (st, sm, pr), c in free.items():
                if st and pr % sm == 0:
                    total += c
            # the fully tight path is N itself (N >= 1, so started)
            if t_started and t_pr % t_sm == 0:
                total += 1
            return total

        return count_up_to(r) - count_up_to(l - 1)
```
