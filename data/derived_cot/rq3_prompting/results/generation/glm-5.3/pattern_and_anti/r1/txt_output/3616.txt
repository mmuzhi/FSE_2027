```python
class Solution:
    def countValidSelections(self, nums: List[int]) -> int:
        total = sum(nums)
        left = 0
        ans = 0
        for i, v in enumerate(nums):
            right = total - left - v
            if v == 0:
                d = abs(left - right)
                if d == 0:
                    ans += 2
                elif d == 1:
                    ans += 1
            left += v
        return ans
```
