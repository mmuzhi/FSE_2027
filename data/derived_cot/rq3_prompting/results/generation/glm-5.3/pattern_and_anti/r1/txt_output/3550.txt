```python
class Solution:
    def maximumValueSum(self, board: List[List[int]]) -> int:
        # Key insight: only the top-3 cells of each row can ever be used,
        # since at most 2 other columns are blocked by the other two rooks.
        candidates = []
        for i, row in enumerate(board):
            top_cols = sorted(range(len(row)), key=lambda j: -row[j])[:3]
            for j in top_cols:
                candidates.append((row[j], i, j))
        # Sort descending by value to enable pruning
        candidates.sort(key=lambda x: -x[0])
        n = len(candidates)
        best = float('-inf')
        for a in range(n):
            va, ra, ca = candidates[a]
            if 3 * va <= best:
                break
            for b in range(a + 1, n):
                vb, rb, cb = candidates[b]
                if rb == ra or cb == ca:
                    continue
                if va + 2 * vb <= best:
                    break
                for c in range(b + 1, n):
                    vc, rc, cc = candidates[c]
                    if va + vb + vc <= best:
                        break
                    if rc == ra or rc == rb or cc == ca or cc == cb:
                        continue
                    s = va + vb + vc
                    if s > best:
                        best = s
        return best
```
