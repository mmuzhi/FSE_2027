class Solution:
    def shiftDistance(self, s: str, t: str, nextCost: List[int], previousCost: List[int]) -> int:
        nextPrefix = [0] * 27
        prevPrefix = [0] * 27
        for j in range(26):
            nextPrefix[j+1] = nextPrefix[j] + nextCost[j]
            prevPrefix[j+1] = prevPrefix[j] + previousCost[j]
        totalNext = nextPrefix[26]
        totalPrev = prevPrefix[26]
        
        ans = 0
        for cs, ct in zip(s, t):
            a = ord(cs) - 97
            b = ord(ct) - 97
            # forward cost: sum nextCost[a..b-1] with wraparound
            if b >= a:
                fwd = nextPrefix[b] - nextPrefix[a]
            else:

                back = prevPrefix[a+1] - prevPrefix[b+1]
            else:
                back = totalPrev - prevPrefix[b+1] + prevPrefix[a+1]
            ans += min(fwd, back)
        return ans

class Solution:
    def shiftDistance(self, s: str, t: str, nextCost: List[int], previousCost: List[int]) -> int:
        # Prefix sums: nextPrefix[j] = nextCost[0..j-1], prevPrefix[j] = previousCost[0..j-1]
        nextPrefix = [0] * 27
        prevPrefix = [0] * 27
        for j in range(26):
            nextPrefix[j + 1] = nextPrefix[j] + nextCost[j]
            prevPrefix[j + 1] = prevPrefix[j] + previousCost[j]
        totalNext = nextPrefix[26]
        totalPrev = prevPrefix[26]

        ans = 0
        for cs, ct in zip(s, t):
            a = ord(cs) - 97
            b = ord(ct) - 97

            # Forward cost: sum of nextCost over a, a+1, ..., b-1 (mod 26)
            if b >= a:
                fwd = nextPrefix[b] - nextPrefix[a]
            else:
                fwd = totalNext - nextPrefix[a] + nextPrefix[b]

            # Backward cost: sum of previousCost over a, a-1, ..., b+1 (mod 26)
            if b <= a:
                back = prevPrefix[a + 1] - prevPrefix[b + 1]
            else:
                back = totalPrev - prevPrefix[b + 1] + prevPrefix[a + 1]

            ans += min(fwd, back)
        return ans
