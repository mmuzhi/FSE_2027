```python
import sys
from collections import deque

def main():
    input_data = sys.stdin.read().split()
    idx = 0
    N = int(input_data[idx]); idx+=1
    M = int(input_data[idx]); idx+=1
    K = int(input_data[idx]); idx+=1
    edges = []
    for _ in range(M):
        u = int(input_data[idx]); idx+=1
        v = int(input_data[idx]); idx+=1
        edges.append((u,v))
    
    # BFS Dmax
    adj = [[] for _ in range(N+1)]
    for u,v in edges:
        adj[u].append(v)
    dist1 = [-1]*(N+1)
    dist1[1] = 0
    q = deque([1])
    while q:
        x = q.popleft()
        for y in adj[x]:
            if dist1[y] == -1:
                dist1[y] = dist1[x]+1
                q.append(y)
    Dmax = dist1[N]
    
    # min cost flow SSP: nodes 1..N
    # graph as adjacency with arcs: to, cap, cost, index for reverse
    n = N
    graph = [[] for _ in range(n+1)]  # each entry: [to, cap, cost, rev_index]
    def add_arc(u, v, cap, cost):
        graph[u].append([v, cap, cost, len(graph[v])])
        graph[v].append([u, 0, -cost, len(graph[u])-1])
    INF = float('inf')
    for u, v in edges:
        add_arc(u, v, 1, 0)
        add_arc(u, v, 10**9, 1)
    
    def bellman_ford_spfa():
        # find shortest path cost from 1 to N in residual
        distv = [float('inf')]*(n+1)
        distv[1] = 0
        inq = [False]*(n+1)
        dq = deque([1])
        inq[1] = True
        # SPFA
        while dq:
            x = dq.popleft()
            inq[x] = False
            dx = distv[x]
            for e in graph[x]:
                v, cap, cost, _ = e
                if cap > 0 and dx + cost < distv[v]:
                    distv[v] = dx + cost
                    if not inq[v]:
                        dq.append(v)
                        inq[v] = True
        return distv[N]
    
    def augment():
        # Bellman-Ford with parent tracking, push bottleneck (at least... we push bottleneck along shortest path)
        distv = [float('inf')]*(n+1)
        distv[1] = 0
        par = [None]*(n+1)  # (node, arc index)
        for _ in range(n):
            updated = False
            for x in range(1, n+1):
                if distv[x] < float('inf'):
                    for i, e in enumerate(graph[x]):
                        v, cap, cost, _ = e
                        if cap > 0 and distv[x] + cost < distv[v]:
                            distv[v] = distv[x] + cost
                            par[v] = (x, i)
                            updated = True
            if not updated:
                break
        if distv[N] == float('inf'):
            return None
        # find bottleneck
        f = float('inf')
        v = N
        while v != 1:
            x, i = par[v]
            f = min(f, graph[x][i][1])
            v = x
        v = N
        while v != 1:
            x, i = par[v]
            graph[x][i][1] -= f
            rev = graph[x][i][3]
            graph[x][rev][1] += f  # wait, rev index is into graph[v]
            v = x
        return distv[N], f
    
    points = [(0, 0)]  # (gamma, cost)
    total_flow = 0
    total_cost = 0
    # augment until marginal >= Dmax or none
    for _ in range(4000):
        res = augment()
        if res is None:
            break
        c, f = res
        if c >= Dmax:
            break
        total_flow += f
        total_cost += c*f
        points.append((total_flow, total_cost))
    
    ans = 0
    for d in range(Dmax, -1, -1):
        # f(d) = max over points of d*gamma - cost
        best = 0
        for g, cst in points:
            val = d*g - cst
            if val > best:
                best = val
        if best <= K:
            ans = d
            break
    print(ans)

main()
```

```python
from collections import deque

def main():

    N = int(data[idx]); idx += 1

    K = int(data[idx]); idx += 1

        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        edges.append((u, v))

    adj = [[] for _ in range(N + 1)]
    for u, v in edges:
        adj[u].append(v)
    dist1 = [-1] * (N + 1)
    dist1[1] = 0
    dq = deque([1])
    while dq:
        x = dq.popleft()
        for y in adj[x]:
            if dist1[y] == -1:
                dist1[y] = dist1[x] + 1
                dq.append(y)
    Dmax = dist1[N]

    # MCMF (SSP with Bellman-Ford)
    graph = [[] for _ in range(N + 1)]  # entries: [to, cap, cost, rev]

    def add_arc(u, v, cap, cost):
        graph[u].append([v, cap, cost, len(graph[v])])
        graph[v].append([u, 0, -cost, len(graph[u]) - 1])

    for u, v in edges:
        add_arc(u, v, 1, 0)
        add_arc(u, v, 1 << 30, 1)

    points = [(0, 0)]

    tot_cost = 0

        INF = float('inf')
        dist = [INF] * (N + 1)
        dist[1] = 0
        pre = [None] * (N + 1)  # (node, index in graph[node])
        for _it in range(N):
            changed = False
            for x in range(1, N + 1):
                dx = dist[x]
                if dx == INF:
                    continue
                for i, e in enumerate(graph[x]):
                    if e[1] > 0 and dx + e[2] < dist[e[0]]:
                        dist[e[0]] = dx + e[2]
                        pre[e[0]] = (x, i)
                        changed = True
            if not changed:
                break
        if dist[N] == INF:
            break
        c = dist[N]
        if c >= Dmax:
            break
        # bottleneck
        f = INF
        v = N
        while v != 1:
            x, i = pre[v]
            if graph[x][i][1] < f:
                f = graph[x][i][1]
            v = x
        v = N
        while v != 1:
            x, i = pre[v]
```

Looking at this problem:

- With 0/1 weights, any shortest path can be made simple, so the answer d ≤ min(K, N−1), and also d ≤ Dmax = unweighted dist(1,N).
- Dist ≥ d ⟺ there exists a potential D: V→{0..d}, D(1)=0, D(N)=d with D(v) ≤ D(u)+1 for every edge (u,v), where the 1-edges must cover all edges with D(v)=D(u)+1 ("ascending" edges). So feasibility(K,d) ⟺ min over such D of #ascending edges ≤ K (extra 1-edges can be added freely since K ≤ M).
- That minimum equals max over flows x of d·val(x) − Σ max(0,x_e−1) (LP duality, totally unimodular system). This is computed by min-cost flow where each edge gets a cap-1 cost-0 arc and an infinite-cap cost-1 arc: f(d) = max_γ (d·γ − g(γ)) with g(γ) = min cost of γ units. Since marginal (augmenting path) costs are non-decreasing and never exceed Dmax, augment while marginal < Dmax (flow stays ≤ M, so ≤ ~100 augmentations), then answer = max d ≤ Dmax with f(d) ≤ K.
