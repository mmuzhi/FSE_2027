class Solution:
    def minDifference(self, nums: List[int]) -> int:
        n = len(nums)
        # positions of fixed
        has_missing = any(v == -1 for v in nums)

    def minDifference(self, nums: List[int]) -> int:
        n = len(nums)

                prev_val = v

        def feasible(d):
