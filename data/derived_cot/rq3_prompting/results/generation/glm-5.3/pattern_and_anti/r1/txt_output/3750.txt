class Solution:
    def solveQueries(self, nums: List[int], queries: List[int]) -> List[int]:
        n = len(nums)
        pos = defaultdict(list)
        for i, v in enumerate(nums):
            pos[v].append(i)
        res = {}
        for v, g in pos.items():
            m = len(g)
            if m < 2:
                continue
            for p in range(m):
                i = g[p]
                a = g[(p - 1) % m]
                b = g[(p + 1) % m]
                d1 = abs(i - a); d1 = min(d1, n - d1)
                d2 = abs(i - b); d2 = min(d2, n - d2)
                res[i] = min(d1, d2)
        return [res.get(q, -1) for q in queries]

                d1 = min(d1, n - d1)
                d2 = abs(i - b)
                d2 = min(d2, n - d2)
                res[i] = min(d1, d2)
