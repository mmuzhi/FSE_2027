def countOfSubstrings(word, k):
    n = len(word)
    vowels = set('aeiou')
    total = 0
    for i in range(n):
        vc = set()
        cons = 0
        for j in range(i, n):
            if word[j] in vowels:
                vc.add(word[j])
            else:
                cons += 1
                if cons > k:
                    break
            if cons == k and len(vc) == 5:
                total += 1
    return total

```python
class Solution:
    def countOfSubstrings(self, word: str, k: int) -> int:
        n = len(word)
        vowels = set('aeiou')
        total = 0
        for i in range(n):
            seen = set()
            cons = 0
            for j in range(i, n):
                c = word[j]
                if c in vowels:
                    seen.add(c)
                else:
                    cons += 1
                    if cons > k:
                        break
                if cons == k and len(seen) == 5:
                    total += 1
        return total
```
