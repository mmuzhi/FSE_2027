import sys
from collections import deque

def main():
    data = sys.stdin.read().split()
    N = int(data[0]); M = int(data[1]); S = data[2]
    MOD = 998244353
    sc = [ord(ch) - 97 for ch in S]

    def step(dp, c):
        ndp = [0]*(N+1)
        prev = 0
        for j in range(1, N+1):
            v = dp[j-1] + (1 if sc[j-1] == c else 0)
            if dp[j] > v: v = dp[j]
            if prev > v: v = prev
            prev = v
            ndp[j] = v
        return tuple(ndp)

    init = tuple([0]*(N+1))
    states = {init: 0}
    state_list = [init]
    raw = []
    q = deque([init])
