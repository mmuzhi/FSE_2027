class Solution:
    def findMaxSum(self, nums1: List[int], nums2: List[int], k: int) -> List[int]:
        n = len(nums1)
        order = sorted(range(n), key=lambda i: nums1[i])
        ans = [0] * n

        cur_sum = 0
        i = 0
        while i < n:
            j = i
            v = nums1[order[i]]
            while j < n and nums1[order[j]] == v:
                ans[order[j]] = cur_sum
                j += 1

                cur_sum += x
                if len(heap) > k:
                    cur_sum -= heapq.heappop(heap)
                i += 1
        return ans

=== ANSWER ===
**Commit:** Sort indices by `nums1`; sweep groups of equal values. Maintain a min-heap of size ≤ k holding the k largest `nums2` values among strictly smaller `nums1` elements, plus a running sum. Each element in a group gets the current sum before the group's values are pushed.

**Invariant:** When answering a group with value `v`, the heap contains exactly the top-k `nums2` values of all indices with `nums1 < v`. Verified against both examples (Example 1 yields `[80,30,0,80,50]`; Example 2 yields all zeros since answers are assigned before pushes within one equal group).

```python
import heapq
from typing import List

class Solution:
    def findMaxSum(self, nums1: List[int], nums2: List[int], k: int) -> List[int]:
        n = len(nums1)
        order = sorted(range(n), key=lambda i: nums1[i])
        ans = [0] * n
        heap = []
        cur_sum = 0
        i = 0
        while i < n:
            v = nums1[order[i]]
            j = i
            # All indices in this equal-value group see the same heap state
            while j < n and nums1[order[j]] == v:
                ans[order[j]] = cur_sum
                j += 1
            # Now add this group's nums2 values into the top-k pool
            while i < j:
                x = nums2[order[i]]
                heapq.heappush(heap, x)
                cur_sum += x
                if len(heap) > k:
                    cur_sum -= heapq.heappop(heap)
                i += 1
        return ans
```
