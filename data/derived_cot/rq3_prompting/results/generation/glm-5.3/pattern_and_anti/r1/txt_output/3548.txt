from collections import Counter

class Solution:
    def countGoodIntegers(self, n: int, k: int) -> int:
        half = (n + 1) // 2
        seen = set()
        total = 0
        for h in range(10 ** (half - 1), 10 ** half):

            if int(p) % k:

            if key in seen:
                continue
            seen.add(key)
            cnt = Counter(p)
            denom = 1
            for v in cnt.values():
                denom *= factorial(v)
            arrangements = factorial(n) // denom
            c0 = cnt.get('0', 0)
            if c0:
                arrangements -= factorial(n - 1) * c0 // denom
            total += arrangements
        return total
