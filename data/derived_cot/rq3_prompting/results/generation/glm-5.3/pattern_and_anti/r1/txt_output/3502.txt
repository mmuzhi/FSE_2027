class Solution:
    def numberOfSubstrings(self, s: str, k: int) -> int:
        n = len(s)
        counts = [0] * 26
        l = 0
        bounded = 0
        for r, ch in enumerate(s):
            ci = ord(ch) - 97
            counts[ci] += 1
            while counts[ci] >= k:
                counts[ord(s[l]) - 97] -= 1
                l += 1
            bounded += r - l + 1
        return n * (n + 1) // 2 - bounded
```

class Solution:
    def numberOfSubstrings(self, s: str, k: int) -> int:
        n = len(s)
        counts = [0] * 26
        l = 0
        bounded = 0  # substrings where every char appears at most k-1 times
        for r, ch in enumerate(s):
            ci = ord(ch) - 97
            counts[ci] += 1
            # only the newly added char can reach count >= k
            while counts[ci] >= k:
                counts[ord(s[l]) - 97] -= 1
                l += 1
            bounded += r - l + 1
        return n * (n + 1) // 2 - bounded
```
