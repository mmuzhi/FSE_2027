- Else, push i onto the stack for character c (so it can be matched by a future mirror character).

```python
class Solution:
    def calculateScore(self, s: str) -> int:
        stacks = [[] for _ in range(26)]
        score = 0
        for i, ch in enumerate(s):
            c = ord(ch) - ord('a')
            m = 25 - c
            if stacks[m]:
                j = stacks[m].pop()
                score += i - j
            else:
                stacks[c].append(i)
        return score
```

```python
class Solution:
    def calculateScore(self, s: str) -> int:
        # stacks[k] holds unmarked indices with character k, in increasing order.
        # At index i with char c (mirror m = 25 - c), the closest unmarked j
        # with s[j] == mirror char is the top of stacks[m].
        stacks = [[] for _ in range(26)]
        score = 0
        for i, ch in enumerate(s):
            c = ord(ch) - ord('a')
            m = 25 - c
            if stacks[m]:
                j = stacks[m].pop()  # mark j and i implicitly
                score += i - j
            else:
                stacks[c].append(i)  # i stays unmarked, available for future
        return score
```
