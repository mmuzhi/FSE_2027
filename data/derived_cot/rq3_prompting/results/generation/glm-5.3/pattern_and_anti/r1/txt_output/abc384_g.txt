```python
import sys
import numpy as np

def main():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx+=1
    A = np.array(data[idx:idx+n], dtype=np.int64); idx+=n
    B = np.array(data[idx:idx+n], dtype=np.int64); idx+=n
    k = int(data[idx]); idx+=1
    qs = []
    for t in range(k):
        x = int(data[idx]); y = int(data[idx+1]); idx+=2
        qs.append((x,y,t))
    SA = np.zeros(n+1, dtype=np.int64); SA[1:] = np.cumsum(A)
    SB = np.zeros(n+1, dtype=np.int64); SB[1:] = np.cumsum(B)
    
    C = 100
    stepC = max(1, n // C)
    ncols = n // stepC + 1
    # group queries by column
    from collections import defaultdict
    colq = defaultdict(list)  # c -> list of (x, y, t)
    for (x,y,t) in qs:
        c = y // stepC
        colq[c].append((x,y,t))
    # build F_c per needed column
    Fcol = {}
    for c, lst in colq.items():
        yc = c * stepC
        maxx = max(item[0] for item in lst)
        if yc > 0:
            sB = np.sort(B[:yc])
            cs = np.zeros(yc+1, dtype=np.int64); np.cumsum(sB, out=cs[1:])
            pos = np.searchsorted(sB, A[:maxx], side='right')
            cnt = pos.astype(np.int64)
            gvals = A[:maxx]*(2*cnt - yc) + SB[yc] - 2*cs[pos]
            F = np.zeros(maxx+1, dtype=np.int64); np.cumsum(gvals, out=F[1:])
        else:
            F = np.zeros(maxx+1, dtype=np.int64)
        Fcol[c] = F
    # process queries sorted by x, maintaining row checkpoint structure
    order = sorted(qs, key=lambda q: q[0])
    R = 100
    stepR = max(1, n // R)
    cur_r = -1
    sortedA = None; csA = None
    ans = [0]*k
    ss = np.searchsorted  # alias
    for (x, y, t) in order:
        r = x // stepR
        if r != cur_r:
            cur_r = r
            xr = r * stepR
            if xr > 0:
                sortedA = np.sort(A[:xr])
                csA = np.zeros(xr+1, dtype=np.int64); np.cumsum(sortedA, out=csA[1:])
            else:
                sortedA = np.zeros(0, dtype=np.int64)
                csA = np.zeros(1, dtype=np.int64)
        xr = cur_r * stepR
        c = y // stepC
        yc = c * stepC
        base = int(Fcol[c][x])
        # Δ2: j in (yc, y], i <= xr
        qn = y - yc
        d2 = 0
        if qn > 0:
            bB = B[yc:y]
            pos = ss(sortedA, bB, side='right')
            tot = (bB * (2*pos - xr)).sum() + qn * int(SA[xr]) - 2 * int(csA[pos].sum())
            d2 = int(tot)
        # Δ3: i in (xr, x], j in (yc, y]
        d3 = 0
        p = x - xr
        if p > 0 and qn > 0:
            bA = A[xr:x]
            bB = B[yc:y]
            sA = np.sort(bA)
            cs3 = np.zeros(p+1, dtype=np.int64); np.cumsum(sA, out=cs3[1:])
            pos = ss(sA, bB, side='right')
            tot = (bB * (2*pos - p)).sum() + qn * int(bA.sum()) - 2 * int(cs3[pos].sum())
            d3 = int(tot)
        ans[t] = base + d2 + d3
    sys.stdout.write('\n'.join(map(str, ans)) + '\n')

main()
```
