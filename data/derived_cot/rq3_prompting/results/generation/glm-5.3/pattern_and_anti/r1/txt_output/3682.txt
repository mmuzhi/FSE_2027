```python
class Solution:
    def countGoodArrays(self, n: int, m: int, k: int) -> int:
        MOD = 10**9 + 7
        # C(n-1, k) * m * (m-1)^(n-k-1)
        # precompute factorials
        N = n
        fact = [1] * (N + 1)
        for i in range(1, N + 1):
            fact[i] = fact[i-1] * i % MOD
        inv_fact = [1] * (N + 1)
        inv_fact[N] = pow(fact[N], MOD - 2, MOD)
        for i in range(N, 0, -1):
            inv_fact[i-1] = inv_fact[i] * i % MOD
        def comb(a, b):
            if b < 0 or b > a:
                return 0
            return fact[a] * inv_fact[b] % MOD * inv_fact[a-b] % MOD
        ans = comb(n-1, k) * (m % MOD) % MOD
        ans = ans * pow(m-1, n-k-1, MOD) % MOD
        return ans
```

```python
class Solution:
    def countGoodArrays(self, n: int, m: int, k: int) -> int:
        MOD = 10**9 + 7
        # Answer = C(n-1, k) * m * (m-1)^(n-k-1)
        # Choosing which k of the n-1 adjacent pairs are "equal" splits the
        # array into n-k runs: first run has m choices, each later run must
        # differ from the previous run -> m-1 choices each.
        fact = [1] * n
        for i in range(1, n):
            fact[i] = fact[i - 1] * i % MOD
        inv_fact = [1] * n
        inv_fact[n - 1] = pow(fact[n - 1], MOD - 2, MOD)
        for i in range(n - 1, 0, -1):
            inv_fact[i - 1] = inv_fact[i] * i % MOD

        def comb(a, b):
            if b < 0 or b > a:
                return 0
            return fact[a] * inv_fact[b] % MOD * inv_fact[a - b] % MOD

        ans = comb(n - 1, k) * (m % MOD) % MOD
        ans = ans * pow(m - 1, n - k - 1, MOD) % MOD
        return ans
```
