```python
import sys

def main():
    data = sys.stdin.read().split()
    n = int(data[0])
    p = list(map(int, data[1:1 + n]))
    out = []
    for i in range(n):
        rank = 1 + sum(1 for j in range(n) if p[j] > p[i])
        out.append(str(rank))
    print("\n".join(out))

main()
```
