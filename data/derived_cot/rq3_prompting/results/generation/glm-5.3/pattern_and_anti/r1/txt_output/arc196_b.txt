from operator import xor

def main():
    data = sys.stdin.buffer.read().split()
    MOD = 998244353
    T = int(data[0])
    idx = 1
    out = []
    tbl = bytes.maketrans(b'AB', b'\x01\x00')
    B = 66
    for _ in range(T):
        H = int(data[idx]); W = int(data[idx + 1]); idx += 2
        rows = data[idx:idx + H]; idx += H
        bad = False
        for r in rows:
            if r.count(65) & 1:

        if not bad:
            X = 0
            for r in rows:
                X ^= int.from_bytes(r, 'little')
            if H & 1:
                if X != int.from_bytes(b'\x42' * W, 'little'):
                    bad = True
            elif X:
                bad = True
        if bad:
            out.append(b'0'); continue
        n = H + W
        parent = list(range(n))
        par = [0] * n
        sz = [1] * n
        comp = n
        ok = True
        Qb = bytes(W)          # Q_{i-1, ·}
        Qint = 0
        for i in range(H):
            s = rows[i]
            bits = s.translate(tbl)     # '1' where A
            k = s.find(B)

                prev = 0

                    p = prev ^ (bits.count(1, pos, k + 1) & 1)
                    c = 1 ^ p ^ Qb[k]
                    # DSU on (i, H + k, c)
                    x = i; pa = 0
                    while True:
                        t = parent[x]
                        if t == x: break
                        u = parent[t]
                        if u == t:

                        par[x] ^= par[t]; parent[x] = u
                        pa ^= par[x]; x = u

                        if t == y: break

                        if u == t:

                        pb ^= par[y]; y = u

Looking at this problem:

- Each tile picks 2 of 4 edges (Type A: adjacent pair, Type B: opposite pair). "No dead ends" means edge endpoints must match across every shared boundary (torus).
- Let x_{i,j} = boundary between (i,j),(i,j+1); y_{i,j} = boundary between (i,j),(i+1,j). A cell with pattern (L,R,U,D)=(x_{i,j-1},x_{i,j},y_{i-1,j},y_{i,j}) is valid iff:
  - Type A: L⊕R=1 and U⊕D=1
  - Type B: L=R, U=D, L≠U
- So along each row, x flips at A cells (needs even #A per row, then x = p_i ⊕ rowprefix); along each column similarly y = q_j ⊕ colprefix. Each B cell gives constraint p_i ⊕ q_j = 1 ⊕ (row prefix) ⊕ (col prefix) =: c_{i,j}.
- Answer = 0 if any row/column has odd #A; otherwise count (p,q) ∈ {0,1}^{H+W} satisfying all B-cell equations = 2^{#components of bipartite graph (rows ∪ cols, edges at B cells)} if consistent (parity DSU), else 0.

Verified by hand against all three samples (2, 0, 2) and several small grids (all-B 2×2 → 2, "AABA"/"AABA" 2×4 → 16, 4×4 A-block → 2).
