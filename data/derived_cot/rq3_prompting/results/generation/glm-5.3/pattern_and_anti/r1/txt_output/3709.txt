```python
class Solution:
    def hasSpecialSubstring(self, s: str, k: int) -> bool:
        i = 0
        n = len(s)
        while i < n:
            j = i
            while j < n and s[j] == s[i]:
                j += 1
            if j - i == k:
                return True
            i = j
        return False
```

```python
class Solution:
    def hasSpecialSubstring(self, s: str, k: int) -> bool:
        # A valid substring of length k must be a maximal run of exactly length k:
        # if the run were longer, one neighbor inside the run would match.
        n = len(s)
        i = 0
        while i < n:
            j = i
            while j < n and s[j] == s[i]:
                j += 1
            if j - i == k:
                return True
            i = j
        return False
```
