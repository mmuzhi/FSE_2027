```python
class Solution:
    def numOfUnplacedFruits(self, fruits: List[int], baskets: List[int]) -> int:
        n = len(fruits)
        size = 1
        while size < n:
            size <<= 1
        # Iterative max segment tree; 0 marks an used basket (capacities >= 1)
        tree = [0] * (2 * size)
        for i, c in enumerate(baskets):
            tree[size + i] = c
        for i in range(size - 1, 0, -1):
            tree[i] = tree[2 * i] if tree[2 * i] >= tree[2 * i + 1] else tree[2 * i + 1]

        unplaced = 0
        for f in fruits:
            if tree[1] < f:
                unplaced += 1
                continue
            # Descend to leftmost leaf with value >= f
            i = 1
            while i < size:
                i <<= 1
                if tree[i] < f:
                    i += 1
            tree[i] = 0  # consume this basket
            i >>= 1
            while i:
                l, r = tree[2 * i], tree[2 * i + 1]
                tree[i] = l if l >= r else r
                i >>= 1
        return unplaced
```
