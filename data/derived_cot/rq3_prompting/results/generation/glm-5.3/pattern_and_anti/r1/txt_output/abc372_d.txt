import sys

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    h = list(map(int, data[1:1+n]))
    stack = []
    res = [0] * n
    for p in range(n - 1, -1, -1):
        res[p] = len(stack)
        v = h[p]
        while stack and stack[-1] < v:
            stack.pop()
        stack.append(v)
    sys.stdout.write(' '.join(map(str, res)))

main()
