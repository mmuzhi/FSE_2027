def solve_numpy(N, M, rest, MOD):
    arr = np.frombuffer(rest, dtype=np.uint8)
    bits = arr[(arr == 48) | (arr == 49)].astype(np.uint8) - 48  # 0/1
    # bits length N*M, element i*M + k is A_{i+1, k+1} → bit position k
    rows = bits.reshape(N, M).astype(np.uint64)
    weights = (np.uint64(1) << np.arange(M, dtype=np.uint64))
    vals = rows @ weights
    nz = vals != 0
    vals = vals[nz]
    if vals.size == 0:
        print(0); return
    low = vals & (vals ^ (vals - np.uint64(1)))
    c = np.log2(low.astype(np.float64)).astype(np.int64)
    u = vals >> c.astype(np.uint64)
    # loop over unique c
    c_arr = c
    u_arr = u
    x_arr = np.zeros(vals.size, dtype=np.int64)
    c_out = np.empty(vals.size, dtype=np.int64)
    r_out = np.empty(vals.size, dtype=np.uint64)
    for cv in np.unique(c_arr):
        sel = np.nonzero(c_arr == cv)[0]
        us = u_arr[sel].copy()
        Mp = M - int(cv)
        T = (Mp - 1).bit_length()
        mask = np.uint64((1 << Mp) - 1)
        x = np.zeros(us.size, dtype=np.int64)
        s = 1
        for t in range(T):
            bm = (us >> np.uint64(s)) & np.uint64(1)
            act = bm == np.uint64(1)
            if act.any():
                x[act] += s
                sub = us[act]
                q = s
                while q < Mp:
                    sub = sub ^ ((sub << np.uint64(q)) & mask)
                    q += q
                us[act] = sub
            s += s
        x_arr[sel] = x
        c_out[sel] = cv
        r_out[sel] = us
    # group ids via lexsort
    order = np.lexsort((r_out.astype(np.int64), c_out))
    cs = c_out[order]; rs = r_out[order].astype(np.int64)
    newkey = np.empty(cs.size, dtype=bool)
    newkey[0] = True
    newkey[1:] = (cs[1:] != cs[:-1]) | (rs[1:] != rs[:-1])
    gid_sorted = np.cumsum(newkey) - 1  # 0..G-1 in sorted order
    # map back to input order
    gid = np.empty(cs.size, dtype=np.int64)
    gid[order] = gid_sorted
