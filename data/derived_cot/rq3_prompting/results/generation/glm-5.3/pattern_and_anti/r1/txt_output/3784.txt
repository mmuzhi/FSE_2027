class Solution:
    def longestCommonPrefix(self, words: List[str], k: int) -> List[int]:
        n = len(words)
        if n <= k:
            return [0] * n
        # build trie
        children = [{}]  # list of dicts
        cnt = [0]
        depth = [0]
        paths = []  # per word, list of node ids (excluding root? include root)
        for w in words:
            cur = 0
            cnt[0] += 1
            path = [0]
            for ch in w:
                nxt = children[cur].get(ch)
                if nxt is None:
                    nxt = len(children)
                    children[cur][ch] = nxt
                    children.append({})
                    cnt.append(0)
                    depth.append(depth[cur] + 1)
                cur = nxt
                cnt[cur] += 1
                path.append(cur)
            paths.append(path)
        # B
        B = -1
        ck = []  # (depth, node) with cnt == k
        for v in range(len(children)):
            if cnt[v] >= k + 1:
                if depth[v] > B:
                    B = depth[v]
            elif cnt[v] == k:
                ck.append((depth[v], v))
        ck.sort(reverse=True)
        # per word sets — but only needed if ck nonempty; also only nodes in ck matter
        # Optimization: only need membership of ck nodes; build per-word set of path ∩ ck_ids
        ck_ids = set(v for _, v in ck)
        ans = []
        for i, w in enumerate(words):
            pset = set(paths[i]) & ck_ids  # or full set
            off = -1
            for d, v in ck:
                if v not in pset:
                    off = d
                    break
            ans.append(max(B, off))
        return ans

class Solution:
    def longestCommonPrefix(self, words: List[str], k: int) -> List[int]:
        n = len(words)
        if n <= k:
            return [0] * n
        
        children = [{}]
        cnt = [0]
        dep = [0]
        paths = []
        for w in words:
            cur = 0
            cnt[0] += 1
            path = [0]
            for ch in w:
                d = children[cur]
                nxt = d.get(ch)
                if nxt is None:
                    nxt = len(children)
                    d[ch] = nxt
                    children.append({})
                    cnt.append(0)
                    dep.append(dep[cur] + 1)
                cur = nxt
                cnt[cur] += 1
                path.append(cur)
            paths.append(path)
        
        B = 0  # root qualifies since n >= k+1
        ck = []
        for v in range(1, len(children)):
            c = cnt[v]
            if c >= k + 1:
                if dep[v] > B:
                    B = dep[v]
            elif c == k:
                ck.append((dep[v], v))
        ck.sort(reverse=True)
        
        if not ck:
            return [B] * n
        
        ck_ids = set(v for _, v in ck)
        ans = [0] * n
        for i in range(n):
            pset = set(paths[i])
            off = -1
            for d, v in ck:
                if v not in pset:
                    off = d
                    break
            ans[i] = off if off > B else B
        return ans

Looking at this problem:

- **Key invariant**: For each trie node (prefix) with `cnt` words passing through it: if `cnt >= k+1`, it contributes its depth to *every* removal (if the removed word passes, `cnt-1 >= k` still; if not, `cnt >= k`). If `cnt == k`, it contributes only when the removed word does **not** pass through it. So `answer[i] = max(B, offpath(i))` where `B = max depth of node with cnt >= k+1` and `offpath(i) = max depth of cnt==k node not on word i's path`.
- **Scan bound**: sort `cnt==k` nodes by depth desc; for word `i`, scan until first off-path node — all skipped nodes lie on path `i` (distinct depths), so ≤ `len(words[i])+1` steps. Total ≤ sum of lengths ≈ 10⁵.
- **Edge cases**: `n == k` → all zeros; verified against both examples and prefix-of-another-word cases (e.g., `["a","ab","ab"], k=2 → [2,1,1]`).

Traced Example 1: B=3 ("run" has cnt 3), cnt==2 nodes are the "jump" chain; words "run" find off-path node at depth 4 → 4; words "jump" find none → 3. Matches `[3,4,4,3,4]`.
