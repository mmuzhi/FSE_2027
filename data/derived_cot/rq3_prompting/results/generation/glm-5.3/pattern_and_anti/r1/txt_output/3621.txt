```python
class Solution:
    def minOperations(self, nums: List[int], k: int) -> int:
        if any(v < k for v in nums):
            return -1
        return len(set(v for v in nums if v > k))
```

=== ANSWER ===
```python
class Solution:
    def minOperations(self, nums: List[int], k: int) -> int:
        # Operations only decrease values, so any value below k makes it impossible.
        if any(v < k for v in nums):
            return -1
        # Each valid operation reduces the count of distinct values > k by at most 1,
        # and collapsing from the largest distinct value downward achieves exactly one per step.
        return len({v for v in nums if v > k})
```
