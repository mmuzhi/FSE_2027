```python
class Solution:
    def numberOfWays(self, n: int, x: int, y: int) -> int:
        MOD = 10**9 + 7
        m = min(n, x)
        dp = [0] * (m + 1)  # dp[j]: functions from first i performers to x stages using exactly j stages
        dp[0] = 1
        for i in range(1, n + 1):
            top = min(i, m)
            for j in range(top, 0, -1):
                dp[j] = (j * dp[j] + (x - j + 1) * dp[j - 1]) % MOD
            dp[0] = 0
        ans = 0
        yk = 1
        for k in range(1, m + 1):
            yk = yk * y % MOD
            ans = (ans + dp[k] * yk) % MOD
        return ans
```

```python
class Solution:
    def numberOfWays(self, n: int, x: int, y: int) -> int:
        MOD = 10**9 + 7
        m = min(n, x)
        # dp[j] = number of ways to assign the first i performers to x labeled
        # stages such that exactly j distinct stages are used.
        # Transition: performer i joins one of the j used stages (j ways),
        # or opens one of the (x - j + 1) unused stages.
        dp = [0] * (m + 1)
        dp[0] = 1
        for i in range(1, n + 1):
            top = min(i, m)
            for j in range(top, 0, -1):
                dp[j] = (j * dp[j] + (x - j + 1) * dp[j - 1]) % MOD
            dp[0] = 0  # with i >= 1 performers, zero used stages is impossible
        ans = 0
        yk = 1
        for k in range(1, m + 1):
            yk = yk * y % MOD
            ans = (ans + dp[k] * yk) % MOD
        return ans
```
