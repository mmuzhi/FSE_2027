class Solution:
    def minimumPairRemoval(self, nums: List[int]) -> int:
        def is_sorted(a):
            return all(a[i] <= a[i + 1] for i in range(len(a) - 1))
        
        nums = list(nums)
        ops = 0
        while not is_sorted(nums):
            best_i = 0
            best_s = nums[0] + nums[1]

                s = nums[i] + nums[i + 1]
                if s < best_s:
                    best_s = s
                    best_i = i
            nums[best_i:best_i + 2] = [best_s]
            ops += 1
        return ops

class Solution:
    def minimumPairRemoval(self, nums: List[int]) -> int:
        def is_nondecreasing(a):
            return all(a[i] <= a[i + 1] for i in range(len(a) - 1))
        
        nums = list(nums)
        ops = 0
        while not is_nondecreasing(nums):
            best_i = 0
            best_s = nums[0] + nums[1]
            for i in range(1, len(nums) - 1):
                s = nums[i] + nums[i + 1]
                if s < best_s:  # strict '<' keeps leftmost on ties
                    best_s = s
                    best_i = i
            nums[best_i:best_i + 2] = [best_s]
            ops += 1
        return ops
