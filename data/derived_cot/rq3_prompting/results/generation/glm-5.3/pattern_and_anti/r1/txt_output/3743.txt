```python
class Solution:
    def maxFreeTime(self, eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
        n = len(startTime)
        gaps = [startTime[0]] + [startTime[i] - endTime[i-1] for i in range(1, n)] + [eventTime - endTime[n-1]]
        w = k + 1
        cur = sum(gaps[:w])
        ans = cur
        for i in range(w, n+1):
            cur += gaps[i] - gaps[i-w]
            ans = max(ans, cur)
        return ans
```

=== ANSWER ===
```python
class Solution:
    def maxFreeTime(self, eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
        n = len(startTime)
        # There are n+1 gaps: before first meeting, between meetings, after last meeting.
        gaps = [startTime[0]]
        gaps += [startTime[i] - endTime[i - 1] for i in range(1, n)]
        gaps.append(eventTime - endTime[n - 1])
        
        # Rescheduling k consecutive meetings merges k+1 adjacent gaps into one block.
        w = k + 1
        cur = sum(gaps[:w])
        ans = cur
        for i in range(w, n + 1):
            cur += gaps[i] - gaps[i - w]
            if cur > ans:
                ans = cur
        return ans
```
