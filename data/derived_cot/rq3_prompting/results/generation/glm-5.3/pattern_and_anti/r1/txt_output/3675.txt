```python
class Solution:
    def maximizeSumOfWeights(self, edges: List[List[int]], k: int) -> int:
        n = len(edges) + 1
        adj = [[] for _ in range(n)]
        for u, v, w in edges:
            adj[u].append((v, w))
            adj[v].append((u, w))
        
        parent = [-1] * n
        parent_w = [0] * n
        order = []
        stack = [0]
        while stack:
            u = stack.pop()
            order.append(u)
            p = parent[u]
            for v, w in adj[u]:
                if v != p:
                    parent[v] = u
                    parent_w[v] = w
                    stack.append(v)
        
        dp0 = [0] * n  # parent edge kept -> at most k-1 children edges
        dp1 = [0] * n  # parent edge cut  -> at most k children edges
        
        for u in reversed(order):
            total = 0
            gains = []
            p = parent[u]
            for v, w in adj[u]:
                if v == p:
                    continue
                total += dp1[v]
                gains.append(dp0[v] + w - dp1[v])
            gains.sort(reverse=True)
            s = 0
            for i in range(min(k, len(gains))):
                if gains[i] > 0:
                    s += gains[i]
                else:
                    break
            dp1[u] = total + s
            s0 = 0
            for i in range(min(k - 1, len(gains))):
                if gains[i] > 0:
                    s0 += gains[i]
                else:
                    break
            dp0[u] = total + s0
        
        return dp1[0]
```

```python
class Solution:
    def maximizeSumOfWeights(self, edges: List[List[int]], k: int) -> int:
        n = len(edges) + 1
        adj = [[] for _ in range(n)]
        for u, v, w in edges:
            adj[u].append((v, w))
            adj[v].append((u, w))

        parent = [-1] * n
        parent_w = [0] * n
        order = []
        stack = [0]
        while stack:
            u = stack.pop()
            order.append(u)
            p = parent[u]
            for v, w in adj[u]:
                if v != p:
                    parent[v] = u
                    parent_w[v] = w
                    stack.append(v)

        # dp0[u]: edge to parent is KEPT -> u may keep at most k-1 child edges
        # dp1[u]: edge to parent is CUT   -> u may keep at most k child edges
        dp0 = [0] * n
        dp1 = [0] * n

        for u in reversed(order):
            total = 0
            gains = []
            p = parent[u]
            for v, w in adj[u]:
                if v == p:
                    continue
                total += dp1[v]
                gains.append(dp0[v] + w - dp1[v])
            gains.sort(reverse=True)

            s = 0
            for i in range(min(k, len(gains))):
                if gains[i] > 0:
                    s += gains[i]
                else:
                    break
            dp1[u] = total + s

            s0 = 0
            for i in range(min(k - 1, len(gains))):
                if gains[i] > 0:
                    s0 += gains[i]
                else:
                    break
            dp0[u] = total + s0

        return dp1[0]
```
