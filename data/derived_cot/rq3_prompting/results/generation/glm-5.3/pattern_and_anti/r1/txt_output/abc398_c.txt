```python
import sys
from collections import Counter

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    a = list(map(int, data[1:1+n]))
    cnt = Counter(a)
    best_val = 0
    best_label = -1
    for i, v in enumerate(a, 1):
        if cnt[v] == 1 and v > best_val:
            best_val = v
            best_label = i
    print(best_label)

main()
```
