```python
class Solution:
    def findMinimumTime(self, strength: List[int], K: int) -> int:
        n = len(strength)
        INF = float('inf')
        dp = [INF] * (1 << n)
        dp[0] = 0
        for mask in range(1 << n):
            if dp[mask] == INF:
                continue
            x = 1 + bin(mask).count('1') * K
            for i in range(n):
                if not mask & (1 << i):
                    t = (strength[i] + x - 1) // x
                    nm = mask | (1 << i)
                    if dp[mask] + t < dp[nm]:
                        dp[nm] = dp[mask] + t
        return dp[(1 << n) - 1]
```

```python
class Solution:
    def findMinimumTime(self, strength: List[int], K: int) -> int:
        n = len(strength)
        INF = float('inf')
        dp = [INF] * (1 << n)
        dp[0] = 0
        for mask in range(1 << n):

                continue
            x = 1 + bin(mask).count('1') * K
            for i in range(n):

                    nm = mask | (1 << i)
                    if dp[mask] + t < dp[nm]:

        return dp[(1 << n) - 1]
```
