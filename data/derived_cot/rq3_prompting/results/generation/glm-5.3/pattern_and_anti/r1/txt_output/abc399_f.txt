import numpy as np
from math import comb

def main():
    data = sys.stdin.buffer.read().split()
    N = int(data[0]); K = int(data[1])
    MOD = 998244353
    A = np.array(data[2:2+N], dtype=np.int64)
    
    S = np.zeros(N+1, dtype=np.int64)
    S[1:] = np.cumsum(A) % MOD  # cumsum max 2e5 * 1e9 = 2e14, fits int64

    P = [None] * (K+1)
    P[0] = np.ones(N+1, dtype=np.int64)
    for j in range(1, K+1):
        P[j] = P[j-1] * S % MOD
    
    # cum[j][i] = sum_{m=0}^{i} P[j][m] mod p
    # T_j[r] = cum[j][r-1] for r >= 1
    cum = [np.cumsum(P[j]) % MOD for j in range(K+1)]

        term = P[K-j][1:] * cum[j][:N] % MOD
        ans = (ans + c * int(term.sum() % MOD)) % MOD
    
    print(ans % MOD)

main()
