from typing import List
from bisect import bisect_left

class Solution:
    def validSequence(self, word1: str, word2: str) -> List[int]:
        n, m = len(word1), len(word2)
        # t0[i]: max number of trailing chars of word2 matchable exactly in word1[i:]
        t0 = [0] * (n + 2)
        for i in range(n - 1, -1, -1):
            k = t0[i + 1]
            if k < m and word1[i] == word2[m - 1 - k]:
                t0[i] = k + 1
            else:
                t0[i] = k
        
        occ = [[] for _ in range(26)]
        for idx, ch in enumerate(word1):
            occ[ord(ch) - 97].append(idx)
        
        res = []
        pos = 0
        j = 0
        used = False
        while j < m:
            if pos >= n:
                return []
            ch = word2[j]
            if not used:
                if word1[pos] == ch:
                    res.append(pos)
                    pos += 1
                    j += 1
                    continue
                # try change at pos
                if t0[pos + 1] >= m - j - 1:
                    res.append(pos)
                    used = True
                    pos += 1
                    j += 1
                    continue
                # next exact occurrence
                o = occ[ord(ch) - 97]
                t = bisect_left(o, pos)
                if t == len(o):
                    return []
                i_eq = o[t]
                res.append(i_eq)
                pos = i_eq + 1
                j += 1
            else:
                o = occ[ord(ch) - 97]
                t = bisect_left(o, pos)
                if t == len(o):
                    return []
                i_eq = o[t]
                res.append(i_eq)
