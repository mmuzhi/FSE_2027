```python
class Solution:
    def countSubstrings(self, s: str) -> int:
        n = len(s)
        digits = [ord(c) - 48 for c in s]
        pre = [0] * (n + 1)
        p = 0
        for i, d in enumerate(digits):

            pre[i + 1] = p

        present = set(digits)
        has3 = 3 in present
        has6 = 6 in present
        has9 = 9 in present
        has7 = 7 in present

        tot3 = [0] * 3
        tot6 = [0] * 6
        tot9 = [0] * 9
        cnt7 = [[0] * 7 for _ in range(6)]

        m7 = [pow(10, r, 7) for r in range(6)]

                if has6:
```

```python
class Solution:
    def countSubstrings(self, s: str) -> int:
        n = len(s)
        digits = [ord(c) - 48 for c in s]

        # prefix values mod 2520 (= lcm(1..9)); substring [i..j] value
        # V satisfies V = X_{j+1} - X_i * 10^(j+1-i), valid mod any d in 1..9
        pre = [0] * (n + 1)
        p = 0
        for i in range(n):
            p = (p * 10 + digits[i]) % 2520
            pre[i + 1] = p

        present = set(digits)
        has3 = 3 in present
        has6 = 6 in present
        has9 = 9 in present
        has7 = 7 in present

        tot3 = [0] * 3   # 10^k ≡ 1 (mod 3)
        tot9 = [0] * 9   # 10^k ≡ 1 (mod 9)
        tot6 = [0] * 6   # 10^k ≡ 4 (mod 6) for k >= 1
        cnt7 = [[0] * 7 for _ in range(6)]  # 10^k mod 7 has period 6

        m7 = [pow(10, r, 7) for r in range(6)]
        invm = [0] * 10
        for m in range(1, 7):
            invm[m] = pow(m, 5, 7)
        b7r = [[(t * invm[m7[r]]) % 7 for t in range(7)] for r in range(6)]

        list6 = [[] for _ in range(6)]
        for b in range(6):
            list6[(4 * b) % 6].append(b)

        ans = 0
        for j in range(n):
            # make prefix index j-2 available for len>=3 substrings ending at j
            if j >= 2:
                pi = pre[j - 2]
                if has3:
                    tot3[pi % 3] += 1
                if has6:
                    tot6[pi % 6] += 1
                if has9:
                    tot9[pi % 9] += 1
                if has7:
                    cnt7[(j - 2) % 6][pi % 7] += 1
            d = digits[j]
            if d == 0:
                continue
            if d == 1:
                ans += j + 1  # everything is divisible by 1
                continue
            ans += 1  # length-1 substring
            if j >= 1 and (10 * digits[j - 1]) % d == 0:
                ans += 1  # length-2 substring
            if j >= 2:
                t = pre[j + 1] % d
                if d == 3:
                    ans += tot3[t]
                elif d == 9:
                    ans += tot9[t]
                elif d == 6:
                    for b in list6[t]:
                        ans += tot6[b]
                elif d == 7:
                    j1 = j + 1
                    for a in range(6):
                        ans += cnt7[a][b7r[(j1 - a) % 6][t]]
                else:
                    # d in {2,4,5,8}: 10^len ≡ 0 (mod d) for len >= 3,
                    # so V ≡ pre[j+1] (mod d)
                    if t == 0:
                        ans += j - 1
        return ans
```
