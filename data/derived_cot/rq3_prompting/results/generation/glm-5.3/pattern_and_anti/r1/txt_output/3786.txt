class Solution:
    def longestPalindromicSubsequence(self, s: str, k: int) -> int:
        n = len(s)
        K = k + 1
        prev2 = [[0]*K]  # diagonal length 0
        prev1 = [[1]*K for _ in range(n)]  # length 1
        for L in range(2, n+1):
            m = n - L + 1
            cur = [None]*m
            for i in range(m):
                j = i + L - 1
                a = ord(s[i]) - 97
                b = ord(s[j]) - 97
                c = abs(a-b)
                c = min(c, 26-c)
                base = [x if x > y else y for x, y in zip(prev1[i+1], prev1[i])]

                    base[c:] = [x if x > y else y for x, y in zip(base[c:], tail)]
                cur[i] = base
            prev2 = prev1
            prev1 = cur

class Solution:
    def longestPalindromicSubsequence(self, s: str, k: int) -> int:
        n = len(s)
        K = k + 1
        if n == 1:
            return 1
        
        # Diagonal DP: D_L[i] = array over budget b (0..k) of best LPS in s[i..i+L-1] using <= b ops
        prev2 = [[0] * K for _ in range(n + 1)]  # diagonal length 0 (empty range)
        prev1 = [[1] * K for _ in range(n)]      # diagonal length 1 (single char)
        
        for L in range(2, n + 1):
            m = n - L + 1
            cur = [None] * m
            for i in range(m):
                j = i + L - 1
                a = ord(s[i]) - 97
                b = ord(s[j]) - 97
                c = abs(a - b)
                if c > 13:
                    c = 26 - c  # cyclic distance to make ends equal (triangle ineq: single-sided move is optimal)
                
                # skip either end
                base = [x if x > y else y for x, y in zip(prev1[i + 1], prev1[i])]
                
                # match ends s[i], s[j] with cost c
                if c <= k:
                    tail = [x + 2 for x in prev2[i + 1][:K - c]]
                    seg = base[c:]
                    base[c:] = [x if x > y else y for x, y in zip(seg, tail)]
                
                cur[i] = base
            prev2 = prev1
            prev1 = cur
        
        return prev1[0][k]
