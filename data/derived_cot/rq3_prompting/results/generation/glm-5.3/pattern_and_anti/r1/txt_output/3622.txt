```python
class Solution:
    def maxFrequency(self, nums: List[int], k: int, numOperations: int) -> int:
        from bisect import bisect_left, bisect_right
        nums.sort()
        n = len(nums)
        ans = 0
        candidates = set()
        for x in nums:
            candidates.add(x)
            candidates.add(x + k)
            candidates.add(x - k)
        for T in candidates:
            A = bisect_right(nums, T) - bisect_left(nums, T)
            W = bisect_right(nums, T + k) - bisect_left(nums, T - k)
            ans = max(ans, min(W, A + numOperations))
        return ans
```
