```python
class Solution:
    def maximumCoins(self, coins: List[List[int]], k: int) -> int:
        coins.sort()
        n = len(coins)
        prefix = [0] * (n + 1)
        for i, (l, r, c) in enumerate(coins):
            prefix[i + 1] = prefix[i] + (r - l + 1) * c
        
        ans = 0
        # Case 1: window starts at l_i
        j = 0
        for i in range(n):
            if j < i:
                j = i
            end = coins[i][0] + k - 1
            while j < n and coins[j][1] <= end:
                j += 1
            total = prefix[j] - prefix[i]
            if j < n and coins[j][0] <= end:
                total += (end - coins[j][0] + 1) * coins[j][2]
            ans = max(ans, total)
        
        # Case 2: window ends at r_i
        j = 0
        for i in range(n):
            start = coins[i][1] - k + 1
            while j < n and coins[j][0] < start:
                j += 1
            total = prefix[i + 1] - prefix[j]
            if j > 0 and coins[j - 1][1] >= start:
                total += (coins[j - 1][1] - start + 1) * coins[j - 1][2]
            ans = max(ans, total)
        
        return ans
```

```python
from typing import List

class Solution:
    def maximumCoins(self, coins: List[List[int]], k: int) -> int:
        coins.sort()
        n = len(coins)
        prefix = [0] * (n + 1)
        for i, (l, r, c) in enumerate(coins):
            prefix[i + 1] = prefix[i] + (r - l + 1) * c

        ans = 0

        # Case 1: window starts exactly at l_i -> [l_i, l_i + k - 1]
        j = 0
        for i in range(n):
            if j < i:
                j = i
            end = coins[i][0] + k - 1
            while j < n and coins[j][1] <= end:
                j += 1
            total = prefix[j] - prefix[i]
            if j < n and coins[j][0] <= end:
                total += (end - coins[j][0] + 1) * coins[j][2]
            if total > ans:
                ans = total

        # Case 2: window ends exactly at r_i -> [r_i - k + 1, r_i]
        j = 0
        for i in range(n):
            start = coins[i][1] - k + 1
            while j < n and coins[j][0] < start:
                j += 1
            total = prefix[i + 1] - prefix[j]
            if j > 0 and coins[j - 1][1] >= start:
                total += (coins[j - 1][1] - start + 1) * coins[j - 1][2]
            if total > ans:
                ans = total

        return ans
```
