```python
class Solution:
    def subsequencePairCount(self, nums: List[int]) -> int:
        MOD = 10**9 + 7
        V = 200
        # D[d] = number of elements divisible by d
        D = [0] * (V + 1)
        for d in range(1, V + 1):
            D[d] = sum(1 for x in nums if x % d == 0)
        # precompute powers of 2 and 3 up to n
        n = len(nums)
        pow2 = [1] * (n + 1)
        pow3 = [1] * (n + 1)
        for i in range(1, n + 1):
            pow2[i] = pow2[i-1] * 2 % MOD
            pow3[i] = pow3[i-1] * 3 % MOD
        # mobius
        mu = [1] * (V + 1)
        # sieve
        is_comp = [False] * (V + 1)
        primes = []
        mu[1] = 1
        for i in range(2, V + 1):
            if not is_comp[i]:
                primes.append(i)
                mu[i] = -1
            for p in primes:
                if i * p > V: break
                is_comp[i*p] = True
                if i % p == 0:
                    mu[i*p] = 0
                    break
                else:
                    mu[i*p] = -mu[i]
        # For each g, relevant squarefree d with D[g*d] > 0
        ans = 0
        for g in range(1, V + 1):
            if D[g] == 0:
                continue
            ds = [d for d in range(1, V // g + 1) if mu[d] != 0 and D[g*d] > 0]
            if not ds:
                continue
            # precompute M table for pairs
            # M(u,v) = 2^{Du+Dv-2Dl} * 3^{Dl} - 2^Du - 2^Dv + 1
            # cache lcm and D
            for d1 in ds:
                u = g * d1
                Du = D[u]
                pu2 = pow2[Du]
                for d2 in ds:
                    v = g * d2
                    Dv = D[v]
                    l = u * v // math.gcd(u, v)
                    Dl = D[l] if l <= V else 0
                    m = (pow2[Du + Dv - 2*Dl] * pow3[Dl] - pu2 - pow2[Dv] + 1) % MOD
                    ans = (ans + mu[d1] * mu[d2] * m) % MOD
        return ans % MOD
```

```python
        V = 200
        n = len(nums)

        # D[d] = number of elements divisible by d
        D = [0] * (V + 1)
        for d in range(1, V + 1):
            D[d] = sum(1 for x in nums if x % d == 0)

        pow2 = [1] * (n + 1)
        pow3 = [1] * (n + 1)
        for i in range(1, n + 1):
            pow2[i] = pow2[i - 1] * 2 % MOD
            pow3[i] = pow3[i - 1] * 3 % MOD

        # Mobius function via trial division
        mu = [0] * (V + 1)
        for m in range(1, V + 1):
            res, x, p = 1, m, 2
            while p * p <= x:
                if x % p == 0:
                    x //= p
                    if x % p == 0:
                        res = 0
                        break
                    res = -res
                p += 1
            if res and x > 1:
                res = -res
            mu[m] = res

        ans = 0
        for g in range(1, V + 1):
            if D[g] == 0:
                continue
            # squarefree d with some element divisible by g*d
            ds = [d for d in range(1, V // g + 1) if mu[d] != 0 and D[g * d] > 0]
            us = [g * d for d in ds]
            Ds = [D[u] for u in us]
            ps = [pow2[c] for c in Ds]
            ms = [mu[d] for d in ds]
            k = len(ds)
            for i in range(k):
                u = us[i]
                Du = Ds[i]
                m1 = ms[i]
                for j in range(k):
                    v = us[j]
                    L = u // gcd(u, v) * v
                    DL = D[L]
                    M = (pow2[Du + Ds[j] - 2 * DL] * pow3[DL] - ps[i] - ps[j] + 1) % MOD
                    ans = (ans + m1 * ms[j] * M) % MOD
        return ans % MOD
```

Indices for pow tables: Du+Dv−2DL ≥ 0 since DL ≤ min(Du,Dv); L ≤ 200 guaranteed because an element divisible by both u and v exists.
