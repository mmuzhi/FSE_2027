class Solution:
    def getLargestOutlier(self, nums: List[int]) -> int:
        total = sum(nums)
        cnt = Counter(nums)
        ans = None
        for x in nums:
            rem = total - x
            if rem % 2 != 0:
                continue
            s = rem // 2
            if s in cnt and (cnt[s] > 1 or s != x):
                if ans is None or x > ans:
                    ans = x
        return ans

class Solution:
    def getLargestOutlier(self, nums: List[int]) -> int:

        cnt = Counter(nums)
        ans = None
        for x in nums:

            # sum element s must exist at a distinct index from x
