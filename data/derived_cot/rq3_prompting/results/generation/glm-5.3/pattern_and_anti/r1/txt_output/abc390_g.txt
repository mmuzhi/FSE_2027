import sys
import heapq
import numpy as np

def main():
    N = int(sys.stdin.buffer.read().split()[0])
    p = 998244353

    L = len(str(N))

    # factorials
    fact = [1] * (N + 1)
    for i in range(1, N + 1):
        fact[i] = fact[i - 1] * i % p
    inv_fact = [1] * (N + 1)
    inv_fact[N] = pow(fact[N], p - 2, p)
    for i in range(N, 0, -1):
        inv_fact[i - 1] = inv_fact[i] * i % p
    factnp = np.array(fact, dtype=np.int64)
    invfactnp = np.array(inv_fact, dtype=np.int64)

    def pow_array(base, n):
        res = np.ones(n, dtype=np.int64)
        base %= p
        if n == 1:
            return res
        filled = 1
        mult = base
        while filled < n:
            cnt = min(filled, n - filled)
            res[filled:filled + cnt] = res[:cnt] * mult % p
            filled += cnt
            if cnt * 2 == filled:
                mult = mult * mult % p
            else:
                mult = mult * int(res[cnt]) % p
        return res

    # digit-length classes
    classes = []
    for l in range(1, L + 1):
        lo = 10 ** (l - 1)
        hi = (10 ** l - 1) if l < L else N
        c = hi - lo + 1
        S = (lo + hi) * c // 2 % p
        classes.append((c, S, l))

    # build leaves (1 + 10^l x)^c
    heap = []
    tid = 0
    for (c, S, l) in classes:
        a = pow(10, l, p)
        powers = pow_array(a, c + 1)
        v = factnp[c] * invfactnp[:c + 1] % p
        v = v * invfactnp[c::-1] % p
        v = v * powers % p
        heapq.heappush(heap, (c + 1, tid, v))
        tid += 1

    C64 = (1 << 64) % p
    C72 = (1 << 72) % p

    def pack(arr):
        n = arr.shape[0]
        m = np.zeros((n, 10), dtype=np.uint8)
        m[:, 0] = arr & 0xFF
        m[:, 1] = (arr >> 8) & 0xFF
        m[:, 2] = (arr >> 16) & 0xFF
        m[:, 3] = (arr >> 24) & 0xFF
        return int.from_bytes(m.tobytes(), 'little')

    def unpack(val, ln):
        buf = val.to_bytes(10 * ln + 16, 'little')
        t = np.frombuffer(buf, dtype=np.uint8, count=10 * ln).reshape(ln, 10).astype(np.uint64)
        low = t[:, 0] + (t[:, 1] << np.uint64(8)) + (t[:, 2] << np.uint64(16)) + (t[:, 3] << np.uint64(24)) \
              + (t[:, 4] << np.uint64(32)) + (t[:, 5] << np.uint64(40)) + (t[:, 6] << np.uint64(48)) + (t[:, 7] << np.uint64(56))
        hi = t[:, 8] * np.uint64(C64) + t[:, 9] * np.uint64(C72)
        out = (low % np.uint64(p) + hi) % np.uint64(p)
        return out.astype(np.int64)

    def polymul(A, B):
        ln = A.shape[0] + B.shape[0] - 1
        return unpack(pack(A) * pack(B), ln)

    while len(heap) > 1:
        _, _, A = heapq.heappop(heap)
        _, _, B = heapq.heappop(heap)
        C = polymul(A, B)
        heapq.heappush(heap, (C.shape[0], tid, C))
        tid += 1

    q = heap[0][2]  # coefficients of Q, length N+1

    w = factnp[:N] * factnp[N - 1::-1] % p

    ans = 0
    for (c, S, l) in classes:

        zinv = pow(z, p - 2, p)

        H = Hrev[::-1]
        t = q[:N] * pow_array(zinv, N) % p
        t = t * H % p
        T = int(t.sum()) % p
        ans = (ans + S * T) % p

    print(ans)

main()

Looking at this problem:

- For element k at a position with exactly m elements after it, the number of permutations where a specific set R (|R|=m, k∉R) lies after k is m!(N-1-m)!, contributing k·10^{d(R)}.
- Grouping numbers by digit length ℓ (≤6 classes, counts c_ℓ, sums S_ℓ), Σ_R 10^{d(R)} = [x^m] Q(x)/(1+10^ℓ x) where Q(x)=Π(1+10^j x)^{c_j}.
- So Answer = Σ_ℓ S_ℓ Σ_m w_m [x^m] Q/(1+10^ℓ x), w_m = m!(N-1-m)!.
- With r_m = Σ_t z^t q_{m-t} (z = -10^ℓ), T_ℓ = Σ_j q_j·z^{-j}·Σ_{s≥j} w_s z^s — fully vectorizable (power arrays by doubling, suffix cumsum).
- Q's coefficients (degree N) come from a product tree over ≤6 binomial factors, using exact big-integer Kronecker multiplication (80-bit slots, values < N·p² < 2^78).

    def pack(arr):
        n = arr.shape[0]
        m = np.zeros((n, 10), dtype=np.uint8)
        m[:, 0] = arr & 0xFF; m[:, 1] = (arr >> 8) & 0xFF
        m[:, 2] = (arr >> 16) & 0xFF; m[:, 3] = (arr >> 24) & 0xFF
        return int.from_bytes(m.tobytes(), 'little')
    def unpack(val, ln):
        buf = val.to_bytes(10*ln + 16, 'little')
        t = np.frombuffer(buf, dtype=np.uint8, count=10*ln).reshape(ln, 10).astype(np.uint64)
        low = t[:,0] + (t[:,1] << np.uint64(8)) + (t[:,2] << np.uint64(16)) + (t[:,3] << np.uint64(24)) \
            + (t[:,4] << np.uint64(32)) + (t[:,5] << np.uint64(40)) + (t[:,6] << np.uint64(48)) + (t[:,7] << np.uint64(56))
        hi = t[:,8]*C64 + t[:,9]*C72
        return ((low % PU + hi) % PU).astype(np.int64)
    def polymul(A, B):
        return unpack(pack(A)*pack(B), A.shape[0]+B.shape[0]-1)

    while len(heap) > 1:
        _, _, A = heapq.heappop(heap)
        _, _, B = heapq.heappop(heap)
        C = polymul(A, B)
        heapq.heappush(heap, (C.shape[0], tid, C)); tid += 1
    q = heap[0][2]

    w = factnp[:N] * factnp[N-1::-1] % p
    ans = 0
    for (c, S, l) in classes:
        z = (p - pow(10, l, p)) % p
        zinv = pow(z, p-2, p)
        G = w * pow_array(z, N) % p
        H = (np.cumsum(G[::-1]) % p)[::-1]
        t = q[:N] * pow_array(zinv, N) % p
        t = t * H % p
        T = int(t.sum()) % p
        ans = (ans + S * T) % p
    print(ans)

main()
