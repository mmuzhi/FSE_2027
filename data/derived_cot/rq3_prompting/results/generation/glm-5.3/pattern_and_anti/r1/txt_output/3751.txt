```python
class Solution:
    def maxFrequency(self, nums: List[int], k: int) -> int:
        n = len(nums)

        ans = total_k  # x = 0 case (or any case where subarray sum ≤ 0)
```

```python
import numpy as np
from typing import List

class Solution:
    def maxFrequency(self, nums: List[int], k: int) -> int:
        arr = np.array(nums, dtype=np.int64)
        n = len(arr)
        total_k = int(np.count_nonzero(arr == k))
        ans = total_k
        if total_k == 0:
            vals, counts = np.unique(arr, return_counts=True)
            return int(counts.max())
        if total_k == n:
            return n
        
        is_k = (arr == k)
        zero = np.zeros(1, dtype=np.int64)
        for v in range(1, 51):
            if v == k:
                continue
            c = (arr == v).astype(np.int64)
            if not c.any():
                continue
            c -= is_k
            prefix = np.cumsum(c)
            prefix = np.concatenate((zero, prefix))
            # best subarray sum = max_j prefix[j+1] - min_{i<=j} prefix[i]
            run_min = np.minimum.accumulate(prefix)
            best = int((prefix[1:] - run_min[:-1]).max())
            if total_k + best > ans:
                ans = total_k + best
        return ans
```

```python
from typing import List

class Solution:
    def maxFrequency(self, nums: List[int], k: int) -> int:
        total_k = nums.count(k)
        best_gain = 0

        # For each value v (v != k), count its occurrences per "gap",
        # where gap t = number of k's seen so far (segments between consecutive k's).
        gap_counts = [dict() for _ in range(51)]
        g = 0
        for x in nums:
            if x == k:
                g += 1
            else:
                d = gap_counts[x]
                d[g] = d.get(g, 0) + 1

        # For each value v, run Kadane on the merged +/-1 sequence:
        # +1 at v positions, -1 at k positions. Inside a gap, cur only rises
        # (a batch of +1s); each k decays cur by 1, clamped at 0.
        # Between events of v, cur decays by exactly (t - prev) k's, floored at 0.
        for v in range(1, 51):
            d = gap_counts[v]
            if not d:
                continue
            cur = 0
            prev = 0
            first = True
            for t, c in sorted(d.items()):
                if first:
                    cur = c
                    first = False
                else:
                    decayed = cur - (t - prev)
                    cur = (decayed if decayed > 0 else 0) + c
                prev = t
                if cur > best_gain:
                    best_gain = cur

        # Answer = existing k's + best (count of v in window) - (count of k in window)
        return total_k + best_gain
```
