- Then S_k = Σ_j C(k,j) M^{k-j} β_j with β_j = 1ᵀÃʲe_∅ (only j = 0..6 matter), and the answer = Σβ_j·T_j(N) − 1 (subtracting the empty sequence), where T_j(N)=Σ_{k=0}^{N} C(k,j)M^{k-j}.
- T_j via [y^j] of ((M(1+y))^{N+1}−1)/((M−1)+My), computed as a power series mod y⁷. M=1 is a special case (answer = N mod p).

I verified by hand: (N=1,M=7)→16, (N=3,M=11)→16095 (β = 1,18,190,1224,...), and (N=2,M=3)→28 against brute force — all match, including catching (and fixing) hand-arithmetic slips by cross-checking S₂=707 for M=11 directly.

```python
import sys

def main():
    MOD = 998244353
    data = sys.stdin.read().split()
    N, M = int(data[0]), int(data[1])
    if M == 1:
        print(N % MOD)
        return
    primes = [2, 3, 5, 7, 11, 13]
    P = len(primes)
    vecs = []
    for a in range(1, M + 1):
        x = a
        v = []
        for q in primes:
            e = 0
            while x % q == 0:
                x //= q
                e += 1
            v.append(e)
        vecs.append(v)
    size = 1 << P
    C = [0] * size
    for mask in range(size):
        s = 0
        for v in vecs:
            prod = 1
            for i in range(P):
                if (mask >> i) & 1:
                    prod *= v[i]
            s += prod
        C[mask] = s % MOD

    # beta_j = sum over T of (A~^j e_empty)_T, j = 0..6 (A~^7 = 0)
    u = [0] * size
    u[0] = 1
    beta = []
    for j in range(7):
        beta.append(sum(u) % MOD)
        nu = [0] * size
        for T in range(size):
            acc = 0
            sub = T
            while True:
                if sub != T:
                    acc += C[T ^ sub] * u[sub]
                if sub == 0:
                    break
                sub = (sub - 1) & T
            nu[T] = acc % MOD
        u = nu

    # T_j = sum_{k=0}^{N} C(k,j) M^{k-j} mod p, via series extraction
    Mn1 = pow(M, N + 1, MOD)
    g = [0] * 7
    g[0] = (Mn1 - 1) % MOD
    binom = 1
    n = N + 1
    for i in range(1, 7):
        binom = binom * ((n - i + 1) % MOD) % MOD * pow(i, MOD - 2, MOD) % MOD
        g[i] = Mn1 * binom % MOD
    invMm1 = pow(M - 1, MOD - 2, MOD)
    r = (-M * invMm1) % MOD
    s = [invMm1 * pow(r, i, MOD) % MOD for i in range(7)]
    invM = pow(M, MOD - 2, MOD)
    ans = 0
    invMj = 1
    for j in range(7):
        Fj = 0
        for i in range(j + 1):
            Fj += g[i] * s[j - i] % MOD
        Tj = Fj % MOD * invMj % MOD
        invMj = invMj * invM % MOD
        ans = (ans + beta[j] * Tj) % MOD
    print((ans - 1) % MOD)

main()
```
