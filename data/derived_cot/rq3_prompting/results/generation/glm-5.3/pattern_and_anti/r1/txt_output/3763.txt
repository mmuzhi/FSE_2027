```python
class Solution:
    def separateSquares(self, squares: List[List[int]]) -> float:
        n = len(squares)
        total = sum(l*l for _,_,l in squares)
        target = total / 2.0
        ys = sorted(set([y for _,y,_ in squares] + [y+l for _,y,l in squares]))
        starts = sorted(squares, key=lambda s: s[1])  # by y
        ends = sorted([(y+l, l) for _,y,l in squares])
        i = j = 0
        width = 0
        base = 0.0
        for k in range(len(ys)-1):
            y0, y1 = ys[k], ys[k+1]
            while i < n and starts[i][1] <= y0:
                width += starts[i][2]
                i += 1
            while j < n and ends[j][0] <= y0:
                width -= ends[j][1]
                j += 1
            if width > 0:
                gain = width * (y1 - y0)
                if base + gain >= target:
                    return y0 + (target - base) / width
                base += gain
        return float(ys[-1])  # shouldn't reach
```

```python
class Solution:
    def separateSquares(self, squares: List[List[int]]) -> float:
        n = len(squares)
        total = sum(l * l for _, _, l in squares)
        target = total / 2.0
        
        # Breakpoints where the slope of "area below" changes
        ys = sorted(set([y for _, y, _ in squares] + [y + l for _, y, l in squares]))
        
        starts = sorted(squares, key=lambda s: s[1])          # squares entering, by bottom y
        ends = sorted((y + l, l) for _, y, l in squares)      # squares leaving, by top y
        
        i = j = 0
        width = 0   # sum of side lengths of squares crossing the current level (slope)
        base = 0.0  # area strictly below current breakpoint
        
        for k in range(len(ys) - 1):
            y0, y1 = ys[k], ys[k + 1]
            while i < n and starts[i][1] <= y0:
                width += starts[i][2]
                i += 1
            while j < n and ends[j][0] <= y0:
                width -= ends[j][1]
                j += 1
            
            if width > 0:
                gain = width * (y1 - y0)
                if base + gain >= target:
                    # Solve base + width * (c - y0) = target exactly on this linear segment
                    return y0 + (target - base) / width
                base += gain
        
        return float(ys[-1])
```
