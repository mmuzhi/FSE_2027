import sys
import numpy as np

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0]); d = int(data[1])
    arr = np.array(data[2:2 + 2*n], dtype=np.int64)
    xs = np.sort(arr[0::2])
    ys = np.sort(arr[1::2])
    
    px = np.concatenate(([0], np.cumsum(xs)))  # length n+1
    py = np.concatenate(([0], np.cumsum(ys)))
    total_x = int(px[-1]); total_y = int(py[-1])
    
    lo_x = int(xs[-1]) - d
    hi_x = int(xs[0]) + d
    if lo_x > hi_x:
        print(0); return
    X = np.arange(lo_x, hi_x + 1, dtype=np.int64)
    idx = np.searchsorted(xs, X, side='right')
    fX = X * (2 * idx - n) - 2 * px[idx] + total_x
    
    lo_y = int(ys[-1]) - d
    hi_y = int(ys[0]) + d
    if lo_y > hi_y:
        print(0); return
    Y = np.arange(lo_y, hi_y + 1, dtype=np.int64)
    idy = np.searchsorted(ys, Y, side='right')
    gY = Y * (2 * idy - n) - 2 * py[idy] + total_y
    
    fX = fX[fX <= d]
    gY = gY[gY <= d]
    if fX.size == 0 or gY.size == 0:
        print(0); return
    
    cntF = np.bincount(fX, minlength=d + 1)
    cntG = np.bincount(gY, minlength=d + 1)
    cumG = np.cumsum(cntG)
    ans = int(np.dot(cntF, cumG[::-1]))
    print(ans)

main()

Looking at this problem:

- The sum splits as f(x) + g(y) where f(x) = Σ|x−x_i|, g(y) = Σ|y−y_i|. Count pairs with f(x)+g(y) ≤ D.
- Since f(x) ≥ |x−x_i| for each i, any valid x must lie in [max(x_i)−D, min(x_i)+D] (length ≤ 2D+1 ≤ 2·10⁶+1), similarly for y.
- Compute f over that range vectorized via sorted xs + prefix sums: with idx = #{x_i ≤ x}, f(x) = x·(2·idx−N) + total − 2·prefix[idx].
- Then bucket values ≤ D into cntF/cntG arrays of size D+1, take prefix sum of cntG, and answer = Σ_v cntF[v]·cumG[D−v] = dot(cntF, cumG reversed).

Verified on sample 1: cntF=[0,2,0,2], cumG=[1,1,3,3], dot with reversed cumG gives 8 ✓. Sample 2: range empty → 0 ✓. N=1 sanity: D=3 gives 25, matching |x|+|y|≤3 lattice count ✓. Overflow safe in int64 (intermediates ≤ ~4·10¹¹, answer ≤ ~2·10¹²).
