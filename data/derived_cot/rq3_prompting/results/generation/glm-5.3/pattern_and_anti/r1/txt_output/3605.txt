```python
class Solution:
    def minBitwiseArray(self, nums: List[int]) -> List[int]:
```

```python
class Solution:
    def minBitwiseArray(self, nums: List[int]) -> List[int]:
        # Key fact: x | (x+1) equals x with its lowest 0 bit set to 1.
        # So num must equal x with its lowest 0 bit (at position k) set to 1,
        # meaning bits 0..k of num are all 1 and higher bits match x.
        # To minimize x, clear the highest such bit: k = (trailing ones of num) - 1.
        # num+1 has its lowest set bit at position t (t = trailing ones of num),
        # so subtract 2^(t-1) = ((num+1) & -(num+1)) >> 1.
        # If num is even (only prime 2), x | (x+1) is always odd -> impossible.
        ans = []
        for num in nums:
            if num % 2 == 0:
                ans.append(-1)
            else:
                low = (num + 1) & (-(num + 1))  # 2^t where t = trailing ones
                ans.append(num - (low >> 1))
        return ans
```
