from bisect import bisect_left, bisect_right

class Solution:
    def shortestMatchingSubstring(self, s: str, p: str) -> int:
        A, B, C = p.split('*')
        n = len(s)
        
        def find_all(pat):
            res = []
            if not pat:
                return res
            start = 0
            find = s.find
            while True:
                idx = find(pat, start)
                if idx == -1:
                    return res
                res.append(idx)
                start = idx + 1
        
        occA = find_all(A)
        occB = find_all(B)
        occC = find_all(C)
        
        la, lb, lc = len(A), len(B), len(C)
        
        # handle pieces
        best = float('inf')
        
        # B occurrences: if B empty, all j in 0..n
        if lb == 0:
            j_list = range(n + 1)
        else:
            j_list = occB
        
        for j in j_list:
            # i: latest A start with i + la <= j
            if la == 0:
                i = j
            else:
                idx = bisect_right(occA, j - la) - 1
                if idx < 0:
                    continue
                i = occA[idx]
            # c: earliest C start >= j + lb
            k = j + lb
            if lc == 0:

class Solution:
    def shortestMatchingSubstring(self, s: str, p: str) -> int:
        A, B, C = p.split('*')
        la, lb, lc = len(A), len(B), len(C)
        n = len(s)

            res = []
            start = 0
            find = s.find
            while True:
                idx = find(pat, start)
                if idx == -1:
                    return res
                res.append(idx)
                start = idx + 1

        occA = find_all(A) if la else []
        occB = find_all(B) if lb else []
        occC = find_all(C) if lc else []

        if (la and not occA) or (lb and not occB) or (lc and not occC):
            return -1

        j_list = range(n + 1) if lb == 0 else occB
        best = -1
        lower = la + lb + lc  # minimum achievable length

                    continue
                c = occC[t2]
            cand = c + lc - i
            if best == -1 or cand < best:
                best = cand
                if best == lower:
                    break

        return best
