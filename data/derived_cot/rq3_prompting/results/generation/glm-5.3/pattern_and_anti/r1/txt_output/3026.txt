class Solution:
    def minimumPossibleSum(self, n: int, target: int) -> int:
        MOD = 10**9 + 7
        h = target // 2
        if n <= h:
            return (n * (n + 1) // 2) % MOD
        m = n - h
        total = h * (h + 1) // 2 + m * target + m * (m - 1) // 2
        return total % MOD
```

class Solution:
    def minimumPossibleSum(self, n: int, target: int) -> int:
        MOD = 10**9 + 7
        h = target // 2
        if n <= h:
            # take 1..n; no two distinct of them can sum to target
            return (n * (n + 1) // 2) % MOD
        # take 1..h, then target, target+1, ..., target+(n-h)-1
        m = n - h
        total = h * (h + 1) // 2 + m * target + m * (m - 1) // 2
        return total % MOD
```
