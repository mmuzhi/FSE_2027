```python
class Solution:
    def makeStringGood(self, s: str) -> int:
        from collections import Counter
        cnt = [0] * 26
        for ch in s:
            cnt[ord(ch) - 97] += 1
        maxc = max(cnt)
        best = len(s)  # upper bound (delete everything... well m=1 will beat; safe init)
        for m in range(1, maxc + 1):
            dp0 = cnt[0]                # t_0 = 0
            dp1 = abs(cnt[0] - m)       # t_0 = m
            sp0 = cnt[0]                # surplus of letter 0 if t=0
            spm = cnt[0] - m if cnt[0] > m else 0
            for i in range(1, 26):
                c = cnt[i]
                d = m - c if c < m else 0
                nd = dp0 if dp0 < dp1 else dp1
                new0 = c + nd
                a = dp0 - (sp0 if sp0 < d else d)
                b = dp1 - (spm if spm < d else d)
                new1 = (d if d else c - m) + (a if a < b else b)
                # abs(c - m): if c < m: m - c = d; else c - m
                dp0, dp1 = new0, new1
                sp0 = c
                spm = c - m if c > m else 0
            if dp0 < best:
                best = dp0
            if dp1 < best:
                best = dp1
        return best
```

```python
class Solution:
    def makeStringGood(self, s: str) -> int:
        cnt = [0] * 26
        for ch in s:
            cnt[ord(ch) - 97] += 1
        maxc = max(cnt)
        best = len(s)
        for m in range(1, maxc + 1):
            c0 = cnt[0]
            dp0 = c0
            dp1 = c0 - m if c0 > m else m - c0
            sp0 = c0
            spm = c0 - m if c0 > m else 0
            for i in range(1, 26):
                c = cnt[i]
                if c < m:
                    d = m - c
                    cm = d
                else:
                    d = 0
                    cm = c - m
                m01 = dp0 if dp0 < dp1 else dp1
                new0 = c + m01
                a = dp0 - (sp0 if sp0 < d else d)
                b = dp1 - (spm if spm < d else d)
                new1 = cm + (a if a < b else b)
                dp0 = new0
                dp1 = new1
                sp0 = c
                spm = c - m if c > m else 0
            if dp0 < best: best = dp0
            if dp1 < best: best = dp1
        return best
```
