```python
import heapq

class Solution:
    def minTimeToReach(self, moveTime: List[List[int]]) -> int:
        n, m = len(moveTime), len(moveTime[0])
        INF = float('inf')
        dist = [[INF] * m for _ in range(n)]
        dist[0][0] = 0
        heap = [(0, 0, 0)]
        while heap:
            t, i, j = heapq.heappop(heap)
            if t > dist[i][j]:
                continue
            if i == n - 1 and j == m - 1:
                return t
            for di, dj in ((1,0),(-1,0),(0,1),(0,-1)):
                x, y = i + di, j + dj
                if 0 <= x < n and 0 <= y < m:
                    nt = max(t, moveTime[x][y]) + 1
                    if nt < dist[x][y]:
                        dist[x][y] = nt
                        heapq.heappush(heap, (nt, x, y))
        return dist[n-1][m-1]
```
