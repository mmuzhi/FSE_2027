class Solution:
    def maxMoves(self, kx: int, ky: int, positions: List[List[int]]) -> int:
        from collections import deque
        
        n = len(positions)
        N = 50
        kmoves = [(1,2),(2,1),(-1,2),(-2,1),(1,-2),(2,-1),(-1,-2),(-2,-1)]
        
        points = [(kx, ky)] + [(x, y) for x, y in positions]
        m = len(points)
        
        # BFS from each point
        grid_dist = []
        for (sx, sy) in points:
            d = [[-1]*N for _ in range(N)]
            d[sx][sy] = 0
            q = deque([(sx, sy)])
            while q:
                x, y = q.popleft()
                dd = d[x][y]
                for dx, dy in kmoves:
                    nx, ny = x+dx, y+dy
                    if 0 <= nx < N and 0 <= ny < N and d[nx][ny] == -1:
                        d[nx][ny] = dd + 1
                        q.append((nx, ny))
            grid_dist.append(d)
        
        startDist = [grid_dist[0][x][y] for (x, y) in points[1:]]
        pawnDist = [[0]*n for _ in range(n)]
        for i in range(n):
            di = grid_dist[i+1]
            for j in range(n):
                if i != j:
                    x, y = points[j+1]
                    pawnDist[i][j] = di[x][y]
        
        size = 1 << n
        BIG = 10**9
        dp = [[0]*n for _ in range(size)]
        for mask in range(size - 2, 0, -1):
            p = bin(mask).count('1')
            alice = (p & 1) == 0
            cand = [(i, mask | (1 << i)) for i in range(n) if not (mask >> i) & 1]
            row = dp[mask]
            for last in range(n):
                if not (mask >> last) & 1:
                    continue
                dl = pawnDist[last]
                if alice:
                    best = -1
                    for i, nxt in cand:
                        val = dl[i] + dp[nxt][i]
                        if val > best:
                            best = val
                else:
                    best = BIG
                    for i, nxt in cand:
                        val = dl[i] + dp[nxt][i]
                        if val < best:
                            best = val
                row[last] = best
        
        ans = -1
        for i in range(n):
            val = startDist[i] + dp[1 << i][i]
            if val > ans:
                ans = val
        return ans

class Solution:
    def maxMoves(self, kx: int, ky: int, positions: List[List[int]]) -> int:
        from collections import deque
        
        n = len(positions)
        N = 50
        kmoves = ((1,2),(2,1),(-1,2),(-2,1),(1,-2),(2,-1),(-1,-2),(-2,-1))
        
        points = [(kx, ky)] + [(x, y) for x, y in positions]
        
        grid_dist = []
        for sx, sy in points:
            d = [[-1] * N for _ in range(N)]
            d[sx][sy] = 0
            q = deque([(sx, sy)])
            while q:
                x, y = q.popleft()
                nd = d[x][y] + 1
                for dx, dy in kmoves:
                    nx, ny = x + dx, y + dy
                    if 0 <= nx < N and 0 <= ny < N and d[nx][ny] < 0:
                        d[nx][ny] = nd
                        q.append((nx, ny))
            grid_dist.append(d)
        
        startDist = [grid_dist[0][x][y] for x, y in points[1:]]
        pawnDist = [[0] * n for _ in range(n)]
        for i in range(n):
            di = grid_dist[i + 1]
            for j in range(n):
                if i != j:
                    x, y = points[j + 1]
                    pawnDist[i][j] = di[x][y]
        
        size = 1 << n
        BIG = 1 << 30
        dp = [[0] * n for _ in range(size)]
        for mask in range(size - 2, 0, -1):
            alice = (bin(mask).count('1') & 1) == 0
            cand = [(i, mask | (1 << i)) for i in range(n) if not (mask >> i) & 1]
            row = dp[mask]
            for last in range(n):
                if not (mask >> last) & 1:
                    continue
                dl = pawnDist[last]
                if alice:
                    best = -1
                    for i, nxt in cand:
                        val = dl[i] + dp[nxt][i]
                        if val > best:
                            best = val
                else:
                    best = BIG
                    for i, nxt in cand:
                        val = dl[i] + dp[nxt][i]
                        if val < best:
                            best = val
                row[last] = best
        
        return max(startDist[i] + dp[1 << i][i] for i in range(n))

- **Decisive constraints**: n ≤ 15 pawns → bitmask game DP over (mask, last captured pawn), since the knight's position after captures is exactly the last captured pawn's square. - **Edge cases**: n = 1 (answer = start distance); mask = full → 0; mask = 0 handled via the virtual start (first move is Alice's, maximize).
