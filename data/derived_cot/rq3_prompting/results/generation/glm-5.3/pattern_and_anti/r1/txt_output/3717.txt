```python
import heapq
from collections import defaultdict

class Solution:
    def minOperations(self, nums: List[int], x: int, k: int) -> int:
        n = len(nums)
        w = x
        half = (w + 1) // 2  # lo size when full
        
        cost = [0] * (n - w + 1)
        
        lo = []  # negated values (max-heap)
        hi = []
        slo = shi = 0  # sums of valid
        nlo = nhi = 0  # valid counts
        # pending deletions per heap: maps value -> count
        dlo = defaultdict(int)
        dhi = defaultdict(int)
        
        def cleanlo():
            while lo and dlo[-lo[0]]:
                dlo[-lo[0]] -= 1
                heapq.heappop(lo)
        def cleanhi():
            while hi and dhi[hi[0]]:
                dhi[hi[0]] -= 1
                heapq.heappop(hi)
        
        for i in range(n):
            v = nums[i]
            # route v
            if nlo and v <= -lo[0]:
                heapq.heappush(lo, -v)
                slo += v; nlo += 1
            else:
                heapq.heappush(hi, v)
                shi += v; nhi += 1
            
            if i >= w:
                u = nums[i - w]
                if nlo and u <= -lo[0]:
                    dlo[u] += 1
                    slo -= u; nlo -= 1
                else:
                    dhi[u] += 1
                    shi -= u; nhi -= 1
                cleanlo(); cleanhi()
            
            # rebalance
            if i >= w - 1:
                while nlo > half:
                    cleanlo()
                    t = -heapq.heappop(lo)
                    slo -= t; nlo -= 1
                    heapq.heappush(hi, t); shi += t; nhi += 1
                while nlo < half and nhi > 0:
                    cleanhi()
                    t = heapq.heappop(hi)
                    shi -= t; nhi -= 1
                    heapq.heappush(lo, -t); slo += t; nlo += 1
                cleanlo(); cleanhi()
                m = -lo[0]
                cost[i - w + 1] = m * nlo - slo + (shi - m * nhi)
        
        # DP
        INF = float('inf')
        # dp[j][i]: min cost of j disjoint windows within prefix ending at index i (0-based, windows fully inside [0..i])
        # dp0 = zeros
        prev = [0] * (n + 1)  # dp[0][i] = 0
        for j in range(1, k + 1):
            cur = [INF] * (n + 1)
            # window starting at s occupies [s, s+w-1]; dp[j][e] where e = s+w-1
            for e in range(w - 1, n + 1):
                a = cur[e - 1]
                b = prev[e - w] + cost[e - w + 1]  # cost index = s = e-w+1
                cur[e] = a if a < b else b
            prev = cur
        return prev[n]
```

import heapq
from collections import defaultdict
from typing import List

class Solution:
    def minOperations(self, nums: List[int], x: int, k: int) -> int:
        n = len(nums)
        w = x
        half = (w + 1) // 2
        m_ = n - w + 1  # number of windows
        
        lo = []  # max-heap via negation, lower half
        hi = []  # min-heap, upper half
        slo = shi = 0
        nlo = nhi = 0
        dlo = defaultdict(int)
        dhi = defaultdict(int)
        
        heappush = heapq.heappush
        heappop = heapq.heappop
        
        cost = [0] * m_
        
        for i in range(n):
            v = nums[i]
            if nlo and v <= -lo[0]:
                heappush(lo, -v)
                slo += v
                nlo += 1
            else:
                heappush(hi, v)
                shi += v
                nhi += 1
            
            if i >= w:
                u = nums[i - w]
                if nlo and u <= -lo[0]:
                    dlo[u] += 1
                    slo -= u
                    nlo -= 1
                else:
                    dhi[u] += 1
                    shi -= u
                    nhi -= 1
                # clean tops
                while lo and dlo[-lo[0]]:
                    dlo[-lo[0]] -= 1
                    heappop(lo)
                while hi and dhi[hi[0]]:
                    dhi[hi[0]] -= 1
                    heappop(hi)
            
            if i >= w - 1:
                while nlo > half:
                    while dlo[-lo[0]]:
                        dlo[-lo[0]] -= 1
                        heappop(lo)
                    t = -heappop(lo)
                    slo -= t; nlo -= 1
                    heappush(hi, t); shi += t; nhi += 1
                while nlo < half and nhi > 0:
                    while dhi[hi[0]]:
                        dhi[hi[0]] -= 1
                        heappop(hi)
                    t = heappop(hi)
                    shi -= t; nhi -= 1
                    heappush(lo, -t); slo += t; nlo += 1
                while lo and dlo[-lo[0]]:
                    dlo[-lo[0]] -= 1
                    heappop(lo)
                while hi and dhi[hi[0]]:
                    dhi[hi[0]] -= 1
                    heappop(hi)
                med = -lo[0]
                cost[i - w + 1] = med * nlo - slo + shi - med * nhi
        
        # DP over windows
        INF = float('inf')
        prev = [0] * (n + 1)
        for j in range(1, k + 1):
            cur = [INF] * (n + 1)
            pv = prev
            cv = cur
            for i in range(j * w, n + 1):
                s = i - w
                b = pv[s] + cost[s]
                a = cv[i - 1]
                cv[i] = a if a < b else b
            prev = cur
        return prev[n]

import heapq
from typing import List

class Solution:
    def minOperations(self, nums: List[int], x: int, k: int) -> int:
        n = len(nums)
        w = x
        half = (w + 1) // 2
        
        lo = []  # max-heap (negated values) - lower half
        hi = []  # min-heap - upper half
        slo = shi = 0
        nlo = nhi = 0
        dlo = {}
        dhi = {}
        dlo_get = dlo.get
        dhi_get = dhi.get
        
        heappush = heapq.heappush
        heappop = heapq.heappop
        
        nw = n - w + 1
        cost = [0] * nw
        
        for i in range(n):
            v = nums[i]
            if nlo and v <= -lo[0]:
                heappush(lo, -v)
                slo += v
                nlo += 1
            else:
                heappush(hi, v)
                shi += v
                nhi += 1
            
            if i >= w:
                u = nums[i - w]
                if nlo and u <= -lo[0]:
                    dlo[u] = dlo.get(u, 0) + 1
                    slo -= u
                    nlo -= 1
                else:
                    dhi[u] = dhi.get(u, 0) + 1
                    shi -= u
                    nhi -= 1
                while lo:
                    t = -lo[0]
                    c = dlo_get(t)
                    if c:
                        dlo[t] = c - 1
                        heappop(lo)
                    else:
                        break
                while hi:
                    t = hi[0]
                    c = dhi_get(t)
                    if c:
                        dhi[t] = c - 1
                        heappop(hi)
                    else:
                        break
            
            if i >= w - 1:
                while nlo > half:
                    while True:
                        t = -lo[0]
                        c = dlo_get(t)
                        if c:
                            dlo[t] = c - 1
                            heappop(lo)
                        else:
                            break
                    t = -heappop(lo)
                    slo -= t; nlo -= 1
                    heappush(hi, t); shi += t; nhi += 1
                while nlo < half and nhi > 0:
                    while True:
                        t = hi[0]
                        c = dhi_get(t)
                        if c:
                            dhi[t] = c - 1
                            heappop(hi)
                        else:
                            break
                    t = heappop(hi)
                    shi -= t; nhi -= 1
                    heappush(lo, -t); slo += t; nlo += 1
                while lo:
                    t = -lo[0]
                    c = dlo_get(t)
                    if c:
                        dlo[t] = c - 1
                        heappop(lo)
                    else:
                        break
                while hi:
                    t = hi[0]
                    c = dhi_get(t)
                    if c:
                        dhi[t] = c - 1
                        heappop(hi)
                    else:
                        break
                med = -lo[0]
                cost[i - w + 1] = med * nlo - slo + shi - med * nhi
        
        INF = float('inf')
        prev = [0] * (n + 1)
        for j in range(1, k + 1):
            cur = [INF] * (n + 1)
            b = INF
            for i in range(j * w, n + 1):
                s = i - w
                c = prev[s] + cost[s]
                if c < b:
                    b = c
                cur[i] = b
            prev = cur
        return prev[n]

import heapq
from typing import List

class Solution:
    def minOperations(self, nums: List[int], x: int, k: int) -> int:
        n = len(nums)
        w = x
        half = (w + 1) // 2  # size of lower half when window is full

        lo = []  # max-heap via negation: lower half
        hi = []  # min-heap: upper half
        slo = shi = 0   # sums of valid elements
        nlo = nhi = 0   # counts of valid elements
        dlo = {}        # pending lazy deletions for lo (by value)
        dhi = {}        # pending lazy deletions for hi (by value)
        dlo_get = dlo.get
        dhi_get = dhi.get

        heappush = heapq.heappush
        heappop = heapq.heappop

        cost = [0] * (n - w + 1)

        for i in range(n):
            v = nums[i]
            if nlo and v <= -lo[0]:
                heappush(lo, -v)
                slo += v
                nlo += 1
            else:
                heappush(hi, v)
                shi += v
                nhi += 1

            if i >= w:
                u = nums[i - w]
                if nlo and u <= -lo[0]:
                    dlo[u] = dlo.get(u, 0) + 1
                    slo -= u
                    nlo -= 1
                else:
                    dhi[u] = dhi.get(u, 0) + 1
                    shi -= u
                    nhi -= 1
                while lo:
                    t = -lo[0]
                    c = dlo_get(t)
                    if c:
                        dlo[t] = c - 1
                        heappop(lo)
                    else:
                        break
                while hi:
                    t = hi[0]
                    c = dhi_get(t)
                    if c:
                        dhi[t] = c - 1
                        heappop(hi)
                    else:
                        break

            if i >= w - 1:
                # rebalance so that lo holds exactly `half` valid elements
                while nlo > half:
                    while True:
                        t = -lo[0]
                        c = dlo_get(t)
                        if c:
                            dlo[t] = c - 1
                            heappop(lo)
                        else:
                            break
                    t = -heappop(lo)
                    slo -= t; nlo -= 1
                    heappush(hi, t); shi += t; nhi += 1
                while nlo < half and nhi > 0:
                    while True:
                        t = hi[0]
                        c = dhi_get(t)
                        if c:
                            dhi[t] = c - 1
                            heappop(hi)
                        else:
                            break
                    t = heappop(hi)
                    shi -= t; nhi -= 1
                    heappush(lo, -t); slo += t; nlo += 1
                while lo:
                    t = -lo[0]
                    c = dlo_get(t)
                    if c:
                        dlo[t] = c - 1
                        heappop(lo)
                    else:
                        break
                while hi:
                    t = hi[0]
                    c = dhi_get(t)
                    if c:
                        dhi[t] = c - 1
                        heappop(hi)
                    else:
                        break
                med = -lo[0]
                cost[i - w + 1] = med * nlo - slo + shi - med * nhi

        # dp over choosing k disjoint windows
        INF = float('inf')
        prev = [0] * (n + 1)
        for j in range(1, k + 1):
            cur = [INF] * (n + 1)
            b = INF
            for i in range(j * w, n + 1):
                s = i - w
                c = prev[s] + cost[s]
                if c < b:
                    b = c
                cur[i] = b
            prev = cur
        return prev[n]
