class Solution:
    def maxDistance(self, s: str, k: int) -> int:
        # Manhattan(x, y) = max over signs (sx, sy) of sx*x + sy*y.
        # For a fixed quadrant (sx, sy), each character contributes +1 or -1
        # to the projection. A -1 contributor can be changed (1 change) into
        # a +1 contributor, gaining 2 per flip. So the best projection at
        # prefix t is p - n + 2*min(n, k) where p/n are +/-1 counts in prefix t.
        ans = 0
        for sx, sy in ((1, 1), (1, -1), (-1, 1), (-1, -1)):
            cur = 0   # unflipped prefix sum of contributions
            neg = 0   # count of -1 contributions so far
            for c in s:
                if c == 'N':
                    v = sy
                elif c == 'S':
                    v = -sy
                elif c == 'E':
                    v = sx
                else:  # 'W'
                    v = -sx
                if v > 0:
                    cur += 1
                else:
                    cur -= 1
                    neg += 1
                cand = cur + 2 * min(neg, k)
                if cand > ans:
                    ans = cand
        return ans
