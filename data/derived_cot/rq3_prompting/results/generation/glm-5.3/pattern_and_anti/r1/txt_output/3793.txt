```python
class Solution:
    def longestPalindrome(self, s: str, t: str) -> int:
        ans = 0
        n, m = len(s), len(t)
        for i in range(n + 1):
            for j in range(i, n + 1):
                a = s[i:j]
                for k in range(m + 1):
                    for l in range(k, m + 1):
                        p = a + t[k:l]
                        if p == p[::-1]:
                            ans = max(ans, len(p))
        return ans
```

```python
class Solution:
    def longestPalindrome(self, s: str, t: str) -> int:
        n, m = len(s), len(t)
        
        def is_pal(u):
            return u == u[::-1]
        
        # palStartS[i]: longest palindromic substring of s starting at i
        palStartS = [0] * (n + 1)
        for i in range(n):
            best = 0
            for j in range(i, n):
                if is_pal(s[i:j+1]):
                    best = j - i + 1
            palStartS[i] = best
        
        # palEndT[e]: longest palindromic substring of t ending at e
        palEndT = [0] * (m)  # palEndT[e] for e in [0, m); handle e=-1 separately as 0
        palEndT = [0] * (m + 1)  # index shift: palEndT[e+1] for e in [-1, m-1]
        for e in range(m):
            best = 0
            for st in range(e + 1):
                if is_pal(t[st:e+1]):
                    best = max(best, e - st + 1)
            palEndT[e + 1] = best
        
        ans = max(max(palStartS), max(palEndT))
        
        for x1 in range(n):

                    cand2 = 2 * b + palEndT[y2 - b + 1]
                    if cand2 > ans:
                        ans = cand2
        return ans
```

```python
class Solution:
    def longestPalindrome(self, s: str, t: str) -> int:
        n, m = len(s), len(t)

        # palStartS[i]: longest palindromic substring of s starting at index i (palStartS[n] = 0)
        palStartS = [0] * (n + 1)
        for i in range(n):
            for j in range(i, n):
                sub = s[i:j + 1]
                if sub == sub[::-1]:
                    palStartS[i] = j - i + 1

        # palEndT[e+1]: longest palindromic substring of t ending at index e (palEndT[0] = 0 for e = -1)
        palEndT = [0] * (m + 1)
        for e in range(m):
            for st in range(e, -1, -1):
                sub = t[st:e + 1]
                if sub == sub[::-1]:
                    palEndT[e + 1] = e - st + 1
                    break

        ans = max(max(palStartS), max(palEndT))

        for x1 in range(n):
            for y2 in range(m):
                k = 0
                while k < n - x1 and k <= y2 and s[x1 + k] == t[y2 - k]:
                    k += 1
                for b in range(1, k + 1):
                    c1 = 2 * b + palStartS[x1 + b]
                    c2 = 2 * b + palEndT[y2 - b + 1]
                    if c1 > ans:
                        ans = c1
                    if c2 > ans:
                        ans = c2
        return ans
```

```python
class Solution:
    def longestPalindrome(self, s: str, t: str) -> int:
        n, m = len(s), len(t)

        # palStartS[i]: longest palindromic substring of s starting at index i (palStartS[n] = 0)
        palStartS = [0] * (n + 1)
        for i in range(n):
            for j in range(i, n):
                sub = s[i:j + 1]
                if sub == sub[::-1]:
                    palStartS[i] = j - i + 1

        # palEndT[e+1]: longest palindromic substring of t ending at index e (palEndT[0] = 0 for e = -1)
        palEndT = [0] * (m + 1)
        for e in range(m):
            for st in range(e, -1, -1):
                sub = t[st:e + 1]
                if sub == sub[::-1]:
                    palEndT[e + 1] = e - st + 1
                    break

        # Single-string palindromes (covers empty counterpart cases)
        ans = max(max(palStartS), max(palEndT))

        # Crossing palindromes: P = A + B with A = s[x1..], B = t[..y2]
        # match: s[x1+u] == t[y2-u] for u < k
        for x1 in range(n):
            for y2 in range(m):
                k = 0
                while k < n - x1 and k <= y2 and s[x1 + k] == t[y2 - k]:
                    k += 1
                for b in range(1, k + 1):
                    # A = reverse(B) + M, M palindrome in s starting at x1+b
                    c1 = 2 * b + palStartS[x1 + b]
                    # B = C + reverse(A), C palindrome in t ending at y2-b
                    c2 = 2 * b + palEndT[y2 - b + 1]
                    if c1 > ans:
                        ans = c1
                    if c2 > ans:
                        ans = c2
        return ans
```
