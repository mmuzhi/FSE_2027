```python
class Solution:
    def minOperations(self, n: int, m: int) -> int:
        LIMIT = 10**4
        # Sieve of Eratosthenes
        is_prime = [False, False] + [True] * (LIMIT - 1)
        for i in range(2, int(LIMIT**0.5) + 1):
            if is_prime[i]:
                for j in range(i*i, LIMIT + 1, i):
                    is_prime[j] = False
        
        if is_prime[n] or is_prime[m]:
            return -1
        
        lo = 10 ** (len(str(n)) - 1)
        hi = 10 ** len(str(n)) - 1
        
        import heapq
        dist = {n: n}
        heap = [(n, n)]
        while heap:
            d, u = heapq.heappop(heap)
            if d > dist.get(u, float('inf')):
                continue
            if u == m:
                return d
            # neighbors: change each digit
            for k in range(len(str(u))):  # since u stays in [lo, hi], same digit count
                delta = 10 ** k
                for nd in (u - delta, u + delta):
                    if lo <= nd <= hi and not is_prime[nd]:
                        if nd not in dist or d + nd < dist[nd]:
                            dist[nd] = d + nd
                            heapq.heappush(heap, (d + nd, nd))
        return -1
```

```python
import heapq

class Solution:
    def minOperations(self, n: int, m: int) -> int:
        LIM = 10000
        # Sieve of Eratosthenes up to 10^4
        is_prime = bytearray([1]) * (LIM + 1)
        is_prime[0] = is_prime[1] = 0
        i = 2
        while i * i <= LIM:
            if is_prime[i]:
                is_prime[i * i::i] = bytearray((LIM - i * i) // i + 1)
            i += 1

        # If start or target is prime, it's impossible
        if is_prime[n] or is_prime[m]:
            return -1

        d = len(str(n))
        lo = 10 ** (d - 1)
        hi = 10 ** d - 1
        INF = float('inf')
        dist = [INF] * (LIM + 1)
        dist[n] = n  # cost counts every value n takes, including the initial one
        pq = [(n, n)]

        while pq:
            cost, u = heapq.heappop(pq)
            if cost > dist[u]:
                continue
            if u == m:
                return cost
            for k in range(d):
                p = 10 ** k
                dig = (u // p) % 10
                if dig < 9:  # increase this digit (no carry since digit < 9)
                    v = u + p
                    if v <= hi and not is_prime[v] and cost + v < dist[v]:
                        dist[v] = cost + v
                        heapq.heappush(pq, (cost + v, v))
                if dig > 0:  # decrease this digit (no borrow since digit > 0)
                    v = u - p
                    if v >= lo and not is_prime[v] and cost + v < dist[v]:
                        dist[v] = cost + v
                        heapq.heappush(pq, (cost + v, v))

        return -1
```
