```python
class Solution:
    def maxSubarraySum(self, nums: List[int]) -> int:
        from collections import defaultdict
        
        n = len(nums)
        
        # Prefix sums
        P = [0] * (n + 1)
        s = 0
        for i in range(n):
            s += nums[i]
            P[i + 1] = s
        
        # Kadane without deletion
        ans = nums[0]
        cur = 0
        for v in nums:
            cur = v if cur < 0 else cur + v
            if cur > ans:
                ans = cur
        
        # Sparse tables for range min / max over P
        m = n + 1
        mn = [P]
        mx = [P]
        j = 1
        while (1 << j) <= m:
            h = 1 << (j - 1)
            a = mn[-1]
            mn.append(list(map(min, a, a[h:])))
            b = mx[-1]
            mx.append(list(map(max, b, b[h:])))
            j += 1
        
        # Group occurrence positions by value
        pos = defaultdict(list)
        for i in range(n):
            pos[nums[i]].append(i)
        
        NEG = -(1 << 62)
        for ps in pos.values():
            if len(ps) == n:
                continue
            G = NEG
            prev = -1
            for p in ps + [n]:
                l = prev + 1
                r = p - 1
                prev = p
                if l > r:
                    continue
                length = r - l + 1
                k = length.bit_length() - 1
                kk = 1 << k
                t = mx[k]
                i1 = l + 1
                i2 = r + 2 - kk
                a = t[i1]
                b2 = t[i2]
                maxp = (a if a > b2 else b2) - P[l]
                t2 = mn[k]
                j1 = l
                j2 = r + 1 - kk
                c = t2[j1]
                d2 = t2[j2]
                mnv = c if c < d2 else d2
                pr = P[r + 1]
                ms = pr - mnv
                if G != NEG:
                    cand = G + maxp
                    if cand > ans:
                        ans = cand
                    g2 = G + pr - P[l]
                    G = ms if ms > g2 else g2
                else:
                    G = ms
        
        return ans
```

```python
from typing import List
from collections import defaultdict


class Solution:
    def maxSubarraySum(self, nums: List[int]) -> int:
        n = len(nums)

        # Prefix sums P[0..n]
        P = [0] * (n + 1)
        s = 0
        for i in range(n):
            s += nums[i]
            P[i + 1] = s

        # Option: no deletion -> standard Kadane (also dominates any subarray
        # lying inside a single run, since that is a subarray of nums itself).
        ans = nums[0]
        cur = 0
        for v in nums:
            cur = v if cur < 0 else cur + v
            if cur > ans:
                ans = cur

        # Sparse tables for range min / range max over P (size n+1)
        m = n + 1
        mn_tab = [P]
        mx_tab = [P]
        j = 1
        while (1 << j) <= m:
            h = 1 << (j - 1)
            a = mn_tab[-1]
            mn_tab.append(list(map(min, a, a[h:])))
            b = mx_tab[-1]
            mx_tab.append(list(map(max, b, b[h:])))
            j += 1

        # Occurrence positions grouped by value
        pos = defaultdict(list)
        for i in range(n):
            pos[nums[i]].append(i)

        NEG = -(1 << 62)
        for ps in pos.values():
            if len(ps) == n:
                continue  # removing all occurrences would empty the array
            # G = best subarray sum (with x deleted) ending exactly at the
            # end of the previous non-empty run of non-x elements.
            G = NEG
            prev = -1
            ps.append(n)
            for p in ps:
                l = prev + 1
                r = p - 1  # run of non-x elements [l, r]
                prev = p
                if l > r:
                    continue
                k = (r - l + 1).bit_length() - 1
                kk = 1 << k
                # max prefix sum of the run: max P[t] for t in [l+1, r+1], minus P[l]
                t = mx_tab[k]
                a1 = t[l + 1]
                a2 = t[r + 2 - kk]
                maxp = (a1 if a1 > a2 else a2) - P[l]
                # max suffix sum of the run: P[r+1] - min P[t] for t in [l, r]
                t2 = mn_tab[k]
                b1 = t2[l]
                b2 = t2[r + 1 - kk]
                ms = P[r + 1] - (b1 if b1 < b2 else b2)
                if G == NEG:
                    G = ms
                else:
                    # spanning subarray: previous run(s) + a prefix of this run
                    cand = G + maxp
                    if cand > ans:
                        ans = cand
                    # extend best-ending through the whole run, or restart inside
                    g2 = G + P[r + 1] - P[l]
                    G = ms if ms > g2 else g2

        return ans
```
