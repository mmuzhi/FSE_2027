import sys
import numpy as np

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0]); X = int(data[1])
    A = np.array(data[2:2+n], dtype=np.int64)
    cnt = np.bincount(A, minlength=X+1)  # A ≤ X guaranteed; length ≥ X+1
    if len(cnt) < X+1: cnt = np.pad(...)  # bincount minlength handles
    f = np.minimum(cnt, 3).astype(np.int64)
    
    # convolution via FFT
    L = 1
    while L < 2*(X+1):
        L <<= 1
    fa = np.fft.rfft(f, L)
    conv = np.rint(np.fft.irfft(fa*fa, L)).astype(np.int64)
    # conv[s] for s in 0..2X
    
    # candidate c
    c_arr = np.arange(1, X-1, dtype=np.int64)  # 1..X-2
    present = f[1:X-1] > 0
    cs = c_arr[present]
    if cs.size == 0:
        print(-1); return
    s = X - cs
    t = X - 2*cs
    f_t = np.where(t >= 1, f[np.clip(t, 0, X)], 0)
    is3c = (t == cs)  # X == 3c
    Sgg = conv[s] - 2*f_t + is3c.astype(np.int64)
    seven = (s % 2 == 0)
    half = s // 2
    g_half = np.where(seven, f[np.clip(half,0,X)], 0) - is3c  # only when s==2c i.e. X==3c; note s==2c ⟺ X==3c ⟺ t==c
    # careful: when not seven, g_half = 0
    cond = (Sgg - g_half*g_half > 0) | (seven & (g_half >= 2))
    # also Sgg could be negative? shouldn't be, it's a true count; ok
    idx = np.nonzero(cond)[0]
    if idx.size == 0:
        print(-1); return
    c = int(cs[idx[0]])
    
    sval = X - c
    g = f.copy(); g[c] -= 1
    # find v in 1..sval-1
    vv = np.arange(1, sval, dtype=np.int64)
    gl = g[1:sval]          # g[v]
    gr = g[sval-1:0:-1]     # g[sval-v] for v=1..sval-1
    ok = (gl >= 1) & (gr >= 1) & ((2*vv != sval) | (gl >= 2))
    v = int(vv[np.nonzero(ok)[0][0]])
    w = sval - v
    
    # k = first occurrence of c
    pos_c = np.nonzero(A == c)[0]
    k = int(pos_c[0]) + 1
    def occ2(val):
        pos = np.nonzero(A == val)[0]
        if val == c:

def main():

    fa = np.fft.rfft(f, L)

Looking at this problem: N, X up to 10^6, values ≤ X. Brute force O(N²) is too slow. Key plan: work at the **value level**. A triple of distinct indices exists iff there are values (a,b,c) with a+b+c=X respecting multiplicities. 

**Approach**: Cap counts at 3, compute the self-convolution of the count array via FFT (numpy rfft, size 2^21, coefficients ≤ ~9·10^6, error ≪ 0.5 so rounding is exact).
