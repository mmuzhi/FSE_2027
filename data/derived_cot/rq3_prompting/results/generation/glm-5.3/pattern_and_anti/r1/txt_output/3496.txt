=== ANSWER ===
```python
from math import isqrt
from typing import List


class Solution:
    def minNumberOfSeconds(self, mountainHeight: int, workerTimes: List[int]) -> int:
        # Binary search on the answer T.
        # Feasible(T): sum over workers of max units reducible within T >= mountainHeight.
        # Worker with time w reduces x units in w*(1+2+...+x) = w*x*(x+1)/2 seconds.
        # Max x: x*(x+1) <= 2T//w  =>  (2x+1)^2 <= 4*q+1 with q = 2T//w
        # => x = (isqrt(4*q + 1) - 1) // 2

        def feasible(T: int) -> bool:
            total = 0
            for w in workerTimes:
                q = (2 * T) // w
                x = (isqrt(4 * q + 1) - 1) // 2
                total += x
                if total >= mountainHeight:
                    return True
            return False

        # Upper bound: the slowest worker alone reduces the whole mountain.
        lo, hi = 1, max(workerTimes) * mountainHeight * (mountainHeight + 1) // 2
        while lo < hi:
            mid = (lo + hi) // 2
            if feasible(mid):
                hi = mid
            else:
                lo = mid + 1
        return lo
```
