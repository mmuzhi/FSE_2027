    def maxRemoval(self, nums: List[int], queries: List[List[int]]) -> int:
        n = len(nums)
        queries.sort()

                r = -heapq.heappop(heap)

                    continue  # expired, discard
                cur += 1
                delta[r + 1] -= 1
                used += 1

import heapq
from typing import List

class Solution:
    def maxRemoval(self, nums: List[int], queries: List[List[int]]) -> int:
        n = len(nums)
        queries.sort()
        heap = []                # max-heap of right ends (negated)
        delta = [0] * (n + 1)    # difference array for coverage of selected queries
        cur = 0                  # current coverage at index i
        used = 0                 # number of queries selected
        j = 0
        for i in range(n):
            cur += delta[i]
            # add all queries that start at or before i
            while j < len(queries) and queries[j][0] <= i:
                heapq.heappush(heap, -queries[j][1])
                j += 1
            # greedily select queries to satisfy nums[i]
            while cur < nums[i]:
                if not heap:
                    return -1
                r = -heapq.heappop(heap)
                if r < i:
                    continue  # expired query, discard
                cur += 1
                delta[r + 1] -= 1
                used += 1
        return len(queries) - used
