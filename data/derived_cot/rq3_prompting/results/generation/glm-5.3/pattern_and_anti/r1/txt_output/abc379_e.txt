import sys

def main():
    data = sys.stdin.read().split()
    n = int(data[0])
    s = data[1]
    # c[k] for k = 1..n (1-indexed): d_k * k
    # A_p = sum_{k=1}^{n-p} c_k for p = 0..n-1; i.e., suffix sums over prefix...
    # A_p = prefix_c[n - p]
    # compute prefix sums
    total = 0
    pref = [0] * (n + 1)
    for idx in range(1, n + 1):
        total += (ord(s[idx-1]) - 48) * idx
        pref[idx] = total
    digits = []
    carry = 0
    for p in range(n):  # p = 0..n-1, coefficient A_p = pref[n - p]
        t = pref[n - p] + carry
        digits.append(t % 10)
        carry = t // 10
    while carry:
        digits.append(carry % 10)
        carry //= 10
    # digits are from least significant; reverse
    sys.stdout.write(''.join(map(str, digits[::-1])))

**Approach.** For digit at position k (1-indexed) with value d_k: it appears in substrings (i, j) with i ≤ k ≤ j, contributing d_k·10^{j−k}, and for each j there are k choices of i. So its total contribution is d_k·k·(1 + 10 + ... + 10^{N−k}) = d_k·k·R(N−k), where R(m) is a repunit of m+1 ones. Grouping by decimal position p: the coefficient at 10^p is A_p = Σ_{k=1}^{N−p} d_k·k = prefix sum of c_k = d_k·k. Since Σc_k ≤ 9·N²/2 ≈ 1.8×10^11 fits in a small int, the whole answer can be produced by simple carry propagation over positions — exact O(N) big-integer-free arithmetic.
