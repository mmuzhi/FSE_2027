```python
from typing import List

class Solution:
    def minCost(self, n: int, cost: List[List[int]]) -> int:
        # Pair up mirror houses (k, n-1-k). State (a, b) = colors of left/right house of a pair, a != b.
        # Transition between consecutive pairs k -> k+1: left colors differ AND right colors differ.
        # For target (a2, b2), best previous = min over rows a1 != a2 of:
        #   - if a1 == b2: row-min of a1 (both entries allowed, since b1 != a1 = b2 automatically)
        #   - else: dp[a1][third color] (only entry with b1 != a1, b1 != b2)
        first, last = cost[0], cost[n - 1]
        d01 = first[0] + last[1]
        d02 = first[0] + last[2]
        d10 = first[1] + last[0]
        d12 = first[1] + last[2]
        d20 = first[2] + last[0]
        d21 = first[2] + last[1]

        for k in range(1, n >> 1):
            ci = cost[k]
            cj = cost[n - 1 - k]
            r0 = d01 if d01 < d02 else d02  # row min for left color 0
            r1 = d10 if d10 < d12 else d12  # row min for left color 1
            r2 = d20 if d20 < d21 else d21  # row min for left color 2
            n01 = (r1 if r1 < d20 else d20) + ci[0] + cj[1]
            n02 = (d10 if d10 < r2 else r2) + ci[0] + cj[2]
            n10 = (r0 if r0 < d21 else d21) + ci[1] + cj[0]
            n12 = (d01 if d01 < r2 else r2) + ci[1] + cj[2]
            n20 = (r0 if r0 < d12 else d12) + ci[2] + cj[0]
            n21 = (d02 if d02 < r1 else r1) + ci[2] + cj[1]
            d01, d02, d10, d12, d20, d21 = n01, n02, n10, n12, n20, n21

        return min(d01, d02, d10, d12, d20, d21)
```
