```python
class Solution:
    def sortMatrix(self, grid: List[List[int]]) -> List[List[int]]:
        n = len(grid)
        for d in range(-(n - 1), n):
            cells = []
            i = max(0, d)
            j = max(0, -d)
            while i < n and j < n:
                cells.append(grid[i][j])
                i += 1
                j += 1
            if d >= 0:
                cells.sort(reverse=True)  # non-increasing
            else:
                cells.sort()  # non-decreasing
            i = max(0, d)
            j = max(0, -d)
            k = 0
            while i < n and j < n:
                grid[i][j] = cells[k]
                i += 1
                j += 1
                k += 1
        return grid
```

```python
class Solution:
    def sortMatrix(self, grid: List[List[int]]) -> List[List[int]]:
        n = len(grid)
        for d in range(-(n - 1), n):
            i, j = max(0, d), max(0, -d)
            cells = []
            while i < n and j < n:
                cells.append(grid[i][j])
                i += 1
                j += 1
            if d >= 0:
                cells.sort(reverse=True)  # bottom-left + middle: non-increasing
            else:
                cells.sort()              # top-right: non-decreasing
            i, j, k = max(0, d), max(0, -d), 0
            while i < n and j < n:
                grid[i][j] = cells[k]
                i += 1
                j += 1
                k += 1
        return grid
```
