```python
import sys
from math import isqrt
from bisect import bisect_right
from itertools import compress

def main():
    N = int(sys.stdin.buffer.read().split()[0])
    M = isqrt(N)

    limit = M
    sieve = bytearray(b'\x01') * (limit + 1)
    sieve[0] = 0
    sieve[1] = 0
    for i in range(2, isqrt(limit) + 1):
        if sieve[i]:
            sieve[i * i::i] = bytearray((limit - i * i) // i + 1)
    primes = list(compress(range(limit + 1), sieve))

    ans = 0

    # Case 1: n = p^8
    for p in primes:
        if p ** 8 <= N:
            ans += 1
        else:
            break

    # Case 2: n = p^2 * q^2, p < q  <=>  p*q <= M
    for p in primes:
        if p * p > M:
            break
        ans += bisect_right(primes, M // p) - bisect_right(primes, p)

    print(ans)

main()
```

```python
import sys
from math import isqrt
from bisect import bisect_right
from itertools import compress

def main():
    N = int(sys.stdin.buffer.read().split()[0])
    M = isqrt(N)

    limit = M
    sieve = bytearray(b'\x01') * (limit + 1)
    sieve[0] = 0
    sieve[1] = 0
    for i in range(2, isqrt(limit) + 1):
        if sieve[i]:
            sieve[i * i::i] = bytearray((limit - i * i) // i + 1)
    primes = list(compress(range(limit + 1), sieve))

    ans = 0

    # Case 1: n = p^8
    for p in primes:
        if p ** 8 <= N:
            ans += 1
        else:
            break

    # Case 2: n = p^2 * q^2 with p < q  <=>  p*q <= M
    for p in primes:
        if p * p > M:
            break
        ans += bisect_right(primes, M // p) - bisect_right(primes, p)

    print(ans)

main()
```
