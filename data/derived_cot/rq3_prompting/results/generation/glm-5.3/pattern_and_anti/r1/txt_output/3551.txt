import numpy as np

class Solution:
    def maximumSubarrayXor(self, nums, queries):
        n = len(nums)
        arr = np.array(nums, dtype=np.int64)
        # Build Fmat
        Fmat = np.empty((n, n), dtype=np.int64)
        # diag 0
        Fmat[np.arange(n), np.arange(n)] = arr
        d_arr = arr
        for d in range(1, n):
            d_arr = d_arr[:-1] ^ d_arr[1:]
            m = n - d
            idx = np.arange(m)
            Fmat[idx, idx + d] = d_arr
        
        # group queries by l
        from collections import defaultdict
        # simpler: bucket lists
        buckets = [[] for _ in range(n)]
        for qi, (l, r) in enumerate(queries):
            buckets[l].append((r, qi))
        
        ans = [0] * len(queries)
        dp = np.empty(n, dtype=np.int64)
        for l in range(n - 1, -1, -1):
            seg = Fmat[l, l:]
            if l == n - 1:
                dp[l] = seg[0]
            else:
                m = n - l
                h = np.empty(m, dtype=np.int64)
                h[0] = seg[0]
                np.maximum(dp[l+1:n], seg[1:], out=h[1:])
                np.maximum.accumulate(h, out=h)
                dp[l:n] = h
            for r, qi in buckets[l]:
                ans[qi] = int(dp[r])
        return ans

```python
import numpy as np
from typing import List

class Solution:
    def maximumSubarrayXor(self, nums: List[int], queries: List[List[int]]) -> List[int]:
        n = len(nums)
        arr = np.array(nums, dtype=np.int64)

        # f(i, j) = XOR score of nums[i..j]
        # f(i, j) = f(i, j-1) ^ f(i+1, j), f(i, i) = nums[i]
        # Computed diagonal by diagonal: diag_d[i] = f(i, i+d) = diag_{d-1}[i] ^ diag_{d-1}[i+1]
        Fmat = np.empty((n, n), dtype=np.int64)
        idx_all = np.arange(n)
        Fmat[idx_all, idx_all] = arr
        d_arr = arr
        for d in range(1, n):
            d_arr = np.bitwise_xor(d_arr[:-1], d_arr[1:])
            m = n - d
            Fmat[idx_all[:m], idx_all[d:]] = d_arr

        q = len(queries)
        # Group queries by their left endpoint l; process rows l from n-1 down to 0.
        buckets = [[] for _ in range(n)]
        for qi in range(q):
            l, r = queries[qi]
            buckets[l].append((r, qi))

        ans = [0] * q
        # dp[r] = best(l, r) = max f(i, j) over l <= i <= j <= r
        # best(l, r) = max(best(l+1, r), best(l, r-1), f(l, r))
        dp = np.empty(n, dtype=np.int64)
        for l in range(n - 1, -1, -1):
            if l == n - 1:
                dp[l] = Fmat[l, l]
            else:
                h = np.empty(n - l, dtype=np.int64)
                h[0] = Fmat[l, l]
                np.maximum(dp[l + 1:], Fmat[l, l + 1:], out=h[1:])
                dp[l:] = np.maximum.accumulate(h)
            for r, qi in buckets[l]:
                ans[qi] = int(dp[r])
        return ans
```
