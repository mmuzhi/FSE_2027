    public static double kappa(int[][] testData, int k) {

        for (int i = 0; i < k; i++) {
            P0 += dataMat[i][i] * 1.0;
        }
        int[] xsum = new int[k];
        int[] ysum = new int[k];
        int sum = 0;
        for (int i = 0; i < k; i++) {
            for (int j = 0; j < k; j++) {
                xsum[i] += dataMat[i][j];
                ysum[j] += dataMat[i][j];
                sum += dataMat[i][j];
            }
        }
        double Pe = 0.0;
        for (int i = 0; i < k; i++) {
            Pe += ysum[i] * xsum[i];
        }
        Pe = Pe / (sum * sum);
        P0 = P0 / sum;
        return (P0 - Pe) / (1 - Pe);
    }

    public static double fleissKappa(int[][] testData, int N, int k, int n) {
        int[][] dataMat = testData;
        double[] P = new double[N];
        double sum = 0.0;
        for (int i = 0; i < N; i++) {
            double temp = 0.0;
            for (int j = 0; j < k; j++) {
                sum += dataMat[i][j];
                temp += dataMat[i][j] * dataMat[i][j];

            temp -= n;
            temp /= (n - 1) * n;
            P[i] = temp;
