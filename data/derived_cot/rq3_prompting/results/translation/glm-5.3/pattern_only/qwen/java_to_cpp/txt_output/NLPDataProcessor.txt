static std::vector<std::string> split(const std::string& s, char delim) {
    std::vector<std::string> tokens;
    size_t start = 0;
    bool matched = false;
    for (size_t i = 0; i < s.size(); ++i) {
        if (s[i] == delim) {
            matched = true;
            tokens.push_back(s.substr(start, i - start));
            start = i + 1;
        }
    }
    if (!matched) {
        tokens.push_back(s); // single element, whole string (even if empty)
        return tokens;
    }
    tokens.push_back(s.substr(start)); // final segment
    while (!tokens.empty() && tokens.back().empty()) tokens.pop_back(); // strip trailing empties
    return tokens;
}
