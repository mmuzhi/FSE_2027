#include <vector>
#include <cmath>
#include <algorithm>
#include <stdexcept>
#include <numeric>

class DataStatistics2 {
private:
    std::vector<std::vector<double>> data_;
    bool is1D_;

    static double round2(double x) {
        // Matches numpy/Python round-half-to-even for 2 decimals
        return std::nearbyint(x * 100.0) / 100.0;
    }

    std::vector<double> flatten() const {
        std::vector<double> out;
        out.reserve(data_.size() * (data_.empty() ? 0 : data_[0].size()));
        for (const auto& row : data_)
            out.insert(out.end(), row.begin(), row.end());
        return out;
    }

public:
    // 1-D constructor (np.array(list))
    DataStatistics2(const std::vector<double>& data) : data_{data}, is1D_(true) {}

    // 2-D constructor (np.array(nested list))
    DataStatistics2(const std::vector<std::vector<double>>& data) : data_(data), is1D_(false) {}

    double get_sum() const {
        double s = 0.0;
        for (const auto& row : data_)
            for (double v : row) s += v;
        return s;
    }

    double get_min() const {
        double m = 0.0;
        bool first = true;
        for (const auto& row : data_)
            for (double v : row) {
                if (first) { m = v; first = false; }
                else m = std::min(m, v);
            }
        if (first) throw std::runtime_error("zero-size array to reduction operation minimum which has no identity");
        return m;
    }

    double get_max() const {
        double m = 0.0;
        bool first = true;
        for (const auto& row : data_)
            for (double v : row) {
                if (first) { m = v; first = false; }
                else m = std::max(m, v);
            }
        if (first) throw std::runtime_error("zero-size array to reduction operation maximum which has no identity");
        return m;
    }

    double get_variance() const {
        std::vector<double> flat = flatten();
        size_t n = flat.size();
        if (n == 0) throw std::runtime_error("variance of empty array");  // numpy: nan with warning
        double mean = std::accumulate(flat.begin(), flat.end(), 0.0) / n;
        double acc = 0.0;
        for (double v : flat) {
            double d = v - mean;
            acc += d * d;
        }
        return round2(acc / static_cast<double>(n));  // ddof=0 (population)
    }

    double get_std_deviation() const {
        std::vector<double> flat = flatten();
        size_t n = flat.size();
        if (n == 0) throw std::runtime_error("std of empty array");
        double mean = std::accumulate(flat.begin(), flat.end(), 0.0) / n;
        double acc = 0.0;
        for (double v : flat) {
            double d = v - mean;
            acc += d * d;
        }
        return round2(std::sqrt(acc / static_cast<double>(n)));
    }

    // np.corrcoef(data, rowvar=False): columns are variables; returns NxN matrix.
    std::vector<std::vector<double>> get_correlation() const {
        size_t rows = data_.size();
        size_t cols = rows ? data_[0].size() : 0;
        if (is1D_) cols = rows ? 1 : 0; // 1-D input -> single variable, 1x1 result
        // treat 1-D as one column of length rows
        size_t nvars = cols;
        size_t nobs = is1D_ ? rows : rows;

        // Extract columns
        std::vector<std::vector<double>> colsData(nvars, std::vector<double>(rows));
        if (is1D_) {
            for (size_t i = 0; i < rows; ++i) colsData[0][i] = data_[i][0];
        } else {
            for (size_t j = 0; j < nvars; ++j)
                for (size_t i = 0; i < rows; ++i)
                    colsData[j][i] = data_[i][j];
        }

        std::vector<std::vector<double>> corr(nvars, std::vector<double>(nvars, 0.0));
        for (size_t a = 0; a < nvars; ++a) {
            for (size_t b = a; b < nvars; ++b) {
                if (rows == 0) { corr[a][b] = corr[b][a] = std::nan(""); continue; }
                double ma = 0.0, mb = 0.0;
                for (size_t i = 0; i < rows; ++i) { ma += colsData[a][i]; mb += colsData[b][i]; }
                ma /= rows; mb /= rows;
                double cov = 0.0, va = 0.0, vb = 0.0;
                for (size_t i = 0; i < rows; ++i) {
                    double da = colsData[a][i] - ma;
                    double db = colsData[b][i] - mb;
                    cov += da * db; va += da * da; vb += db * db;
                }
                double denom = std::sqrt(va) * std::sqrt(vb);
                double r = (denom == 0.0) ? std::nan("") : (cov / denom);
                corr[a][b] = corr[b][a] = r;
            }
        }
        return corr;
    }
};

class DataStatistics2 {
private:
    std::vector<std::vector<double>> data_;  // rows of the (possibly 1-D) dataset
    bool is1D_;                              // tracks np.array(list) vs np.array(nested list)

    // Python round(x, 2) / np.round: round-half-to-even on 2 decimals
    static double round2(double x) {
        return std::nearbyint(x * 100.0) / 100.0;
    }

    std::vector<double> flatten() const {
        std::vector<double> out;
        for (const auto& row : data_)
            out.insert(out.end(), row.begin(), row.end());
        return out;
    }

    double variance_raw() const {
        std::vector<double> flat = flatten();
        size_t n = flat.size();
        if (n == 0) return std::nan("");  // np.var([]) -> nan (with warning)
        double mean = std::accumulate(flat.begin(), flat.end(), 0.0) /
                      static_cast<double>(n);
        double acc = 0.0;
        for (double v : flat) {
            double d = v - mean;
            acc += d * d;
        }
        return acc / static_cast<double>(n);  // ddof=0 (population variance)
    }

public:
    // 1-D input: np.array(data)
    DataStatistics2(const std::vector<double>& data)
        : data_{std::vector<std::vector<double>>{data}}, is1D_(true) {}

    DataStatistics2(const std::vector<std::vector<double>>& data)
        : data_(data), is1D_(false) {}

        double s = 0.0;  // np.sum([]) == 0.0
        for (const auto& row : data_)
            for (double v : row) s += v;
        return s;
    }

    double get_min() const {
        double m = 0.0;
        bool first = true;
        for (const auto& row : data_)
            for (double v : row) {
                if (first) { m = v; first = false; }
                else m = std::min(m, v);
            }
        if (first)
            throw std::runtime_error(
                "zero-size array to reduction operation which has no identity");
        return m;
    }

    double get_max() const {
        double m = 0.0;
        bool first = true;
        for (const auto& row : data_)
            for (double v : row) {
                if (first) { m = v; first = false; }
                else m = std::max(m, v);
            }
        if (first)

        } else {

                for (size_t i = 0; i < nobs; ++i)
                    vars[j][i] = data_[i][j];
        }

        std::vector<std::vector<double>> corr(nvars,
                                              std::vector<double>(nvars, 0.0));
        for (size_t a = 0; a < nvars; ++a) {

                    ma += vars[a][i];
                    mb += vars[b][i];

                ma /= static_cast<double>(nobs);
                mb /= static_cast<double>(nobs);
                double cov = 0.0, va = 0.0, vb = 0.0;
                for (size_t i = 0; i < nobs; ++i) {
                    double da = vars[a][i] - ma;
                    double db = vars[b][i] - mb;
                    cov += da * db;
                    va += da * da;
                    vb += db * db;

                double denom = std::sqrt(va) * std::sqrt(vb);
                // ddof cancels out in Pearson correlation
                double r = (denom == 0.0) ? std::nan("") : (cov / denom);
                corr[a][b] = corr[b][a] = r;
            }
        }
