    // bin(42423)[2:] == "1010010110110111"

            ? -static_cast<unsigned long long>(value)  // magnitude; safe for LLONG_MIN
            : static_cast<unsigned long long>(value);

        std::string result;

            n /= static_cast<unsigned long long>(base);

        if (negative) result.insert(result.begin(), prefix_char);
        return result;

    // ("0b"/"0o"/"0x", any case) and single underscores between digits are
    // accepted; anything else raises ValueError -> std::invalid_argument.
    static long long from_base(const std::string& s, int base, char prefix_char) {
        std::size_t i = 0, j = s.size();

            return c == ' ' || c == '\t' || c == '\n' ||
                   c == '\v' || c == '\f' || c == '\r';

            negative = (s[i] == '-');

            (s[i + 1] == prefix_char ||

            i += 2;

        bool any_digit = false;

        auto fail = [&]() {
            throw std::invalid_argument(
                "invalid literal for int() with base " + std::to_string(base) +
                ": '" + s + "'");
        };

        for (std::size_t k = i; k < j; ++k) {
            char c = s[k];
            if (c == '_') {
                if (!allow_underscore || k + 1 == j) fail();
                allow_underscore = false;
                continue;
            }
            int d;
            if (c >= '0' && c <= '9') d = c - '0';
            else if (c >= 'a' && c <= 'z') d = c - 'a' + 10;
            else if (c >= 'A' && c <= 'Z') d = c - 'A' + 10;
            else fail();
            if (d >= base) fail();
            any_digit = true;
            allow_underscore = true;
            acc = acc * static_cast<unsigned long long>(base) +
                  static_cast<unsigned long long>(d);
        }
        if (!any_digit) fail();

        if (negative) acc = (~acc) + 1ULL;  // two's-complement negate
        return static_cast<long long>(acc);
    }
