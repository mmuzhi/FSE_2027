namespace pyfmt {

// Formats a double the way Python's str()/repr() does:
// shortest decimal representation that round-trips back to the same
// double, using fixed notation for exponents in [-4, 16) and
// scientific notation otherwise.
inline std::string repr(double value) {
    if (std::isnan(value)) return "nan";
    if (std::isinf(value)) return (value < 0) ? "-inf" : "inf";

    char buf[64];
    int prec;
    for (prec = 0; prec <= 16; ++prec) {
        std::snprintf(buf, sizeof(buf), "%.*e", prec, value);
        if (std::strtod(buf, nullptr) == value) break;
    }

    std::string s(buf);
    std::size_t i = 0;
    bool negative = false;
    if (!s.empty() && s[0] == '-') {
        negative = true;
        i = 1;
    }

    std::string digits;
    digits += s[i++];  // first significant digit
    if (i < s.size() && s[i] == '.') {
        ++i;
        while (i < s.size() && s[i] != 'e') digits += s[i++];
    }
    ++i;  // skip 'e'
    int exp10 = std::atoi(s.c_str() + i);

    // Strip trailing zeros (defensive; shortest form has none).
    while (digits.size() > 1 && digits.back() == '0') digits.pop_back();

    const std::string sign = negative ? "-" : "";
    if (exp10 >= -4 && exp10 < 16) {
        if (exp10 >= 0) {
            if (static_cast<std::size_t>(exp10) + 1 >= digits.size()) {
                return sign + digits + std::string(exp10 + 1 - digits.size(), '0') + ".0";
            }
            return sign + digits.substr(0, exp10 + 1) + "." + digits.substr(exp10 + 1);
        }
        return sign + "0." + std::string(-exp10 - 1, '0') + digits;
    }

    std::string mantissa = digits.substr(0, 1);
    if (digits.size() > 1) mantissa += "." + digits.substr(1);
    char ebuf[8];
    std::snprintf(ebuf, sizeof(ebuf), "%+03d", exp10);
    return sign + mantissa + "e" + ebuf;
}

}  // namespace pyfmt

class VendingMachine {
public:
    struct ItemInfo {
        double price;
        int quantity;
    };

    // inventory is kept in insertion order, like a Python dict.
    std::vector<std::pair<std::string, ItemInfo>> inventory;
    double balance;

    VendingMachine() : balance(0) {}

    void add_item(const std::string& item_name, double price, int quantity) {
        if (!restock_item(item_name, quantity)) {
            inventory.emplace_back(item_name, ItemInfo{price, quantity});
        }
    }

    double insert_coin(double amount) {
        balance += amount;
        return balance;
    }

    // Returns the balance after the purchase, or no value (Python False)
    // when the item is missing, out of stock, or too expensive.
    std::optional<double> purchase_item(const std::string& item_name) {
        auto it = find(item_name);
        if (it != inventory.end()) {
            ItemInfo& item = it->second;
            if (item.quantity > 0 && balance >= item.price) {
                balance -= item.price;
                item.quantity -= 1;
                return balance;
            }
        }
        return std::nullopt;
    }

    bool restock_item(const std::string& item_name, int quantity) {
        auto it = find(item_name);
        if (it != inventory.end()) {
            it->second.quantity += quantity;
            return true;
        }
        return false;
    }

    // Returns the item listing, or no value (Python False) when empty.
    std::optional<std::string> display_items() const {
        if (inventory.empty()) {
            return std::nullopt;
        }
        std::string items;
        bool first = true;
        for (const auto& entry : inventory) {
            if (!first) items += "\n";
            first = false;
            items += entry.first + " - $" + pyfmt::repr(entry.second.price) +
                     " [" + std::to_string(entry.second.quantity) + "]";
        }
        return items;
    }

private:
    std::vector<std::pair<std::string, ItemInfo>>::iterator find(const std::string& item_name) {
        for (auto it = inventory.begin(); it != inventory.end(); ++it) {
            if (it->first == item_name) return it;
        }
        return inventory.end();
    }
    std::vector<std::pair<std::string, ItemInfo>>::const_iterator find(const std::string& item_name) const {
        for (auto it = inventory.begin(); it != inventory.end(); ++it) {
            if (it->first == item_name) return it;
        }
        return inventory.end();
    }
};

// the shortest decimal string that round-trips to the same double,
// in fixed notation when the decimal exponent is in [-4, 16) and in
// scientific notation otherwise ("1.25", "1.0", "1e+16", "1.5e-05", ...).
inline std::string repr(double value) {
    if (std::isnan(value)) return "nan";
    if (std::isinf(value)) return value < 0 ? "-inf" : "inf";

    // Shortest number of significant digits that round-trips.
    char buf[64];
    for (int prec = 0; prec <= 16; ++prec) {
        std::snprintf(buf, sizeof(buf), "%.*e", prec, value);
        if (std::strtod(buf, nullptr) == value) break;
    }

    // Parse "[-]d[.ddd]e[+-]XX" produced above.
    const std::string s(buf);
    std::size_t i = 0;
    bool negative = false;
    if (!s.empty() && s[0] == '-') {
        negative = true;
        i = 1;
    }
    std::string digits(1, s[i++]);
    if (i < s.size() && s[i] == '.') {
        ++i;
        while (i < s.size() && s[i] != 'e') digits += s[i++];
    }
    ++i;  // skip 'e'
    const int exp10 = std::atoi(s.c_str() + i);

    while (digits.size() > 1 && digits.back() == '0') digits.pop_back();

    const std::string sign = negative ? "-" : "";
    if (exp10 >= -4 && exp10 < 16) {
        if (exp10 >= 0) {
            if (static_cast<std::size_t>(exp10) + 1 >= digits.size()) {
                return sign + digits + std::string(static_cast<std::size_t>(exp10) + 1 - digits.size(), '0') + ".0";
            }
            return sign + digits.substr(0, static_cast<std::size_t>(exp10) + 1) + "." + digits.substr(static_cast<std::size_t>(exp10) + 1);
        }
        return sign + "0." + std::string(static_cast<std::size_t>(-exp10) - 1, '0') + digits;
    }

    std::string mantissa = digits.substr(0, 1);
    if (digits.size() > 1) mantissa += "." + digits.substr(1);
    char ebuf[8];
    std::snprintf(ebuf, sizeof(ebuf), "%+03d", exp10);
    return sign + mantissa + "e" + ebuf;
}

}  // namespace py_float_repr

// Simulates a vending machine: adding products, inserting coins,
// purchasing products, viewing the balance, replenishing stock and
// displaying product information.
class VendingMachine {
public:
    struct ItemInfo {
        double price;
        int quantity;
    };

    // Insertion-ordered inventory (Python dicts keep insertion order,
    // and display_items() iterates in that order).
    std::vector<std::pair<std::string, ItemInfo>> inventory;
    double balance;

    VendingMachine() : balance(0) {}

    // Adds a product to the inventory; if the product already exists,
    // only its quantity is replenished (the price is left unchanged).
    void add_item(const std::string& item_name, double price, int quantity) {
        if (!restock_item(item_name, quantity)) {
            inventory.emplace_back(item_name, ItemInfo{price, quantity});
        }
    }

    // Inserts coins and returns the resulting balance.
    double insert_coin(double amount) {
        balance += amount;
        return balance;
    }

    // Purchases a product. Returns the balance after the purchase, or
    // no value (the Python version's False) when the item is unknown,
    // out of stock, or too expensive.
    std::optional<double> purchase_item(const std::string& item_name) {
        auto it = std::find_if(inventory.begin(), inventory.end(),
                               [&](const std::pair<std::string, ItemInfo>& entry) {
                                   return entry.first == item_name;
                               });
        if (it != inventory.end()) {
            ItemInfo& item = it->second;
            if (item.quantity > 0 && balance >= item.price) {
                balance -= item.price;
                item.quantity -= 1;
                return balance;
            } else {
                return std::nullopt;
            }
        } else {
            return std::nullopt;
        }
    }

    // Replenishes the inventory of a product already present.
    bool restock_item(const std::string& item_name, int quantity) {
        auto it = std::find_if(inventory.begin(), inventory.end(),
                               [&](const std::pair<std::string, ItemInfo>& entry) {
                                   return entry.first == item_name;
                               });
        if (it != inventory.end()) {
            it->second.quantity += quantity;
            return true;
        } else {
            return false;
        }
    }

    // Displays the products. Returns the listing string, or no value
    // (the Python version's False) when the machine is empty.
    std::optional<std::string> display_items() const {
        if (inventory.empty()) {
            return std::nullopt;
        }
        std::string items;
        for (std::size_t idx = 0; idx < inventory.size(); ++idx) {
            if (idx > 0) items += '\n';
            const auto& entry = inventory[idx];
            items += entry.first + " - $" + py_float_repr::repr(entry.second.price) +
                     " [" + std::to_string(entry.second.quantity) + "]";
        }
        return items;
    }
};
