package org.example;

import java.lang.Math;

public class AreaCalculator {

    private double radius;

    public AreaCalculator(double radius) {
        this.radius = radius;
    }

    public double calculateCircleArea() {
        return Math.PI * Math.pow(this.radius, 2);
    }

    public double calculateSphereArea() {
        return 4 * Math.PI * Math.pow(this.radius, 2);
    }

    public double calculateCylinderArea(double height) {
        return 2 * Math.PI * this.radius * (this.radius + height);
    }

    public double calculateSectorArea(double angle) {
        return Math.pow(this.radius, 2) * angle / 2;
    }

    public double calculateAnnulusArea(double innerRadius, double outerRadius) {
        return Math.PI * (Math.pow(outerRadius, 2) - Math.pow(innerRadius, 2));
    }

    public static void main(String[] args) {
        AreaCalculator areaCalculator = new AreaCalculator(2);

        System.out.println("Circle Area: " + areaCalculator.calculateCircleArea());
        System.out.println("Sphere Area: " + areaCalculator.calculateSphereArea());
        System.out.println("Cylinder Area: " + areaCalculator.calculateCylinderArea(3));
        System.out.println("Sector Area: " + areaCalculator.calculateSectorArea(Math.PI));
        System.out.println("Annulus Area: " + areaCalculator.calculateAnnulusArea(2, 3));
    }
}

    if (std::isnan(d)) return "NaN";
    if (std::isinf(d)) return d < 0 ? "-Infinity" : "Infinity";
    if (d == 0.0) return std::signbit(d) ? "-0.0" : "0.0";

    bool negative = d < 0;
    double a = std::fabs(d);

    int precision;
    for (precision = 1; precision < 17; ++precision) {
        std::snprintf(buf, sizeof buf, "%.*e", precision, a);
        if (std::strtod(buf, nullptr) == a) break;
    }
    // precision 17 always round-trips
    if (precision == 17) std::snprintf(buf, sizeof buf, "%.17e", a);

    // Parse "d.ddddde±XX"
    // buf format: first char digit, '.', digits, 'e', sign, exponent digits
    std::string digits; // significant digits without decimal point
    int e10 = 0;
    {
        const char* p = buf;
        digits.push_back(*p++);
        if (*p == '.') {
            ++p;
            while (*p != 'e' && *p != '\0') digits.push_back(*p++);
        }
        if (*p == 'e') {
            ++p;
            e10 = std::atoi(p);
        }
    }
    // a = 0.digits × 10^(e10+1), or first digit has place value 10^e10.

    std::string result;
    if (negative) result.push_back('-');

    if (e10 >= -3 && e10 < 7) {
        // Plain decimal notation, Java style.
        // Place decimal point after the (e10+1)-th digit.
        if (e10 >= 0) {
            int intDigits = e10 + 1;
            if ((int)digits.size() <= intDigits) {
                // Pad with zeros, then ".0"
                std::string intPart = digits + std::string(intDigits - digits.size(), '0');
                result += intPart + ".0";
            } else {
                result += digits.substr(0, intDigits) + "." + digits.substr(intDigits);
            }
        } else {
            // e10 < 0: 0.000ddd
            result += "0." + std::string(-e10 - 1, '0') + digits;
        }
    } else {
        // Scientific notation: d.dddEn
        result += digits.substr(0, 1) + "." + digits.substr(1) + "E" + std::to_string(e10);
    }
    return result;
