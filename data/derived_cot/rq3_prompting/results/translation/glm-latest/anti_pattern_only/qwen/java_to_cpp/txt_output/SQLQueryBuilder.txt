import java.util.LinkedHashMap;
import java.util.Map;
import java.util.Set;

public class SQLQueryBuilder {

    public static String select(String table, String columns, Map<String, String> where) {
        if (columns == null) {
            columns = "*";
        }
        return select(table, columns.split(",\\s*"), where);
    }

    public static String select(String table, String[] columns, Map<String, String> where) {
        StringBuilder query = new StringBuilder("SELECT ");
        if (columns.length > 0) {
            query.append(String.join(", ", columns));
        } else {
            query.append("*");
        }
        query.append(" FROM ").append(table);

        if (where != null && !where.isEmpty()) {
            query.append(" WHERE ");
            Set<Map.Entry<String, String>> entrySet = where.entrySet();
            boolean first = true;
            for (Map.Entry<String, String> entry : entrySet) {
                if (!first) {
                    query.append(" AND ");
                }
                query.append(entry.getKey()).append("='").append(entry.getValue()).append("'");
                first = false;
            }
        }
        return query.toString();
    }

    public static String insert(String table, Map<String, String> data) {
        StringBuilder query = new StringBuilder("INSERT INTO ").append(table).append(" (");
        StringBuilder values = new StringBuilder(" VALUES (");

// Minimal insertion-ordered map mirroring java.util.LinkedHashMap<String, String>.
class LinkedHashMap {
public:
    using Entry = std::pair<std::string, std::string>;

    // Java Map.put: replaces the value of an existing key (keeping its original
    // insertion position) and returns the previous value; otherwise appends.
    std::string put(const std::string& key, const std::string& value) {
        for (Entry& e : entries_) {
            if (e.first == key) {
                std::string previous = e.second;
                e.second = value;
                return previous;
            }
        }
        entries_.emplace_back(key, value);
        return std::string();
    }

    // Java Map.get
    const std::string* get(const std::string& key) const {
        for (const Entry& e : entries_) {
            if (e.first == key) return &e.second;
        }
        return nullptr;
    }

    bool containsKey(const std::string& key) const {
        return get(key) != nullptr;
    }

    bool isEmpty() const { return entries_.empty(); }

    std::size_t size() const { return entries_.size(); }

    const std::vector<Entry>& entrySet() const { return entries_; }

private:
    std::vector<Entry> entries_;
};

class SQLQueryBuilder {
public:
    static std::string select(const std::string& table, const char* columns, const LinkedHashMap* where) {
        std::string cols = (columns != nullptr) ? columns : "*";
        return select(table, splitColumns(cols), where);
    }

    static std::string select(const std::string& table, const std::vector<std::string>& columns, const LinkedHashMap* where) {
        std::string query = "SELECT ";
        if (!columns.empty()) {
            query += join(columns, ", ");
        } else {
            query += "*";
        }
        query += " FROM ";
        query += table;

        appendWhere(query, where);
        return query;
    }

    static std::string insert(const std::string& table, const LinkedHashMap& data) {
        std::string query = "INSERT INTO ";
        query += table;
        query += " (";
        std::string values = " VALUES (";

        bool first = true;
        for (const LinkedHashMap::Entry& entry : data.entrySet()) {
            if (!first) {
                query += ", ";
                values += ", ";
            }
            query += entry.first;
            values += "'";
            values += entry.second;
            values += "'";
            first = false;
        }

        query += ")";
        values += ")";

        query += values;
        return query;
    }

    // NOTE: named 'delete_' because 'delete' is a reserved keyword in C++.
    static std::string delete_(const std::string& table, const LinkedHashMap* where) {
        std::string query = "DELETE FROM ";
        query += table;

        appendWhere(query, where);
        return query;
    }

    static std::string update(const std::string& table, const LinkedHashMap& data, const LinkedHashMap* where) {
        std::string query = "UPDATE ";
        query += table;
        query += " SET ";

        bool first = true;
        for (const LinkedHashMap::Entry& entry : data.entrySet()) {
            if (!first) {
                query += ", ";
            }
            query += entry.first;
            query += "='";
            query += entry.second;
            query += "'";
            first = false;
        }

        appendWhere(query, where);
        return query;
    }

private:
    // Replicates Java's String.split(",\\s*", 0): pieces separated by a comma
    // followed by optional ASCII whitespace; trailing empty pieces are dropped;
    // if no separator occurs, the whole string is returned as the only piece.
    static std::vector<std::string> splitColumns(const std::string& s) {
        static const std::string WHITESPACE = " \t\n\v\f\r"; // Java's \s
        std::vector<std::string> parts;
        bool foundSeparator = false;
        std::size_t index = 0;
        std::size_t i = 0;
        while (i < s.size()) {
            if (s[i] == ',') {
                foundSeparator = true;
                parts.push_back(s.substr(index, i - index));
                ++i;
                i = s.find_first_not_of(WHITESPACE, i);
                if (i == std::string::npos) {
                    i = s.size();
                }
                index = i;
            } else {
                ++i;
            }
        }
        if (!foundSeparator) {
            return std::vector<std::string>{s};
        }
        parts.push_back(s.substr(index));
        while (!parts.empty() && parts.back().empty()) {
            parts.pop_back();
        }
        return parts;
    }

    static std::string join(const std::vector<std::string>& parts, const std::string& delimiter) {
        std::string result;
        for (std::size_t i = 0; i < parts.size(); ++i) {
            if (i > 0) result += delimiter;
            result += parts[i];
        }
        return result;
    }

    static void appendWhere(std::string& query, const LinkedHashMap* where) {
        if (where != nullptr && !where->isEmpty()) {
            query += " WHERE ";
            bool first = true;
            for (const LinkedHashMap::Entry& entry : where->entrySet()) {
                if (!first) {
                    query += " AND ";
                }
                query += entry.first;
                query += "='";
                query += entry.second;
                query += "'";
                first = false;
            }
        }
    }
};

#include <cstddef>
#include <string>
#include <utility>

namespace org {
namespace example {

// Minimal stand-in for java.util.LinkedHashMap<String, String>: a map that
// keeps its entries in insertion order (a re-put of an existing key replaces
// the value in place, keeping the key's original position, like Java).
class LinkedHashMap {
public:
    using Entry = std::pair<std::string, std::string>;

    // Equivalent of Map.put: replaces an existing key's value (keeping its
    // original insertion position) and returns the previous value, or nullopt
    // if the key was absent.
    std::optional<std::string> put(const std::string& key, const std::string& value) {
        for (Entry& entry : entries_) {
            if (entry.first == key) {
                std::string previous = entry.second;
                entry.second = value;
                return previous;
            }
        }
        entries_.emplace_back(key, value);
        return std::nullopt;
    }

    // Equivalent of Map.get: pointer to the value, or nullptr if absent.
    const std::string* get(const std::string& key) const {
        for (const Entry& entry : entries_) {
            if (entry.first == key) {
                return &entry.second;
            }
        }
        return nullptr;
    }

    bool containsKey(const std::string& key) const {
        return get(key) != nullptr;
    }

    bool isEmpty() const {
        return entries_.empty();
    }

    std::size_t size() const {
        return entries_.size();
    }

    const std::vector<Entry>& entrySet() const {
        return entries_;
    }

private:
    std::vector<Entry> entries_;
};

class SQLQueryBuilder {
public:
    // Java: select(String table, String columns, Map<String, String> where)
    // 'columns' may be nullptr (Java null) and is then treated as "*".
    static std::string select(const std::string& table, const char* columns,
                              const LinkedHashMap* where) {
        const std::string effectiveColumns = (columns != nullptr) ? columns : "*";
        return select(table, splitColumns(effectiveColumns), where);
    }

    // Java: select(String table, String[] columns, Map<String, String> where)
    static std::string select(const std::string& table,
                              const std::vector<std::string>& columns,
                              const LinkedHashMap* where) {
        std::string query = "SELECT ";
        if (!columns.empty()) {
            query += join(columns, ", ");
        } else {
            query += "*";
        }
        query += " FROM ";
        query += table;

        appendWhereClause(query, where);
        return query;
    }

    // Java: insert(String table, Map<String, String> data)
    static std::string insert(const std::string& table, const LinkedHashMap& data) {
        std::string query = "INSERT INTO ";
        query += table;
        query += " (";
        std::string values = " VALUES (";

        bool first = true;
        for (const LinkedHashMap::Entry& entry : data.entrySet()) {
            if (!first) {
                query += ", ";
                values += ", ";
            }
            query += entry.first;
            values += "'";
            values += entry.second;
            values += "'";
            first = false;
        }

        query += ")";
        values += ")";

        query += values;
        return query;
    }

    // Java: delete(String table, Map<String, String> where)
    // Renamed to delete_ because 'delete' is a reserved keyword in C++.
    static std::string delete_(const std::string& table, const LinkedHashMap* where) {
        std::string query = "DELETE FROM ";
        query += table;

        appendWhereClause(query, where);
        return query;
    }

    // Java: update(String table, Map<String, String> data, Map<String, String> where)
    static std::string update(const std::string& table, const LinkedHashMap& data,
                              const LinkedHashMap* where) {
        std::string query = "UPDATE ";
        query += table;
        query += " SET ";

        bool first = true;
        for (const LinkedHashMap::Entry& entry : data.entrySet()) {
            if (!first) {
                query += ", ";
            }
            query += entry.first;
            query += "='";
            query += entry.second;
            query += "'";
            first = false;
        }

        appendWhereClause(query, where);
        return query;
    }

private:
    // Equivalent of Java's columns.split(",\\s*") (limit 0): splits on a comma
    // followed by any run of ASCII whitespace ('\s' in Java), drops trailing
    // empty pieces, and returns the whole string as a single piece when it
    // contains no separator at all.
    static std::vector<std::string> splitColumns(const std::string& columns) {
        static const char* const JAVA_WHITESPACE = " \t\n\v\f\r";

        std::vector<std::string> pieces;
        bool separatorFound = false;
        std::size_t index = 0;
        std::size_t i = 0;
        while (i < columns.size()) {
            if (columns[i] == ',') {
                separatorFound = true;
                pieces.push_back(columns.substr(index, i - index));
                ++i;
                std::size_t next = columns.find_first_not_of(JAVA_WHITESPACE, i);
                i = (next == std::string::npos) ? columns.size() : next;
                index = i;
            } else {
                ++i;
            }
        }
        if (!separatorFound) {
            return std::vector<std::string>(1, columns);
        }
        pieces.push_back(columns.substr(index));
        while (!pieces.empty() && pieces.back().empty()) {
            pieces.pop_back();
        }
        return pieces;
    }

    // Equivalent of String.join(delimiter, parts).
    static std::string join(const std::vector<std::string>& parts,
                            const std::string& delimiter) {
        std::string result;
        for (std::size_t i = 0; i < parts.size(); ++i) {
            if (i > 0) {
                result += delimiter;
            }
            result += parts[i];
        }
        return result;
    }

    // Shared " WHERE k='v' AND ..." construction (Java: where != null && !where.isEmpty()).
    static void appendWhereClause(std::string& query, const LinkedHashMap* where) {
        if (where != nullptr && !where->isEmpty()) {
            query += " WHERE ";
            bool first = true;
            for (const LinkedHashMap::Entry& entry : where->entrySet()) {
                if (!first) {
                    query += " AND ";
                }
                query += entry.first;
                query += "='";
                query += entry.second;
                query += "'";
                first = false;
            }
        }
    }
};

}  // namespace example
