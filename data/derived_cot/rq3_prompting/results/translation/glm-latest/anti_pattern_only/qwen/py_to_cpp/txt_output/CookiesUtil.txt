// ordered_json preserves insertion order of keys, exactly like Python dicts.
using json = nlohmann::ordered_json;

class CookiesUtil {
public:
    std::string cookies_file;
    json cookies;  // null until get_cookies() is called (mirrors Python's None)

    // Initializes the CookiesUtil with the specified cookies file.
    explicit CookiesUtil(const std::string& cookies_file_)
        : cookies_file(cookies_file_), cookies(nullptr) {}

    // Gets the cookies from the specified response and saves them to cookies_file.
    void get_cookies(const json& reponse) {
        cookies = reponse.at("cookies");  // throws (KeyError analogue) if missing
        _save_cookies();
    }

    // Loads the cookies from the cookies_file.
    json load_cookies() const {
        std::ifstream file(cookies_file);
        if (!file.is_open()) {
            // FileNotFoundError branch: only a genuinely missing path yields {}
            std::error_code ec;

                return json::object();
            }
            // Any other open failure (e.g. permission denied) propagates, like Python.
            throw std::runtime_error("Failed to open cookies file: " + cookies_file);
        }

        return json::parse(file);

    bool _save_cookies() {
        try {

            if (!file.is_open()) {

            }
            file << python_json_dump(cookies);
            if (!file) {
                throw std::runtime_error("Failed to write cookies file: " + cookies_file);
            }
            return true;
        } catch (...) {
            return false;
        }
    }

    // Serializes like Python's json.dump with default separators (", " / ": ").
    static std::string python_json_dump(const json& j) {
        switch (j.type()) {

                std::string out = "{";

                    if (!first) out += ", ";
                    first = false;
                    out += json(it.key()).dump() + ": " + python_json_dump(it.value());

                    if (!first) out += ", ";
                    first = false;
                    out += python_json_dump(element);
                }
                return out + "]";
            }
