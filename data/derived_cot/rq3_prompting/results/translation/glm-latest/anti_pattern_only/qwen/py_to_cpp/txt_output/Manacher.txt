#include <stdexcept>
#include <string>
#include <utility>

// This class implements a (simplified) Manacher-style algorithm to find the
// longest palindromic substring in a given string.
class Manacher {
public:
    // Initializes the Manacher class with the given input_string.
    explicit Manacher(std::string input_string)
        : input_string_(std::move(input_string)) {}

    // Recursively calculates the length of the palindromic substring based on
    // a given center, difference value, and input string.
    int palindromic_length(int center, int diff, const std::string& str) const {
        if (center - diff == -1
                || center + diff == static_cast<int>(str.length())
                || python_index(str, center - diff) != python_index(str, center + diff)) {
            return 0;
        }
        return 1 + palindromic_length(center, diff + 1, str);
    }

    // Finds the longest palindromic substring in the given string.
    std::string palindromic_string() const {
        int max_length = 0;

        std::string new_input_string;
        std::string output_string;

        // Interleave '|' between all characters except the last one:
        //   for i in input_string[:len(input_string) - 1]: new_input_string += i + "|"
        for (std::size_t i = 0; i + 1 < input_string_.size(); ++i) {
            new_input_string += input_string_[i];
            new_input_string += '|';
        }
        // new_input_string += input_string[-1]  (IndexError on empty input)
        if (input_string_.empty()) {
            throw std::out_of_range("string index out of range");
        }
        new_input_string += input_string_.back();

        int start = 0;
        bool start_assigned = false;

        for (int i = 0; i < static_cast<int>(new_input_string.size()); ++i) {
            int length = palindromic_length(i, 1, new_input_string);

            if (max_length < length) {
                max_length = length;
                start = i;
                start_assigned = true;
            }
        }

        // Python raises UnboundLocalError here when 'start' was never assigned.
        if (!start_assigned) {
            throw std::runtime_error(
                "local variable 'start' referenced before assignment");
        }

        for (int i = start - max_length; i <= start + max_length; ++i) {
            if (new_input_string[i] != '|') {
                output_string += new_input_string[i];
            }
        }

        return output_string;
    }

private:
    std::string input_string_;

    // Mimics Python's string indexing: negative indices wrap around,
    // out-of-range access raises (IndexError ~ std::out_of_range).
    static char python_index(const std::string& str, int index) {
        const int size = static_cast<int>(str.size());
        if (index < 0) {
            index += size;
        }
        if (index < 0 || index >= size) {
            throw std::out_of_range("string index out of range");
        }
        return str[static_cast<std::size_t>(index)];
    }
};

#include <stdexcept>
#include <string>
#include <utility>

/*
 * This class implements a manacher algorithm to find the Longest palindromic
 * substring in a given string.
 */
class Manacher {
public:
    /*
     * Initializes the Manacher class with the given input_string.
     * :param input_string: The input_string to be searched, str.
     */
    explicit Manacher(std::string input_string)
        : input_string_(std::move(input_string)) {}

    /*
     * Recursively calculates the length of the palindromic substring based on
     * a given center, difference value, and input string.
     * :param center: The center of the palindromic substring, int.
     * :param diff: The difference between the center and the current position, int.
     * :param str: The string to be searched, str.
     * :return: The length of the palindromic substring, int.
     */
    int palindromic_length(int center, int diff, const std::string& str) const {
        if (center - diff == -1
                || center + diff == static_cast<int>(str.length())
                || python_index(str, center - diff) != python_index(str, center + diff)) {
            return 0;
        }
        return 1 + palindromic_length(center, diff + 1, str);
    }

    /*
     * Finds the longest palindromic substring in the given string.
     * :return: The longest palindromic substring, str.
     */
    std::string palindromic_string() const {
        int max_length = 0;

        std::string new_input_string;
        std::string output_string;

        // new_input_string = "|".join-style interleaving:
        //   for i in input_string[:len(input_string) - 1]: new_input_string += i + "|"
        //   new_input_string += input_string[-1]
        const int n = static_cast<int>(input_string_.size());
        for (int i = 0; i < n - 1; ++i) {
            new_input_string += input_string_[i];
            new_input_string += '|';
        }
        if (input_string_.empty()) {
            // Python: self.input_string[-1] raises IndexError on an empty string.
            throw std::out_of_range("string index out of range");
        }
        new_input_string += input_string_.back();

        int start = 0;
        bool start_assigned = false;

        for (int i = 0; i < static_cast<int>(new_input_string.size()); ++i) {
            int length = palindromic_length(i, 1, new_input_string);

            if (max_length < length) {
                max_length = length;
                start = i;
                start_assigned = true;
            }
        }

        if (!start_assigned) {
            // Python: 'start' is only bound inside the loop above; referencing it
            // here raises UnboundLocalError when no palindrome was found.
            throw std::runtime_error(
                "local variable 'start' referenced before assignment");
        }

        for (int i = start - max_length; i <= start + max_length; ++i) {
            if (new_input_string[i] != '|') {
                output_string += new_input_string[i];
            }
        }

        return output_string;
    }

private:
    std::string input_string_;

    // Mimics Python's str indexing: negative indices count from the end and
    // out-of-range access raises IndexError (mapped to std::out_of_range).
    static char python_index(const std::string& str, int index) {
        const int size = static_cast<int>(str.size());
        if (index < 0) {
            index += size;
        }
        if (index < 0 || index >= size) {
            throw std::out_of_range("string index out of range");
        }
        return str[static_cast<std::size_t>(index)];
    }
};
