class AvgPartition {
private:
    std::vector<int> lst;
    int limit;

    // Python-style floor division (raises like ZeroDivisionError on b == 0).
    static int floorDiv(int a, int b) {
        if (b == 0) {
            throw std::runtime_error("integer division or modulo by zero");
        }
        int q = a / b;
        int r = a % b;
        if (r != 0 && ((a < 0) != (b < 0))) {
            --q;
        }
        return q;
    }

    // Python-style modulo (result takes the sign of the divisor).
    static int floorMod(int a, int b) {
        return a - floorDiv(a, b) * b;
    }

    // Python-style lst[start:end] slicing (step 1), with negative-index
    // handling and clamping to the container bounds.
    std::vector<int> slice(int start, int end) const {
        int n = static_cast<int>(lst.size());
        if (start < 0) {
            start += n;
            if (start < 0) start = 0;
        } else if (start > n) {
            start = n;
        }
        if (end < 0) {
            end += n;
            if (end < 0) end = 0;
        } else if (end > n) {
            end = n;
        }
        if (start >= end) {
            return {};
        }
        return std::vector<int>(lst.begin() + start, lst.begin() + end);
    }

public:

        : lst(lst), limit(limit) {}
