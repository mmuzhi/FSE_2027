struct BigNum {
    std::vector<int> d; // little-endian decimal digits, empty == 0
    BigNum() {}
    BigNum(unsigned long long v) { if (v == 0) return; while (v) { d.push_back(v % 10); v /= 10; } }
    void mul_add(unsigned long long m, unsigned long long a) {
        // this = this * m + a
        unsigned long long carry = a;
        for (size_t i = 0; i < d.size(); ++i) {
            unsigned long long cur = (unsigned long long)d[i] * m + carry;
            d[i] = (int)(cur % 10);
            carry = cur / 10;
        }
        while (carry) { d.push_back((int)(carry % 10)); carry /= 10; }
    }
    void add(const BigNum& o) {
        int carry = 0;
        size_t n = std::max(d.size(), o.d.size());
        d.resize(n, 0);
        for (size_t i = 0; i < n; ++i) {
            int s = d[i] + (i < o.d.size() ? o.d[i] : 0) + carry;
            d[i] = s % 10;
            carry = s / 10;
        }
        if (carry) d.push_back(carry);
    }
    std::string str() const {
        if (d.empty()) return "0";
        std::string s;
        for (size_t i = d.size(); i-- > 0; ) s += char('0' + d[i]);
        return s;
    }
};

namespace {

// Minimal arbitrary-precision unsigned integer (enough for the arithmetic
// performed by Words2Numbers: multiply by a small scale, add a small value,
// add another big value, and convert to decimal string).
struct BigNum {
    std::vector<int> digits;  // little-endian decimal digits; empty == 0

    BigNum() {}

    void mul_add(unsigned long long multiplier, unsigned long long addend) {
        unsigned long long carry = addend;
        for (std::size_t i = 0; i < digits.size(); ++i) {
            unsigned long long cur = static_cast<unsigned long long>(digits[i]) * multiplier + carry;
            digits[i] = static_cast<int>(cur % 10);
            carry = cur / 10;
        }
        while (carry) {
            digits.push_back(static_cast<int>(carry % 10));
            carry /= 10;
        }
    }

    BigNum plus(const BigNum& other) const {
        BigNum r;
        r.digits = digits;
        int carry = 0;
        std::size_t n = r.digits.size() > other.digits.size() ? r.digits.size() : other.digits.size();
        r.digits.resize(n, 0);
        for (std::size_t i = 0; i < n; ++i) {
            int s = r.digits[i] + (i < other.digits.size() ? other.digits[i] : 0) + carry;
            r.digits[i] = s % 10;
            carry = s / 10;
        }
        if (carry) r.digits.push_back(carry);
        return r;
    }

    std::string str() const {
        if (digits.empty()) return "0";
        std::string s;
        s.reserve(digits.size());
        for (std::size_t i = digits.size(); i-- > 0;)
            s.push_back(static_cast<char>('0' + digits[i]));
        return s;
    }
};

// Minimal arbitrary-precision unsigned integer supporting exactly the
// operations needed here: value = value * small + small, value = value + value,
// and conversion to a decimal string. This mirrors Python's unbounded ints.
class BigNum {
public:
    BigNum() {}

    // this = this * multiplier + addend
    void mul_add(unsigned long long multiplier, unsigned long long addend) {
        unsigned long long carry = addend;
        for (std::size_t i = 0; i < digits_.size(); ++i) {
            unsigned long long cur =
                static_cast<unsigned long long>(digits_[i]) * multiplier + carry;
            digits_[i] = static_cast<int>(cur % 10);
            carry = cur / 10;
        }
        while (carry > 0) {
            digits_.push_back(static_cast<int>(carry % 10));
            carry /= 10;
        }
    }

    // returns this + other
    BigNum plus(const BigNum& other) const {
        BigNum r;
        r.digits_ = digits_;
        if (r.digits_.size() < other.digits_.size())
            r.digits_.resize(other.digits_.size(), 0);
        int carry = 0;
        for (std::size_t i = 0; i < r.digits_.size(); ++i) {
            int sum = r.digits_[i] + carry;
            if (i < other.digits_.size())
                sum += other.digits_[i];
            r.digits_[i] = sum % 10;
            carry = sum / 10;
        }
        if (carry > 0)
            r.digits_.push_back(carry);
        return r;
    }

    std::string str() const {
        if (digits_.empty())
            return "0";
        std::string s;
        s.reserve(digits_.size());
        for (std::size_t i = digits_.size(); i-- > 0;)
            s.push_back(static_cast<char>('0' + digits_[i]));
        return s;
    }

private:
    std::vector<int> digits_;  // little-endian decimal digits; empty means 0
};

std::vector<std::string> split_words(const std::string& s) {
    // Equivalent of Python's str.split(): splits on runs of whitespace,
    // producing no empty tokens.
    std::vector<std::string> words;
    std::istringstream iss(s);
    std::string w;
    while (iss >> w)
        words.push_back(w);
    return words;
}

bool ends_with(const std::string& word, const std::string& ending) {
    return word.size() >= ending.size() &&
           word.compare(word.size() - ending.size(), ending.size(), ending) == 0;
}

std::string replace_hyphens_with_spaces(const std::string& s) {
    std::string t = s;
    for (std::size_t i = 0; i < t.size(); ++i) {
        if (t[i] == '-')
            t[i] = ' ';
    }
    return t;
}
