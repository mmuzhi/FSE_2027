    // Returns: the y-coordinates of the interpolation points.

                                              const std::vector<double>& y,
                                              const std::vector<double>& x_interp) {

            for (std::size_t i = 0; i + 1 < x.size(); ++i) {

                        throw std::runtime_error("ZeroDivisionError: division by zero");
