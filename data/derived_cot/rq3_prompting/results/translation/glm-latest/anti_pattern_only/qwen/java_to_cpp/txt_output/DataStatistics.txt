```java
package org.example;

import java.util.*;
import java.util.stream.Collectors;

public class DataStatistics {

    public double mean(int[] data) {
        double sum = Arrays.stream(data).sum();
        return Math.round((sum / data.length) * 100.0) / 100.0;
    }

    public double median(int[] data) {
        int[] sortedData = Arrays.copyOf(data, data.length);
        Arrays.sort(sortedData);
        int n = sortedData.length;

        if (n % 2 == 0) {
            int middle = n / 2;
            return Math.round(((sortedData[middle - 1] + sortedData[middle]) / 2.0) * 100.0) / 100.0;
        } else {
            int middle = n / 2;
            return sortedData[middle];
        }
    }

    public List<Integer> mode(int[] data) {
        Map<Integer, Long> frequencyMap = Arrays.stream(data)
                .boxed()
                .collect(Collectors.groupingBy(e -> e, Collectors.counting()));

        long maxCount = Collections.max(frequencyMap.values());

        return frequencyMap.entrySet()
                .stream()
                .filter(entry -> entry.getValue() == maxCount)
                .map(Map.Entry::getKey)
                .sorted()
                .collect(Collectors.toList());
```
```cpp
    }
```
```cpp
}
```

```cpp
private:
    // Replicates java.lang.Math.round(double): the closest long to the
    // argument, with ties rounding toward positive infinity.
    static long javaMathRound(double a) {
        const std::uint64_t EXP_BIT_MASK    = 0x7FF0'0000'0000'0000ULL;
        const std::uint64_t SIGNIF_BIT_MASK = 0x000F'FFFF'FFFF'FFFFULL;
        const int SIGNIFICAND_WIDTH = 53; // includes the implicit bit
        const int EXP_BIAS = 1023;

        std::uint64_t longBits;
        std::memcpy(&longBits, &a, sizeof(longBits));

        long biasedExp = static_cast<long>((longBits & EXP_BIT_MASK) >> (SIGNIFICAND_WIDTH - 1));
        long shift = static_cast<long>(SIGNIFICAND_WIDTH - 2 + EXP_BIAS) - biasedExp;
        if ((shift & -64) == 0) { // 0 <= shift < 64
            // a is finite with 2^-64 <= ulp(a) < 1
            long r = static_cast<long>((longBits & SIGNIF_BIT_MASK) | (SIGNIF_BIT_MASK + 1));
            if ((longBits >> 63) != 0) {
                r = -r;
            }
            // (r) is a / ulp(a); ((r >> shift) + 1) >> 1 is floor(a + 1/2)
            return ((r >> shift) + 1) >> 1;
        }
        // |a| < 2^-64 (rounds to 0), or a is a mathematical integer,
        // or a is infinite/NaN — fall back to Java's double->long cast.
        return javaDoubleToLong(a);
    }

    // Replicates Java's (long) double cast: NaN -> 0, saturation at Long bounds.
    static long javaDoubleToLong(double a) {
        if (std::isnan(a)) return 0;
        if (a >= 9223372036854775808.0) return 9223372036854775807L; //  2^63
        if (a <= -9223372036854775808.0) return -9223372036854775807L - 1; // -2^63
        return static_cast<long>(a);
    }
```

        constexpr std::uint64_t kSignifBitMask = 0x000FFFFFFFFFFFFFULL;
        constexpr int kSignificandWidth = 53;  // includes the implicit leading bit

        std::uint64_t longBits = 0;
