import numpy as np
from gensim import matutils
from numpy import dot, array


class VectorUtil:
    @staticmethod
    def similarity(vector_1, vector_2):
        return dot(matutils.unitvec(vector_1), matutils.unitvec(vector_2))

    @staticmethod
    def cosine_similarities(vector_1, vectors_all):
        norm = np.linalg.norm(vector_1)
        all_norms = np.linalg.norm(vectors_all, axis=1)
        dot_products = dot(vectors_all, vector_1)
        similarities = dot_products / (norm * all_norms)
        return similarities

    @staticmethod
    def n_similarity(vector_list_1, vector_list_2):
        if not (len(vector_list_1) and len(vector_list_2)):
            raise ZeroDivisionError('At least one of the passed list is empty.')

        return dot(matutils.unitvec(array(vector_list_1).mean(axis=0)),
                   matutils.unitvec(array(vector_list_2).mean(axis=0)))

    @staticmethod
    def compute_idf_weight_dict(total_num, number_dict):
        index_2_key_map = {}

        index = 0

        count_list = []
        for key, count in number_dict.items():
            index_2_key_map[index] = key
            count_list.append(count)
            index = index + 1

        a = np.array(count_list)
        ## smooth, in case the divide by zero error
        a = np.log((total_num + 1) / (a + 1))
        result = {}

        for index, w in enumerate(a):
            key = index_2_key_map[index]
            result[key] = w

        return result

#include <cmath>
#include <cstddef>
#include <map>
#include <stdexcept>
#include <string>
#include <vector>

class VectorUtil {
public:
    /// Compute the cosine similarity between one vector and another vector.
    static double similarity(const std::vector<double>& vector_1,
                             const std::vector<double>& vector_2) {
        return dot(unitvec(vector_1), unitvec(vector_2));
    }

    /// Compute cosine similarities between one vector and a set of other vectors.
    static std::vector<double> cosine_similarities(const std::vector<double>& vector_1,
                                                   const std::vector<std::vector<double>>& vectors_all) {
        const double norm_1 = norm(vector_1);
        std::vector<double> similarities;
        similarities.reserve(vectors_all.size());
        for (const std::vector<double>& row : vectors_all) {
            const double row_norm = norm(row);
            const double dot_product = dot(row, vector_1);
            similarities.push_back(dot_product / (norm_1 * row_norm));
        }
        return similarities;
    }

    static double n_similarity(const std::vector<std::vector<double>>& vector_list_1,
                               const std::vector<std::vector<double>>& vector_list_2) {
        if (vector_list_1.empty() || vector_list_2.empty()) {
            throw std::runtime_error("At least one of the passed list is empty.");
        }
        return dot(unitvec(mean(vector_list_1)), unitvec(mean(vector_list_2)));
    }

    static std::map<std::string, double> compute_idf_weight_dict(
            int total_num, const std::map<std::string, double>& number_dict) {

        for (const auto& entry : number_dict) {
            result[entry.first] = std::log((static_cast<double>(total_num) + 1.0) / (entry.second + 1.0));
        }
        return result;

    static std::vector<double> unitvec(const std::vector<double>& vec) {
        const double veclen = norm(vec);
        if (veclen > 0.0) {
            const double scale = 1.0 / veclen;
            std::vector<double> unit;
            unit.reserve(vec.size());
            for (const double x : vec) {
                unit.push_back(x * scale);
            }
            return unit;
        }
        return vec;
    }

    /// Dot product of two vectors of equal length.
    static double dot(const std::vector<double>& a, const std::vector<double>& b) {
        if (a.size() != b.size()) {
            throw std::invalid_argument("shapes not aligned for dot product");
        }
        double result = 0.0;
        for (std::size_t i = 0; i < a.size(); ++i) {
            result += a[i] * b[i];
        }
        return result;
    }

    /// Euclidean (l2) norm.
    static double norm(const std::vector<double>& vec) {
        double sum_of_squares = 0.0;
        for (const double x : vec) {
            sum_of_squares += x * x;
        }
        return std::sqrt(sum_of_squares);
    }

            for (std::size_t i = 0; i < dim; ++i) {

            }

        for (std::size_t i = 0; i < dim; ++i) {
            mean_vec[i] /= count;
        }
        return mean_vec;

#include <cmath>

/**
 * Provides vector operations, including calculating similarity, cosine
 * similarities, average similarity, and IDF weights.
 */
class VectorUtil {
public:
    /**
     * Compute the cosine similarity between one vector and another vector.
     *
     * Mirrors: dot(matutils.unitvec(vector_1), matutils.unitvec(vector_2))
     */
    static double similarity(const std::vector<double>& vector_1,
                             const std::vector<double>& vector_2) {
        return dot(unitvec(vector_1), unitvec(vector_2));
    }

    /**
     * Compute cosine similarities between one vector and a set of other vectors.
     *
     * Mirrors: matutils.cosine_similarities(vector_1, vectors_all)
     */
    static std::vector<double> cosine_similarities(
            const std::vector<double>& vector_1,
            const std::vector<std::vector<double>>& vectors_all) {
        const double norm = l2_norm(vector_1);
        std::vector<double> similarities;
        similarities.reserve(vectors_all.size());
        for (const std::vector<double>& row : vectors_all) {
            const double row_norm = l2_norm(row);
            const double dot_product = dot(row, vector_1);
            // Division by zero yields inf/nan, matching numpy semantics.
            similarities.push_back(dot_product / (norm * row_norm));
        }
        return similarities;
    }

    /**
     * Compute cosine similarity between two sets of vectors.
     */
    static double n_similarity(const std::vector<std::vector<double>>& vector_list_1,
                               const std::vector<std::vector<double>>& vector_list_2) {
        if (vector_list_1.empty() || vector_list_2.empty()) {
            throw std::runtime_error("At least one of the passed list is empty.");
        }
        return dot(unitvec(mean_axis0(vector_list_1)),
                   unitvec(mean_axis0(vector_list_2)));
    }

    /**
     * Calculate log((total_num + 1) / (count + 1)) for each count in number_dict.
     */
    static std::map<std::string, double> compute_idf_weight_dict(
            int total_num, const std::map<std::string, double>& number_dict) {
        std::map<std::string, double> result;
        for (const auto& entry : number_dict) {
            // The +1 smoothing avoids divide-by-zero / log(0) issues.
            result[entry.first] =
                std::log((static_cast<double>(total_num) + 1.0) / (entry.second + 1.0));
        }
        return result;
    }

private:
    // gensim.matutils.unitvec with the default l2 norm: multiply by 1/||vec||;
    // a zero-length (or zero) vector is returned unchanged.
    static std::vector<double> unitvec(const std::vector<double>& vec) {
        const double veclen = l2_norm(vec);
        if (veclen > 0.0) {
            const double scale = 1.0 / veclen;
            std::vector<double> unit;
            unit.reserve(vec.size());
            for (const double x : vec) {
                unit.push_back(x * scale);
            }
            return unit;
        }
        return vec;
    }

    // numpy.dot for 1-D vectors: raises on length mismatch like numpy does.
    static double dot(const std::vector<double>& a, const std::vector<double>& b) {
        if (a.size() != b.size()) {
            throw std::invalid_argument("vectors are not aligned for dot product");
        }
        double result = 0.0;
        for (std::size_t i = 0; i < a.size(); ++i) {
            result += a[i] * b[i];
        }
        return result;
    }

    // np.linalg.norm (l2) for a 1-D vector.
    static double l2_norm(const std::vector<double>& vec) {
        double sum_of_squares = 0.0;
        for (const double x : vec) {
            sum_of_squares += x * x;
        }
        return std::sqrt(sum_of_squares);
    }

    // numpy: array(vectors).mean(axis=0) — element-wise mean of equally sized vectors.
    static std::vector<double> mean_axis0(const std::vector<std::vector<double>>& vectors) {
        if (vectors.empty()) {
            return {};
        }
        const std::size_t dim = vectors.front().size();
        std::vector<double> mean(dim, 0.0);
        for (const std::vector<double>& v : vectors) {
            if (v.size() != dim) {
                throw std::invalid_argument("vectors must all have the same length");
            }
            for (std::size_t i = 0; i < dim; ++i) {
                mean[i] += v[i];
            }
        }
        const double count = static_cast<double>(vectors.size());
        for (std::size_t i = 0; i < dim; ++i) {
            mean[i] /= count;
        }
        return mean;
    }
};
