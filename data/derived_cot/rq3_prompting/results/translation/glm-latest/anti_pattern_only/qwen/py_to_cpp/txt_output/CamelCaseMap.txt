public:
    using iterator_category = std::forward_iterator_tag;
    using value_type = const std::string;
    using difference_type = std::ptrdiff_t;
    using pointer = const std::string*;
    using reference = const std::string&;

    explicit ConstKeyIterator(std::vector<std::pair<std::string, std::string>>::const_iterator it) : _it(it) {}

    reference operator*() const { return _it->first; }
    pointer operator->() const { return &_it->first; }
    ConstKeyIterator& operator++() { ++_it; return *this; }
    ConstKeyIterator operator++(int) { ConstKeyIterator tmp = *this; ++_it; return tmp; }
    bool operator==(const ConstKeyIterator& other) const { return _it == other._it; }
    bool operator!=(const ConstKeyIterator& other) const { return _it != other._it; }

private:
    std::vector<std::pair<std::string, std::string>>::const_iterator _it;
};

ConstKeyIterator begin() const { return ConstKeyIterator(_data.begin()); }

static std::string python_title(const std::string& s) {
    std::string result = s;
    bool prev_cased = false;
    for (char& c : result) {
        const bool is_lower = (c >= 'a' && c <= 'z');
        const bool is_upper = (c >= 'A' && c <= 'Z');
        if (is_lower || is_upper) {  // cased (ASCII letters)

                if (is_upper) c = static_cast<char>(c - 'A' + 'a');  // lowercase
            } else {

        }

// Python's str.title() for ASCII input: the first cased character of every
// "word" (maximal run of cased characters) is upper-cased, the rest are
// lower-cased.
static inline std::string py_title_ascii(const std::string& s) {
    std::string result = s;
    bool prev_cased = false;
    for (char& c : result) {
        const bool lower = (c >= 'a' && c <= 'z');
        const bool upper = (c >= 'A' && c <= 'Z');
        if (lower || upper) {
            if (prev_cased) {
                if (upper) c = static_cast<char>(c - 'A' + 'a');
            } else {
                if (lower) c = static_cast<char>(c - 'a' + 'A');
            }
            prev_cased = true;
        } else {
            prev_cased = false;
        }
    }
    return result;
}
