package org.example;

import java.util.*;
import java.util.stream.Collectors;

public class DataStatistics {

    public double mean(int[] data) {
        double sum = Arrays.stream(data).sum();
        return Math.round((sum / data.length) * 100.0) / 100.0;
    }

    public double median(int[] data) {
        int[] sortedData = Arrays.copyOf(data, data.length);
        Arrays.sort(sortedData);
        int n = sortedData.length;

        if (n % 2 == 0) {
            int middle = n / 2;
            return Math.round(((sortedData[middle - 1] + sortedData[middle]) / 2.0) * 100.0) / 100.0;
        } else {
            int middle = n / 2;
            return sortedData[middle];
        }
    }

    public List<Integer> mode(int[] data) {
        Map<Integer, Long> frequencyMap = Arrays.stream(data)
                .boxed()
                .collect(Collectors.groupingBy(e -> e, Collectors.counting()));

        long maxCount = Collections.max(frequencyMap.values());

        return frequencyMap.entrySet()
                .stream()
                .filter(entry -> entry.getValue() == maxCount)
                .map(Map.Entry::getKey)
                .sorted()
                .collect(Collectors.toList());
    }
}

```python
import math
from collections import Counter


class DataStatistics:

    def mean(self, data):
        """Return the mean rounded to 2 decimal places (half-up, like Java's Math.round)."""
        total = sum(data)
        if not data:
            # Java: 0.0 / 0 -> NaN; Math.round(NaN) -> 0; 0 / 100.0 -> 0.0
            return 0.0
        return math.floor(total / len(data) * 100.0 + 0.5) / 100.0

    def median(self, data):
        sorted_data = sorted(data)
        n = len(sorted_data)

        if n % 2 == 0:
            middle = n // 2
            value = (sorted_data[middle - 1] + sorted_data[middle]) / 2.0
            return math.floor(value * 100.0 + 0.5) / 100.0
        else:
            middle = n // 2
            return float(sorted_data[middle])

    def mode(self, data):
        frequency = Counter(data)
        # Raises ValueError on empty input, mirroring Java's NoSuchElementException
        max_count = max(frequency.values())
        return sorted(key for key, count in frequency.items() if count == max_count)
```
