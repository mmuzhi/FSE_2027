import numpy as np

class MovieBookingSystem:
    def __init__(self):
        self.movies = []

    def add_movie(self, name, price, start_time, end_time, n):
        movie = {
            'name': name,
            'price': price,
            'start_time': datetime.strptime(start_time, '%H:%M'),
            'end_time': datetime.strptime(end_time, '%H:%M'),
            'seats': np.zeros((n, n))
        }
        self.movies.append(movie)

    def book_ticket(self, name, seats_to_book):
        for movie in self.movies:

                for seat in seats_to_book:

                        movie['seats'][seat[0]][seat[1]] = 1
                    else:

                return "Booking success."

        start_time = datetime.strptime(start_time, '%H:%M')

        for movie in self.movies:
            if start_time <= movie['start_time'] and movie['end_time'] <= end_time:
                available_movies.append(movie['name'])

        return available_movies

class MovieBookingSystem {
private:
    struct Movie {
        std::string name;
        double price;
        int startTime;   // minutes since midnight (datetime(1900,1,1,H,M))
        int endTime;     // minutes since midnight
        std::vector<std::vector<double>> seats;
    };

    std::vector<Movie> movies;

    // Parse "HH:MM" like datetime.strptime(s, "%H:%M") -> minutes since midnight.
    // Throws std::invalid_argument on malformed input (Python raises ValueError).
    static int parseTimeToMinutes(const std::string& timeStr) {
        std::size_t i = 0;

        auto readField = [&](int maxDigits, int maxValue, int& value) -> bool {
            int digits = 0;
            value = 0;
            while (i < timeStr.size() && digits < maxDigits &&
                   std::isdigit(static_cast<unsigned char>(timeStr[i]))) {
                value = value * 10 + (timeStr[i] - '0');
                ++i;
                ++digits;
            }
            if (digits == 0) return false;
            if (digits == maxDigits && value > maxValue) return false;
            return true;
        };

        int hour = 0;
        int minute = 0;
        if (!readField(2, 23, hour)) {
            throw std::invalid_argument("time data '" + timeStr + "' does not match format '%H:%M'");
        }
        if (i >= timeStr.size() || timeStr[i] != ':') {
            throw std::invalid_argument("time data '" + timeStr + "' does not match format '%H:%M'");
        }
        ++i; // skip ':'
        if (!readField(2, 59, minute)) {
            throw std::invalid_argument("time data '" + timeStr + "' does not match format '%H:%M'");
        }
        if (i != timeStr.size()) {
            throw std::invalid_argument("unconverted data remains: '" + timeStr.substr(i) + "'");
        }
        return hour * 60 + minute;
    }
