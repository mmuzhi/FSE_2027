                + "gender TEXT, "
                + "grade INTEGER"

        } catch (SQLException e) {
            e.printStackTrace();
        }

            pstmt.setString(1, studentData.getName());
            pstmt.setInt(2, studentData.getAge());
            pstmt.setString(3, studentData.getGender());
            pstmt.setInt(4, studentData.getGrade());

        } catch (SQLException e) {
            e.printStackTrace();
        }

class SqliteConnection {
public:
    explicit SqliteConnection(const std::string& filename) {
        const int rc = sqlite3_open(filename.c_str(), &db_);
        if (rc != SQLITE_OK) {
            std::string msg = db_ ? sqlite3_errmsg(db_) : "out of memory";
            if (db_) {
                sqlite3_close(db_);
                db_ = nullptr;
            }
            throw SqlException("Error opening database '" + filename + "': " + msg);
        }
    }

    ~SqliteConnection() {
        if (db_ != nullptr) {
            sqlite3_close(db_);
        }
    }

    SqliteConnection(const SqliteConnection&) = delete;
    SqliteConnection& operator=(const SqliteConnection&) = delete;

    sqlite3* handle() const { return db_; }

private:
    sqlite3* db_ = nullptr;
};

class SqliteStatement {
public:
    SqliteStatement(sqlite3* db, const std::string& sql) {
        const int rc = sqlite3_prepare_v2(db, sql.c_str(), -1, &stmt_, nullptr);
        if (rc != SQLITE_OK) {
            stmt_ = nullptr;
            throw SqlException(std::string("SQL error in prepare: ") + sqlite3_errmsg(db));
        }
    }

    ~SqliteStatement() {
        if (stmt_ != nullptr) {
            sqlite3_finalize(stmt_);
        }
    }

    SqliteStatement(const SqliteStatement&) = delete;
    SqliteStatement& operator=(const SqliteStatement&) = delete;

    sqlite3_stmt* handle() const { return stmt_; }

private:
    sqlite3_stmt* stmt_ = nullptr;
};

// RAII wrapper around sqlite3*, analogous to a JDBC Connection
// (a connection is opened and later closed automatically).
class SqliteConnection {
public:
    explicit SqliteConnection(const std::string& filename) {
        const int rc = sqlite3_open(filename.c_str(), &db_);
        if (rc != SQLITE_OK) {
            const std::string message = (db_ != nullptr) ? sqlite3_errmsg(db_) : "out of memory";
            if (db_ != nullptr) {
                sqlite3_close(db_);
                db_ = nullptr;
            }
            throw SQLException("Error opening database '" + filename + "': " + message);
        }
    }

    SqliteConnection(const SqliteConnection&) = delete;
    SqliteConnection& operator=(const SqliteConnection&) = delete;

    SqliteConnection(SqliteConnection&& other) noexcept : db_(other.db_) {
        other.db_ = nullptr;
    }

    SqliteConnection& operator=(const SqliteConnection&) = delete;
    SqliteConnection& operator=(SqliteConnection&&) = delete;

    ~SqliteConnection() {
        if (db_ != nullptr) {
            sqlite3_close(db_);
        }
    }

    sqlite3* handle() const { return db_; }

private:
    sqlite3* db_ = nullptr;
};

// RAII wrapper around sqlite3_stmt*, analogous to a JDBC Statement /
// PreparedStatement (the statement is finalized automatically).
class SqliteStatement {
public:
    SqliteStatement(sqlite3* db, const std::string& sql) {
        const int rc = sqlite3_prepare_v2(db, sql.c_str(), -1, &stmt_, nullptr);
        if (rc != SQLITE_OK) {
            stmt_ = nullptr;
            throw SQLException(std::string("SQL error: ") + sqlite3_errmsg(db));
        }
    }

    SqliteStatement(const SqliteStatement&) = delete;
    SqliteStatement& operator=(const SqliteStatement&) = delete;

    SqliteStatement(SqliteStatement&& other) noexcept : stmt_(other.stmt_) {
        other.stmt_ = nullptr;
    }

    SqliteStatement& operator=(const SqliteStatement&) = delete;
    SqliteStatement& operator=(SqliteStatement&&) = delete;

    ~SqliteStatement() {
        if (stmt_ != nullptr) {
            sqlite3_finalize(stmt_);
        }
    }

    sqlite3_stmt* handle() const { return stmt_; }

private:
    sqlite3_stmt* stmt_ = nullptr;
};

public:
    explicit SQLException(std::string message)
        : std::runtime_error(std::move(message)) {}

    // Analog of Throwable.printStackTrace(): reports the error on stderr.
    void printStackTrace() const {
        std::cerr << "SQLException: " << what() << std::endl;

// RAII wrapper around sqlite3*; plays the role of a JDBC Connection that is
// automatically closed (try-with-resources).
class SqliteConnection {
public:
    explicit SqliteConnection(const std::string& filename)
        : db_(nullptr) {
        const int rc = sqlite3_open(filename.c_str(), &db_);
        if (rc != SQLITE_OK) {
            const std::string message =
                (db_ != nullptr) ? sqlite3_errmsg(db_) : "out of memory";
            if (db_ != nullptr) {
                sqlite3_close(db_);
                db_ = nullptr;
            }
            throw SQLException("Error opening database '" + filename +
                               "': " + message);
        }
    }

    SqliteConnection(const SqliteConnection&) = delete;
    SqliteConnection& operator=(const SqliteConnection&) = delete;

    SqliteConnection(SqliteConnection&& other) noexcept
        : db_(other.db_) {
        other.db_ = nullptr;
    }

    SqliteConnection& operator=(SqliteConnection&&) = delete;

    ~SqliteConnection() {
        if (db_ != nullptr) {
            sqlite3_close(db_);
        }
    }

    sqlite3* handle() const { return db_; }

private:
    sqlite3* db_;
};

// RAII wrapper around sqlite3_stmt*; plays the role of JDBC Statement /
// PreparedStatement / ResultSet (the statement doubles as the result cursor
// and is finalized automatically).
class SqliteStatement {
public:
    SqliteStatement(sqlite3* db, const std::string& sql)
        : stmt_(nullptr) {
        const int rc =
            sqlite3_prepare_v2(db, sql.c_str(), -1, &stmt_, nullptr);
        if (rc != SQLITE_OK) {
            stmt_ = nullptr;
            throw SQLException(std::string("SQL error: ") +
                               sqlite3_errmsg(db));
        }
    }

    SqliteStatement(const SqliteStatement&) = delete;
    SqliteStatement& operator=(const SqliteStatement&) = delete;

    SqliteStatement(SqliteStatement&& other) noexcept
        : stmt_(other.stmt_) {
        other.stmt_ = nullptr;
    }

    SqliteStatement& operator=(SqliteStatement&&) = delete;

    ~SqliteStatement() {
        if (stmt_ != nullptr) {
            sqlite3_finalize(stmt_);
        }
    }

    sqlite3_stmt* handle() const { return stmt_; }

private:
    sqlite3_stmt* stmt_;
};
