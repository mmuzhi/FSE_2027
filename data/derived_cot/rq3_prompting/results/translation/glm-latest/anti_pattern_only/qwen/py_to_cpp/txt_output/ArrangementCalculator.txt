#include <optional>
#include <stdexcept>
#include <utility>

/**
 * The ArrangementCalculator class provides permutation calculations
 * and selection operations for a given set of data elements.
 */
class ArrangementCalculator {
public:
    // Initializes the ArrangementCalculator object with a list of datas.
    ArrangementCalculator(std::vector<int> datas) : datas(std::move(datas)) {}

    // Counts the number of arrangements by choosing m items from n items.
    // If m is not provided or n equals m, returns factorial(n).
    static long long count(int n, std::optional<int> m = std::nullopt) {
        if (!m.has_value() || n == m.value()) {
            return factorial(n);
        }
        return factorial(n) / factorial(n - m.value());
    }

    // Counts the total number of all possible arrangements by choosing at
    // least 1 item and at most n items from n items.
    static long long count_all(int n) {
        long long total = 0;
        for (int i = 1; i <= n; ++i) {
            total += count(n, i);
        }
        return total;
    }

    // Generates a list of arrangements by selecting m items from the internal
    // datas. If m is not provided, selects all items.
    // Permutations are generated over positions (like itertools.permutations),
    // in lexicographic index order.
    std::vector<std::vector<int>> select(std::optional<int> m = std::nullopt) const {
        int r;
        if (m.has_value()) {
            r = m.value();
            if (r < 0) {
                // itertools.permutations raises ValueError for negative r.
                throw std::invalid_argument("r must be non-negative");
            }
        } else {
            r = static_cast<int>(datas.size());
        }

        std::vector<std::vector<int>> result;
        const int n = static_cast<int>(datas.size());
        if (r > n) {
            return result;  // itertools.permutations yields nothing when r > n
        }
        std::vector<int> current;
        current.reserve(static_cast<size_t>(r));
        std::vector<bool> used(static_cast<size_t>(n), false);
        permute(r, current, used, result);
        return result;
    }

    // Generates a list of all arrangements by selecting at least 1 item and
    // at most the number of internal datas.
