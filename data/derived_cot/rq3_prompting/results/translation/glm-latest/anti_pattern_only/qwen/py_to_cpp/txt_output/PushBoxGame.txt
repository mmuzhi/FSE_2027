        init_game();
    }

    void init_game() {
        for (int row = 0; row < static_cast<int>(map.size()); ++row) {
            for (int col = 0; col < static_cast<int>(map[row].size()); ++col) {
                if (map[row][col] == 'O') {
                    player_row = row;
                    player_col = col;
                } else if (map[row][col] == 'G') {
                    targets.push_back(std::make_pair(row, col));
                    target_count += 1;
                } else if (map[row][col] == 'X') {
                    boxes.push_back(std::make_pair(row, col));
                }
            }
        }
    }

    bool check_win() {
        int box_on_target_count = 0;
        for (const auto& box : boxes) {
            if (std::find(targets.begin(), targets.end(), box) != targets.end()) {
                box_on_target_count += 1;
            }
        }
        if (box_on_target_count == target_count) {
            is_game_over = true;
        }
        return is_game_over;
    }

    bool move(const std::string& direction) {
        int new_player_row = player_row;
        int new_player_col = player_col;

        if (direction == "w") {
            new_player_row -= 1;
        } else if (direction == "s") {
            new_player_row += 1;
        } else if (direction == "a") {
            new_player_col -= 1;

            new_player_col += 1;

    void init_game() {
        for (std::size_t row = 0; row < map.size(); ++row) {
            const std::string& line = map[row];
            for (std::size_t col = 0; col < line.size(); ++col) {
                const char cell = line[col];
                if (cell == 'O') {
                    player_row = static_cast<int>(row);
                    player_col = static_cast<int>(col);

                    boxes.push_back(std::make_pair(static_cast<int>(row),

            }

            is_game_over = true;
        }
        return is_game_over;

    // check whether the game is won.

        int new_player_col = player_col;

        if (direction == "w") {
            new_player_row -= 1;

            new_player_row += 1;
        } else if (direction == "a") {

            new_player_col += 1;
