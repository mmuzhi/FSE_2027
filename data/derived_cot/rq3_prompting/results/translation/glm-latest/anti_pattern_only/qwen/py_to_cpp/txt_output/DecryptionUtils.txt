#include <cctype>
#include <stdexcept>

/**
 * This is a class that provides methods for decryption, including
 * the Caesar cipher, Vigenere cipher, and Rail Fence cipher.
 */
class DecryptionUtils {
public:
    /**
     * Initializes the decryption utility with a key.
     * @param key The key to use for decryption.
     */
    explicit DecryptionUtils(std::string key)
        : key_(std::move(key)) {}

    /**
     * Deciphers the given ciphertext using the Caesar cipher.
     * @param ciphertext The ciphertext to decipher.
     * @param shift The shift to use for decryption.
     * @return The deciphered plaintext.
     */
    std::string caesar_decipher(const std::string& ciphertext, long long shift) const {
        std::string plaintext;
        plaintext.reserve(ciphertext.size());
        for (char ch : ciphertext) {
            unsigned char uc = static_cast<unsigned char>(ch);
            if (std::isalpha(uc)) {
                int ascii_offset = std::isupper(uc) ? 65 : 97;
                long long shifted = (static_cast<long long>(uc) - ascii_offset - shift) % 26;
                if (shifted < 0) shifted += 26;
                plaintext += static_cast<char>(shifted + ascii_offset);
            } else {
                plaintext += ch;
            }
        }
        return plaintext;
    }

    /**
     * Deciphers the given ciphertext using the Vigenere cipher.
     * @param ciphertext The ciphertext to decipher.
     * @return The deciphered plaintext.
     */
    std::string vigenere_decipher(const std::string& ciphertext) const {
        std::string decrypted_text;
        decrypted_text.reserve(ciphertext.size());
        std::size_t key_index = 0;
        for (char ch : ciphertext) {
            unsigned char uc = static_cast<unsigned char>(ch);
            if (std::isalpha(uc)) {
                if (key_.empty()) {
                    // Python raises ZeroDivisionError (key_index % len(key) with len(key) == 0).
                    throw std::runtime_error("integer division or modulo by zero");
                }
                unsigned char key_ch = static_cast<unsigned char>(key_[key_index % key_.size()]);
                int shift = std::tolower(key_ch) - 'a';
                long long value = (std::tolower(uc) - 'a' - shift) % 26;
                if (value < 0) value += 26;
                char decrypted_char = static_cast<char>(value + 'a');
                decrypted_text += std::isupper(uc)
                                      ? static_cast<char>(std::toupper(static_cast<unsigned char>(decrypted_char)))
                                      : decrypted_char;
                ++key_index;
            } else {
                decrypted_text += ch;
            }
        }
        return decrypted_text;
    }

    /**
     * Deciphers the given ciphertext using the Rail Fence cipher.
     * @param encrypted_text The ciphertext to decipher.
     * @param rails The number of rails to use for decryption.
     * @return The deciphered plaintext.
     */
    std::string rail_fence_decipher(const std::string& encrypted_text, long long rails) const {
        const std::size_t n = encrypted_text.size();
        // Python builds `rails` rows (an empty list of rows when rails <= 0)
        // and initializes every cell to '\n'. Out-of-range accesses raise
        // IndexError in Python; .at() throws std::out_of_range here.
        std::vector<std::vector<char>> fence(
            rails > 0 ? static_cast<std::size_t>(rails) : 0,
            std::vector<char>(n, '\n'));

        long long direction = -1;
        long long row = 0;
        std::size_t col = 0;

        // Mark the zig-zag path (Python marks cells with the empty string '').
        for (std::size_t i = 0; i < n; ++i) {
            if (row == 0 || row == rails - 1) {
                direction = -direction;
            }
            fence.at(static_cast<std::size_t>(row)).at(col) = '\0';
            ++col;
            row += direction;
        }

        std::size_t index = 0;
        for (std::size_t i = 0; i < fence.size(); ++i) {
            for (std::size_t j = 0; j < n; ++j) {
                if (fence[i][j] == '\0') {
                    fence[i][j] = encrypted_text[index];
                    ++index;
                }
            }
        }

        std::string plain_text;
        plain_text.reserve(n);
        direction = -1;
        row = 0;
        col = 0;
        for (std::size_t i = 0; i < n; ++i) {
            if (row == 0 || row == rails - 1) {
                direction = -direction;
            }
            plain_text += fence.at(static_cast<std::size_t>(row)).at(col);
            ++col;
            row += direction;
        }

        return plain_text;
    }

private:
    std::string key_;
};

// This is a class that provides methods for decryption, including
// the Caesar cipher, Vigenere cipher, and Rail Fence cipher.

class DecryptionUtils {
public:
    // Initializes the decryption utility with a key.
    explicit DecryptionUtils(std::string key)
        : key_(std::move(key)) {}

    // Deciphers the given ciphertext using the Caesar cipher.
    // Returns the deciphered plaintext.
    std::string caesar_decipher(const std::string& ciphertext, long long shift) const {
        std::string plaintext;
        plaintext.reserve(ciphertext.size());
        for (char ch : ciphertext) {
            unsigned char uc = static_cast<unsigned char>(ch);
            if (std::isalpha(uc)) {
                // Python: 65 if char.isupper() else 97
                const int ascii_offset = std::isupper(uc) ? 65 : 97;
                // Python's % always yields a non-negative result; C++'s may be
                // negative, so normalize it.
                long long shifted = (static_cast<long long>(uc) - ascii_offset - shift) % 26;
                if (shifted < 0) shifted += 26;
                plaintext += static_cast<char>(shifted + ascii_offset);
            } else {
                plaintext += ch;
            }
        }
        return plaintext;
    }

    // Deciphers the given ciphertext using the Vigenere cipher.
    // Returns the deciphered plaintext.
    std::string vigenere_decipher(const std::string& ciphertext) const {
        std::string decrypted_text;
        decrypted_text.reserve(ciphertext.size());
        std::size_t key_index = 0;
        for (char ch : ciphertext) {
            unsigned char uc = static_cast<unsigned char>(ch);
            if (std::isalpha(uc)) {
                // Python: key_index % len(self.key) raises ZeroDivisionError
                // when the key is empty; mirror that with an exception.
                if (key_.empty()) {
                    throw std::runtime_error("integer division or modulo by zero");
                }
                const unsigned char key_char =
                    static_cast<unsigned char>(key_[key_index % key_.size()]);
                const int shift = std::tolower(key_char) - 'a';
                int value = (std::tolower(uc) - 'a' - shift) % 26;
                if (value < 0) value += 26;  // Python-style modulo
                const char decrypted_char = static_cast<char>(value + 'a');
                decrypted_text += std::isupper(uc)
                                      ? static_cast<char>(std::toupper(
                                            static_cast<unsigned char>(decrypted_char)))
                                      : decrypted_char;
                ++key_index;
            } else {
                decrypted_text += ch;
            }
        }
        return decrypted_text;
    }

    // Deciphers the given ciphertext using the Rail Fence cipher.
    // Returns the deciphered plaintext.
    std::string rail_fence_decipher(const std::string& encrypted_text, long long rails) const {
        const std::size_t n = encrypted_text.size();
        // Python builds `rails` rows (an empty list of rows when rails <= 0),
        // each initialized to '\n'. Python raises IndexError on out-of-range
        // access; .at() throws std::out_of_range analogously.
        std::vector<std::vector<char>> fence(
            rails > 0 ? static_cast<std::size_t>(rails) : 0,
            std::vector<char>(n, '\n'));

        long long direction = -1;
        long long row = 0;
        std::size_t col = 0;

        // Mark the zig-zag path (Python marks cells with the empty string '').
        for (std::size_t i = 0; i < n; ++i) {
            if (row == 0 || row == rails - 1) {
                direction = -direction;
            }
            fence.at(static_cast<std::size_t>(row)).at(col) = '\0';
            ++col;
            row += direction;
        }

        // Fill the marked cells with the ciphertext, row by row.
        std::size_t index = 0;
        for (std::size_t i = 0; i < fence.size(); ++i) {
            for (std::size_t j = 0; j < n; ++j) {
                if (fence[i][j] == '\0') {
                    fence[i][j] = encrypted_text[index];
                    ++index;
                }
            }
        }

        // Read the fence back off in zig-zag order.
        std::string plain_text;
        plain_text.reserve(n);
        direction = -1;
        row = 0;
        col = 0;
        for (std::size_t i = 0; i < n; ++i) {
            if (row == 0 || row == rails - 1) {
                direction = -direction;
            }
            plain_text += fence.at(static_cast<std::size_t>(row)).at(col);
            ++col;
            row += direction;
        }

        return plain_text;
    }

private:
    std::string key_;
};
