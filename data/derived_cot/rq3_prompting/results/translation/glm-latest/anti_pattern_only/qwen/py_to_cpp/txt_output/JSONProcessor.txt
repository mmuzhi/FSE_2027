class JSONProcessor:

        """

        """

    def write_json(self, data, file_path):
        """
        Write data to a JSON file and save it to the given path.

        :param data: dict, the data to be written to the JSON file.
        :param file_path: str, the path of the JSON file.
        :return: 1 if the writing process is successful, or -1, if an error occurs during the writing process.
        >>> json.write_json({'key1': 'value1', 'key2': 'value2'}, 'test.json')
        1
        >>> json.read_json('test.json')
        {'key1': 'value1', 'key2': 'value2'}
        """

    def process_json(self, file_path, remove_key):
        """
        read a JSON file and process the data by removing a specified key and rewrite the modified data back to the file.

        :param file_path: str, the path of the JSON file.
        :param remove_key: str, the key to be removed.
        :return: 1, if the specified key is successfully removed and the data is written back.
                    0, if the file does not exist or the specified key does not exist in the data.

        {'key1': 'value1', 'key2': 'value2'}
        >>> json.process_json('test.json', 'key1')
        1
        >>> json.read_json('test.json')
        {'key2': 'value2'}
        """

import json
import os


class JSONProcessor:
    def read_json(self, file_path):
        if not os.path.exists(file_path):
            return 0
        try:
            with open(file_path, 'r') as file:
                data = json.load(file)
            return data
        except:
            return -1

    def write_json(self, data, file_path):
        try:
            with open(file_path, 'w') as file:
                json.dump(data, file)
            return 1
        except:
            return -1

    def process_json(self, file_path, remove_key):
        data = self.read_json(file_path)
        if data == 0 or data == -1:
            return 0
        if remove_key in data:
            del data[remove_key]
            self.write_json(data, file_path)
            return 1
        else:
            return 0

// Appends the 4 lowercase hex digits of `value` (16 bits used) to `out`.
void appendHex4(std::uint32_t value, std::string& out) {
    static const char* digits = "0123456789abcdef";
    out += digits[(value >> 12) & 0xF];
    out += digits[(value >> 8) & 0xF];
    out += digits[(value >> 4) & 0xF];
    out += digits[value & 0xF];
}

// Serializes a string exactly like Python's json.dumps with default settings
// (ensure_ascii=True): escapes ", \, control characters and every character
// outside the printable ASCII range as \uXXXX (with surrogate pairs above BMP).
void dumpPyString(const std::string& s, std::string& out) {
    out += '"';
    std::size_t i = 0;
    while (i < s.size()) {
        const unsigned char c = static_cast<unsigned char>(s[i]);
        std::uint32_t cp = 0;
        std::size_t extra = 0;
        if (c < 0x80) {
            cp = c;
        } else if ((c & 0xE0) == 0xC0) {
            cp = c & 0x1Fu;
            extra = 1;
        } else if ((c & 0xF0) == 0xE0) {
            cp = c & 0x0Fu;
            extra = 2;
        } else if ((c & 0xF8) == 0xF0) {
            cp = c & 0x07u;
            extra = 3;
        } else {
            cp = 0xFFFDu;  // invalid byte: replacement char fallback
        }
        bool ok = true;
        for (std::size_t k = 0; k < extra; ++k) {
            if (i + 1 + k >= s.size() ||
                (static_cast<unsigned char>(s[i + 1 + k]) & 0xC0) != 0x80) {
                ok = false;
                break;
            }
        }
        if (ok) {
            for (std::size_t k = 1; k <= extra; ++k) {
                cp = (cp << 6) | (static_cast<unsigned char>(s[i + k]) & 0x3Fu);
            }
            i += 1 + extra;
        } else {
            cp = 0xFFFDu;
            i += 1;
        }

        switch (cp) {
            case '"': out += "\\\""; break;
            case '\\': out += "\\\\"; break;
            case '\b': out += "\\b"; break;
            case '\f': out += "\\f"; break;
            case '\n': out += "\\n"; break;
            case '\r': out += "\\r"; break;
            case '\t': out += "\\t"; break;
            default:
                if (cp < 0x20 || cp > 0x7Eu) {
                    if (cp <= 0xFFFF) {
                        out += "\\u";
                        appendHex4(cp, out);
                    } else {
                        const std::uint32_t v = cp - 0x10000u;
                        out += "\\u";
                        appendHex4(0xD800u + (v >> 10), out);
                        out += "\\u";
                        appendHex4(0xDC00u + (v & 0x3FFu), out);
                    }
                } else {
                    out += static_cast<char>(cp);
                }
        }
    }
    out += '"';
}

// Serializes a JSON value exactly like Python's json.dump with default
// settings: separators (', ', ': ') and ensure_ascii=True.
void dumpPy(const json& value, std::string& out) {
    switch (value.type()) {
        case json::value_t::object: {
            out += '{';
            bool first = true;
            for (auto it = value.begin(); it != value.end(); ++it) {
                if (!first) {
                    out += ", ";
                }
                first = false;
                dumpPyString(it.key(), out);
                out += ": ";
                dumpPy(it.value(), out);
            }
            out += '}';
            break;
        }
        case json::value_t::array: {
            out += '[';
            bool first = true;
            for (const auto& item : value) {
                if (!first) {
                    out += ", ";
                }
                first = false;
                dumpPy(item, out);
            }
            out += ']';
            break;
        }
        case json::value_t::string:
            dumpPyString(value.get_ref<const json::string_t&>(), out);
            break;
        default:
            out += value.dump();
            break;
    }
}

}  // namespace
