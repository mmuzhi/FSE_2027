public:
    std::string binary_string;

    explicit BinaryDataProcessor(const std::string& binary_string_)

        clean_non_binary_chars();

    void clean_non_binary_chars() {

    std::map<std::string, std::variant<double, int>> calculate_binary_info() const {

        const int ones_count = static_cast<int>(std::count(binary_string.begin(), binary_string.end(), '1'));

            {"Ones", ones_percentage},
            {"Bit length", total_length}
        };
