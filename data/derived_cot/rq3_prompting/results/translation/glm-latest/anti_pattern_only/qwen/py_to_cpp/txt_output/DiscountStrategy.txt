#include <functional>
#include <map>
#include <set>
#include <stdexcept>
#include <string>
#include <utility>
#include <variant>
#include <vector>

/*
 * This is a class that allows using different discount strategies based on
 * shopping credit or shopping cart in a supermarket.
 *
 * Python dicts are modelled with std::map<std::string, Value>, where a Value
 * can hold a string, an int or a float. Missing keys throw std::out_of_range
 * (the analogue of Python's KeyError) and values of the wrong type throw
 * std::bad_variant_access (the analogue of Python's TypeError).
 */
class DiscountStrategy {
public:
    using Value = std::variant<std::string, int, double>;  // a Python dict value
    using Dict = std::map<std::string, Value>;             // a Python dict
    using Cart = std::vector<Dict>;                        // a Python list of dicts

    // A Python function taking the order and returning the discount.
    using Promotion = std::function<double(DiscountStrategy&)>;

    Dict customer;   // customer information, e.g. {'name': ..., 'fidelity': ...}
    Cart cart;       // cart of items, e.g. {'product': ..., 'quantity': ..., 'price': ...}
    Promotion promotion;  // optional promotion applied to the order

    /*
     * Initialize the DiscountStrategy with customer information, a cart of
     * items, and an optional promotion.
     */
    DiscountStrategy(Dict customer_, Cart cart_, Promotion promotion_ = nullptr)
        : customer(std::move(customer_)),
          cart(std::move(cart_)),
          promotion(std::move(promotion_)) {
        total_ = total();
    }

    /*
     * Calculate the total cost of items in the cart.
     * Returns the total cost of items (e.g. 14 * 23.5 -> 329.0).
     */
    double total() {
        double sum = 0.0;
        for (const Dict& item : cart) {
            sum += number(item, "quantity") * number(item, "price");
        }
        total_ = sum;
        return total_;
    }

    /*
     * Calculate the final amount to be paid after applying the discount
     * (e.g. 329.0 - 16.45 -> 312.55).
     */
    double due() {
        double discount = 0.0;
        if (promotion) {
            discount = promotion(*this);
        }
        return total_ - discount;
    }

    /*
     * Discount based on the fidelity points of the customer. Customers with
     * over 1000 points enjoy a 5% discount on the entire order
     * (e.g. 329.0 * 0.05 -> 16.45).
     */
    static double FidelityPromo(DiscountStrategy& order) {
        if (number(order.customer, "fidelity") >= 1000) {
            return order.total() * 0.05;
        }
        return 0.0;
    }

    /*
     * Discount based on bulk item quantity in the order. In the same order,
     * if the quantity of a single item reaches 20 or more, each such item
     * enjoys a 10% discount (e.g. 20 * 23.5 * 0.1 -> 47.0).
     */
    static double BulkItemPromo(DiscountStrategy& order) {
        double discount = 0.0;
        for (const Dict& item : order.cart) {
            if (number(item, "quantity") >= 20) {
                discount += number(item, "quantity") * number(item, "price") * 0.1;
            }
        }
        return discount;
    }

    /*
     * Discount based on the number of different products in the order. If the
     * order contains 10 or more different products, the entire order enjoys a
     * 7% discount.
     */
    static double LargeOrderPromo(DiscountStrategy& order) {
        std::set<std::string> products;
        for (const Dict& item : order.cart) {
            products.insert(text(item, "product"));
        }
        if (products.size() >= 10) {
            return order.total() * 0.07;
        }
        return 0.0;
    }

private:
    // Python's name-mangled self.__total
    double total_ = 0.0;

    // dict['key'] for a numeric value (int or float).
    static double number(const Dict& dict, const std::string& key) {
        const Value& value = dict.at(key);  // std::out_of_range ~ KeyError
        if (const double* d = std::get_if<double>(&value)) {
            return *d;
        }
        return static_cast<double>(std::get<int>(value));  // bad_variant_access ~ TypeError
    }

    // dict['key'] for a string value.
    static const std::string& text(const Dict& dict, const std::string& key) {
        return std::get<std::string>(dict.at(key));  // out_of_range / bad_variant_access
    }
};
