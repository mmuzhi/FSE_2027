                return email;

            }
        }
    }

class EmailClient {
private:
    std::string addr;
    double capacity;
    InboxRef inbox;

    static std::string nowTimestamp() {
        const std::time_t now =
            std::chrono::system_clock::to_time_t(std::chrono::system_clock::now());
        std::tm localTm{};
        const std::tm* const tmPtr = std::localtime(&now);
        if (tmPtr != nullptr) {
            localTm = *tmPtr;
        }
        std::ostringstream oss;
        oss << std::put_time(&localTm, "%Y-%m-%d %H:%M:%S");
        return oss.str();
    }

    static bool tryAsNumber(const std::any& value, double& out) {
        if (!value.has_value()) {
            return false;
        }
        const std::type_info& ti = value.type();
        if (ti == typeid(double)) { out = std::any_cast<double>(value); return true; }
        if (ti == typeid(float))  { out = std::any_cast<float>(value);  return true; }
        if (ti == typeid(int))    { out = std::any_cast<int>(value);    return true; }
        if (ti == typeid(long))   { out = static_cast<double>(std::any_cast<long>(value)); return true; }
        if (ti == typeid(long long)) { out = static_cast<double>(std::any_cast<long long>(value)); return true; }
        if (ti == typeid(short))  { out = std::any_cast<short>(value);  return true; }
        return false;

        const auto it = email.find("size");
        if (it == email.end()) {
            return false;
        }
        return tryAsNumber(it->second, out);
    }

public:

        if (!recv.isFullWithOneMoreEmail(size)) {
            const std::string timestamp = nowTimestamp();

            email.emplace("sender", this->addr);
            email.emplace("receiver", recv.addr);
            email.emplace("content", content);
            email.emplace("size", size);
            email.emplace("time", timestamp);
            email.emplace("state", std::string("unread"));

            return true;

            this->clearInbox(size);
            if (recv.isFullWithOneMoreEmail(size)) {
                return false;
            }
            return sendTo(recv, content, size); // Retry sending after clearing
        }
    }

        if (this->inbox->empty()) {
            return nullptr;
        }
        for (const EmailRef& email : *this->inbox) {
            const auto it = email->find("state");
            if (it == email->end()) {
                continue;
            }
            const std::string* const state = std::any_cast<std::string>(&it->second);
            if (state != nullptr && *state == "unread") {
                (*email)["state"] = std::string("read");
                return email;
            }
        }
        return nullptr;
    }

    bool isFullWithOneMoreEmail(double size) const {
        const double occupiedSize = this->getOccupiedSize();
        return occupiedSize + size > this->capacity;
    }

    double getOccupiedSize() const {
        double occupiedSize = 0;
        double sz = 0;
        for (const EmailRef& email : *this->inbox) {
            if (tryGetSize(*email, sz)) {
                occupiedSize += sz;
            }
        }
        return occupiedSize;
    }

    void clearInbox(double size) {
        if (this->addr.empty()) {
            return;
        }
        double freedSpace = 0;
        double sz = 0;
        while (freedSpace < size && !this->inbox->empty()) {
            const EmailRef email = this->inbox->front();
            this->inbox->erase(this->inbox->begin());
            if (tryGetSize(*email, sz)) {
                freedSpace += sz;
            }
        }
    }

    InboxRef getInbox() const {
        return this->inbox;
    }

    void setInbox(InboxRef inbox) {
        this->inbox = std::move(inbox);
    }
};
