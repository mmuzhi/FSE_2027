    auto isDigit = [](char c) {
        return c >= '0' && c <= '9';

    size_t i = 0;
    auto readFixedTwoDigits = [&](const char* field) -> int {
        if (i + 2 > text.size() || !isDigit(text[i]) || !isDigit(text[i + 1])) {
            throw std::invalid_argument(failMsg + ": expected two digits for " + field);

        int value = (text[i] - '0') * 10 + (text[i + 1] - '0');
        i += 2;
        return value;
