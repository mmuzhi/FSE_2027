```py
class DataStatistics4:

        :return: The correlation coefficient, float.

        """

        """
        Calculate the skewness of a set of data.

        >>> DataStatistics4.skewness([1, 2, 5])

        :return: The kurtosis, float.

        -1.5000000000000007

        """

    @staticmethod

        Calculate the probability density function (PDF) of a set of data under a normal distribution.
        :param data: The input data list, list.
        :param mu: The mean of the normal distribution, float.
        :param sigma: The standard deviation of the normal distribution, float.
        :return: The probability density function (PDF), list.

import math

class DataStatistics4:

    @staticmethod
    def correlation_coefficient(data1, data2):
        n = len(data1)
        mean1 = sum(data1) / n
        mean2 = sum(data2) / n

        numerator = sum((data1[i] - mean1) * (data2[i] - mean2) for i in range(n))
        denominator = math.sqrt(sum((data1[i] - mean1) ** 2 for i in range(n))) * math.sqrt(sum((data2[i] - mean2) ** 2 for i in range(n)))

        return numerator / denominator if denominator != 0 else 0
    
    @staticmethod
    def skewness(data):
        n = len(data)
        mean = sum(data) / n
        variance = sum((x - mean) ** 2 for x in data) / n
        std_deviation = math.sqrt(variance)

        skewness = sum((x - mean) ** 3 for x in data) * n / ((n - 1) * (n - 2) * std_deviation ** 3) if std_deviation != 0 else 0

        return skewness
    
    @staticmethod
    def kurtosis(data):

        n = len(data)
        mean = sum(data) / n
        std_dev = math.sqrt(sum((x - mean) ** 2 for x in data) / n)

        if std_dev == 0:
            return math.nan

        centered_data = [(x - mean) for x in data]
        fourth_moment = sum(x ** 4 for x in centered_data) / n

        kurtosis_value = (fourth_moment / std_dev ** 4) - 3

        return kurtosis_value
    
    @staticmethod
    def pdf(data, mu, sigma):
        pdf_values = [1 / (sigma * math.sqrt(2 * math.pi)) * math.exp(-0.5 * ((x - mu) / sigma) ** 2) for x in data]
        return pdf_values
```

```cpp
#include <cmath>
#include <stdexcept>
#include <vector>

namespace {

const double PI = 3.14159265358979323846;

// Mirrors Python's ZeroDivisionError for float/int division by zero.
class ZeroDivisionError : public std::runtime_error {
public:
    explicit ZeroDivisionError(const char* message) : std::runtime_error(message) {}
};

}  // namespace

class DataStatistics4 {
public:
    static double correlation_coefficient(const std::vector<double>& data1,
                                          const std::vector<double>& data2) {
        const long long n = static_cast<long long>(data1.size());
        if (n == 0) {
            throw ZeroDivisionError("division by zero");  // sum([]) / 0
        }

        double sum1 = 0.0;
        double sum2 = 0.0;
        for (double v : data1) sum1 += v;
        for (double v : data2) sum2 += v;

        const double mean1 = sum1 / static_cast<double>(n);
        const double mean2 = sum2 / static_cast<double>(n);

        double numerator = 0.0;
        double sum_sq1 = 0.0;
        double sum_sq2 = 0.0;
        for (long long i = 0; i < n; ++i) {
            const double d1 = data1[static_cast<size_t>(i)] - mean1;
            const double d2 = data2.at(static_cast<size_t>(i)) - mean2;  // IndexError-equivalent
            numerator += d1 * d2;
            sum_sq1 += std::pow(d1, 2.0);
            sum_sq2 += std::pow(d2, 2.0);
        }

        const double denominator = std::sqrt(sum_sq1) * std::sqrt(sum_sq2);
        return denominator != 0.0 ? numerator / denominator : 0.0;
    }

    static double skewness(const std::vector<double>& data) {
        const long long n = static_cast<long long>(data.size());
        if (n == 0) {
            throw ZeroDivisionError("division by zero");  // sum([]) / 0
        }

        double sum = 0.0;
        for (double x : data) sum += x;
        const double mean = sum / static_cast<double>(n);

        double sum_sq = 0.0;
        for (double x : data) sum_sq += std::pow(x - mean, 2.0);
        const double variance = sum_sq / static_cast<double>(n);
        const double std_deviation = std::sqrt(variance);

        if (std_deviation != 0.0) {
            double sum_cube = 0.0;
            for (double x : data) sum_cube += std::pow(x - mean, 3.0);

            const double denominator =
                static_cast<double>((n - 1) * (n - 2)) * std::pow(std_deviation, 3.0);
            if (denominator == 0.0) {
                throw ZeroDivisionError("float division by zero");
            }
            return sum_cube * static_cast<double>(n) / denominator;
        }
        return 0.0;
    }

    static double kurtosis(const std::vector<double>& data) {
        const long long n = static_cast<long long>(data.size());
        if (n == 0) {
            throw ZeroDivisionError("division by zero");  // sum([]) / 0
        }

        double sum = 0.0;
        for (double x : data) sum += x;
        const double mean = sum / static_cast<double>(n);

        double sum_sq = 0.0;
        for (double x : data) sum_sq += std::pow(x - mean, 2.0);
        const double std_dev = std::sqrt(sum_sq / static_cast<double>(n));

        if (std_dev == 0.0) {
            return std::nan("");
        }

        double sum_fourth = 0.0;
        for (double x : data) sum_fourth += std::pow(x - mean, 4.0);
        const double fourth_moment = sum_fourth / static_cast<double>(n);

        const double denom = std::pow(std_dev, 4.0);
        if (denom == 0.0) {
            throw ZeroDivisionError("float division by zero");
        }
        return fourth_moment / denom - 3.0;
    }

    static std::vector<double> pdf(const std::vector<double>& data, double mu, double sigma) {
        std::vector<double> pdf_values;
        pdf_values.reserve(data.size());
        for (double x : data) {

            }
            pdf_values.push_back(1.0 / denom * std::exp(-0.5 * std::pow((x - mu) / sigma, 2.0)));
        }
        return pdf_values;
    }
};
```
