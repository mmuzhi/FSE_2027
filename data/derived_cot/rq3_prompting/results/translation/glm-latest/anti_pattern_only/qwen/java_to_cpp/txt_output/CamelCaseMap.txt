        return camelCaseString.toString();

}

// C++ translation of the Java CamelCaseMap.
//
// Mapping notes:
// - java.util.LinkedHashMap<String, Object>  -> std::vector<std::pair<std::string, std::any>>
//   preserving insertion order (re-put of an existing key updates the value
//   in place and keeps the original position).
// - Java's Object/null                        -> std::any (an empty std::any plays
//   the role of Java's null; get() therefore returns an empty std::any for a
//   missing key, just as Java returns null).
// - Java's String cannot be null in C++ (std::string), so the null-key branch of
//   _convertKey has no counterpart here.
class CamelCaseMap {
public:
    std::any get(const std::string& key) const {
        const std::string convertedKey = _convertKey(key);
        const auto it = findEntry(convertedKey);
        if (it == data.end()) {
            return std::any{}; // Java: data.get(...) == null when absent
        }
        return it->second;
    }

    void put(const std::string& key, std::any value) {
        const std::string convertedKey = _convertKey(key);
        const auto it = findEntry(convertedKey);
        if (it != data.end()) {
            it->second = std::move(value); // LinkedHashMap keeps the original insertion position
        } else {
            data.emplace_back(convertedKey, std::move(value));
        }
    }

    void remove(const std::string& key) {
        const std::string convertedKey = _convertKey(key);
        const auto it = findEntry(convertedKey);
        if (it != data.end()) {
            data.erase(it);
        }
    }

    // Package-private in Java; public here since C++ has no package access.
    std::string _convertKey(const std::string& key) const {
        // Java checks key == null here; std::string cannot be null.
        return _toCamelCase(key);
    }

    // Package-private (static) in Java; public here since C++ has no package access.
    static std::string _toCamelCase(const std::string& key) {
        const std::vector<std::string> parts = splitOnUnderscore(key);
        // Java: parts[0] throws ArrayIndexOutOfBoundsException when the split
        // produced no elements (key made up entirely of underscores, e.g. "_").
        std::string camelCaseString = parts.at(0);
        for (std::size_t i = 1; i < parts.size(); ++i) {
            const std::string& part = parts[i];
            // Java: part.substring(0, 1) throws StringIndexOutOfBoundsException
            // for an empty part (e.g. key "a__b").
            if (part.empty()) {
                throw std::out_of_range("String index out of range: 1");
            }
            camelCaseString += toUpper(part.substr(0, 1));
            camelCaseString += toLower(part.substr(1));
        }
        return camelCaseString;
    }

    // Java returns a live view (Set<String>) in insertion order;
    // the C++ counterpart returns the keys in insertion order by value.
    std::vector<std::string> keySet() const {
        std::vector<std::string> keys;
        keys.reserve(data.size());
        for (const auto& entry : data) {
            keys.push_back(entry.first);
        }
        return keys;
    }

    int size() const {
        return static_cast<int>(data.size());
    }

private:
    std::vector<std::pair<std::string, std::any>> data;

    std::vector<std::pair<std::string, std::any>>::const_iterator
    findEntry(const std::string& convertedKey) const {
        return std::find_if(data.begin(), data.end(),
                            [&convertedKey](const std::pair<std::string, std::any>& entry) {
                                return entry.first == convertedKey;
                            });
    }

    std::vector<std::pair<std::string, std::any>>::iterator
    findEntry(const std::string& convertedKey) {
        return std::find_if(data.begin(), data.end(),
                            [&convertedKey](const std::pair<std::string, std::any>& entry) {
                                return entry.first == convertedKey;
                            });
    }

    // Replicates Java's String.split("_") (limit 0): split on every '_',
    // drop trailing empty parts; if '_' never occurs, the result is the
    // whole string as a single element ([""] for an empty input).
    static std::vector<std::string> splitOnUnderscore(const std::string& s) {
        std::vector<std::string> parts;
        std::string current;
        bool found = false;
        for (char c : s) {
            if (c == '_') {
                found = true;
                parts.push_back(current);
                current.clear();
            } else {
                current.push_back(c);
            }
        }
        if (!found) {
            return {s};
        }
        parts.push_back(current);
        while (!parts.empty() && parts.back().empty()) {
            parts.pop_back();
        }
        return parts;
    }

    static std::string toUpper(const std::string& s) {
        std::string result(s);
        for (char& c : result) {
            c = static_cast<char>(std::toupper(static_cast<unsigned char>(c)));
        }
        return result;
    }

    static std::string toLower(const std::string& s) {
        std::string result(s);
        for (char& c : result) {
            c = static_cast<char>(std::tolower(static_cast<unsigned char>(c)));
        }
        return result;
    }
};

} // namespace example
} // namespace org
