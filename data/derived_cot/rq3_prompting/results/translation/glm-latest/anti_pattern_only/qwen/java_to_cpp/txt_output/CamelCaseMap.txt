public class CamelCaseMap {

        data.put(_convertKey(key), value);
    }

        data.remove(_convertKey(key));
    }

    String _convertKey(String key) {
        if (key == null) {
            return null;
        }
        return _toCamelCase(key);
    }

    static String _toCamelCase(String key) {
        String[] parts = key.split("_");
        StringBuilder camelCaseString = new StringBuilder(parts[0]);
        for (int i = 1; i < parts.length; i++) {

        return camelCaseString.toString();

    }

        return data.size();
    }

    Matcher m = matcher(input);

class CamelCaseMap {
public:
    using Key = std::optional<std::string>;   // nullable string key (Java: String, may be null)
    using Value = std::any;                    // Java: Object (empty any == null)

    // Returns the stored value, or an empty std::any (Java: null) if absent.
    Value get(const Key& key) const {
        auto it = index_.find(_convertKey(key));
        if (it != index_.end()) {
            return it->second->second;
        }
        return Value{};
    }

    void put(const Key& key, Value value) {
        Key converted = _convertKey(key);
        auto it = index_.find(converted);
        if (it != index_.end()) {
            // LinkedHashMap keeps the original insertion position on re-put.
            it->second->second = std::move(value);
        } else {
            entries_.emplace_back(converted, std::move(value));
            index_.emplace(converted, std::prev(entries_.end()));
        }
    }

    void remove(const Key& key) {
        auto it = index_.find(_convertKey(key));
        if (it != index_.end()) {
            entries_.erase(it->second);
            index_.erase(it);
        }
    }

    // Package-private in Java; public here (C++ has no package access).
    Key _convertKey(const Key& key) const {
        if (!key.has_value()) {
            return Key{};
        }
        return _toCamelCase(*key);
    }

    static std::string _toCamelCase(const std::string& key) {
        std::vector<std::string> parts = _splitUnderscore(key);
        if (parts.empty()) {
            // Java: parts[0] on a zero-length array -> ArrayIndexOutOfBoundsException
            throw std::out_of_range("Index 0 out of bounds for length 0");
        }
        std::string camelCaseString = parts[0];
        for (std::size_t i = 1; i < parts.size(); ++i) {
            const std::string& part = parts[i];
            if (part.empty()) {
                // Java: "".substring(0, 1) -> StringIndexOutOfBoundsException
                throw std::out_of_range("begin 0, end 1, length 0");
            }
            // part.substring(0, 1).toUpperCase()
            camelCaseString.push_back(
                static_cast<char>(std::toupper(static_cast<unsigned char>(part[0]))));
            // part.substring(1).toLowerCase()
            for (std::size_t j = 1; j < part.size(); ++j) {
                camelCaseString.push_back(
                    static_cast<char>(std::tolower(static_cast<unsigned char>(part[j]))));
            }
        }
        return camelCaseString;
    }

    std::vector<Key> keySet() const {
        std::vector<Key> keys;
        keys.reserve(entries_.size());
        for (const auto& entry : entries_) {
            keys.push_back(entry.first);
        }
        return keys;
    }

    int size() const {
        return static_cast<int>(entries_.size());

    using Entry = std::pair<Key, Value>;
    using EntryIterator = std::list<Entry>::iterator;

    // - no '_' at all -> single element with the whole string (even "")
    // - otherwise split on every '_', then drop trailing empty parts
    static std::vector<std::string> _splitUnderscore(const std::string& key) {
        if (key.find('_') == std::string::npos) {
            return {key};
        }
        std::vector<std::string> parts;
        std::string current;
        for (char c : key) {
            if (c == '_') {
                parts.push_back(current);
                current.clear();
            } else {
                current.push_back(c);
            }
        }
        parts.push_back(current);
        while (!parts.empty() && parts.back().empty()) {
            parts.pop_back();
        }
        return parts;
    }

    // Insertion-ordered map (LinkedHashMap equivalent).
    std::list<Entry> entries_;

class CamelCaseMap {
public:
    using Key = std::optional<std::string>;
    using Value = std::any;

    // Returns the stored value, or an empty std::any (Java: null) if absent.
    Value get(const Key& key) const {
        auto it = index_.find(_convertKey(key));
        if (it != index_.end()) {
            return it->second->second;
        }
        return Value{};
    }

        Key converted = _convertKey(key);
        auto it = index_.find(converted);
        if (it != index_.end()) {
            // LinkedHashMap: re-inserting an existing key keeps its position.
            it->second->second = std::move(value);

            index_.emplace(converted, entryIt);
        }

        if (it != index_.end()) {

            parts.pop_back();
