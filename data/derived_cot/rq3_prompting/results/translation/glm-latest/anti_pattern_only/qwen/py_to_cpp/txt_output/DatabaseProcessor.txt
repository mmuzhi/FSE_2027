private:
    std::string database_name_;

    // RAII wrapper around sqlite3*, mirroring sqlite3.connect()/close().
    class Connection {
    public:
        explicit Connection(const std::string& name) {
            if (sqlite3_open(name.c_str(), &handle_) != SQLITE_OK) {
                const std::string message =
                    handle_ ? sqlite3_errmsg(handle_) : "unable to open database";
                if (handle_) sqlite3_close(handle_);
                handle_ = nullptr;
                throw std::runtime_error(message);
            }
        }
        ~Connection() {
            if (handle_) sqlite3_close(handle_);
        }
        Connection(const Connection&) = delete;
        Connection& operator=(const Connection&) = delete;

        sqlite3* get() const { return handle_; }

        void execute(const std::string& sql) {
            char* error_message = nullptr;
            if (sqlite3_exec(handle_, sql.c_str(), nullptr, nullptr,
                             &error_message) != SQLITE_OK) {
                const std::string message =
                    error_message ? error_message : sqlite3_errmsg(handle_);
                sqlite3_free(error_message);
                throw std::runtime_error(message);
            }
        }

    private:
        sqlite3* handle_ = nullptr;
    };

    // RAII wrapper around sqlite3_stmt*.
    class Statement {
    public:
        Statement(sqlite3* connection, const std::string& sql) {
            if (sqlite3_prepare_v2(connection, sql.c_str(), -1, &handle_,
                                   nullptr) != SQLITE_OK) {
                handle_ = nullptr;
                throw std::runtime_error(sqlite3_errmsg(connection));
            }
        }
        ~Statement() {
            if (handle_) sqlite3_finalize(handle_);
        }
        Statement(const Statement&) = delete;
        Statement& operator=(const Statement&) = delete;

        sqlite3_stmt* get() const { return handle_; }

    private:
        sqlite3_stmt* handle_ = nullptr;
    };

    // Bind a Python-like value (str -> TEXT, int -> INTEGER).
    static void bind(sqlite3_stmt* stmt, int index, const Value& value) {
        int rc;
        if (const std::string* text = std::get_if<std::string>(&value)) {
            rc = sqlite3_bind_text(stmt, index, text->c_str(),
                                   static_cast<int>(text->size()),
                                   SQLITE_TRANSIENT);
        } else {
            rc = sqlite3_bind_int64(stmt, index, std::get<long long>(value));
        }
        if (rc != SQLITE_OK) {
            throw std::runtime_error("binding parameter failed");
        }
    }

    static std::string column_text(sqlite3_stmt* stmt, int index) {
        const unsigned char* text = sqlite3_column_text(stmt, index);
        return text ? std::string(reinterpret_cast<const char*>(text)) : std::string();
    }

// This is a class for processing a database, supporting to create tables,
// insert data into the database, search for data based on name, and delete
// data from the database.
//
// Requires linking against SQLite 3: -lsqlite3

#include <sqlite3.h>

#include <map>
#include <optional>
#include <stdexcept>
#include <string>
#include <tuple>
#include <utility>
#include <variant>
#include <vector>

class DatabaseProcessor {
public:
    // Equivalent of the Python dict values used for a row of data:
    // either a str (TEXT) or an int (INTEGER).
    using Value = std::variant<std::string, long long>;
    // Equivalent of one Python dict, e.g. {'name': 'John', 'age': 25}.
    using Row = std::map<std::string, Value>;
    // One row of a search result: (id, name, age), e.g. (1, "John", 25).
    using ResultRow = std::tuple<long long, std::string, long long>;

    // Initialize database name of database processor.
    explicit DatabaseProcessor(std::string database_name)
        : database_name_(std::move(database_name)) {}

    const std::string& database_name() const { return database_name_; }

    // Create a new table in the database if it doesn't exist.
    // id (INTEGER) is the PRIMARY KEY, key1 is TEXT and key2 is INTEGER.
    void create_table(const std::string& table_name, const std::string& key1,
                      const std::string& key2) {
        Connection connection(database_name_);
        connection.execute("CREATE TABLE IF NOT EXISTS " + table_name +
                           " (id INTEGER PRIMARY KEY, " + key1 + " TEXT, " +
                           key2 + " INTEGER)");
    }

    // Insert data into the specified table in the database.
    // data: a list of dictionaries where each dictionary represents a row.
    void insert_into_database(const std::string& table_name,
                              const std::vector<Row>& data) {
        Connection connection(database_name_);
        // Python's sqlite3 wraps DML statements in a transaction that is
        // only committed by conn.commit(); replicate this so a failure part
        // way through leaves no partial inserts behind.
        connection.execute("BEGIN");
        {
            const std::string query =
                "INSERT INTO " + table_name + " (name, age) VALUES (?, ?)";
            Statement statement(connection.get(), query);
            for (const Row& item : data) {
                sqlite3_reset(statement.get());
                // Raises (like KeyError) if the key is missing.
                const Value& name = item.at("name");
                const Value& age = item.at("age");
                bind(statement.get(), 1, name);
                bind(statement.get(), 2, age);
                if (sqlite3_step(statement.get()) != SQLITE_DONE) {
                    throw std::runtime_error(sqlite3_errmsg(connection.get()));
                }
            }
        }
        connection.execute("COMMIT");
    }

    // Search the specified table for rows with a matching name.
    // Returns the matching rows, or std::nullopt (None) if there are none.
    std::optional<std::vector<ResultRow>> search_database(
        const std::string& table_name, const std::string& name) {
        Connection connection(database_name_);
        const std::string query = "SELECT * FROM " + table_name + " WHERE name = ?";
        Statement statement(connection.get(), query);
        bind(statement.get(), 1, name);

        std::vector<ResultRow> result;
        int status = sqlite3_step(statement.get());
        while (status == SQLITE_ROW) {
            result.emplace_back(sqlite3_column_int64(statement.get(), 0),
                                column_text(statement.get(), 1),
                                sqlite3_column_int64(statement.get(), 2));
            status = sqlite3_step(statement.get());
        }
        if (status != SQLITE_DONE) {
            throw std::runtime_error(sqlite3_errmsg(connection.get()));
        }

        if (result.empty()) {
            return std::nullopt;
        }
        return result;
    }

    // Delete rows from the specified table with a matching name.
    void delete_from_database(const std::string& table_name,
                              const std::string& name) {
        Connection connection(database_name_);
        const std::string query = "DELETE FROM " + table_name + " WHERE name = ?";
        Statement statement(connection.get(), query);
        bind(statement.get(), 1, name);
        if (sqlite3_step(statement.get()) != SQLITE_DONE) {
            throw std::runtime_error(sqlite3_errmsg(connection.get()));
        }
    }

private:
    std::string database_name_;

    // RAII wrapper around sqlite3*, mirroring sqlite3.connect()/conn.close().
    class Connection {
    public:
        explicit Connection(const std::string& database_name) {
            if (sqlite3_open(database_name.c_str(), &handle_) != SQLITE_OK) {
                const std::string message =
                    handle_ != nullptr ? sqlite3_errmsg(handle_)
                                       : "unable to open database file";
                if (handle_ != nullptr) {
                    sqlite3_close(handle_);
                }
                handle_ = nullptr;
                throw std::runtime_error(message);
            }
        }

        ~Connection() {
            if (handle_ != nullptr) {
                sqlite3_close(handle_);
            }
        }

        Connection(const Connection&) = delete;
        Connection& operator=(const Connection&) = delete;

        sqlite3* get() const { return handle_; }

        void execute(const std::string& sql) {
            char* error_message = nullptr;
            if (sqlite3_exec(handle_, sql.c_str(), nullptr, nullptr,
                             &error_message) != SQLITE_OK) {
                const std::string message =
                    error_message != nullptr ? error_message
                                             : sqlite3_errmsg(handle_);
                sqlite3_free(error_message);
                throw std::runtime_error(message);
            }
        }

    private:
        sqlite3* handle_ = nullptr;
    };

    // RAII wrapper around sqlite3_stmt*.
    class Statement {
    public:
        Statement(sqlite3* connection, const std::string& sql) {
            if (sqlite3_prepare_v2(connection, sql.c_str(), -1, &handle_,
                                   nullptr) != SQLITE_OK) {
                handle_ = nullptr;
                throw std::runtime_error(sqlite3_errmsg(connection));
            }
        }

        ~Statement() {
            if (handle_ != nullptr) {
                sqlite3_finalize(handle_);
            }
        }

        Statement(const Statement&) = delete;
        Statement& operator=(const Statement&) = delete;

        sqlite3_stmt* get() const { return handle_; }

    private:
        sqlite3_stmt* handle_ = nullptr;
    };

    // Bind a value the way Python's sqlite3 does: str -> TEXT, int -> INTEGER.
    static void bind(sqlite3_stmt* statement, int index, const Value& value) {
        int status = SQLITE_OK;
        if (const std::string* text = std::get_if<std::string>(&value)) {
            status = sqlite3_bind_text(statement, index, text->c_str(),
                                       static_cast<int>(text->size()),
                                       SQLITE_TRANSIENT);
        } else if (const long long* integer = std::get_if<long long>(&value)) {
            status = sqlite3_bind_int64(statement, index, *integer);
        }
        if (status != SQLITE_OK) {
            throw std::runtime_error("unable to bind parameter " +
                                     std::to_string(index));
        }
    }

    static std::string column_text(sqlite3_stmt* statement, int index) {
        const unsigned char* text = sqlite3_column_text(statement, index);
        return text != nullptr
                   ? std::string(reinterpret_cast<const char*>(text))
                   : std::string();
    }
};
