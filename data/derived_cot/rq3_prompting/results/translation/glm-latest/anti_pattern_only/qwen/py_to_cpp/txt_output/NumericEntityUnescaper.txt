class NumericEntityUnescaper {
public:

        std::string out;
        size_t pos = 0;
        size_t length = str.length();

        while (pos + 2 < length) {
            if (str[pos] == '&' && str[pos + 1] == '#') {
                size_t start = pos + 2;
                bool is_hex = false;
                char first_char = str[start];

                if (first_char == 'x' || first_char == 'X') {
                    start += 1;
                    is_hex = true;
                }

                if (start == length) {
                    return out;
                }

                size_t end = start;
                while (end < length && is_hex_char(str[end])) {
                    end += 1;
                }

                if (end < length && str[end] == ';') {

                    // int() fails when the slice is empty, or (base 10) when

                        for (size_t i = start; i < end; ++i) {
                            if (str[i] < '0' || str[i] > '9') {
                                parseable = false;
                                break;
                            }
                        }
                    }
                    if (!parseable) {
                        return out;
                    }

                    unsigned long long entity_value = 0;
                    bool out_of_range = false;
                    unsigned base = is_hex ? 16u : 10u;
                    for (size_t i = start; i < end; ++i) {
                        int d;

                        if (c >= '0' && c <= '9') {
