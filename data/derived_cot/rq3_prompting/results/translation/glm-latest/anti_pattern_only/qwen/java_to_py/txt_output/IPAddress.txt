package org.example;

import java.util.ArrayList;
import java.util.List;

public class IPAddress {
    private String ipAddress;

    public IPAddress(String ipAddress) {
        this.ipAddress = ipAddress;
    }

    public boolean isValid() {
        String[] octets = ipAddress.split("\\.");
        if (octets.length != 4) {
            return false;
        }
        for (String octet : octets) {
            try {
                int num = Integer.parseInt(octet);
                if (num < 0 || num > 255) {
                    return false;
                }
            } catch (NumberFormatException e) {
                return false;
            }
        }
        return true;
    }

    public List<String> getOctets() {
        List<String> octetsList = new ArrayList<>();
        if (isValid()) {
            String[] octets = ipAddress.split("\\.");
            for (String octet : octets) {
                octetsList.add(octet);
            }
        }
        return octetsList;
    }

    public String getBinary() {
        if (isValid()) {
            StringBuilder binaryString = new StringBuilder();
            String[] octets = ipAddress.split("\\.");
            for (String octet : octets) {
                int num = Integer.parseInt(octet);
                String binaryOctet = String.format("%08d", Integer.parseInt(Integer.toBinaryString(num)));
                binaryString.append(binaryOctet).append(".");
            }
            return binaryString.substring(0, binaryString.length() - 1);
        }
        return "";
    }
}

import re


class IPAddress:
    # Emulates Java's Integer.parseInt grammar: an optional sign followed by
    # one or more digits (no whitespace, no underscores, no separators).
    _INT_PATTERN = re.compile(r"[+-]?\d+\Z")

    def __init__(self, ip_address):
        self._ip_address = ip_address

    def _split_octets(self):
        # Java's String.split("\\.") drops trailing empty strings and returns
        # the whole input (as a single element) when no separator is present.
        if "." not in self._ip_address:
            return [self._ip_address]
        parts = self._ip_address.split(".")
        while parts and parts[-1] == "":
            parts.pop()
        return parts

import re


class IPAddress:
    # Grammar accepted by Java's Integer.parseInt: an optional ASCII sign
    # followed by one or more decimal digits.  This rejects surrounding
    # whitespace and underscore separators, which Python's int() would
    # otherwise silently accept.
    _INT_RE = re.compile(r"[+-]?\d+")

    def __init__(self, ip_address):
        self._ip_address = ip_address

    def _split_octets(self):
        # Emulate Java's String.split("\\."): trailing empty strings are
        # dropped, and a string containing no separator yields the whole
        # string as a single element.
        if "." not in self._ip_address:
            return [self._ip_address]
        parts = self._ip_address.split(".")
        while parts and parts[-1] == "":
            parts.pop()
        return parts

    @classmethod
    def _parse_int(cls, value):
        # Mirrors Integer.parseInt, raising ValueError (the Python analogue
        # of NumberFormatException) for anything outside its grammar.
        if cls._INT_RE.fullmatch(value) is None:
            raise ValueError(f'For input string: "{value}"')
        return int(value)

    def is_valid(self):
        octets = self._split_octets()
        if len(octets) != 4:
            return False
        for octet in octets:
            try:
                num = self._parse_int(octet)
                if num < 0 or num > 255:
                    return False
            except ValueError:
                return False
        return True

    def get_octets(self):
        if self.is_valid():
            return self._split_octets()
        return []

    def get_binary(self):
