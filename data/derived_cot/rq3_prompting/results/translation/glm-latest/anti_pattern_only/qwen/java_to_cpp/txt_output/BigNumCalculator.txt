```java
public class BigNumCalculator {

    public static String add(String num1, String num2) {
        int maxLength = Math.max(num1.length(), num2.length());
```

```cpp
#include <algorithm>
#include <iostream>
#include <string>
#include <vector>

namespace org {
namespace example {

// Equivalent of Character.getNumericValue(char) for ASCII characters:
// '0'-'9' -> 0-9, 'a'-'z'/'A'-'Z' -> 10-35, otherwise -1.
static int getNumericValue(char c) {
    if (c >= '0' && c <= '9') {
        return c - '0';
    }
    if (c >= 'a' && c <= 'z') {
        return c - 'a' + 10;
    }
    if (c >= 'A' && c <= 'Z') {
        return c - 'A' + 10;
    }
    return -1;
}

// Equivalent of String.format("%" + width + "s", s).replace(' ', '0'):
// right-align s within the given width, then turn every space into '0'.
static std::string formatAndZeroPad(const std::string& s, std::size_t width) {
    std::string formatted = s;
    if (formatted.length() < width) {
        formatted.insert(0, width - formatted.length(), ' ');
    }
    std::replace(formatted.begin(), formatted.end(), ' ', '0');
    return formatted;
}

std::string add(std::string num1, std::string num2) {
    std::size_t maxLength = std::max(num1.length(), num2.length());
    num1 = formatAndZeroPad(num1, maxLength);
    num2 = formatAndZeroPad(num2, maxLength);

    int carry = 0;
    std::string result;
    for (int i = static_cast<int>(maxLength) - 1; i >= 0; i--) {
        int digitSum = getNumericValue(num1[i]) + getNumericValue(num2[i]) + carry;
        carry = digitSum / 10;
        int digit = digitSum % 10;
        result.insert(0, std::to_string(digit));
    }

    if (carry > 0) {
        result.insert(0, std::to_string(carry));
    }

    return result;
}

std::string subtract(std::string num1, std::string num2) {
    bool negative = false;
    if (num1.length() < num2.length() ||
        (num1.length() == num2.length() && num1 < num2)) {
        std::swap(num1, num2);
        negative = true;
    }

    std::size_t maxLength = std::max(num1.length(), num2.length());
    num1 = formatAndZeroPad(num1, maxLength);
    num2 = formatAndZeroPad(num2, maxLength);

    int borrow = 0;
    std::string result;
    for (int i = static_cast<int>(maxLength) - 1; i >= 0; i--) {
        int digitDiff = getNumericValue(num1[i]) - getNumericValue(num2[i]) - borrow;

        if (digitDiff < 0) {
            digitDiff += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }

        result.insert(0, std::to_string(digitDiff));
    }

    while (result.length() > 1 && result[0] == '0') {
        result.erase(0, 1);
    }

    if (negative) {
        result.insert(0, 1, '-');
    }

    return result;
}

std::string multiply(const std::string& num1, const std::string& num2) {
    int len1 = static_cast<int>(num1.length());
    int len2 = static_cast<int>(num2.length());
    std::vector<int> result(len1 + len2, 0);

    for (int i = len1 - 1; i >= 0; i--) {
        for (int j = len2 - 1; j >= 0; j--) {
            int mul = getNumericValue(num1[i]) * getNumericValue(num2[j]);
            int p1 = i + j;
            int p2 = i + j + 1;
            int total = mul + result[p2];

            result[p1] += total / 10;
            result[p2] = total % 10;
        }
    }

    std::string sb;
    int start = 0;
    while (start < static_cast<int>(result.size()) - 1 && result[start] == 0) {
        start++;
    }

    for (std::size_t i = static_cast<std::size_t>(start); i < result.size(); i++) {
        sb += std::to_string(result[i]);
    }

    return sb;
}

} // namespace example
} // namespace org

int main() {

    std::cout << org::example::subtract("12345678901234567890", "98765432109876543210") << std::endl;
```
