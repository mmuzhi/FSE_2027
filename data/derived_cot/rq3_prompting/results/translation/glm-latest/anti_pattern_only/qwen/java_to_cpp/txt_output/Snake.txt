    final int SCREEN_HEIGHT;

        this.SCREEN_HEIGHT = SCREEN_HEIGHT;

            }

class Snake {
public:
    class Position {
    public:

        int getY() const { return y; }

        bool operator==(const Position& other) const {
            return x == other.x && y == other.y;
        }

            return !(*this == other);
        }

            return static_cast<int>(31u * static_cast<unsigned int>(x)

        }

    private:
        int x;
        int y;
    };

    public:

            // (nanoTime ^ seedUniquifier); approximate with clock-based seeding.

                return static_cast<int>((static_cast<int64_t>(bound) * r) >> 31);

                bits = next(31);
                val = bits % bound;

                // exactly with a 64-bit comparison against INT32_MAX.
            } while (static_cast<int64_t>(bits) - val + m
                     > static_cast<int64_t>(std::numeric_limits<int>::max()));
            return val;
        }

    private:
        uint64_t seed;

        int next(int bits) {
            seed = (seed * 0x5DEECE66DULL + 0xBULL) & ((1ULL << 48) - 1);
            return static_cast<int>(static_cast<uint32_t>(seed >> (48 - bits)));
        }
    };

- `java.util.Random` was ported exactly (48-bit LCG, `nextInt(bound)` rejection-sampling with overflow semantics), since `randomFoodPosition()` depends on it directly; the no-arg constructor seeds nondeterministically, as in Java.
- `positions.at(0)` mirrors `positions.get(0)` throwing on an empty list; `std::find` over the range starting at index 2 mirrors `subList(2, size).contains(...)`.
- `getPositions()` returns a live reference, matching Java's returned list reference.
- Integer `%` and `/` semantics match Java (truncation toward zero), so wraparound behavior for negative coordinates is preserved.
