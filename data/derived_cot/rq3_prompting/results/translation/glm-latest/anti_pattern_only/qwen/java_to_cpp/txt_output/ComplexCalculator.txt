package org.example;

public class ComplexCalculator {

    public ComplexNumber add(ComplexNumber c1, ComplexNumber c2) {
        double real = c1.getReal() + c2.getReal();
        double imaginary = c1.getImaginary() + c2.getImaginary();
        return new ComplexNumber(real, imaginary);
    }

    public ComplexNumber subtract(ComplexNumber c1, ComplexNumber c2) {
        double real = c1.getReal() - c2.getReal();
        double imaginary = c1.getImaginary() - c2.getImaginary();
        return new ComplexNumber(real, imaginary);
    }

    public ComplexNumber multiply(ComplexNumber c1, ComplexNumber c2) {
        double real = c1.getReal() * c2.getReal() - c1.getImaginary() * c2.getImaginary();
        double imaginary = c1.getReal() * c2.getImaginary() + c1.getImaginary() * c2.getReal();
        return new ComplexNumber(real, imaginary);
    }

    public ComplexNumber divide(ComplexNumber c1, ComplexNumber c2) {
        double denominator = c2.getReal() * c2.getReal() + c2.getImaginary() * c2.getImaginary();
        double real = (c1.getReal() * c2.getReal() + c1.getImaginary() * c2.getImaginary()) / denominator;
        double imaginary = (c1.getImaginary() * c2.getReal() - c1.getReal() * c2.getImaginary()) / denominator;
        return new ComplexNumber(real, imaginary);
    }

    public static class ComplexNumber {
        private double real;
        private double imaginary;

        public ComplexNumber(double real, double imaginary) {
            this.real = real;
            this.imaginary = imaginary;
        }

        public double getReal() {
            return real;
        }

        public double getImaginary() {
            return imaginary;
        }

        @Override
        public boolean equals(Object obj) {
            if (this == obj) return true;
            if (obj == null || getClass() != obj.getClass()) return false;
            ComplexNumber that = (ComplexNumber) obj;
            return Double.compare(that.real, real) == 0 && Double.compare(that.imaginary, imaginary) == 0;
        }

        @Override
        public String toString() {
            return real + (imaginary >= 0 ? "+" : "") + imaginary + "j";
        }
    }
}

   #include <cstdlib>

   namespace org { namespace example {
   
   namespace detail {
   
   // Produces the same string as Java's Double.toString(double).
   inline std::string javaDoubleToString(double d) {
       if (std::isnan(d)) return "NaN";
       if (std::isinf(d)) return d > 0 ? "Infinity" : "-Infinity";
       
       const bool neg = std::signbit(d);
       const double a = std::fabs(d);
       
       if (a == 0.0) return neg ? "-0.0" : "0.0";
       
       // Find the shortest decimal digit string (correctly rounded) that
       // round-trips to `a`, in the form "d.dddde±XX".
       char buf[64];
       for (int prec = 1; prec <= 17; ++prec) {
           std::snprintf(buf, sizeof buf, "%.*e", prec - 1, a);
           if (std::strtod(buf, nullptr) == a) break;
       }
       
       // Parse "[-]d[.ddd]e±XX".
       const char* p = buf;
       std::string digits;
       // p points at first digit (a > 0 so no '-').
       digits.push_back(*p++);
       if (*p == '.') {
           ++p;
           while (*p != 'e' && *p != 'E') digits.push_back(*p++);
       }
       // skip 'e'
       ++p;
       int exp10 = std::atoi(p);
       
       // digits = d1...dn, a = d1.d2...dn * 10^exp10
       const int n = static_cast<int>(digits.size());
       std::string out;
       if (neg) out.push_back('-');
       
       if (exp10 >= -3 && exp10 < 7) {
           // Plain decimal notation.
           if (exp10 >= 0) {
               if (exp10 + 1 >= n) {
                   // d1..dn followed by (exp10+1-n) zeros, then ".0"
                   out += digits;
                   out.append(exp10 + 1 - n, '0');
                   out += ".0";
               } else {
                   // d1..d(exp10+1) '.' d(exp10+2)..dn
                   out.append(digits, 0, exp10 + 1);
                   out.push_back('.');
                   out.append(digits, exp10 + 1, std::string::npos);
               }
           } else {
               // "0." + (-exp10-1) zeros + digits
               out += "0.";
               out.append(-exp10 - 1, '0');
               out += digits;
           }
       } else {
           // Computerized scientific notation: d1 '.' d2..dn 'E' exp10
           out.push_back(digits[0]);
           out.push_back('.');
           if (n > 1) out.append(digits, 1, std::string::npos);
           else out.push_back('0');
           out.push_back('E');
           out += std::to_string(exp10);
       }
       return out;
   }

       if (a != b) return false;
       if (a == 0.0) return std::signbit(a) == std::signbit(b);

       if (std::isnan(d)) return "NaN";
       if (std::isinf(d)) return d > 0.0 ? "Infinity" : "-Infinity";
       
       const bool negative = std::signbit(d);
       const double magnitude = std::fabs(d);
       if (magnitude == 0.0) return negative ? "-0.0" : "0.0";
       
       // Shortest round-trip digits in scientific form: "d.ddde±XX".
       char buffer[64];
       std::to_chars_result result = std::to_chars(buffer, buffer + sizeof(buffer), magnitude, std::chars_format::scientific);
       const std::string sci(buffer, result.ptr);
       
       // Split significand and exponent.
       std::string::size_type ePos = sci.find('e');
       std::string digits = sci.substr(0, ePos);
       const int exponent = std::atoi(sci.c_str() + ePos + 1);
       
       // Remove the decimal point.
       std::string::size_type dotPos = digits.find('.');
       if (dotPos != std::string::npos) digits.erase(dotPos, 1);
       
       const std::string::size_type digitCount = digits.size();
       std::string out = negative ? "-" : "";
       
       if (exponent >= -3 && exponent <= 6) {
           if (exponent >= 0) {
               if (static_cast<std::string::size_type>(exponent) + 1 >= digitCount) {
                   out += digits;
                   out.append(exponent + 1 - digitCount, '0');
                   out += ".0";
               } else {
                   out.append(digits, 0, exponent + 1);
                   out += '.';
                   out.append(digits, exponent + 1, std::string::npos);

               out.append(-exponent - 1, '0');

           out += 'E';

       }
