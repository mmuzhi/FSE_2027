        self.required = set()
        self.types = {}

                key_value = arg[2:].split('=')
                if len(key_value) == 2:

                else:
                    self.arguments[key_value[0]] = True
            elif arg.startswith('-'):
                key = arg[1:]
                if i + 1 < len(args) and not args[i + 1].startswith('-'):
                    self.arguments[key] = self._convert_type(key, args[i + 1])
                else:
                    self.arguments[key] = True
        missing_args = self.required - set(self.arguments.keys())
        if missing_args:
            return False, missing_args

        return True, None

    def get_argument(self, key):
        return self.arguments.get(key)

    def add_argument(self, arg, required=False, arg_type=str):
        if required:

        self.types[arg] = arg_type

    def _convert_type(self, arg, value):
        try:
            return self.types[arg](value)
        except (ValueError, KeyError):
            return value

    // Returns (true, {}) on success, or (false, missing_args) when required
    // arguments are missing.
    std::pair<bool, std::set<std::string>> parse_arguments(const std::string& command_string) {
        std::vector<std::string> args = split_tokens(command_string);
        if (!args.empty()) {
            args.erase(args.begin());  // [1:] -- drop the program name
        }

        for (std::size_t i = 0; i < args.size(); ++i) {
            const std::string& arg = args[i];

                const std::string body = arg.substr(2);
                const std::size_t eq = body.find('=');
                if (eq == std::string::npos) {
                    // key_value == [body] -> flag
                    arguments[body] = true;
                } else {
                    const std::string key = body.substr(0, eq);
                    const std::string value = body.substr(eq + 1);
                    if (value.find('=') != std::string::npos) {
                        // key_value has more than 2 parts -> flag
                        arguments[key] = true;
                    } else {
                        arguments[key] = convert_type(key, value);
                    }
                }

                    arguments[key] = convert_type(key, args[i + 1]);
                } else {
                    arguments[key] = true;
                }
            }
        }

        for (const std::string& req : required) {
            if (arguments.find(req) == arguments.end()) {
                missing_args.insert(req);
            }

        }
        return {true, {}};

        }

        if (required) {
            this->required.insert(arg);
        }
        types[arg] = arg_type;
    }

    // Falls back to the raw string when the argument is unknown or the
    // conversion fails (Python: except (ValueError, KeyError)).
    Value convert_type(const std::string& arg, const std::string& value) const {
        auto it = types.find(arg);
        if (it == types.end()) {
            return value;  // KeyError -> return value unchanged
        }
        try {
            switch (it->second) {
                case Type::Str:
                    return value;  // str(value) on a str is the identity
                case Type::Int:
                    return parse_int(value);
                case Type::Float:
                    return parse_float(value);
                case Type::Bool:
                    return !value.empty();  // bool(s) is True for non-empty strings
            }
        } catch (const std::exception&) {
            return value;  // ValueError -> return value unchanged
        }
        return value;
    }

    const std::map<std::string, Value>& get_arguments() const { return arguments; }
    const std::set<std::string>& get_required() const { return required; }
    const std::map<std::string, Type>& get_types() const { return types; }

private:
    std::map<std::string, Value> arguments;
    std::set<std::string> required;
    std::map<std::string, Type> types;

    static bool starts_with(const std::string& s, const std::string& prefix) {
        return s.size() >= prefix.size() && s.compare(0, prefix.size(), prefix) == 0;
    }

    static std::vector<std::string> split_tokens(const std::string& s) {
        std::vector<std::string> tokens;
        std::istringstream iss(s);
        std::string token;
        while (iss >> token) {
            tokens.push_back(token);
        }
        return tokens;
    }

    // Python int(s): optional sign, digits, surrounding whitespace ignored,
    // everything else is a ValueError.
    static int parse_int(const std::string& value) {
        const std::string trimmed = strip(value);
        if (trimmed.empty()) {
            throw std::invalid_argument("invalid literal for int()");
        }
        std::size_t pos = 0;
        if (trimmed[0] == '+' || trimmed[0] == '-') {
            pos = 1;
        }
        if (pos >= trimmed.size()) {
            throw std::invalid_argument("invalid literal for int()");
        }
        for (std::size_t i = pos; i < trimmed.size(); ++i) {
            if (!std::isdigit(static_cast<unsigned char>(trimmed[i]))) {
                throw std::invalid_argument("invalid literal for int()");
            }
        }
        return std::stoi(trimmed);
    }

    static double parse_float(const std::string& value) {
        const std::string trimmed = strip(value);
        if (trimmed.empty()) {
            throw std::invalid_argument("invalid literal for float()");
        }
        std::size_t pos = 0;
        double result = std::stod(trimmed, &pos);
        if (pos != trimmed.size()) {
            throw std::invalid_argument("invalid literal for float()");
        }
        return result;
    }

    static std::string strip(const std::string& s) {
        std::size_t begin = 0;
        std::size_t end = s.size();
        while (begin < end && std::isspace(static_cast<unsigned char>(s[begin]))) ++begin;
        while (end > begin && std::isspace(static_cast<unsigned char>(s[end - 1]))) --end;
        return s.substr(begin, end - begin);
    }
};

// A class for parsing command line arguments.
//
// Values are dynamically typed (mirroring Python's behaviour):
//   - flags are stored as bool (true)
//   - "--key=value" / "-key value" are converted with the declared type
class ArgumentParser {
public:
    // Argument type specifiers (Python builtins: str, int, float, bool).
    enum class Type { Str, Int, Float, Bool };

    // A dynamically typed argument value.
    using Value = std::variant<std::string, bool, int, double>;

    // Mirrors the public Python attributes.
    std::map<std::string, Value> arguments;
    std::set<std::string> required;
    std::map<std::string, Type> types;

    ArgumentParser() = default;

    // Parses the given command line argument string and stores the parsed
    // result (converted to the declared type) in `arguments`.
    // Returns {true, {}} on success, or {false, missing} when required
    // arguments are missing.
    std::pair<bool, std::set<std::string>> parse_arguments(const std::string& command_string) {
        std::vector<std::string> args = tokenize(command_string);
        if (!args.empty()) {
            args.erase(args.begin());  // drop the leading token, i.e. [1:]
        }

        for (std::size_t i = 0; i < args.size(); ++i) {
            const std::string& arg = args[i];

            if (starts_with(arg, "--")) {
                const std::string body = arg.substr(2);
                const std::size_t eq = body.find('=');
                if (eq == std::string::npos) {
                    // No '=': "--flag" -> flag = True
                    arguments[body] = true;
                } else {
                    const std::string key = body.substr(0, eq);
                    const std::string rest = body.substr(eq + 1);
                    if (rest.find('=') != std::string::npos) {
                        // More than one '=': "--a=b=c" -> a = True
                        arguments[key] = true;
                    } else {
                        arguments[key] = _convert_type(key, rest);
                    }
                }
            } else if (starts_with(arg, "-")) {
                const std::string key = arg.substr(1);
                if (i + 1 < args.size() && !starts_with(args[i + 1], "-")) {
                    arguments[key] = _convert_type(key, args[i + 1]);
                } else {
                    arguments[key] = true;
                }
            }
        }

        // missing_args = self.required - set(self.arguments.keys())
        std::set<std::string> missing_args;
        for (const std::string& name : required) {
            if (arguments.find(name) == arguments.end()) {
                missing_args.insert(name);
            }
        }
        if (!missing_args.empty()) {
            return {false, missing_args};
        }
        return {true, {}};
    }

    // Retrieves the value of the specified argument, or std::nullopt if the
    // argument does not exist.
    std::optional<Value> get_argument(const std::string& key) const {
        const auto it = arguments.find(key);
        if (it == arguments.end()) {
            return std::nullopt;
        }
        return it->second;
    }

    // Declares an argument: registers its type; adds it to `required` when
    // `required` is true.
    void add_argument(const std::string& arg, bool required = false, Type arg_type = Type::Str) {
        if (required) {
            this->required.insert(arg);
        }
        types[arg] = arg_type;
    }

    // Tries to convert the raw string `value` using the declared type of
    // `arg`. Returns the raw string when the argument is unknown (KeyError)
    // or the conversion fails (ValueError).
    Value _convert_type(const std::string& arg, const std::string& value) const {
        const auto it = types.find(arg);
        if (it == types.end()) {
            return value;  // KeyError: unknown argument
        }
        try {
            switch (it->second) {
                case Type::Str:
                    return value;                 // str(value) on a str
                case Type::Int:
                    return python_int(value);     // int(value)
                case Type::Float:
                    return python_float(value);   // float(value)
                case Type::Bool:
                    return !value.empty();        // bool(value)
            }
        } catch (const std::exception&) {
            return value;  // ValueError: conversion failed
        }
        return value;  // unreachable
    }

private:
    static bool starts_with(const std::string& s, const std::string& prefix) {
        return s.size() >= prefix.size() && s.compare(0, prefix.size(), prefix) == 0;
    }

    // str.split(): splits on runs of whitespace, no empty tokens.
    static std::vector<std::string> tokenize(const std::string& s) {
        std::vector<std::string> tokens;
        std::istringstream iss(s);
        std::string token;
        while (iss >> token) {
            tokens.push_back(token);
        }
        return tokens;
    }

    static std::string trim(const std::string& s) {
        std::size_t b = 0;
        std::size_t e = s.size();
        while (b < e && std::isspace(static_cast<unsigned char>(s[b]))) ++b;
        while (e > b && std::isspace(static_cast<unsigned char>(s[e - 1]))) --e;
        return s.substr(b, e - b);
    }

    // int(s): optional sign + digits, surrounding whitespace ignored.
    static int python_int(const std::string& value) {
        const std::string t = trim(value);
        std::size_t start = 0;
        if (!t.empty() && (t[0] == '+' || t[0] == '-')) {
            start = 1;
        }
        if (start >= t.size()) {
            throw std::invalid_argument("invalid literal for int(): " + value);
        }
        for (std::size_t i = start; i < t.size(); ++i) {
            if (!std::isdigit(static_cast<unsigned char>(t[i]))) {
                throw std::invalid_argument("invalid literal for int(): " + value);
            }
        }
        return std::stoi(t);
    }

    // float(s): full string must be consumed by the conversion.
    static double python_float(const std::string& value) {
        const std::string t = trim(value);
        if (t.empty()) {
            throw std::invalid_argument("invalid literal for float(): " + value);
        }
        std::size_t pos = 0;
        const double result = std::stod(t, &pos);
        if (pos != t.size()) {
            throw std::invalid_argument("invalid literal for float(): " + value);
        }
        return result;
    }
};
