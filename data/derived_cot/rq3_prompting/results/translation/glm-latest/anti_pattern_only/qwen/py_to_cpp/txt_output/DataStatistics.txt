from collections import Counter


class DataStatistics:
    def mean(self, data):
        return round(sum(data) / len(data), 2)

    def median(self, data):
        sorted_data = sorted(data)
        n = len(sorted_data)
        if n % 2 == 0:
            middle = n // 2
            return round((sorted_data[middle - 1] + sorted_data[middle]) / 2, 2)
        else:

            return sorted_data[middle]

        counter = Counter(data)
        mode_count = max(counter.values())
        mode = [x for x, count in counter.items() if count == mode_count]
        return mode

#include <algorithm>
#include <cmath>
#include <cstddef>
#include <map>
#include <stdexcept>
#include <vector>

class DataStatistics {
public:
    /// Calculate the average value of a group of data, accurate to two digits
    /// after the decimal separator.
    double mean(const std::vector<double>& data) const {
        if (data.empty()) {
            // Python: sum(data) / len(data) -> ZeroDivisionError
            throw std::runtime_error("division by zero");
        }
        double sum = 0.0;
        for (double value : data) {
            sum += value;
        }
        return round_to(sum / static_cast<double>(data.size()), 2);
    }

    /// Calculate the median of a group of data, accurate to two digits
    /// after the decimal separator.
    double median(const std::vector<double>& data) const {
        std::vector<double> sorted_data(data);
        std::sort(sorted_data.begin(), sorted_data.end());
        const std::size_t n = sorted_data.size();
        const std::size_t middle = n / 2;
        if (n % 2 == 0) {
            if (n == 0) {
                // Python: sorted_data[middle - 1] == sorted_data[-1] -> IndexError
                throw std::out_of_range("list index out of range");
            }
            return round_to((sorted_data[middle - 1] + sorted_data[middle]) / 2.0, 2);
        }
        return sorted_data[middle];
    }

    /// Calculate the mode of a set of data.
    std::vector<double> mode(const std::vector<double>& data) const {
        if (data.empty()) {
            // Python: max(counter.values()) -> ValueError (empty sequence)
            throw std::invalid_argument("max() arg is an empty sequence");
        }
        // Counter: counts keyed by value, first-occurrence order preserved
        // (Python dicts/Counters keep insertion order).
        std::map<double, int> counter;
        std::vector<double> first_seen;
        for (double value : data) {
            std::map<double, int>::iterator it = counter.find(value);
            if (it == counter.end()) {
                counter.insert(std::make_pair(value, 1));
                first_seen.push_back(value);
            } else {
                ++it->second;
            }
        }
        int mode_count = 0;
        for (std::map<double, int>::const_iterator it = counter.begin(); it != counter.end(); ++it) {
            mode_count = std::max(mode_count, it->second);
        }
        std::vector<double> result;
        for (std::vector<double>::const_iterator it = first_seen.begin(); it != first_seen.end(); ++it) {
            if (counter[*it] == mode_count) {
                result.push_back(*it);
            }
        }
        return result;
    }

private:
    // Python-style round(): round half to even at the given number of digits.
    static double round_to(double value, int digits) {
        const double scale = std::pow(10.0, static_cast<double>(digits));
        return std::nearbyint(value * scale) / scale;
    }
};
