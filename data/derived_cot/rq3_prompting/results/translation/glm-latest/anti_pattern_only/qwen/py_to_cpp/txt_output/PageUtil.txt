// PageUtil is a versatile utility for handling pagination and search
// functionalities in an efficient and convenient manner.
class PageUtil {
public:
    using Data = std::vector<int>;
    // A heterogeneous "dict" value, mirroring Python's dict values.
    using Value = std::variant<int, bool, std::string, Data>;
    using Dict = std::map<std::string, Value>;

    // Initialize the PageUtil object with the given data and page size.
    // Throws std::invalid_argument if page_size == 0 (Python raises
    // ZeroDivisionError while computing total_pages).
    PageUtil(Data data, int page_size)
        : data_(std::move(data)),
          page_size_(page_size),
          total_items_(static_cast<int>(data_.size())) {
        if (page_size_ == 0) {
            throw std::invalid_argument("integer division or modulo by zero");
        }
        total_pages_ = floor_div(total_items_ + page_size_ - 1, page_size_);
    }

    // Retrieve a specific page of data; returns an empty vector for
    // out-of-range page numbers.
    Data get_page(int page_number) const {
        if (page_number < 1 || page_number > total_pages_) {
            return {};
        }
        int start_index = (page_number - 1) * page_size_;
        int end_index = start_index + page_size_;
        return slice(start_index, end_index);
    }

    // Retrieve information about a specific page; returns an empty dict
    // for out-of-range page numbers.
    Dict get_page_info(int page_number) const {
        if (page_number < 1 || page_number > total_pages_) {
            return {};
        }
        int start_index = (page_number - 1) * page_size_;
        int end_index = std::min(start_index + page_size_, total_items_);
        Data page_data = slice(start_index, end_index);

        Dict page_info;
        page_info["current_page"] = page_number;
        page_info["per_page"] = page_size_;
        page_info["total_pages"] = total_pages_;
        page_info["total_items"] = total_items_;
        page_info["has_previous"] = page_number > 1;
        page_info["has_next"] = page_number < total_pages_;
        page_info["data"] = std::move(page_data);
        return page_info;
    }

    // Search for items whose string representation contains the keyword.
    Dict search(const std::string& keyword) const {
        Data results;
        for (int item : data_) {
            if (std::to_string(item).find(keyword) != std::string::npos) {
                results.push_back(item);
            }
        }
        int num_results = static_cast<int>(results.size());
        int num_pages = floor_div(num_results + page_size_ - 1, page_size_);

        Dict search_info;
        search_info["keyword"] = keyword;
        search_info["total_results"] = num_results;
        search_info["total_pages"] = num_pages;
        search_info["results"] = std::move(results);
        return search_info;
    }

private:
    Data data_;
    int page_size_;
    int total_items_;
    int total_pages_;

    // Python-style floor division (//) for ints.
    static int floor_div(int a, int b) {
        int q = a / b;
        if ((a % b != 0) && ((a < 0) != (b < 0))) {
            --q;
        }
        return q;
    }

    // Python-style slice data_[start:end] (step 1).
    Data slice(int start, int end) const {
        int n = total_items_;
        if (start < 0) start += n;
        if (end < 0) end += n;
        start = std::clamp(start, 0, n);
        end = std::clamp(end, 0, n);
        if (start >= end) {
            return {};
        }
        return Data(data_.begin() + start, data_.begin() + end);

};
