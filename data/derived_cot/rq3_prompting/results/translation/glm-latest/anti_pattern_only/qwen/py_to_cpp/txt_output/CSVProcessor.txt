```py
import csv

class CSVProcessor:

    def __init__(self):

    def read_csv(self, file_name):
        """
        Read the csv file by file_name, get the title and data from it

        :return title, data: (list, list), first row is title, the rest is data

        """

        """
        Write data into a csv file.
        :param file_name: str, name of the csv file
        :return:int, if success return 1, or 0 otherwise
        >>> csvProcessor = CSVProcessor()
        >>> csvProcessor.write_csv([['a', 'b', 'c', 'd'], ['1', '2', '3', '4']], 'write_test.csv')
        1
        """

    def process_csv_data(self, N, save_file_name):
        """
        Read a csv file into variable title and data.
        Only remain the N th (from 0) column data and Capitalize them, store the title and new data into a new csv file.
        Add '_process' suffix after old file name, as a new file name.
        :param N: int, the N th column(from 0)
        :param save_file_name, the name of file that needs to be processed.
        :return:int, if success return 1, or 0 otherwise
        >>> csvProcessor = CSVProcessor()
        >>> csvProcessor.read_csv('read_test.csv')
        (['a', 'b', 'c', 'd'], [['hElLo', 'YoU', 'ME', 'LoW']])
        >>> csvProcessor.process_csv_data(0, 'read_test.csv')
        1
        >>> csvProcessor.read_csv('read_test_process.csv')
        (['a', 'b', 'c', 'd'], [['HELLO']])
        """
'''

import csv


class CSVProcessor:

    def __init__(self):
        pass

    def read_csv(self, file_name):
        data = []
        with open(file_name, 'r') as file:
            reader = csv.reader(file)
            title = next(reader)
            for row in reader:
                data.append(row)
        return title, data

    def write_csv(self, data, file_name):
        try:
            with open(file_name, 'w', newline='') as file:
                writer = csv.writer(file)
                writer.writerows(data)
            return 1
        except:
            return 0

    def process_csv_data(self, N, save_file_name):
        title, data = self.read_csv(save_file_name)
        column_data = [row[N] for row in data]
        column_data = [row.upper() for row in column_data]
        new_data = [title, column_data]
        return self.write_csv(new_data, save_file_name.split('.')[0] + '_process.csv')
```

```cpp
static std::vector<std::vector<std::string>> parse_csv(const std::string& s) {
    std::vector<std::vector<std::string>> rows;
    std::vector<std::string> row;
    std::string field;
    bool in_quoted = false;      // inside a quoted field
    bool after_quote = false;    // saw a closing quote (QUOTE_IN_QUOTED_FIELD)
    bool any_field_saved = false; // whether current row has at least one field saved or started

    auto end_record = [&]() {
        rows.push_back(row);
        row.clear();
        any_field_saved = false;
    };

    size_t i = 0;
    const size_t n = s.size();
    // We'll iterate with explicit handling of '\r\n'.
    while (i <= n) {
        bool at_eof = (i == n);
        char c = at_eof ? '\0' : s[i];

        if (in_quoted) {
            if (at_eof) {
                // unterminated quoted field: _csv.c would... error "unexpected end of data"
                // Actually Python raises Error: unexpected end of data
                // For simplicity, save field and record.
                row.push_back(field);
                end_record();
                break;
            }
            if (c == '"') {
                if (i + 1 < n && s[i+1] == '"') {
                    field += '"';
                    i += 2;
                } else {
                    in_quoted = false;
                    after_quote = true;
                    ++i;
                }
            } else {
                field += c;
                ++i;
            }
            continue;
        }

        if (after_quote) {
            // QUOTE_IN_QUOTED_FIELD
            if (at_eof) {
                row.push_back(field); field.clear();
                end_record();
                break;
            }
            if (c == '"') {
                // doubled quote after close? Actually in Python: QUOTE_IN_QUOTED_FIELD + quotechar -> add quotechar, back to IN_QUOTED_FIELD
                field += '"';
                after_quote = false;
                in_quoted = true;
                ++i;
```
