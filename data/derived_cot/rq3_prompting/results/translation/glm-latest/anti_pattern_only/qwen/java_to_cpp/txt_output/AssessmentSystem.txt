    public:

            : name_(std::move(name)), grade_(grade), major_(std::move(major)) {}

        const std::string& getName() const { return name_; }

        void addCourseScore(const std::string& course, int score) {
            courses_[course] = score; // or insert_or_assign
        }

        std::optional<double> calculateGPA() const {
            if (courses_.empty()) return std::nullopt;
            int totalScore = 0;
            for (const auto& [course, score] : courses_) totalScore += score;

        }

            auto it = courses_.find(course);
            if (it != courses_.end()) return it->second;
            return std::nullopt;
        }

        auto it = students_.find(name);

            it->second.addCourseScore(course, score);
        }
    }

        auto it = students_.find(name);
        if (it != students_.end()) {
            return it->second.calculateGPA();
        }
        return std::nullopt;
    }

            }
        }
        return failingStudents;

            : name(std::move(name)), grade(grade), major(std::move(major)) {}

// Formats a double the way Java's Double.toString / System.out.println(double) does
// for the values produced by this program (decimal form with at least one fraction digit).
std::string javaToString(double value) {
    if (std::isnan(value)) return "NaN";
    if (std::isinf(value)) return value > 0 ? "Infinity" : "-Infinity";
    std::ostringstream oss;
    oss << value;
    std::string result = oss.str();
    if (result.find('.') == std::string::npos &&
        result.find('e') == std::string::npos &&
        result.find('E') == std::string::npos) {
        result += ".0";
    }
    return result;
}

std::string javaToString(const std::string& value) { return value; }

// Formats a list the way Java's AbstractCollection.toString does: "[a, b]".
std::string javaToString(const std::vector<std::string>& list) {
    std::string result = "[";
    for (std::size_t i = 0; i < list.size(); ++i) {
        if (i > 0) result += ", ";
        result += list[i];
    }
    result += "]";
    return result;
}

// Java's println prints "null" for null references (e.g. missing student / no courses).
template <typename T>
std::string javaToString(const std::optional<T>& value) {
    if (value.has_value()) {
        return javaToString(*value);
    }
    return "null";
}
