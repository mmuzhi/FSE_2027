#include <algorithm>
#include <cmath>
#include <cstddef>
#include <random>
#include <stdexcept>
#include <utility>
#include <vector>

namespace {

// Module-level random source, mirroring Python's `random` module.
std::mt19937& random_engine() {
    static std::mt19937 engine{std::random_device{}()};
    return engine;
}

// random.randint(a, b): uniform integer in [a, b] (inclusive).
int randint(int lo, int hi) {
    if (lo > hi) {
        throw std::invalid_argument("empty range for randrange()");
    }
    std::uniform_int_distribution<int> dist(lo, hi);
    return dist(random_engine());
}

// Python floor division (`//`) for integers.
int floor_div(int a, int b) {
    if (b == 0) {
        throw std::runtime_error("integer division or modulo by zero");
    }
    int q = a / b;
    if ((a % b != 0) && ((a < 0) != (b < 0))) {
        --q;
    }
    return q;
}

// Python modulo (`%`): result has the sign of the divisor.
double py_mod(double a, double m) {
    if (m == 0.0) {
        throw std::runtime_error("float modulo");
    }
    double r = std::fmod(a, m);
    if (r != 0.0 && ((r < 0.0) != (m < 0.0))) {
        r += m;
    }
    return r;
}

}  // namespace

class Snake {
public:
    // Positions use double coordinates: Python's `SCREEN_WIDTH / 2` is true
    // (float) division, so every snake position is a float in Python.
    using Position = std::pair<double, double>;

    int length;
    int SCREEN_WIDTH;
    int SCREEN_HEIGHT;
    int BLOCK_SIZE;
    std::vector<Position> positions;
    int score;
    Position food_position;

    Snake(int SCREEN_WIDTH_, int SCREEN_HEIGHT_, int BLOCK_SIZE_, const Position& food_position_)
        : length(1),
          SCREEN_WIDTH(SCREEN_WIDTH_),
          SCREEN_HEIGHT(SCREEN_HEIGHT_),
          BLOCK_SIZE(BLOCK_SIZE_),
          positions(1, Position(SCREEN_WIDTH_ / 2.0, SCREEN_HEIGHT_ / 2.0)),
          score(0),
          food_position(food_position_) {}

    void move(const Position& direction) {
        if (positions.empty()) {
            throw std::out_of_range("list index out of range");
        }
        const Position cur = positions.front();
        const double x = direction.first;
        const double y = direction.second;

        const Position new_pos(
            py_mod(cur.first + x * static_cast<double>(BLOCK_SIZE),
                   static_cast<double>(SCREEN_WIDTH)),
            py_mod(cur.second + y * static_cast<double>(BLOCK_SIZE),
                   static_cast<double>(SCREEN_HEIGHT)));

        if (new_pos == food_position) {
            eat_food();
        }

        if (positions.size() > 2
            && std::find(positions.begin() + 2, positions.end(), new_pos) != positions.end()) {
            reset();
        } else {
            positions.insert(positions.begin(), new_pos);
            if (static_cast<long long>(positions.size()) > length) {
                positions.pop_back();
            }
        }
    }

    void random_food_position() {
        while (std::find(positions.begin(), positions.end(), food_position) != positions.end()) {
            const int x = randint(0, floor_div(SCREEN_WIDTH, BLOCK_SIZE) - 1);
            const int y = randint(0, floor_div(SCREEN_HEIGHT, BLOCK_SIZE) - 1);
            food_position = Position(static_cast<double>(x) * static_cast<double>(BLOCK_SIZE),
                                     static_cast<double>(y) * static_cast<double>(BLOCK_SIZE));
        }
    }

    void reset() {
        length = 1;
        positions.assign(1, Position(SCREEN_WIDTH / 2.0, SCREEN_HEIGHT / 2.0));
        score = 0;
        random_food_position();
    }

    void eat_food() {
        length += 1;
        score += 100;
        random_food_position();
    }
};
