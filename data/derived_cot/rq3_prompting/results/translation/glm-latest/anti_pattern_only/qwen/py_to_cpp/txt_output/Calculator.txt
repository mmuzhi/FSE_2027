#include <cctype>
#include <cmath>
#include <cstdlib>
#include <functional>
#include <map>
#include <optional>
#include <stdexcept>
#include <string>
#include <utility>
#include <vector>

class Calculator {
public:
    Calculator() {
        operators_['+'] = [](double x, double y) { return x + y; };
        operators_['-'] = [](double x, double y) { return x - y; };
        operators_['*'] = [](double x, double y) { return x * y; };
        operators_['/'] = [this](double x, double y) -> double {
            if (y == 0.0) {
                throw std::runtime_error("float division by zero");
            }
            return x / y;
        };
        operators_['^'] = [](double x, double y) { return std::pow(x, y); };
    }

    std::optional<double> calculate(const std::string& expression) const {
        std::vector<double> operand_stack;
        std::vector<char> operator_stack;
        std::string num_buffer;

        for (char c : expression) {
            if (std::isdigit(static_cast<unsigned char>(c)) || c == '.') {
                num_buffer += c;
            } else {
                if (!num_buffer.empty()) {
                    operand_stack.push_back(to_float(num_buffer));
                    num_buffer.clear();
                }

                if (is_operator_char(c)) {
                    while (!operator_stack.empty() &&
                           operator_stack.back() != '(' &&
                           precedence(operator_stack.back()) >= precedence(c)) {
                        apply_operator(operand_stack, operator_stack);
                    }
                    operator_stack.push_back(c);
                } else if (c == '(') {
                    operator_stack.push_back(c);
                } else if (c == ')') {
                    while (!operator_stack.empty() && operator_stack.back() != '(') {
                        apply_operator(operand_stack, operator_stack);
                    }
                    pop_back(operator_stack);  // mirrors operator_stack.pop()
                }
            }
        }

        if (!num_buffer.empty()) {
            operand_stack.push_back(to_float(num_buffer));
        }

        while (!operator_stack.empty()) {
            apply_operator(operand_stack, operator_stack);
        }

        if (operand_stack.empty()) {
            return std::nullopt;
        }
        return operand_stack.back();
    }

    int precedence(char op) const {
        switch (op) {
            case '+':
            case '-':
                return 1;
            case '*':
            case '/':
                return 2;
            case '^':
                return 3;
            default:
                return 0;
        }
    }

    void apply_operator(std::vector<double>& operand_stack,
                        std::vector<char>& operator_stack) const {
        char op = pop_back(operator_stack);
        double operand2 = pop_back(operand_stack);
        double operand1 = pop_back(operand_stack);
        double result = operators_.at(op)(operand1, operand2);
        operand_stack.push_back(result);
    }

private:
    std::map<char, std::function<double(double, double)>> operators_;

    static bool is_operator_char(char c) {
        return c == '+' || c == '-' || c == '*' || c == '/' || c == '^';
    }

    template <typename T>
    static T pop_back(std::vector<T>& stack) {
        if (stack.empty()) {
            throw std::out_of_range("pop from empty list");
        }
        T value = stack.back();
        stack.pop_back();
        return value;
    }

    static double to_float(const std::string& s) {
        const char* begin = s.c_str();
        char* end = nullptr;
        double value = std::strtod(begin, &end);
        if (end == begin || *end != '\0') {
            throw std::invalid_argument(
                "could not convert string to float: '" + s + "'");
        }
        return value;
    }
