```
from typing import List

class Solution:
    def numMovesStonesII(self, stones: List[int]) -> List[int]:
        stones.sort()
        n = len(stones)
        max_moves = max(stones[-1] - stones[1] - n + 2, stones[-2] - stones[0] - n + 2)
        min_moves = n
        j = 0
        for i in range(n):
            while j + 1 < n and stones[j+1] - stones[i] < n:
                j += 1
            count = j - i + 1
            if count == n - 1 and stones[j] - stones[i] == n - 2:
                min_moves = min(min_moves, 2)
            else:
                min_moves = min(min_moves, n - count)
        return [min_moves, max_moves]
```

```
from typing import List

class Solution:
    def numMovesStonesII(self, stones: List[int]) -> List[int]:
        stones.sort()
        n = len(stones)

        max_moves = max(stones[-1] - stones[1] - n + 2,
                        stones[-2] - stones[0] - n + 2)

        if stones[-2] - stones[0] - n + 2 == 0 or stones[-1] - stones[1] - n + 2 == 0:
            return [min(2, max_moves), max_moves]

        best = 0
        left = 0
        for right, stone in enumerate(stones):
            while stones[left] <= stone - n:
```

```
from typing import List

class Solution:
    def numMovesStonesII(self, stones: List[int]) -> List[int]:
        stones.sort()
        n = len(stones)

        move_penultimate = stones[-2] - stones[0] - n + 2
        move_final = stones[-1] - stones[1] - n + 2
        most_moves = max(move_penultimate, move_final)

        if move_penultimate == 0 or move_final == 0:
            return [min(2, most_moves), most_moves]

        max_legal_moves = 0
        starting_index = 0

        for index, stone in enumerate(stones):
            while stones[starting_index] <= stone - n:
                starting_index += 1
            max_legal_moves = max(max_legal_moves, index - starting_index + 1)

        return [n - max_legal_moves, most_moves]
```
