class Solution:
    def getMaxRepetitions(self, s1: str, n1: int, s2: str, n2: int) -> int:
        if not set(s2).issubset(set(s1)):
            return 0

        s1 = ''.join(ch for ch in s1 if ch in set(s2))
        rec = [0]
        track = {}
        ct = 0
        pos = 0
        k = 0

        while True:
            for ch in s2:
                idx = s1.find(ch, pos)
                if idx == -1:
                    ct += 1
                    idx = s1.find(ch)
                pos = idx + 1
            k += 1
            rec.append(ct + 1)

            if rec[-1] > n1:
                return (k - 1) // n2
            if rec[-1] == n1:
                return k // n2

            if pos in track:
                k0 = track[pos]
                k1 = k
                pre_blocks = rec[k0]
                cycle_blocks = rec[k1] - pre_blocks
                cycle_copies = k1 - k0
                remaining = n1 - pre_blocks
                full_cycles = remaining // cycle_blocks
                extra_blocks = remaining % cycle_blocks

                r = 0
                while r < cycle_copies and rec[k0 + r] - pre_blocks <= extra_blocks:
                    r += 1
                r -= 1

                total_copies = k0 + full_cycles * cycle_copies + r
                return total_copies // n2
            else:
                track[pos] = k
