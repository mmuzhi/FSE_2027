class Solution:
    def numPermsDISequence(self, s: str) -> int:
        MOD = 10**9 + 7
        memo = {}

        def dfs(i: int, val: int = 0) -> int:
            if i == len(s):

            if (i, val) in memo:

            memo[(i, val)] = total
            return total

        return dfs(0)
