```
from collections import deque
from typing import List

class Solution:
    def frogPosition(self, n: int, edges: List[List[int]], t: int, target: int) -> float:
        adj = [[] for _ in range(n+1)]
        for u,v in edges:
            adj[u].append(v)
            adj[v].append(u)
        visited = [False]*(n+1)
        q = deque([(1, 1.0)])
        for _ in range(t):
            size = len(q)
            for _ in range(size):
                node, prob = q.popleft()
                visited[node] = True
                cnt = 0
                for nb in adj[node]:
                    if not visited[nb]:
                        cnt += 1
                if cnt == 0:
                    q.append((node, prob))
                else:
                    for nb in adj[node]:
                        if not visited[nb]:
                            q.append((nb, prob / cnt))
        for node, prob in q:
            if node == target:
                return prob
        return 0.0
```

```
class Solution:
    def frogPosition(self, n: int, edges: List[List[int]], t: int, target: int) -> float:

            return 0.0

        q = Queue()
```
