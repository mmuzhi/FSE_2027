class Solution:
    def numPermsDISequence(self, s: str) -> int:
        MOD = 10**9 + 7
        mem = {}

        def dfs(i: int, val: int = 0) -> int:
            if i == len(s):
                return 1
            key = (i, val)
            if key in mem:
                return mem[key]

            total = 0
            if s[i] == "D":
                for nxt in range(val + 1):
                    total += dfs(i + 1, nxt)
            else:
                for nxt in range(val + 1, i + 2):
                    total += dfs(i + 1, nxt)

            total %= MOD
            mem[key] = total
            return total

        return dfs(0)
```
