from typing import List
from collections import Counter

class Solution:
    def findSubstring(self, s: str, words: List[str]) -> List[int]:
        if not s or not words:
            return []

        word_len = len(words[0])
        total_len = word_len * len(words)

        if total_len > len(s):
            return []

        target = Counter(words)
        res = []

        for start in range(word_len):
            if start + total_len > len(s):
                continue

            current = Counter()
            for i in range(start, start + total_len, word_len):
                w = s[i:i + word_len]
                if w in target:
                    current[w] += 1

            if current == target:
                res.append(start)

            for i in range(start + word_len, len(s) - total_len + 1, word_len):
                old_word = s[i - word_len:i]
                new_word = s[i + total_len - word_len:i + total_len]

                if old_word in target:
                    current[old_word] -= 1
                    if current[old_word] == 0:
                        del current[old_word]

                if new_word in target:
                    current[new_word] += 1

                if current == target:
                    res.append(i)

        return res

from typing import List
from collections import Counter

class Solution:
    def findSubstring(self, s: str, words: List[str]) -> List[int]:
        if not s or not words:
            return []

        wlen = len(words[0])
        slen = wlen * len(words)

        if slen > len(s):
            return []

        occ = Counter(words)
        res = []

        def test(track):
            for key, val in occ.items():
                if track.get(key, 0) != val:
                    return False
            return True

        for k in range(wlen):
            if k + slen > len(s):
                continue

            track = {word: 0 for word in words}

            for i in range(k, k + slen, wlen):
                w = s[i:i + wlen]
                if w in occ:
                    track[w] += 1

            if test(track):
                res.append(k)

            for i in range(k + wlen, len(s) - slen + 1, wlen):

from typing import List
from collections import Counter

class Solution:
    def findSubstring(self, s: str, words: List[str]) -> List[int]:
        if not s or not words:
            return []

        wlen = len(words[0])
        slen = wlen * len(words)

        if slen > len(s):
            return []

        occ = Counter(words)
        res = []

        def test(track):
            for key, val in occ.items():
                if track.get(key, 0) != val:
                    return False
            return True

        for k in range(wlen):
            if k + slen > len(s):
                continue

            track = {word: 0 for word in words}

            for i in range(k, k + slen, wlen):
                w = s[i:i + wlen]
                if w in occ:
                    track[w] += 1

            if test(track):
                res.append(k)

            for i in range(k + wlen, len(s) - slen + 1, wlen):
                nw = s[i + slen - wlen:i + slen]
                pw = s[i - wlen:i]

                if nw in occ:
                    track[nw] += 1
                if pw in occ:
                    track[pw] -= 1

                if test(track):
                    res.append(i)

        return res
