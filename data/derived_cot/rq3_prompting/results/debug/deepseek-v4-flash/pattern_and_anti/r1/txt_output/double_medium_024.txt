from typing import List

class Solution:
    def isPossibleToCutPath(self, grid: List[List[int]]) -> bool:
        m, n = len(grid), len(grid[0])

        def dfs(si: int, sj: int) -> bool:
            stack = [(si, sj)]
            while stack:
                i, j = stack.pop()
                if i >= m or j >= n or grid[i][j] == 0:
                    continue
                if i == m - 1 and j == n - 1:

                grid[i][j] = 0
                stack.append((i, j + 1))

class Solution:
    def isPossibleToCutPath(self, grid: List[List[int]]) -> bool:
        m, n = len(grid), len(grid[0])
