class Solution:
    def secondGreaterElement(self, nums: List[int]) -> List[int]:
        n = len(nums)
        res = [-1] * n
        stack1 = []
        stack2 = []
        for i in range(n):
            while stack2 and nums[stack2[-1]] < nums[i]:
                res[stack2.pop()] = nums[i]
            tmp = []

                tmp.append(stack1.pop())
            while tmp:
                stack2.append(tmp.pop())
            stack1.append(i)
        return res
