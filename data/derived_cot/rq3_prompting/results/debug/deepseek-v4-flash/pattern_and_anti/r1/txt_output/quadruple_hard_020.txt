```
from collections import Counter, defaultdict, deque

class Solution:
    def isTransformable(self, s: str, t: str) -> bool:
        if Counter(s) != Counter(t):
            return False
        pos = defaultdict(deque)
        for i, ch in enumerate(t):
            pos[ch].append(i)
        for ch in s:
            idx = pos[ch].popleft()
            for d in range(ord(ch)-ord('0')+1, 10):
                if pos[str(d)] and pos[str(d)][0] < idx:
                    return False
        return True
```
