class Solution:
    def minimumTotalCost(self, nums1: List[int], nums2: List[int]) -> int:
        n = len(nums1)
        ans = 0
        cnt = Counter()
        for i, (a,b) in enumerate(zip(nums1, nums2)):
            if a == b:
                cnt[a] += 1
                ans += i
        max_cnt = max(cnt.values(), default=0)
        max_val = max(cnt, key=cnt.get) if cnt else -1
        total = sum(cnt.values())
        if max_cnt <= total - max_cnt:
            return ans
        need = max_cnt - (total - max_cnt)
        for i, (a,b) in enumerate(zip(nums1, nums2)):
            if a != b and a != max_val and b != max_val:
                ans += i
                need -= 1
                if need == 0:
                    return ans
        return -1

```
from typing import List
from collections import Counter, defaultdict

class Solution:
    def minimumTotalCost(self, nums1: List[int], nums2: List[int]) -> int:
        n = len(nums1)
        cnt1 = Counter(nums1)
        cnt2 = Counter(nums2)

        for v in set(cnt1) | set(cnt2):
            if cnt1[v] + cnt2[v] > n:
                return -1

        equal = Counter()
        pos = defaultdict(list)
        ans = 0

        for i in range(n):
            if nums1[i] == nums2[i]:
                v = nums1[i]
                equal[v] += 1
                ans += i
                pos[v].append(i)

        if not equal:
            return 0

        most = equal.most_common()
        max_count = most[0][1]
        rest_sum = sum(c for _, c in most[1:])

        if rest_sum >= max_count:
            return ans

        need = max_count - rest_sum
        majority = most[0][0]

        used = [False] * n
        for v, _ in most[1:]:
            for idx in pos[v]:
                used[idx] = True

        for i in range(n):
            if need == 0:
                break
            if not used[i] and nums1[i] != majority and nums2[i] != majority:
                ans += i
                need -= 1

        return ans if need == 0 else -1
```

        cnt1 = Counter(nums1)
        cnt2 = Counter(nums2)

        for v in set(cnt1) | set(cnt2):
            if cnt1[v] + cnt2[v] > n:
                return -1

        equal = Counter()
        pos = defaultdict(list)
        ans = 0

        for i in range(n):
            if nums1[i] == nums2[i]:
                v = nums1[i]
                equal[v] += 1
                ans += i
                pos[v].append(i)

        if not equal:
            return 0

        most = equal.most_common()
        max_count = most[0][1]
        rest_sum = sum(c for _, c in most[1:])

        if rest_sum >= max_count:
            return ans

        need = max_count - rest_sum
        majority = most[0][0]

        used = [False] * n

        for i in range(n):

            if not used[i] and nums1[i] != majority and nums2[i] != majority:
                ans += i
                need -= 1

        return ans if need == 0 else -1

    def minimumTotalCost(self, nums1: List[int], nums2: List[int]) -> int:
        n = len(nums1)
        ans = 0
        cnt = Counter()
        for i, (a,b) in enumerate(zip(nums1, nums2)):
            if a == b:
                cnt[a] += 1
                ans += i
        if not cnt:
            return 0
        max_val = max(cnt, key=cnt.get)
        max_cnt = cnt[max_val]
        rest = sum(cnt.values()) - max_cnt
        if rest >= max_cnt:
            return ans
        need = max_cnt - rest
        for i, (a,b) in enumerate(zip(nums1, nums2)):
            if a != b and a != max_val and b != max_val:
                ans += i
                need -= 1
                if need == 0:
                    return ans
        return -1

from typing import List
from collections import Counter

class Solution:
    def minimumTotalCost(self, nums1: List[int], nums2: List[int]) -> int:
        n = len(nums1)
        ans = 0
        cnt = Counter()

        for i in range(n):
            if nums1[i] == nums2[i]:
                cnt[nums1[i]] += 1
                ans += i

        if not cnt:
            return 0

        max_val = max(cnt, key=cnt.get)
        max_cnt = cnt[max_val]
        rest = sum(cnt.values()) - max_cnt

        if rest >= max_cnt:
            return ans

        need = max_cnt - rest

        for i in range(n):
            if need == 0:
                return ans
            a, b = nums1[i], nums2[i]
            if a != b and a != max_val and b != max_val:
                ans += i
                need -= 1

        return -1
