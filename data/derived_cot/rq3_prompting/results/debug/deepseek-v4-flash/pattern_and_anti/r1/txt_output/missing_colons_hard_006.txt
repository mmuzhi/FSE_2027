class Solution:

        q = deque([s1])

class Solution:

                i = 0
                while s[i] == s2[i]:
                    i += 1

                for j in range(i + 1, len(s)):
                    if s[j] == s2[i] and s[j] != s2[j]:
                        nxt = s[:i] + s[j] + s[i+1:j] + s[i] + s[j+1:]
                        if nxt not in seen:
                            seen.add(nxt)
                            q.append(nxt)

            ans += 1

        return -1
