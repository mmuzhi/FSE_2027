    def palindromePairs(self, words: List[str]) -> List[List[int]]:
        rev = {word[::-1]: i for i, word in enumerate(words)}
        res = []
        for i, word in enumerate(words):

from typing import List

class Solution:
    def palindromePairs(self, words: List[str]) -> List[List[int]]:
        backward = {word[::-1]: i for i, word in enumerate(words)}
        res = set()

        for i, word in enumerate(words):
            for j in range(len(word) + 1):
                prefix = word[:j]
                suffix = word[j:]

                if prefix in backward and suffix == suffix[::-1] and backward[prefix] != i:
                    res.add((i, backward[prefix]))

                if suffix in backward and prefix == prefix[::-1] and backward[suffix] != i:
                    res.add((backward[suffix], i))

        return [list(pair) for pair in res]

from typing import List
from collections import defaultdict

class Solution:
    def palindromePairs(self, words: List[str]) -> List[List[int]]:
        backward = defaultdict(list)
        for i, word in enumerate(words):
            backward[word[::-1]].append(i)
        res = set()
        for i, word in enumerate(words):
            for j in range(len(word) + 1):
                prefix = word[:j]
                suffix = word[j:]
                if prefix in backward and suffix == suffix[::-1]:
                    for k in backward[prefix]:
                        if k != i:
                            res.add((i, k))
                if suffix in backward and prefix == prefix[::-1]:
                    for k in backward[suffix]:
                        if k != i:
                            res.add((k, i))
        return [list(p) for p in res]

from typing import List

class Solution:
    def palindromePairs(self, words: List[str]) -> List[List[int]]:
        backward = {word[::-1]: i for i, word in enumerate(words)}
        res = set()

        for i, word in enumerate(words):
            for j in range(len(word) + 1):
                prefix = word[:j]
                suffix = word[j:]

                if prefix in backward and suffix == suffix[::-1] and backward[prefix] != i:
                    res.add((i, backward[prefix]))

                if suffix in backward and prefix == prefix[::-1] and backward[suffix] != i:
                    res.add((backward[suffix], i))

        return [list(pair) for pair in res]

from typing import List

class Solution:
    def palindromePairs(self, words: List[str]) -> List[List[int]]:
        backward = {word[::-1]: i for i, word in enumerate(words)}
        res = []

        for i, word in enumerate(words):
            if word in backward and backward[word] != i:
                res.append([i, backward[word]])

            if word != "" and "" in backward and word == word[::-1]:
                res.append([i, backward[""]])
                res.append([backward[""], i])

            for j in range(1, len(word)):
                prefix = word[:j]
                suffix = word[j:]

                if suffix in backward and prefix == prefix[::-1]:
                    res.append([backward[suffix], i])

                if prefix in backward and suffix == suffix[::-1]:
                    res.append([i, backward[prefix]])

        return res

from typing import List

class Solution:
    def palindromePairs(self, words: List[str]) -> List[List[int]]:
        backward = {word[::-1]: i for i, word in enumerate(words)}
        res = []

        for i, word in enumerate(words):
            if word in backward and backward[word] != i:
                res.append([i, backward[word]])

            if word != "" and "" in backward and word == word[::-1]:
                res.append([i, backward[""]])

                if prefix in backward and suffix == suffix[::-1] and backward[prefix] != i:
                    res.append([i, backward[prefix]])

        return res

from typing import List

class Solution:
    def palindromePairs(self, words: List[str]) -> List[List[int]]:
        backward = {}
        for i, word in enumerate(words):
            backward.setdefault(word[::-1], []).append(i)

        res = set()

        for i, word in enumerate(words):
            for j in range(len(word) + 1):
                prefix = word[:j]
                suffix = word[j:]

                if prefix in backward and suffix == suffix[::-1]:
                    for k in backward[prefix]:
                        if k != i:
                            res.add((i, k))

                if suffix in backward and prefix == prefix[::-1]:
                    for k in backward[suffix]:
