```
from typing import List

class Solution:
    def longestCycle(self, edges: List[int]) -> int:
        visited = [0] * len(edges)
        ans = -1
        for i in range(len(edges)):
            pos = i
            step = 1
            while pos != -1:
                if visited[pos]:
                    if visited[pos][0] == i:
                        ans = max(ans, step - visited[pos][1])
                    break
                visited[pos] = (i, step)
                step += 1
                pos = edges[pos]
        return ans
```
