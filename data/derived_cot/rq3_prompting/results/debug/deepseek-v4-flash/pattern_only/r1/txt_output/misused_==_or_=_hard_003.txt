from typing import List
from functools import lru_cache

class Solution:
    def ways(self, pizza: List[str], k: int) -> int:
        MOD = 10**9 + 7
        m = len(pizza)
        n = len(pizza[0])

        pref = [[0] * (n + 1) for _ in range(m + 1)]
        for i in range(m):
            for j in range(n):
                pref[i+1][j+1] = (pref[i][j+1] + pref[i+1][j] - pref[i][j] +
                                  (1 if pizza[i][j] == 'A' else 0))

        def count(r1, c1, r2, c2):
            if r1 > r2 or c1 > c2:
                return 0
            return (pref[r2+1][c2+1] - pref[r1][c2+1] -
                    pref[r2+1][c1] + pref[r1][c1])

        @lru_cache(None)
        def dp(i, j, pieces):
            if pieces == 1:
                return 1 if count(i, j, m - 1, n - 1) > 0 else 0

            ans = 0
            # horizontal cut
            for r in range(i, m - 1):
                if count(i, j, r, n - 1) > 0:
                    ans += dp(r + 1, j, pieces - 1)

            # vertical cut
            for c in range(j, n - 1):
                if count(i, j, m - 1, c) > 0:
                    ans += dp(i, c + 1, pieces - 1)

            return ans % MOD

        return dp(0, 0, k) % MOD

```
class Solution:
    def ways(self, pizza: List[str], k: int) -> int:
        m, n = len(pizza), len(pizza[0])
        pref = [[0]*(n+1) for _ in range(m+1)]
        for i in range(m):
            for j in range(n):
                pref[i+1][j+1] = pref[i+1][j] + pref[i][j+1] - pref[i][j] + (pizza[i][j]=='A')
        @lru_cache(None)
        def dp(i,j,k):
            if k == 0: return 0
            if k == 1: return 1 if pref[m][n] - pref[i][n] - pref[m][j] + pref[i][j] > 0 else 0
            ans = 0
            for x in range(i+1, m):
                if pref[x][n] - pref[i][n] - pref[x][j] + pref[i][j] > 0:
                    ans += dp(x,j,k-1)
            for y in range(j+1, n):
                if pref[m][y] - pref[i][y] - pref[m][j] + pref[i][j] > 0:
                    ans += dp(i,y,k-1)
            return ans
        return dp(0,0,k)
```

```python
    def ways(self, pizza: List[str], k: int) -> int:
```
