class Solution:
    def decodeCiphertext(self, encoded_text: str, rows: int) -> str:

        cols = n // rows
        res = []

        for start_col in range(cols):

            while r < rows and c < cols:

                r += 1
                c += 1

        return ''.join(res).rstrip()
