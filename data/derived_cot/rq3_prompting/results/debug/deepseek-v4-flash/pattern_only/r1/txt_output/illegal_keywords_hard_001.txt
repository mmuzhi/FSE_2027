```
class BIT:

    def isTransformable(self, s: str, t: str) -> bool:
        if Counter(s) != Counter(t):

        pos = [deque() for _ in range(10)]

            d = ord(ch)-48

            bit.update(p, 1)
```

```
def isTransformable(self, s: str, t: str) -> bool:
    if Counter(s) != Counter(t): return False
    pos = defaultdict(deque)
    for i, ch in enumerate(t):
        pos[ch].append(i)
    last = [-1]*10
    for ch in s:
        p = pos[ch].popleft()
        for d in range(int(ch)):
            if last[d] > p:
                return False
        last[int(ch)] = max(last[int(ch)], p)
    return True
```

```
class Solution:
    def isTransformable(self, s: str, t: str) -> bool:
```
```
        if sorted(s) != sorted(t):

        n = len(s)
```
