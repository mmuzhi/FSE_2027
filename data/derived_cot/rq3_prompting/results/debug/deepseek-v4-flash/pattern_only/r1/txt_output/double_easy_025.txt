class Solution:
    def canThreePartsEqualSum(self, arr: List[int]) -> bool:
        total = sum(arr)
        if total % 3 != 0:

        parts = 0

        for x in arr:

            current_sum += x
            if current_sum == target:
                current_sum = 0
                parts += 1

        return False
