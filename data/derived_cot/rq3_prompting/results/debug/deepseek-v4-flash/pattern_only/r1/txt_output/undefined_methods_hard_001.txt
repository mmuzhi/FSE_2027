class Solution:
    def getMaxRepetitions(self, s1: str, n1: int, s2: str, n2: int) -> int:
        if not set(s2).issubset(set(s1)):
            return 0

        idx = 0
        cnt = 0
        seen = {0: 0}
        cnt_seen = [0]

        for i in range(1, n1 + 1):
            for ch in s1:
                if ch == s2[idx]:
                    idx += 1
                    if idx == len(s2):
                        cnt += 1
                        idx = 0

            if idx in seen:
                start = seen[idx]
                cycle_len = i - start
                cycle_cnt = cnt - cnt_seen[start]

                total_cnt = cnt_seen[start] + ((n1 - start) // cycle_len) * cycle_cnt
                idx = idx  # same as seen[idx]
                # simulate remaining partial blocks
                for _ in range((n1 - start) % cycle_len):
                    for ch in s1:
                        if ch == s2[idx]:

                            if idx == len(s2):
                                total_cnt += 1
                                idx = 0
                return total_cnt // n2

            seen[idx] = i
            cnt_seen.append(cnt)

        return cnt // n2

class Solution:
    def getMaxRepetitions(self, s1: str, n1: int, s2: str, n2: int) -> int:
        if not set(s2).issubset(set(s1)):
            return 0

        idx = 0
        cnt = 0
        seen = {0: (0, 0)}
        m = len(s2)

        for i in range(1, n1 + 1):
            for ch in s1:
                if ch == s2[idx]:
                    idx += 1
                    if idx == m:
                        cnt += 1
                        idx = 0

            if idx in seen:
                start, start_cnt = seen[idx]
                cycle_len = i - start
                cycle_cnt = cnt - start_cnt

                total_cnt = start_cnt + ((n1 - start) // cycle_len) * cycle_cnt

                for _ in range((n1 - start) % cycle_len):
                    for ch in s1:
                        if ch == s2[idx]:
                            idx += 1
                            if idx == m:
                                total_cnt += 1
                                idx = 0

                return total_cnt // n2

            seen[idx] = (i, cnt)

        return cnt // n2

    def getMaxRepetitions(self, s1: str, n1: int, s2: str, n2: int) -> int:
        if not set(s2).issubset(set(s1)):
            return 0
        s1 = ''.join(c for c in s1 if c in set(s2))

                pos = s1.find(ch, start)
                if pos == -1:
                    ct += 1
                    pos = s1.find(ch)
                start = pos + 1
            rec.append(ct + 1)
            if rec[-1] > n1:
                return (len(rec) - 2) // n2
            state = start - 1  # or start? 
            if state not in track:
                track[state] = len(rec) - 1
            else:
                break
        j = track[state]
        L = len(rec) - 1
        c1 = rec[L] - rec[j]
        c2 = L - j
        # rec[j] <= n1
        full = (n1 - rec[j]) // c1
        rem_blocks = (n1 - rec[j]) % c1
        q = j + full * c2
        # partial
        t = j
        while t < L and rec[t+1] - rec[j] <= rem_blocks:
            t += 1
            q += 1
        return q // n2

class Solution:
    def getMaxRepetitions(self, s1: str, n1: int, s2: str, n2: int) -> int:
        if not set(s2).issubset(set(s1)):
            return 0

        idx = 0
        cnt = 0
        seen = {0: (0, 0)}
        m = len(s2)

        for i in range(1, n1 + 1):
            for ch in s1:
                if ch == s2[idx]:
                    idx += 1
                    if idx == m:
                        cnt += 1
                        idx = 0
