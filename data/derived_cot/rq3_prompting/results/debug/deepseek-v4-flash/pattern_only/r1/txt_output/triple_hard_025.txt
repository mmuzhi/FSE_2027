class Solution:
    def numPermsDISequence(self, s: str) -> int:
        n = len(s)
        MOD = 10**9 + 7
        dp = [1]  # length 1, only permutation [0], last rank 0
        for ch in s:
            m = len(dp)
            pref = [0] * (m + 1)
            for i, v in enumerate(dp):
                pref[i+1] = (pref[i] + v) % MOD
            ndp = [0] * (m + 1)
            if ch == 'D':
                for j in range(m + 1):
                    ndp[j] = (pref[m] - pref[j]) % MOD

                for j in range(m + 1):
                    ndp[j] = pref[j]
            dp = ndp
        return sum(dp) % MOD
