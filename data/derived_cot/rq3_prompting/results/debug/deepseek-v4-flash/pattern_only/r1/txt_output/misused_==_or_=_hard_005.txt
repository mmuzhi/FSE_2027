```python
from typing import List
from collections import Counter

class Solution:
    def minimumTotalCost(self, nums1: List[int], nums2: List[int]) -> int:
        n = len(nums1)
        cnt = Counter()
        ans = 0
        for i in range(n):
            if nums1[i] == nums2[i]:
                cnt[nums1[i]] += 1
                ans += i

        if not cnt:
            return 0

        total = sum(cnt.values())
        max_freq = max(cnt.values())
        majority = max(cnt, key=cnt.get)

        if max_freq <= total - max_freq:
            return ans

        need = max_freq - (total - max_freq)
        for i in range(n):
            if need == 0:
                break
            if nums1[i] != nums2[i] and nums1[i] != majority and nums2[i] != majority:
                ans += i
                need -= 1

        return -1 if need > 0 else ans
```

```
        ans = 0
        n = len(nums1)

            if nums1[i] == nums2[i]:
                cnt[nums1[i]] = cnt.get(nums1[i], 0) + 1
```

```
    def minimumTotalCost(self, nums1: List[int], nums2: List[int]) -> int:
```
