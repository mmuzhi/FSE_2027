    def smallestString(self, s: str) -> str:
        ct1 = 0
        for ch in s:
            if ch == 'a':
                ct1 += 1
            else:
                break

        ans = 'a' * ct1
        s = s[ct1:]

        ct2 = 0

        for ch in s:
            ct2 += 1
            if ch != 'a':

                change = True
                continue

                if change:

        if change:
