class Solution:
    def ways(self, pizza: List[str], k: int) -> int:
        m, n = len(pizza), len(pizza[0])
        dp = [[[0]*(n+1) for _ in range(m+1)] for _ in range(k)]
        pre = [[0]*(n+1) for _ in range(m+1)]
        for i in range(m-1,-1,-1):
            for j in range(n-1,-1,-1):
                pre[i][j] = pre[i+1][j] + pre[i][j+1] - pre[i+1][j+1] + (pizza[i][j]=='A')
        for i in range(m-1,-1,-1):
            for j in range(n-1,-1,-1):
                if pre[i][j] > 0:
                    dp[0][i][j] = 1
        for kk in range(1,k):
            for i in range(m-1,-1,-1):
                for j in range(n-1,-1,-1):
                    for x in range(i+1,m):
                        if pre[i][j] - pre[x][j] > 0:

                    for y in range(j+1,n):

    def ways(self, pizza: List[str], k: int) -> int:
        MOD = 1000000007
        R, C = len(pizza), len(pizza[0])
        pref = [[0] * (C + 1) for _ in range(R + 1)]
        for i in range(R):
            for j in range(C):

from functools import lru_cache

    def ways(self, pizza: List[str], k: int) -> int:
        MOD = 1000000007
        R, C = len(pizza), len(pizza[0])

        pref = [[0] * (C + 1) for _ in range(R + 1)]
        for i in range(R):
            for j in range(C):

from typing import List
from functools import lru_cache

class Solution:
    def ways(self, pizza: List[str], k: int) -> int:
        MOD = 1000000007
        R, C = len(pizza), len(pizza[0])

        pref = [[0] * (C + 1) for _ in range(R + 1)]
        for i in range(R):
            for j in range(C):
                pref[i + 1][j + 1] = (
                    pref[i][j + 1] + pref[i + 1][j] - pref[i][j]
                    + (1 if pizza[i][j] == 'A' else 0)
                )

        def has_apple(r1: int, c1: int, r2: int, c2: int) -> bool:
            return pref[r2 + 1][c2 + 1] - pref[r1][c2 + 1] - pref[r2 + 1][c1] + pref[r1][c1] > 0

        @lru_cache(None)
        def dp(r: int, c: int, pieces: int) -> int:
            if not has_apple(r, c, R - 1, C - 1):
                return 0
            if pieces == 1:
                return 1

            ans = 0

            for nr in range(r, R - 1):
                if has_apple(r, c, nr, C - 1):
                    ans += dp(nr + 1, c, pieces - 1)

            for nc in range(c, C - 1):
                if has_apple(r, c, R - 1, nc):
                    ans += dp(r, nc + 1, pieces - 1)

            return ans % MOD

        if pref[R][C] < k:
            return 0

        return dp(0, 0, k)
