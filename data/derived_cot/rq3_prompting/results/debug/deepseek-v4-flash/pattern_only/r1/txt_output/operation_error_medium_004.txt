            node = stack.pop()
            order.append(node)
            for nxt in graph[node]:
                if not visited[nxt]:
                    visited[nxt] = True
                    parent[nxt] = node
                    stack.append(nxt)

        subtree = [1] * n
        ans = 0

        for node in reversed(order):
            if node == 0:
                continue
            ans += (subtree[node] + seats - 1) // seats
            subtree[parent[node]] += subtree[node]

        return ans
