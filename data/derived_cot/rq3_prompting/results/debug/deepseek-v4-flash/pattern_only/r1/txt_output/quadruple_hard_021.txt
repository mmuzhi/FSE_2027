class Solution:
    def getMaxRepetitions(self, s1: str, n1: int, s2: str, n2: int) -> int:
        if n1 == 0: return 0
        if not set(s2).issubset(set(s1)): return 0
        repeatCount = [0] * (n1 + 1)
        nextIndex = [0] * (n1 + 1)
        for i in range(1, n1 + 1):
            idx = nextIndex[i - 1]
            count = repeatCount[i - 1]
            for ch in s2:
                pos = s1.find(ch, idx)
                if pos == -1:
                    count += 1
                    pos = s1.find(ch)
                idx = pos + 1
            repeatCount[i] = count
            nextIndex[i] = idx
            for j in range(i):
                if nextIndex[j] == idx:
                    prevCount = repeatCount[j]
                    patternCount = (repeatCount[i] - repeatCount[j]) * ((n1 - j) // (i - j))
                    remainCount = repeatCount[j + (n1 - j) % (i - j)] - repeatCount[j]
                    return (prevCount + patternCount + remainCount) // n2
        return repeatCount[n1] // n2

```python
from collections import defaultdict

class Solution:
    def getMaxRepetitions(self, s1: str, n1: int, s2: str, n2: int) -> int:
        if n1 == 0:
            return 0
        if not set(s2).issubset(set(s1)):
            return 0

        s1 = ''.join(c for c in s1 if c in set(s2))

        rec = [0]          # rec[k] = s1 copies consumed after k full s2 copies
        track = {}         # start position -> k
        start = 0
        ct = 0
        k = 0

        while True:
            for ch in s2:
                pos = s1.find(ch, start)
                if pos == -1:
                    ct += 1
                    pos = s1.find(ch)
                start = pos + 1

            k += 1
            rec.append(ct + 1)

            if rec[-1] > n1:
                return (k - 1) // n2
            if rec[-1] == n1:
                return k // n2

            if start in track:
                prev_k = track[start]
                cycle_len = k - prev_k
                cycle_copies = rec[k] - rec[prev_k]

                remaining = n1 - rec[prev_k]
                full_cycles = remaining // cycle_copies
                rem_copies = remaining % cycle_copies

                extra = 0
                for i in range(prev_k + 1, k + 1):
                    if rec[i] - rec[prev_k] <= rem_copies:
                        extra += 1
                    else:
                        break

                total_s2 = prev_k + full_cycles * cycle_len + extra
                return total_s2 // n2
            else:
                track[start] = k
```

class Solution:
    def getMaxRepetitions(self, s1: str, n1: int, s2: str, n2: int) -> int:
        if n1 == 0:
            return 0
        if not set(s2).issubset(set(s1)):
            return 0

        s1 = ''.join(c for c in s1 if c in set(s2))
        L = len(s1)
        total = L * n1

        rec = [0]
        track = {}
        pos = 0
        k = 0

        while True:
            for ch in s2:
                r = pos % L
                p = s1.find(ch, r)
                if p == -1:
                    p = s1.find(ch)
                    pos = (pos // L + 1) * L + p
                else:
                    pos = (pos // L) * L + p
                pos += 1

            k += 1
            rec.append(pos)

            if pos > total:
                return (k - 1) // n2
            if pos == total:
                return k // n2

            state = pos % L
            if state in track:
                prev_k = track[state]
                cycle_len = k - prev_k
                cycle_dist = rec[k] - rec[prev_k]
                remaining = total - rec[prev_k]
                full_cycles = remaining // cycle_dist
                rem_dist = remaining % cycle_dist

                extra = 0
                for i in range(prev_k + 1, k + 1):
                    if rec[i] - rec[prev_k] <= rem_dist:
                        extra += 1
                    else:
                        break

                return (prev_k + full_cycles * cycle_len + extra) // n2
            else:
                track[state] = k

```
class Solution:
    def getMaxRepetitions(self, s1: str, n1: int, s2: str, n2: int) -> int:
        if n1 == 0:
            return 0

        allowed = set(s2)
        if not allowed.issubset(s1):
            return 0

        s1 = ''.join(c for c in s1 if c in allowed)
        L = len(s1)
        total = L * n1

        rec = [0]
        track = {}
        pos = 0
        k = 0

        while True:
            for ch in s2:
                r = pos % L
                p = s1.find(ch, r)
                if p == -1:
                    p = s1.find(ch)
                    pos = (pos // L + 1) * L + p
                else:
                    pos = (pos // L) * L + p
                pos += 1

            k += 1
            rec.append(pos)

            if pos > total:
                return (k - 1) // n2
            if pos == total:
                return k // n2

            state = pos % L
            if state in track:
                prev_k = track[state]
                cycle_len = k - prev_k
                cycle_dist = rec[k] - rec[prev_k]
                remaining = total - rec[prev_k]

                full_cycles = remaining // cycle_dist
                rem_dist = remaining % cycle_dist

                extra = 0
                for i in range(prev_k + 1, k + 1):
                    if rec[i] - rec[prev_k] <= rem_dist:
                        extra += 1
                    else:
                        break

                return (prev_k + full_cycles * cycle_len + extra) // n2
            else:
                track[state] = k
```

class Solution:
    def getMaxRepetitions(self, s1: str, n1: int, s2: str, n2: int) -> int:
        if n1 == 0:
            return 0

        allowed = set(s2)
        if not allowed.issubset(set(s1)):
            return 0

        s1 = ''.join(c for c in s1 if c in allowed)
        L = len(s1)
        total = L * n1

        rec = [0]
        track = {}
        pos = 0
        k = 0

        while True:
            for ch in s2:
                r = pos % L
                p = s1.find(ch, r)
                if p == -1:
                    p = s1.find(ch)
                    pos = (pos // L + 1) * L + p
                else:
                    pos = (pos // L) * L + p
                pos += 1

            k += 1
            rec.append(pos)

            if pos > total:
                return (k - 1) // n2
            if pos == total:
                return k // n2

            state = pos % L
            if state in track:
                prev_k = track[state]
                cycle_len = k - prev_k
                cycle_dist = rec[k] - rec[prev_k]
                remaining = total - rec[prev_k]

                full_cycles = remaining // cycle_dist
                rem_dist = remaining % cycle_dist

                extra = 0
                for i in range(prev_k + 1, k + 1):
                    if rec[i] - rec[prev_k] <= rem_dist:
                        extra += 1
                    else:
                        break

                return (prev_k + full_cycles * cycle_len + extra) // n2
            else:
                track[state] = k

class Solution:
    def getMaxRepetitions(self, s1: str, n1: int, s2: str, n2: int) -> int:
        if n1 == 0:
            return 0

        allowed = set(s2)
        if not allowed.issubset(set(s1)):
            return 0

        s1 = ''.join(c for c in s1 if c in allowed)
        L = len(s1)
        total = L * n1

        rec = [0]
        track = {}
        pos = 0
        k = 0

        while True:
            for ch in s2:
                r = pos % L
                p = s1.find(ch, r)
                if p == -1:
                    p = s1.find(ch)
                    pos = (pos // L + 1) * L + p
                else:
                    pos = (pos // L) * L + p
                pos += 1

            k += 1
            rec.append(pos)

            if pos > total:
                return (k - 1) // n2
            if pos == total:
                return k // n2

            state = pos % L
            if state in track:
                prev_k = track[state]
                cycle_len = k - prev_k
                cycle_dist = rec[k] - rec[prev_k]
                remaining = total - rec[prev_k]

                full_cycles = remaining // cycle_dist
                rem_dist = remaining % cycle_dist

                extra = 0

                        extra += 1
                    else:
                        break

                return (prev_k + full_cycles * cycle_len + extra) // n2
            else:
                track[state] = k
