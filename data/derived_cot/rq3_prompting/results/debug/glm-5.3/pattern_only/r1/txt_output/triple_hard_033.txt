class Solution:
    def findSubstring(self, s: str, words: List[str]) -> List[int]:
        wlen = len(words[0])
        slen = wlen * len(words)
        track = dict()
        occ = Counter(words)

                if val != occ[key]:
                    return False
            return True

                    track.update({nw: track[nw] + 1})
                if pw in words:
                    track.update({pw: track[pw] - 1})
                if test():
                    res.append(i)
        return res
