class Solution:
    def dp(self, i, s, prev, k, ct, n, dct):
        if k < 0:
            return float("infinity")
        if i >= n:
            x = 0
            if ct > 1:
                x = len(str(ct)) + 1
            elif ct == 1:
                x = 1

        if s[i] == prev:
