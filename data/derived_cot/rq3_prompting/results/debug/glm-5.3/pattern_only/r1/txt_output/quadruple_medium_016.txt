from typing import List

class Solution():
    def restoreIpAddresses(self, s: str) -> List[str]:
        res = []
        cur = []

        def backtrack(i):
            if i == len(s) and len(cur) == 4:

                return

                    backtrack(i + j + 1)
                    cur.pop()
                j += 1

        backtrack(0)
        return res
