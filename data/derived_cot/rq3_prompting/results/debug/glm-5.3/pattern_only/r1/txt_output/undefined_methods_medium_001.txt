```python
class Solution:
    def getSumAbsoluteDifferences(self, nums: List[int]) -> List[int]:
        n = len(nums)
        total = sum(nums)
        prefix = 0
        result = []
        for i in range(n):
            left = nums[i] * i - prefix
            right = (total - prefix - nums[i]) - nums[i] * (n - 1 - i)
            result.append(left + right)
            prefix += nums[i]
        return result
```
