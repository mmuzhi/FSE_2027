```java
public boolean isTransformable(String s, String t) {
    // Queue of indices for each digit in s
    Deque<Integer>[] idx = new Deque[10];
    for (int i = 0; i < 10; i++) idx[i] = new ArrayDeque<>();
    for (int i = 0; i < s.length(); i++) idx[s.charAt(i) - '0'].addLast(i);
    for (char c : t.toCharArray()) {
        int d = c - '0';
        if (idx[d].isEmpty()) return false;
        int pos = idx[d].pollFirst();
        for (int i = 0; i < d; i++) {
            if (idx[i].isEmpty() || idx[i].peekFirst() > pos) continue; // ok
            return false; // a smaller digit is before pos, can't move d ahead of it
        }
    }
    return true;
}
```
