            d[i[0]] = d.get(i[0], []) + [i[1]]
            d[i[1]] = d.get(i[1], []) + [i[0]]

        visit = [0] * (n + 1)

        q = Queue()
        q.put([1, 1.0])

        for dur in range(t):

            l = q.qsize()
            for i in range(l):
                temp = q.get()

                count = 0
                for ele in d[temp[0]]:
                    if visit[ele] == 0:
                        count += 1

                if temp[0] == target and count == 0:
                    return float(temp[1])

                if visit[temp[0]] != 0:
                    continue

                visit[temp[0]] = 1

                for ele in d[temp[0]]:
                    if visit[ele] == 0:
                        q.put([ele, temp[1] * (1 / count)])

        l = q.qsize()
        for i in range(l):
            temp = q.get()
            if temp[0] == target:
                return float(temp[1])

        return 0.0
