```python
def findKthNumber(self, n, k):
    def getReqNum(a, b):
        gap = 0
        while a <= n:
            gap += min(n + 1, b) - a
            a *= 10
            b *= 10
        return gap

    cur = 1
    k -= 1
    while k > 0:
        reqNum = getReqNum(cur, cur + 1)
        if reqNum <= k:
            k -= reqNum
            cur += 1
        else:
            k -= 1
            cur *= 10
    return cur
```
