class Solution:
    def findTheString(self, lcp: List[List[int]]) -> str:
        # simple validation
        n = len(lcp)

        for i in range(n):
            for j in range(i + 1, n):
                if lcp[i][j] != lcp[j][i]:
                    return ''
                if lcp[i][j] > n - j:

            next_el_ind += 1

                if lcp[i][j] > 0:
