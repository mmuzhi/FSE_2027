import collections
from typing import List


class Solution:
    def findSubstring(self, s: str, words: List[str]) -> List[int]:
        wlen = len(words[0])
        slen = wlen * len(words)
        track = dict()

        occ = collections.Counter(words)

        def test():

                if val != occ[key]:
                    return False
            return True

        res = []

            for i in words:
