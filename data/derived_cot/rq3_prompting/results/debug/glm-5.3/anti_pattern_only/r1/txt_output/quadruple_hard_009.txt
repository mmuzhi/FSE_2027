        self.c = len(pizza[0])
        tot_apples = 0
        self.pfsum_row = []
        self.pfsum_col = []
        for i in range(self.r):

            pfs_r = [0] * self.c
            pfs_c = [0] * self.c

                    break

                return 1
            return 0
