public int getMaxRepetitions(String s1, int n1, String s2, int n2) {
    if (n1 == 0) return 0;
    int s1cnt = 0, index = 0, s2cnt = 0;
    Map<Integer, int[]> recall = new HashMap<>();
    int[] preLoop = new int[2], inLoop = new int[2];
    while (true) {
        s1cnt++;
        for (int i = 0; i < s1.length(); i++) {
            if (s1.charAt(i) == s2.charAt(index)) {
                index++;
                if (index == s2.length()) {
                    index = 0;

            preLoop = recall.get(index);
            inLoop = new int[]{s1cnt - preLoop[0], s2cnt - preLoop[1]};
            break;
        } else {
            recall.put(index, new int[]{s1cnt, s2cnt});

    }

class Solution:
    def getMaxRepetitions(self, s1: str, n1: int, s2: str, n2: int) -> int:

        s1 = ''.join(c for c in s1 if c in set(s2))

        ct = start = ptr2 = 0
        while True:
            for c in s2:

                if ptr == -1:
                    ct += 1
                    ptr = s1.find(c)
                start = ptr + 1
            rec.append(ct + 1)
            if rec[-1] > n1: return (len(rec) - 2) // n2

                track[ptr] = len(rec) - 1
            else:

        cycleStart = rec[track[ptr]]
        cycle1, cycle2 = ct + 1 - cycleStart, len(rec) - 1 - track[ptr]
        rest = n1 - cycleStart
        rem = cycleStart + rest % cycle1
        while rec[ptr2] <= rem:
            ptr2 += 1
        return (cycle2 * (rest // cycle1) + ptr2 - 1) // n2

from collections import defaultdict

class Solution:
    def getMaxRepetitions(self, s1: str, n1: int, s2: str, n2: int) -> int:
        rec, track = [0], defaultdict(int)
        ct = start = ptr1 = ptr2 = 0

        if not set(s2).issubset(set(s1)): return 0

        s1 = ''.join(char for char in s1 if char in set(s2))

        while True:
            for char in s2:
                ptr = s1.find(char, start)
                if ptr == -1:
                    ct += 1
                    ptr = s1.find(char)
                start = ptr + 1
            rec.append(ct + 1)

            if rec[-1] > n1: return (len(rec) - 2) // n2

            if ptr not in track: track[ptr] = len(rec) - 1

            else: break

        cycleStart = rec[track[ptr]]
        cycle1, cycle2 = ct + 1 - cycleStart, len(rec) - 1 - track[ptr]
        rest = n1 - cycleStart

        rem = cycleStart + rest % cycle1

        while rec[ptr2] <= rem: ptr2 += 1

        return (cycle2 * (rest // cycle1) + ptr2 - 1) // n2

                    ct += 1
                    ptr = s1.find(char)
                start = ptr + 1
            rec.append(ct + 1)

            if rec[-1] > n1: return (len(rec) - 2) // n2

            if ptr not in track: track[ptr] = len(rec) - 1

            else: break

        cycleStart = rec[track[ptr]]
        cycle1, cycle2 = ct + 1 - cycleStart, len(rec) - 1 - track[ptr]
        rest = n1 - cycleStart

        rem = cycleStart + rest % cycle1

        while rec[ptr2] <= rem: ptr2 += 1

        return (cycle2 * (rest // cycle1) + ptr2 - 1) // n2
