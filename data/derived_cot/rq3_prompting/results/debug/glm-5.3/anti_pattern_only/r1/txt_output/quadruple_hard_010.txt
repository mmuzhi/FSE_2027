from typing import List


class Node:
    __slots__ = ("val", "prio", "left", "right", "size")

    def __init__(self, val: int):
        self.val = val
        self.prio = random.random()
        self.left = None

    t.size = size(t.left) + size(t.right) + 1
    return t

        return upd(a)
    b.left = merge(a, b.left)
