class Solution:
    def isPossibleToCutPath(self, grid: List[List[int]]) -> bool:
        m, n = len(grid), len(grid[0])
        
        # DFS from bottom-right to top-left, marking reachable cells as 2
        def dfs(i, j):
            grid[i][j] = 2
            for di, dj in ((-1, 0), (0, -1)):
                ni, nj = i + di, j + dj
                if 0 <= ni < m and 0 <= nj < n and grid[ni][nj] == 1:
                    dfs(ni, nj)
        
        dfs(m - 1, n - 1)
        
        # BFS from top-left, level by level
        q = collections.deque([(0, 0)])
        grid[0][0] = 0
        while q:
            size = len(q)
            for _ in range(size):
                i, j = q.popleft()
                if i == m - 1 and j == n - 1:
                    return False
                for di, dj in ((1, 0), (0, 1)):
                    ni, nj = i + di, j + dj
                    if 0 <= ni < m and 0 <= nj < n and grid[ni][nj] == 2:
                        grid[ni][nj] = 0
                        q.append((ni, nj))
            if len(q) == 1 and q[0] != (m - 1, n - 1):
                return True
        return True
