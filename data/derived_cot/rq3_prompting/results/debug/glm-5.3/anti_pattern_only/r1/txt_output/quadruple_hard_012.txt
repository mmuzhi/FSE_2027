            if i == len(s):
                if prev_count == 0:
                    return 0
                elif prev_count == 1:
                    return 1
                else:
                    return len(str(prev_count)) + 1
            if (i, prev_char, prev_count, k) in memo:
                return memo[(i, prev_char, prev_count, k)]
            if s[i] == prev_char:
                keep = dp(i+1, prev_char, prev_count+1, k)
            else:
                cost = 0 if prev_count == 0 else (1 if prev_count == 1 else len(str(prev_count))+1)
                keep = cost + dp(i+1, s[i], 1, k)
            delete = dp(i+1, prev_char, prev_count, k-1)
            memo[(i, prev_char, prev_count, k)] = min(keep, delete)
            return memo[(i, prev_char, prev_count, k)]
