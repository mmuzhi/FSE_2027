```java
class Solution {
    public long minimumTotalCost(int[] nums1, int[] nums2) {
        long ans = 0L;
        Map<Integer, Integer> cnt = new HashMap<>();
        int sameCnt = 0, modeCnt = 0, modeVal = -1;
        int n = nums1.length;
        for (int i = 0; i < n; ++i) {
            if (nums1[i] == nums2[i]) {
                ans += i;
                sameCnt++;
                cnt.put(nums1[i], cnt.getOrDefault(nums1[i], 0) + 1);
                if (cnt.get(nums1[i]) > modeCnt) {
                    modeCnt = cnt.get(nums1[i]);
                    modeVal = nums1[i];
                }
            }
        }
        for (int i = 0; i < n && modeCnt * 2 > sameCnt; ++i) {
            if (nums1[i] != modeVal && nums2[i] != modeVal) {
                ans += i;
                sameCnt++;
            }
        }
        return modeCnt * 2 > sameCnt ? -1 : ans;
    }
}
```

class Solution:
    def minimumTotalCost(self, nums1: List[int], nums2: List[int]) -> int:
        n=len(nums1)
        z=Counter(nums1)
        z1=Counter(nums2)
        for i in z:
            if(n-z1[i]<z[i]):
                return -1
            if(z[i]>=n//2+1 and z1[i]>=n//2+1):
                return -1
        for i in z1:
            if(n-z[i]<z1[i]):
                return -1
            if(z[i]>=n//2+1 and z1[i]>=n//2+1):
                return -1
        z=Counter([])
        ans=0
        flag=0
        d=defaultdict(list)
        vis=[0 for i in range(n)]
        for i in range(n):
            if(nums1[i]==nums2[i]):
                z[nums2[i]]+=1
                ans+=i
                flag=1
                d[nums2[i]].append(i)
        t=0
        l=z.most_common(len(z))
        a=0
        for i in range(1,len(l)):

            z[l[0][0]]=x

                    if(vis[i]==0 and nums1[i]!=j and nums2[i]!=j and x):

                            ans+=i
