```python
class Solution:
    def isTransformable(self, s: str, t: str) -> bool:
        if len(s) != len(t):
            return False
        idx = defaultdict(deque)
        for i, c in enumerate(s):
            idx[int(c)].append(i)
        for i, c in enumerate(t):
            d = int(c)
            if not idx[d]:
                return False
            pos = idx[d].popleft()
            for d2 in range(d):
                if idx[d2] and idx[d2][0] < pos:
                    return False
        return True
```

```python
from collections import Counter, defaultdict, deque


class BIT:
    def __init__(self, n):
        self.n = n
        self.tree = [0] * (n + 1)

    def sum(self, i):
        ans = 0
        i += 1
        while i > 0:
            ans += self.tree[i]
            i -= (i & (-i))
        return ans

    def update(self, i, value):
        i += 1
        while i <= self.n:
            self.tree[i] += value
            i += (i & (-i))
```

```python
class BIT:
    def __init__(self, n):
        self.n = n
        self.tree = [0] * (n + 1)

    def sum(self, i):

        i += 1
        while i > 0:

            i -= (i & (-i))
        return ans

        i += 1
        while i <= self.n:
            self.tree[i] += value
            i += (i & (-i))
```
