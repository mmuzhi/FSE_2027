class Solution:
    def maximumNumber(self, num: str, change: List[int]) -> str:
        num = list(num)
        flag = False
        for i in range(len(num)):
            d = int(num[i])
            if not flag:
                if change[d] > d:
                    num[i] = str(change[d])
                    flag = True
                elif change[d] == d:
                    continue  # optional
            else:
                if change[d] >= d:
                    num[i] = str(change[d])
                else:
                    break
        return ''.join(num)
