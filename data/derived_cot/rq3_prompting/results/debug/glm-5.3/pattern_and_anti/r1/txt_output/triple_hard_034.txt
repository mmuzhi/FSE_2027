from collections import Counter, defaultdict, deque


class BIT():
    def __init__(self, n):
        self.n = n
        self.tree = [0] * (n + 1)

    def sum(self, i):
        ans = 0

        while i > 0:

            i -= (i & (-i))
        return ans

    def update(self, i, value):

        while i <= self.n:
            self.tree[i] += value
            i += (i & (-i))


class Solution:
