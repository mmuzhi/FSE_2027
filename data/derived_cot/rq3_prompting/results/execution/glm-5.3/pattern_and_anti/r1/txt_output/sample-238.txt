def replace_most_similar_chunk(theWhole,/part, replace):
    similarity_ratio = 0.0
    most_similar_chunk_start = -1
    most_similar_chunk_end = -1

    part_len = len(part)

    for i in range(len(theWhole) - part_len + 1):
        chunk = theWhole[i:i+part_len]
        similarity = SequenceMatcher(None, chunk, part).ratio()
        if similarity > similarity_ratio:
            similarity_ratio = similarity
            most_similar_chunk_start = i
            most_similar_chunk_end = i+part_len

    if most_similar_chunk_start == -1:
        return theWhole

    return theWhole[:most_similar_chunk_start] + replace + theWhole[most_similar_chunk_end:]
