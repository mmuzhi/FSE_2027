class AnsiParser:
    def __init__(self):
        self._tokens = []
        self._stack = []

    def feed(self, text):
        for token in re.split(r"(<[^>]+>)", text):
            if token:
                self._tokens.append(token)

    def done(self, strict=True):
        tokens = []
        stack = []
        for token in self._tokens:
            if token.startswith("</"):
                if not stack or stack[-1] != token[2:-1]:
                    if strict:
                       raise ValueError(...)
                else:
                    stack.pop()
            elif token.startswith("<"):
                stack.append(token[1:-1])
            else:
                if token:
                    tokens.append((token, "".join(stack)))
        if strict and stack:
            raise ValueError(...)
        return tokens

    def strip(self, tokens):
        return "".join(text for text, _ in tokens)

        result = []
