```python
def last_word(text, include='many_punctuations'):

    if not text:  # Empty string
        return ''

    if text[-1].isspace():
        return ''
    else:
        regex = {'many_punctuations': r'[\w`!@#$%^&*()_+=\[\]{},./\\:-]*"?$',
```

```python
from __future__ import print_function
import sqlparse
from sqlparse.sql import Identifier
from parseutils import extract_tables

def suggest_type(full_text, text_before_cursor):
    """Suggests the type of completion...
    ...
    """
    word_before_cursor = last_word(text_before_cursor,
            include='many_punctuations')

    identifier = None

    # If we've partially typed a word, word_before_cursor won't be an empty
    # string. In that case we want to remove the partially typed string before
    # sending it to the sqlparser. Otherwise the last token will always be the
    # partially typed string which renders the smart completion useless because
    # it will always return the list of keywords as completion.
    if word_before_cursor:
        if word_before_cursor.endswith(
                '(') or word_before_cursor.startswith('\\'):
            parsed = sqlparse.parse(text_before_cursor)
        else:
            parsed = sqlparse.parse(
                    text_before_cursor[:-len(word_before_cursor)])

            # word_before_cursor may include a schema qualification, like
            # "schema_name.table_name" or "schema_name.table_name_partial", we
            # want the second part of it
            p = sqlparse.parse(word_before_cursor)[0]

            if p.tokens and isinstance(p.tokens[0], Identifier):
                identifier = p.tokens[0]
    else:
        parsed = sqlparse.parse(text_before_cursor)

    if len(parsed) > 1:
        # Multiple statements being edited.  Isolate the current statement by
        # cumulatively summing the lengths of the statements.
        current_pos = len(text_before_cursor)
        stmt_start, stmt_end = 0, 0

        for statement in parsed:
            stmt_len = len(str(statement))
            stmt_start, stmt_end = stmt_end, stmt_end + stmt_len

            if stmt_end >= current_pos:
                text_before_cursor = full_text[stmt_start:current_pos]
                full_text = full_text[stmt_start:]
                break

    elif parsed:
        statement = parsed[0]
    else:
        statement = None

    # Check if the current statement is starting with a complete word that is
    # a known command
    if statement:
        tok1 = statement.token_first()
        if tok1 and (tok1.value == 'source' or tok1.value.startswith('\\')):
            return suggest_special(text_before_cursor)

    last_token = statement and statement.token_prev(len(statement.tokens))[1] or ''

    return suggest_based_on_last_token(last_token, text_before_cursor,
                                       full_text, identifier)
```

```python
from __future__ import print_function
import sys
import sqlparse
from sqlparse.sql import Identifier
from .parseutils import last_word, extract_tables


def suggest_type(full_text, text_before_cursor):
    """Takes the full_text that is typed so far and also the text before the
    cursor to suggest completion type and scope.

    For example if `SELECT * FROM users WHERE (user.status = 'ACTIVE' AND user.id > 1000) OR `
    is passed in as `full_text` and `user.id > 1000) OR ` is passed in as `text_before_cursor`,
    then this function should return the suggestions as:
    ...
    """
    word_before_cursor = last_word(text_before_cursor, include='many_punctuations')

    identifier = None

    try:
        # If we've partially typed a word, word_before_cursor won't be an empty
        # string. In that case we want to remove the partially typed string before
        # sending it to the sqlparser. Otherwise the last token will always be the
        # partially typed string which renders the smart completion useless because
        # it will always return the list of keywords as completion.
        if word_before_cursor:
            if word_before_cursor.endswith(
                    '(') or word_before_cursor.startswith('\\'):
                parsed = sqlparse.parse(text_before_cursor)
            else:
                parsed = sqlparse.parse(
                        text_before_cursor[:-len(word_before_cursor)])

                # word_before_cursor may include a schema qualification, like
                # "schema_name.table_name" or "schema_name.table_name_partial", we
                # want the second part of it
                p = sqlparse.parse(word_before_cursor)[0]

                if p.tokens and isinstance(p.tokens[0], Identifier):
                    identifier = p.tokens[0]
        else:
            parsed = sqlparse.parse(text_before_cursor)
    except (TypeError, AttributeError):
        return [{'type': 'keyword'}]

    if len(parsed) > 1:
        # Multiple statements being edited.  Isolate the current statement by
        # cumulatively summing the lengths of the statements.
        current_pos = len(text_before_cursor)
        stmt_start, stmt_end = 0, 0

        for statement in parsed:
            stmt_len = len(str(statement))
            stmt_start, stmt_end = stmt_end, stmt_end + stmt_len

            if stmt_end >= current_pos:
                text_before_cursor = full_text[stmt_start:current_pos]
                full_text = full_text[stmt_start:]
                break

    elif parsed:
        statement = parsed[0]
    else:
        # Empty string, no suggestions
        statement = None

    # Check if the current statement is starting with a complete word that is
    # a known command
    if statement:
        tok1 = statement.token_first()
        if tok1 and (tok1.value == 'source' or tok1.value.startswith('\\')):
            return suggest_special(text_before_cursor)

    last_token = statement and statement.token_prev(len(statement.tokens))[1] or ''

    return suggest_based_on_last_token(last_token, text_before_cursor,
                                       full_text, identifier)
```

```python
def extract_tables(sql):
    """Extract the table names from an SQL statement.

    Returns the table names in a list
    """
    parsed = sqlparse.parse(sql)
    if not parsed:
        return []

    # INSERT statements must stop looking for tables at the sign of a
    # SELECT. Otherwise the tables from the enclosed SELECT statement would
    # be reported.
    is_insert = inserted(parsed)

    tbl_prefix_seen = False
    result = set()

    for statement in parsed:
        if statement.get_type() != 'UNKNOWN':
            stream = chain(statement.tokens,
                           _find_subselect(statement))
        else:
            stream = statement.tokens

        # Iterate over tokens
        for tok in stream:
            if tok.value in INSERT:
                ...
            if tok.value in PREFIX:
                tbl_prefix_seen = True
            elif isinstance(tok, IdentifierList):
                for name in tok.get_identifiers():
                    # Sometimes Keywords (such as FROM ) are classified as
                    # identifiers in the identifier list. For example:
                    # "SELECT * FROM foo, bar ASC, baz"
                    # In this case the identifier list has 5 identifiers instead of
                    # 3. (We also have a bug where SQL builtins would be treated as
```
