    """
    Convert an Excel number (where float(xldate) is the number of days since
    either 1900-01-01 or 1904-01-01, depending on datemode) to a Gregorian
    date in (year, month, day, hour, minute, nearest_second).

    :param xldate: The Excel number
    :param datemode: 0: 1900-based, 1: 1904-based.
    ...
    """
    if datemode not in (0, 1):
        raise XLDateBadDatemode(datemode)
    if xldate == 0.00:
        return (0, 0, 0, 0, 0, 0)
    if xldate < 0.00:
        raise XLDateNegative(xldate)
    xldays = int(xldate)
    frac = xldate - xldays
    seconds = int(round(frac * 86400.0))
    assert 0 <= seconds <= 86400
    if seconds == 86400:
        hour = minute = second = 0
        xldays += 1
    else:

        hour, minute = divmod(minutes, 60)

        raise XLDateTooLarge(xldate)

    mp >>= 14
    if mp >= 10:
