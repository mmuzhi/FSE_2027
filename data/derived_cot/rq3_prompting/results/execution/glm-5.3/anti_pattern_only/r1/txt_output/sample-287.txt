```python
def xform_name(name, _xform_cache={}):
    """Convert camel case to snake case. e.g. MyName -> my_name

    Note: This function is not intended to be used for converting
    module names. For that, see the function smart_wrapper_generator.py.
    """
    if name in _xform_cache:
        return _xform_cache[name]

    new_name = name[0].lower() if name else ''
    for c in name[1:]:
        if c.isupper():
            new_name += '_' + c.lower()
        else:
            new_name += c
    _xform_cache[name] = new_name
    return new_name
```
