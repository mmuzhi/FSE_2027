    # Modify the AST to add line number prints
    tree = ast.parse(code)

class AddLinePrints(ast.NodeTransformer):
    """
    Transformer to insert print statements indicating the line number
    """

    def insert_print_statement(self, line_number: int | None) -> ast.Expr:
        return ast.Expr(
            value=ast.Call(
                func=ast.Name(id="print", ctx=ast.Load()),
                args=[
                    ast.Constant(value=f"Active line {line_number}"),
                ],
                keywords=[],
            )
        )

    def insert_multiple_print_statement(self, *line_numbers: int | None) -> list[ast.Expr]:
        return [self.insert_print_statement(ln) for ln in line_numbers]

    def visit_body(self, node: ast.List[ast.stmt]) -> ast.List[ast.stmt]:
        new_body: list[ast.stmt] = []
        for index, child in enumerate(node):
            if isinstance(child, (ast.If, ast.For, ast.While, ast.With, ast.Try)):
                new_body.extend(self.insert_multiple_print_statement(
                    child.lineno,
                    child.end_lineno
                ))
            elif isinstance(child, (ast.FunctionDef, ast.ClassDef)):

                        ln + 1

                    child.lineno,
                ))
            new_body.append(child)
        return new_body

        node.body = self.visit_body(node.body)
        ast.fix_missing_locations(node)
        return node
