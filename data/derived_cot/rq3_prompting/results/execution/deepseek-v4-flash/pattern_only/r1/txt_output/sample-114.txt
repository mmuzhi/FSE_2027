def sanitize_dataframe(df):
    """Sanitize a DataFrame to prepare it for serialization.

    * Make a copy of the DataFrame
    * Rename columns to strings
    * Convert datetime/timedelta columns to strings
    * Replace NaN/Inf with None
    * Convert categorical columns to strings
    """
    df = df.copy()

    if isinstance(df.columns, pd.RangeIndex):
        names = [str(i) for i in df.columns]
    else:
        names = [str(col) for col in df.columns]
    df.columns = names

    for col in df.columns:
        dtype = df[col].dtype
        if pd.api.types.is_integer_dtype(dtype):
            df[col] = df[col].astype(object)
        elif pd.api.types.is_float_dtype(dtype):
            df[col] = df[col].astype(object)
            df[col] = df[col].where(df[col].notnull(), None)
        elif pd.api.types.is_datetime64_any_dtype(dtype):
            df[col] = df[col].astype(str).str.replace(" ", "T", regex=False)
        elif pd.api.types.is_timedelta64_dtype(dtype):
            df[col] = df[col].astype(str)
        elif pd.api.types.is_categorical_dtype(dtype):
            df[col] = df[col].astype(str)
        elif dtype == object:
            df[col] = df[col].fillna(value=None)
    return df
```

    """Sanitize a DataFrame to prepare it for serialization.

    * Make a copy of the DataFrame
    * Convert column names to strings
    * Replace non-finite values with None
    * Convert datetime/timedelta to strings
    * Convert categorical to strings
    """
    df = df.copy()

    if isinstance(df.columns, pd.RangeIndex):
        names = [str(i) for i in df.columns]
    else:
        names = [str(col) for col in df.columns]
    df.columns = names

    for col in df.columns:
        dtype = df[col].dtype
        if pd.api.types.is_integer_dtype(dtype):
            df[col] = df[col].astype(object)
        elif pd.api.types.is_float_dtype(dtype):
            df[col] = df[col].astype(object)
            df[col] = df[col].where(df[col].notnull(), None)
        elif pd.api.types.is_datetime64_any_dtype(dtype):

        elif pd.api.types.is_timedelta64_dtype(dtype):
            df[col] = df[col].astype(str)
        elif pd.api.types.is_categorical_dtype(dtype):

def sanitize_dataframe(df):
    """Sanitize a DataFrame to prepare it for serialization.

    * Make a copy
    * Convert RangeIndex to strings
    * Convert unsupported dtypes to object
    * Replace NaN/Inf with None
    * Convert categorical to strings
    """
    df = df.copy()

    if isinstance(df.columns, pd.RangeIndex):
        names = [str(i) for i in df.columns]
    else:
        names = [str(col) for col in df.columns]
    df.columns = names

    for col in df.columns:
        dtype = df[col].dtype
        if pd.api.types.is_integer_dtype(dtype):
            df[col] = df[col].astype(object)
        elif pd.api.types.is_float_dtype(dtype):
            df[col] = df[col].astype(object)
            df[col] = df[col].where(df[col].notnull(), None)
        elif pd.api.types.is_datetime64_any_dtype(dtype):
            df[col] = df[col].astype(str).str.replace(" ", "T", regex=False)
        elif pd.api.types.is_timedelta64_dtype(dtype):
            df[col] = df[col].astype(str)
        elif pd.api.types.is_categorical_dtype(dtype):
            df[col] = df[col].astype(str)
        elif dtype == object:
            df[col] = df[col].fillna(value=None)
    return df
```
